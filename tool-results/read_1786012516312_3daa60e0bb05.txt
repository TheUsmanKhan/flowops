     1→'use client'
     2→
     3→/**
     4→ * SendExchangeShipmentModal
     5→ * -------------------------
     6→ * Reusable Dialog used by ExchangeDetailView to dispatch a new replacement
     7→ * item to the customer via a courier.
     8→ *
     9→ * Two modes (driven by exchangeMethod + isExchangeReplacement):
    10→ *   - courier_replacement  → POST /api/exchanges/{id}/dispatch-new-item
    11→ *                            (immediate dispatch — old item NOT yet received)
    12→ *   - customer_self_return → POST /api/exchanges/{id}/dispatch-replacement
    13→ *                            (post-verification dispatch)
    14→ *
    15→ * The dialog collects 6 fields in strict sequence:
    16→ *   1. Courier integration (must be selected first — drives the city provider)
    17→ *   2. Delivery city (uses CityAutocomplete with the selected courier's
    18→ *      providerKey so only operational cities are offered)
    19→ *   3. Shipping address (dropdown of customer's existing addresses + "Add New")
    20→ *   4. Shipping phone (dropdown of customer's existing phones + "Add New")
    21→ *   5. Invoice / COD amount (defaults to defaultInvoiceAmount)
    22→ *   6. Quantity (defaults to defaultQuantity)
    23→ *
    24→ * The Add New flows POST to /api/customers/{customerId}/addresses|phones,
    25→ * then refetch + auto-select the new record.
    26→ *
    27→ * On submit, the collected payload is POSTed to the dispatch endpoint. The
    28→ * backend (Prompt 5 parent integration) reads these fields to create the
    29→ * ExchangeShipment, reserve stock, and call the courier adapter.
    30→ *
    31→ * City-mismatch resolution is intentionally NOT implemented inline — the
    32→ * backend returns a clear error which we surface via toast.
    33→ */
    34→
    35→import { useEffect, useMemo, useState } from 'react'
    36→import { useMutation, useQuery, useQueryClient } from '@tanstack/react-query'
    37→import { toast } from 'sonner'
    38→import { api, FetchError } from '@/lib/api-client'
    39→import { Button } from '@/components/ui/button'
    40→import { Input } from '@/components/ui/input'
    41→import { Label } from '@/components/ui/label'
    42→import { Checkbox } from '@/components/ui/checkbox'
    43→import { Badge } from '@/components/ui/badge'
    44→import {
    45→  Dialog,
    46→  DialogContent,
    47→  DialogDescription,
    48→  DialogFooter,
    49→  DialogHeader,
    50→  DialogTitle,
    51→} from '@/components/ui/dialog'
    52→import {
    53→  Select,
    54→  SelectContent,
    55→  SelectItem,
    56→  SelectTrigger,
    57→  SelectValue,
    58→} from '@/components/ui/select'
    59→import {
    60→  Truck,
    61→  Loader2,
    62→  MapPin,
    63→  Phone,
    64→  Plus,
    65→  Banknote,
    66→  Package,
    67→  CheckCircle2,
    68→  AlertCircle,
    69→} from 'lucide-react'
    70→import { cn } from '@/lib/utils'
    71→import { CityAutocomplete } from '@/components/couriers/city-autocomplete'
    72→import { getErrorMessage } from './_shared'
    73→
    74→export interface SendExchangeShipmentModalProps {
    75→  open: boolean
    76→  onOpenChange: (v: boolean) => void
    77→  exchangeId: string
    78→  exchangeMethod: 'courier_replacement' | 'customer_self_return'
    79→  /** true = courier_replacement (immediate dispatch); false = customer_self_return (post-verification) */
    80→  isExchangeReplacement: boolean
    81→  defaultCustomerId: string
    82→  defaultVariantId: string
    83→  defaultQuantity: number
    84→  defaultInvoiceAmount: number
    85→  onSuccess?: () => void
    86→}
    87→
    88→// ─────────────────────────────────────────────────────────────────────────────
    89→// API response shapes
    90→// ─────────────────────────────────────────────────────────────────────────────
    91→
    92→interface IntegrationProvider {
    93→  id: string
    94→  providerKey: string
    95→  providerName: string
    96→  category: string
    97→  logoUrl: string | null
    98→}
    99→
   100→interface CompanyIntegration {
   101→  id: string
   102→  connectionName: string
   103→  isActive: boolean
   104→  provider: IntegrationProvider
   105→}
   106→
   107→interface IntegrationsResponse {
   108→  providers: IntegrationProvider[]
   109→  integrations: CompanyIntegration[]
   110→}
   111→
   112→interface CustomerPhoneDTO {
   113→  id: string
   114→  phoneRaw: string
   115→  label: string | null
   116→  isPrimary: boolean
   117→}
   118→
   119→interface CustomerAddressDTO {
   120→  id: string
   121→  label: string | null
   122→  address: string
   123→  city: string
   124→  isDefault: boolean
   125→}
   126→
   127→interface CustomerDetail {
   128→  id: string
   129→  name: string
   130→  phones: CustomerPhoneDTO[]
   131→  addresses: CustomerAddressDTO[]
   132→}
   133→
   134→// ─────────────────────────────────────────────────────────────────────────────
   135→// Constants
   136→// ─────────────────────────────────────────────────────────────────────────────
   137→
   138→const ADD_NEW_SENTINEL = '__add_new__'
   139→
   140→// ─────────────────────────────────────────────────────────────────────────────
   141→// Main component
   142→// ─────────────────────────────────────────────────────────────────────────────
   143→
   144→export function SendExchangeShipmentModal({
   145→  open,
   146→  onOpenChange,
   147→  exchangeId,
   148→  exchangeMethod,
   149→  isExchangeReplacement,
   150→  defaultCustomerId,
   151→  defaultVariantId,
   152→  defaultQuantity,
   153→  defaultInvoiceAmount,
   154→  onSuccess,
   155→}: SendExchangeShipmentModalProps) {
   156→  const queryClient = useQueryClient()
   157→
   158→  // ── Form state ──────────────────────────────────────────────────────────
   159→  const [companyIntegrationId, setCompanyIntegrationId] = useState<string>('')
   160→  const [deliveryCity, setDeliveryCity] = useState<string>('')
   161→  const [addressId, setAddressId] = useState<string>('')
   162→  const [phoneId, setPhoneId] = useState<string>('')
   163→  const [invoiceAmount, setInvoiceAmount] = useState<string>(
   164→    String(defaultInvoiceAmount ?? 0),
   165→  )
   166→  const [quantity, setQuantity] = useState<string>(
   167→    String(defaultQuantity ?? 1),
   168→  )
   169→
   170→  // Inline "Add New" sub-forms
   171→  const [addingAddress, setAddingAddress] = useState(false)
   172→  const [newAddressLabel, setNewAddressLabel] = useState('')
   173→  const [newAddressLine, setNewAddressLine] = useState('')
   174→  const [newAddressCity, setNewAddressCity] = useState('')
   175→  const [newAddressIsDefault, setNewAddressIsDefault] = useState(false)
   176→
   177→  const [addingPhone, setAddingPhone] = useState(false)
   178→  const [newPhoneRaw, setNewPhoneRaw] = useState('')
   179→  const [newPhoneLabel, setNewPhoneLabel] = useState('')
   180→  const [newPhoneIsPrimary, setNewPhoneIsPrimary] = useState(false)
   181→
   182→  // Reset form when modal closes
   183→  useEffect(() => {
   184→    if (!open) {
   185→      setCompanyIntegrationId('')
   186→      setDeliveryCity('')
   187→      setAddressId('')
   188→      setPhoneId('')
   189→      setInvoiceAmount(String(defaultInvoiceAmount ?? 0))
   190→      setQuantity(String(defaultQuantity ?? 1))
   191→      setAddingAddress(false)
   192→      setAddingPhone(false)
   193→      setNewAddressLabel('')
   194→      setNewAddressLine('')
   195→      setNewAddressCity('')
   196→      setNewAddressIsDefault(false)
   197→      setNewPhoneRaw('')
   198→      setNewPhoneLabel('')
   199→      setNewPhoneIsPrimary(false)
   200→    }
   201→  }, [open, defaultInvoiceAmount, defaultQuantity])
   202→
   203→  // ── Data: integrations ──────────────────────────────────────────────────
   204→  const integrationsQuery = useQuery<IntegrationsResponse>({
   205→    queryKey: ['integrations', 'courier'],
   206→    queryFn: () => api.get<IntegrationsResponse>('/api/integrations?category=courier'),
   207→    enabled: open,
   208→    staleTime: 30_000,
   209→  })
   210→
   211→  const courierIntegrations = useMemo(
   212→    () =>
   213→      (integrationsQuery.data?.integrations ?? []).filter(
   214→        (i) => i.isActive && i.provider?.category === 'courier',
   215→      ),
   216→    [integrationsQuery.data],
   217→  )
   218→
   219→  const selectedIntegration = useMemo(
   220→    () => courierIntegrations.find((i) => i.id === companyIntegrationId),
   221→    [courierIntegrations, companyIntegrationId],
   222→  )
   223→  const selectedProviderKey = selectedIntegration?.provider?.providerKey ?? ''
   224→
   225→  // ── Data: customer (addresses + phones) ─────────────────────────────────
   226→  const customerQuery = useQuery<CustomerDetail>({
   227→    queryKey: ['customer-detail', defaultCustomerId],
   228→    queryFn: () => api.get<CustomerDetail>(`/api/customers/${defaultCustomerId}`),
   229→    enabled: open && !!defaultCustomerId,
   230→    staleTime: 0,
   231→  })
   232→
   233→  const addresses = customerQuery.data?.addresses ?? []
   234→  const phones = customerQuery.data?.phones ?? []
   235→
   236→  // Default new-address city to the selected delivery city
   237→  useEffect(() => {
   238→    if (addingAddress && !newAddressCity && deliveryCity) {
   239→      setNewAddressCity(deliveryCity)
   240→    }
   241→  }, [addingAddress, newAddressCity, deliveryCity])
   242→
   243→  // ── Mutations: Add new address / phone ──────────────────────────────────
   244→  const addAddressMutation = useMutation({
   245→    mutationFn: async () => {
   246→      if (!newAddressLine.trim()) throw new Error('Address line is required')
   247→      if (!newAddressCity.trim()) throw new Error('City is required')
   248→      return api.post<{ addressId: string }>(
   249→        `/api/customers/${defaultCustomerId}/addresses`,
   250→        {
   251→          label: newAddressLabel.trim() || undefined,
   252→          address: newAddressLine.trim(),
   253→          city: newAddressCity.trim(),
   254→          is_default: newAddressIsDefault,
   255→        },
   256→      )
   257→    },
   258→    onSuccess: (data) => {
   259→      toast.success('Address added to customer record.')
   260→      // Reset inline form
   261→      setNewAddressLabel('')
   262→      setNewAddressLine('')
   263→      setNewAddressIsDefault(false)
   264→      setAddingAddress(false)
   265→      // Refetch customer, then auto-select the new address
   266→      queryClient.invalidateQueries({
   267→        queryKey: ['customer-detail', defaultCustomerId],
   268→      })
   269→      setAddressId(data.addressId)
   270→    },
   271→    onError: (err) => toast.error(getErrorMessage(err)),
   272→  })
   273→
   274→  const addPhoneMutation = useMutation({
   275→    mutationFn: async () => {
   276→      if (!newPhoneRaw.trim()) throw new Error('Phone number is required')
   277→      return api.post<{ phoneId: string }>(
   278→        `/api/customers/${defaultCustomerId}/phones`,
   279→        {
   280→          phone: newPhoneRaw.trim(),
   281→          label: newPhoneLabel.trim() || undefined,
   282→          is_primary: newPhoneIsPrimary,
   283→        },
   284→      )
   285→    },
   286→    onSuccess: (data) => {
   287→      toast.success('Phone added to customer record.')
   288→      setNewPhoneRaw('')
   289→      setNewPhoneLabel('')
   290→      setNewPhoneIsPrimary(false)
   291→      setAddingPhone(false)
   292→      queryClient.invalidateQueries({
   293→        queryKey: ['customer-detail', defaultCustomerId],
   294→      })
   295→      setPhoneId(data.phoneId)
   296→    },
   297→    onError: (err) => toast.error(getErrorMessage(err)),
   298→  })
   299→
   300→  // ── Mutation: dispatch ──────────────────────────────────────────────────
   301→  const dispatchMutation = useMutation({
   302→    mutationFn: async () => {
   303→      const endpoint = isExchangeReplacement
   304→        ? `/api/exchanges/${exchangeId}/dispatch-new-item`
   305→        : `/api/exchanges/${exchangeId}/dispatch-replacement`
   306→      return api.post(endpoint, {
   307→        companyIntegrationId,
   308→        deliveryCity: deliveryCity.trim(),
   309→        shippingAddressId: addressId || undefined,
   310→        shippingPhoneId: phoneId || undefined,
   311→        invoiceAmount: Number(invoiceAmount) || 0,
   312→        quantity: Number(quantity) || 1,
   313→        variantId: defaultVariantId,
   314→      })
   315→    },
   316→    onSuccess: () => {
   317→      const msg = isExchangeReplacement
   318→        ? 'Replacement item dispatched — ExchangeShipment created.'
   319→        : 'Replacement shipment dispatched.'
   320→      toast.success(msg)
   321→      // Invalidate exchange + shipments so the parent refetches
   322→      queryClient.invalidateQueries({ queryKey: ['exchanges'] })
   323→      queryClient.invalidateQueries({ queryKey: ['exchange', exchangeId] })
   324→      queryClient.invalidateQueries({ queryKey: ['inventory-pools'] })
   325→      onSuccess?.()
   326→      onOpenChange(false)
   327→    },
   328→    onError: (err) => {
   329→      // Keep the dialog open on failure so the user can fix inputs.
   330→      toast.error(
   331→        err instanceof FetchError
   332→          ? err.message
   333→          : getErrorMessage(err),
   334→      )
   335→    },
   336→  })
   337→
   338→  // ── Validation ──────────────────────────────────────────────────────────
   339→  const canSubmit =
   340→    !!companyIntegrationId &&
   341→    !!deliveryCity.trim() &&
   342→    !!addressId &&
   343→    !!phoneId &&
   344→    !addingAddress &&
   345→    !addingPhone &&
   346→    !dispatchMutation.isPending
   347→
   348→  const methodLabel = isExchangeReplacement
   349→    ? 'Courier Replacement — Immediate Dispatch'
   350→    : 'Customer Self-Return — Post-Verification Dispatch'
   351→
   352→  return (
   353→    <Dialog open={open} onOpenChange={(v) => !v && onOpenChange(false)}>
   354→      <DialogContent className="sm:max-w-lg max-h-[90vh] overflow-y-auto scrollbar-thin">
   355→        <DialogHeader>
   356→          <DialogTitle className="flex items-center gap-2">
   357→            <Truck className="h-4 w-4" /> Dispatch Replacement Shipment
   358→          </DialogTitle>
   359→          <DialogDescription>
   360→            {methodLabel}. Fill in courier + delivery details to create an
   361→            ExchangeShipment (EXCH-#####) and book with the courier.
   362→          </DialogDescription>
   363→        </DialogHeader>
   364→
   365→        <div className="space-y-4 py-2">
   366→          {/* Method badge */}
   367→          <div className="flex items-center gap-2">
   368→            <Badge
   369→              variant="outline"
   370→              className={cn(
   371→                'text-[10px]',
   372→                isExchangeReplacement
   373→                  ? 'bg-violet-50 text-violet-700 border-violet-200'
   374→                  : 'bg-sky-50 text-sky-700 border-sky-200',
   375→              )}
   376→            >
   377→              {exchangeMethod.replace(/_/g, ' ')}
   378→            </Badge>
   379→            <span className="text-xs text-muted-foreground">
   380→              Exchange {exchangeId.slice(0, 8)}…
   381→            </span>
   382→          </div>
   383→
   384→          {/* 1. Courier integration */}
   385→          <div className="space-y-1.5">
   386→            <Label className="text-xs flex items-center gap-1">
   387→              <Truck className="h-3 w-3" /> 1. Courier *
   388→            </Label>
   389→            <Select
   390→              value={companyIntegrationId}
   391→              onValueChange={(v) => {
   392→                setCompanyIntegrationId(v)
   393→                // Reset city when courier changes (different provider → different cities)
   394→                setDeliveryCity('')
   395→              }}
   396→              disabled={integrationsQuery.isLoading}
   397→            >
   398→              <SelectTrigger>
   399→                <SelectValue
   400→                  placeholder={
   401→                    integrationsQuery.isLoading
   402→                      ? 'Loading couriers…'
   403→                      : courierIntegrations.length === 0
   404→                        ? 'No courier integrations connected'
   405→                        : 'Select courier integration'
   406→                  }
   407→                />
   408→              </SelectTrigger>
   409→              <SelectContent>
   410→                {courierIntegrations.map((i) => (
   411→                  <SelectItem key={i.id} value={i.id}>
   412→                    <span className="flex items-center gap-2">
   413→                      <span className="font-medium">
   414→                        {i.provider.providerName}
   415→                      </span>
   416→                      <span className="text-xs text-muted-foreground">
   417→                        ({i.connectionName})
   418→                      </span>
   419→                    </span>
   420→                  </SelectItem>
   421→                ))}
   422→              </SelectContent>
   423→            </Select>
   424→            {integrationsQuery.isError && (
   425→              <p className="text-xs text-rose-600 flex items-center gap-1">
   426→                <AlertCircle className="h-3 w-3" /> Failed to load courier
   427→                integrations.
   428→              </p>
   429→            )}
   430→          </div>
   431→
   432→          {/* 2. City */}
   433→          <div className="space-y-1.5">
   434→            <Label className="text-xs flex items-center gap-1">
   435→              <MapPin className="h-3 w-3" /> 2. Delivery City *
   436→            </Label>
   437→            <CityAutocomplete
   438→              providerKey={selectedProviderKey}
   439→              value={deliveryCity}
   440→              onChange={setDeliveryCity}
   441→              placeholder={
   442→                selectedProviderKey
   443→                  ? 'Search city…'
   444→                  : 'Select a courier first'
   445→              }
   446→              disabled={!selectedProviderKey}
   447→            />
   448→            {!selectedProviderKey && (
   449→              <p className="text-xs text-muted-foreground">
   450→                City options are filtered to the courier's operational cities.
   451→              </p>
   452→            )}
   453→          </div>
   454→
   455→          {/* 3. Customer address */}
   456→          <div className="space-y-1.5">
   457→            <Label className="text-xs flex items-center gap-1">
   458→              <MapPin className="h-3 w-3" /> 3. Shipping Address *
   459→            </Label>
   460→            <Select
   461→              value={addressId}
   462→              onValueChange={(v) => {
   463→                if (v === ADD_NEW_SENTINEL) {
   464→                  setAddingAddress(true)
   465→                  setAddressId('')
   466→                } else {
   467→                  setAddingAddress(false)
   468→                  setAddressId(v)
   469→                }
   470→              }}
   471→              disabled={customerQuery.isLoading}
   472→            >
   473→              <SelectTrigger>
   474→                <SelectValue
   475→                  placeholder={
   476→                    customerQuery.isLoading
   477→                      ? 'Loading addresses…'
   478→                      : addresses.length === 0
   479→                        ? 'No addresses — add new'
   480→                        : 'Select address'
   481→                  }
   482→                />
   483→              </SelectTrigger>
   484→              <SelectContent>
   485→                {addresses.map((a) => (
   486→                  <SelectItem key={a.id} value={a.id}>
   487→                    <span className="flex flex-col">
   488→                      <span>
   489→                        {a.address}
   490→                        {a.isDefault && (
   491→                          <span className="ml-1 text-[10px] text-emerald-600">
   492→                            (default)
   493→                          </span>
   494→                        )}
   495→                      </span>
   496→                      <span className="text-xs text-muted-foreground">
   497→                        {a.city}
   498→                        {a.label ? ` · ${a.label}` : ''}
   499→                      </span>
   500→                    </span>
   501→                  </SelectItem>
   502→                ))}
   503→                <SelectItem value={ADD_NEW_SENTINEL}>
   504→                  <span className="flex items-center gap-1 text-primary">
   505→                    <Plus className="h-3 w-3" /> Add new address
   506→                  </span>
   507→                </SelectItem>
   508→              </SelectContent>
   509→            </Select>
   510→
   511→            {/* Inline Add New Address form */}
   512→            {addingAddress && (
   513→              <div className="rounded-md border border-primary/30 bg-primary/5 p-3 space-y-2">
   514→                <p className="text-xs font-medium flex items-center gap-1">
   515→                  <Plus className="h-3 w-3" /> New Address
   516→                </p>
   517→                <Input
   518→                  placeholder="Address line *"
   519→                  value={newAddressLine}
   520→                  onChange={(e) => setNewAddressLine(e.target.value)}
   521→                  className="text-xs h-8"
   522→                />
   523→                <Input
   524→                  placeholder="City *"
   525→                  value={newAddressCity}
   526→                  onChange={(e) => setNewAddressCity(e.target.value)}
   527→                  className="text-xs h-8"
   528→                />
   529→                <Input
   530→                  placeholder="Label (optional, e.g. Home, Office)"
   531→                  value={newAddressLabel}
   532→                  onChange={(e) => setNewAddressLabel(e.target.value)}
   533→                  className="text-xs h-8"
   534→                />
   535→                <div className="flex items-center gap-2">
   536→                  <Checkbox
   537→                    id="new-addr-default"
   538→                    checked={newAddressIsDefault}
   539→                    onCheckedChange={(v) => setNewAddressIsDefault(v === true)}
   540→                  />
   541→                  <Label
   542→                    htmlFor="new-addr-default"
   543→                    className="text-xs text-muted-foreground"
   544→                  >
   545→                    Set as default address
   546→                  </Label>
   547→                </div>
   548→                <div className="flex gap-2 pt-1">
   549→                  <Button
   550→                    size="sm"
   551→                    variant="ghost"
   552→                    className="h-7 text-xs"
   553→                    onClick={() => setAddingAddress(false)}
   554→                    disabled={addAddressMutation.isPending}
   555→                  >
   556→                    Cancel
   557→                  </Button>
   558→                  <Button
   559→                    size="sm"
   560→                    className="h-7 text-xs"
   561→                    onClick={() => addAddressMutation.mutate()}
   562→                    disabled={
   563→                      addAddressMutation.isPending ||
   564→                      !newAddressLine.trim() ||
   565→                      !newAddressCity.trim()
   566→                    }
   567→                  >
   568→                    {addAddressMutation.isPending ? (
   569→                      <>
   570→                        <Loader2 className="h-3 w-3 animate-spin" /> Saving…
   571→                      </>
   572→                    ) : (
   573→                      'Save Address'
   574→                    )}
   575→                  </Button>
   576→                </div>
   577→              </div>
   578→            )}
   579→          </div>
   580→
   581→          {/* 4. Customer phone */}
   582→          <div className="space-y-1.5">
   583→            <Label className="text-xs flex items-center gap-1">
   584→              <Phone className="h-3 w-3" /> 4. Customer Phone *
   585→            </Label>
   586→            <Select
   587→              value={phoneId}
   588→              onValueChange={(v) => {
   589→                if (v === ADD_NEW_SENTINEL) {
   590→                  setAddingPhone(true)
   591→                  setPhoneId('')
   592→                } else {
   593→                  setAddingPhone(false)
   594→                  setPhoneId(v)
   595→                }
   596→              }}
   597→              disabled={customerQuery.isLoading}
   598→            >
   599→              <SelectTrigger>
   600→                <SelectValue
   601→                  placeholder={
   602→                    customerQuery.isLoading
   603→                      ? 'Loading phones…'
   604→                      : phones.length === 0
   605→                        ? 'No phones — add new'
   606→                        : 'Select phone'
   607→                  }
   608→                />
   609→              </SelectTrigger>
   610→              <SelectContent>
   611→                {phones.map((p) => (
   612→                  <SelectItem key={p.id} value={p.id}>
   613→                    <span className="flex items-center gap-2">
   614→                      <span className="font-mono">{p.phoneRaw}</span>
   615→                      {p.isPrimary && (
   616→                        <span className="text-[10px] text-emerald-600">
   617→                          (primary)
   618→                        </span>
   619→                      )}
   620→                      {p.label && (
   621→                        <span className="text-xs text-muted-foreground">
   622→                          · {p.label}
   623→                        </span>
   624→                      )}
   625→                    </span>
   626→                  </SelectItem>
   627→                ))}
