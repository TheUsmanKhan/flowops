    41→    41→// ──────────────────────────────────────────────────────────────
    42→    42→// 1. generatePostExLoadSheet — standalone batch action
    43→    43→// ──────────────────────────────────────────────────────────────
    44→    44→
    45→    45→/**
    46→    46→ * Generate a PostEx load sheet (pickup manifest) for a batch of tracking numbers.
    47→    47→ *
    48→    48→ * Can be called:
    49→    49→ *   (a) Manually from a "Generate Load Sheet" button for multiple Unbooked orders.
    50→    50→ *   (b) Automatically right after a successful bookShipment() call (chained).
    51→    51→ *
    52→    52→ * @param companyIntegrationId - The PostEx company integration to use
    53→    53→ * @param trackingNumbers - List of PostEx tracking numbers to include
    54→    54→ * @param pickupAddress - Optional pickup address text (from the address book)
    55→    55→ */
    56→    56→export async function generatePostExLoadSheet(
    57→    57→  companyIntegrationId: string,
    58→    58→  trackingNumbers: string[],
    59→    59→  pickupAddress?: string,
    60→    60→): Promise<ActionResult> {
    61→    61→  try {
    62→    62→    const ctx = await getWorkspace()
    63→    63→    if (!isElevated(ctx)) {
    64→    64→      return { success: false, error: 'Only elevated roles can generate load sheets.' }
    65→    65→    }
    66→    66→
    67→    67→    const integration = await db.companyIntegration.findFirst({
    68→    68→      where: {
    69→    69→        id: companyIntegrationId,
    70→    70→        companyId: ctx.company.id,
    71→    71→        provider: { providerKey: 'postex' },
    72→    72→      },
    73→    73→      include: { provider: true },
    74→    74→    })
    75→    75→    if (!integration) {
    76→    76→      return { success: false, error: 'PostEx integration not found.' }
    77→    77→    }
    78→    78→
    79→    79→    const credentials = decryptCredentials(integration.credentialsEncrypted!)
    80→    80→    const adapter = getCourierAdapter('postex', credentials)
    81→    81→
    82→    82→    if (!adapter.generateLoadSheet) {
    83→    83→      return { success: false, error: 'PostEx adapter does not support load sheet generation.' }
    84→    84→    }
    85→    85→
    86→    86→    const result = await executeLoggedIntegrationAction<{ success: boolean; rawResponse?: unknown; error?: string }>({
    87→    87→      companyIntegrationId,
    88→    88→      organizationId: integration.organizationId,
    89→    89→      actionType: 'generate_load_sheet',
    90→    90→      direction: 'outbound',
    91→    91→      fn: async () => adapter.generateLoadSheet!(trackingNumbers, pickupAddress),
    92→    92→    })
    93→    93→
    94→    94→    insertAuditLog({
    95→    95→      action: 'postex.load_sheet_generated',
    96→    96→      entityType: 'company_integration',
    97→    97→      entityId: companyIntegrationId,
    98→    98→      companyId: ctx.company.id,
    99→    99→      organizationId: ctx.company.organizationId,
   100→   100→      userId: ctx.user.id,
   101→   101→      employeeId: ctx.employee.id,
   102→   102→      newValues: {
   103→   103→        trackingNumbersCount: trackingNumbers.length,
   104→   104→        pickupAddress: pickupAddress ?? null,
   105→   105→      },
   106→   106→    })
   107→   107→
   108→   108→    return { success: result.success, data: result }
   109→   109→  } catch (err) {
   110→   110→    return {
   111→   111→      success: false,
   112→   112→      error: err instanceof Error ? err.message : 'Failed to generate load sheet',
   113→   113→    }
   114→   114→  }
   115→   115→}

   119→   119→// ──────────────────────────────────────────────────────────────
   120→   120→
   121→   121→/**
   122→   122→ * Poll PostEx for status updates on all active orders AND exchange_shipments.
   123→   123→ *
   124→   124→ * Queries all records with a PostEx courier integration where status is NOT
   125→   125→ * 'delivered' or 'rto' (permanent exclusion — these are never re-polled).
   126→   126→ * Batches tracking numbers, calls trackBulkShipments(), and for each result:
   127→   127→ *   - If status changed → triggers the appropriate OMS function
   128→   128→ *   - If status unchanged → only updates lastPolledAt
   129→   129→ *
   130→   130→ * IDEMPOTENT: calling twice with no underlying change produces no duplicate
   131→   131→ * audit/metric entries beyond the tracking API call log itself.
   132→   132→ *
   133→   133→ * SCHEDULING: needs to run every 30 minutes. Infrastructure-level scheduling
   134→   134→ * (cron, Vercel Cron, etc.) still needs to be connected — same pattern as
   135→   135→ * the city-sync job.
   136→   136→ */

   138→   138→// ──────────────────────────────────────────────────────────────
   139→   139→// 2b. trackSingleOrderStatus — single-order immediate tracking (uses trackShipment)
   140→   140→// ──────────────────────────────────────────────────────────────
   141→   141→
   142→   142→/**
   143→   143→ * Track a SINGLE order's courier status immediately via the adapter's
   144→   144→ * trackShipment() method (NOT the bulk API). This is the function wired to the
   145→   145→ * manual "Refresh Courier Status" button on the Order Detail page.
   146→   146→ *
   147→   147→ * Why a separate function (not just calling pollPostExOrderStatuses)?
   148→   148→ *   - The bulk poll iterates ALL active orders across ALL companies — wasteful
   149→   149→ *     for a single user-initiated check.
   150→   150→ *   - trackShipment() is the adapter's single-tracking method (GET /v1/track-order/{tn}),
   151→   151→ *     which is faster and cheaper than the bulk API for one order.
   152→   152→ *   - This gives the user an immediate, per-order refresh without waiting for
   153→   153→ *     the next cron cycle.
   154→   154→ *
   155→   155→ * Reuses the SAME status-transition logic as the bulk poll (performOrderDispatch
   156→   156→ * for in_transit, mark delivered for delivered, RTO handling, etc.) by calling
   157→   157→ * the shared functions directly.
   158→   158→ *
   159→   159→ * @param orderId - The order to track
   160→   160→ * @returns { success, data: { status, subStatus, updated } }
   161→   161→ */
   162→   162→export async function trackSingleOrderStatus(orderId: string): Promise<ActionResult<{
   163→   163→  status: string
   164→   164→  subStatus: string | null
   165→   165→  updated: boolean
   166→   166→}>> {
   167→   167→  try {
   168→   168→    const ctx = await getWorkspace()
   169→   169→    if (!isElevated(ctx)) {
   170→   170→      return { success: false, error: 'Only elevated roles can refresh courier status.' }
   171→   171→    }
   172→   172→
   173→   173→    // Fetch the order (workspace-scoped)
   174→   174→    const order = await db.order.findFirst({
   175→   175→      where: { id: orderId, companyId: ctx.company.id },
   176→   176→      select: {
   177→   177→        id: true,
   178→   178→        trackingNumber: true,
   179→   179→        courierCompanyIntegrationId: true,
   180→   180→        status: true,
   181→   181→        courierSubStatus: true,
   182→   182→        flowopsOrderNumber: true,
   183→   183→      },
   184→   184→    })
   185→   185→    if (!order) return { success: false, error: 'Order not found' }
   186→   186→    if (!order.trackingNumber) return { success: false, error: 'Order has no tracking number' }
   187→   187→    if (!order.courierCompanyIntegrationId) {
   188→   188→      return { success: false, error: 'Order has no courier integration' }
   189→   189→    }
   190→   190→
   191→   191→    // Fetch the integration + decrypt credentials
   192→   192→    const integration = await db.companyIntegration.findUnique({
   193→   193→      where: { id: order.courierCompanyIntegrationId },
   194→   194→      include: { provider: true },
   195→   195→    })
   196→   196→    if (!integration || !integration.credentialsEncrypted) {
   197→   197→      return { success: false, error: 'Courier integration not found or no credentials' }
   198→   198→    }
   199→   199→
   200→   200→    const credentials = decryptCredentials(integration.credentialsEncrypted)
   201→   201→    const providerKey = integration.provider.providerKey
   202→   202→    const adapter = getCourierAdapter(providerKey, credentials)
   203→   203→
   204→   204→    // Call trackShipment() via the logged wrapper (single-order, NOT bulk)
   205→   205→    const trackResult = await executeLoggedIntegrationAction<TrackShipmentResult>({
   206→   206→      companyIntegrationId: integration.id,
   207→   207→      organizationId: integration.organizationId,
   208→   208→      actionType: 'track_shipment',
   209→   209→      direction: 'outbound',
   210→   210→      relatedEntityType: 'order',
   211→   211→      relatedEntityId: order.id,
   212→   212→      fn: async () => adapter.trackShipment(order.trackingNumber!),
   213→   213→    })
   214→   214→
   215→   215→    // Extract mapped subStatus from raw response
   216→   216→    const raw = trackResult.rawResponse as Record<string, unknown> | undefined
   217→   217→    const mappedSubStatus = (raw?.mappedSubStatus as string) ?? null
   218→   218→    const needsShipperAdvice = (raw?.needsShipperAdvice as boolean) ?? false
   219→   219→    const unrecognized = (raw?.unrecognized as boolean) ?? false
   220→   220→
   221→   221→    const subStatusChanged = mappedSubStatus !== order.courierSubStatus
   222→   222→    const now = new Date()
   223→   223→
   224→   224→    // Update lastPolledAt + courierSubStatus + flags (same as bulk poll)
   225→   225→    await db.order.update({
   226→   226→      where: { id: order.id },
   227→   227→      data: {
   228→   228→        lastPolledAt: now,
   229→   229→        courierSubStatus: mappedSubStatus,
   230→   230→        needsShipperAdvice,
   231→   231→        unrecognizedCourierStatus: unrecognized,
   232→   232→      },
   233→   233→    })
   234→   234→
   235→   235→    // Apply the SAME status transitions as the bulk poll
   236→   236→    if (subStatusChanged) {
   237→   237→      // in_transit → auto-dispatch (with inventory deduction)
   238→   238→      if (trackResult.status === 'in_transit') {
   239→   239→        if (order.status === 'confirmed' || order.status === 'processing') {
   240→   240→          const dispatchResult = await performOrderDispatch(order.id, { source: 'auto_poll' })
   241→   241→          if (!dispatchResult.success && !dispatchResult.skipped) {
   242→   242→            console.error(`[trackSingle] Failed to auto-dispatch order ${order.id}: ${dispatchResult.error}`)
   243→   243→          }
   244→   244→        }
   245→   245→      }
   246→   246→
   247→   247→      // delivered → mark as delivered
   248→   248→      if (trackResult.status === 'delivered') {
   249→   249→        const freshOrder = await db.order.findUnique({ where: { id: order.id }, select: { status: true } })
   250→   250→        if (freshOrder && (freshOrder.status === 'confirmed' || freshOrder.status === 'processing')) {
   251→   251→          await performOrderDispatch(order.id, { source: 'auto_poll' }).catch(() => {})
   252→   252→        }
   253→   253→        if (freshOrder && freshOrder.status !== 'delivered' && freshOrder.status !== 'cancelled' && freshOrder.status !== 'refunded') {
   254→   254→          await db.order.update({
   255→   255→            where: { id: order.id },
   256→   256→            data: { status: 'delivered', deliveredAt: new Date() },
   257→   257→          })
   258→   258→        }
   259→   259→      }
   260→   260→
   261→   261→      // returned → RTO (restock inventory for both reserved AND dispatched items)
   262→   262→      if (trackResult.status === 'returned') {
   263→   263→        const freshOrder = await db.order.findUnique({
   264→   264→          where: { id: order.id },
   265→   265→          select: { status: true, dispatchLocationId: true, organizationId: true, companyId: true },
   266→   266→        })
   267→   267→        if (freshOrder && freshOrder.status !== 'rto' && freshOrder.status !== 'cancelled' && freshOrder.status !== 'refunded') {
   268→   268→          // Restock inventory — handles BOTH confirmed/processing (unreserve)
   269→   269→          // AND dispatched (return_resellable/return_stitched_received) cases.
   270→   270→          // Previously the dispatched case was a GAP (onHand never restored).
   271→   271→          const restockResult = await restockOrderForRto(order.id, {
   272→   272→            organizationId: freshOrder.organizationId,
   273→   273→            companyId: freshOrder.companyId ?? integration.companyId,
   274→   274→            employeeId: integration.createdBy ?? null,
   275→   275→            returnReason: `PostEx courier returned (status: returned, detected via trackSingleOrderStatus)`,
   276→   276→          })
   277→   277→          if (restockResult.itemsRestocked > 0) {
   278→   278→            console.log(`[trackSingle] Restocked ${restockResult.itemsRestocked} item(s) for RTO order ${order.flowopsOrderNumber}`)
   279→   279→          }
   280→   280→          await db.order.update({
   281→   281→            where: { id: order.id },
   282→   282→            data: { status: 'rto', returnedAt: new Date() },
   283→   283→          })
   284→   284→        }
   285→   285→      }
   286→   286→    }
   287→   287→
   288→   288→    return {
   289→   289→      success: true,
   290→   290→      data: {
   291→   291→        status: trackResult.status ?? 'unknown',
   292→   292→        subStatus: mappedSubStatus,
   293→   293→        updated: subStatusChanged,
   294→   294→      },
   295→   295→    }
   296→   296→  } catch (err) {
   297→   297→    return {
   298→   298→      success: false,
   299→   299→      error: err instanceof Error ? err.message : 'Failed to track single order',
   300→   300→    }
   301→   301→  }
   302→   302→}
   303→   303→
   304→   304→export async function pollPostExOrderStatuses(): Promise<ActionResult<{
   305→   305→  polledOrders: number
   306→   306→  polledShipments: number
   307→   307→  statusChanges: number
   308→   308→  errors: string[]
   309→   309→}>> {
   310→   310→  const errors: string[] = []
   311→   311→  let polledOrders = 0
   312→   312→  let polledShipments = 0
   313→   313→  let statusChanges = 0
   314→   314→
   315→   315→  try {
   316→   316→    // Find all active PostEx company integrations
   317→   317→    const postexIntegrations = await db.companyIntegration.findMany({
   318→   318→      where: {
   319→   319→        isActive: true,
   320→   320→        provider: { providerKey: 'postex' },
   321→   321→        credentialsEncrypted: { not: null },
   322→   322→      },
   323→   323→      include: { provider: true },
   324→   324→    })
   325→   325→
   326→   326→    if (postexIntegrations.length === 0) {
   327→   327→      return { success: true, data: { polledOrders: 0, polledShipments: 0, statusChanges: 0, errors: [] } }
   328→   328→    }
   329→   329→
   330→   330→    for (const integration of postexIntegrations) {
   331→   331→      try {
   332→   332→        const credentials = decryptCredentials(integration.credentialsEncrypted!)
   333→   333→        const adapter = getCourierAdapter('postex', credentials)
   334→   334→
   335→   335→        if (!adapter.trackBulkShipments) {
   336→   336→          errors.push(`PostEx adapter for integration ${integration.id} does not support bulk tracking.`)
   337→   337→          continue
   338→   338→        }
   339→   339→
   340→   340→        // ── Fetch active Orders (not delivered/rto) with PostEx tracking ──
   341→   341→        const activeOrders = await db.order.findMany({
   342→   342→          where: {
   343→   343→            courierCompanyIntegrationId: integration.id,
   344→   344→            status: { notIn: ['delivered', 'rto', 'cancelled', 'refunded'] },
   345→   345→            trackingNumber: { not: null },
   346→   346→          },
   347→   347→          select: {
   348→   348→            id: true,
   349→   349→            trackingNumber: true,
   350→   350→            courierSubStatus: true, // we'll use this to detect changes
   351→   351→            status: true,
   352→   352→          },
   353→   353→        })
   354→   354→
   355→   355→        // ── Fetch active Exchange Shipments (not delivered/rto) with PostEx tracking ──
   356→   356→        const activeShipments = await db.exchangeShipment.findMany({
   357→   357→          where: {
   358→   358→            courierCompanyIntegrationId: integration.id,
   359→   359→            status: { notIn: ['delivered', 'rto', 'cancelled'] },
   360→   360→            trackingNumber: { not: null },
   361→   361→          },
   362→   362→          select: {
   363→   363→            id: true,
   364→   364→            trackingNumber: true,
   365→   365→            courierSubStatus: true,
   366→   366→            status: true,
   367→   367→          },
   368→   368→        })
   369→   369→
   370→   370→        // Combine all tracking numbers for bulk API call
   371→   371→        const allTrackingNumbers: Array<{ trackingNumber: string; type: 'order' | 'shipment'; id: string; currentSubStatus: string | null; currentStatus: string }> = [
   372→   372→          ...activeOrders.map((o) => ({
   373→   373→            trackingNumber: o.trackingNumber!,
   374→   374→            type: 'order' as const,
   375→   375→            id: o.id,
   376→   376→            currentSubStatus: o.courierSubStatus ?? null,
   377→   377→            currentStatus: o.status,
   378→   378→          })),
   379→   379→          ...activeShipments.map((s) => ({
   380→   380→            trackingNumber: s.trackingNumber!,
   381→   381→            type: 'shipment' as const,
   382→   382→            id: s.id,
   383→   383→            currentSubStatus: s.courierSubStatus ?? null,
   384→   384→            currentStatus: s.status,
   385→   385→          })),
   386→   386→        ]
   387→   387→
   388→   388→        if (allTrackingNumbers.length === 0) continue
   389→   389→
   390→   390→        // Call the bulk tracking API via the logged wrapper
   391→   391→        const trackingResults = await executeLoggedIntegrationAction<TrackShipmentResult[]>({
   392→   392→          companyIntegrationId: integration.id,
   393→   393→          organizationId: integration.organizationId,
   394→   394→          actionType: 'track_shipment_bulk',
   395→   395→          direction: 'outbound',
   396→   396→          fn: async () => adapter.trackBulkShipments!(allTrackingNumbers.map((t) => t.trackingNumber)),
   397→   397→        })
   398→   398→
   399→   399→        // Map results back to records by tracking number
   400→   400→        const resultsByTrackingNumber = new Map<string, TrackShipmentResult>()
   401→   401→        for (let i = 0; i < allTrackingNumbers.length; i++) {
   402→   402→          if (trackingResults[i]) {
   403→   403→            resultsByTrackingNumber.set(allTrackingNumbers[i].trackingNumber, trackingResults[i])
   404→   404→          }
   405→   405→        }
   406→   406→
   407→   407→        const now = new Date()
   408→   408→
   409→   409→        for (const entry of allTrackingNumbers) {
   410→   410→          const result = resultsByTrackingNumber.get(entry.trackingNumber)
   411→   411→          if (!result || !result.success) {
   412→   412→            // Tracking failed for this number — update lastPolledAt only
   413→   413→            if (entry.type === 'order') {
   414→   414→              await db.order.update({ where: { id: entry.id }, data: { lastPolledAt: now } }).catch((e) => {
   415→   415→                console.error(`[poll] Failed to update lastPolledAt for order ${entry.id}:`, e)
   416→   416→              })
   417→   417→              polledOrders++
   418→   418→            } else {
   419→   419→              await db.exchangeShipment.update({ where: { id: entry.id }, data: { lastPolledAt: now } }).catch((e) => {
   420→   420→                console.error(`[poll] Failed to update lastPolledAt for shipment ${entry.id}:`, e)
   421→   421→              })
   422→   422→              polledShipments++
   423→   423→            }
   424→   424→            continue
   425→   425→          }
   426→   426→
   427→   427→          // Extract the mapped subStatus from the raw response
   428→   428→          const raw = result.rawResponse as Record<string, unknown> | undefined
   429→   429→          const mappedSubStatus = (raw?.mappedSubStatus as string) ?? null
   430→   430→          const needsShipperAdvice = (raw?.needsShipperAdvice as boolean) ?? false
   431→   431→          const unrecognized = (raw?.unrecognized as boolean) ?? false
   432→   432→
   433→   433→          // Check if status changed (compare subStatus)
   434→   434→          const subStatusChanged = mappedSubStatus !== entry.currentSubStatus
   435→   435→
   436→   436→          if (entry.type === 'order') {
   437→   437→            // Update lastPolledAt + courierSubStatus + flags
   438→   438→            // NOTE: we also persist unrecognizedCourierStatus on Order (was
   439→   439→            // previously only persisted on ExchangeShipment — bug fix).
   440→   440→            await db.order.update({
   441→   441→              where: { id: entry.id },
   442→   442→              data: {
   443→   443→                lastPolledAt: now,
   444→   444→                courierSubStatus: mappedSubStatus,
   445→   445→                needsShipperAdvice: needsShipperAdvice,
   446→   446→                unrecognizedCourierStatus: unrecognized,
   447→   447→              },
   448→   448→            }).catch((e) => {
   449→   449→              console.error(`[poll] Failed to update courier status for order ${entry.id}:`, e)
   450→   450→            })
   451→   451→            polledOrders++
   452→   452→
   453→   453→            if (subStatusChanged) {
   454→   454→              statusChanges++
   455→   455→
   456→   456→              // ── "Picked By PostEx" → auto-dispatch ──
   457→   457→              // When PostEx picks up the package, the item has physically left
   458→   458→              // the warehouse. Call performOrderDispatch() which runs the FULL
   459→   459→              // inventory deduction (processInventoryTransaction('sale_dispatched')
