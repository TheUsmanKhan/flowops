    91→// ──────────────────────────────────────────────────────────────
    92→// Processes ALL order_items for an order and attempts to reserve
    93→// stock for each based on its fulfillment_type_snapshot.
    94→//
    95→// For stock_based items:
    96→//   - If sufficient available stock → reserveStockForOrder(), fulfillment_status='reserved'
    97→//   - If insufficient + inventory_policy='continue' → fulfillment_status='backordered'
    98→//   - If insufficient + inventory_policy='deny' → outcome='failed'
    99→//
   100→// For made_to_order items:
   101→//   - checkAndFulfillMadeToOrderVariant() checks returned stock first
   102→//   - If existing_stock → fulfillment_status='reserved', returned_stitched_used=true
   103→//   - If fresh_production → fulfillment_status='reserved', production_order_id set
   104→//   - If error → outcome='failed'
   105→//
   106→// After processing: if ANY item is 'backordered', order status → 'partially_backordered'.
   107→// If ALL items are 'reserved', order stays 'confirmed'.
   108→//
   109→// Called from:
   110→//   - createManualOrder() IF the order auto-confirmed at creation
   111→//   - confirmOrder() when a 'pending' order is manually confirmed
   112→//   - dispatchOrderAction() if status is still 'pending' at dispatch time
   113→
   114→async function reserveOrderStock(
   115→  orderId: string,
   116→  ctx: { company: { id: string; organizationId: string }; user: { id: string }; employee: { id: string } },
   117→): Promise<{
   118→  success: boolean
   119→  results: Array<{ orderItemId: string; outcome: 'reserved' | 'backordered' | 'failed'; reason?: string }>
   120→}> {
   121→  const order = await db.order.findUnique({
   122→    where: { id: orderId },
   123→    select: { dispatchLocationId: true, companyId: true, organizationId: true },
   124→  })
   125→  if (!order) return { success: false, results: [] }
   126→
   127→  const items = await db.orderItem.findMany({
   128→    where: { orderId },
   129→    include: {
   130→      orgVariant: {
   131→        select: { id: true, fulfillmentType: true, inventoryPolicy: true, stitchingCharges: true },
   132→      },
   133→    },
   134→  })
   135→
   136→  const results: Array<{ orderItemId: string; outcome: 'reserved' | 'backordered' | 'failed'; reason?: string }> = []
   137→  let hasBackordered = false
   138→
   139→  for (const item of items) {
   140→    if (item.fulfillmentStatus === 'reserved' || item.fulfillmentStatus === 'dispatched') {
   141→      // Already processed (e.g. re-confirmation attempt) — skip
   142→      results.push({ orderItemId: item.id, outcome: 'reserved' })
   143→      continue
   144→    }
   145→
   146→    const locationId = item.reservedLocationId ?? order.dispatchLocationId
   147→    if (!locationId) {
   148→      results.push({ orderItemId: item.id, outcome: 'failed', reason: 'No dispatch location set' })
   149→      continue
   150→    }
   151→
   152→    if (item.fulfillmentTypeSnapshot === 'stock_based') {
   153→      // Check available stock at the dispatch location
   154→      const pool = await db.inventoryPool.findUnique({
   155→        where: {
   156→          orgVariantId_locationId: {
   157→            orgVariantId: item.orgVariantId,
   158→            locationId,
   159→          },
   160→        },
   161→      })
   162→
   163→      const available = pool ? pool.onHand - pool.reserved : 0
   164→
   165→      if (available >= item.quantity) {
   166→        // Sufficient stock — reserve it
   167→        const reserveResult = await reserveStockForOrder({
   168→          orgVariantId: item.orgVariantId,
   169→          locationId,
   170→          organizationId: order.organizationId,
   171→          companyId: order.companyId,
   172→          employeeId: ctx.employee.id,
   173→          quantity: item.quantity,
   174→          orderId,
   175→        })
   176→
   177→        if (reserveResult.success) {
   178→          await db.orderItem.update({
   179→            where: { id: item.id },
   180→            data: { fulfillmentStatus: 'reserved', reservedLocationId: locationId },
   181→          })
   182→          results.push({ orderItemId: item.id, outcome: 'reserved' })
   183→        } else {
   184→          results.push({ orderItemId: item.id, outcome: 'failed', reason: reserveResult.error })
   185→        }
   186→      } else if (item.orgVariant.inventoryPolicy === 'continue') {
   187→        // Insufficient stock but backordering allowed
   188→        await db.orderItem.update({
   189→          where: { id: item.id },
   190→          data: { fulfillmentStatus: 'backordered', backorderedAt: new Date() },
   191→        })
   192→        await insertMetricEvent({
   193→          companyId: order.companyId,
   194→          entityType: 'product',
   195→          entityId: item.orgVariantId,
   196→          metricKey: 'order.backordered',
   197→          numericValue: item.quantity,
   198→          dimensions: { order_id: orderId },
   199→        }).catch(() => {})
   200→        hasBackordered = true
   201→        results.push({
   202→          orderItemId: item.id,
   203→          outcome: 'backordered',
   204→          reason: `Available: ${available}, required: ${item.quantity}`,
   205→        })
   206→      } else {
   207→        // Insufficient stock + policy='deny' — fail
   208→        results.push({
   209→          orderItemId: item.id,
   210→          outcome: 'failed',
   211→          reason: `Insufficient stock (available: ${available}, required: ${item.quantity}) and inventory_policy='deny'`,
   212→        })
   213→      }
   214→    } else if (item.fulfillmentTypeSnapshot === 'made_to_order') {
   215→      // Made-to-order: check returned stock first, then trigger production
   216→      const mtoResult = await checkAndFulfillMadeToOrderVariant(
   217→        item.orgVariantId,
   218→        item.quantity,
   219→        order.companyId,
   220→        locationId,
   221→      )
   222→
   223→      if (mtoResult.source === 'existing_stock' && mtoResult.locationId) {
   224→        // Returned stock available — reserve it
   225→        const reserveResult = await reserveStockForOrder({
   226→          orgVariantId: item.orgVariantId,
   227→          locationId: mtoResult.locationId,
   228→          organizationId: order.organizationId,
   229→          companyId: order.companyId,
   230→          employeeId: ctx.employee.id,
   231→          quantity: item.quantity,
   232→          orderId,
   233→        })
   234→
   235→        if (reserveResult.success) {
   236→          await db.orderItem.update({
   237→            where: { id: item.id },
   238→            data: {
   239→              fulfillmentStatus: 'reserved',
   240→              returnedStitchedUsed: true,
   241→              reservedLocationId: mtoResult.locationId,
   242→            },
   243→          })
   244→          await insertMetricEvent({
   245→            companyId: order.companyId,
   246→            entityType: 'product',
   247→            entityId: item.orgVariantId,
   248→            metricKey: 'order.made_to_order_from_returned_stock',
   249→            numericValue: 1,
   250→            dimensions: { order_id: orderId, stitching_cost_saved: Number(item.orgVariant.stitchingCharges) || 0 },
   251→          }).catch(() => {})
   252→          results.push({ orderItemId: item.id, outcome: 'reserved' })
   253→        } else {
   254→          results.push({ orderItemId: item.id, outcome: 'failed', reason: reserveResult.error })
   255→        }
   256→      } else if (mtoResult.source === 'fresh_production' && mtoResult.productionOrderId) {
   257→        // Fresh production triggered — link the production order
   258→        await db.orderItem.update({
   259→          where: { id: item.id },
   260→          data: {
   261→            fulfillmentStatus: 'reserved',
   262→            productionOrderId: mtoResult.productionOrderId,
   263→          },
   264→        })
   265→        // Also link the order_item back to the production order
   266→        await db.productionOrder.update({
   267→          where: { id: mtoResult.productionOrderId },
   268→          data: { orderItemId: item.id },
   269→        })
   270→        await insertMetricEvent({
   271→          companyId: order.companyId,
   272→          entityType: 'product',
   273→          entityId: item.orgVariantId,
   274→          metricKey: 'order.made_to_order_production_triggered',
   275→          numericValue: 1,
   276→          dimensions: { order_id: orderId, production_order_id: mtoResult.productionOrderId },
   277→        }).catch(() => {})
   278→        results.push({ orderItemId: item.id, outcome: 'reserved' })
   279→      } else {
   280→        // Error (no fabric source, insufficient fabric, etc.)
   281→        results.push({
   282→          orderItemId: item.id,
   283→          outcome: 'failed',
   284→          reason: mtoResult.error ?? 'Made-to-order fulfillment failed',
   285→        })
   286→      }
   287→    }
   288→  }
   289→
   290→  // Update order status based on results
   291→  if (hasBackordered) {
   292→    await db.order.update({
   293→      where: { id: orderId },
   294→      data: { status: 'partially_backordered' },
   295→    })
   296→  }
   297→  // If no backordered items, order stays 'confirmed' (or whatever it was)
   298→
   299→  return { success: true, results }
   300→}

   305→
   306→export async function createManualOrder(
   307→  input: CreateManualOrderInput,
   308→): Promise<ActionResult<{
   309→  orderId: string
   310→  flowopsOrderNumber: string
   311→  orderItems: Array<{ id: string; orgVariantId: string; quantity: number }>
   312→}>> {
   313→  try {
   314→    const ctx = await getWorkspace()
   315→    await requirePermission(ctx, PERMISSIONS.ORDERS_CREATE)
   316→
   317→    // 1. Validate
   318→    const parsed = createManualOrderSchema.safeParse(input)
   319→    if (!parsed.success) {
   320→      return {
   321→        success: false,
   322→        error: parsed.error.issues[0]?.message ?? 'Invalid order data',
   323→      }
   324→    }
   325→    const d = parsed.data
   326→
   327→    // 2. Resolve customer (existing customer_id OR inline new_customer).
   328→    //    Also resolve optional saved address/phone selection and recipient_name.
   329→    let customerId: string
   330→    let customerName: string
   331→    let usedCustomerAddressId: string | null = null
   332→    let usedCustomerPhoneId: string | null = null
   333→    let saveAddressForNextTime = d.save_address_for_next_time ?? false
   334→    // Whether the delivery_address came from a saved customer_addresses row
   335→    // (vs. typed as a one-off). Used to decide whether to bump lastUsedAt.
   336→    let selectedSavedAddressId: string | null = null
   337→
   338→    if (d.customer_id) {
   339→      // Existing customer path — verify they belong to this org.
   340→      const existing = await db.customer.findFirst({
   341→        where: { id: d.customer_id, organizationId: ctx.company.organizationId },
   342→        select: { id: true, name: true },
   343→      })
   344→      if (!existing) return { success: false, error: 'Customer not found' }
   345→      customerId = existing.id
   346→      customerName = existing.name
   347→
   348→      // If a saved address was selected, verify it belongs to this customer.
   349→      if (d.used_customer_address_id) {
   350→        const addr = await db.customerAddress.findFirst({
   351→          where: {
   352→            id: d.used_customer_address_id,
   353→            customerId: existing.id,
   354→            organizationId: ctx.company.organizationId,
   355→          },
   356→          select: { id: true, address: true, city: true },
   357→        })
   358→        if (addr) {
   359→          usedCustomerAddressId = addr.id
   360→          selectedSavedAddressId = addr.id
   361→          // The delivery_address snapshot is the order's own editable copy.
   362→          // If the caller passed the same address text, we trust it; if they
   363→          // passed different text (edited it), we still record the link to
   364→          // the saved address for lastUsedAt tracking but keep the edited text
   365→          // as the order's snapshot. Either way, delivery_address comes from
   366→          // the input — we do NOT override it here.
   367→        }
   368→      }
   369→
   370→      // If a saved phone was selected, verify it belongs to this customer.
   371→      if (d.used_customer_phone_id) {
   372→        const phone = await db.customerPhone.findFirst({
   373→          where: {
   374→            id: d.used_customer_phone_id,
   375→            customerId: existing.id,
   376→            organizationId: ctx.company.organizationId,
   377→          },
   378→          select: { id: true },
   379→        })
   380→        if (phone) {
   381→          usedCustomerPhoneId = phone.id
   382→        }
   383→      }
   384→    } else if (d.new_customer) {
   385→      // Inline new-customer path — create the customer + their phones +
   386→      // addresses now via the Customer Management System. The first phone is
   387→      // primary and the first address is default (validated by createCustomerSchema).
   388→      const createResult = await createCustomer(d.new_customer)
   389→      if (!createResult.success || !createResult.data) {
   390→        return { success: false, error: createResult.error ?? 'Failed to create customer' }
   391→      }
   392→      customerId = createResult.data.customerId
   393→      customerName = d.new_customer.name.trim()
   394→      // For a brand-new customer, the first address they provided IS the
   395→      // saved default address. Mark it as the selected address so we bump
   396→      // its lastUsedAt after order creation.
   397→      const newAddr = await db.customerAddress.findFirst({
   398→        where: { customerId, isDefault: true },
   399→        select: { id: true },
   400→      })
   401→      if (newAddr) {
   402→        usedCustomerAddressId = newAddr.id
   403→        selectedSavedAddressId = newAddr.id
   404→      }
   405→      // Same for the primary phone.
   406→      const newPhone = await db.customerPhone.findFirst({
   407→        where: { customerId, isPrimary: true },
   408→        select: { id: true },
   409→      })
   410→      if (newPhone) {
   411→        usedCustomerPhoneId = newPhone.id
   412→      }
   413→      // For a brand-new customer, we already saved the address as part of
   414→      // createCustomer — don't double-save it.
   415→      saveAddressForNextTime = false
   416→    } else {
   417→      return { success: false, error: 'Either customer_id or new_customer is required' }
   418→    }
   419→
   420→    // 3. recipient_name defaults to the customer's name if not explicitly overridden.
   421→    const recipientName = (d.recipient_name && d.recipient_name.trim()) || customerName
   422→
   423→    // 3. Fetch variants + their pricing for this company + fulfillment_type snapshot
   424→    const variantIds = d.items.map((i) => i.org_variant_id)
   425→    const variants = await db.orgProductVariant.findMany({
   426→      where: { id: { in: variantIds }, organizationId: ctx.company.organizationId },
   427→      include: {
   428→        product: { select: { title: true } },
   429→        companyPricing: { where: { companyId: ctx.company.id } },
   430→      },
   431→    })
   432→
   433→    if (variants.length !== variantIds.length) {
   434→      return { success: false, error: 'One or more variants not found in this organization' }
   435→    }
   436→
   437→    // 4. Compute subtotal + build order items data
   438→    let subtotal = 0
   439→    const orderItemsData: Array<{
   440→      orgVariantId: string
   441→      quantity: number
   442→      unitPrice: number
   443→      lineTotal: number
   444→      fulfillmentTypeSnapshot: string
   445→    }> = []
   446→
   447→    // Build the order_detail string as we iterate — universal courier
   448→    // reference field (migration 015). Format:
   449→    //   "Product Title (SKU-001, Size: M, Color: Blue) ×2, ..."
   450→    // Auto-generated but editable from the order-create form. If the caller
   451→    // supplies an explicit order_detail, we honour it verbatim.
   452→    const orderDetailParts: string[] = []
   453→
   454→    for (const item of d.items) {
   455→      const variant = variants.find((v) => v.id === item.org_variant_id)
   456→      if (!variant) continue
   457→
   458→      // Use provided unit_price OR fall back to company pricing OR variant cost
   459→      const unitPrice =
   460→        item.unit_price ??
   461→        (variant.companyPricing[0]?.salePrice
   462→          ? Number(variant.companyPricing[0].salePrice)
   463→          : Number(variant.costPrice))
   464→
   465→      const lineTotal = item.quantity * unitPrice
   466→      subtotal += lineTotal
   467→
   468→      orderItemsData.push({
   469→        orgVariantId: item.org_variant_id,
   470→        quantity: item.quantity,
   471→        unitPrice,
   472→        lineTotal,
   473→        fulfillmentTypeSnapshot: variant.fulfillmentType,
   474→      })
   475→
   476→      // Append to orderDetail: "Product Title (SKU-001, Size: M) ×2"
   477→      const attrParts: string[] = []
   478→      try {
   479→        const attrs = JSON.parse(variant.attributeValues || '{}') as Record<string, string>
   480→        for (const [k, v] of Object.entries(attrs)) {
   481→          if (v) attrParts.push(`${k}: ${v}`)
   482→        }
   483→      } catch {
   484→        // ignore parse errors
   485→      }
   486→      const inner = [variant.sku, ...attrParts].filter(Boolean).join(', ')
   487→      orderDetailParts.push(
   488→        `${variant.product.title}${inner ? ` (${inner})` : ''} ×${item.quantity}`,
   489→      )
   490→    }
   491→
   492→    // 5. Compute totals
   493→    const discountAmount = d.discount_amount ?? 0
   494→    const courierCharges = d.courier_charges ?? 0
   495→    const estimatedDeliveryCharge = d.estimated_delivery_charge ?? 0
   496→    const taxAmount = d.tax_amount ?? 0
   497→    // Total = subtotal + courier charges + delivery charge + tax - discount
   498→    // (delivery charge and tax are ADDITIVE — not absorbed into subtotal)
   499→    const totalOrderValue = subtotal + courierCharges + estimatedDeliveryCharge + taxAmount - discountAmount
   500→
   501→    // 6. Determine payment status + source
   502→    let paymentStatus: string = 'cod_pending'
   503→    let paymentSource: string = 'cod_native'
   504→    let advanceAmount: number | null = null
   505→    let advancePaymentMethod: string | null = null
   506→    let advancePaymentReference: string | null = null
   507→    let advancePaidAt: Date | null = null
   508→
   509→    if (d.payment_type === 'partial_advance') {
   510→      paymentStatus = 'advance_paid'
   511→      paymentSource = 'manual_conversion'
   512→      advanceAmount = d.advance_amount ?? 0
   513→      advancePaymentMethod = d.advance_payment_method || null
   514→      advancePaymentReference = d.advance_payment_reference || null
   515→      advancePaidAt = new Date()
   516→    } else if (d.payment_type === 'fully_prepaid') {
   517→      paymentStatus = 'fully_prepaid'
   518→      paymentSource = 'manual_conversion'
   519→      advanceAmount = totalOrderValue
   520→      advancePaymentMethod = d.advance_payment_method || null
   521→      advancePaymentReference = d.advance_payment_reference || null
   522→      advancePaidAt = new Date()
   523→    }
   524→
   525→    // remainingCodAmount is NOT a GENERATED column in the DB — the application
   526→    // MUST compute it as totalOrderValue - (advanceAmount ?? 0).
   527→    const remainingCodAmount = totalOrderValue - (advanceAmount ?? 0)
   528→
   529→    // 7. Determine initial order status
   530→    let orderStatus: string = 'pending'
   531→    let confirmedAt: Date | null = null
   532→    let skippedConfirmation = false
   533→
   534→    if (paymentStatus === 'advance_paid' || paymentStatus === 'fully_prepaid') {
   535→      // Payment itself is the confirmation signal — bypasses require_order_confirmation
   536→      orderStatus = 'confirmed'
   537→      confirmedAt = new Date()
   538→    } else {
   539→      // COD order — check company_order_settings
   540→      const settings = await db.companyOrderSetting.findUnique({
   541→        where: { companyId: ctx.company.id },
   542→      })
   543→      if (!settings?.requireOrderConfirmation) {
   544→        orderStatus = 'confirmed'
   545→        confirmedAt = new Date()
   546→        skippedConfirmation = true
   547→      }
   548→    }
   549→
   550→    // 8. Generate order number
   551→    const flowopsOrderNumber = await generateOrderNumber(ctx.company.id)
   552→
   553→    // 8a. Universal courier reference fields (migration 015).
   554→    // orderRefNumber: defaults to flowopsOrderNumber if the caller left it
   555→    // blank — every courier has a "reference" field so this is a core OMS
   556→    // field, mapped to the courier's own ref at booking time.
   557→    const orderRefNumber =
   558→      (d.order_ref_number && d.order_ref_number.trim()) || flowopsOrderNumber
   559→    // orderDetail: caller-provided takes precedence, otherwise use the
   560→    // auto-generated string from the cart items (product title + SKU +
   561→    // variant attributes + quantity).
   562→    const orderDetail =
   563→      (d.order_detail && d.order_detail.trim()) || orderDetailParts.join(', ')
   564→

   595→        advancePaymentMethod,
   596→        advancePaymentReference,
   597→        advancePaidAt,
   598→        remainingCodAmount,

   603→        recommendedCourierCompanyIntegrationId: d.courier_company_integration_id || null,
   604→        dispatchLocationId: d.dispatch_location_id,
   605→        notesForCourier: d.notes_for_courier || null,
   606→        // Universal courier reference fields (migration 015)
   607→        orderRefNumber,
   608→        orderDetail,
   609→        skippedConfirmation,
   610→        confirmedAt,
   611→        createdBy: ctx.employee.id,
   612→      },
   613→    })
   614→

   616→    const createdItems: Array<{ id: string; orgVariantId: string; quantity: number }> = []
   617→    for (const itemData of orderItemsData) {
   618→      const item = await db.orderItem.create({

   620→          orderId: order.id,
   621→          orgVariantId: itemData.orgVariantId,

   623→          quantity: itemData.quantity,
   624→          unitPrice: itemData.unitPrice,
   625→          lineTotal: itemData.lineTotal,

   627→          fulfillmentTypeSnapshot: itemData.fulfillmentTypeSnapshot,

   630→      })

   632→        id: item.id,

   640→      action: 'order.created',
   641→      entityType: 'order',
   642→      entityId: order.id,
   643→      companyId: ctx.company.id,
   644→      organizationId: ctx.company.organizationId,
   645→      userId: ctx.user.id,
   646→      employeeId: ctx.employee.id,
   647→      newValues: {
   648→        flowopsOrderNumber,
   649→        status: orderStatus,
   650→        paymentType: d.payment_type,
   651→        paymentStatus,
   652→        itemCount: orderItemsData.length,
   653→        totalOrderValue,
   654→        customerId,
   655→      },
   656→    })
   657→
   658→    // 12. Customer Management System integration:
   659→    //   a. If a saved address was selected, bump its lastUsedAt to NOW() so
   660→    //      the UI can sort saved addresses by recency.
   661→    //   b. If a brand-new one-off address was typed AND the caller set
   662→    //      save_address_for_next_time=true, persist it as a new
   663→    //      customer_addresses row (and link it via usedCustomerAddressId
   664→    //      for future tracking). This is the "Save this address for next
   665→    //      time" UX from Step 3's frontend.
   666→    //   c. Recompute cached customer stats (total_orders_count, etc.) and
   667→    //      auto-flag if RTO threshold crossed.
   668→    if (selectedSavedAddressId) {
   669→      await markAddressAsUsed(selectedSavedAddressId)
   670→    } else if (saveAddressForNextTime && d.delivery_address && d.delivery_city) {
   671→      // One-off address typed + user opted to save it for next time.
   672→      // Persist as a non-default customer_addresses row (default stays as-is
   673→      // unless the user explicitly changes it via the customer profile).
   674→      const savedAddr = await db.customerAddress.create({
   675→        data: {
   676→          customerId,
   677→          organizationId: ctx.company.organizationId,
   678→          label: null,
   679→          address: d.delivery_address,
   680→          city: d.delivery_city,
   681→          isDefault: false,
   682→          lastUsedAt: new Date(),
   683→        },
   684→      })
   685→      // Link the order to the newly-saved address for future tracking.
   686→      await db.order.update({
   687→        where: { id: order.id },
   688→        data: { usedCustomerAddressId: savedAddr.id },
   689→      })
   690→    }
   691→
   692→    await updateCustomerStats(customerId)
   693→
   694→    // 13. If the order auto-confirmed (payment-driven or company setting),
   695→    // run the stock reservation logic immediately.
   696→    if (orderStatus === 'confirmed') {
   697→      await reserveOrderStock(order.id, ctx)
   698→    }
   699→
   700→    await insertMetricEvent({
   701→      companyId: ctx.company.id,
   702→      entityType: 'order',
   703→      entityId: order.id,
   704→      metricKey: 'order.created',
   705→      numericValue: Number(totalOrderValue),
   706→      dimensions: { order_source: 'manual', payment_type: d.payment_type, company_id: ctx.company.id },
   707→    }).catch(() => {})
   708→
   709→    return {
   710→      success: true,
   711→      data: { orderId: order.id, flowopsOrderNumber, orderItems: createdItems },
   712→    }
   713→  } catch (err) {
   714→    return {
   715→      success: false,
   716→      error: err instanceof Error ? err.message : 'Failed to create order',
   717→    }
   718→  }
   719→}

  1002→export async function confirmOrder(orderId: string): Promise<ActionResult> {
  1003→  try {
  1004→    const ctx = await getWorkspace()
  1005→    await requirePermission(ctx, PERMISSIONS.ORDERS_MANAGE)
  1006→
  1007→    const order = await db.order.findFirst({
  1008→      where: { id: orderId, companyId: ctx.company.id },
  1009→    })
  1010→    if (!order) return { success: false, error: 'Order not found' }
  1011→    if (order.status !== 'pending') {
  1012→      return { success: false, error: `Order is already ${order.status} (cannot confirm)` }
  1013→    }
  1014→
  1015→    await db.order.update({
  1016→      where: { id: orderId },
  1017→      data: { status: 'confirmed', confirmedAt: new Date() },
  1018→    })
  1019→
  1020→    await insertAuditLog({
  1021→      action: 'order.confirmed',
  1022→      entityType: 'order',
  1023→      entityId: orderId,
  1024→      companyId: ctx.company.id,
  1025→      organizationId: ctx.company.organizationId,
  1026→      userId: ctx.user.id,
  1027→      employeeId: ctx.employee.id,
  1028→      oldValues: { status: 'pending' },
  1029→      newValues: { status: 'confirmed' },
  1030→    })
  1031→
  1032→    // Step 3: Run stock reservation for all items.
  1033→    // This may flip the order to 'partially_backordered' if some items
  1034→    // can't be fulfilled immediately.
  1035→    const reserveResult = await reserveOrderStock(orderId, ctx)
  1036→    if (!reserveResult.success) {
  1037→      // Non-fatal — the order is confirmed, but reservation had issues.
  1038→      // The results array has per-item details.
  1039→    }
  1040→
  1041→    await insertMetricEvent({
  1042→      companyId: ctx.company.id,
  1043→      entityType: 'order',
  1044→      entityId: orderId,
  1045→      metricKey: 'order.confirmed',
  1046→      numericValue: Number(order.totalOrderValue),
  1047→      dimensions: { confirmation_method: 'manual' },
  1048→    }).catch(() => {})
  1049→
  1050→    // Recompute cached customer stats (order count may change if this was a
  1051→    // re-confirmation, and the status transition affects value/rto calcs).
  1052→    // Non-fatal — never break the confirm action on a stats failure.
  1053→    await updateCustomerStats(order.customerId).catch(() => {})
  1054→
  1055→    return { success: true }
  1056→  } catch (err) {
  1057→    return {
  1058→      success: false,
  1059→      error: err instanceof Error ? err.message : 'Failed to confirm order',
  1060→    }
  1061→  }
  1062→}

  1305→export async function cancelOrder(input: CancelOrderInput): Promise<ActionResult> {
  1306→  try {
  1307→    const ctx = await getWorkspace()
  1308→    await requirePermission(ctx, PERMISSIONS.ORDERS_CANCEL)
  1309→
  1310→    const parsed = cancelOrderSchema.safeParse(input)
  1311→    if (!parsed.success) {
  1312→      return { success: false, error: parsed.error.issues[0]?.message ?? 'Invalid input' }
  1313→    }
  1314→    const d = parsed.data
  1315→
  1316→    const order = await db.order.findFirst({
  1317→      where: { id: d.order_id, companyId: ctx.company.id },
  1318→    })
  1319→    if (!order) return { success: false, error: 'Order not found' }
  1320→
  1321→    const nonCancellableStatuses = ['dispatched', 'delivered', 'rto', 'cancelled', 'refunded']
  1322→    if (nonCancellableStatuses.includes(order.status)) {
  1323→      return {
  1324→        success: false,
  1325→        error: `Cannot cancel an order that is already ${order.status} (use the returns flow instead)`,
  1326→      }
  1327→    }
  1328→
  1329→    await db.order.update({
  1330→      where: { id: d.order_id },
  1331→      data: {
  1332→        status: 'cancelled',
  1333→        cancelledAt: new Date(),
  1334→        cancellationReason: d.cancellation_reason,
  1335→      },
  1336→    })
  1337→
  1338→    // Step 3: Unreserve stock for any items with fulfillment_status='reserved'.
  1339→    // Items with fulfillment_status='backordered' need no inventory action
  1340→    // (no reservation ever existed for them) — they'll be orphaned since
  1341→    // the order is now cancelled and won't be picked up by future
  1342→    // checkAndFulfillBackorders() runs (which skip cancelled orders).
  1343→    const reservedItems = await db.orderItem.findMany({
  1344→      where: { orderId: d.order_id, fulfillmentStatus: 'reserved' },
  1345→    })
  1346→
  1347→    for (const item of reservedItems) {
  1348→      const locationId = item.reservedLocationId ?? order.dispatchLocationId
  1349→      if (!locationId) continue
  1350→
  1351→      await unreserveStockForOrder({
  1352→        orgVariantId: item.orgVariantId,
  1353→        locationId,
  1354→        organizationId: order.organizationId,
  1355→        companyId: ctx.company.id,
  1356→        employeeId: ctx.employee.id,
  1357→        quantity: item.quantity,
  1358→        orderId: d.order_id,
  1359→      })
  1360→    }
  1361→
  1362→    await insertAuditLog({
  1363→      action: 'order.cancelled',
  1364→      entityType: 'order',
  1365→      entityId: d.order_id,
  1366→      companyId: ctx.company.id,
  1367→      organizationId: ctx.company.organizationId,
  1368→      userId: ctx.user.id,
  1369→      employeeId: ctx.employee.id,
  1370→      oldValues: { status: order.status },
  1371→      newValues: { status: 'cancelled', reason: d.cancellation_reason, unreservedItems: reservedItems.length },
  1372→    })
  1373→
  1374→    await insertMetricEvent({
  1375→      companyId: ctx.company.id,
  1376→      entityType: 'order',
  1377→      entityId: d.order_id,
  1378→      metricKey: 'order.cancelled',
  1379→      numericValue: Number(order.totalOrderValue),
  1380→      dimensions: { cancellation_reason: d.cancellation_reason, had_reserved_items: reservedItems.length > 0 },
  1381→    }).catch(() => {})
  1382→
  1383→    // Recompute cached customer stats (cancelled orders are excluded from
  1384→    // total_orders_count, so this count drops). Non-fatal.
  1385→    await updateCustomerStats(order.customerId).catch(() => {})
  1386→
  1387→    return { success: true }
  1388→  } catch (err) {
  1389→    return {
  1390→      success: false,
  1391→      error: err instanceof Error ? err.message : 'Failed to cancel order',
  1392→    }
  1393→  }
  1394→}

  1439→    const offset = filters.offset ?? 0

  1442→      companyId: ctx.company.id,

  1444→
  1445→    // Status filter — prefer multi-select `statuses`, fall back to single `status`
  1446→    if (filters.statuses && filters.statuses.length > 0) {
  1447→      where.status = { in: filters.statuses }
  1448→    } else if (filters.status) {
  1449→      where.status = filters.status
  1450→    }
  1451→
  1452→    // Payment type filter — prefer multi-select `paymentTypes`, fall back to single `paymentType`
  1453→    if (filters.paymentTypes && filters.paymentTypes.length > 0) {
  1454→      where.paymentType = { in: filters.paymentTypes }
  1455→    } else if (filters.paymentType) {
  1456→      where.paymentType = filters.paymentType
  1457→    }
  1458→
  1459→    // Payment status filter (multi-select preferred)
  1460→    if (filters.paymentStatuses && filters.paymentStatuses.length > 0) {
  1461→      where.paymentStatus = { in: filters.paymentStatuses }
  1462→    } else if (filters.paymentStatus) {
  1463→      where.paymentStatus = filters.paymentStatus
  1464→    }
  1465→
  1466→    // Order source filter — prefer multi-select `orderSources`, fall back to single `orderSource`
  1467→    if (filters.orderSources && filters.orderSources.length > 0) {
  1468→      where.orderSource = { in: filters.orderSources }
  1469→    } else if (filters.orderSource) {
  1470→      where.orderSource = filters.orderSource
  1471→    }
  1472→
  1473→    // Courier filter — prefer multi-select `courierNames`, fall back to single `courierName`
  1474→    if (filters.courierNames && filters.courierNames.length > 0) {
  1475→      where.courierName = { in: filters.courierNames }
  1476→    } else if (filters.courierName) {
  1477→      where.courierName = filters.courierName
  1478→    }
  1479→
  1480→    if (filters.customerId) where.customerId = filters.customerId
  1481→
  1482→    // Delivery city filter — case-insensitive contains on delivery_city
  1483→    if (filters.deliveryCity) {
  1484→      where.deliveryCity = { contains: filters.deliveryCity, mode: 'insensitive' }
  1485→    }
  1486→

  1490→      if (filters.dateTo) {
