   167→    queryKey: ['product', productId],

   481→function PricingTab({
   482→  productId,
   483→  variants,
   484→  onUpdate,
   485→}: {
   486→  productId: string
   487→  variants: ProductVariant[]
   488→  onUpdate: () => void
   489→}) {
   490→  const [drafts, setDrafts] = useState<Record<string, { salePrice: number | null; comparePrice: number | null }>>(
   491→    () =>
   492→      Object.fromEntries(
   493→        variants.map((v) => [
   494→          v.id,
   495→          { salePrice: v.salePrice, comparePrice: v.comparePrice },
   496→        ]),
   497→      ),
   498→  )
   499→  const [saving, setSaving] = useState<string | null>(null)
   500→
   501→  function setDraft(
   502→    variantId: string,
   503→    field: 'salePrice' | 'comparePrice',
   504→    value: number | null,
   505→  ) {
   506→    setDrafts((d) => ({
   507→      ...d,
   508→      [variantId]: { ...d[variantId], [field]: value },
   509→    }))
   510→  }
   511→
   512→  async function save(variantId: string) {
   513→    const draft = drafts[variantId]
   514→    if (!draft) return
   515→    if (draft.salePrice == null || draft.salePrice <= 0) {
   516→      toast.error('Sale price must be a positive number.')
   517→      return
   518→    }
   519→    setSaving(variantId)
   520→    try {
   521→      // POST /api/products/[id]/pricing expects { pricing: [{ org_variant_id, sale_price, compare_price? }] }
   522→      await api.post(`/api/products/${productId}/pricing`, {
   523→        pricing: [
   524→          {
   525→            org_variant_id: variantId,
   526→            sale_price: draft.salePrice,
   527→            ...(draft.comparePrice != null ? { compare_price: draft.comparePrice } : {}),
   528→          },
   529→        ],
   530→      })
   531→      toast.success('Pricing updated.')
   532→      onUpdate()
   533→    } catch (err) {
   534→      toast.error(err instanceof FetchError ? err.message : 'Failed to update pricing.')
   535→    } finally {
   536→      setSaving(null)
   537→    }
   538→  }
   539→
   540→  return (
   541→    <Card>
   542→      <CardHeader>
   543→        <CardTitle className="text-base flex items-center gap-2">
   544→          <Tag className="h-4 w-4 text-primary" /> Your company pricing
   545→        </CardTitle>
   546→        <CardDescription>
   547→          Set the sale and compare-at prices for your storefront.
   548→        </CardDescription>
   549→      </CardHeader>
   550→      <CardContent className="p-0">
   551→        <div className="overflow-x-auto">
   552→          <Table>
   553→            <TableHeader>
   554→              <TableRow>
   555→                <TableHead>SKU</TableHead>
   556→                <TableHead className="text-right">Cost</TableHead>
   557→                <TableHead>Sale price</TableHead>
   558→                <TableHead>Compare price</TableHead>
   559→                <TableHead className="text-right">Action</TableHead>
   560→              </TableRow>
   561→            </TableHeader>
   562→            <TableBody>
   563→              {variants.map((v) => {
   564→                const draft = drafts[v.id] ?? { salePrice: v.salePrice, comparePrice: v.comparePrice }
   565→                const dirty =
   566→                  draft.salePrice !== v.salePrice ||
   567→                  draft.comparePrice !== v.comparePrice
   568→                const canSave =
   569→                  dirty &&
   570→                  draft.salePrice != null &&
   571→                  draft.salePrice > 0 &&
   572→                  (draft.comparePrice == null || draft.comparePrice > draft.salePrice)
   573→                return (
   574→                  <TableRow key={v.id}>
   575→                    <TableCell className="font-mono text-xs">{v.sku}</TableCell>
   576→                    <TableCell className="text-right text-xs text-muted-foreground">
   577→                      {formatMoney(v.costPrice)}
   578→                    </TableCell>
   579→                    <TableCell>
   580→                      <Input
   581→                        type="number"
   582→                        min="0"
   583→                        step="0.01"
   584→                        value={draft.salePrice ?? ''}
   585→                        onChange={(e) =>
   586→                          setDraft(
   587→                            v.id,
   588→                            'salePrice',
   589→                            e.target.value ? Number(e.target.value) : null,
   590→                          )
   591→                        }
   592→                        className="h-8 w-32 text-xs"
   593→                      />
   594→                    </TableCell>
   595→                    <TableCell>
   596→                      <Input
   597→                        type="number"
   598→                        min="0"
   599→                        step="0.01"
   600→                        value={draft.comparePrice ?? ''}
   601→                        onChange={(e) =>
   602→                          setDraft(
   603→                            v.id,
   604→                            'comparePrice',
   605→                            e.target.value ? Number(e.target.value) : null,
   606→                          )
   607→                        }
   608→                        className="h-8 w-32 text-xs"
   609→                      />
   610→                    </TableCell>
   611→                    <TableCell className="text-right">
   612→                      <Button
   613→                        size="sm"
   614→                        variant={dirty ? 'default' : 'outline'}
   615→                        disabled={!canSave || saving === v.id}
   616→                        onClick={() => save(v.id)}
   617→                      >
   618→                        {saving === v.id ? (
   619→                          <Loader2 className="h-3.5 w-3.5 animate-spin" />

  1237→/** Interactive is_active switch with optimistic update + revert on error. */
  1238→function VariantActiveSwitch({
  1239→  productId,
  1240→  variantId,
  1241→  isActive,
  1242→  canEdit,
  1243→}: {
  1244→  productId: string
  1245→  variantId: string
  1246→  isActive: boolean
  1247→  canEdit: boolean
  1248→}) {
  1249→  const queryClient = useQueryClient()
  1250→
  1251→  const mutation = useMutation({
  1252→    mutationFn: (newVal: boolean) =>
  1253→      api.post(`/api/products/${productId}/variants/${variantId}/toggle`, {
  1254→        is_active: newVal,
  1255→      }),
  1256→    onMutate: async (newVal) => {
  1257→      // Optimistic update — flip the cached variant's isActive immediately.
  1258→      await queryClient.cancelQueries({ queryKey: ['product', productId] })
  1259→      const prev = queryClient.getQueryData<{ product: ProductDetail }>(['product', productId])
  1260→      if (prev) {
  1261→        queryClient.setQueryData<{ product: ProductDetail }>(['product', productId], {
  1262→          product: {
  1263→            ...prev.product,
  1264→            variants: prev.product.variants.map((v) =>
  1265→              v.id === variantId ? { ...v, isActive: newVal } : v
  1266→            ),
  1267→          },
  1268→        })
  1269→      }
  1270→      return { prev }
  1271→    },
  1272→    onError: (err, _newVal, context) => {
  1273→      // Revert the optimistic update.
  1274→      if (context?.prev) {
  1275→        queryClient.setQueryData(['product', productId], context.prev)
  1276→      }
  1277→      toast.error(err instanceof FetchError ? err.message : 'Failed to toggle variant.')
  1278→    },
  1279→    onSuccess: () => {
  1280→      queryClient.invalidateQueries({ queryKey: ['product', productId] })
  1281→      toast.success('Variant status updated.')
  1282→    },
  1283→  })
  1284→
  1285→  return (
  1286→    <Switch
  1287→      checked={isActive}
  1288→      disabled={!canEdit || mutation.isPending}
  1289→      onCheckedChange={(v) => mutation.mutate(v)}
  1290→      aria-label="Toggle variant active"
  1291→    />
  1292→  )
  1293→}

  1534→function ImagesTab({

  1544→  const fileInputRef = useRef<HTMLInputElement>(null)
  1545→  const [uploading, setUploading] = useState(false)
  1546→  const [deletingId, setDeletingId] = useState<string | null>(null)
  1547→
  1548→  // NOTE: The backend route `PATCH /api/products/[id]/images/[imageId]` for
  1549→  // setting is_primary does not exist yet. Upload already auto-sets the first
  1550→  // image as primary, so the manual primary toggle is intentionally skipped
  1551→  // for now. When that route lands, wire a "Set Primary" star button per
  1552→  // non-primary image here.
  1553→
  1554→  const uploadMutation = useMutation({
  1555→    mutationFn: async (file: File) => {
  1556→      // Use raw fetch — multipart FormData must NOT have a JSON Content-Type.
  1557→      const fd = new FormData()
  1558→      fd.append('file', file)
  1559→      const res = await fetch(`/api/products/${productId}/images`, {
  1560→        method: 'POST',
  1561→        body: fd,
  1562→        cache: 'no-store',

  1564→      const text = await res.text()
  1565→      let body: unknown = null
  1566→      if (text) {
  1567→        try {
  1568→          body = JSON.parse(text)
  1569→        } catch {
  1570→          body = text
  1571→        }
  1572→      }
  1573→      if (!res.ok) {
  1574→        const msg =
  1575→          body && typeof body === 'object' && 'error' in body
  1576→            ? String((body as { error: unknown }).error)
  1577→            : typeof body === 'string'
  1578→              ? body
  1579→              : 'Upload failed'
  1580→        throw new FetchError(res.status, msg)
  1581→      }
  1582→      return body as { image_id: string; public_url: string; is_primary: boolean }
  1583→    },

  1585→      queryClient.invalidateQueries({ queryKey: ['product', productId] })

  1592→      setUploading(false)

  1596→  const deleteMutation = useMutation({

  1619→      toast.error('File too large. Maximum 5 MB.')

  1621→    }

  1642→              onChange={onPickFile}
  1643→              disabled={uploading}
  1644→            />
  1645→            <Button
  1646→              size="sm"

  1655→              Upload

  1658→        )}
  1659→      </CardHeader>
  1660→      <CardContent>

  1674→          <div className="grid gap-4 grid-cols-2 sm:grid-cols-3 lg:grid-cols-4">
