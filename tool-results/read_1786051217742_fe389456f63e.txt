   150→function getErrorMessage(err: unknown): string {
   151→  if (err instanceof FetchError) return err.message
   152→  if (err instanceof Error) return err.message
   153→  return 'Something went wrong. Please try again.'
   154→}

   156→function parseQty(v: string): number {
   157→  if (v === '') return 0
   158→  const n = Number(v)
   159→  return Number.isFinite(n) && n >= 0 ? Math.floor(n) : 0
   160→}
   161→
   162→function parsePrice(v: string): number {
   163→  if (v === '') return 0
   164→  const n = Number(v)
   165→  return Number.isFinite(n) && n >= 0 ? n : 0
   166→}

   732→  // ── Upload payment proof + persist URL to the order ───────────────────────
   733→  // Returns true on success, false on any failure. NEVER throws.
   734→  async function uploadPaymentProof(orderId: string): Promise<boolean> {
   735→    if (!paymentProofFile) return true
   736→
   737→    setUploadingProof(true)
   738→    try {
   739→      const fd = new FormData()
   740→      fd.append('file', paymentProofFile)
   741→      const uploadRes = await fetch(
   742→        `/api/upload?type=payment-proofs&id=${orderId}`,
   743→        { method: 'POST', body: fd, credentials: 'include', headers: getSessionToken() ? { Authorization: `Bearer ${getSessionToken()}` } : {} },
   744→      )
   745→      const uploadText = await uploadRes.text()
   746→      let uploadBody: unknown = null
   747→      if (uploadText) {
   748→        try {
   749→          uploadBody = JSON.parse(uploadText)
   750→        } catch {
   751→          uploadBody = uploadText
   752→        }
   753→      }
   754→      if (!uploadRes.ok) {
   755→        const msg =
   756→          uploadBody && typeof uploadBody === 'object' && 'error' in uploadBody
   757→            ? String((uploadBody as { error: unknown }).error)
   758→            : typeof uploadBody === 'string'
   759→              ? uploadBody
   760→              : `Upload failed (HTTP ${uploadRes.status})`
   761→        throw new Error(msg)
   762→      }
   763→      const { url } = uploadBody as { url: string }
   764→
   765→      await api.post(`/api/orders/${orderId}/payment-proof`, {
   766→        advance_payment_screenshot_url: url,
   767→      })
   768→      return true
   769→    } catch {
   770→      return false
   771→    } finally {
   772→      setUploadingProof(false)
   773→    }
   774→  }

  1563→function CrmStatsWidget({ customer }: { customer: SelectedCustomer }) {
  1564→  const totalOrders = customer.totalOrdersCount
  1565→  const totalRto = customer.totalRtoCount
  1566→  // Derive a delivery rate from the cached totals (returned orders / total).
  1567→  // This is informational — not the source of truth (the live query is).
  1568→  const deliveryRate =
  1569→    totalOrders > 0 ? ((totalOrders - totalRto) / totalOrders) * 100 : 100
  1570→  const rtoRate = totalOrders > 0 ? (totalRto / totalOrders) * 100 : 0
  1571→  const deliveredCount = Math.max(0, totalOrders - totalRto)
  1572→
  1573→  return (
  1574→    <div className="space-y-3">
  1575→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-2">

  1578→          label="Delivered"
  1579→          value={String(deliveredCount)}
  1580→          sub={`${deliveryRate.toFixed(1)}%`}
  1581→          tone="emerald"
  1582→        />
  1583→        <StatCell
  1584→          label="Returned"
  1585→          value={String(totalRto)}
  1586→          sub={`${rtoRate.toFixed(1)}%`}
  1587→          tone="rose"
  1588→        />
  1589→        <StatCell
  1590→          label="Delivery Rate"
  1591→          value={`${deliveryRate.toFixed(1)}%`}
  1592→          tone="emerald"
  1593→        />
  1594→      </div>
  1595→
  1596→      {totalOrders > 0 && (
  1597→        <p className="text-xs text-muted-foreground flex items-center gap-1.5">
  1598→          <History className="h-3 w-3" />
  1599→          Returning customer — verify address before dispatch.
  1600→        </p>
  1601→      )}
  1602→
  1603→      {/* Saved address history (from customer.addresses[] with lastUsedAt) */}
  1604→      {customer.addresses.length > 0 && (
  1605→        <div className="space-y-1.5">
  1606→          <p className="text-[10px] uppercase tracking-wide text-muted-foreground flex items-center gap-1">
  1607→            <MapPin className="h-3 w-3" /> Saved addresses
  1608→          </p>
  1609→          <ul className="space-y-1 max-h-32 overflow-y-auto pr-1 scrollbar-thin">
  1610→            {customer.addresses.slice(0, 5).map((a) => (
  1611→              <li
  1612→                key={a.id}
  1613→                className="text-xs flex items-center justify-between gap-2 rounded bg-background/60 px-2 py-1"
  1614→              >

  1638→function formatLastUsedShort(iso: string | null): string {
  1639→  if (!iso) return 'Never used'
  1640→  const date = new Date(iso)
  1641→  const now = new Date()
  1642→  const diffMs = now.getTime() - date.getTime()
  1643→  const diffDays = Math.floor(diffMs / 86_400_000)
  1644→  if (diffDays < 1) return 'Today'
  1645→  if (diffDays === 1) return '1 day ago'
  1646→  if (diffDays < 30) return `${diffDays} days ago`
  1647→  return date.toLocaleDateString('en-PK', { day: '2-digit', month: 'short' })
  1648→}
