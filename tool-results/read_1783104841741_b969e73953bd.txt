   280→   280→   Otherwise check role_permissions table.

   298→   298→
   299→   299→profiles:
   300→   300→  SELECT: auth.uid() = id
   301→   301→  UPDATE: auth.uid() = id
   302→   302→
   303→   303→organizations:
   304→   304→  SELECT: owner_id = auth.uid()
   305→   305→          OR EXISTS (SELECT 1 FROM employees e 
   306→   306→                     JOIN companies c ON c.id = e.company_id 
   307→   307→                     WHERE c.organization_id = organizations.id 
   308→   308→                     AND e.user_id = auth.uid() 
   309→   309→                     AND e.status = 'active')
   310→   310→  INSERT: auth.uid() IS NOT NULL (any logged in user can create an org)
   311→   311→  UPDATE: owner_id = auth.uid()
   312→   312→  DELETE: owner_id = auth.uid()
   313→   313→
   314→   314→companies:
   315→   315→  SELECT: company_id = get_active_company_id()
   316→   316→          AND EXISTS (SELECT 1 FROM employees 
   317→   317→                      WHERE company_id = companies.id 
   318→   318→                      AND user_id = auth.uid() 
   319→   319→                      AND status = 'active')
   320→   320→  INSERT: organization owner can create companies
   321→   321→  UPDATE: is_elevated_employee(id) = TRUE
   322→   322→  DELETE: DISABLED (use is_active = FALSE instead)
   323→   323→
   324→   324→employees:
   325→   325→  SELECT: company_id = get_active_company_id()
   326→   326→          AND company_id IN (SELECT company_id FROM employees 
   327→   327→                              WHERE user_id = auth.uid() 
   328→   328→                              AND status = 'active')
   329→   329→  INSERT: has_permission(company_id, 'employees.invite') = TRUE
   330→   330→  UPDATE: is_elevated_employee(company_id) = TRUE
   331→   331→          OR (auth.uid() = user_id AND -- employee can update own limited fields
   332→   332→              -- Only allow updating: avatar, phone on own profile)
   333→   333→  DELETE: DISABLED (use status = 'terminated' instead)
   334→   334→
   335→   335→roles:
   336→   336→  SELECT: company_id = get_active_company_id()
   337→   337→  INSERT: has_permission(company_id, 'settings.roles.manage') = TRUE
   338→   338→  UPDATE: has_permission(company_id, 'settings.roles.manage') = TRUE
   339→   339→          AND is_system_role = FALSE
   340→   340→  DELETE: has_permission(company_id, 'settings.roles.manage') = TRUE
   341→   341→          AND is_system_role = FALSE
   342→   342→
   343→   343→role_permissions:
   344→   344→  SELECT: company_id = get_active_company_id()
   345→   345→  INSERT: is_elevated_employee(company_id) = TRUE

   505→   505→  1. supabase.auth.signUp({ email, password, options: { data: { full_name } } })
   506→   506→  2. Supabase trigger auto-creates profiles + user_settings rows
   507→   507→  3. Return success → redirect to /onboarding
   508→   508→  4. Insert audit_log: action "auth.registered"

   512→   512→─────────────────────────────
   513→   513→Form fields: email, password
   514→   514→Server Action:
   515→   515→  1. supabase.auth.signInWithPassword({ email, password })
   516→   516→  2. Check if profiles.is_onboarded = TRUE
   517→   517→  3. Check if user has pending invitations for their email
   518→   518→  4. If is_onboarded = FALSE → redirect to /onboarding (pass pending invites)
   519→   519→  5. If is_onboarded = TRUE → check user_settings.active_company_id
   520→   520→  6. If active_company_id is NULL → redirect to /onboarding
   521→   521→  7. Else → redirect to /dashboard
   522→   522→  8. Insert audit_log: action "auth.login"

   534→   534→FLOW 4: FORGOT PASSWORD (OTP Code Flow)

   536→   536→This is a 3-step flow:

   538→   538→STEP 1 — /forgot-password

   543→   543→    })
   544→   544→  UI: Show "Check your email. We sent a 6-digit code." message.
   545→   545→  Note: Supabase sends an email with a magic link that contains a token.
   546→   546→
   547→   547→STEP 2 — /reset-password (user arrives via email link)
   548→   548→  The email link brings user to /reset-password?token_hash=xxx&type=recovery
   549→   549→  Server Action in route:
   550→   550→    1. Exchange token hash for session:
   551→   551→       supabase.auth.verifyOtp({ token_hash, type: 'recovery' })
   552→   552→    2. On success: show "Create New Password" form
   553→   553→
   554→   554→STEP 3 — New Password Form
   555→   555→  Form fields: new_password, confirm_password
   556→   556→  Zod validation: min 8 chars, passwords must match
   557→   557→  Server Action:
   558→   558→    1. supabase.auth.updateUser({ password: new_password })
   559→   559→    2. Sign out all other sessions for security
   560→   560→    3. Redirect to /login with success toast
   561→   561→    4. Insert audit_log: action "auth.password_reset"

   570→   570→  4. Protect all /dashboard/* and /employees/* and /roles/* routes
