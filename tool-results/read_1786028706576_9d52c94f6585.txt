   191→/// Company-scoped roles. System roles (Owner/Founder/Co-Founder/Investor) are
   192→/// elevated and bypass all permission checks. Custom roles use role_permissions.
   193→model Role {
   194→  id            String   @id @default(cuid())
   195→  companyId     String
   196→  company       Company  @relation(fields: [companyId], references: [id], onDelete: Cascade)
   197→  name          String
   198→  description   String?
   199→  roleTier      String   @default("standard") // elevated | standard
   200→  isSystemRole  Boolean  @default(false)
   201→  systemRoleKey String?  // owner | founder | co_founder | investor | null
   202→  // NOTE: systemRoleKey is unique PER COMPANY (not globally), so every
   203→  // company can seed its own owner/founder/co_founder/investor roles.
   204→  isActive      Boolean  @default(true)
   205→  createdById   String?
   206→  createdBy     Profile? @relation("RoleCreator", fields: [createdById], references: [id])
   207→  createdAt     DateTime @default(now())
   208→  updatedAt     DateTime @updatedAt
   209→
   210→  employees       Employee[]
   211→  rolePermissions RolePermission[]
   212→  invitations     Invitation[]
   213→
   214→  @@unique([companyId, name])
   215→  @@unique([companyId, systemRoleKey])
   216→}
