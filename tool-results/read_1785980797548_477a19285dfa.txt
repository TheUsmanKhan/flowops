     3→/**
     4→ * BookingWorkbenchView
     5→ * ---------------------
     6→ * Bulk booking workbench for external-platform orders that have NOT yet been
     7→ * booked with a courier (i.e. courierCompanyIntegrationId === null).
     8→ *
     9→ * Fetches confirmed/processing orders via:
    10→ *   GET /api/orders?statuses=confirmed,processing&limit=100
    11→ *
    12→ * Then client-side filters to orders where courierCompanyIntegrationId is null
    13→ * (the API doesn't yet expose this as a query param — adding it would require
    14→ * a /lib change which is out-of-scope for this task).
    15→ *
    16→ * Each row exposes editable inputs for customer name / phone / address,
    17→ * a CityAutocomplete (providerKey from the batch-level courier dropdown),
    18→ * COD amount, and an order type dropdown (Normal/Overland/Replacement —
    19→ * pre-computed client-side via the same logic as the PostEx adapter for
    20→ * convenience, but editable so the operator can override).
    21→ *
    22→ * The top toolbar has:
    23→ *   - Courier integration dropdown (drives the CityAutocomplete providerKey)
    24→ *   - "Select All" / "Select None"
    25→ *   - "Upload Booking" button — fires POST /api/booking-workbench/book for
    26→ *     each CHECKED row independently. Failures do NOT block other rows.
    27→ *
    28→ * Per-row result: ✅ tracking number, or ❌ error message.
    29→ */
    30→

    78→  flowopsOrderNumber: string
    79→  externalOrderReference: string | null
    80→  orderSource: string
    81→  status: string
    82→  paymentType: string
    83→  totalOrderValue: number
    84→  remainingCodAmount: number | null
    85→  courierCompanyIntegrationId: string | null
    86→  customerId: string
    87→  deliveryCity: string | null
    88→  customerName: string
    89→  customerPhone: string | null
    90→  itemCount: number

   123→// ─────────────────────────────────────────────────────────────────────────────
   124→// Types — booking workbench book endpoint
   125→// ─────────────────────────────────────────────────────────────────────────────
   126→
   127→interface BookRequest {
   128→  orderId: string
   129→  companyIntegrationId: string
   130→  customerName?: string
   131→  customerPhone?: string
   132→  deliveryAddress?: string
   133→  deliveryCity?: string
   134→  codAmount?: number
   135→  orderType?: string
   136→}

   138→interface BookSuccess {
   139→  success: true
   140→  trackingNumber: string
   141→  orderType: string
   142→  providerStatus: string | null

   145→type BookResult =
   146→  | { ok: true; trackingNumber: string; orderType: string }
   147→  | { ok: false; error: string }

   149→// ─────────────────────────────────────────────────────────────────────────────

   153→interface RowState {

   155→  customerPhone: string
   156→  deliveryAddress: string
   157→  deliveryCity: string
   158→  codAmount: string
   159→  orderType: 'Normal' | 'Overland' | 'Replacement'
   160→  checked: boolean
   161→  // Per-row booking result (cleared on edit)
   162→  result: BookResult | null
   163→  isBooking: boolean
   164→}

   166→type OrderType = 'Normal' | 'Overland' | 'Replacement'

   181→// ─────────────────────────────────────────────────────────────────────────────
   182→// Main component
   183→// ─────────────────────────────────────────────────────────────────────────────
   184→
   185→export function BookingWorkbenchView() {
   186→  const queryClient = useQueryClient()
   187→
   188→  // Batch-level state
   189→  const [companyIntegrationId, setCompanyIntegrationId] = useState<string>('')
   190→  const [search, setSearch] = useState('')
   191→
   192→  // Per-row editable state keyed by order id
   193→  const [rowStates, setRowStates] = useState<Record<string, RowState>>({})
   194→
   195→  // ── Data: orders ────────────────────────────────────────────────────────
   196→  const ordersQuery = useQuery<OrdersListResponse>({
   197→    queryKey: ['orders', 'booking-workbench'],
   198→    queryFn: () =>
   199→      api.get<OrdersListResponse>(
   200→        '/api/orders?statuses=confirmed,processing&limit=100',
   201→      ),
   202→    staleTime: 15_000,
   203→  })
   204→
   205→  // Client-side filter: only unbooked orders (no courier integration)
   206→  const unbookedOrders = useMemo(() => {
   207→    if (!ordersQuery.data?.orders) return []
   208→    return ordersQuery.data.orders.filter(
   209→      (o) => !o.courierCompanyIntegrationId,
   210→    )
   211→  }, [ordersQuery.data])
   212→
   213→  // Apply search filter (order # / customer / phone)
   214→  const filteredOrders = useMemo(() => {
   215→    if (!search.trim()) return unbookedOrders
   216→    const q = search.trim().toLowerCase()
   217→    return unbookedOrders.filter(
   218→      (o) =>
   219→        o.flowopsOrderNumber.toLowerCase().includes(q) ||
   220→        o.externalOrderReference?.toLowerCase().includes(q) ||
   221→        o.customerName.toLowerCase().includes(q) ||
   222→        o.customerPhone?.toLowerCase().includes(q),
   223→    )
   224→  }, [unbookedOrders, search])
   225→
   226→  // ── Data: courier integrations ──────────────────────────────────────────
   227→  const integrationsQuery = useQuery<IntegrationsResponse>({
   228→    queryKey: ['integrations', 'courier'],
   229→    queryFn: () => api.get<IntegrationsResponse>('/api/integrations?category=courier'),
   230→    staleTime: 30_000,
   231→  })
   232→
   233→  const courierIntegrations = useMemo(
   234→    () =>
   235→      (integrationsQuery.data?.integrations ?? []).filter(
   236→        (i) => i.isActive && i.provider?.category === 'courier',
   237→      ),
   238→    [integrationsQuery.data],
   239→  )
   240→
   241→  const selectedIntegration = useMemo(
   242→    () => courierIntegrations.find((i) => i.id === companyIntegrationId),
   243→    [courierIntegrations, companyIntegrationId],
   244→  )
   245→  const selectedProviderKey = selectedIntegration?.provider?.providerKey ?? ''
   246→
   247→  // ── Row state helpers ───────────────────────────────────────────────────
   248→  function getRowState(order: OrderRow): RowState {
   249→    if (rowStates[order.id]) return rowStates[order.id]
   250→    return {
   251→      customerName: order.customerName ?? '',
   252→      customerPhone: order.customerPhone ?? '',
   253→      deliveryAddress: '', // not returned by listOrders endpoint — user fills in
   254→      deliveryCity: order.deliveryCity ?? '',
   255→      codAmount: String(
   256→        order.remainingCodAmount ?? order.totalOrderValue ?? 0,
   257→      ),
   258→      orderType: 'Normal',
   259→      checked: false,
   260→      result: null,
   261→      isBooking: false,
   262→    }
   263→  }
   264→
   265→  function patchRow(orderId: string, patch: Partial<RowState>) {
   266→    setRowStates((prev) => ({
   267→      ...prev,
   268→      [orderId]: {
   269→        ...(prev[orderId] ?? getRowStateLookup(orderId)),
   270→        ...patch,
   271→        // Clear any previous result when the user edits a field
   272→        result:
   273→          patch.result !== undefined ? patch.result : null,
   274→      },
   275→    }))
   276→  }
   277→
   278→  // Lookup that doesn't depend on filteredOrders (for patchRow safety)
   279→  function getRowStateLookup(orderId: string): RowState {
   280→    if (rowStates[orderId]) return rowStates[orderId]
   281→    const order = ordersQuery.data?.orders?.find((o) => o.id === orderId)
   282→    if (order) {
   283→      return {
   284→        customerName: order.customerName ?? '',
   285→        customerPhone: order.customerPhone ?? '',
   286→        deliveryAddress: '',
   287→        deliveryCity: order.deliveryCity ?? '',
   288→        codAmount: String(
   289→          order.remainingCodAmount ?? order.totalOrderValue ?? 0,
   290→        ),
   291→        orderType: 'Normal' as const,
   292→        checked: false,
   293→        result: null,
   294→        isBooking: false,
   295→      }
   296→    }
   297→    return {
   298→      customerName: '',
   299→      customerPhone: '',
   300→      deliveryAddress: '',
   301→      deliveryCity: '',
   302→      codAmount: '0',
   303→      orderType: 'Normal' as const,
   304→      checked: false,
   305→      result: null,
   306→      isBooking: false,
   307→    }
   308→  }
   309→
   310→  // ── Select-all helpers ──────────────────────────────────────────────────
   311→  const checkedCount = filteredOrders.filter(
   312→    (o) => getRowState(o).checked,
   313→  ).length
   314→  const allChecked =
   315→    filteredOrders.length > 0 && checkedCount === filteredOrders.length
   316→
   317→  function toggleSelectAll() {
   318→    const next: Record<string, RowState> = { ...rowStates }
   319→    const target = !allChecked
   320→    for (const o of filteredOrders) {
   321→      const cur = next[o.id] ?? getRowState(o)
   322→      next[o.id] = { ...cur, checked: target }
   323→    }
   324→    setRowStates(next)
   325→  }
   326→
   327→  // ── Booking mutation (per-row) ──────────────────────────────────────────
   328→  const bookMutation = useMutation({
   329→    mutationFn: async (input: {
   330→      orderId: string
   331→      body: BookRequest
   332→    }): Promise<BookSuccess> => {
   333→      return api.post<BookSuccess>('/api/booking-workbench/book', input.body)
   334→    },
   335→    // We handle success/failure in the batchSubmit loop below, not here.
   336→  })
   337→
   338→  // ── Batch submit ────────────────────────────────────────────────────────
   339→  async function handleUploadBooking() {
   340→    if (!companyIntegrationId) {
   341→      toast.error('Select a courier integration first.')
   342→      return
   343→    }
   344→    const checkedRows = filteredOrders.filter((o) => {
   345→      const s = getRowState(o)
   346→      return s.checked && !s.isBooking
   347→    })
   348→    if (checkedRows.length === 0) {
   349→      toast.error('No rows selected. Tick the checkbox on at least one order.')
   350→      return
   351→    }
   352→
   353→    toast.info(`Booking ${checkedRows.length} order(s) with ${selectedIntegration?.provider?.providerName ?? 'courier'}…`)
   354→
   355→    let successCount = 0
   356→    let failureCount = 0
   357→
   358→    // Fire requests sequentially — booking calls external courier APIs which
   359→    // typically don't tolerate parallel hits well, and per-row errors must
   360→    // not block other rows.
   361→    for (const order of checkedRows) {
   362→      const state = getRowState(order)
   363→      const body: BookRequest = {
   364→        orderId: order.id,
   365→        companyIntegrationId,
   366→        customerName: state.customerName.trim() || undefined,
   367→        customerPhone: state.customerPhone.trim() || undefined,
   368→        deliveryAddress: state.deliveryAddress.trim() || undefined,
   369→        deliveryCity: state.deliveryCity.trim() || undefined,
   370→        codAmount: Number(state.codAmount) || 0,
   371→        orderType: state.orderType,
   372→      }
   373→
   374→      // Mark as booking in progress
   375→      patchRow(order.id, { isBooking: true, result: null })
   376→
   377→      try {
   378→        const result = await bookMutation.mutateAsync({
   379→          orderId: order.id,
   380→          body,
   381→        })
   382→        successCount++
   383→        patchRow(order.id, {
   384→          isBooking: false,
   385→          checked: false,
   386→          result: {
   387→            ok: true,
   388→            trackingNumber: result.trackingNumber,
   389→            orderType: result.orderType,
   390→          },
   391→        })
   392→      } catch (err) {
   393→        failureCount++
   394→        const message =
   395→          err instanceof FetchError
   396→            ? err.message
   397→            : err instanceof Error
   398→              ? err.message
   399→              : 'Booking failed'
   400→        patchRow(order.id, {
   401→          isBooking: false,
   402→          result: { ok: false, error: message },
   403→        })
   404→      }
   405→    }
   406→
   407→    // Final toast + invalidate orders list so successfully booked rows
   408→    // disappear (they now have courierCompanyIntegrationId set).
   409→    if (successCount > 0 && failureCount === 0) {
   410→      toast.success(
   411→        `All ${successCount} order(s) booked successfully.`,
   412→      )
   413→    } else if (successCount > 0 && failureCount > 0) {
   414→      toast.warning(
   415→        `${successCount} booked, ${failureCount} failed. See per-row status.`,
   416→      )
   417→    } else {
   418→      toast.error(`All ${failureCount} booking(s) failed. See per-row errors.`)
   419→    }
   420→
   421→    if (successCount > 0) {
   422→      queryClient.invalidateQueries({ queryKey: ['orders', 'booking-workbench'] })
   423→      queryClient.invalidateQueries({ queryKey: ['orders'] })
   424→    }
   425→  }
   426→
   427→  // ── Render: loading / error / empty ─────────────────────────────────────
   428→  if (ordersQuery.isLoading) {
   429→    return (
   430→      <div className="space-y-4">
   431→        <PageHeader
   432→          title="Booking Workbench"
   433→          description="Bulk-book external-platform orders with a courier."
   434→        />
   435→        <Card>
   436→          <CardContent className="p-10 text-center">
   437→            <Loader2 className="h-6 w-6 animate-spin mx-auto text-muted-foreground" />
   438→            <p className="text-sm text-muted-foreground mt-2">
   439→              Loading unbooked orders…
   440→            </p>
   441→          </CardContent>
   442→        </Card>
   443→      </div>
   444→    )
   445→  }
   446→
   447→  if (ordersQuery.isError) {
   448→    return (
   449→      <div className="space-y-4">
   450→        <PageHeader
   451→          title="Booking Workbench"
   452→          description="Bulk-book external-platform orders with a courier."
   453→        />
   454→        <Card className="border-rose-200">
   455→          <CardContent className="p-10 text-center space-y-3">
   456→            <AlertCircle className="h-6 w-6 mx-auto text-rose-600" />
   457→            <p className="text-sm text-rose-700">
   458→              {getErrorMessage(ordersQuery.error)}
   459→            </p>
   460→            <Button
   461→              variant="outline"
   462→              size="sm"
   463→              onClick={() => ordersQuery.refetch()}
   464→            >
   465→              <RefreshCw className="h-3.5 w-3.5" /> Retry
   466→            </Button>
   467→          </CardContent>
   468→        </Card>
   469→      </div>
   470→    )
   471→  }
   472→
   473→  return (
   474→    <div className="space-y-4">
   475→      <PageHeader
   476→        title="Booking Workbench"
   477→        description="Bulk-book confirmed external-platform orders that haven't yet been booked with a courier."
   478→        actions={
   479→          <Button
   480→            variant="outline"
   481→            size="sm"
   482→            onClick={() => ordersQuery.refetch()}
   483→            disabled={ordersQuery.isFetching}
   484→          >
   485→            <RefreshCw
   486→              className={cn(
   487→                'h-3.5 w-3.5',
   488→                ordersQuery.isFetching && 'animate-spin',
   489→              )}
   490→            />
   491→            Refresh
   492→          </Button>
   493→        }
   494→      />
   495→
   496→      {/* Batch toolbar */}
   497→      <Card>
   498→        <CardContent className="p-4 space-y-3">
   499→          <div className="grid gap-3 md:grid-cols-[1fr_1fr_auto] md:items-end">
   500→            {/* Courier integration */}
   501→            <div className="space-y-1.5">
   502→              <Label className="text-xs flex items-center gap-1">
   503→                <Truck className="h-3 w-3" /> Courier Integration
   504→              </Label>
   505→              <Select
   506→                value={companyIntegrationId}
   507→                onValueChange={(v) => {
   508→                  setCompanyIntegrationId(v)
   509→                  // City provider changes — clear per-row city overrides so
   510→                  // they re-query against the new courier's operational cities.
   511→                  const next: Record<string, RowState> = {}
   512→                  for (const [id, s] of Object.entries(rowStates)) {
   513→                    next[id] = { ...s, deliveryCity: '', result: null }
   514→                  }
   515→                  setRowStates(next)
   516→                }}
   517→                disabled={integrationsQuery.isLoading}
   518→              >
   519→                <SelectTrigger>
   520→                  <SelectValue
   521→                    placeholder={
   522→                      integrationsQuery.isLoading
   523→                        ? 'Loading…'
   524→                        : courierIntegrations.length === 0
   525→                          ? 'No courier integrations'
   526→                          : 'Select courier'
   527→                    }
   528→                  />
   529→                </SelectTrigger>
   530→                <SelectContent>
   531→                  {courierIntegrations.map((i) => (
   532→                    <SelectItem key={i.id} value={i.id}>
   533→                      <span className="flex items-center gap-2">
   534→                        <span className="font-medium">
   535→                          {i.provider.providerName}
   536→                        </span>
   537→                        <span className="text-xs text-muted-foreground">
   538→                          ({i.connectionName})
   539→                        </span>
   540→                      </span>
   541→                    </SelectItem>
   542→                  ))}
   543→                </SelectContent>
   544→              </Select>
   545→            </div>
   546→
   547→            {/* Search */}
   548→            <div className="space-y-1.5">
   549→              <Label className="text-xs flex items-center gap-1">
   550→                <Search className="h-3 w-3" /> Search
   551→              </Label>
   552→              <Input
   553→                placeholder="Order #, customer, phone…"
   554→                value={search}
   555→                onChange={(e) => setSearch(e.target.value)}
   556→              />
   557→            </div>
   558→
   559→            {/* Upload button */}
   560→            <Button
   561→              onClick={handleUploadBooking}
   562→              disabled={
   563→                !companyIntegrationId ||
   564→                checkedCount === 0 ||
   565→                bookMutation.isPending
   566→              }
   567→              className="md:w-auto"
   568→            >
   569→              {bookMutation.isPending ? (
   570→                <>
   571→                  <Loader2 className="h-4 w-4 animate-spin" /> Booking…
   572→                </>
   573→              ) : (
   574→                <>
   575→                  <Upload className="h-4 w-4" /> Upload Booking ({checkedCount})
   576→                </>
   577→              )}
   578→            </Button>
   579→          </div>
   580→
   581→          {/* Helper text */}
   582→          {courierIntegrations.length === 0 && !integrationsQuery.isLoading && (
   583→            <p className="text-xs text-amber-700 flex items-center gap-1">
   584→              <AlertCircle className="h-3 w-3" /> No courier integrations
   585→              connected. Connect one in Settings → Integrations before booking.
   586→            </p>
   587→          )}
   588→          {companyIntegrationId && (
   589→            <p className="text-xs text-muted-foreground">
   590→              Using{' '}
   591→              <strong className="text-foreground">
   592→                {selectedIntegration?.provider?.providerName}
   593→              </strong>{' '}
   594→              — city dropdown is filtered to this courier's operational cities.
   595→            </p>
   596→          )}
   597→        </CardContent>
   598→      </Card>
   599→
   600→      {/* Orders table */}
   601→      <Card>
   602→        <CardContent className="p-0">
   603→          {filteredOrders.length === 0 ? (
   604→            <div className="p-10 text-center space-y-2">
   605→              <Inbox className="h-8 w-8 mx-auto text-muted-foreground" />
   606→              <p className="text-sm font-medium">
   607→                {unbookedOrders.length === 0
   608→                  ? 'All caught up!'
   609→                  : 'No matching orders'}
   610→              </p>
   611→              <p className="text-xs text-muted-foreground">
   612→                {unbookedOrders.length === 0
   613→                  ? 'No confirmed/processing orders are awaiting courier booking.'
   614→                  : 'Try a different search term.'}
   615→              </p>
   616→            </div>
   617→          ) : (
   618→            <div className="overflow-x-auto">
   619→              <Table>
   620→                <TableHeader>
   621→                  <TableRow>
   622→                    <TableHead className="w-10">
   623→                      <Checkbox
   624→                        checked={allChecked}
   625→                        onCheckedChange={toggleSelectAll}
