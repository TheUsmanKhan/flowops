   204→   204→/// Company-scoped roles. System roles (Owner/Founder/Co-Founder/Investor) are
   205→   205→/// elevated and bypass all permission checks. Custom roles use role_permissions.
   206→   206→model Role {
   207→   207→  id            String   @id @default(cuid())
   208→   208→  companyId     String
   209→   209→  company       Company  @relation(fields: [companyId], references: [id], onDelete: Cascade)
   210→   210→  name          String
   211→   211→  description   String?
   212→   212→  roleTier      String   @default("standard") // elevated | standard
   213→   213→  isSystemRole  Boolean  @default(false)
   214→   214→  systemRoleKey String?  // owner | founder | co_founder | investor | null
   215→   215→  // NOTE: systemRoleKey is unique PER COMPANY (not globally), so every
   216→   216→  // company can seed its own owner/founder/co_founder/investor roles.
   217→   217→  isActive      Boolean  @default(true)
   218→   218→  createdById   String?
   219→   219→  createdBy     Profile? @relation("RoleCreator", fields: [createdById], references: [id])
   220→   220→  createdAt     DateTime @default(now())
   221→   221→  updatedAt     DateTime @updatedAt
   222→   222→
   223→   223→  employees       Employee[]
   224→   224→  rolePermissions RolePermission[]
   225→   225→  invitations     Invitation[]
   226→   226→
   227→   227→  @@unique([companyId, name])
   228→   228→  @@unique([companyId, systemRoleKey])
   229→   229→}
