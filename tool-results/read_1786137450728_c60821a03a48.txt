    51→    51→-- ADAPTATIONS FROM SPEC (necessary to integrate with the live schema)
    52→    52→-- ============================================================================
    53→    53→-- The spec was written in idealized PostgreSQL terms (lowercase table names,
    54→    54→-- UUID primary keys, snake_case columns, `organizations(id)` references).
    55→    55→-- The LIVE FlowOps schema uses the opposite conventions because it was built
    56→    56→-- with Prisma on an existing multi-tenant SaaS. To produce SQL that actually
    57→    57→-- runs on this live instance WITHOUT breaking the 92 existing orders, the
    58→    58→-- following adaptations are made (each is the correct production call):
    59→    59→--
    60→    60→--   1. Table name "Customer" (PascalCase) instead of `customers`.
    61→    61→--      The table already exists with 5 rows + 92 FK references from "Order".
    62→    62→--      Creating a parallel `customers` table would orphan every order. We
    63→    63→--      ALTER the existing table in place. The three NEW child tables use
    64→    64→--      lowercase names (customer_phones, customer_addresses,
    65→    65→--      customer_external_identities) exactly as the spec names them.
    66→    66→--
    67→    67→--   2. All IDs are TEXT (cuid) instead of UUID.
    68→    68→--      Every existing PK/FK in the schema (Organization, Employee, Customer,
    69→    69→--      Order, OrderItem, ...) is TEXT. A UUID customer_phones.customerId FK
    70→    70→--      could not reference the TEXT "Customer".id. New child tables use
    71→    71→--      TEXT PKs with `DEFAULT gen_random_uuid()::text` so raw-SQL inserts
    72→    72→--      get a DB-generated ID while Prisma client inserts override with a
    73→    73→--      cuid (same pattern the rest of the app uses).
    74→    74→--
    75→    75→--   3. All columns are camelCase (Prisma convention) instead of snake_case,
    76→    76→--      AND are double-quoted everywhere so PostgreSQL preserves the casing
    77→    77→--      (unquoted identifiers are folded to lowercase). camelCase columns on
    78→    78→--      the new tables: "phoneRaw", "phoneNormalized", "isPrimary",
    79→    79→--      "isDefault", "lastUsedAt", "externalCustomerId", "matchedVia",
    80→    80→--      "organizationId", "customerId", "createdAt", "updatedAt". This
    81→    81→--      matches the existing "Customer"/"Order" tables which Prisma created
    82→    82→--      with quoted camelCase columns.
    83→    83→--
    84→    84→--   4. FK references use PascalCase: REFERENCES "Organization"(id),
    85→    85→--      "Employee"(id), "Customer"(id).
    86→    86→--
    87→    87→--   5. RLS uses ENABLE (not FORCE ROW LEVEL SECURITY). The application
    88→    88→--      connects via the `postgres` role, which BYPASSES RLS by default.
    89→    89→--      This means:
    90→    90→--        - The Prisma app keeps working exactly as before (no GUC-setting
    91→    91→--          middleware required for the app to function).
    92→    92→--        - Direct Supabase API access (anon/authenticated roles) IS subject
    93→    93→--          to these RLS policies -> defense-in-depth for any future direct
    94→    94→--          client access, exactly as the spec intends.
    95→    95→--      When Step 2 wires GUC-setting middleware (app.active_org_id etc.),
    96→    96→--      the policies become actively enforced for non-bypass roles.
    97→    97→--
    98→    98→--   6. normalize_phone() and match_or_create_customer() accept/return TEXT
    99→    99→--      (not UUID) for the same ID-compatibility reason.
   100→   100→--
   101→   101→-- All other semantics — table structure, constraints, indexes, the layered
   102→   102→-- matching strategy, RLS policy logic — follow the spec exactly.

   107→   107→-- ============================================================================
   108→   108→-- PART 0: normalize_phone() FUNCTION
   109→   109→-- ============================================================================
   110→   110→-- Strips all non-digit characters, then canonicalizes Pakistani phone numbers
   111→   111→-- to E.164-style "+92XXXXXXXXXX". Used both at storage time (so every
   112→   112→-- customer_phones."phoneNormalized" is canonical) and at lookup time (so
   113→   113→-- matching is format-independent).
   114→   114→--
   115→   115→-- Examples:
   116→   116→--   normalize_phone('0300-1234567')    -> '+923001234567'
   117→   117→--   normalize_phone('+92 300 1234567') -> '+923001234567'
   118→   118→--   normalize_phone('923001234567')    -> '+923001234567'
   119→   119→--   normalize_phone('3001234567')      -> '+923001234567'  (10-digit local)
   120→   120→--   normalize_phone(NULL)              -> NULL
   121→   121→--   normalize_phone('')                -> NULL
   122→   122→--
   123→   123→-- Marked IMMUTABLE so it can be used in indexes and CHECK constraints.
   124→   124→
   125→   125→CREATE OR REPLACE FUNCTION normalize_phone(p_raw_phone TEXT)
   126→   126→RETURNS TEXT
   127→   127→LANGUAGE plpgsql
   128→   128→IMMUTABLE
   129→   129→AS $$
   130→   130→DECLARE
   131→   131→  v_digits TEXT;
   132→   132→  v_len    INT;
   133→   133→BEGIN
   134→   134→  IF p_raw_phone IS NULL OR BTRIM(p_raw_phone) = '' THEN
   135→   135→    RETURN NULL;
   136→   136→  END IF;
   137→   137→
   138→   138→  -- Keep digits only
   139→   139→  v_digits := REGEXP_REPLACE(p_raw_phone, '[^0-9]', '', 'g');
   140→   140→  v_len    := LENGTH(v_digits);
   141→   141→
   142→   142→  IF v_len = 0 THEN
   143→   143→    RETURN NULL;
   144→   144→  END IF;
   145→   145→
   146→   146→  -- 12 digits starting with '92' -> already has country code, just prefix '+'
   147→   147→  IF v_len = 12 AND v_digits LIKE '92%' THEN
   148→   148→    RETURN '+' || v_digits;
   149→   149→  END IF;
   150→   150→
   151→   151→  -- 11 digits starting with '0' -> local Pakistani format (03XXXXXXXXX)
   152→   152→  IF v_len = 11 AND v_digits LIKE '0%' THEN
   153→   153→    RETURN '+92' || SUBSTRING(v_digits FROM 2);
   154→   154→  END IF;
   155→   155→
   156→   156→  -- 10 digits without leading 0 (e.g. 3001234567) -> treat as local, prefix +92
   157→   157→  IF v_len = 10 THEN
   158→   158→    RETURN '+92' || v_digits;
   159→   159→  END IF;
   160→   160→
   161→   161→  -- Fallback for unusual lengths (10-13 digits): strip a leading 0 if present
   162→   162→  -- and prefix +92. Best-effort consistency for storage/matching.
   163→   163→  IF v_len BETWEEN 10 AND 13 THEN
   164→   164→    IF v_digits LIKE '0%' THEN
   165→   165→      v_digits := SUBSTRING(v_digits FROM 2);
   166→   166→    END IF;
   167→   167→    RETURN '+92' || v_digits;
   168→   168→  END IF;
   169→   169→
   170→   170→  -- Anything outside 10-13 digits: return +digits as-is (rare, best-effort)
   171→   171→  RETURN '+' || v_digits;
   172→   172→END;
   173→   173→$$;

   264→   264→--
   265→   265→-- "matchedVia" records HOW each mapping was established (for audit/debugging):
   266→   266→--   exact_identity — first time this external ID was seen (Layer 4 of matching)
   267→   267→--   phone_match    — matched an existing customer by normalized phone (Layer 2)
   268→   268→--   email_match    — matched an existing customer by email (Layer 3)
   269→   269→--   manual         — an operator explicitly linked the accounts
   270→   270→
   271→   271→CREATE TABLE IF NOT EXISTS customer_external_identities (
   272→   272→  id                    TEXT PRIMARY KEY DEFAULT gen_random_uuid()::text,

   274→   274→  "organizationId"      TEXT NOT NULL REFERENCES "Organization"(id),

   276→   276→  platform              TEXT NOT NULL
   277→   277→                          CHECK (platform IN ('shopify','daraz','instagram')),
   278→   278→
   279→   279→  "externalCustomerId"  TEXT NOT NULL,
   280→   280→  -- the platform's own customer identifier
   281→   281→
   282→   282→  "matchedVia"          TEXT NOT NULL DEFAULT 'manual'
   283→   283→                          CHECK ("matchedVia" IN
   284→   284→                            ('exact_identity','phone_match','email_match','manual')),
   285→   285→
   286→   286→  "createdAt"           TIMESTAMPTZ NOT NULL DEFAULT NOW(),

   294→   294→

   296→   296→  ON customer_external_identities ("organizationId", platform, "externalCustomerId");
   297→   297→
   298→   298→-- ============================================================================
   299→   299→-- PART 4: ADD NEW COLUMNS TO "Customer" + DB-SIDE ID DEFAULT
   300→   300→-- ============================================================================
   301→   301→-- flaggedAt / flaggedBy: track WHEN and WHO flagged a customer for RTO fraud.
   302→   302→-- createdBy: audit trail for manual customer creation.
   303→   303→-- (The spec's cached stats — totalOrdersCount, totalOrderValue, totalRtoCount,
   304→   304→--  isFlagged, flaggedReason — already exist on the live table and are kept.)
   305→   305→--
   306→   306→-- ID DEFAULT: Prisma's @default(cuid()) generates IDs CLIENT-SIDE, so the
   307→   307→-- "Customer".id column has NO DB-side default. The match_or_create_customer()
   308→   308→-- SQL function (Part 10) inserts into "Customer" directly (no Prisma), so it
   309→   309→-- needs a DB-side default to generate an id. We set DEFAULT gen_random_uuid()::text
   310→   310→-- — Prisma will continue to override this with cuids on its own inserts, so
   311→   311→-- existing behavior is unchanged; only raw-SQL inserts (from the function)
   312→   312→-- use the DB default. The resulting id is a TEXT value either way (cuid or
   313→   313→-- uuid-as-text), which is what every FK in the schema expects.
   314→   314→
   315→   315→ALTER TABLE "Customer"
   316→   316→  ADD COLUMN IF NOT EXISTS "flaggedAt" TIMESTAMPTZ;
   317→   317→ALTER TABLE "Customer"
   318→   318→  ADD COLUMN IF NOT EXISTS "flaggedBy" TEXT REFERENCES "Employee"(id) ON DELETE SET NULL;
   319→   319→ALTER TABLE "Customer"
   320→   320→  ADD COLUMN IF NOT EXISTS "createdBy" TEXT REFERENCES "Employee"(id) ON DELETE SET NULL;
   321→   321→
   322→   322→ALTER TABLE "Customer"
   323→   323→  ALTER COLUMN id SET DEFAULT gen_random_uuid()::text;
   324→   324→
   325→   325→-- "updatedAt" is NOT NULL but Prisma's @updatedAt only sets it CLIENT-SIDE,
   326→   326→-- so the DB column has no default. Raw SQL inserts (from the function) need
   327→   327→-- a DB-side default so the NOT NULL constraint is satisfied on INSERT. Prisma
   328→   328→-- will continue to set updatedAt explicitly on its own writes, so this is safe.
   329→   329→ALTER TABLE "Customer"
   330→   330→  ALTER COLUMN "updatedAt" SET DEFAULT NOW();
   331→   331→
   332→   332→-- Index for the "flagged customers" list view (per organization).
   333→   333→CREATE INDEX IF NOT EXISTS "Customer_organizationId_isFlagged_idx"

   336→   336→-- ============================================================================
   337→   337→-- PART 5: BACKFILL customer_phones FROM LEGACY Customer.phone (IF PRESENT)
   338→   338→-- ============================================================================
   339→   339→-- Migrate every customer's primary phone + alternate phone into customer_phones
   340→   340→-- BEFORE the legacy columns are dropped. The primary phone gets
   341→   341→-- "isPrimary"=TRUE (the partial unique index guarantees one primary per
   342→   342→-- customer). Alternate phones get "isPrimary"=FALSE.
   343→   343→--
   344→   344→-- Wrapped in DO blocks that first check information_schema.columns so the
   345→   345→-- migration is SAFE TO RE-RUN from any state — if the legacy columns were
   346→   346→-- already dropped by a prior partial run, the backfill is a no-op rather than
   347→   347→-- a hard error.
   348→   348→--
   349→   349→-- ON CONFLICT ("organizationId", "phoneNormalized") DO NOTHING handles the
   350→   350→-- rare case where a customer's alternatePhone collides with another
   351→   351→-- customer's already-migrated primary phone — the primary wins, the duplicate
   352→   352→-- alternate is silently skipped (it is, by definition, the same contact).
   353→   353→
   354→   354→DO $$
   355→   355→BEGIN
   356→   356→  IF EXISTS (
   357→   357→    SELECT 1 FROM information_schema.columns
   358→   358→    WHERE table_name = 'Customer' AND column_name = 'phone'
   359→   359→  ) THEN
   360→   360→    INSERT INTO customer_phones
   361→   361→      ("customerId", "organizationId", "phoneRaw", "phoneNormalized", "isPrimary", label, "createdAt")
   362→   362→    SELECT
   363→   363→      c.id,
   364→   364→      c."organizationId",
   365→   365→      c.phone,
   366→   366→      normalize_phone(c.phone),
   367→   367→      TRUE,
   368→   368→      'Primary',
   369→   369→      c."createdAt"
   370→   370→    FROM "Customer" c
   371→   371→    WHERE c.phone IS NOT NULL AND BTRIM(c.phone) <> ''
   372→   372→    ON CONFLICT ("organizationId", "phoneNormalized") DO NOTHING;
   373→   373→  END IF;
   374→   374→END $$;
   375→   375→
   376→   376→DO $$
   377→   377→BEGIN
   378→   378→  IF EXISTS (
   379→   379→    SELECT 1 FROM information_schema.columns
   380→   380→    WHERE table_name = 'Customer' AND column_name = 'alternatePhone'
   381→   381→  )
   382→   382→  AND EXISTS (
   383→   383→    SELECT 1 FROM information_schema.columns
   384→   384→    WHERE table_name = 'Customer' AND column_name = 'phone'
   385→   385→  ) THEN
   386→   386→    INSERT INTO customer_phones
   387→   387→      ("customerId", "organizationId", "phoneRaw", "phoneNormalized", "isPrimary", label, "createdAt")
   388→   388→    SELECT
   389→   389→      c.id,
   390→   390→      c."organizationId",
   391→   391→      c."alternatePhone",
   392→   392→      normalize_phone(c."alternatePhone"),
   393→   393→      FALSE,
   394→   394→      'Alternate',
   395→   395→      c."createdAt"
   396→   396→    FROM "Customer" c
   397→   397→    WHERE c."alternatePhone" IS NOT NULL
   398→   398→      AND BTRIM(c."alternatePhone") <> ''
   399→   399→      AND normalize_phone(c."alternatePhone") IS DISTINCT FROM normalize_phone(c.phone)
   400→   400→    ON CONFLICT ("organizationId", "phoneNormalized") DO NOTHING;
   401→   401→  END IF;
   402→   402→END $$;
   403→   403→
   404→   404→-- ============================================================================
   405→   405→-- PART 6: BACKFILL customer_addresses FROM LEGACY JSON (IF PRESENT)
   406→   406→-- ============================================================================
   407→   407→-- Parse the flat JSONB { "address": "...", "city": "..." } that was stored in
   408→   408→-- Customer.shippingAddress and Customer.billingAddress, and migrate each into
   409→   409→-- a customer_addresses row.
   410→   410→--
   411→   411→--  - Shipping address -> "isDefault" = TRUE (it is the primary delivery address)
   412→   412→--  - Billing address  -> "isDefault" = FALSE, and ONLY if it differs from the
   413→   413→--    shipping address (no point storing an exact duplicate)
   414→   414→--
   415→   415→-- "lastUsedAt" is set to the customer's most recent order's createdAt, so the
   416→   416→-- UI can sort addresses by recency immediately after migration.
   417→   417→--
   418→   418→-- Wrapped in DO blocks (same re-run-safety rationale as Part 5).
   419→   419→
   420→   420→DO $$
   421→   421→BEGIN
   422→   422→  IF EXISTS (
   423→   423→    SELECT 1 FROM information_schema.columns
   424→   424→    WHERE table_name = 'Customer' AND column_name = 'shippingAddress'
   425→   425→  ) THEN
   426→   426→    INSERT INTO customer_addresses
   427→   427→      ("customerId", "organizationId", label, address, city, "isDefault", "lastUsedAt", "createdAt", "updatedAt")
   428→   428→    SELECT
   429→   429→      c.id,
   430→   430→      c."organizationId",
   431→   431→      'Shipping',

   540→   540→-- ============================================================================
   541→   541→-- PART 10: match_or_create_customer() FUNCTION
   542→   542→-- ============================================================================
   543→   543→-- Implements the layered cross-platform customer identity matching strategy.
   544→   544→-- Called by Step 2's server actions and, in a future Shopify sprint, directly
   545→   545→-- from a webhook handler.
   546→   546→--
   547→   547→-- Layered strategy:
   548→   548→--   1. EXACT IDENTITY — look up customer_external_identities for an existing
   549→   549→--      ("organizationId", platform, "externalCustomerId") mapping. Fastest
   550→   550→--      and most reliable. Return immediately if found.
   551→   551→--   2. PHONE MATCH — if p_phone is provided, normalize it and look up
   552→   552→--      customer_phones within this org. If found, establish a NEW external
   553→   553→--      identity mapping ("matchedVia"='phone_match') so future orders from
   554→   554→--      the same platform hit Layer 1, and return that customer_id.
   555→   555→--   3. EMAIL MATCH — if no phone match and p_email is provided, look up
   556→   556→--      Customer.email within this org. If found, establish an external
   557→   557→--      identity mapping ("matchedVia"='email_match') and return.
   558→   558→--   4. CREATE — if nothing matches, create a new Customer row, a primary
   559→   559→--      customer_phones row (if p_phone provided), and an external identity
   560→   560→--      mapping ("matchedVia"='exact_identity'). Return the new customer_id.
   561→   561→--
   562→   562→-- Concurrency: Layers 1-3 are read-only lookups. Before Layer 4 creates
   563→   563→-- anything, we acquire a transaction-scoped advisory lock keyed on
   564→   564→-- ("organizationId", platform, "externalCustomerId") and RE-RUN Layer 1.
   565→   565→-- This guarantees that two simultaneous webhook calls for the same external
   566→   566→-- ID cannot create duplicate customers.
   567→   567→--
   568→   568→-- SECURITY DEFINER + SET search_path=public: the function must insert into
   569→   569→-- customer_phones and customer_external_identities even when the caller
   570→   570→-- lacks INSERT permission. search_path is pinned to prevent search_path
   571→   571→-- injection (standard SECURITY DEFINER hardening).
   572→   572→
   573→   573→CREATE OR REPLACE FUNCTION match_or_create_customer(
   574→   574→  p_organization_id      TEXT,
   575→   575→  p_platform             TEXT,
   576→   576→  p_external_customer_id TEXT,
   577→   577→  p_phone                TEXT DEFAULT NULL,
   578→   578→  p_email                TEXT DEFAULT NULL,
   579→   579→  p_name                 TEXT DEFAULT NULL
   580→   580→) RETURNS TEXT
   581→   581→LANGUAGE plpgsql
   582→   582→SECURITY DEFINER
   583→   583→SET search_path = public
   584→   584→AS $$
   585→   585→DECLARE
   586→   586→  v_customer_id      TEXT;
   587→   587→  v_normalized_phone TEXT;
   588→   588→  v_lock_key         BIGINT;
   589→   589→BEGIN
   590→   590→  -- ------------------------------------------------------------------
   591→   591→  -- Layer 1: exact external identity match (fast path)
   592→   592→  -- ------------------------------------------------------------------
   593→   593→  SELECT cei."customerId" INTO v_customer_id
   594→   594→  FROM customer_external_identities cei
   595→   595→  WHERE cei."organizationId" = p_organization_id
   596→   596→    AND cei.platform = p_platform
   597→   597→    AND cei."externalCustomerId" = p_external_customer_id
   598→   598→  LIMIT 1;
   599→   599→
   600→   600→  IF v_customer_id IS NOT NULL THEN
   601→   601→    RETURN v_customer_id;
   602→   602→  END IF;
   603→   603→
   604→   604→  -- ------------------------------------------------------------------
   605→   605→  -- Layer 2: phone match
   606→   606→  -- ------------------------------------------------------------------
   607→   607→  IF p_phone IS NOT NULL AND BTRIM(p_phone) <> '' THEN
   608→   608→    v_normalized_phone := normalize_phone(p_phone);
   609→   609→    IF v_normalized_phone IS NOT NULL THEN
   610→   610→      SELECT cp."customerId" INTO v_customer_id
   611→   611→      FROM customer_phones cp
   612→   612→      WHERE cp."organizationId" = p_organization_id
   613→   613→        AND cp."phoneNormalized" = v_normalized_phone
   614→   614→      LIMIT 1;
   615→   615→
   616→   616→      IF v_customer_id IS NOT NULL THEN
   617→   617→        INSERT INTO customer_external_identities
   618→   618→          ("customerId", "organizationId", platform, "externalCustomerId", "matchedVia")
   619→   619→        VALUES
   620→   620→          (v_customer_id, p_organization_id, p_platform, p_external_customer_id, 'phone_match')
   621→   621→        ON CONFLICT ("organizationId", platform, "externalCustomerId") DO NOTHING;
   622→   622→        RETURN v_customer_id;
   623→   623→      END IF;
   624→   624→    END IF;
   625→   625→  END IF;
   626→   626→
   627→   627→  -- ------------------------------------------------------------------
   628→   628→  -- Layer 3: email match
   629→   629→  -- ------------------------------------------------------------------
   630→   630→  IF p_email IS NOT NULL AND BTRIM(p_email) <> '' THEN
   631→   631→    SELECT c.id INTO v_customer_id
   632→   632→    FROM "Customer" c
   633→   633→    WHERE c."organizationId" = p_organization_id
   634→   634→      AND c.email = p_email
   635→   635→    LIMIT 1;
   636→   636→
   637→   637→    IF v_customer_id IS NOT NULL THEN
   638→   638→      INSERT INTO customer_external_identities
   639→   639→        ("customerId", "organizationId", platform, "externalCustomerId", "matchedVia")
   640→   640→      VALUES
   641→   641→        (v_customer_id, p_organization_id, p_platform, p_external_customer_id, 'email_match')
   642→   642→      ON CONFLICT ("organizationId", platform, "externalCustomerId") DO NOTHING;
   643→   643→      RETURN v_customer_id;
   644→   644→    END IF;
   645→   645→  END IF;
   646→   646→
   647→   647→  -- ------------------------------------------------------------------
   648→   648→  -- Layer 4: create new customer (race-protected)
   649→   649→  -- ------------------------------------------------------------------
   650→   650→  v_lock_key := hashtextextended(
   651→   651→    p_organization_id || ':' || p_platform || ':' || p_external_customer_id,
   652→   652→    0
   653→   653→  );
   654→   654→  PERFORM pg_advisory_xact_lock(v_lock_key);
   655→   655→
   656→   656→  -- Re-check Layer 1 under the lock.
   657→   657→  SELECT cei."customerId" INTO v_customer_id
   658→   658→  FROM customer_external_identities cei
   659→   659→  WHERE cei."organizationId" = p_organization_id
   660→   660→    AND cei.platform = p_platform
   661→   661→    AND cei."externalCustomerId" = p_external_customer_id
   662→   662→  LIMIT 1;
   663→   663→
   664→   664→  IF v_customer_id IS NOT NULL THEN
   665→   665→    RETURN v_customer_id;
   666→   666→  END IF;
   667→   667→
   668→   668→  -- Create the new customer. Name defaults to 'Unknown Customer' if the
   669→   669→  -- platform did not supply one (Shopify sometimes doesn't for guest orders).
   670→   670→  INSERT INTO "Customer" ("organizationId", name, email)
   671→   671→  VALUES (p_organization_id, COALESCE(NULLIF(BTRIM(p_name), ''), 'Unknown Customer'), NULLIF(BTRIM(p_email), ''))
   672→   672→  RETURNING id INTO v_customer_id;
   673→   673→
   674→   674→  -- Create the primary phone row (if provided). ON CONFLICT DO NOTHING guards
   675→   675→  -- against the very rare case where the normalized phone already exists for
   676→   676→  -- a different customer in this org.
   677→   677→  IF p_phone IS NOT NULL AND BTRIM(p_phone) <> '' THEN
   678→   678→    v_normalized_phone := normalize_phone(p_phone);
   679→   679→    IF v_normalized_phone IS NOT NULL THEN
   680→   680→      INSERT INTO customer_phones
   681→   681→        ("customerId", "organizationId", "phoneRaw", "phoneNormalized", "isPrimary", label)
   682→   682→      VALUES
   683→   683→        (v_customer_id, p_organization_id, p_phone, v_normalized_phone, TRUE, 'Primary')
   684→   684→      ON CONFLICT ("organizationId", "phoneNormalized") DO NOTHING;
   685→   685→    END IF;
   686→   686→  END IF;
   687→   687→
   688→   688→  -- Establish the external identity mapping (first time this external ID is
   689→   689→  -- seen, hence 'exact_identity').
   690→   690→  INSERT INTO customer_external_identities
   691→   691→    ("customerId", "organizationId", platform, "externalCustomerId", "matchedVia")
   692→   692→  VALUES
   693→   693→    (v_customer_id, p_organization_id, p_platform, p_external_customer_id, 'exact_identity');
   694→   694→
   695→   695→  RETURN v_customer_id;
   696→   696→END;
   697→   697→$$;
