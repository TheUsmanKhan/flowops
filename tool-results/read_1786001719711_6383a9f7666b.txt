     3→     3→/**
     4→     4→ * BookingWorkbenchView
     5→     5→ * ---------------------
     6→     6→ * Bulk booking workbench for external-platform orders that have NOT yet been
     7→     7→ * booked with a courier (i.e. courierCompanyIntegrationId === null).
     8→     8→ *
     9→     9→ * Fetches confirmed/processing orders via:
    10→    10→ *   GET /api/orders?statuses=confirmed,processing&limit=100
    11→    11→ *
    12→    12→ * Then client-side filters to orders where courierCompanyIntegrationId is null
    13→    13→ * (the API doesn't yet expose this as a query param — adding it would require
    14→    14→ * a /lib change which is out-of-scope for this task).
    15→    15→ *
    16→    16→ * Each row exposes editable inputs for customer name / phone / address,
    17→    17→ * a CityAutocomplete (providerKey from the batch-level courier dropdown),
    18→    18→ * COD amount, and an order type dropdown (Normal/Overland/Replacement —
    19→    19→ * pre-computed client-side via the same logic as the PostEx adapter for
    20→    20→ * convenience, but editable so the operator can override).
    21→    21→ *
    22→    22→ * The top toolbar has:
    23→    23→ *   - Courier integration dropdown (drives the CityAutocomplete providerKey)
    24→    24→ *   - "Select All" / "Select None"
    25→    25→ *   - "Upload Booking" button — fires POST /api/booking-workbench/book for
    26→    26→ *     each CHECKED row independently. Failures do NOT block other rows.
    27→    27→ *
    28→    28→ * Per-row result: ✅ tracking number, or ❌ error message.
    29→    29→ */

   185→   185→
   186→   186→export function BookingWorkbenchView() {
   187→   187→  const queryClient = useQueryClient()
   188→   188→
   189→   189→  // Batch-level state
   190→   190→  const [companyIntegrationId, setCompanyIntegrationId] = useState<string>('')
   191→   191→  const [search, setSearch] = useState('')
   192→   192→
   193→   193→  // Per-row editable state keyed by order id
   194→   194→  const [rowStates, setRowStates] = useState<Record<string, RowState>>({})
   195→   195→
   196→   196→  // ── Data: orders ────────────────────────────────────────────────────────
   197→   197→  const ordersQuery = useQuery<OrdersListResponse>({
   198→   198→    queryKey: ['orders', 'booking-workbench'],
   199→   199→    queryFn: () =>
   200→   200→      api.get<OrdersListResponse>(
   201→   201→        '/api/orders?statuses=confirmed,processing&limit=100',
   202→   202→      ),
   203→   203→    staleTime: 15_000,
   204→   204→  })
   205→   205→
   206→   206→  // Client-side filter: only unbooked orders (no courier integration)
   207→   207→  const unbookedOrders = useMemo(() => {
   208→   208→    if (!ordersQuery.data?.orders) return []
   209→   209→    return ordersQuery.data.orders.filter(
   210→   210→      (o) => !o.courierCompanyIntegrationId,
   211→   211→    )
   212→   212→  }, [ordersQuery.data])
   213→   213→
   214→   214→  // Apply search filter (order # / customer / phone)
   215→   215→  const filteredOrders = useMemo(() => {
   216→   216→    if (!search.trim()) return unbookedOrders
   217→   217→    const q = search.trim().toLowerCase()
   218→   218→    return unbookedOrders.filter(
   219→   219→      (o) =>
   220→   220→        o.flowopsOrderNumber.toLowerCase().includes(q) ||
   221→   221→        o.externalOrderReference?.toLowerCase().includes(q) ||
   222→   222→        o.customerName.toLowerCase().includes(q) ||
   223→   223→        o.customerPhone?.toLowerCase().includes(q),
   224→   224→    )
   225→   225→  }, [unbookedOrders, search])
   226→   226→
   227→   227→  // ── Data: courier integrations ──────────────────────────────────────────
   228→   228→  const integrationsQuery = useQuery<IntegrationsResponse>({
   229→   229→    queryKey: ['integrations', 'courier'],
   230→   230→    queryFn: () => api.get<IntegrationsResponse>('/api/integrations?category=courier'),
   231→   231→    staleTime: 30_000,
   232→   232→  })
   233→   233→
   234→   234→  const courierIntegrations = useMemo(
   235→   235→    () =>
   236→   236→      (integrationsQuery.data?.integrations ?? []).filter(
   237→   237→        (i) => i.isActive && i.provider?.category === 'courier',
   238→   238→      ),
   239→   239→    [integrationsQuery.data],
   240→   240→  )
   241→   241→
   242→   242→  const selectedIntegration = useMemo(
   243→   243→    () => courierIntegrations.find((i) => i.id === companyIntegrationId),
   244→   244→    [courierIntegrations, companyIntegrationId],
   245→   245→  )
   246→   246→  const selectedProviderKey = selectedIntegration?.provider?.providerKey ?? ''
   247→   247→
   248→   248→  // ── Row state helpers ───────────────────────────────────────────────────
   249→   249→  function getRowState(order: OrderRow): RowState {
   250→   250→    if (rowStates[order.id]) return rowStates[order.id]
   251→   251→    return {
   252→   252→      customerName: order.customerName ?? '',
   253→   253→      customerPhone: order.customerPhone ?? '',
   254→   254→      deliveryAddress: order.deliveryAddress ?? '',
   255→   255→      deliveryCity: order.deliveryCity ?? '',
   256→   256→      codAmount: String(
   257→   257→        order.remainingCodAmount ?? order.totalOrderValue ?? 0,
   258→   258→      ),
   259→   259→      orderType: 'Normal',
   260→   260→      checked: false,
   261→   261→      result: null,
   262→   262→      isBooking: false,
   263→   263→    }
   264→   264→  }
   265→   265→
   266→   266→  function patchRow(orderId: string, patch: Partial<RowState>) {
   267→   267→    setRowStates((prev) => ({
   268→   268→      ...prev,
   269→   269→      [orderId]: {
   270→   270→        ...(prev[orderId] ?? getRowStateLookup(orderId)),
   271→   271→        ...patch,
   272→   272→        // Clear any previous result when the user edits a field
   273→   273→        result:
   274→   274→          patch.result !== undefined ? patch.result : null,
   275→   275→      },
   276→   276→    }))
   277→   277→  }
   278→   278→
   279→   279→  // Lookup that doesn't depend on filteredOrders (for patchRow safety)
   280→   280→  function getRowStateLookup(orderId: string): RowState {
   281→   281→    if (rowStates[orderId]) return rowStates[orderId]
   282→   282→    const order = ordersQuery.data?.orders?.find((o) => o.id === orderId)
   283→   283→    if (order) {
   284→   284→      return {
   285→   285→        customerName: order.customerName ?? '',
   286→   286→        customerPhone: order.customerPhone ?? '',
   287→   287→        deliveryAddress: order.deliveryAddress ?? '',
   288→   288→        deliveryCity: order.deliveryCity ?? '',
   289→   289→        codAmount: String(
   290→   290→          order.remainingCodAmount ?? order.totalOrderValue ?? 0,
   291→   291→        ),
   292→   292→        orderType: 'Normal' as const,
   293→   293→        checked: false,
   294→   294→        result: null,
   295→   295→        isBooking: false,
   296→   296→      }
   297→   297→    }
   298→   298→    return {
   299→   299→      customerName: '',
   300→   300→      customerPhone: '',
   301→   301→      deliveryAddress: '',
   302→   302→      deliveryCity: '',
   303→   303→      codAmount: '0',
   304→   304→      orderType: 'Normal' as const,
   305→   305→      checked: false,
   306→   306→      result: null,
   307→   307→      isBooking: false,
   308→   308→    }
   309→   309→  }
   310→   310→
   311→   311→  // ── Select-all helpers ──────────────────────────────────────────────────
   312→   312→  const checkedCount = filteredOrders.filter(
   313→   313→    (o) => getRowState(o).checked,
   314→   314→  ).length
   315→   315→  const allChecked =
   316→   316→    filteredOrders.length > 0 && checkedCount === filteredOrders.length
   317→   317→
   318→   318→  function toggleSelectAll() {
   319→   319→    const next: Record<string, RowState> = { ...rowStates }
   320→   320→    const target = !allChecked
   321→   321→    for (const o of filteredOrders) {
   322→   322→      const cur = next[o.id] ?? getRowState(o)
   323→   323→      next[o.id] = { ...cur, checked: target }
   324→   324→    }
   325→   325→    setRowStates(next)
   326→   326→  }
   327→   327→
   328→   328→  // ── Booking mutation (per-row) ──────────────────────────────────────────
   329→   329→  const bookMutation = useMutation({
   330→   330→    mutationFn: async (input: {
   331→   331→      orderId: string
   332→   332→      body: BookRequest
   333→   333→    }): Promise<BookSuccess> => {
   334→   334→      return api.post<BookSuccess>('/api/booking-workbench/book', input.body)
   335→   335→    },
   336→   336→    // We handle success/failure in the batchSubmit loop below, not here.
   337→   337→  })
   338→   338→
   339→   339→  // ── Batch submit ────────────────────────────────────────────────────────
   340→   340→  async function handleUploadBooking() {
   341→   341→    if (!companyIntegrationId) {
   342→   342→      toast.error('Select a courier integration first.')
   343→   343→      return
   344→   344→    }
   345→   345→    const checkedRows = filteredOrders.filter((o) => {
   346→   346→      const s = getRowState(o)
   347→   347→      return s.checked && !s.isBooking
   348→   348→    })
   349→   349→    if (checkedRows.length === 0) {
   350→   350→      toast.error('No rows selected. Tick the checkbox on at least one order.')
   351→   351→      return
   352→   352→    }
   353→   353→
   354→   354→    toast.info(`Booking ${checkedRows.length} order(s) with ${selectedIntegration?.provider?.providerName ?? 'courier'}…`)
   355→   355→
   356→   356→    let successCount = 0
   357→   357→    let failureCount = 0
   358→   358→
   359→   359→    // Fire requests sequentially — booking calls external courier APIs which
   360→   360→    // typically don't tolerate parallel hits well, and per-row errors must
   361→   361→    // not block other rows.
   362→   362→    for (const order of checkedRows) {
   363→   363→      const state = getRowState(order)
   364→   364→      const body: BookRequest = {
   365→   365→        orderId: order.id,
   366→   366→        companyIntegrationId,
   367→   367→        customerName: state.customerName.trim() || undefined,
   368→   368→        customerPhone: state.customerPhone.trim() || undefined,
   369→   369→        deliveryAddress: state.deliveryAddress.trim() || undefined,
   370→   370→        deliveryCity: state.deliveryCity.trim() || undefined,
   371→   371→        codAmount: Number(state.codAmount) || 0,
   372→   372→        orderType: state.orderType,
   373→   373→      }
   374→   374→
   375→   375→      // Mark as booking in progress
   376→   376→      patchRow(order.id, { isBooking: true, result: null })
   377→   377→
   378→   378→      try {
   379→   379→        const result = await bookMutation.mutateAsync({
   380→   380→          orderId: order.id,
   381→   381→          body,
   382→   382→        })
   383→   383→        successCount++
   384→   384→        patchRow(order.id, {
   385→   385→          isBooking: false,
   386→   386→          checked: false,
   387→   387→          result: {
   388→   388→            ok: true,
   389→   389→            trackingNumber: result.trackingNumber,
   390→   390→            orderType: result.orderType,
   391→   391→          },
   392→   392→        })
   393→   393→      } catch (err) {
   394→   394→        failureCount++
   395→   395→        const message =
   396→   396→          err instanceof FetchError
   397→   397→            ? err.message
   398→   398→            : err instanceof Error
   399→   399→              ? err.message
   400→   400→              : 'Booking failed'
   401→   401→        patchRow(order.id, {
   402→   402→          isBooking: false,
   403→   403→          result: { ok: false, error: message },
   404→   404→        })
   405→   405→      }
   406→   406→    }
   407→   407→
   408→   408→    // Final toast + invalidate orders list so successfully booked rows
   409→   409→    // disappear (they now have courierCompanyIntegrationId set).
   410→   410→    if (successCount > 0 && failureCount === 0) {
   411→   411→      toast.success(
   412→   412→        `All ${successCount} order(s) booked successfully.`,
   413→   413→      )
   414→   414→    } else if (successCount > 0 && failureCount > 0) {
   415→   415→      toast.warning(
   416→   416→        `${successCount} booked, ${failureCount} failed. See per-row status.`,
   417→   417→      )
   418→   418→    } else {
   419→   419→      toast.error(`All ${failureCount} booking(s) failed. See per-row errors.`)
   420→   420→    }
   421→   421→
   422→   422→    if (successCount > 0) {
   423→   423→      queryClient.invalidateQueries({ queryKey: ['orders', 'booking-workbench'] })
   424→   424→      queryClient.invalidateQueries({ queryKey: ['orders'] })
   425→   425→    }
   426→   426→  }
   427→   427→
   428→   428→  // ── Render: loading / error / empty ─────────────────────────────────────
   429→   429→  if (ordersQuery.isLoading) {
   430→   430→    return (
   431→   431→      <div className="space-y-4">
   432→   432→        <PageHeader
   433→   433→          title="Booking Workbench"
   434→   434→          description="Bulk-book external-platform orders with a courier."
   435→   435→        />
   436→   436→        <Card>
   437→   437→          <CardContent className="p-10 text-center">
   438→   438→            <Loader2 className="h-6 w-6 animate-spin mx-auto text-muted-foreground" />
   439→   439→            <p className="text-sm text-muted-foreground mt-2">
   440→   440→              Loading unbooked orders…
   441→   441→            </p>
   442→   442→          </CardContent>
   443→   443→        </Card>
   444→   444→      </div>
   445→   445→    )
   446→   446→  }
   447→   447→
   448→   448→  if (ordersQuery.isError) {
   449→   449→    return (
   450→   450→      <div className="space-y-4">
   451→   451→        <PageHeader
   452→   452→          title="Booking Workbench"
   453→   453→          description="Bulk-book external-platform orders with a courier."
   454→   454→        />
   455→   455→        <Card className="border-rose-200">
   456→   456→          <CardContent className="p-10 text-center space-y-3">
   457→   457→            <AlertCircle className="h-6 w-6 mx-auto text-rose-600" />
   458→   458→            <p className="text-sm text-rose-700">
   459→   459→              {getErrorMessage(ordersQuery.error)}
   460→   460→            </p>
   461→   461→            <Button
   462→   462→              variant="outline"
   463→   463→              size="sm"
   464→   464→              onClick={() => ordersQuery.refetch()}
   465→   465→            >
   466→   466→              <RefreshCw className="h-3.5 w-3.5" /> Retry
   467→   467→            </Button>
   468→   468→          </CardContent>
   469→   469→        </Card>
   470→   470→      </div>
   471→   471→    )
   472→   472→  }
   473→   473→
   474→   474→  return (
   475→   475→    <div className="space-y-4">
   476→   476→      <PageHeader
   477→   477→        title="Booking Workbench"
   478→   478→        description="Bulk-book confirmed external-platform orders that haven't yet been booked with a courier."
   479→   479→        actions={
   480→   480→          <Button
   481→   481→            variant="outline"
   482→   482→            size="sm"
   483→   483→            onClick={() => ordersQuery.refetch()}
   484→   484→            disabled={ordersQuery.isFetching}
   485→   485→          >
   486→   486→            <RefreshCw
   487→   487→              className={cn(
   488→   488→                'h-3.5 w-3.5',
   489→   489→                ordersQuery.isFetching && 'animate-spin',
   490→   490→              )}
   491→   491→            />
   492→   492→            Refresh
   493→   493→          </Button>
   494→   494→        }
   495→   495→      />
   496→   496→
   497→   497→      {/* Batch toolbar */}
   498→   498→      <Card>
   499→   499→        <CardContent className="p-4 space-y-3">
   500→   500→          <div className="grid gap-3 md:grid-cols-[1fr_1fr_auto] md:items-end">
   501→   501→            {/* Courier integration */}
   502→   502→            <div className="space-y-1.5">
   503→   503→              <Label className="text-xs flex items-center gap-1">
   504→   504→                <Truck className="h-3 w-3" /> Courier Integration
   505→   505→              </Label>
