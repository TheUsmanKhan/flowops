    91→    91→// ──────────────────────────────────────────────────────────────
    92→    92→// Processes ALL order_items for an order and attempts to reserve
    93→    93→// stock for each based on its fulfillment_type_snapshot.
    94→    94→//
    95→    95→// For stock_based items:
    96→    96→//   - If sufficient available stock → reserveStockForOrder(), fulfillment_status='reserved'
    97→    97→//   - If insufficient + inventory_policy='continue' → fulfillment_status='backordered'
    98→    98→//   - If insufficient + inventory_policy='deny' → outcome='failed'
    99→    99→//
   100→   100→// For made_to_order items:
   101→   101→//   - checkAndFulfillMadeToOrderVariant() checks returned stock first
   102→   102→//   - If existing_stock → fulfillment_status='reserved', returned_stitched_used=true
   103→   103→//   - If fresh_production → fulfillment_status='reserved', production_order_id set
   104→   104→//   - If error → outcome='failed'
   105→   105→//
   106→   106→// After processing: if ANY item is 'backordered', order status → 'partially_backordered'.
   107→   107→// If ALL items are 'reserved', order stays 'confirmed'.
   108→   108→//
   109→   109→// Called from:
   110→   110→//   - createManualOrder() IF the order auto-confirmed at creation
   111→   111→//   - confirmOrder() when a 'pending' order is manually confirmed
   112→   112→//   - dispatchOrderAction() if status is still 'pending' at dispatch time
   113→   113→
   114→   114→async function reserveOrderStock(
   115→   115→  orderId: string,
   116→   116→  ctx: { company: { id: string; organizationId: string }; user: { id: string }; employee: { id: string } },
   117→   117→): Promise<{
   118→   118→  success: boolean
   119→   119→  results: Array<{ orderItemId: string; outcome: 'reserved' | 'backordered' | 'failed'; reason?: string }>
   120→   120→}> {
   121→   121→  const order = await db.order.findUnique({
   122→   122→    where: { id: orderId },
   123→   123→    select: { dispatchLocationId: true, companyId: true, organizationId: true },
   124→   124→  })
   125→   125→  if (!order) return { success: false, results: [] }
   126→   126→
   127→   127→  const items = await db.orderItem.findMany({
   128→   128→    where: { orderId },
   129→   129→    include: {
   130→   130→      orgVariant: {
   131→   131→        select: { id: true, fulfillmentType: true, inventoryPolicy: true, stitchingCharges: true },
   132→   132→      },
   133→   133→    },
   134→   134→  })
   135→   135→
   136→   136→  const results: Array<{ orderItemId: string; outcome: 'reserved' | 'backordered' | 'failed'; reason?: string }> = []
   137→   137→  let hasBackordered = false
   138→   138→
   139→   139→  for (const item of items) {
   140→   140→    if (item.fulfillmentStatus === 'reserved' || item.fulfillmentStatus === 'dispatched') {
   141→   141→      // Already processed (e.g. re-confirmation attempt) — skip
   142→   142→      results.push({ orderItemId: item.id, outcome: 'reserved' })
   143→   143→      continue
   144→   144→    }
   145→   145→
   146→   146→    const locationId = item.reservedLocationId ?? order.dispatchLocationId
   147→   147→    if (!locationId) {
   148→   148→      results.push({ orderItemId: item.id, outcome: 'failed', reason: 'No dispatch location set' })
   149→   149→      continue
   150→   150→    }
   151→   151→
   152→   152→    if (item.fulfillmentTypeSnapshot === 'stock_based') {
   153→   153→      // Check available stock at the dispatch location
   154→   154→      const pool = await db.inventoryPool.findUnique({
   155→   155→        where: {
   156→   156→          orgVariantId_locationId: {
   157→   157→            orgVariantId: item.orgVariantId,
   158→   158→            locationId,
   159→   159→          },
   160→   160→        },
   161→   161→      })
   162→   162→
   163→   163→      const available = pool ? pool.onHand - pool.reserved : 0
   164→   164→
   165→   165→      if (available >= item.quantity) {
   166→   166→        // Sufficient stock — reserve it
   167→   167→        const reserveResult = await reserveStockForOrder({
   168→   168→          orgVariantId: item.orgVariantId,
   169→   169→          locationId,
   170→   170→          organizationId: order.organizationId,
   171→   171→          companyId: order.companyId,
   172→   172→          employeeId: ctx.employee.id,
   173→   173→          quantity: item.quantity,
   174→   174→          orderId,
   175→   175→        })
   176→   176→
   177→   177→        if (reserveResult.success) {
   178→   178→          await db.orderItem.update({
   179→   179→            where: { id: item.id },
   180→   180→            data: { fulfillmentStatus: 'reserved', reservedLocationId: locationId },
   181→   181→          })
   182→   182→          results.push({ orderItemId: item.id, outcome: 'reserved' })
   183→   183→        } else {
   184→   184→          results.push({ orderItemId: item.id, outcome: 'failed', reason: reserveResult.error })
   185→   185→        }
   186→   186→      } else if (item.orgVariant.inventoryPolicy === 'continue') {
   187→   187→        // Insufficient stock but backordering allowed
   188→   188→        await db.orderItem.update({
   189→   189→          where: { id: item.id },
   190→   190→          data: { fulfillmentStatus: 'backordered', backorderedAt: new Date() },
   191→   191→        })
   192→   192→        await insertMetricEvent({
   193→   193→          companyId: order.companyId,
   194→   194→          entityType: 'product',
   195→   195→          entityId: item.orgVariantId,
   196→   196→          metricKey: 'order.backordered',
   197→   197→          numericValue: item.quantity,
   198→   198→          dimensions: { order_id: orderId },
   199→   199→        }).catch(() => {})
   200→   200→        hasBackordered = true
   201→   201→        results.push({
   202→   202→          orderItemId: item.id,
   203→   203→          outcome: 'backordered',
   204→   204→          reason: `Available: ${available}, required: ${item.quantity}`,
   205→   205→        })
   206→   206→      } else {
   207→   207→        // Insufficient stock + policy='deny' — fail
   208→   208→        results.push({
   209→   209→          orderItemId: item.id,
   210→   210→          outcome: 'failed',
   211→   211→          reason: `Insufficient stock (available: ${available}, required: ${item.quantity}) and inventory_policy='deny'`,
   212→   212→        })
   213→   213→      }
   214→   214→    } else if (item.fulfillmentTypeSnapshot === 'made_to_order') {
   215→   215→      // Made-to-order: check returned stock first, then trigger production
   216→   216→      const mtoResult = await checkAndFulfillMadeToOrderVariant(
   217→   217→        item.orgVariantId,
   218→   218→        item.quantity,
   219→   219→        order.companyId,
   220→   220→        locationId,
   221→   221→      )
   222→   222→
   223→   223→      if (mtoResult.source === 'existing_stock' && mtoResult.locationId) {
   224→   224→        // Returned stock available — reserve it
   225→   225→        const reserveResult = await reserveStockForOrder({
   226→   226→          orgVariantId: item.orgVariantId,
   227→   227→          locationId: mtoResult.locationId,
   228→   228→          organizationId: order.organizationId,
   229→   229→          companyId: order.companyId,
   230→   230→          employeeId: ctx.employee.id,
   231→   231→          quantity: item.quantity,
   232→   232→          orderId,
   233→   233→        })
   234→   234→
   235→   235→        if (reserveResult.success) {
   236→   236→          await db.orderItem.update({
   237→   237→            where: { id: item.id },
   238→   238→            data: {
   239→   239→              fulfillmentStatus: 'reserved',
   240→   240→              returnedStitchedUsed: true,
   241→   241→              reservedLocationId: mtoResult.locationId,
   242→   242→            },
   243→   243→          })
   244→   244→          await insertMetricEvent({
   245→   245→            companyId: order.companyId,
   246→   246→            entityType: 'product',
   247→   247→            entityId: item.orgVariantId,
   248→   248→            metricKey: 'order.made_to_order_from_returned_stock',
   249→   249→            numericValue: 1,
   250→   250→            dimensions: { order_id: orderId, stitching_cost_saved: Number(item.orgVariant.stitchingCharges) || 0 },
   251→   251→          }).catch(() => {})
   252→   252→          results.push({ orderItemId: item.id, outcome: 'reserved' })
   253→   253→        } else {
   254→   254→          results.push({ orderItemId: item.id, outcome: 'failed', reason: reserveResult.error })
   255→   255→        }
   256→   256→      } else if (mtoResult.source === 'fresh_production' && mtoResult.productionOrderId) {
   257→   257→        // Fresh production triggered — link the production order
   258→   258→        await db.orderItem.update({
   259→   259→          where: { id: item.id },
   260→   260→          data: {
   261→   261→            fulfillmentStatus: 'reserved',
   262→   262→            productionOrderId: mtoResult.productionOrderId,
   263→   263→          },
   264→   264→        })
   265→   265→        // Also link the order_item back to the production order
   266→   266→        await db.productionOrder.update({
   267→   267→          where: { id: mtoResult.productionOrderId },
   268→   268→          data: { orderItemId: item.id },
   269→   269→        })
   270→   270→        await insertMetricEvent({
   271→   271→          companyId: order.companyId,
   272→   272→          entityType: 'product',
   273→   273→          entityId: item.orgVariantId,
   274→   274→          metricKey: 'order.made_to_order_production_triggered',
   275→   275→          numericValue: 1,
   276→   276→          dimensions: { order_id: orderId, production_order_id: mtoResult.productionOrderId },
   277→   277→        }).catch(() => {})
   278→   278→        results.push({ orderItemId: item.id, outcome: 'reserved' })
   279→   279→      } else {
   280→   280→        // Error (no fabric source, insufficient fabric, etc.)
   281→   281→        results.push({
   282→   282→          orderItemId: item.id,
   283→   283→          outcome: 'failed',
   284→   284→          reason: mtoResult.error ?? 'Made-to-order fulfillment failed',
   285→   285→        })
   286→   286→      }
   287→   287→    }
   288→   288→  }
   289→   289→
   290→   290→  // Update order status based on results
   291→   291→  if (hasBackordered) {
   292→   292→    await db.order.update({
   293→   293→      where: { id: orderId },
   294→   294→      data: { status: 'partially_backordered' },
   295→   295→    })
   296→   296→  }
   297→   297→  // If no backordered items, order stays 'confirmed' (or whatever it was)
   298→   298→
   299→   299→  return { success: true, results }
   300→   300→}
   301→   301→

   303→   303→// createManualOrder

   305→   305→
   306→   306→export async function createManualOrder(
   307→   307→  input: CreateManualOrderInput,
   308→   308→): Promise<ActionResult<{
   309→   309→  orderId: string
   310→   310→  flowopsOrderNumber: string
   311→   311→  orderItems: Array<{ id: string; orgVariantId: string; quantity: number }>
   312→   312→}>> {
   313→   313→  try {
   314→   314→    const ctx = await getWorkspace()
   315→   315→    await requirePermission(ctx, PERMISSIONS.ORDERS_CREATE)
   316→   316→
   317→   317→    // 1. Validate
   318→   318→    const parsed = createManualOrderSchema.safeParse(input)
   319→   319→    if (!parsed.success) {
   320→   320→      return {
   321→   321→        success: false,
   322→   322→        error: parsed.error.issues[0]?.message ?? 'Invalid order data',
   323→   323→      }
   324→   324→    }
   325→   325→    const d = parsed.data
   326→   326→
   327→   327→    // 2. Resolve customer (existing customer_id OR inline new_customer).
   328→   328→    //    Also resolve optional saved address/phone selection and recipient_name.
   329→   329→    let customerId: string
   330→   330→    let customerName: string
   331→   331→    let usedCustomerAddressId: string | null = null
   332→   332→    let usedCustomerPhoneId: string | null = null
   333→   333→    let saveAddressForNextTime = d.save_address_for_next_time ?? false
   334→   334→    // Whether the delivery_address came from a saved customer_addresses row
   335→   335→    // (vs. typed as a one-off). Used to decide whether to bump lastUsedAt.
   336→   336→    let selectedSavedAddressId: string | null = null
   337→   337→
   338→   338→    if (d.customer_id) {
   339→   339→      // Existing customer path — verify they belong to this org.
   340→   340→      const existing = await db.customer.findFirst({
   341→   341→        where: { id: d.customer_id, organizationId: ctx.company.organizationId },
   342→   342→        select: { id: true, name: true },
   343→   343→      })
   344→   344→      if (!existing) return { success: false, error: 'Customer not found' }
   345→   345→      customerId = existing.id
   346→   346→      customerName = existing.name
   347→   347→
   348→   348→      // If a saved address was selected, verify it belongs to this customer.
   349→   349→      if (d.used_customer_address_id) {
   350→   350→        const addr = await db.customerAddress.findFirst({
   351→   351→          where: {
   352→   352→            id: d.used_customer_address_id,
   353→   353→            customerId: existing.id,
   354→   354→            organizationId: ctx.company.organizationId,
   355→   355→          },
   356→   356→          select: { id: true, address: true, city: true },
   357→   357→        })
   358→   358→        if (addr) {
   359→   359→          usedCustomerAddressId = addr.id
   360→   360→          selectedSavedAddressId = addr.id
   361→   361→          // The delivery_address snapshot is the order's own editable copy.
   362→   362→          // If the caller passed the same address text, we trust it; if they
   363→   363→          // passed different text (edited it), we still record the link to
   364→   364→          // the saved address for lastUsedAt tracking but keep the edited text
   365→   365→          // as the order's snapshot. Either way, delivery_address comes from
   366→   366→          // the input — we do NOT override it here.
   367→   367→        }
   368→   368→      }
   369→   369→
   370→   370→      // If a saved phone was selected, verify it belongs to this customer.
   371→   371→      if (d.used_customer_phone_id) {
   372→   372→        const phone = await db.customerPhone.findFirst({
   373→   373→          where: {
   374→   374→            id: d.used_customer_phone_id,
   375→   375→            customerId: existing.id,
   376→   376→            organizationId: ctx.company.organizationId,
   377→   377→          },
   378→   378→          select: { id: true },
   379→   379→        })
   380→   380→        if (phone) {
   381→   381→          usedCustomerPhoneId = phone.id
   382→   382→        }
   383→   383→      }
   384→   384→    } else if (d.new_customer) {
   385→   385→      // Inline new-customer path — create the customer + their phones +
   386→   386→      // addresses now via the Customer Management System. The first phone is
   387→   387→      // primary and the first address is default (validated by createCustomerSchema).
   388→   388→      const createResult = await createCustomer(d.new_customer)
   389→   389→      if (!createResult.success || !createResult.data) {
   390→   390→        return { success: false, error: createResult.error ?? 'Failed to create customer' }
   391→   391→      }
   392→   392→      customerId = createResult.data.customerId
   393→   393→      customerName = d.new_customer.name.trim()
   394→   394→      // For a brand-new customer, the first address they provided IS the
   395→   395→      // saved default address. Mark it as the selected address so we bump
   396→   396→      // its lastUsedAt after order creation.
   397→   397→      const newAddr = await db.customerAddress.findFirst({
   398→   398→        where: { customerId, isDefault: true },
   399→   399→        select: { id: true },
   400→   400→      })
   401→   401→      if (newAddr) {
   402→   402→        usedCustomerAddressId = newAddr.id
   403→   403→        selectedSavedAddressId = newAddr.id
   404→   404→      }
   405→   405→      // Same for the primary phone.
   406→   406→      const newPhone = await db.customerPhone.findFirst({
   407→   407→        where: { customerId, isPrimary: true },
   408→   408→        select: { id: true },
   409→   409→      })
   410→   410→      if (newPhone) {
   411→   411→        usedCustomerPhoneId = newPhone.id
   412→   412→      }
   413→   413→      // For a brand-new customer, we already saved the address as part of
   414→   414→      // createCustomer — don't double-save it.
   415→   415→      saveAddressForNextTime = false
   416→   416→    } else {
   417→   417→      return { success: false, error: 'Either customer_id or new_customer is required' }
   418→   418→    }
   419→   419→
   420→   420→    // 3. recipient_name defaults to the customer's name if not explicitly overridden.
   421→   421→    const recipientName = (d.recipient_name && d.recipient_name.trim()) || customerName
   422→   422→
   423→   423→    // 3. Fetch variants + their pricing for this company + fulfillment_type snapshot
   424→   424→    const variantIds = d.items.map((i) => i.org_variant_id)
   425→   425→    const variants = await db.orgProductVariant.findMany({
   426→   426→      where: { id: { in: variantIds }, organizationId: ctx.company.organizationId },
   427→   427→      include: {
   428→   428→        product: { select: { title: true } },
   429→   429→        companyPricing: { where: { companyId: ctx.company.id } },
   430→   430→      },
   431→   431→    })
   432→   432→
   433→   433→    if (variants.length !== variantIds.length) {
   434→   434→      return { success: false, error: 'One or more variants not found in this organization' }
   435→   435→    }
   436→   436→
   437→   437→    // 4. Compute subtotal + build order items data
   438→   438→    let subtotal = 0
   439→   439→    const orderItemsData: Array<{
   440→   440→      orgVariantId: string
   441→   441→      quantity: number
   442→   442→      unitPrice: number
   443→   443→      lineTotal: number
   444→   444→      fulfillmentTypeSnapshot: string
   445→   445→    }> = []
   446→   446→
   447→   447→    // Build the order_detail string as we iterate — universal courier
   448→   448→    // reference field (migration 015). Format:
   449→   449→    //   "Product Title (SKU-001, Size: M, Color: Blue) ×2, ..."
   450→   450→    // Auto-generated but editable from the order-create form. If the caller
   451→   451→    // supplies an explicit order_detail, we honour it verbatim.
   452→   452→    const orderDetailParts: string[] = []
   453→   453→
   454→   454→    for (const item of d.items) {
   455→   455→      const variant = variants.find((v) => v.id === item.org_variant_id)
   456→   456→      if (!variant) continue
   457→   457→
   458→   458→      // Use provided unit_price OR fall back to company pricing OR variant cost
   459→   459→      const unitPrice =
   460→   460→        item.unit_price ??
   461→   461→        (variant.companyPricing[0]?.salePrice
   462→   462→          ? Number(variant.companyPricing[0].salePrice)
   463→   463→          : Number(variant.costPrice))
   464→   464→
   465→   465→      const lineTotal = item.quantity * unitPrice
   466→   466→      subtotal += lineTotal
   467→   467→
   468→   468→      orderItemsData.push({
   469→   469→        orgVariantId: item.org_variant_id,
   470→   470→        quantity: item.quantity,
   471→   471→        unitPrice,
   472→   472→        lineTotal,
   473→   473→        fulfillmentTypeSnapshot: variant.fulfillmentType,
   474→   474→      })
   475→   475→
   476→   476→      // Append to orderDetail: "Product Title (SKU-001, Size: M) ×2"

   542→   542→      })

   709→   709→    return {

   713→   713→  } catch (err) {

   716→   716→      error: err instanceof Error ? err.message : 'Failed to create order',
   717→   717→    }
   718→   718→  }
