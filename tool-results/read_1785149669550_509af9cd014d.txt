   176→   176→function parseQty(v: string): number {
   177→   177→  if (v === '') return 0

   197→   197→/** Pick the shipping address from a customer's saved addresses array.
   198→   198→ *  Honors the legacy fallback: untyped addresses are treated as shipping. */
   199→   199→function pickShippingAddress(addresses: CustomerAddress[] | undefined): CustomerAddress | null {
   200→   200→  if (!addresses || addresses.length === 0) return null
   201→   201→  return (
   202→   202→    addresses.find((a) => a.type === 'shipping') ??
   203→   203→    addresses.find((a) => !a.type) ??
   204→   204→    addresses[0]
   205→   205→  )
   206→   206→}

   571→   571→  // ── Upload payment proof + persist URL to the order ───────────────────────
   572→   572→  // Returns true on success, false on any failure. NEVER throws.
   573→   573→  async function uploadPaymentProof(orderId: string): Promise<boolean> {
   574→   574→    if (!paymentProofFile) return true // nothing to upload — trivially "success"
   575→   575→
   576→   576→    setUploadingProof(true)
   577→   577→    try {
   578→   578→      // Step 1: upload the raw file via /api/upload
   579→   579→      const fd = new FormData()
   580→   580→      fd.append('file', paymentProofFile)
   581→   581→      const uploadRes = await fetch(
   582→   582→        `/api/upload?type=payment-proofs&id=${orderId}`,
   583→   583→        { method: 'POST', body: fd },
   584→   584→      )
   585→   585→      const uploadText = await uploadRes.text()
   586→   586→      let uploadBody: unknown = null
   587→   587→      if (uploadText) {
   588→   588→        try {
   589→   589→          uploadBody = JSON.parse(uploadText)
   590→   590→        } catch {
   591→   591→          uploadBody = uploadText
   592→   592→        }
   593→   593→      }
   594→   594→      if (!uploadRes.ok) {
   595→   595→        const msg =
   596→   596→          uploadBody && typeof uploadBody === 'object' && 'error' in uploadBody
   597→   597→            ? String((uploadBody as { error: unknown }).error)
   598→   598→            : typeof uploadBody === 'string'
   599→   599→              ? uploadBody
   600→   600→              : `Upload failed (HTTP ${uploadRes.status})`
   601→   601→        throw new Error(msg)
   602→   602→      }
   603→   603→      const { url } = uploadBody as { url: string }
   604→   604→
   605→   605→      // Step 2: persist the URL on the order via the dedicated endpoint
   606→   606→      // (NOT convert-payment — the order already has its payment type set).
   607→   607→      await api.post(`/api/orders/${orderId}/payment-proof`, {
   608→   608→        advance_payment_screenshot_url: url,
   609→   609→      })
   610→   610→      return true
   611→   611→    } catch {
   612→   612→      return false
   613→   613→    } finally {
   614→   614→      setUploadingProof(false)
   615→   615→    }
   616→   616→  }
   617→   617→
