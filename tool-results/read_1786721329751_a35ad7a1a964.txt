   268→function toISODate(d: Date): string {
   269→  const y = d.getFullYear()
   270→  const m = String(d.getMonth() + 1).padStart(2, '0')
   271→  const day = String(d.getDate()).padStart(2, '0')
   272→  return `${y}-${m}-${day}`
   273→}
   274→
   275→function resolveTimeSpan(value: TimeSpanValue, custom?: DateRange): DateRange {
   276→  const now = new Date()
   277→  const today = new Date(now.getFullYear(), now.getMonth(), now.getDate())
   278→
   279→  switch (value) {
   280→    case 'today':
   281→      return { from: toISODate(today), to: toISODate(today) }
   282→    case 'yesterday': {
   283→      const y = new Date(today)
   284→      y.setDate(y.getDate() - 1)
   285→      return { from: toISODate(y), to: toISODate(y) }
   286→    }
   287→    case 'last_7_days': {
   288→      const from = new Date(today)
   289→      from.setDate(from.getDate() - 6)
   290→      return { from: toISODate(from), to: toISODate(today) }
   291→    }
   292→    case 'last_30_days': {
   293→      const from = new Date(today)
   294→      from.setDate(from.getDate() - 29)
   295→      return { from: toISODate(from), to: toISODate(today) }
   296→    }
   297→    case 'this_month': {
   298→      const from = new Date(now.getFullYear(), now.getMonth(), 1)
   299→      return { from: toISODate(from), to: toISODate(today) }
   300→    }
   301→    case 'last_month': {
   302→      const firstOfThis = new Date(now.getFullYear(), now.getMonth(), 1)
   303→      const from = new Date(firstOfThis)
   304→      from.setMonth(from.getMonth() - 1)
   305→      const to = new Date(firstOfThis.getFullYear(), firstOfThis.getMonth(), 0)
   306→      return { from: toISODate(from), to: toISODate(to) }
   307→    }
   308→    case 'this_year': {
   309→      const from = new Date(now.getFullYear(), 0, 1)
   310→      return { from: toISODate(from), to: toISODate(today) }
   311→    }
   312→    case 'custom':
   313→      return {
   314→        from: custom?.from ?? toISODate(today),
   315→        to: custom?.to ?? toISODate(today),
   316→      }
   317→  }
   318→}

   320→function spanDayCount(range: DateRange): number {
   321→  const from = new Date(range.from + 'T00:00:00')
   322→  const to = new Date(range.to + 'T23:59:59')
   323→  const diff = to.getTime() - from.getTime()
   324→  return Math.max(1, Math.round(diff / (1000 * 60 * 60 * 24)) + 1)
   325→}
   326→
   327→function isMonthlyGranularity(range: DateRange): boolean {
   328→  return spanDayCount(range) > 30
   329→}

   439→function toggleArray(arr: string[], value: string): string[] {
   440→  return arr.includes(value) ? arr.filter((v) => v !== value) : [...arr, value]

   443→function countActiveFilters(f: FilterState): number {
   444→  let n = 0
   445→  if (f.statuses.length) n++
   446→  if (f.paymentTypes.length) n++
   447→  if (f.paymentStatuses.length) n++
   448→  if (f.orderSources.length) n++
   449→  if (f.dateFrom || f.dateTo) n++

   454→  if (f.courier.trim()) n++
   455→  return n

   471→function applyPreset(preset: DatePresetValue): { from: string; to: string } {
   472→  const now = new Date()
   473→  const to = new Date(now)
   474→  const from = new Date(now)
   475→  if (preset === 'today') {
   476→    // from = today
   477→  } else if (preset === '7d') {
   478→    from.setDate(now.getDate() - 6)
   479→  } else if (preset === '30d') {
   480→    from.setDate(now.getDate() - 29)
   481→  } else if (preset === 'this_month') {
   482→    from.setDate(1)
   483→  }
   484→  return { from: toISODate(from), to: toISODate(to) }
   485→}

  1245→function buildChartData(
  1246→  orders: OrderRow[],
  1247→  range: DateRange,
  1248→  monthly: boolean,
  1249→  metric: ChartMetric,
  1250→): ChartPoint[] {
  1251→  // Build the list of interval buckets
  1252→  const buckets = buildIntervalBuckets(range, monthly)
  1253→
  1254→  // For each order, compute its bucket key (yyyy-mm-dd or yyyy-mm)
  1255→  for (const o of orders) {
  1256→    const created = new Date(o.createdAt)
  1257→    if (Number.isNaN(created.getTime())) continue
  1258→    const key = monthly
  1259→      ? `${created.getFullYear()}-${String(created.getMonth() + 1).padStart(2, '0')}`
  1260→      : toISODate(created)
  1261→    const b = buckets.find((x) => x.key === key)
  1262→    if (!b) continue
  1263→    b.total += 1
  1264→    if (o.status === 'rto') b.rto += 1
  1265→    if (o.status === 'cancelled') b.cancelled += 1
  1266→    if (o.status !== 'cancelled') b.revenue += o.totalOrderValue
  1267→    if (o.status === 'pending') b.pending += 1
  1268→    if (o.status === 'partially_backordered') b.backordered += 1
  1269→  }
  1270→
  1271→  return buckets.map((b) => {
  1272→    let value = 0
  1273→    let count: number | undefined
  1274→    let total: number | undefined
  1275→    switch (metric) {
  1276→      case 'total_orders':
  1277→        value = b.total
  1278→        break
  1279→      case 'revenue':
  1280→        value = b.revenue
  1281→        break
  1282→      case 'rto_rate':
  1283→        value = b.total > 0 ? (b.rto / b.total) * 100 : 0
  1284→        count = b.rto
  1285→        total = b.total
  1286→        break
  1287→      case 'cancellation_rate':
  1288→        value = b.total > 0 ? (b.cancelled / b.total) * 100 : 0
  1289→        count = b.cancelled
  1290→        total = b.total
  1291→        break
  1292→      case 'pending_current':
  1293→        value = b.pending
  1294→        break
  1295→      case 'backordered_current':
  1296→        value = b.backordered
  1297→        break
  1298→    }
  1299→    return { label: b.label, value, count, total }
  1300→  })
  1301→}

  1303→interface Bucket {
  1304→  key: string
  1305→  label: string
  1306→  total: number
  1307→  revenue: number
  1308→  rto: number
  1309→  cancelled: number
  1310→  pending: number
  1311→  backordered: number
  1312→}

  1314→function buildIntervalBuckets(range: DateRange, monthly: boolean): Bucket[] {
  1315→  const out: Bucket[] = []
  1316→  const start = new Date(range.from + 'T00:00:00')
  1317→  const end = new Date(range.to + 'T23:59:59')
  1318→
  1319→  if (monthly) {
  1320→    // Iterate by month from start month to end month (inclusive)
  1321→    const cursor = new Date(start.getFullYear(), start.getMonth(), 1)
  1322→    const endMonth = new Date(end.getFullYear(), end.getMonth(), 1)
  1323→    while (cursor <= endMonth) {
  1324→      const key = `${cursor.getFullYear()}-${String(cursor.getMonth() + 1).padStart(2, '0')}`
  1325→      const label = cursor.toLocaleDateString('en-PK', { month: 'short', year: '2-digit' })
  1326→      out.push({ key, label, total: 0, revenue: 0, rto: 0, cancelled: 0, pending: 0, backordered: 0 })
  1327→      cursor.setMonth(cursor.getMonth() + 1)
  1328→    }
  1329→  } else {
  1330→    // Iterate by day from start to end (inclusive)
  1331→    const cursor = new Date(start.getFullYear(), start.getMonth(), start.getDate())
  1332→    const endDay = new Date(end.getFullYear(), end.getMonth(), end.getDate())
  1333→    while (cursor <= endDay) {
  1334→      const key = toISODate(cursor)
  1335→      const label = cursor.toLocaleDateString('en-PK', { day: '2-digit', month: 'short' })
  1336→      out.push({ key, label, total: 0, revenue: 0, rto: 0, cancelled: 0, pending: 0, backordered: 0 })
  1337→      cursor.setDate(cursor.getDate() + 1)
  1338→    }
  1339→  }
  1340→  return out
  1341→}
