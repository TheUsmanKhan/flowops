    41→// ──────────────────────────────────────────────────────────────
    42→// Phone format conversion helper
    43→// ──────────────────────────────────────────────────────────────
    44→
    45→/**
    46→ * Convert a Pakistani phone number to PostEx's required format: 03XXXXXXXXX.
    47→ *
    48→ * PostEx's API requires customerPhone in format "03xxxxxxxxx" (11 digits,
    49→ * starting with 03). Our CRM stores phones in E.164 format (+92XXXXXXXXXX).
    50→ *
    51→ * Conversion rules:
    52→ *   +92 3XX XXXXXXX → 03XX XXXXXXX  (strip +92, prepend 0)
    53→ *   92 3XX XXXXXXX  → 03XX XXXXXXX  (strip 92, prepend 0)
    54→ *   03XXXXXXXXX     → 03XXXXXXXXX  (already correct)
    55→ *   3XXXXXXXXX      → 03XXXXXXXXX  (prepend 0)
    56→ *   Anything else   → return as-is (let PostEx validate/reject)
    57→ */
    58→function convertToPostExPhone(phone: string): string {
    59→  const trimmed = phone.trim()
    60→
    61→  // Already in 03XXXXXXXXX format
    62→  if (/^03\d{9}$/.test(trimmed)) {
    63→    return trimmed
    64→  }
    65→
    66→  // +92 3XX... → 03XX...
    67→  if (trimmed.startsWith('+92')) {
    68→    const rest = trimmed.substring(3) // remove +92
    69→    if (rest.startsWith('3')) {
    70→      return '0' + rest
    71→    }
    72→    return '0' + rest // fallback
    73→  }
    74→
    75→  // 92 3XX... → 03XX...
    76→  if (trimmed.startsWith('92') && trimmed.length >= 11) {
    77→    const rest = trimmed.substring(2) // remove 92
    78→    if (rest.startsWith('3')) {
    79→      return '0' + rest
    80→    }
    81→    return '0' + rest
    82→  }
    83→
    84→  // 3XX... (missing leading 0) → 03XX...
    85→  if (/^3\d{9}$/.test(trimmed)) {
    86→    return '0' + trimmed
    87→  }
    88→
    89→  // Unknown format — return as-is, let PostEx validate
    90→  return trimmed
    91→}

   158→  // ──────────────────────────────────────────────────────────────

   160→  async bookShipment(input: BookShipmentInput): Promise<BookShipmentResult> {
   161→    // NOTE: City validation is handled by the CALLER (booking-workbench/book
   162→    // route or the auto-booking server action) via revalidateCityAtBookingTime()
   163→    // with the companyIntegrationId — which enables the live PostEx fallback.
   164→    // The adapter doesn't have the integration ID, so it can't do the live
   165→    // fallback. We trust the caller's pre-validation.
   166→
   167→    // Build the request body — ONLY fields that exist in the confirmed API.
   168→    // NO weight/handling/itemsQty/paymentMethod/orderTags fields.
   169→    const body: Record<string, unknown> = {
   170→      cityName: input.deliveryCity,
   171→      customerName: input.recipientName,
   172→      customerPhone: convertToPostExPhone(input.recipientPhone),
   173→      deliveryAddress: input.deliveryAddress,
   174→      invoiceDivision: 1, // default 1 (split airway bills for larger orders)
   175→      invoicePayment: input.codAmount,
   176→      items: input.quantity ?? 1,
   177→      orderDetail: input.itemDescription || 'Order',
   178→      orderRefNumber: input.orderNumber,
   179→      orderType: input.orderType ?? 'Normal',
   180→      transactionNotes: input.transactionNotes ?? '',
   181→      pickupAddressCode: input.pickupAddressCode ?? '',
   182→    }
   183→    // NOTE: storeAddressCode is intentionally OMITTED. Sending it (even as
   184→    // empty string) causes PostEx to validate it and reject with
   185→    // "INVALID MERCHANT STORE ADDRESS CODE". PostEx uses its default store
   186→    // address when the field is absent.
   187→
   188→    const response = await fetch(`${POSTEX_BASE_URL}/v3/create-order`, {
   189→      method: 'POST',
   190→      headers: {
   191→        'Content-Type': 'application/json',
   192→        token: this.token,
   193→      },
   194→      body: JSON.stringify(body),
   195→    })
   196→
   197→    const json: PostExApiResponse<PostExCreateOrderDist> = await response.json()
   198→
   199→    if (json.statusCode === 200 || json.statusCode === '200') {
   200→      const trackingNumber = json.dist?.trackingNumber
   201→      if (!trackingNumber) {
   202→        return {
   203→          success: false,
   204→          error: 'PostEx returned success but no tracking number in response.',
   205→          rawResponse: json,
   206→        }
   207→      }
   208→
   209→      return {
   210→        success: true,
   211→        trackingNumber,
   212→        providerStatus: json.dist?.orderStatus ?? 'Unbooked',
   213→        rawResponse: json,
   214→      }
   215→    }
   216→
   217→    return {
   218→      success: false,
   219→      error: json.statusMessage || `PostEx API returned statusCode ${json.statusCode}`,
   220→      rawResponse: json,
   221→    }
   222→  }

   225→  // 2. trackShipment — GET v1/track-order/{trackingNumber}
   226→  // ──────────────────────────────────────────────────────────────
   227→
   228→  async trackShipment(trackingNumber: string): Promise<TrackShipmentResult> {
   229→    const response = await fetch(
   230→      `${POSTEX_BASE_URL}/v1/track-order/${encodeURIComponent(trackingNumber)}`,
   231→      {
   232→        method: 'GET',
   233→        headers: {
   234→          token: this.token,
   235→        },
   236→      },
   237→    )
   238→
   239→    const json: PostExApiResponse<PostExTrackOrderDist> = await response.json()
   240→
   241→    if (json.statusCode === 200 || json.statusCode === '200') {
   242→      const dist = json.dist
   243→      if (!dist) {
   244→        return {
   245→          success: false,
   246→          error: 'PostEx returned success but no dist in response.',
   247→          rawResponse: json,
   248→        }
   249→      }
   250→
   251→      const postexStatus = dist.transactionStatus || 'Unbooked'
   252→      const mapped = mapPostExStatus(postexStatus)
   253→
   254→      // Map to the generic TrackShipmentResult status enum
   255→      let genericStatus: TrackShipmentResult['status'] = 'booked'
   256→      if (mapped.triggerDelivered) genericStatus = 'delivered'
   257→      else if (mapped.triggerRto) genericStatus = 'returned'
   258→      else if (mapped.triggerDispatch) genericStatus = 'in_transit'
   259→      else if (mapped.orderStatus === 'cancelled') genericStatus = 'failed' // reuse 'failed' for cancelled-by-merchant/expired
   260→
   261→      return {
   262→        success: true,
   263→        status: genericStatus,
   264→        lastUpdateAt: (dist.transactionDate as string) ?? undefined,
   265→        rawResponse: {
   266→          ...dist, // includes transactionStatusHistory for audit/display
   267→          mappedSubStatus: mapped.courierSubStatus,
   268→          needsShipperAdvice: mapped.needsShipperAdvice,
   269→          unrecognized: mapped.unrecognized,
   270→        },
   271→      }
   272→    }
   273→
   274→    return {
   275→      success: false,
   276→      error: json.statusMessage || `PostEx API returned statusCode ${json.statusCode}`,
   277→      rawResponse: json,
   278→    }
   279→  }

   281→  // ──────────────────────────────────────────────────────────────
   282→  // 3. trackBulkShipments — GET v1/track-bulk-order
   283→  // ──────────────────────────────────────────────────────────────
   284→
   285→  async trackBulkShipments(trackingNumbers: string[]): Promise<TrackShipmentResult[]> {
   286→    if (trackingNumbers.length === 0) return []
   287→
   288→    // Chunk into groups of 50 (default — PostEx docs don't specify a limit,
   289→    // but 50 is a reasonable batch size to avoid URL length issues)
   290→    const CHUNK_SIZE = 50
   291→    const results: TrackShipmentResult[] = []
   292→
   293→    for (let i = 0; i < trackingNumbers.length; i += CHUNK_SIZE) {
   294→      const chunk = trackingNumbers.slice(i, i + CHUNK_SIZE)
   295→
   296→      // PostEx's bulk tracking API uses GET with query params.
   297→      // The parameter name is "TrackingNumbers" (capital T, capital N)
   298→      // — confirmed from the live API error message:
   299→      // "Required List parameter 'TrackingNumbers' is not present"
   300→      const queryString = chunk.map((t) => `TrackingNumbers=${encodeURIComponent(t)}`).join('&')
   301→
   302→      const response = await fetch(
   303→        `${POSTEX_BASE_URL}/v1/track-bulk-order?${queryString}`,
   304→        {
   305→          method: 'GET',
   306→          headers: {
   307→            token: this.token,
   308→          },
   309→        },
   310→      )
   311→
   312→      const json: PostExApiResponse<PostExBulkTrackItem[]> = await response.json()
   313→
   314→      if (json.statusCode === 200 || json.statusCode === '200') {
   315→        const dist = json.dist ?? []
   316→        for (const item of dist) {
   317→          if (!item.trackingResponse) {
   318→            results.push({
   319→              success: false,
   320→              error: item.message || 'No tracking response for this number.',
   321→              rawResponse: item,
   322→            })
   323→            continue
   324→          }
   325→
   326→          const postexStatus = item.trackingResponse.transactionStatus || 'Unbooked'
   327→          const mapped = mapPostExStatus(postexStatus)
   328→
   329→          let genericStatus: TrackShipmentResult['status'] = 'booked'
   330→          if (mapped.triggerDelivered) genericStatus = 'delivered'
   331→          else if (mapped.triggerRto) genericStatus = 'returned'
   332→          else if (mapped.triggerDispatch) genericStatus = 'in_transit'
   333→          else if (mapped.orderStatus === 'cancelled') genericStatus = 'failed' // cancelled-by-merchant/expired
   334→
   335→          results.push({
   336→            success: true,
   337→            status: genericStatus,
   338→            lastUpdateAt: (item.trackingResponse.transactionDate as string) ?? undefined,
   339→            rawResponse: {
   340→              ...item.trackingResponse,
   341→              mappedSubStatus: mapped.courierSubStatus,
   342→              needsShipperAdvice: mapped.needsShipperAdvice,
   343→              unrecognized: mapped.unrecognized,
   344→            },
   345→          })
   346→        }
   347→      } else {
   348→        // API error for this chunk — add error results for all tracking numbers in the chunk
   349→        for (const _tn of chunk) {
   350→          results.push({
   351→            success: false,
   352→            error: json.statusMessage || `PostEx bulk API returned statusCode ${json.statusCode}`,
   353→            rawResponse: json,
   354→          })
   355→        }
   356→      }
   357→    }
   358→
   359→    return results
   360→  }

   366→  async cancelShipment(trackingNumber: string): Promise<CancelShipmentResult> {
   367→    // NOTE: Only meaningful while PostEx status is Unbooked or Booked (per
   368→    // confirmed cancel-window rule). The CALLER is responsible for checking
   369→    // this before invoking — this adapter is a thin API wrapper.
   370→    const response = await fetch(`${POSTEX_BASE_URL}/v1/cancel-order`, {
   371→      method: 'PUT',
   372→      headers: {
   373→        'Content-Type': 'application/json',
   374→        token: this.token,
   375→      },
   376→      body: JSON.stringify({ trackingNumber }),
   377→    })
   378→
   379→    if (response.status === 200) {
   380→      return { success: true }
   381→    }
   382→
   383→    if (response.status === 404) {
   384→      return { success: false, error: 'Order not found on PostEx.' }
   385→    }
   386→
   387→    let errorMessage = `PostEx cancel API returned HTTP ${response.status}`
   388→    try {
   389→      const json = await response.json()
   390→      errorMessage = json.statusMessage || errorMessage
   391→    } catch {
   392→      // Response body not JSON — use the HTTP status message
   393→    }
   394→
   395→    return { success: false, error: errorMessage }
   396→  }

   424→  // ──────────────────────────────────────────────────────────────

   426→  /**
   427→   * Fetch ALL operational cities from PostEx.
   428→   *
   429→   * PostEx's v2/get-operational-city endpoint accepts an optional
   430→   * `operationalCityType` query param (Pickup | Delivery | null), but in
   431→   * practice PostEx's API returns errors ("statusCode undefined") when this
   432→   * param is set. Only the unfiltered call (no param) works reliably.
   433→   *
   434→   * We make a SINGLE unfiltered call with a 30-second timeout. This returns
   435→   * all cities with their isPickupCity/isDeliveryCity flags already set.
   436→   */
   437→  async fetchOperationalCities(): Promise<OperationalCity[]> {
   438→    const url = `${POSTEX_BASE_URL}/v2/get-operational-city`
   439→
   440→    const controller = new AbortController()
   441→    const timeout = setTimeout(() => controller.abort(), 30_000)
   442→    try {
   443→      const response = await fetch(url, {
   444→        method: 'GET',
   445→        headers: { token: this.token },
   446→        signal: controller.signal,
   447→      })
   448→      const json: PostExApiResponse<PostExOperationalCity[]> = await response.json()
   449→
   450→      if (json.statusCode === 200 || json.statusCode === '200') {
   451→        const dist = json.dist ?? []
   452→        if (dist.length === 0) {
   453→          throw new Error('PostEx cities API returned 0 cities. Check the integration token.')
   454→        }
   455→        return dist.map((c) => ({
   456→          cityName: c.operationalCityName,
   457→          cityId: undefined,
   458→          isPickupCity: c.isPickupCity === true || c.isPickupCity === 'true',
   459→          isDeliveryCity: c.isDeliveryCity === true || c.isDeliveryCity === 'true',
   460→        }))
   461→      }
   462→
   463→      throw new Error(
   464→        json.statusMessage || `PostEx cities API returned statusCode ${json.statusCode}`,
   465→      )
   466→    } finally {
   467→      clearTimeout(timeout)
   468→    }
   469→  }

   506→    }

   599→  }
   600→
   601→  // ──────────────────────────────────────────────────────────────
   602→  // 10. generateLoadSheet — POST v2/generate-load-sheet
   603→  // ──────────────────────────────────────────────────────────────
   604→
   605→  async generateLoadSheet(
   606→    trackingNumbers: string[],
   607→    pickupAddress?: string,
   608→  ): Promise<{ success: boolean; rawResponse?: unknown; error?: string; pdfBase64?: string }> {
   609→    if (trackingNumbers.length === 0) {
   610→      return { success: false, error: 'No tracking numbers provided.' }
   611→    }
   612→
   613→    const body = {
   614→      pickupAddress: pickupAddress ?? '',
   615→      trackingNumbers,
   616→    }
   617→
   618→    const response = await fetch(
   619→      `${POSTEX_BASE_URL}/v2/generate-load-sheet`,
   620→      {
   621→        method: 'POST',
   622→        headers: {
   623→          'Content-Type': 'application/json',
   624→          token: this.token,
   625→        },
   626→        body: JSON.stringify(body),
   627→      },
   628→    )
   629→
   630→    // PostEx returns a PDF file for this endpoint, not JSON.
   631→    if (response.status === 200) {
   632→      // The response is a PDF binary — capture it as base64 so the caller
   633→      // can store it in our own file storage (not an external courier URL
   634→      // that might expire — same principle as Proof of Delivery).
   635→      const arrayBuffer = await response.arrayBuffer()
   636→      const buffer = Buffer.from(arrayBuffer)
   637→      const pdfBase64 = buffer.toString('base64')
   638→
   639→      return {
   640→        success: true,
   641→        pdfBase64,
   642→        rawResponse: {
   643→          contentType: response.headers.get('content-type'),
   644→          contentLength: response.headers.get('content-length'),
   645→          status: response.status,
   646→          pdfSizeBytes: buffer.length,
   647→        },
   648→      }
   649→    }
   650→
   651→    // Non-200: try to parse as JSON for the error message
   652→    let errorMsg = `PostEx load-sheet API returned HTTP ${response.status}`
   653→    try {
   654→      const json = await response.json()
   655→      errorMsg = json.statusMessage || json.message || errorMsg
   656→    } catch {
   657→      // Not JSON — use the HTTP status error
   658→    }
   659→
   660→    return {
   661→      success: false,
   662→      error: errorMsg,
   663→    }
   664→  }
   665→
   666→  // ──────────────────────────────────────────────────────────────
   667→  // 11. fetchPaymentStatus — GET v1/payment-status/{trackingNumber}
   668→  // ──────────────────────────────────────────────────────────────
   669→
   670→  /**
   671→   * Fetch the payment/settlement status for a tracking number.
   672→   *
   673→   * PostEx's Payment Status API response:
   674→   *   { statusCode, statusMessage, dist: { orderRefNumber, trackingNumber,
   675→   *     settle (boolean), settlementDate, upfrontPaymentDate,
   676→   *     cprNumber_1, reservePaymentDate, cprNumber_2 } }
   677→   *
   678→   * IMPORTANT: PostEx's Payment Status API does NOT break out delivery charge
   679→   * as a separate field. It only provides settlement status (boolean), dates,
   680→   * and CPR numbers. The `actualDeliveryCharge` field on Order/exchange_shipments
   681→   * CANNOT be auto-populated from this API — it would need to come from a
   682→   * different source (e.g. manual entry or a reconciliation report).
   683→   */
   684→  async fetchPaymentStatus(trackingNumber: string): Promise<{
   685→    success: boolean
   686→    settled: boolean
   687→    settlementDate: string | null
   688→    upfrontPaymentDate: string | null
   689→    cprNumber1: string | null
   690→    cprNumber2: string | null
   691→    error?: string
   692→  }> {
   693→    const response = await fetch(
   694→      `${POSTEX_BASE_URL}/v1/payment-status/${encodeURIComponent(trackingNumber)}`,
   695→      {
   696→        method: 'GET',
   697→        headers: {
   698→          token: this.token,
   699→        },
   700→      },
   701→    )
   702→
   703→    if (response.status === 404) {
   704→      return { success: false, settled: false, settlementDate: null, upfrontPaymentDate: null, cprNumber1: null, cprNumber2: null, error: 'Order not found on PostEx.' }
   705→    }
   706→
   707→    const json: PostExApiResponse<{
   708→      orderRefNumber: string
   709→      trackingNumber: string
   710→      settle: boolean
   711→      settlementDate: string
   712→      upfrontPaymentDate: string
   713→      cprNumber_1: string
   714→      reservePaymentDate: string
   715→      cprNumber_2: string
   716→    }> = await response.json()
   717→
   718→    if (json.statusCode === 200 || json.statusCode === '200') {
   719→      const dist = json.dist
   720→      return {
   721→        success: true,
   722→        settled: dist?.settle ?? false,
   723→        settlementDate: dist?.settlementDate ?? null,
   724→        upfrontPaymentDate: dist?.upfrontPaymentDate ?? null,
   725→        cprNumber1: dist?.cprNumber_1 ?? null,
   726→        cprNumber2: dist?.cprNumber_2 ?? null,
   727→      }
   728→    }
   729→
   730→    return {
   731→      success: false,
   732→      settled: false,
   733→      settlementDate: null,
   734→      upfrontPaymentDate: null,
   735→      cprNumber1: null,
   736→      cprNumber2: null,
   737→      error: json.statusMessage || `PostEx payment-status API returned statusCode ${json.statusCode}`,
   738→    }
   739→  }
   740→}
