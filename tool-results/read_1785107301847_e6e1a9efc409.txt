   141→   141→function parseQty(v: string): number {
   142→   142→  if (v === '') return 0
   143→   143→  const n = Number(v)
   144→   144→  return Number.isFinite(n) && n >= 0 ? Math.floor(n) : 0
   145→   145→}
   146→   146→
   147→   147→function parsePrice(v: string): number {
   148→   148→  if (v === '') return 0
   149→   149→  const n = Number(v)
   150→   150→  return Number.isFinite(n) && n >= 0 ? n : 0
   151→   151→}

   174→   174→export function OrderCreateView() {
   175→   175→  const navigate = useAppStore((s) => s.navigate)
   176→   176→  const can = useCan()
   177→   177→  const queryClient = useQueryClient()
   178→   178→  const canCreate = can(PERMISSIONS.ORDERS_CREATE)
   179→   179→
   180→   180→  const [step, setStep] = useState(1)
   181→   181→
   182→   182→  // ── Form state ────────────────────────────────────────────────────────────
   183→   183→  // Customer
   184→   184→  const [customerMode, setCustomerMode] = useState<'existing' | 'new'>('existing')
   185→   185→  const [phoneSearch, setPhoneSearch] = useState('')
   186→   186→  const [selectedCustomer, setSelectedCustomer] = useState<CustomerRow | null>(null)
   187→   187→  const [newCustomer, setNewCustomer] = useState({
   188→   188→    name: '',
   189→   189→    phone: '',
   190→   190→    alternate_phone: '',
   191→   191→    email: '',
   192→   192→    address: '',
   193→   193→    city: '',
   194→   194→  })
   195→   195→
   196→   196→  // Items
   197→   197→  const [cart, setCart] = useState<CartItem[]>([])
   198→   198→  const [variantSearch, setVariantSearch] = useState('')
   199→   199→
   200→   200→  // Payment
   201→   201→  const [paymentType, setPaymentType] = useState<PaymentType>('full_cod')
   202→   202→  const [advanceAmount, setAdvanceAmount] = useState('')
   203→   203→  const [advancePaymentMethod, setAdvancePaymentMethod] = useState('')
   204→   204→  const [advancePaymentReference, setAdvancePaymentReference] = useState('')
   205→   205→  const [advancePaymentScreenshotUrl, setAdvancePaymentScreenshotUrl] = useState('')
   206→   206→
   207→   207→  // Delivery
   208→   208→  const [deliveryAddress, setDeliveryAddress] = useState('')
   209→   209→  const [deliveryCity, setDeliveryCity] = useState('')
   210→   210→  const [courierName, setCourierName] = useState('')
   211→   211→  const [dispatchLocationId, setDispatchLocationId] = useState('')
   212→   212→  const [notesForCourier, setNotesForCourier] = useState('')
   213→   213→  const [discountAmount, setDiscountAmount] = useState('')
   214→   214→  const [discountReason, setDiscountReason] = useState('')
   215→   215→
   216→   216→  // ── Data queries ──────────────────────────────────────────────────────────
   217→   217→  const locationsQuery = useQuery<LocationsResponse>({
   218→   218→    queryKey: ['locations'],
   219→   219→    queryFn: () => api.get<LocationsResponse>('/api/inventory-locations'),
   220→   220→    staleTime: 30_000,
   221→   221→  })
   222→   222→
   223→   223→  const productsQuery = useQuery<ProductsResponse>({
   224→   224→    queryKey: ['products', 'for-order-create'],
   225→   225→    queryFn: () => api.get<ProductsResponse>('/api/products?pageSize=100'),
   226→   226→    staleTime: 60_000,
   227→   227→  })
   228→   228→
   229→   229→  // Customer phone search — debounced via staleTime + query key
   230→   230→  const trimmedPhone = phoneSearch.trim()
   231→   231→  const customersQuery = useQuery<CustomersSearchResponse>({
   232→   232→    queryKey: ['customers', 'search', trimmedPhone],
   233→   233→    queryFn: () =>
   234→   234→      api.get<CustomersSearchResponse>(
   235→   235→        `/api/customers?search=${encodeURIComponent(trimmedPhone)}&limit=10`,
   236→   236→      ),
   237→   237→    enabled: trimmedPhone.length >= 4,
   238→   238→    staleTime: 10_000,
   239→   239→  })
   240→   240→
   241→   241→  // ── Variant options (flatten products → variants) ─────────────────────────
   242→   242→  const variantOptions: VariantOption[] = useMemo(() => {
   243→   243→    const list: VariantOption[] = []
   244→   244→    for (const p of productsQuery.data?.products ?? []) {
   245→   245→      for (const v of p.variants) {
   246→   246→        list.push({
   247→   247→          variantId: v.id,
   248→   248→          sku: v.sku,
   249→   249→          productTitle: p.title,
   250→   250→          costPrice: v.costPrice,
   251→   251→          salePrice: v.salePrice,
   252→   252→          fulfillmentType: v.fulfillmentType,
   253→   253→        })
   254→   254→      }
   255→   255→    }
   256→   256→    return list
   257→   257→  }, [productsQuery.data])
   258→   258→
   259→   259→  const variantSearchResults = useMemo(() => {
   260→   260→    if (!variantSearch.trim()) return []
   261→   261→    const q = variantSearch.toLowerCase()
   262→   262→    return variantOptions
   263→   263→      .filter(
   264→   264→        (v) => v.sku.toLowerCase().includes(q) || v.productTitle.toLowerCase().includes(q),
   265→   265→      )
   266→   266→      .slice(0, 8)
   267→   267→  }, [variantOptions, variantSearch])
   268→   268→
   269→   269→  // ── Computed totals ───────────────────────────────────────────────────────
   270→   270→  const subtotal = useMemo(
   271→   271→    () => cart.reduce((s, i) => s + i.unitPrice * i.quantity, 0),
   272→   272→    [cart],
   273→   273→  )
   274→   274→  const discount = parsePrice(discountAmount)
   275→   275→  const total = Math.max(0, subtotal - discount)
   276→   276→
   277→   277→  // ── Auto-fill delivery from customer ──────────────────────────────────────
   278→   278→  useEffect(() => {
   279→   279→    if (customerMode === 'existing' && selectedCustomer) {
   280→   280→      // No addresses returned by customer search — let user type.
   281→   281→    }
   282→   282→  }, [customerMode, selectedCustomer])
   283→   283→
   284→   284→  // When switching to "new" customer, pre-fill delivery fields from their input.
   285→   285→  useEffect(() => {
   286→   286→    if (customerMode === 'new') {
   287→   287→      if (newCustomer.address && !deliveryAddress) {
   288→   288→        setDeliveryAddress(newCustomer.address)
   289→   289→      }
   290→   290→      if (newCustomer.city && !deliveryCity) {
   291→   291→        setDeliveryCity(newCustomer.city)
   292→   292→      }
   293→   293→      if (newCustomer.phone && !courierName) {
   294→   294→        // leave courier blank — not derivable
   295→   295→      }
   296→   296→    }
   297→   297→  }, [customerMode, newCustomer.address, newCustomer.city, newCustomer.phone, courierName, deliveryAddress, deliveryCity])
   298→   298→
   299→   299→  // ── Auto-select default dispatch location ─────────────────────────────────
   300→   300→  useEffect(() => {
   301→   301→    if (!dispatchLocationId && locationsQuery.data?.locations.length) {
   302→   302→      const def = locationsQuery.data.locations.find((l) => l.isDefault)
   303→   303→      setDispatchLocationId(def?.id ?? locationsQuery.data.locations[0].id)
   304→   304→    }
   305→   305→  }, [dispatchLocationId, locationsQuery.data])
   306→   306→
   307→   307→  // ── Item manipulation ─────────────────────────────────────────────────────
   308→   308→  const addVariant = (v: VariantOption) => {
   309→   309→    setCart((prev) => {
   310→   310→      if (prev.some((i) => i.variantId === v.variantId)) {
   311→   311→        toast.info('That variant is already in the cart.')
   312→   312→        return prev
   313→   313→      }
   314→   314→      const unitPrice = v.salePrice ?? v.costPrice
   315→   315→      return [
   316→   316→        ...prev,
   317→   317→        {
   318→   318→          variantId: v.variantId,
   319→   319→          sku: v.sku,
   320→   320→          productTitle: v.productTitle,
   321→   321→          unitPrice,
   322→   322→          quantity: 1,
   323→   323→          fulfillmentType: v.fulfillmentType,
   324→   324→        },
   325→   325→      ]
   326→   326→    })
   327→   327→    setVariantSearch('')
   328→   328→  }
   329→   329→
   330→   330→  const removeItem = (variantId: string) => {
   331→   331→    setCart((prev) => prev.filter((i) => i.variantId !== variantId))
   332→   332→  }
   333→   333→
   334→   334→  const updateItem = (variantId: string, patch: Partial<CartItem>) => {
   335→   335→    setCart((prev) =>
   336→   336→      prev.map((i) => (i.variantId === variantId ? { ...i, ...patch } : i)),
   337→   337→    )
   338→   338→  }
   339→   339→
   340→   340→  // ── Step validation ───────────────────────────────────────────────────────
   341→   341→  const customerStepValid =
   342→   342→    customerMode === 'existing'
   343→   343→      ? !!selectedCustomer
   344→   344→      : newCustomer.name.trim().length >= 2 &&
   345→   345→        newCustomer.phone.trim().length >= 7 &&
   346→   346→        newCustomer.address.trim().length >= 2 &&
   347→   347→        newCustomer.city.trim().length >= 2
   348→   348→
   349→   349→  const itemsStepValid = cart.length > 0 && cart.every((i) => i.quantity > 0 && i.unitPrice >= 0)
   350→   350→
   351→   351→  const paymentStepValid =
   352→   352→    paymentType === 'full_cod' ||
   353→   353→    (paymentType === 'partial_advance' && parsePrice(advanceAmount) > 0) ||
   354→   354→    paymentType === 'fully_prepaid'
   355→   355→
   356→   356→  const deliveryStepValid =
   357→   357→    deliveryAddress.trim().length >= 2 &&
   358→   358→    deliveryCity.trim().length >= 2 &&
   359→   359→    !!dispatchLocationId
   360→   360→
   361→   361→  const canGoNext = (): boolean => {
   362→   362→    if (step === 1) return customerStepValid
   363→   363→    if (step === 2) return itemsStepValid
   364→   364→    if (step === 3) return paymentStepValid
   365→   365→    if (step === 4) return deliveryStepValid
   366→   366→    return true
   367→   367→  }
   368→   368→
   369→   369→  // ── Mutation: create order ────────────────────────────────────────────────
   370→   370→  const createMutation = useMutation({
   371→   371→    mutationFn: async (payload: unknown) =>
   372→   372→      api.post<CreateOrderResponse>('/api/orders', payload),
   373→   373→    onSuccess: (data) => {
   374→   374→      toast.success(`Order ${data.flowopsOrderNumber} created successfully.`)
   375→   375→      void queryClient.invalidateQueries({ queryKey: ['orders'] })
   376→   376→      navigate({ name: 'order-detail', id: data.orderId })
   377→   377→    },
   378→   378→    onError: (err) => toast.error(getErrorMessage(err)),
   379→   379→  })
   380→   380→
   381→   381→  // ── Permission gate (after all hooks) ─────────────────────────────────────
   382→   382→  if (!canCreate) {
   383→   383→    return (
   384→   384→      <div className="space-y-6">
   385→   385→        <PageHeader
   386→   386→          title="Create Order"
   387→   387→          description="Manually create a customer order"
   388→   388→          actions={
   389→   389→            <Button variant="outline" size="sm" onClick={() => navigate({ name: 'orders' })}>
   390→   390→              <ArrowLeft className="h-4 w-4" /> Back to Orders
   391→   391→            </Button>
   392→   392→          }
   393→   393→        />
   394→   394→        <Card>
   395→   395→          <CardContent className="p-10 text-center">
   396→   396→            <AlertCircle className="h-10 w-10 mx-auto text-muted-foreground/40 mb-3" />
   397→   397→            <p className="text-sm text-muted-foreground">
   398→   398→              You don&apos;t have permission to create orders.
   399→   399→            </p>
   400→   400→          </CardContent>
   401→   401→        </Card>
   402→   402→      </div>
   403→   403→    )
   404→   404→  }
   405→   405→
   406→   406→  // ── Build payload + submit ────────────────────────────────────────────────
   407→   407→  const handleSubmit = () => {
   408→   408→    if (!customerStepValid || !itemsStepValid || !paymentStepValid || !deliveryStepValid) {
   409→   409→      toast.error('Please complete all required fields before creating the order.')
   410→   410→      return
   411→   411→    }
   412→   412→
   413→   413→    const payload: Record<string, unknown> = {
   414→   414→      items: cart.map((i) => ({
   415→   415→        org_variant_id: i.variantId,
   416→   416→        quantity: i.quantity,
   417→   417→        unit_price: i.unitPrice,
   418→   418→      })),
   419→   419→      payment_type: paymentType,
   420→   420→      delivery_address: deliveryAddress.trim(),
   421→   421→      delivery_city: deliveryCity.trim(),
   422→   422→      courier_name: courierName.trim() || undefined,
   423→   423→      dispatch_location_id: dispatchLocationId,
   424→   424→      notes_for_courier: notesForCourier.trim() || undefined,
   425→   425→      discount_amount: discount > 0 ? discount : undefined,
   426→   426→      discount_reason: discountReason.trim() || undefined,
   427→   427→    }
   428→   428→
   429→   429→    if (paymentType === 'partial_advance' || paymentType === 'fully_prepaid') {
   430→   430→      payload.advance_amount =
   431→   431→        paymentType === 'fully_prepaid' ? total : parsePrice(advanceAmount)
   432→   432→      payload.advance_payment_method = advancePaymentMethod || undefined
   433→   433→      payload.advance_payment_reference = advancePaymentReference.trim() || undefined
   434→   434→      payload.advance_payment_screenshot_url = advancePaymentScreenshotUrl.trim() || undefined
   435→   435→    }
   436→   436→
   437→   437→    if (customerMode === 'existing' && selectedCustomer) {
   438→   438→      payload.customer_id = selectedCustomer.id
   439→   439→    } else {
   440→   440→      payload.customer = {
   441→   441→        name: newCustomer.name.trim(),
   442→   442→        phone: newCustomer.phone.trim(),
   443→   443→        alternate_phone: newCustomer.alternate_phone.trim() || undefined,
   444→   444→        email: newCustomer.email.trim() || undefined,
   445→   445→        addresses: [
   446→   446→          {
   447→   447→            label: 'Home',
   448→   448→            address: newCustomer.address.trim(),
   449→   449→            city: newCustomer.city.trim(),
   450→   450→            is_default: true,
   451→   451→          },
   452→   452→        ],
   453→   453→      }
   454→   454→    }
   455→   455→
   456→   456→    createMutation.mutate(payload)
   457→   457→  }
   458→   458→
   459→   459→  return (
   460→   460→    <div className="space-y-6">
   461→   461→      <PageHeader
   462→   462→        title="Create Order"
   463→   463→        description="Manually create a customer order"
   464→   464→        actions={
   465→   465→          <Button variant="outline" size="sm" onClick={() => navigate({ name: 'orders' })}>
   466→   466→            <ArrowLeft className="h-4 w-4" /> Back to Orders
   467→   467→          </Button>
   468→   468→        }
   469→   469→      />
   470→   470→
   471→   471→      {/* ── Stepper ─────────────────────────────────────────────────────────── */}
   472→   472→      <Stepper currentStep={step} />
   473→   473→
   474→   474→      {/* ── Step content ────────────────────────────────────────────────────── */}
   475→   475→      {step === 1 && (
   476→   476→        <CustomerStep
   477→   477→          customerMode={customerMode}
   478→   478→          setCustomerMode={setCustomerMode}
   479→   479→          phoneSearch={phoneSearch}
   480→   480→          setPhoneSearch={setPhoneSearch}
   481→   481→          selectedCustomer={selectedCustomer}
   482→   482→          setSelectedCustomer={setSelectedCustomer}
   483→   483→          newCustomer={newCustomer}
   484→   484→          setNewCustomer={setNewCustomer}
   485→   485→          customers={customersQuery.data?.customers ?? []}
   486→   486→          isSearching={customersQuery.isFetching}
   487→   487→        />
   488→   488→      )}
   489→   489→
   490→   490→      {step === 2 && (
   491→   491→        <ItemsStep
   492→   492→          cart={cart}
   493→   493→          variantSearch={variantSearch}
   494→   494→          setVariantSearch={setVariantSearch}
   495→   495→          variantSearchResults={variantSearchResults}
   496→   496→          addVariant={addVariant}
