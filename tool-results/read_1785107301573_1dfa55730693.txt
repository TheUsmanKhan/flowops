   225→   225→const PKR = new Intl.NumberFormat('en-PK', { maximumFractionDigits: 0 })
   226→   226→
   227→   227→function formatPKR(n: number): string {
   228→   228→  return `Rs. ${PKR.format(n)}`
   229→   229→}
   230→   230→
   231→   231→function formatDate(iso: string | null): string {
   232→   232→  if (!iso) return '—'
   233→   233→  try {
   234→   234→    return new Date(iso).toLocaleDateString('en-PK', {
   235→   235→      day: '2-digit',
   236→   236→      month: 'short',
   237→   237→      year: 'numeric',
   238→   238→    })
   239→   239→  } catch {
   240→   240→    return iso
   241→   241→  }
   242→   242→}
   243→   243→
   244→   244→function formatDateTime(iso: string | null): string {
   245→   245→  if (!iso) return '—'
   246→   246→  try {
   247→   247→    return new Date(iso).toLocaleString('en-PK', {
   248→   248→      day: '2-digit',
   249→   249→      month: 'short',
   250→   250→      hour: '2-digit',
   251→   251→      minute: '2-digit',
   252→   252→    })
   253→   253→  } catch {
   254→   254→    return iso
   255→   255→  }
   256→   256→}
   257→   257→
   258→   258→function getErrorMessage(err: unknown): string {
   259→   259→  if (err instanceof FetchError) return err.message
   260→   260→  if (err instanceof Error) return err.message
   261→   261→  return 'Something went wrong.'
   262→   262→}
   263→   263→
   264→   264→function formatAttributeValues(values: Record<string, string>): string {
   265→   265→  if (!values || Object.keys(values).length === 0) return ''
   266→   266→  return Object.entries(values)
   267→   267→    .map(([k, v]) => `${k}: ${v}`)
   268→   268→    .join(' · ')
   269→   269→}

   435→   435→        <PageHeader
   436→   436→          title="Order detail"

   458→   458→  const { order, customer, items } = orderQuery.data
   459→   459→
   460→   460→  // ── Status + action availability ──────────────────────────────────────────

   469→   469→    canFulfill &&
   470→   470→    !order.packedAt

   472→   472→    (status === 'confirmed' ||
   473→   473→      status === 'partially_backordered' ||
   474→   474→      status === 'processing') &&
   475→   475→    canFulfill
