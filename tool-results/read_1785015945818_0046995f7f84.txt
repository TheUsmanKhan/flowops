    73→// ──────────────────────────────────────────────────────────────
    74→// Helper: reserveOrderStock — shared internal function
    75→// ──────────────────────────────────────────────────────────────
    76→// Processes ALL order_items for an order and attempts to reserve
    77→// stock for each based on its fulfillment_type_snapshot.
    78→//
    79→// For stock_based items:
    80→//   - If sufficient available stock → reserveStockForOrder(), fulfillment_status='reserved'
    81→//   - If insufficient + inventory_policy='continue' → fulfillment_status='backordered'
    82→//   - If insufficient + inventory_policy='deny' → outcome='failed'
    83→//
    84→// For made_to_order items:
    85→//   - checkAndFulfillMadeToOrderVariant() checks returned stock first
    86→//   - If existing_stock → fulfillment_status='reserved', returned_stitched_used=true
    87→//   - If fresh_production → fulfillment_status='reserved', production_order_id set
    88→//   - If error → outcome='failed'
    89→//
    90→// After processing: if ANY item is 'backordered', order status → 'partially_backordered'.
    91→// If ALL items are 'reserved', order stays 'confirmed'.
    92→//
    93→// Called from:
    94→//   - createManualOrder() IF the order auto-confirmed at creation
    95→//   - confirmOrder() when a 'pending' order is manually confirmed
    96→//   - dispatchOrderAction() if status is still 'pending' at dispatch time
    97→
    98→async function reserveOrderStock(
    99→  orderId: string,
   100→  ctx: { company: { id: string; organizationId: string }; user: { id: string }; employee: { id: string } },
   101→): Promise<{
   102→  success: boolean
   103→  results: Array<{ orderItemId: string; outcome: 'reserved' | 'backordered' | 'failed'; reason?: string }>
   104→}> {
   105→  const order = await db.order.findUnique({
   106→    where: { id: orderId },
   107→    select: { dispatchLocationId: true, companyId: true, organizationId: true },
   108→  })
   109→  if (!order) return { success: false, results: [] }
   110→
   111→  const items = await db.orderItem.findMany({
   112→    where: { orderId },
   113→    include: {
   114→      orgVariant: {
   115→        select: { id: true, fulfillmentType: true, inventoryPolicy: true },
   116→      },
   117→    },
   118→  })
   119→
   120→  const results: Array<{ orderItemId: string; outcome: 'reserved' | 'backordered' | 'failed'; reason?: string }> = []
   121→  let hasBackordered = false
   122→
   123→  for (const item of items) {
   124→    if (item.fulfillmentStatus === 'reserved' || item.fulfillmentStatus === 'dispatched') {
   125→      // Already processed (e.g. re-confirmation attempt) — skip
   126→      results.push({ orderItemId: item.id, outcome: 'reserved' })
   127→      continue
   128→    }
   129→
   130→    const locationId = item.reservedLocationId ?? order.dispatchLocationId
   131→    if (!locationId) {
   132→      results.push({ orderItemId: item.id, outcome: 'failed', reason: 'No dispatch location set' })
   133→      continue
   134→    }
   135→
   136→    if (item.fulfillmentTypeSnapshot === 'stock_based') {
   137→      // Check available stock at the dispatch location
   138→      const pool = await db.inventoryPool.findUnique({
   139→        where: {
   140→          orgVariantId_locationId: {
   141→            orgVariantId: item.orgVariantId,
   142→            locationId,
   143→          },
   144→        },
   145→      })
   146→
   147→      const available = pool ? pool.onHand - pool.reserved : 0
   148→
   149→      if (available >= item.quantity) {
   150→        // Sufficient stock — reserve it
   151→        const reserveResult = await reserveStockForOrder({
   152→          orgVariantId: item.orgVariantId,
   153→          locationId,
   154→          organizationId: order.organizationId,
   155→          companyId: order.companyId,
   156→          employeeId: ctx.employee.id,
   157→          quantity: item.quantity,
   158→          orderId,
   159→        })
   160→
   161→        if (reserveResult.success) {
   162→          await db.orderItem.update({
   163→            where: { id: item.id },
   164→            data: { fulfillmentStatus: 'reserved', reservedLocationId: locationId },
   165→          })
   166→          results.push({ orderItemId: item.id, outcome: 'reserved' })
   167→        } else {
   168→          results.push({ orderItemId: item.id, outcome: 'failed', reason: reserveResult.error })
   169→        }
   170→      } else if (item.orgVariant.inventoryPolicy === 'continue') {
   171→        // Insufficient stock but backordering allowed
   172→        await db.orderItem.update({
   173→          where: { id: item.id },
   174→          data: { fulfillmentStatus: 'backordered', backorderedAt: new Date() },
   175→        })
   176→        hasBackordered = true
   177→        results.push({
   178→          orderItemId: item.id,
   179→          outcome: 'backordered',
   180→          reason: `Available: ${available}, required: ${item.quantity}`,
   181→        })
   182→      } else {
   183→        // Insufficient stock + policy='deny' — fail
   184→        results.push({
   185→          orderItemId: item.id,
   186→          outcome: 'failed',
   187→          reason: `Insufficient stock (available: ${available}, required: ${item.quantity}) and inventory_policy='deny'`,
   188→        })
   189→      }
   190→    } else if (item.fulfillmentTypeSnapshot === 'made_to_order') {
   191→      // Made-to-order: check returned stock first, then trigger production
   192→      const mtoResult = await checkAndFulfillMadeToOrderVariant(
   193→        item.orgVariantId,
   194→        item.quantity,
   195→        order.companyId,
   196→        locationId,
   197→      )
   198→
   199→      if (mtoResult.source === 'existing_stock' && mtoResult.locationId) {
   200→        // Returned stock available — reserve it
   201→        const reserveResult = await reserveStockForOrder({
   202→          orgVariantId: item.orgVariantId,
   203→          locationId: mtoResult.locationId,
   204→          organizationId: order.organizationId,
   205→          companyId: order.companyId,
   206→          employeeId: ctx.employee.id,
   207→          quantity: item.quantity,
   208→          orderId,
   209→        })
   210→
   211→        if (reserveResult.success) {
   212→          await db.orderItem.update({
   213→            where: { id: item.id },
   214→            data: {
   215→              fulfillmentStatus: 'reserved',
   216→              returnedStitchedUsed: true,
   217→              reservedLocationId: mtoResult.locationId,
   218→            },
   219→          })
   220→          results.push({ orderItemId: item.id, outcome: 'reserved' })
   221→        } else {
   222→          results.push({ orderItemId: item.id, outcome: 'failed', reason: reserveResult.error })
   223→        }
   224→      } else if (mtoResult.source === 'fresh_production' && mtoResult.productionOrderId) {
   225→        // Fresh production triggered — link the production order
   226→        await db.orderItem.update({
   227→          where: { id: item.id },
   228→          data: {
   229→            fulfillmentStatus: 'reserved',
   230→            productionOrderId: mtoResult.productionOrderId,
   231→          },
   232→        })
   233→        // Also link the order_item back to the production order
   234→        await db.productionOrder.update({
   235→          where: { id: mtoResult.productionOrderId },
   236→          data: { orderItemId: item.id },
   237→        })
   238→        results.push({ orderItemId: item.id, outcome: 'reserved' })
   239→      } else {
   240→        // Error (no fabric source, insufficient fabric, etc.)
   241→        results.push({
   242→          orderItemId: item.id,
   243→          outcome: 'failed',
   244→          reason: mtoResult.error ?? 'Made-to-order fulfillment failed',
   245→        })
   246→      }
   247→    }
   248→  }
   249→
   250→  // Update order status based on results
   251→  if (hasBackordered) {
   252→    await db.order.update({
   253→      where: { id: orderId },
   254→      data: { status: 'partially_backordered' },
   255→    })
   256→  }
   257→  // If no backordered items, order stays 'confirmed' (or whatever it was)
   258→
   259→  return { success: true, results }
   260→}

   263→// createManualOrder

   265→
   266→export async function createManualOrder(
   267→  input: CreateManualOrderInput,
   268→): Promise<ActionResult<{
   269→  orderId: string
   270→  flowopsOrderNumber: string
   271→  orderItems: Array<{ id: string; orgVariantId: string; quantity: number }>
   272→}>> {
   273→  try {
   274→    const ctx = await getWorkspace()
   275→    await requirePermission(ctx, PERMISSIONS.ORDERS_CREATE)
   276→
   277→    // 1. Validate
   278→    const parsed = createManualOrderSchema.safeParse(input)
   279→    if (!parsed.success) {
   280→      return {
   281→        success: false,
   282→        error: parsed.error.issues[0]?.message ?? 'Invalid order data',
   283→      }
   284→    }
   285→    const d = parsed.data
   286→
   287→    // 2. Find or create customer
   288→    let customerId: string
   289→    if (d.customer_id) {
   290→      // Verify existing customer belongs to this org
   291→      const existing = await db.customer.findFirst({
   292→        where: { id: d.customer_id, organizationId: ctx.company.organizationId },
   293→      })
   294→      if (!existing) return { success: false, error: 'Customer not found' }
   295→      customerId = existing.id
   296→    } else if (d.customer) {
   297→      const customerResult = await findOrCreateCustomer({
   298→        ...d.customer,
   299→        organizationId: ctx.company.organizationId,
   300→        companyId: ctx.company.id,
   301→      })
   302→      if (!customerResult.success || !customerResult.data) {
   303→        return { success: false, error: customerResult.error ?? 'Failed to create customer' }
   304→      }
   305→      customerId = customerResult.data.customerId
   306→    } else {
   307→      return { success: false, error: 'Either customer or customer_id is required' }
   308→    }
   309→
   310→    // 3. Fetch variants + their pricing for this company + fulfillment_type snapshot
   311→    const variantIds = d.items.map((i) => i.org_variant_id)
   312→    const variants = await db.orgProductVariant.findMany({
   313→      where: { id: { in: variantIds }, organizationId: ctx.company.organizationId },
   314→      include: {
   315→        companyPricing: { where: { companyId: ctx.company.id } },
   316→      },
   317→    })
   318→
   319→    if (variants.length !== variantIds.length) {
   320→      return { success: false, error: 'One or more variants not found in this organization' }
   321→    }
   322→
   323→    // 4. Compute subtotal + build order items data
   324→    let subtotal = 0
   325→    const orderItemsData: Array<{
   326→      orgVariantId: string
   327→      quantity: number
   328→      unitPrice: number
   329→      lineTotal: number
   330→      fulfillmentTypeSnapshot: string
   331→    }> = []
   332→
   333→    for (const item of d.items) {
   334→      const variant = variants.find((v) => v.id === item.org_variant_id)
   335→      if (!variant) continue
   336→
   337→      // Use provided unit_price OR fall back to company pricing OR variant cost
   338→      const unitPrice =
   339→        item.unit_price ??
   340→        (variant.companyPricing[0]?.salePrice
   341→          ? Number(variant.companyPricing[0].salePrice)
   342→          : Number(variant.costPrice))
   343→
   344→      const lineTotal = item.quantity * unitPrice
   345→      subtotal += lineTotal
   346→
   347→      orderItemsData.push({
   348→        orgVariantId: item.org_variant_id,
   349→        quantity: item.quantity,
   350→        unitPrice,
   351→        lineTotal,
   352→        fulfillmentTypeSnapshot: variant.fulfillmentType,
   353→      })
   354→    }
   355→
   356→    // 5. Compute totals
   357→    const discountAmount = d.discount_amount ?? 0
   358→    const courierCharges = d.courier_charges ?? 0
   359→    const totalOrderValue = subtotal + courierCharges - discountAmount
   360→
   361→    // 6. Determine payment status + source
   362→    let paymentStatus: string = 'cod_pending'
   363→    let paymentSource: string = 'cod_native'
   364→    let advanceAmount: number | null = null
   365→    let advancePaymentMethod: string | null = null
   366→    let advancePaymentReference: string | null = null
   367→    let advancePaidAt: Date | null = null
   368→
   369→    if (d.payment_type === 'partial_advance') {
   370→      paymentStatus = 'advance_paid'
   371→      paymentSource = 'manual_conversion'
   372→      advanceAmount = d.advance_amount ?? 0
   373→      advancePaymentMethod = d.advance_payment_method || null
   374→      advancePaymentReference = d.advance_payment_reference || null
   375→      advancePaidAt = new Date()
   376→    } else if (d.payment_type === 'fully_prepaid') {
   377→      paymentStatus = 'fully_prepaid'
   378→      paymentSource = 'manual_conversion'
   379→      advanceAmount = totalOrderValue
   380→      advancePaymentMethod = d.advance_payment_method || null
   381→      advancePaymentReference = d.advance_payment_reference || null
   382→      advancePaidAt = new Date()
   383→    }
   384→
   385→    // 7. Determine initial order status
   386→    let orderStatus: string = 'pending'
   387→    let confirmedAt: Date | null = null
   388→    let skippedConfirmation = false
   389→
   390→    if (paymentStatus === 'advance_paid' || paymentStatus === 'fully_prepaid') {
   391→      // Payment itself is the confirmation signal — bypasses require_order_confirmation
   392→      orderStatus = 'confirmed'
   393→      confirmedAt = new Date()
   394→    } else {
   395→      // COD order — check company_order_settings
   396→      const settings = await db.companyOrderSetting.findUnique({
   397→        where: { companyId: ctx.company.id },
   398→      })
   399→      if (!settings?.requireOrderConfirmation) {
   400→        orderStatus = 'confirmed'
   401→        confirmedAt = new Date()
   402→        skippedConfirmation = true
   403→      }
   404→    }
   405→
   406→    // 8. Generate order number
   407→    const flowopsOrderNumber = await generateOrderNumber(ctx.company.id)
   408→
   409→    // 9. Create the order
   410→    const order = await db.order.create({
   411→      data: {
   412→        organizationId: ctx.company.organizationId,
   413→        companyId: ctx.company.id,
   414→        flowopsOrderNumber,
   415→        orderSource: 'manual',
   416→        customerId,
   417→        status: orderStatus,
   418→        paymentType: d.payment_type,
   419→        paymentStatus,
   420→        paymentSource,
   421→        subtotal,
   422→        discountAmount: discountAmount || null,
   423→        discountReason: d.discount_reason || null,
   424→        courierCharges: courierCharges || null,
   425→        totalOrderValue,
   426→        advanceAmount,
   427→        advancePaymentMethod,
   428→        advancePaymentReference,
   429→        advancePaidAt,
   430→        deliveryAddress: d.delivery_address,
   431→        deliveryCity: d.delivery_city,
   432→        courierName: d.courier_name || null,
   433→        dispatchLocationId: d.dispatch_location_id,
   434→        notesForCourier: d.notes_for_courier || null,
   435→        skippedConfirmation,
   436→        confirmedAt,
   437→        createdBy: ctx.employee.id,
   438→      },
   439→    })
   440→
   441→    // 10. Create order items (fulfillment_status = 'reserved' as PLACEHOLDER)
   442→    const createdItems: Array<{ id: string; orgVariantId: string; quantity: number }> = []
   443→    for (const itemData of orderItemsData) {
   444→      const item = await db.orderItem.create({
   445→        data: {
   446→          orderId: order.id,
   447→          orgVariantId: itemData.orgVariantId,
   448→          organizationId: ctx.company.organizationId,
   449→          quantity: itemData.quantity,
   450→          unitPrice: itemData.unitPrice,
   451→          lineTotal: itemData.lineTotal,
   452→          fulfillmentStatus: 'reserved', // PLACEHOLDER — Step 3 will validate/adjust
   453→          fulfillmentTypeSnapshot: itemData.fulfillmentTypeSnapshot,
   454→          reservedLocationId: d.dispatch_location_id,
   455→        },
   456→      })
   457→      createdItems.push({
   458→        id: item.id,
   459→        orgVariantId: item.orgVariantId,
   460→        quantity: item.quantity,
   461→      })
   462→    }
   463→
   464→    // 11. Audit log
   465→    await insertAuditLog({
   466→      action: 'order.created',
   467→      entityType: 'order',
   468→      entityId: order.id,
   469→      companyId: ctx.company.id,
   470→      organizationId: ctx.company.organizationId,
   471→      userId: ctx.user.id,
   472→      employeeId: ctx.employee.id,
   473→      newValues: {
   474→        flowopsOrderNumber,
   475→        status: orderStatus,
   476→        paymentType: d.payment_type,
   477→        paymentStatus,
   478→        itemCount: orderItemsData.length,
   479→        totalOrderValue,
   480→        customerId,
   481→      },
   482→    })
   483→
   484→    // 12. Update customer stats
   485→    await updateCustomerStats(customerId)
   486→
   487→    // 13. If the order auto-confirmed (payment-driven or company setting),
   488→    // run the stock reservation logic immediately.
   489→    if (orderStatus === 'confirmed') {
   490→      await reserveOrderStock(order.id, ctx)
   491→    }
   492→
   493→    return {
   494→      success: true,
   495→      data: { orderId: order.id, flowopsOrderNumber, orderItems: createdItems },
   496→    }
   497→  } catch (err) {
   498→    return {
   499→      success: false,
   500→      error: err instanceof Error ? err.message : 'Failed to create order',
   501→    }
   502→  }
   503→}

   508→
   509→export async function createOrderFromShopifyWebhook(
   510→  payload: ShopifyOrderWebhook,
   511→  companyId: string,
   512→  organizationId: string,
   513→): Promise<ActionResult<{ orderId: string; flowopsOrderNumber: string }>> {
   514→  try {
   515→    const ctx = await getWorkspace()
   516→    // This function would be called by a future webhook handler.
   517→    // For now it's structured and unit-testable with a mock payload.
   518→
   519→    const parsed = shopifyOrderWebhookSchema.safeParse(payload)
   520→    if (!parsed.success) {
   521→      return { success: false, error: 'Invalid Shopify webhook payload' }
   522→    }
   523→    const d = parsed.data
   524→
   525→    // Map Shopify financial_status → FlowOps payment_status
   526→    let paymentStatus: string
   527→    let paymentSource: string = 'shopify_gateway'
   528→    let paymentType: string
   529→    let orderStatus: string = 'pending'
   530→    let confirmedAt: Date | null = null
   531→
   532→    switch (d.financial_status) {
   533→      case 'paid':
   534→        paymentStatus = 'fully_prepaid'
   535→        paymentType = 'fully_prepaid'
   536→        orderStatus = 'confirmed'
   537→        confirmedAt = new Date()
   538→        break
   539→      case 'partially_paid':
   540→        paymentStatus = 'advance_paid'
   541→        paymentType = 'partial_advance'
   542→        orderStatus = 'confirmed'
   543→        confirmedAt = new Date()
   544→        break
   545→      default:
   546→        paymentStatus = 'cod_pending'
   547→        paymentType = 'full_cod'
   548→        paymentSource = 'cod_native'
   549→        // Check company settings for COD orders
   550→        const settings = await db.companyOrderSetting.findUnique({ where: { companyId } })
   551→        if (!settings?.requireOrderConfirmation) {
   552→          orderStatus = 'confirmed'
   553→          confirmedAt = new Date()
   554→        }
   555→    }
   556→
   557→    // Find or create customer from Shopify customer data
   558→    const customerPhone = d.customer.phone || '0000000000' // fallback
   559→    let customer = await db.customer.findFirst({
   560→      where: { organizationId, phone: customerPhone },
   561→    })
   562→
   563→    if (!customer && d.customer.first_name) {
   564→      customer = await db.customer.create({
   565→        data: {
   566→          organizationId,
   567→          name: `${d.customer.first_name} ${d.customer.last_name ?? ''}`.trim(),
   568→          phone: customerPhone,
   569→          email: d.customer.email || null,
   570→          addresses: JSON.stringify(
   571→            d.customer.default_address
   572→              ? [{
   573→                  address: d.customer.default_address.address1 || '',
   574→                  city: d.customer.default_address.city || '',
   575→                  province: d.customer.default_address.province || '',
   576→                  is_default: true,
   577→                }]
   578→              : [],
   579→          ),
   580→        },
   581→      })
   582→    }
   583→
   584→    if (!customer) {
   585→      return { success: false, error: 'Could not resolve customer from Shopify payload' }
   586→    }
   587→
   588→    // Resolve variants by SKU
   589→    const skus = d.line_items.map((li) => li.sku).filter(Boolean) as string[]
   590→    const variants = await db.orgProductVariant.findMany({
   591→      where: { sku: { in: skus }, organizationId },
   592→    })
   593→
   594→    // Compute subtotal
   595→    let subtotal = 0
   596→    const orderItemsData: Array<{
   597→      orgVariantId: string
   598→      quantity: number
   599→      unitPrice: number
   600→      fulfillmentTypeSnapshot: string
   601→    }> = []
   602→
   603→    for (const li of d.line_items) {
   604→      const variant = variants.find((v) => v.sku === li.sku)
   605→      if (!variant) continue // skip unmatched items for now
   606→      const unitPrice = parseFloat(li.price)
   607→      subtotal += unitPrice * li.quantity
   608→      orderItemsData.push({
   609→        orgVariantId: variant.id,
   610→        quantity: li.quantity,
   611→        unitPrice,
   612→        fulfillmentTypeSnapshot: variant.fulfillmentType,
   613→      })
   614→    }
   615→
   616→    if (orderItemsData.length === 0) {
   617→      return { success: false, error: 'No matching variants found for Shopify line items' }
   618→    }
   619→
   620→    const totalOrderValue = parseFloat(d.total_price)
   621→    const flowopsOrderNumber = await generateOrderNumber(companyId)
   622→
   623→    const order = await db.order.create({
   624→      data: {
   625→        organizationId,
   626→        companyId,
   627→        flowopsOrderNumber,
   628→        orderSource: 'shopify',
   629→        externalOrderReference: d.name,
   630→        externalOrderId: String(d.id),
   631→        customerId: customer.id,
   632→        status: orderStatus,
   633→        paymentType,
   634→        paymentStatus,
   635→        paymentSource,
   636→        subtotal,
   637→        totalOrderValue,
   638→        confirmedAt,
   639→        deliveryAddress: d.customer.default_address?.address1 || null,
   640→        deliveryCity: d.customer.default_address?.city || null,
   641→      },
   642→    })
   643→
   644→    for (const itemData of orderItemsData) {
   645→      await db.orderItem.create({
   646→        data: {
   647→          orderId: order.id,
   648→          orgVariantId: itemData.orgVariantId,
   649→          organizationId,
   650→          quantity: itemData.quantity,
   651→          unitPrice: itemData.unitPrice,
   652→          lineTotal: itemData.quantity * itemData.unitPrice,
   653→          fulfillmentStatus: 'reserved',
   654→          fulfillmentTypeSnapshot: itemData.fulfillmentTypeSnapshot,
   655→        },
   656→      })
   657→    }
   658→
   659→    await insertAuditLog({
   660→      action: 'order.created_from_shopify',
   661→      entityType: 'order',
   662→      entityId: order.id,
   663→      companyId,
   664→      organizationId,
   665→      userId: ctx.user.id,
   666→      employeeId: ctx.employee.id,
   667→      newValues: {
   668→        flowopsOrderNumber,
   669→        externalOrderReference: d.name,
   670→        financialStatus: d.financial_status,
   671→        totalOrderValue,
   672→      },
   673→    })
   674→
   675→    await updateCustomerStats(customer.id)
   676→
   677→    return { success: true, data: { orderId: order.id, flowopsOrderNumber } }
   678→  } catch (err) {
   679→    return {
   680→      success: false,
   681→      error: err instanceof Error ? err.message : 'Failed to create order from Shopify webhook',
   682→    }
   683→  }
   684→}

   878→export async function cancelOrder(input: CancelOrderInput): Promise<ActionResult> {
   879→  try {
   880→    const ctx = await getWorkspace()
   881→    await requirePermission(ctx, PERMISSIONS.ORDERS_CANCEL)
   882→
   883→    const parsed = cancelOrderSchema.safeParse(input)
   884→    if (!parsed.success) {
   885→      return { success: false, error: parsed.error.issues[0]?.message ?? 'Invalid input' }
   886→    }
   887→    const d = parsed.data
   888→

   890→      where: { id: d.order_id, companyId: ctx.company.id },
   891→    })
   892→    if (!order) return { success: false, error: 'Order not found' }
   893→
   894→    const nonCancellableStatuses = ['dispatched', 'delivered', 'rto', 'cancelled', 'refunded']
   895→    if (nonCancellableStatuses.includes(order.status)) {
   896→      return {
   897→        success: false,
   898→        error: `Cannot cancel an order that is already ${order.status} (use the returns flow instead)`,
   899→      }
   900→    }
   901→
   902→    await db.order.update({
   903→      where: { id: d.order_id },
   904→      data: {

   908→      },
   909→    })
   910→
   911→    // Step 3: Unreserve stock for any items with fulfillment_status='reserved'.
   912→    // Items with fulfillment_status='backordered' need no inventory action
   913→    // (no reservation ever existed for them) — they'll be orphaned since
   914→    // the order is now cancelled and won't be picked up by future
   915→    // checkAndFulfillBackorders() runs (which skip cancelled orders).
   916→    const reservedItems = await db.orderItem.findMany({
   917→      where: { orderId: d.order_id, fulfillmentStatus: 'reserved' },
   918→    })
   919→
   920→    for (const item of reservedItems) {
   921→      const locationId = item.reservedLocationId ?? order.dispatchLocationId
   922→      if (!locationId) continue
   923→
   924→      await unreserveStockForOrder({
   925→        orgVariantId: item.orgVariantId,
   926→        locationId,
   927→        organizationId: order.organizationId,
   928→        companyId: ctx.company.id,
   929→        employeeId: ctx.employee.id,
   930→        quantity: item.quantity,
   931→        orderId: d.order_id,
   932→      })
   933→    }
   934→
   935→    await insertAuditLog({
   936→      action: 'order.cancelled',
   937→      entityType: 'order',
   938→      entityId: d.order_id,
   939→      companyId: ctx.company.id,
   940→      organizationId: ctx.company.organizationId,
   941→      userId: ctx.user.id,
   942→      employeeId: ctx.employee.id,
   943→      oldValues: { status: order.status },
   944→      newValues: { status: 'cancelled', reason: d.cancellation_reason, unreservedItems: reservedItems.length },
   945→    })
   946→
   947→    return { success: true }
   948→  } catch (err) {
   949→    return {
   950→      success: false,
   951→      error: err instanceof Error ? err.message : 'Failed to cancel order',
   952→    }
   953→  }
   954→}

  1165→ * Dispatch an order — deducts stock from inventory, sets tracking info.
  1166→ *
  1167→ * Business rules:
  1168→ *   - If order.status = 'pending' (confirmation was skipped): run the
  1169→ *     full confirmation/reservation logic inline first, then proceed.
  1170→ *   - If any item is still 'backordered' after reservation: BLOCK dispatch
  1171→ *     with a clear error (hard rule — no split shipments).
  1172→ *   - For each reserved item: call dispatchOrder() (inventory) which
  1173→ *     deducts on_hand, releases the reservation, and locks COGS.
  1174→ *   - The backfill_order_timestamps() trigger auto-sets confirmed_at/
  1175→ *     packed_at if they were still NULL.
  1176→ *   - If company_order_settings.require_packing_step = TRUE, verify
  1177→ *     order.packed_at IS NOT NULL before proceeding.
  1178→ */
  1179→export async function dispatchOrderAction(
  1180→  orderId: string,
  1181→  trackingNumber: string,
  1182→  courierName?: string,
  1183→): Promise<ActionResult> {
  1184→  try {
  1185→    const ctx = await getWorkspace()
  1186→    await requirePermission(ctx, PERMISSIONS.ORDERS_FULFILL)
  1187→
  1188→    const order = await db.order.findFirst({
  1189→      where: { id: orderId, companyId: ctx.company.id },
  1190→    })
  1191→    if (!order) return { success: false, error: 'Order not found' }
  1192→
  1193→    // Cannot dispatch already-dispatched/delivered/cancelled orders
  1194→    if (['dispatched', 'delivered', 'rto', 'cancelled', 'refunded'].includes(order.status)) {
  1195→      return { success: false, error: `Cannot dispatch an order that is already ${order.status}` }
  1196→    }
  1197→
  1198→    // Check packing requirement
  1199→    const settings = await db.companyOrderSetting.findUnique({
  1200→      where: { companyId: ctx.company.id },
  1201→    })
  1202→    if (settings?.requirePackingStep && !order.packedAt) {
  1203→      return {
  1204→        success: false,
  1205→        error: 'This order must be marked as packed before dispatching. Use markOrderPacked() first.',
  1206→      }
  1207→    }
  1208→
  1209→    // If order is still 'pending' (confirmation was skipped), run the
  1210→    // full confirmation + reservation logic inline first.
  1211→    if (order.status === 'pending') {
  1212→      await db.order.update({
  1213→        where: { id: orderId },
  1214→        data: { status: 'confirmed', confirmedAt: new Date() },
  1215→      })
  1216→      await reserveOrderStock(orderId, ctx)
  1217→    }
  1218→
  1219→    // Re-fetch the order to get the updated status after reservation
  1220→    const updatedOrder = await db.order.findFirst({
  1221→      where: { id: orderId },
  1222→      select: { status: true, dispatchLocationId: true, organizationId: true },
  1223→    })
  1224→    if (!updatedOrder) return { success: false, error: 'Order not found after reservation' }
  1225→
  1226→    // Check for backordered items — BLOCK dispatch if any exist
  1227→    const backorderedItems = await db.orderItem.findMany({
  1228→      where: { orderId, fulfillmentStatus: 'backordered' },
  1229→      include: {
  1230→        orgVariant: { select: { sku: true } },
  1231→      },
  1232→    })
  1233→
  1234→    if (backorderedItems.length > 0) {
  1235→      const itemList = backorderedItems
  1236→        .map((i) => `${i.orgVariant.sku} (qty: ${i.quantity})`)
  1237→        .join(', ')
  1238→      return {
  1239→        success: false,
  1240→        error: `Cannot dispatch: ${backorderedItems.length} item(s) are still backordered: ${itemList}. Receive stock or cancel those items first.`,
  1241→      }
  1242→    }
  1243→
  1244→    // Dispatch each reserved item
  1245→    const itemsToDispatch = await db.orderItem.findMany({
  1246→      where: { orderId, fulfillmentStatus: 'reserved' },
  1247→    })
  1248→
  1249→    for (const item of itemsToDispatch) {
  1250→      const locationId = item.reservedLocationId ?? updatedOrder.dispatchLocationId
  1251→      if (!locationId) {
  1252→        return { success: false, error: `Order item ${item.id} has no dispatch location` }
  1253→      }
  1254→
  1255→      const dispatchResult = await dispatchInventory({
  1256→        orgVariantId: item.orgVariantId,
  1257→        locationId,
  1258→        organizationId: updatedOrder.organizationId,
  1259→        companyId: ctx.company.id,
  1260→        employeeId: ctx.employee.id,
  1261→        quantity: item.quantity,
  1262→        orderId,
  1263→      })
  1264→
  1265→      if (!dispatchResult.success) {
  1266→        return {
  1267→          success: false,
  1268→          error: `Failed to dispatch item: ${dispatchResult.error}`,
  1269→        }
  1270→      }
  1271→
  1272→      // Mark item as dispatched
  1273→      await db.orderItem.update({
  1274→        where: { id: item.id },
  1275→        data: { fulfillmentStatus: 'dispatched', fulfilledAt: new Date() },
  1276→      })
  1277→    }
  1278→
  1279→    // Update order status — the backfill_order_timestamps() trigger
  1280→    // will auto-set confirmedAt/packedAt if they were still NULL.
  1281→    await db.order.update({
  1282→      where: { id: orderId },
  1283→      data: {
  1284→        status: 'dispatched',
  1285→        dispatchedAt: new Date(),
  1286→        trackingNumber: trackingNumber || null,
  1287→        courierName: courierName || order.courierName,
  1288→      },
  1289→    })
  1290→
  1291→    await insertAuditLog({
  1292→      action: 'order.dispatched',
  1293→      entityType: 'order',
  1294→      entityId: orderId,
  1295→      companyId: ctx.company.id,
  1296→      organizationId: ctx.company.organizationId,
  1297→      userId: ctx.user.id,
  1298→      employeeId: ctx.employee.id,
  1299→      oldValues: { status: updatedOrder.status },
  1300→      newValues: {
  1301→        status: 'dispatched',
  1302→        trackingNumber,
  1303→        courierName,
  1304→        itemsDispatched: itemsToDispatch.length,
  1305→      },
  1306→    })
  1307→
  1308→    // Update customer stats
  1309→    const orderWithCustomer = await db.order.findUnique({
  1310→      where: { id: orderId },
  1311→      select: { customerId: true },
  1312→    })
  1313→    if (orderWithCustomer) {
  1314→      await updateCustomerStats(orderWithCustomer.customerId)
  1315→    }
  1316→
  1317→    return { success: true }
  1318→  } catch (err) {
  1319→    return {
  1320→      success: false,
  1321→      error: err instanceof Error ? err.message : 'Failed to dispatch order',
  1322→    }
  1323→  }
  1324→}

  1418→/**
  1419→ * Mark a dispatched order as delivered.
  1420→ */
  1421→export async function markOrderDelivered(orderId: string): Promise<ActionResult> {
  1422→  try {
  1423→    const ctx = await getWorkspace()
  1424→    await requirePermission(ctx, PERMISSIONS.ORDERS_FULFILL)
  1425→
  1426→    const order = await db.order.findFirst({
  1427→      where: { id: orderId, companyId: ctx.company.id },
  1428→    })
  1429→    if (!order) return { success: false, error: 'Order not found' }
  1430→    if (order.status !== 'dispatched') {
  1431→      return { success: false, error: `Order must be dispatched to mark as delivered (current: ${order.status})` }
  1432→    }
  1433→
  1434→    await db.order.update({
  1435→      where: { id: orderId },
  1436→      data: { status: 'delivered', deliveredAt: new Date() },
  1437→    })
  1438→
  1439→    await insertAuditLog({
  1440→      action: 'order.delivered',
  1441→      entityType: 'order',
  1442→      entityId: orderId,
  1443→      companyId: ctx.company.id,
  1444→      organizationId: ctx.company.organizationId,
  1445→      userId: ctx.user.id,
  1446→      employeeId: ctx.employee.id,
  1447→      oldValues: { status: 'dispatched' },
  1448→      newValues: { status: 'delivered' },
  1449→    })
  1450→
  1451→    // Metric: order.delivered
  1452→    const deliveryDays = order.dispatchedAt
  1453→      ? Math.round((Date.now() - new Date(order.dispatchedAt).getTime()) / (1000 * 60 * 60 * 24))
  1454→      : 0
  1455→
  1456→    await insertMetricEvent({
  1457→      companyId: ctx.company.id,
  1458→      entityType: 'order',
  1459→      entityId: orderId,
  1460→      metricKey: 'order.delivered',
  1461→      numericValue: Number(order.totalOrderValue),
  1462→      dimensions: { delivery_days: deliveryDays },
  1463→    }).catch(() => {})
  1464→
  1465→    return { success: true }
  1466→  } catch (err) {
  1467→    return {
  1468→      success: false,
  1469→      error: err instanceof Error ? err.message : 'Failed to mark order as delivered',
  1470→    }
  1471→  }
  1472→}
