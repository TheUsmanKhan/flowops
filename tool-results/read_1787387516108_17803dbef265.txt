   107→/**
   108→ * Fetches live inventory per variant for this product and returns a map keyed
   109→ * by variantId. Used by the parent-child table to display on-hand stock
   110→ * alongside cost/price columns. Reads from the same inventory_pools data as
   111→ * the Inventory tab and the Inventory Dashboard — no parallel data source.
   112→ */
   113→function useVariantInventoryMap(productId: string) {
   114→  const { data } = useQuery<InventorySummaryResponse>({
   115→    queryKey: ['product-inventory', productId],
   116→    queryFn: () => api.get(`/api/inventory/summary?product_id=${productId}`),
   117→    staleTime: 15_000,
   118→  })
   119→  const map = new Map<string, InventorySummaryVariant>()
   120→  for (const v of data?.variants ?? []) {
   121→    map.set(v.variantId, v)
   122→  }
   123→  return map
   124→}

   126→/** Compact stock cell — shows on_hand, with available in muted subtext. */
   127→function StockCell({ inv }: { inv?: InventorySummaryVariant }) {
   128→  if (!inv) {
   129→    return <span className="text-xs text-muted-foreground">—</span>
   130→  }
   131→  const onHand = inv.totalOnHand
   132→  const available = inv.totalAvailable
   133→  if (onHand === 0) {
   134→    return <span className="text-xs text-muted-foreground">0</span>
   135→  }
   136→  const lowStock = available > 0 && available <= 5
   137→  return (
   138→    <div className="text-xs leading-tight">
   139→      <span className={cn('font-medium', lowStock ? 'text-amber-700' : 'text-foreground')}>{onHand}</span>
   140→      {available !== onHand && (
   141→        <div className="text-[10px] text-muted-foreground">{available} avail.</div>
   142→      )}
   143→    </div>
   144→  )
   145→}

   148→ * Bug 2 fix: badge that reads trackInventory (mutable), NOT fulfillmentType
   149→ * (immutable). A made_to_order variant with opening stock added shows

   151→ * fulfillmentType-based display if trackInventory is undefined (e.g. older

   182→export function ParentChildVariantTable({
   183→  productId,
   184→  mode = 'both',
   185→}: {
   186→  productId: string
   187→  mode?: 'cost' | 'pricing' | 'both'
   188→}) {
   189→  const can = useCan()
   190→  const queryClient = useQueryClient()
   191→  const [expandedGroups, setExpandedGroups] = useState<Set<string>>(new Set())
   192→  const [editingVariant, setEditingVariant] = useState<ChildVariant | null>(null)
   193→
   194→  const canEditCost = can(PERMISSIONS.PRODUCTS_EDIT)
   195→  const canEditPrice = can(PERMISSIONS.PRODUCTS_PRICING)
   196→  const showCost = mode === 'cost' || mode === 'both'
   197→  const showPricing = mode === 'pricing' || mode === 'both'
   198→
   199→  const { data, isLoading, isError, refetch } = useQuery<VariantGroupsResponse>({
   200→    queryKey: ['variant-groups', productId],
   201→    queryFn: () => api.get(`/api/products/${productId}/variant-groups`),
   202→    staleTime: 15_000,
   203→  })
   204→
   205→  // Live inventory per variant — powers the read-only Stock column.
   206→  const inventoryMap = useVariantInventoryMap(productId)
   207→
   208→  function toggleGroup(parentValue: string) {
   209→    setExpandedGroups((prev) => {
   210→      const next = new Set(prev)
   211→      if (next.has(parentValue)) next.delete(parentValue)
   212→      else next.add(parentValue)
   213→      return next
   214→    })
   215→  }
   216→
   217→  // Expand all groups by default on first load
   218→  if (data && expandedGroups.size === 0 && data.groups.length > 0) {
   219→    setExpandedGroups(new Set(data.groups.map((g) => g.parentValue)))
   220→  }
   221→
   222→  if (isLoading) {
   223→    return (
   224→      <Card>
   225→        <CardContent className="p-6 space-y-3">
   226→          {[0, 1, 2].map((i) => (
   227→            <div key={i} className="h-20 rounded bg-muted/50 animate-pulse" />
   228→          ))}
   229→        </CardContent>
   230→      </Card>
   231→    )
   232→  }
   233→
   234→  if (isError) {
   235→    return (
   236→      <Card>
   237→        <CardContent className="p-6 text-center">
   238→          <AlertCircle className="h-8 w-8 mx-auto text-muted-foreground mb-2" />
   239→          <p className="text-sm text-muted-foreground mb-3">Failed to load variant groups.</p>
   240→          <Button variant="outline" size="sm" onClick={() => refetch()}>Retry</Button>
   241→        </CardContent>
   242→      </Card>
   243→    )
   244→  }
   245→
   246→  if (!data || data.groups.length === 0) {
   247→    return (
   248→      <Card>
   249→        <CardContent className="p-6 text-center text-sm text-muted-foreground">
   250→          No variants found for this product.
   251→        </CardContent>
   252→      </Card>
   253→    )
   254→  }
   255→
   256→  // If only one attribute, render flat table (no grouping)
   257→  if (!data.hasMultipleAttributes) {
   258→    return (
   259→      <>
   260→        <FlatVariantTable
   261→          groups={data.groups}
   262→          productId={productId}
   263→          showCost={showCost}
   264→          showPricing={showPricing}
   265→          canEditCost={canEditCost}
   266→          canEditPrice={canEditPrice}
   267→          queryClient={queryClient}
   268→          onEdit={(v) => setEditingVariant(v)}
   269→          inventoryMap={inventoryMap}
   270→        />
   271→        {editingVariant && (
   272→          <VariantEditDialog
   273→            variant={editingVariant}
   274→            productId={productId}
   275→            onClose={() => setEditingVariant(null)}
   276→            queryClient={queryClient}
   277→          />
   278→        )}
   279→      </>
   280→    )
   281→  }
   282→
   283→  // Grouped table with parent-child cascade
   284→  return (
   285→    <div className="space-y-3">
   286→      {data.parentAttributeName && (
   287→        <p className="text-xs text-muted-foreground">
   288→          Grouped by <span className="font-medium text-foreground">{data.parentAttributeDisplayName || data.parentAttributeName}</span> —
   289→          set prices once per group, or override individual variants.
   290→        </p>
   291→      )}
   292→      {data.groups.map((group) => (
   293→        <GroupCard
   294→          key={group.parentValue}
   295→          group={group}
   296→          productId={productId}
   297→          parentAttributeName={data.parentAttributeName!}
   298→          expanded={expandedGroups.has(group.parentValue)}
   299→          onToggle={() => toggleGroup(group.parentValue)}
   300→          showCost={showCost}
   301→          showPricing={showPricing}
   302→          canEditCost={canEditCost}
   303→          canEditPrice={canEditPrice}
   304→          queryClient={queryClient}
   305→          onEditVariant={(v) => setEditingVariant(v)}
   306→          inventoryMap={inventoryMap}
   307→        />
   308→      ))}
   309→      {editingVariant && (
   310→        <VariantEditDialog
   311→          variant={editingVariant}
   312→          productId={productId}
   313→          onClose={() => setEditingVariant(null)}
   314→          queryClient={queryClient}
   315→        />
   316→      )}
   317→    </div>
   318→  )
   319→}

   324→function GroupCard({
   325→  group,
   326→  productId,
   327→  parentAttributeName,
   328→  expanded,
   329→  onToggle,
   330→  showCost,
   331→  showPricing,
   332→  canEditCost,
   333→  canEditPrice,
   334→  queryClient,
   335→  onEditVariant,
   336→  inventoryMap,
   337→}: {
   338→  group: VariantGroup
   339→  productId: string
   340→  parentAttributeName: string
   341→  expanded: boolean
   342→  onToggle: () => void
   343→  showCost: boolean
   344→  showPricing: boolean
   345→  canEditCost: boolean
   346→  canEditPrice: boolean
   347→  queryClient: ReturnType<typeof useQueryClient>
   348→  onEditVariant: (v: ChildVariant) => void
   349→  inventoryMap: Map<string, InventorySummaryVariant>
   350→}) {
   351→  const [parentCost, setParentCost] = useState('')
   352→  const [parentSale, setParentSale] = useState('')
   353→  const [parentCompare, setParentCompare] = useState('')
   354→  const [parentWeight, setParentWeight] = useState('')
   355→  const [applying, setApplying] = useState(false)
   356→
   357→  const firstChild = group.children[0]
   358→  if (parentCost === '' && firstChild) setParentCost(String(firstChild.costPrice))
   359→  if (parentSale === '' && firstChild?.salePrice != null) setParentSale(String(firstChild.salePrice))
   360→  if (parentCompare === '' && firstChild?.comparePrice != null) setParentCompare(String(firstChild.comparePrice))
   361→  if (parentWeight === '' && firstChild?.weightKg != null) setParentWeight(String(firstChild.weightKg))
   362→
   363→  // Bug 1 fix: ONE handler cascades ALL THREE fields (cost, sale, compare).
   364→  // Calls both server endpoints (cost + sale-price) in sequence, then
   365→  // invalidates the cache once. Each endpoint only updates children whose
   366→  // relevant synced flag is true, so the three flags remain INDEPENDENT.
   367→  // Weight cascade added — calls the weight endpoint in the same sequence.
   368→  async function applyAllToGroup() {
   369→    const cost = Number(parentCost)
   370→    const sale = Number(parentSale)
   371→    const compare = parentCompare ? Number(parentCompare) : null
   372→    const weight = parentWeight ? Number(parentWeight) : null
   373→
   374→    if (isNaN(cost) || cost < 0) { toast.error('Enter a valid cost price'); return }
   375→    if (isNaN(sale) || sale < 0) { toast.error('Enter a valid sale price'); return }
   376→    if (parentWeight && (isNaN(weight as number) || (weight as number) < 0)) { toast.error('Enter a valid weight'); return }
   377→
   378→    setApplying(true)
   379→    let costCount = 0
   380→    let saleCount = 0
   381→    let weightCount = 0
   382→    try {
   383→      // Cascade cost to cost-synced children
   384→      const costRes = await api.post<{ success: boolean; updated_count: number }>(
   385→        `/api/products/${productId}/variant-groups/dummy/cost`,
   386→        { cost_price: cost, parent_attribute_name: parentAttributeName, parent_value: group.parentValue },
   387→      )
   388→      costCount = costRes.updated_count
   389→
   390→      // Cascade sale + compare to their respective synced children
   391→      const saleRes = await api.post<{ success: boolean; updated_count: number }>(
   392→        `/api/products/${productId}/variant-groups/dummy/sale-price`,
   393→        { sale_price: sale, compare_price: compare, parent_attribute_name: parentAttributeName, parent_value: group.parentValue },
   394→      )
   395→      saleCount = saleRes.updated_count
   396→
   397→      // Cascade weight to weight-synced children (only if a weight was entered)
   398→      if (parentWeight && weight != null) {
   399→        const weightRes = await api.post<{ success: boolean; updated_count: number }>(
   400→          `/api/products/${productId}/variant-groups/dummy/weight`,
   401→          { weightKg: weight, parent_attribute_name: parentAttributeName, parent_value: group.parentValue },
   402→        )
   403→        weightCount = weightRes.updated_count
   404→      }
   405→
   406→      toast.success(`Applied to group — Cost: ${costCount}, Sale+Compare: ${saleCount}${parentWeight ? `, Weight: ${weightCount}` : ''} variant(s)`)
   407→      queryClient.invalidateQueries({ queryKey: ['variant-groups', productId] })
   408→      queryClient.invalidateQueries({ queryKey: ['product', productId] })
   409→    } catch (err) {
   410→      toast.error(err instanceof FetchError ? err.message : 'Failed to apply to group')
   411→    } finally { setApplying(false) }
   412→  }
   413→
   414→  const childAttrKeys = group.children.length > 0
   415→    ? Object.keys(group.children[0].attributeValues).filter((k) => k !== parentAttributeName)
   416→    : []
   417→
   418→  return (
   419→    <Card>
   420→      <CardHeader className="pb-3 cursor-pointer" onClick={onToggle}>
   421→        <div className="flex items-center justify-between">
   422→          <div className="flex items-center gap-2">
   423→            {expanded ? <ChevronDown className="h-4 w-4 text-muted-foreground" /> : <ChevronRight className="h-4 w-4 text-muted-foreground" />}
   424→            <CardTitle className="text-base">{group.parentValue}</CardTitle>
   425→            <Badge variant="secondary" className="text-[10px]">{group.childCount} variant{group.childCount === 1 ? '' : 's'}</Badge>
   426→          </div>
   427→        </div>
   428→      </CardHeader>
   429→
   430→      {expanded && (
   431→        <CardContent className="space-y-4">
   432→          {/* Parent group inputs — uses the shared ParentGroupInputs component
   433→              with a single "Apply to Group" button that cascades ALL fields
   434→              (cost, sale, compare, weight) independently. Bug 1 fix + weight extension. */}
   435→          <ParentGroupInputs
   436→            parentCost={parentCost}
   437→            parentSale={parentSale}
   438→            parentCompare={parentCompare}
   439→            parentWeight={parentWeight}
   440→            onCostChange={setParentCost}
   441→            onSaleChange={setParentSale}
   442→            onCompareChange={setParentCompare}
   443→            onWeightChange={setParentWeight}
   444→            onApplyAll={applyAllToGroup}
   445→            showCost={showCost}
   446→            showPricing={showPricing}
   447→            showWeight={canEditCost}
   448→            canEditCost={canEditCost}
   449→            canEditPrice={canEditPrice}
   450→            canEditWeight={canEditCost}
   451→            applying={applying}
   452→          />
   453→
   454→          {/* Children table */}
   455→          <div className="rounded-md border overflow-x-auto scrollbar-thin">
   456→            <table className="w-full text-sm">
   457→              <thead className="bg-muted/30">
   458→                <tr className="text-left text-xs text-muted-foreground">
   459→                  {childAttrKeys.map((k) => (<th key={k} className="px-3 py-2 font-medium">{k}</th>))}
   460→                  <th className="px-3 py-2 font-medium">SKU</th>
   461→                  <th className="px-3 py-2 font-medium">Fulfillment</th>
   462→                  <th className="px-3 py-2 font-medium text-center">Stock</th>
   463→                  {showCost && <th className="px-3 py-2 font-medium text-right">Cost</th>}
   464→                  {showPricing && <th className="px-3 py-2 font-medium text-right">Sale</th>}
   465→                  {showPricing && <th className="px-3 py-2 font-medium text-right">Compare</th>}
   466→                  {canEditCost && <th className="px-3 py-2 font-medium text-right">Weight (kg)</th>}
   467→                  <th className="px-3 py-2 font-medium text-center">Active</th>
   468→                  <th className="px-3 py-2 font-medium">Actions</th>
   469→                </tr>
   470→              </thead>
   471→              <tbody>
   472→                {group.children.map((child) => (
   473→                  <ChildRow
   474→                    key={child.variantId}
   475→                    child={child}
   476→                    childAttrKeys={childAttrKeys}
   477→                    productId={productId}
   478→                    showCost={showCost}
   479→                    showPricing={showPricing}
   480→                    canEditCost={canEditCost}
   481→                    canEditPrice={canEditPrice}
   482→                    queryClient={queryClient}
   483→                    onEdit={() => onEditVariant(child)}
   484→                    inventory={inventoryMap.get(child.variantId)}
   485→                  />
   486→                ))}
   487→              </tbody>
   488→            </table>
   489→          </div>
   490→        </CardContent>
   491→      )}
   492→    </Card>
   493→  )
   494→}

   496→// ──────────────────────────────────────────────────────────────

   499→function ChildRow({
   500→  child,
   501→  childAttrKeys,
   502→  productId,
   503→  showCost,
   504→  showPricing,
   505→  canEditCost,
   506→  canEditPrice,
   507→  queryClient,
   508→  onEdit,
   509→  inventory,
   510→}: {
   511→  child: ChildVariant
   512→  childAttrKeys: string[]
   513→  productId: string
   514→  showCost: boolean
   515→  showPricing: boolean
   516→  canEditCost: boolean
   517→  canEditPrice: boolean
   518→  queryClient: ReturnType<typeof useQueryClient>
   519→  onEdit: () => void
   520→  inventory?: InventorySummaryVariant
   521→}) {
   522→  const [costValue, setCostValue] = useState(String(child.costPrice))
   523→  const [saleValue, setSaleValue] = useState(child.salePrice != null ? String(child.salePrice) : '')
   524→  const [compareValue, setCompareValue] = useState(child.comparePrice != null ? String(child.comparePrice) : '')
   525→  const [weightValue, setWeightValue] = useState(child.weightKg != null ? String(child.weightKg) : '')
   526→  const [saving, setSaving] = useState(false)
   527→  const [toggling, setToggling] = useState(false)
   528→
   529→  async function saveCost() {
   530→    const newCost = Number(costValue)
   531→    if (isNaN(newCost) || newCost < 0) return
   532→    if (newCost === child.costPrice) return
   533→    setSaving(true)
   534→    try {
   535→      await api.post(`/api/products/${productId}/variants/${child.variantId}/override-cost`, { cost_price: newCost })
   536→      toast.success('Cost overridden — no longer synced with parent')
   537→      queryClient.invalidateQueries({ queryKey: ['variant-groups', productId] })
   538→      queryClient.invalidateQueries({ queryKey: ['product', productId] })
   539→    } catch (err) {
   540→      toast.error(err instanceof FetchError ? err.message : 'Failed to override cost')
   541→      setCostValue(String(child.costPrice))
   542→    } finally { setSaving(false) }
   543→  }
   544→
   545→  async function saveWeight() {
   546→    const newWeight = weightValue ? Number(weightValue) : null
   547→    if (weightValue && (isNaN(newWeight as number) || (newWeight as number) < 0)) return
   548→    if (newWeight === child.weightKg) return
   549→    setSaving(true)
   550→    try {
   551→      await api.post(`/api/products/${productId}/variants/${child.variantId}/override-weight`, { weightKg: newWeight })
   552→      toast.success('Weight overridden — no longer synced with parent')
   553→      queryClient.invalidateQueries({ queryKey: ['variant-groups', productId] })
   554→      queryClient.invalidateQueries({ queryKey: ['product', productId] })
   555→    } catch (err) {
   556→      toast.error(err instanceof FetchError ? err.message : 'Failed to override weight')
   557→      setWeightValue(child.weightKg != null ? String(child.weightKg) : '')
   558→    } finally { setSaving(false) }
   559→  }
   560→
   561→  async function resyncWeight() {
   562→    setSaving(true)
   563→    try {

   573→  async function savePrice() {

   583→    } catch (err) {

   585→      setSaleValue(child.salePrice != null ? String(child.salePrice) : '')
   586→      setCompareValue(child.comparePrice != null ? String(child.comparePrice) : '')
   587→    } finally { setSaving(false) }

   729→function VariantEditDialog({
   730→  variant,
   731→  productId,
   732→  onClose,
   733→  queryClient,
   734→}: {
   735→  variant: ChildVariant
   736→  productId: string
   737→  onClose: () => void
   738→  queryClient: ReturnType<typeof useQueryClient>
   739→}) {
   740→  const [form, setForm] = useState({
   741→    sku: variant.sku,
   742→    barcode: variant.barcode ?? '',
   743→    costPrice: String(variant.costPrice),
   744→    weightGrams: String(variant.weightGrams ?? 0),
   745→    weightKg: variant.weightKg != null ? String(variant.weightKg) : '',
   746→    stitchingCharges: String(variant.stitchingCharges ?? 0),
   747→    productionDays: String(variant.productionDays ?? 0),
   748→    isTaxable: variant.isTaxable ?? true,
   749→    requiresShipping: variant.requiresShipping ?? true,
   750→  })
   751→  const [saving, setSaving] = useState(false)
   752→  const [confirmSkuChange, setConfirmSkuChange] = useState(false)
   753→
   754→  async function save() {
   755→    // If SKU changed, show confirmation
   756→    if (form.sku !== variant.sku && !confirmSkuChange) {
   757→      setConfirmSkuChange(true)
   758→      return
   759→    }
   760→
   761→    setSaving(true)
   762→    try {
   763→      const patch: Record<string, unknown> = {}
   764→      if (form.sku !== variant.sku) patch.sku = form.sku
   765→      if (form.barcode !== (variant.barcode ?? '')) patch.barcode = form.barcode
   766→      if (Number(form.costPrice) !== variant.costPrice) patch.cost_price = Number(form.costPrice)
   767→      if (Number(form.weightGrams) !== (variant.weightGrams ?? 0)) patch.weight_grams = Number(form.weightGrams)
   768→      // Weight (kg) — null when empty, otherwise the number
   769→      const newWeightKg = form.weightKg === '' ? null : Number(form.weightKg)
   770→      const oldWeightKg = variant.weightKg ?? null
   771→      if (newWeightKg !== oldWeightKg) patch.weight_kg = newWeightKg
   772→      if (Number(form.stitchingCharges) !== (variant.stitchingCharges ?? 0)) patch.stitching_charges = Number(form.stitchingCharges)
   773→      if (Number(form.productionDays) !== (variant.productionDays ?? 0)) patch.production_days = Number(form.productionDays)
   774→      if (form.isTaxable !== (variant.isTaxable ?? true)) patch.is_taxable = form.isTaxable
   775→      if (form.requiresShipping !== (variant.requiresShipping ?? true)) patch.requires_shipping = form.requiresShipping
   776→
   777→      if (Object.keys(patch).length === 0) {
   778→        onClose()
   779→        return
   780→      }
   781→
   782→      await api.patch(`/api/products/${productId}/variants/${variant.variantId}`, patch)
   783→      toast.success('Variant updated')
   784→      queryClient.invalidateQueries({ queryKey: ['variant-groups', productId] })
   785→      queryClient.invalidateQueries({ queryKey: ['product', productId] })
   786→      onClose()
   787→    } catch (err) {
   788→      toast.error(err instanceof FetchError ? err.message : 'Failed to update variant')
   789→    } finally {
   790→      setSaving(false)
   791→      setConfirmSkuChange(false)
   792→    }
   793→  }
   794→
   795→  return (
   796→    <Dialog open onOpenChange={(v) => !v && onClose()}>
   797→      <DialogContent className="max-w-lg">
   798→        <DialogHeader>
   799→          <DialogTitle>Edit Variant</DialogTitle>
   800→        </DialogHeader>
   801→
   802→        {confirmSkuChange && (
   803→          <div className="rounded-lg border border-amber-300 bg-amber-50 p-3 text-sm text-amber-800">
   804→            <p className="font-medium">Change SKU?</p>
   805→            <p className="text-xs mt-1">Changing the SKU won&apos;t affect history but may cause confusion with existing labels. Continue?</p>
   806→          </div>
   807→        )}
   808→
   809→        <div className="grid grid-cols-2 gap-4">
   810→          <div className="space-y-1.5">
   811→            <Label className="text-xs">SKU</Label>
   812→            <Input value={form.sku} onChange={(e) => setForm({ ...form, sku: e.target.value })} className="h-9 text-sm font-mono" />
   813→          </div>
   814→          <div className="space-y-1.5">
   815→            <Label className="text-xs">Barcode</Label>
   816→            <Input value={form.barcode} onChange={(e) => setForm({ ...form, barcode: e.target.value })} className="h-9 text-sm" placeholder="UPC/EAN" />
   817→          </div>
   818→          <div className="space-y-1.5">
   819→            <Label className="text-xs">Cost Price</Label>
   820→            <Input type="number" min="0" step="0.01" value={form.costPrice} onChange={(e) => setForm({ ...form, costPrice: e.target.value })} className="h-9 text-sm" />
   821→          </div>
   822→          <div className="space-y-1.5">
   823→            <Label className="text-xs">Weight (grams)</Label>
   824→            <Input type="number" min="0" step="1" value={form.weightGrams} onChange={(e) => setForm({ ...form, weightGrams: e.target.value })} className="h-9 text-sm" />
   825→          </div>
   826→          <div className="space-y-1.5">
   827→            <Label className="text-xs">Weight (kg)</Label>
   828→            <Input type="number" min="0" step="0.001" value={form.weightKg} onChange={(e) => setForm({ ...form, weightKg: e.target.value })} className="h-9 text-sm" placeholder="0.000" />
   829→          </div>
   830→          <div className="space-y-1.5">
   831→            <Label className="text-xs">Stitching Charges</Label>
   832→            <Input type="number" min="0" step="0.01" value={form.stitchingCharges} onChange={(e) => setForm({ ...form, stitchingCharges: e.target.value })} className="h-9 text-sm" />
   833→          </div>
   834→          <div className="space-y-1.5">
   835→            <Label className="text-xs">Production Days</Label>
   836→            <Input type="number" min="0" step="1" value={form.productionDays} onChange={(e) => setForm({ ...form, productionDays: e.target.value })} className="h-9 text-sm" />
   837→          </div>
   838→          <div className="flex items-center gap-2">
   839→            <Switch checked={form.isTaxable} onCheckedChange={(v) => setForm({ ...form, isTaxable: v })} />
   840→            <Label className="text-xs">Taxable</Label>
   841→          </div>
   842→          <div className="flex items-center gap-2">
   843→            <Switch checked={form.requiresShipping} onCheckedChange={(v) => setForm({ ...form, requiresShipping: v })} />
   844→            <Label className="text-xs">Requires Shipping</Label>
   845→          </div>
   846→        </div>
   847→
   848→        <div className="flex items-center gap-2 pt-2 border-t">
   849→          <FulfillmentTypeBadge type={variant.fulfillmentType} />
   850→          <span className="text-xs text-muted-foreground">Fulfillment type cannot be changed after creation</span>
   851→        </div>
   852→
   853→        <DialogFooter>
   854→          <Button variant="ghost" onClick={onClose}>Cancel</Button>
   855→          <Button onClick={save} disabled={saving}>
   856→            {saving ? <Loader2 className="h-4 w-4 animate-spin" /> : <Check className="h-4 w-4" />}
   857→            {confirmSkuChange ? 'Confirm SKU Change & Save' : 'Save Changes'}
   858→          </Button>
   859→        </DialogFooter>
   860→      </DialogContent>
   861→    </Dialog>
   862→  )
   863→}

   937→function FlatRow({
   938→  child,
   939→  attrValues,
   940→  productId,
   941→  showCost,
   942→  showPricing,
   943→  canEditCost,
   944→  canEditPrice,
   945→  queryClient,
   946→  onEdit,
   947→  inventory,
   948→}: {
   949→  child: ChildVariant
   950→  attrValues: string
   951→  productId: string
   952→  showCost: boolean
   953→  showPricing: boolean
   954→  canEditCost: boolean
   955→  canEditPrice: boolean
   956→  queryClient: ReturnType<typeof useQueryClient>
   957→  onEdit: () => void
   958→  inventory?: InventorySummaryVariant
   959→}) {
   960→  const [costValue, setCostValue] = useState(String(child.costPrice))
   961→  const [saleValue, setSaleValue] = useState(child.salePrice != null ? String(child.salePrice) : '')
   962→  const [compareValue, setCompareValue] = useState(child.comparePrice != null ? String(child.comparePrice) : '')

   966→  async function saveCost() {

   969→    try {
   970→      await api.post(`/api/products/${productId}/variants/${child.variantId}/override-cost`, { cost_price: newCost })
   971→      toast.success('Cost updated')
   972→      queryClient.invalidateQueries({ queryKey: ['variant-groups', productId] })
   973→    } catch (err) { toast.error(err instanceof FetchError ? err.message : 'Failed') }
   974→  }
   975→
   976→  async function saveWeight() {
   977→    const newWeight = weightValue ? Number(weightValue) : null
   978→    if (weightValue && (isNaN(newWeight as number) || (newWeight as number) < 0)) return
   979→    if (newWeight === child.weightKg) return
   980→    try {
   981→      await api.post(`/api/products/${productId}/variants/${child.variantId}/override-weight`, { weightKg: newWeight })
   982→      toast.success('Weight updated')
   983→      queryClient.invalidateQueries({ queryKey: ['variant-groups', productId] })
   984→    } catch (err) { toast.error(err instanceof FetchError ? err.message : 'Failed') }
   985→  }
   986→
   987→  async function savePrice() {
   988→    const newSale = Number(saleValue)
   989→    if (isNaN(newSale) || newSale < 0) return
   990→    try {
   991→      await api.post(`/api/products/${productId}/variants/${child.variantId}/override-price`, { sale_price: newSale, compare_price: compareValue ? Number(compareValue) : null })
   992→      toast.success('Price updated')
   993→      queryClient.invalidateQueries({ queryKey: ['variant-groups', productId] })
   994→    } catch (err) { toast.error(err instanceof FetchError ? err.message : 'Failed') }
   995→  }
   996→
   997→  async function toggleActive() {
   998→    setToggling(true)
   999→    try {
  1000→      await api.post(`/api/products/${productId}/variants/${child.variantId}/toggle`, { is_active: !child.isActive })
  1001→      toast.success(child.isActive ? 'Deactivated' : 'Activated')
  1002→      queryClient.invalidateQueries({ queryKey: ['variant-groups', productId] })
