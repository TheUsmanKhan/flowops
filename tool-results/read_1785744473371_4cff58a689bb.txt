    92→interface GeneratedVariant {
    93→  sku: string
    94→  attribute_values: Record<string, string>
    95→  cost_price: number

   102→  is_default: boolean

   104→  is_active: boolean
   105→  compare_price: number | null
   106→  // Parent-child pricing cascade flags — default true for freshly generated
   107→  // variants; flipped to false when a user overrides an individual child.
   108→  cost_price_synced_with_parent: boolean

   110→  compare_price_synced_with_parent: boolean
   111→  opening_stock_qty?: number
   112→  opening_stock_cost?: number
   113→  opening_stock_location_id?: string

   115→  // fulfillment_type. fulfillment_type never changes (describes HOW the
   116→  // variant is normally fulfilled); track_inventory changes when opening
   117→  // stock is added to a made_to_order variant (flips to true). The badge
   118→  // in the Opening Stock table reads THIS field, not fulfillment_type.
   119→  // Default: stock_based → true, made_to_order → false. Flipped to true
   120→  // optimistically in local state when the user enters Qty > 0 for an MTO
   121→  // variant's opening stock.
   122→  track_inventory: boolean
   123→}

   283→    // eslint-disable-next-line react-hooks/exhaustive-deps

   303→      setSavingDraft(false)

   305→  }, [draftId, step, title, shortDescription, description, productType,

   339→  // ---- Helpers
   340→  function validateStep(s: number): string | null {
   341→    if (s === 0) {
   342→      if (title.trim().length < 2) return 'Product title must be at least 2 characters.'
   343→      if (shortDescription.length > 500) return 'Short description must be 500 characters or fewer.'
   344→    }
   345→    if (s === 1) {
   346→      const variants = collectVariantsForValidation()

   349→          return 'Pick at least one attribute and value to generate variants.'
   350→        }
   351→        return 'Add at least one variant.'
   352→      }
   353→      for (const v of variants) {
   354→        if (!v.sku.trim()) return 'Every variant needs an SKU.'
   355→        if (typeof v.sale_price !== 'number' || v.sale_price < 0) {
   356→          return `Variant ${v.sku || '(unnamed)'} needs a valid sale price.`
   357→        }
   358→        if (typeof v.cost_price !== 'number' || v.cost_price < 0) {
   359→          return `Variant ${v.sku || '(unnamed)'} needs a valid cost price.`

   361→      }
   362→    }
   363→    return null
   364→  }
   365→
   366→  function collectVariantsForValidation(): VariantDraft[] {
   367→    if (productType === 'simple') return [simpleVariant]
   368→    if (productType === 'variable' && isStitchable) {
   369→      return generatedVariants.map((g) => ({
   370→        sku: g.sku,
   371→        barcode: '',
   372→        attribute_values: g.attribute_values,
   373→        cost_price: g.cost_price,
   374→        stitching_charges: g.stitching_charges,
   375→        compare_price: g.compare_price,
   376→        weight_grams: 0,
   377→        fulfillment_type: g.fulfillment_type as 'stock_based' | 'made_to_order',
   378→        stitching_type: g.stitching_type as VariantDraft['stitching_type'],
   379→        production_days: g.production_days,
   380→        sale_price: g.sale_price,
   381→        is_active: g.is_active,
   382→        is_default: g.is_default,
   383→        // CRITICAL: propagate the user-entered opening stock values from the
   384→        // Mode B preview table — previously these were hardcoded to zero/empty,

   386→        // Only stock_based variants qualify for opening stock. A variant is

   390→          g.fulfillment_type === 'stock_based' &&
   391→          !!g.opening_stock_location_id &&
   392→          Number(g.opening_stock_qty ?? 0) > 0,
   393→        opening_stock_qty: Number(g.opening_stock_qty ?? 0),
   394→        opening_stock_cost: Number(g.opening_stock_cost ?? 0),
   395→        opening_stock_location_id: g.opening_stock_location_id ?? '',
   396→        fabric_source_variant_id: null,
   397→      }))
   398→    }
   399→    if (productType === 'variable') return regularVariants
   400→    // bundle / service — use simple variant form
   401→    return [simpleVariant]
   402→  }

   404→  function goNext() {
   405→    setSubmitError(null)
   406→    const err = validateStep(step)
   407→    if (err) {
   408→      setSubmitError(err)
   409→      return
   410→    }
   411→    setStep((s) => Math.min(s + 1, STEPS.length - 1))
   412→  }
   413→
   414→  // ---- Generic attribute mode: regenerate variants whenever the attribute
   415→  // selection changes. Calls the generate endpoint with the selection and a
   416→  // dummy product id (the route doesn't use the id — it accepts product_slug
   417→  // in the body and returns pure combinations).
   418→  const lastReqIdRef = useRef(0)
   419→  const handleAttributeChange = useCallback(
   420→    (selection: SelectionState) => {
   421→      setAttributeSelection(selection)
   422→      if (selection.selectedAttributes.length === 0) {
   423→        setGeneratedVariants([])
   424→        return
   425→      }
   426→      const reqId = ++lastReqIdRef.current
   427→      setGenerating(true)
   428→      api
   429→        .post<{
   430→          combinations: Array<{
   431→            attribute_values: Record<string, string>
   432→            suggested_sku: string
   433→            suggested_fulfillment_type: string
   434→          }>
   435→        }>(`/api/products/new/variants/generate`, {
   436→          product_slug: slug,
   437→          base_sku: baseSku.trim() || undefined,
   438→          selected_attributes: selection.selectedAttributes,
   439→        })
   440→        .then((res) => {
   441→          // Stale-response guard — keep only the latest reply.
   442→          if (reqId !== lastReqIdRef.current) return
   443→          // Preserve user edits (sale_price, is_active, cost_price) for
   444→          // existing SKUs; default new ones with a 30% markup on cost (=0).
   445→          setGeneratedVariants((prev) => {
   446→            const prevBySku = new Map(prev.map((v) => [v.sku, v]))
   447→            return res.combinations.map((c, i) => {
   448→              const existing = prevBySku.get(c.suggested_sku)
   449→              return {
   450→                sku: c.suggested_sku,
   451→                attribute_values: c.attribute_values,
   452→                cost_price: existing?.cost_price ?? 0,
   453→                stitching_charges: existing?.stitching_charges ?? 0,
   454→                fulfillment_type: c.suggested_fulfillment_type,
   455→                stitching_type:
   456→                  existing?.stitching_type ??
   457→                  (c.suggested_fulfillment_type === 'made_to_order'
   458→                    ? 'stitched_basic'
   459→                    : 'unstitched'),
   460→                production_days: existing?.production_days ?? 0,
   461→                requires_shipping: existing?.requires_shipping ?? true,
   462→                allow_backorder: existing?.allow_backorder ?? false,
   463→                is_default: existing?.is_default ?? i === 0,
   464→                sale_price: existing?.sale_price ?? 0,
   465→                is_active: existing?.is_active ?? true,
   466→                compare_price: existing?.compare_price ?? null,
   467→                // Default synced=true for new variants; preserve for existing
   468→                cost_price_synced_with_parent: existing?.cost_price_synced_with_parent ?? true,
   469→                sale_price_synced_with_parent: existing?.sale_price_synced_with_parent ?? true,
   470→                compare_price_synced_with_parent: existing?.compare_price_synced_with_parent ?? true,
   471→                // Preserve opening stock edits across regenerations
   472→                opening_stock_qty: existing?.opening_stock_qty,
   473→                opening_stock_cost: existing?.opening_stock_cost,
   474→                opening_stock_location_id: existing?.opening_stock_location_id,
   475→                // Bug 2 fix: track_inventory defaults based on fulfillment_type
   476→                // (stock_based → true, made_to_order → false). Preserved for
   477→                // existing variants so the user's opening stock entry isn't
   478→                // lost on regeneration.
   479→                track_inventory:
   480→                  existing?.track_inventory ??
   481→                  c.suggested_fulfillment_type === 'stock_based',
   482→              }
   483→            })
   484→          })
   485→        })
   486→        .catch((err) => {
   487→          if (reqId !== lastReqIdRef.current) return
   488→          const msg =
   489→            err instanceof FetchError
   490→              ? err.message
   491→              : 'Failed to generate variant combinations.'
   492→          setSubmitError(msg)
   493→          toast.error(msg)
   494→        })
   495→        .finally(() => {
   496→          if (reqId === lastReqIdRef.current) setGenerating(false)
   497→        })
   498→    },
   499→    [slug, baseSku],
   500→  )
   501→
   502→  // ---- Submit final product
   503→  async function submit() {
   504→    setSubmitError(null)
   505→    for (const s of [0, 1, 2]) {
   506→      const err = validateStep(s)
   507→      if (err) {
   508→        setSubmitError(err)
   509→        setStep(s)
   510→        return
   511→      }
   512→    }
   513→
   514→    // Clone the drafts so we can normalize the default flag without mutating state.
   515→    const variants = collectVariantsForValidation().map<VariantDraft>((v) => ({ ...v }))
   516→
   517→    // Ensure at least one variant is the default.
   518→    if (!variants.some((v) => v.is_default) && variants.length > 0) {
   519→      variants[0].is_default = true
   520→    }
   521→
   522→    const payload = {
   523→      title: title.trim(),
   524→      base_sku: baseSku.trim() || undefined,
   525→      description: description.trim() || undefined,
   526→      short_description: shortDescription.trim() || undefined,
   527→      product_type: productType,
   528→      category_id: categoryId || undefined,
   529→      brand_id: brandId || undefined,
   530→      product_scope: productScope,
   531→      is_stitchable: isStitchable,
   532→      stitching_base_price: 0,
   533→      has_size_variants:
   534→        isStitchable &&
   535→        productType === 'variable' &&
   536→        attributeSelection.selectedAttributes.some(
   537→          (a) => a.attribute_name.toLowerCase() === 'size',
   538→        ),
   539→      is_active: true,
   540→      is_featured: isFeatured,
   541→      variants: variants.map((v) => ({
   542→        sku: v.sku.trim(),
   543→        barcode: v.barcode?.trim() || undefined,
   544→        attribute_values: v.attribute_values,
   545→        cost_price: Number(v.cost_price) || 0,
   546→        stitching_charges: Number(v.stitching_charges) || 0,
   547→        compare_price: v.compare_price ?? undefined,
   548→        weight_grams: Number(v.weight_grams) || 0,
   549→        fulfillment_type: v.fulfillment_type,
   550→        stitching_type: v.stitching_type ?? undefined,
   551→        production_days: Number(v.production_days) || 0,
   552→        allow_backorder: false,
   553→        requires_shipping: true,
   554→        is_taxable: true,
   555→        is_active: v.is_active,
   556→        is_default: v.is_default,
   557→        sale_price: Number(v.sale_price) || 0,
   558→        fabric_source_variant_id: v.fabric_source_variant_id || undefined,

   571→      // 2. Identify variants that have opening stock data filled in. We use

   573→      //    Each variant can have its OWN location_id now — no more forced
   574→      //    single-location batching.
   575→      const variantsWithOpeningStock = variants
   576→        .map((v, i) => ({ variant: v, variantId: res.variantIds?.[i] }))
   577→        .filter(
   578→          ({ variant, variantId }) =>
   579→            variantId &&
   580→            variant.has_opening_stock &&
   581→            variant.opening_stock_qty > 0 &&
   582→            variant.opening_stock_location_id,
   583→        )
   584→
   585→      // 3. Call the dedicated opening-stock endpoint PER VARIANT ROW. Each
   586→      //    call is awaited (not fire-and-forget) so we can surface per-variant
   587→      //    failures clearly. The endpoint wraps processInventoryTransaction()
   588→      //    — the SAME function every other inventory movement uses — so this
   589→      //    produces real inventory_pools + inventory_transactions rows
   590→      //    identical to Receive Stock / PO receiving.
   591→      const failedVariants: Array<{ sku: string; reason: string }> = []
   592→      const succeededVariants: Array<{ sku: string; locationId: string }> = []
   593→      const touchedLocationIds = new Set<string>()
   594→
   595→      for (const { variant, variantId } of variantsWithOpeningStock) {
   596→        try {
   597→          const result = await api.post<{
   598→            success: boolean
   599→            transaction_id: string
   600→            location_id: string
   601→          }>('/api/inventory/opening-stock', {
   602→            org_variant_id: variantId,
   603→            location_id: variant.opening_stock_location_id,
   604→            quantity: variant.opening_stock_qty,
   605→            // Cost is set EXACTLY ONCE per variant — via the parent-group
   606→            // cascade or individual override in the grouped pricing table above.
   607→            // The Opening Stock section no longer has its own cost input; it
   608→            // uses the variant's current cost_price directly, so the two can
   609→            // never diverge. See Bug 2 fix.
   610→            cost_per_unit: variant.cost_price,
   611→            notes: `Opening stock for ${res.title}`,
   612→          })
   613→          if (result?.success) {
   614→            succeededVariants.push({ sku: variant.sku, locationId: variant.opening_stock_location_id })
   615→            touchedLocationIds.add(variant.opening_stock_location_id)
   616→          } else {
   617→            failedVariants.push({ sku: variant.sku, reason: 'Unknown error' })

   619→        } catch (err) {

   621→          failedVariants.push({ sku: variant.sku, reason })

   626→      //    was created successfully, but if any opening stock entry failed,
   627→      //    the user MUST see exactly which variant failed and why.

   645→      //    dashboard, transactions ledger, product detail (multiple keys
   646→      //    including the pricing tab + inventory tab), per-location views,
   647→      //    and the products list. This guarantees no stale data anywhere.
   648→      queryClient.invalidateQueries({ queryKey: ['products'] })
   649→      queryClient.invalidateQueries({ queryKey: ['inventory-pools'] })
   650→      queryClient.invalidateQueries({ queryKey: ['inventory-transactions'] })
   651→      queryClient.invalidateQueries({ queryKey: ['inventory-dashboard'] })
   652→      queryClient.invalidateQueries({ queryKey: ['product', res.id] })
   653→      queryClient.invalidateQueries({ queryKey: ['product-inventory', res.id] })

   656→      for (const locId of touchedLocationIds) {
   657→        queryClient.invalidateQueries({ queryKey: ['location', locId] })
   658→      }
   659→
   660→      toast.success(`"${res.title}" created.`)
   661→      setHasChanges(false) // Reset guard — no false-positive prompt after saving
   662→      // Phase 10: Delete the draft now that the real product is created
   663→      if (draftId) {
   664→        await api.delete(`/api/drafts?id=${draftId}`).catch(() => {})
   665→        setDraftId(undefined)
   666→      }
   667→      navigate({ name: 'product-detail', id: res.id })
   668→    } catch (err) {
   669→      const msg =
   670→        err instanceof FetchError ? err.message : 'Failed to create product.'
   671→      setSubmitError(msg)
   672→      toast.error(msg)
   673→    } finally {
   674→      setSubmitting(false)
   675→    }
