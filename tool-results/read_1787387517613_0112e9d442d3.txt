   107→   107→/**
   108→   108→ * Fetches live inventory per variant for this product and returns a map keyed
   109→   109→ * by variantId. Used by the parent-child table to display on-hand stock
   110→   110→ * alongside cost/price columns. Reads from the same inventory_pools data as
   111→   111→ * the Inventory tab and the Inventory Dashboard — no parallel data source.
   112→   112→ */
   113→   113→function useVariantInventoryMap(productId: string) {
   114→   114→  const { data } = useQuery<InventorySummaryResponse>({
   115→   115→    queryKey: ['product-inventory', productId],
   116→   116→    queryFn: () => api.get(`/api/inventory/summary?product_id=${productId}`),
   117→   117→    staleTime: 15_000,
   118→   118→  })
   119→   119→  const map = new Map<string, InventorySummaryVariant>()
   120→   120→  for (const v of data?.variants ?? []) {
   121→   121→    map.set(v.variantId, v)
   122→   122→  }
   123→   123→  return map
   124→   124→}

   126→   126→/** Compact stock cell — shows on_hand, with available in muted subtext. */
   127→   127→function StockCell({ inv }: { inv?: InventorySummaryVariant }) {
   128→   128→  if (!inv) {
   129→   129→    return <span className="text-xs text-muted-foreground">—</span>
   130→   130→  }
   131→   131→  const onHand = inv.totalOnHand
   132→   132→  const available = inv.totalAvailable
   133→   133→  if (onHand === 0) {
   134→   134→    return <span className="text-xs text-muted-foreground">0</span>
   135→   135→  }
   136→   136→  const lowStock = available > 0 && available <= 5
   137→   137→  return (
   138→   138→    <div className="text-xs leading-tight">
   139→   139→      <span className={cn('font-medium', lowStock ? 'text-amber-700' : 'text-foreground')}>{onHand}</span>
   140→   140→      {available !== onHand && (
   141→   141→        <div className="text-[10px] text-muted-foreground">{available} avail.</div>
   142→   142→      )}
   143→   143→    </div>
   144→   144→  )
   145→   145→}

   182→   182→export function ParentChildVariantTable({
   183→   183→  productId,
   184→   184→  mode = 'both',
   185→   185→}: {
   186→   186→  productId: string
   187→   187→  mode?: 'cost' | 'pricing' | 'both'
   188→   188→}) {
   189→   189→  const can = useCan()
   190→   190→  const queryClient = useQueryClient()
   191→   191→  const [expandedGroups, setExpandedGroups] = useState<Set<string>>(new Set())
   192→   192→  const [editingVariant, setEditingVariant] = useState<ChildVariant | null>(null)
   193→   193→
   194→   194→  const canEditCost = can(PERMISSIONS.PRODUCTS_EDIT)
   195→   195→  const canEditPrice = can(PERMISSIONS.PRODUCTS_PRICING)
   196→   196→  const showCost = mode === 'cost' || mode === 'both'
   197→   197→  const showPricing = mode === 'pricing' || mode === 'both'
   198→   198→
   199→   199→  const { data, isLoading, isError, refetch } = useQuery<VariantGroupsResponse>({
   200→   200→    queryKey: ['variant-groups', productId],
   201→   201→    queryFn: () => api.get(`/api/products/${productId}/variant-groups`),
   202→   202→    staleTime: 15_000,
   203→   203→  })
   204→   204→
   205→   205→  // Live inventory per variant — powers the read-only Stock column.
   206→   206→  const inventoryMap = useVariantInventoryMap(productId)
   207→   207→
   208→   208→  function toggleGroup(parentValue: string) {
   209→   209→    setExpandedGroups((prev) => {
   210→   210→      const next = new Set(prev)
   211→   211→      if (next.has(parentValue)) next.delete(parentValue)
   212→   212→      else next.add(parentValue)
   213→   213→      return next
   214→   214→    })
   215→   215→  }
   216→   216→
   217→   217→  // Expand all groups by default on first load
   218→   218→  if (data && expandedGroups.size === 0 && data.groups.length > 0) {
   219→   219→    setExpandedGroups(new Set(data.groups.map((g) => g.parentValue)))
   220→   220→  }
   221→   221→
   222→   222→  if (isLoading) {
   223→   223→    return (
   224→   224→      <Card>
   225→   225→        <CardContent className="p-6 space-y-3">
   226→   226→          {[0, 1, 2].map((i) => (
   227→   227→            <div key={i} className="h-20 rounded bg-muted/50 animate-pulse" />
   228→   228→          ))}
   229→   229→        </CardContent>
   230→   230→      </Card>
   231→   231→    )
   232→   232→  }
   233→   233→
   234→   234→  if (isError) {
   235→   235→    return (
   236→   236→      <Card>
   237→   237→        <CardContent className="p-6 text-center">
   238→   238→          <AlertCircle className="h-8 w-8 mx-auto text-muted-foreground mb-2" />
   239→   239→          <p className="text-sm text-muted-foreground mb-3">Failed to load variant groups.</p>
   240→   240→          <Button variant="outline" size="sm" onClick={() => refetch()}>Retry</Button>
   241→   241→        </CardContent>
   242→   242→      </Card>
   243→   243→    )
   244→   244→  }
   245→   245→
   246→   246→  if (!data || data.groups.length === 0) {
   247→   247→    return (
   248→   248→      <Card>
   249→   249→        <CardContent className="p-6 text-center text-sm text-muted-foreground">
   250→   250→          No variants found for this product.
   251→   251→        </CardContent>
   252→   252→      </Card>
   253→   253→    )
   254→   254→  }
   255→   255→
   256→   256→  // If only one attribute, render flat table (no grouping)
   257→   257→  if (!data.hasMultipleAttributes) {
   258→   258→    return (
   259→   259→      <>
   260→   260→        <FlatVariantTable
   261→   261→          groups={data.groups}
   262→   262→          productId={productId}
   263→   263→          showCost={showCost}
   264→   264→          showPricing={showPricing}
   265→   265→          canEditCost={canEditCost}
   266→   266→          canEditPrice={canEditPrice}
   267→   267→          queryClient={queryClient}
   268→   268→          onEdit={(v) => setEditingVariant(v)}
   269→   269→          inventoryMap={inventoryMap}
   270→   270→        />
   271→   271→        {editingVariant && (
   272→   272→          <VariantEditDialog
   273→   273→            variant={editingVariant}
   274→   274→            productId={productId}
   275→   275→            onClose={() => setEditingVariant(null)}
   276→   276→            queryClient={queryClient}
   277→   277→          />
   278→   278→        )}
   279→   279→      </>
   280→   280→    )
   281→   281→  }
   282→   282→
   283→   283→  // Grouped table with parent-child cascade
   284→   284→  return (
   285→   285→    <div className="space-y-3">
   286→   286→      {data.parentAttributeName && (
   287→   287→        <p className="text-xs text-muted-foreground">
   288→   288→          Grouped by <span className="font-medium text-foreground">{data.parentAttributeDisplayName || data.parentAttributeName}</span> —
   289→   289→          set prices once per group, or override individual variants.
   290→   290→        </p>
   291→   291→      )}
   292→   292→      {data.groups.map((group) => (
   293→   293→        <GroupCard
   294→   294→          key={group.parentValue}
   295→   295→          group={group}
   296→   296→          productId={productId}
   297→   297→          parentAttributeName={data.parentAttributeName!}
   298→   298→          expanded={expandedGroups.has(group.parentValue)}
   299→   299→          onToggle={() => toggleGroup(group.parentValue)}
   300→   300→          showCost={showCost}
   301→   301→          showPricing={showPricing}
   302→   302→          canEditCost={canEditCost}
   303→   303→          canEditPrice={canEditPrice}
   304→   304→          queryClient={queryClient}
   305→   305→          onEditVariant={(v) => setEditingVariant(v)}
   306→   306→          inventoryMap={inventoryMap}
   307→   307→        />
   308→   308→      ))}
   309→   309→      {editingVariant && (
   310→   310→        <VariantEditDialog
   311→   311→          variant={editingVariant}
   312→   312→          productId={productId}
   313→   313→          onClose={() => setEditingVariant(null)}
   314→   314→          queryClient={queryClient}
   315→   315→        />
   316→   316→      )}
   317→   317→    </div>
   318→   318→  )

   324→   324→function GroupCard({
   325→   325→  group,
   326→   326→  productId,
   327→   327→  parentAttributeName,
   328→   328→  expanded,
   329→   329→  onToggle,
   330→   330→  showCost,
   331→   331→  showPricing,
   332→   332→  canEditCost,
   333→   333→  canEditPrice,
   334→   334→  queryClient,
   335→   335→  onEditVariant,
   336→   336→  inventoryMap,
   337→   337→}: {
   338→   338→  group: VariantGroup
   339→   339→  productId: string
   340→   340→  parentAttributeName: string
   341→   341→  expanded: boolean
   342→   342→  onToggle: () => void
   343→   343→  showCost: boolean
   344→   344→  showPricing: boolean
   345→   345→  canEditCost: boolean
   346→   346→  canEditPrice: boolean
   347→   347→  queryClient: ReturnType<typeof useQueryClient>
   348→   348→  onEditVariant: (v: ChildVariant) => void
   349→   349→  inventoryMap: Map<string, InventorySummaryVariant>
   350→   350→}) {
   351→   351→  const [parentCost, setParentCost] = useState('')
   352→   352→  const [parentSale, setParentSale] = useState('')
   353→   353→  const [parentCompare, setParentCompare] = useState('')
   354→   354→  const [parentWeight, setParentWeight] = useState('')
   355→   355→  const [applying, setApplying] = useState(false)
   356→   356→
   357→   357→  const firstChild = group.children[0]
   358→   358→  if (parentCost === '' && firstChild) setParentCost(String(firstChild.costPrice))
   359→   359→  if (parentSale === '' && firstChild?.salePrice != null) setParentSale(String(firstChild.salePrice))
   360→   360→  if (parentCompare === '' && firstChild?.comparePrice != null) setParentCompare(String(firstChild.comparePrice))
   361→   361→  if (parentWeight === '' && firstChild?.weightKg != null) setParentWeight(String(firstChild.weightKg))
   362→   362→
   363→   363→  // Bug 1 fix: ONE handler cascades ALL THREE fields (cost, sale, compare).
   364→   364→  // Calls both server endpoints (cost + sale-price) in sequence, then
   365→   365→  // invalidates the cache once. Each endpoint only updates children whose
   366→   366→  // relevant synced flag is true, so the three flags remain INDEPENDENT.
   367→   367→  // Weight cascade added — calls the weight endpoint in the same sequence.
   368→   368→  async function applyAllToGroup() {
   369→   369→    const cost = Number(parentCost)
   370→   370→    const sale = Number(parentSale)
   371→   371→    const compare = parentCompare ? Number(parentCompare) : null
   372→   372→    const weight = parentWeight ? Number(parentWeight) : null
   373→   373→
   374→   374→    if (isNaN(cost) || cost < 0) { toast.error('Enter a valid cost price'); return }
   375→   375→    if (isNaN(sale) || sale < 0) { toast.error('Enter a valid sale price'); return }
   376→   376→    if (parentWeight && (isNaN(weight as number) || (weight as number) < 0)) { toast.error('Enter a valid weight'); return }
   377→   377→
   378→   378→    setApplying(true)
   379→   379→    let costCount = 0
   380→   380→    let saleCount = 0
   381→   381→    let weightCount = 0
   382→   382→    try {
   383→   383→      // Cascade cost to cost-synced children
   384→   384→      const costRes = await api.post<{ success: boolean; updated_count: number }>(
   385→   385→        `/api/products/${productId}/variant-groups/dummy/cost`,
   386→   386→        { cost_price: cost, parent_attribute_name: parentAttributeName, parent_value: group.parentValue },
   387→   387→      )
   388→   388→      costCount = costRes.updated_count
   389→   389→
   390→   390→      // Cascade sale + compare to their respective synced children
   391→   391→      const saleRes = await api.post<{ success: boolean; updated_count: number }>(
   392→   392→        `/api/products/${productId}/variant-groups/dummy/sale-price`,
   393→   393→        { sale_price: sale, compare_price: compare, parent_attribute_name: parentAttributeName, parent_value: group.parentValue },
   394→   394→      )
   395→   395→      saleCount = saleRes.updated_count
   396→   396→
   397→   397→      // Cascade weight to weight-synced children (only if a weight was entered)
   398→   398→      if (parentWeight && weight != null) {
   399→   399→        const weightRes = await api.post<{ success: boolean; updated_count: number }>(
   400→   400→          `/api/products/${productId}/variant-groups/dummy/weight`,
   401→   401→          { weightKg: weight, parent_attribute_name: parentAttributeName, parent_value: group.parentValue },
   402→   402→        )
   403→   403→        weightCount = weightRes.updated_count
   404→   404→      }
   405→   405→
   406→   406→      toast.success(`Applied to group — Cost: ${costCount}, Sale+Compare: ${saleCount}${parentWeight ? `, Weight: ${weightCount}` : ''} variant(s)`)
   407→   407→      queryClient.invalidateQueries({ queryKey: ['variant-groups', productId] })
   408→   408→      queryClient.invalidateQueries({ queryKey: ['product', productId] })
   409→   409→    } catch (err) {
   410→   410→      toast.error(err instanceof FetchError ? err.message : 'Failed to apply to group')
   411→   411→    } finally { setApplying(false) }
   412→   412→  }
   413→   413→
   414→   414→  const childAttrKeys = group.children.length > 0
   415→   415→    ? Object.keys(group.children[0].attributeValues).filter((k) => k !== parentAttributeName)
   416→   416→    : []
   417→   417→
   418→   418→  return (
   419→   419→    <Card>
   420→   420→      <CardHeader className="pb-3 cursor-pointer" onClick={onToggle}>
   421→   421→        <div className="flex items-center justify-between">
   422→   422→          <div className="flex items-center gap-2">
   423→   423→            {expanded ? <ChevronDown className="h-4 w-4 text-muted-foreground" /> : <ChevronRight className="h-4 w-4 text-muted-foreground" />}
   424→   424→            <CardTitle className="text-base">{group.parentValue}</CardTitle>
   425→   425→            <Badge variant="secondary" className="text-[10px]">{group.childCount} variant{group.childCount === 1 ? '' : 's'}</Badge>
   426→   426→          </div>
   427→   427→        </div>
   428→   428→      </CardHeader>
   429→   429→
   430→   430→      {expanded && (
   431→   431→        <CardContent className="space-y-4">
   432→   432→          {/* Parent group inputs — uses the shared ParentGroupInputs component
   433→   433→              with a single "Apply to Group" button that cascades ALL fields
   434→   434→              (cost, sale, compare, weight) independently. Bug 1 fix + weight extension. */}
   435→   435→          <ParentGroupInputs
   436→   436→            parentCost={parentCost}
   437→   437→            parentSale={parentSale}
   438→   438→            parentCompare={parentCompare}
   439→   439→            parentWeight={parentWeight}
   440→   440→            onCostChange={setParentCost}
   441→   441→            onSaleChange={setParentSale}
   442→   442→            onCompareChange={setParentCompare}
   443→   443→            onWeightChange={setParentWeight}
   444→   444→            onApplyAll={applyAllToGroup}
   445→   445→            showCost={showCost}
   446→   446→            showPricing={showPricing}
   447→   447→            showWeight={canEditCost}
   448→   448→            canEditCost={canEditCost}
   449→   449→            canEditPrice={canEditPrice}
   450→   450→            canEditWeight={canEditCost}
   451→   451→            applying={applying}
   452→   452→          />
   453→   453→
   454→   454→          {/* Children table */}
   455→   455→          <div className="rounded-md border overflow-x-auto scrollbar-thin">
   456→   456→            <table className="w-full text-sm">
   457→   457→              <thead className="bg-muted/30">
   458→   458→                <tr className="text-left text-xs text-muted-foreground">
   459→   459→                  {childAttrKeys.map((k) => (<th key={k} className="px-3 py-2 font-medium">{k}</th>))}
   460→   460→                  <th className="px-3 py-2 font-medium">SKU</th>
   461→   461→                  <th className="px-3 py-2 font-medium">Fulfillment</th>
   462→   462→                  <th className="px-3 py-2 font-medium text-center">Stock</th>
   463→   463→                  {showCost && <th className="px-3 py-2 font-medium text-right">Cost</th>}
   464→   464→                  {showPricing && <th className="px-3 py-2 font-medium text-right">Sale</th>}
   465→   465→                  {showPricing && <th className="px-3 py-2 font-medium text-right">Compare</th>}
   466→   466→                  {canEditCost && <th className="px-3 py-2 font-medium text-right">Weight (kg)</th>}
   467→   467→                  <th className="px-3 py-2 font-medium text-center">Active</th>

   735→   735→  variant: ChildVariant
   736→   736→  productId: string
   737→   737→  onClose: () => void
   738→   738→  queryClient: ReturnType<typeof useQueryClient>
   739→   739→}) {
   740→   740→  const [form, setForm] = useState({
   741→   741→    sku: variant.sku,
   742→   742→    barcode: variant.barcode ?? '',
   743→   743→    costPrice: String(variant.costPrice),
   744→   744→    weightGrams: String(variant.weightGrams ?? 0),

   753→   753→
   754→   754→  async function save() {

   757→   757→      setConfirmSkuChange(true)

   759→   759→    }

   763→   763→      const patch: Record<string, unknown> = {}
   764→   764→      if (form.sku !== variant.sku) patch.sku = form.sku
   765→   765→      if (form.barcode !== (variant.barcode ?? '')) patch.barcode = form.barcode
   766→   766→      if (Number(form.costPrice) !== variant.costPrice) patch.cost_price = Number(form.costPrice)
   767→   767→      if (Number(form.weightGrams) !== (variant.weightGrams ?? 0)) patch.weight_grams = Number(form.weightGrams)

   779→   779→        return
   780→   780→      }

   782→   782→      await api.patch(`/api/products/${productId}/variants/${variant.variantId}`, patch)
   783→   783→      toast.success('Variant updated')
   784→   784→      queryClient.invalidateQueries({ queryKey: ['variant-groups', productId] })
   785→   785→      queryClient.invalidateQueries({ queryKey: ['product', productId] })

   787→   787→    } catch (err) {

   791→   791→      setConfirmSkuChange(false)
   792→   792→    }
   793→   793→  }

   802→   802→        {confirmSkuChange && (
   803→   803→          <div className="rounded-lg border border-amber-300 bg-amber-50 p-3 text-sm text-amber-800">
   804→   804→            <p className="font-medium">Change SKU?</p>
   805→   805→            <p className="text-xs mt-1">Changing the SKU won&apos;t affect history but may cause confusion with existing labels. Continue?</p>
   806→   806→          </div>
   807→   807→        )}
   808→   808→
   809→   809→        <div className="grid grid-cols-2 gap-4">
   810→   810→          <div className="space-y-1.5">
   811→   811→            <Label className="text-xs">SKU</Label>
   812→   812→            <Input value={form.sku} onChange={(e) => setForm({ ...form, sku: e.target.value })} className="h-9 text-sm font-mono" />
   813→   813→          </div>
   814→   814→          <div className="space-y-1.5">
   815→   815→            <Label className="text-xs">Barcode</Label>
   816→   816→            <Input value={form.barcode} onChange={(e) => setForm({ ...form, barcode: e.target.value })} className="h-9 text-sm" placeholder="UPC/EAN" />
   817→   817→          </div>
   818→   818→          <div className="space-y-1.5">
   819→   819→            <Label className="text-xs">Cost Price</Label>
   820→   820→            <Input type="number" min="0" step="0.01" value={form.costPrice} onChange={(e) => setForm({ ...form, costPrice: e.target.value })} className="h-9 text-sm" />

   822→   822→          <div className="space-y-1.5">

   937→   937→function FlatRow({
   938→   938→  child,
   939→   939→  attrValues,
   940→   940→  productId,
   941→   941→  showCost,
   942→   942→  showPricing,
   943→   943→  canEditCost,
   944→   944→  canEditPrice,
   945→   945→  queryClient,
   946→   946→  onEdit,
   947→   947→  inventory,
   948→   948→}: {
   949→   949→  child: ChildVariant
   950→   950→  attrValues: string
   951→   951→  productId: string
   952→   952→  showCost: boolean
   953→   953→  showPricing: boolean
   954→   954→  canEditCost: boolean
   955→   955→  canEditPrice: boolean
   956→   956→  queryClient: ReturnType<typeof useQueryClient>
   957→   957→  onEdit: () => void
   958→   958→  inventory?: InventorySummaryVariant
   959→   959→}) {
   960→   960→  const [costValue, setCostValue] = useState(String(child.costPrice))
   961→   961→  const [saleValue, setSaleValue] = useState(child.salePrice != null ? String(child.salePrice) : '')
   962→   962→  const [compareValue, setCompareValue] = useState(child.comparePrice != null ? String(child.comparePrice) : '')
   963→   963→  const [weightValue, setWeightValue] = useState(child.weightKg != null ? String(child.weightKg) : '')
   964→   964→  const [toggling, setToggling] = useState(false)
   965→   965→
   966→   966→  async function saveCost() {
   967→   967→    const newCost = Number(costValue)
   968→   968→    if (isNaN(newCost) || newCost < 0 || newCost === child.costPrice) return
   969→   969→    try {
   970→   970→      await api.post(`/api/products/${productId}/variants/${child.variantId}/override-cost`, { cost_price: newCost })
   971→   971→      toast.success('Cost updated')
   972→   972→      queryClient.invalidateQueries({ queryKey: ['variant-groups', productId] })
   973→   973→    } catch (err) { toast.error(err instanceof FetchError ? err.message : 'Failed') }
   974→   974→  }
   975→   975→
   976→   976→  async function saveWeight() {
   977→   977→    const newWeight = weightValue ? Number(weightValue) : null
   978→   978→    if (weightValue && (isNaN(newWeight as number) || (newWeight as number) < 0)) return
   979→   979→    if (newWeight === child.weightKg) return
   980→   980→    try {
   981→   981→      await api.post(`/api/products/${productId}/variants/${child.variantId}/override-weight`, { weightKg: newWeight })
   982→   982→      toast.success('Weight updated')
   983→   983→      queryClient.invalidateQueries({ queryKey: ['variant-groups', productId] })
   984→   984→    } catch (err) { toast.error(err instanceof FetchError ? err.message : 'Failed') }
   985→   985→  }
   986→   986→
   987→   987→  async function savePrice() {
   988→   988→    const newSale = Number(saleValue)
   989→   989→    if (isNaN(newSale) || newSale < 0) return
   990→   990→    try {
   991→   991→      await api.post(`/api/products/${productId}/variants/${child.variantId}/override-price`, { sale_price: newSale, compare_price: compareValue ? Number(compareValue) : null })
   992→   992→      toast.success('Price updated')
   993→   993→      queryClient.invalidateQueries({ queryKey: ['variant-groups', productId] })
   994→   994→    } catch (err) { toast.error(err instanceof FetchError ? err.message : 'Failed') }
   995→   995→  }
   996→   996→
   997→   997→  async function toggleActive() {
   998→   998→    setToggling(true)
   999→   999→    try {
  1000→  1000→      await api.post(`/api/products/${productId}/variants/${child.variantId}/toggle`, { is_active: !child.isActive })
  1001→  1001→      toast.success(child.isActive ? 'Deactivated' : 'Activated')
  1002→  1002→      queryClient.invalidateQueries({ queryKey: ['variant-groups', productId] })
  1003→  1003→      queryClient.invalidateQueries({ queryKey: ['product', productId] })
  1004→  1004→    } catch (err) { toast.error(err instanceof FetchError ? err.message : 'Failed') }
  1005→  1005→    finally { setToggling(false) }
  1006→  1006→  }
  1007→  1007→
  1008→  1008→  return (
  1009→  1009→    <tr className="border-t hover:bg-muted/20">
  1010→  1010→      <td className="px-3 py-2 text-xs">
  1011→  1011→        {attrValues || '—'}
  1012→  1012→        {child.isDefault && <Badge variant="secondary" className="ml-1 text-[9px]">DEFAULT</Badge>}
  1013→  1013→      </td>
  1014→  1014→      <td className="px-3 py-2">
  1015→  1015→        <span className="font-mono text-xs">{child.sku}</span>
  1016→  1016→        {child.barcode && <div className="text-[10px] text-muted-foreground">{child.barcode}</div>}
  1017→  1017→      </td>
  1018→  1018→      <td className="px-3 py-2"><TrackingBadge trackInventory={child.trackInventory} fulfillmentType={child.fulfillmentType} /></td>
  1019→  1019→      <td className="px-3 py-2 text-center"><StockCell inv={inventory} /></td>
  1020→  1020→      {showCost && (
  1021→  1021→        <td className="px-3 py-2 text-right">
  1022→  1022→          {canEditCost ? (
  1023→  1023→            <Input type="number" min="0" step="0.01" value={costValue} onChange={(e) => setCostValue(e.target.value)} onBlur={saveCost} className="h-7 w-20 text-xs text-right" />
  1024→  1024→          ) : <span className="text-xs">{child.costPrice}</span>}
  1025→  1025→        </td>
  1026→  1026→      )}
  1027→  1027→      {showPricing && (
  1028→  1028→        <td className="px-3 py-2 text-right">
  1029→  1029→          {canEditPrice ? (
  1030→  1030→            <Input type="number" min="0" step="0.01" value={saleValue} onChange={(e) => setSaleValue(e.target.value)} onBlur={savePrice} className="h-7 w-20 text-xs text-right" />
  1031→  1031→          ) : <span className="text-xs">{child.salePrice ?? '—'}</span>}
  1032→  1032→        </td>
  1033→  1033→      )}
  1034→  1034→      {showPricing && (
  1035→  1035→        <td className="px-3 py-2 text-right text-xs text-muted-foreground">{child.comparePrice ?? '—'}</td>
  1036→  1036→      )}
  1037→  1037→      {canEditCost && (
  1038→  1038→        <td className="px-3 py-2 text-right">
  1039→  1039→          {canEditCost ? (
  1040→  1040→            <Input
  1041→  1041→              type="number"
  1042→  1042→              min="0"
  1043→  1043→              step="0.001"
  1044→  1044→              value={weightValue}
  1045→  1045→              onChange={(e) => setWeightValue(e.target.value)}
  1046→  1046→              onBlur={saveWeight}
  1047→  1047→              className="h-7 w-20 text-xs text-right"
  1048→  1048→              placeholder="—"
  1049→  1049→            />
  1050→  1050→          ) : (
  1051→  1051→            <span className="text-xs">{child.weightKg ?? '—'}</span>
  1052→  1052→          )}
  1053→  1053→        </td>
  1054→  1054→      )}
  1055→  1055→      <td className="px-3 py-2 text-center">
  1056→  1056→        {canEditCost ? (
  1057→  1057→          <Switch checked={child.isActive} onCheckedChange={toggleActive} disabled={toggling} />
  1058→  1058→        ) : (
  1059→  1059→          <Badge variant={child.isActive ? 'secondary' : 'outline'} className="text-[10px]">{child.isActive ? 'Active' : 'Inactive'}</Badge>
  1060→  1060→        )}
  1061→  1061→      </td>
  1062→  1062→      {canEditCost && (
  1063→  1063→        <td className="px-3 py-2">
  1064→  1064→          <Button size="sm" variant="ghost" className="h-7 w-7 p-0" onClick={onEdit} title="Edit variant details">
  1065→  1065→            <Pencil className="h-3 w-3" />
  1066→  1066→          </Button>
  1067→  1067→        </td>
  1068→  1068→      )}
  1069→  1069→    </tr>
  1070→  1070→  )
  1071→  1071→}
