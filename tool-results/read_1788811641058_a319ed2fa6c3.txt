     1→/**
     2→ * Leopard Courier Adapter — REAL implementation.
     3→ *
     4→ * Replaces the previous stub. Implements all required CourierAdapter methods
     5→ * plus the optional capabilities (fetchOperationalCities, createPickupAddress,
     6→ * fetchExistingPickupAddresses).
     7→ *
     8→ * Base URLs:
     9→ *   Staging:    https://merchantapistaging.leopardscourier.com/api/
    10→ *   Production: https://merchantapi.leopardscourier.com/api/
    11→ *
    12→ * Auth: api_key + api_password, sent as part of the request body (JSON) — NOT
    13→ * as a header. Both fields stored in the encrypted credentials blob.
    14→ *
    15→ * All endpoints use JSON format (/format/json/).
    16→ *
    17→ * Weight: Leopard requires GRAMS. The internal system always uses KG — the
    18→ * adapter converts at the boundary (weightKg × 1000 → integer grams).
    19→ *
    20→ * City IDs: Leopard uses NUMERIC city IDs (integers). The adapter resolves
    21→ * city names to Leopard's numeric IDs via the cached courier_operational_cities
    22→ * table (cityId column stores the numeric ID as a string).
    23→ *
    24→ * No orderType/Normal-Overland-Replacement concept applies to Leopard —
    25→ * Leopard's shipment_type field is a different thing (optional, defaults to
    26→ * "overnight"). Do NOT build any Leopard-specific order-type logic.
    27→ */

   142→  constructor(credentials: Record<string, string>) {
   143→    this.apiKey = credentials.api_key ?? ''
   144→    this.apiPassword = credentials.api_password ?? ''
   145→    if (!this.apiKey || !this.apiPassword) {
   146→      throw new Error('Leopard adapter requires both api_key and api_password in credentials.')
   147→    }
   148→    // Use production if credentials explicitly say so, otherwise staging.
   149→    // The isProduction field is a boolean in configSchema, but it can arrive as:
   150→    //   - "true" / "false" (string from JSON)
   151→    //   - true / false (boolean — JSON.parse keeps it as boolean)
   152→    //   - "on" (HTML checkbox default)
   153→    //   - undefined (field left empty — defaults to staging)
   154→    const isProd = credentials.isProduction === true
   155→      || credentials.isProduction === 'true'
   156→      || credentials.isProduction === 'on'
   157→      || credentials.isProduction === '1'
   158→      || credentials.isProduction === 1
   159→    this.baseUrl = isProd ? LEOPARD_PRODUCTION_BASE : LEOPARD_STAGING_BASE
   160→  }

   176→
   177→  private async post<T = unknown>(endpoint: string, body: Record<string, unknown>): Promise<LeopardResponse<T>> {
   178→    const url = `${this.baseUrl}${endpoint}/format/json/`
   179→    const response = await fetch(url, {
   180→      method: 'POST',
   181→      headers: { 'Content-Type': 'application/json' },
   182→      body: JSON.stringify(body),
   183→    })
   184→
   185→    const json = (await response.json()) as LeopardResponse<T>
   186→    return json
   187→  }

   193→  private async getWithParams<T = unknown>(endpoint: string, params: Record<string, string>): Promise<LeopardResponse<T>> {
   194→    const queryString = new URLSearchParams(params).toString()
   195→    const url = `${this.baseUrl}${endpoint}/format/json/?${queryString}`
   196→    const response = await fetch(url, {
   197→      method: 'GET',
   198→      headers: { 'Content-Type': 'application/json' },
   199→    })
   200→
   201→    const json = (await response.json()) as LeopardResponse<T>
   202→    return json
   203→  }

   206→  // Helper: extract error message from Leopard response
   207→  // ──────────────────────────────────────────────────────────────
   208→
   209→  private extractError(resp: LeopardResponse): string {
   210→    if (typeof resp.error === 'string') return resp.error
   211→    if (resp.error && typeof resp.error === 'object') {
   212→      // Error is an object mapping field → message
   213→      return Object.entries(resp.error)
   214→        .map(([k, v]) => `${k}: ${v}`)
   215→        .join('; ')
   216→    }
   217→    return resp.message || 'Unknown Leopard API error'
   218→  }

   221→  // 1. fetchOperationalCities — GET/POST getAllCities

   224→  async fetchOperationalCities(): Promise<OperationalCity[]> {
   225→    const resp = await this.post<LeopardCity[]>('getAllCities', this.authBody())
   226→
   227→    if (resp.status !== 1 && resp.status !== '1') {
   228→      throw new Error(this.extractError(resp))
   229→    }
   230→
   231→    const cities = resp.city_list
   232→    if (!cities || !Array.isArray(cities) || cities.length === 0) {
   233→      throw new Error('Leopard cities API returned 0 cities. Check the API credentials.')
   234→    }
   235→
   236→    return cities.map((c) => ({
   237→      cityName: c.name,
   238→      // Store the numeric ID as a string (matches the cityId TEXT column)
   239→      cityId: String(c.id),
   240→      // allow_as_origin = pickup city, allow_as_destination = delivery city
   241→      isPickupCity: c.allow_as_origin === true,
   242→      isDeliveryCity: c.allow_as_destination === true,
   243→      // Store the shipment_type array as a JSON string (the sync action
   244→      // will persist it to the shipmentTypes column)
   245→      // Note: OperationalCity type doesn't have shipmentTypes, but the
   246→      // sync action reads rawResponse for extra fields. We include it
   247→      // there via a custom approach — see syncCourierOperationalCities.
   248→    }))

   265→  // ──────────────────────────────────────────────────────────────
   266→  // 2. bookShipment — POST bookPacket
   267→  // ──────────────────────────────────────────────────────────────
   268→
   269→  async bookShipment(input: BookShipmentInput): Promise<BookShipmentResult> {
   270→    // City validation is handled by the CALLER via revalidateCityAtBookingTime()
   271→    // — the adapter doesn't have the integration ID for the live fallback.
   272→
   273→    // ── Resolve city IDs ──
   274→    // Leopard requires NUMERIC city IDs (integers), NOT city name strings.
   275→    // 'self' = use the shipper's own city. For origin, we use 'self' if no
   276→    // explicit pickup city override is needed. For destination, we resolve
   277→    // the delivery city name to Leopard's numeric ID from the cached
   278→    // courier_operational_cities table.
   279→    //
   280→    // The caller (booking action) passes pickupLocationCity + deliveryCity
   281→    // as city NAME strings. We need to resolve them to numeric IDs.
   282→    // However, the adapter doesn't have DB access — so we accept the
   283→    // pickupAddressCode (which Leopard maps to the shipper's registered
   284→    // city via shipment_id) and use 'self' for origin.
   285→    //
   286→    // For destination, the caller should pass the numeric cityId via the
   287→    // deliveryCity field IF it's already resolved, OR we use a special
   288→    // convention: if deliveryCity is a numeric string, send it as-is;
   289→    // otherwise, we can't resolve it here and must throw.
   290→    //
   291→    // DESIGN DECISION: The booking action (caller) is responsible for
   292→    // resolving the delivery city NAME to Leopard's numeric cityId BEFORE
   293→    // calling the adapter. It passes the numeric ID in a new field on
   294→    // BookShipmentInput. But since we can't add fields to the interface
   295→    // without breaking other adapters, we use the existing deliveryCity
   296→    // field: if it's numeric, send as integer; if it's a name, throw with
   297→    // a clear error.
   298→    //
   299→    // ACTUALLY: The cleanest approach is to resolve in the adapter using
   300→    // the DB. But the adapter shouldn't have DB access. So we resolve
   301→    // in the booking action and pass the numeric ID as deliveryCity
   302→    // (overloading the field). This is documented in the booking action.
   303→
   304→    const destinationCity = input.deliveryCity
   305→    // If destinationCity is not numeric, we can't proceed
   306→    if (!/^\d+$/.test(destinationCity)) {
   307→      return {
   308→        success: false,
   309→        error: `Leopard booking requires a numeric destination city ID, but got '${destinationCity}'. The booking action must resolve the city name to Leopard's numeric cityId before calling the adapter.`,
   310→      }
   311→    }
   312→
   313→    // Origin city: use 'self' (shipper's own city) by default
   314→    const originCity = 'self'
   315→
   316→    // ── Weight conversion: KG → grams ──
   317→    // input.weightGrams is ALREADY in grams (despite the name, the booking
   318→    // action passes grams). But to be safe, we ensure it's an integer.
   319→    const weightGrams = Math.max(1, Math.round(input.weightGrams))
   320→
   321→    // ── Build the request body ──
   322→    const body: Record<string, unknown> = {
   323→      ...this.authBody(),
   324→      booked_packet_weight: weightGrams,
   325→      booked_packet_no_piece: input.quantity ?? 1,
   326→      booked_packet_collect_amount: Math.round(input.codAmount),
   327→      booked_packet_order_id: input.orderNumber,
   328→      origin_city: originCity,
   329→      destination_city: parseInt(destinationCity, 10),
   330→      // Shipper info: use 'self' so Leopard uses the shipper's registered info
   331→      shipment_id: input.pickupAddressCode ?? undefined, // numeric shipper ID
   332→      shipment_name_eng: 'self',
   333→      shipment_email: 'self',
   334→      shipment_phone: 'self',
   335→      shipment_address: 'self',
   336→      // Consignee (customer) info
   337→      consignment_name_eng: input.recipientName,
   338→      consignment_email: '', // optional
   339→      consignment_phone: input.recipientPhone,
   340→      consignment_phone_two: '', // optional
   341→      consignment_phone_three: '', // optional
   342→      consignment_address: input.deliveryAddress,
   343→      special_instructions: input.transactionNotes ?? '',
   344→      // Shipment type: optional, defaults to "overnight" if empty
   345→      shipment_type: input.shipmentType ?? '',
   346→    }
   347→
   348→    // Optional: return address override (Leopard-specific)
   349→    if (input.returnAddressOverride) {
   350→      body.return_address = input.returnAddressOverride.address
   351→      // return_city needs to be a numeric city ID — resolve from the override
   352→      // For now, if the override's cityName is numeric, use it; otherwise omit
   353→      // (the booking action should resolve it)
   354→      if (/^\d+$/.test(input.returnAddressOverride.cityName)) {
   355→        body.return_city = parseInt(input.returnAddressOverride.cityName, 10)
   356→      }
   357→    }
   358→
   359→    // Remove undefined fields (Leopard rejects some empty fields)
   360→    const cleanBody: Record<string, unknown> = {}
   361→    for (const [k, v] of Object.entries(body)) {
   362→      if (v !== undefined && v !== '') {
   363→        cleanBody[k] = v
   364→      }
   365→    }
   366→
   367→    const resp = await this.post('bookPacket', cleanBody)
   368→
   369→    if (resp.status === 1 || resp.status === '1') {
   370→      const trackingNumber = resp.track_number
   371→      if (!trackingNumber) {
   372→        return {
   373→          success: false,
   374→          error: 'Leopard returned success but no track_number in response.',
   375→          rawResponse: resp,
   376→        }
   377→      }
   378→
   379→      return {
   380→        success: true,
   381→        trackingNumber,
   382→        providerStatus: 'Booked', // Leopard doesn't return an initial status string
   383→        slipLink: resp.slip_link ?? undefined,
   384→        rawResponse: resp,
   385→      }
   386→    }
   387→
   388→    return {
   389→      success: false,
   390→      error: this.extractError(resp),
   391→      rawResponse: resp,
   392→    }
   393→  }

   396→  // 3. trackShipment — GET/POST trackBookedPacket
   397→  // ──────────────────────────────────────────────────────────────
   398→
   399→  async trackShipment(trackingNumber: string): Promise<TrackShipmentResult> {
   400→    const body = {
   401→      ...this.authBody(),
   402→      track_numbers: trackingNumber,
   403→    }
   404→
   405→    const resp = await this.post<LeopardTrackPacket[]>('trackBookedPacket', body)
   406→
   407→    if (resp.status !== 1 && resp.status !== '1') {
   408→      return {
   409→        success: false,
   410→        error: this.extractError(resp),
   411→        rawResponse: resp,
   412→      }
   413→    }
   414→
   415→    const packets = resp.packet_list
   416→    if (!packets || packets.length === 0) {
   417→      return {
   418→        success: false,
   419→        error: 'Leopard returned success but no packet data.',
   420→        rawResponse: resp,
   421→      }
   422→    }
   423→
   424→    const packet = packets[0]
   425→    const rawStatus = packet.booked_packet_status ?? ''
   426→
   427→    // Map the raw status string to our generic status enum.
   428→    // Full status mapping will be built in Prompt 7 — for now, pass through
   429→    // the raw status string and do a basic mapping.
   430→    let genericStatus: TrackShipmentResult['status'] = 'booked'
   431→    const lowerStatus = rawStatus.toLowerCase()
   432→    if (lowerStatus.includes('deliver') && !lowerStatus.includes('undeliver')) {
   433→      genericStatus = 'delivered'
   434→    } else if (lowerStatus.includes('return') || lowerStatus.includes('rto')) {
   435→      genericStatus = 'returned'
   436→    } else if (lowerStatus.includes('cancel') || lowerStatus.includes('reject')) {
   437→      genericStatus = 'failed'
   438→    } else if (lowerStatus.includes('transit') || lowerStatus.includes('dispatch') || lowerStatus.includes('pickup')) {
   439→      genericStatus = 'in_transit'
   440→    }
   441→
   442→    return {
   443→      success: true,
   444→      status: genericStatus,
   445→      lastUpdateAt: packet.activity_date ?? undefined,
   446→      rawResponse: {
   447→        ...packet,
   448→        rawStatus, // pass through the raw status string for Prompt 7's mapping
   449→        trackingDetail: packet['Tracking Detail'] ?? [],
   450→      },
   451→    }
   452→  }

   487→  // ──────────────────────────────────────────────────────────────

   489→  /**
   490→   * Parse Leopard's webhook push payload.
   491→   *
   492→   * Leopard pushes: { "data": [{ cn_number, status, receiver_name, reason, activity_date }, ...] }
   493→   *
   494→   * Returns the FIRST update's tracking number + mapped status (for the
   495→   * generic webhook route's ParseStatusWebhookResult compatibility). The
   496→   * FULL array is processed by processLeopardWebhookUpdates() in the webhook
   497→   * route handler, which handles ALL updates.
   498→   *
   499→   * If the payload has no data array or is empty, throws a clear error.
   500→   */
   501→  async parseStatusWebhook(rawPayload: unknown): Promise<ParseStatusWebhookResult> {
   502→    const payload = rawPayload as { data?: LeopardWebhookUpdate[] }
   503→
   504→    if (!payload?.data || !Array.isArray(payload.data) || payload.data.length === 0) {
   505→      throw new Error('Leopard webhook payload missing or empty "data" array')
   506→    }
   507→
   508→    // Return the first update (for compatibility with the generic webhook route).
   509→    // The full array is processed by processLeopardWebhookUpdates() in the route handler.
   510→    const firstUpdate = payload.data[0]
   511→    if (!firstUpdate.cn_number || !firstUpdate.status) {
   512→      throw new Error('Leopard webhook update missing cn_number or status')
   513→    }
   514→
   515→    // Map the status to our generic status enum
   516→    const { mapLeopardStatus } = await import('./leopard.status-map')
   517→    const mapping = mapLeopardStatus(firstUpdate.status)
   518→
   519→    let genericStatus: ParseStatusWebhookResult['status'] = 'booked'
   520→    if (mapping.triggerDelivered) genericStatus = 'delivered'
   521→    else if (mapping.triggerRto) genericStatus = 'returned'
   522→    else if (mapping.triggerDispatch) genericStatus = 'in_transit'
   523→
   524→    return {
   525→      trackingNumber: firstUpdate.cn_number,
   526→      status: genericStatus,
   527→      lastUpdateAt: firstUpdate.activity_date,
   528→    }
   529→  }

   549→    return true
   550→  }

   560→  async createPickupAddress(input: PickupAddressInput): Promise<PickupAddressResult> {
   561→    // Leopard's createShipper requires a city_id (numeric). The caller passes
   562→    // cityName — we need to resolve it to a numeric ID. But the adapter doesn't
   563→    // have DB access. The caller (address-book action) should resolve the city
   564→    // name to Leopard's numeric cityId BEFORE calling the adapter, and pass it
   565→    // via cityName (overloading the field to accept numeric IDs).
   566→    const cityId = input.cityName
   567→    if (!/^\d+$/.test(cityId)) {
   568→      return {
   569→        success: false,
   570→        error: `Leopard createShipper requires a numeric city_id, but got '${cityId}'. The address-book action must resolve the city name to Leopard's numeric cityId before calling the adapter.`,
   571→      }
   572→    }
   573→
   574→    const body: Record<string, unknown> = {
   575→      ...this.authBody(),
   576→      shipment_name: input.contactPersonName,
   577→      shipment_email: '', // optional
   578→      shipment_phone: input.phone1,
   579→      shipment_address: input.address,
   580→      city_id: parseInt(cityId, 10),
   581→      // Optional bank fields — omitted (not required for address creation)
   582→      // Optional return_address — if the caller provides it via the override
   583→    }
   584→
   585→    // Add return_address if provided in the input (Leopard-specific extension)
   586→    // The PickupAddressInput doesn't have returnAddressOverride, so we check
   587→    // if the caller passed it via a custom field. For now, we omit it —
   588→    // the address-book action will handle the override separately.
   589→
   590→    const resp = await this.post<LeopardShipper[]>('createShipper', body)
   591→
   592→    if (resp.status === 1 || resp.status === '1') {
   593→      // Extract the shipment_id from the response data
   594→      const data = resp.data
   595→      let shipmentId: string | undefined
   596→
   597→      if (Array.isArray(data) && data.length > 0) {
   598→        shipmentId = String(data[0].shipment_id)
   599→      } else if (data && typeof data === 'object' && !Array.isArray(data)) {
   600→        // Single object response
   601→        const shipper = data as LeopardShipper
   602→        shipmentId = String(shipper.shipment_id)
   603→      }
   604→
   605→      if (!shipmentId) {
   606→        return {
   607→          success: false,
   608→          error: 'Leopard createShipper returned success but no shipment_id in response.',
   609→        }
   610→      }
   611→
   612→      return {
   613→        success: true,
   614→        providerAddressCode: shipmentId,
   615→      }
   616→    }
   617→
   618→    // Check for "already exists" case
   619→    if (resp.message && resp.message.toLowerCase().includes('already exists')) {
   620→      return {
   621→        success: false,
   622→        error: 'A shipper with these details already exists. Use getShipperDetails to find the existing shipment_id.',
   623→      }
   624→    }
   625→
   626→    return {
   627→      success: false,
   628→      error: this.extractError(resp),
   629→    }
   630→  }
   631→
   632→  // ──────────────────────────────────────────────────────────────
   633→  // 9. fetchExistingPickupAddresses — GET getShipperDetails (ALL)
   634→  // ──────────────────────────────────────────────────────────────
   635→
   636→  async fetchExistingPickupAddresses(): Promise<Array<{
   637→    providerAddressCode: string
   638→    label?: string
   639→    address: string
   640→    cityName: string
   641→    contactPersonName: string
   642→    phone1: string
   643→    phone2?: string
   644→  }>> {
   645→    // getShipperDetails uses GET with query params
   646→    const params = {
   647→      api_key: this.apiKey,
   648→      api_password: this.apiPassword,
   649→      // request_param and request_value are optional — if omitted, returns all shippers
   650→    }
   651→
   652→    const resp = await this.getWithParams<LeopardShipper[] | LeopardShipper>('getShipperDetails', params)
   653→
   654→    if (resp.status !== 1 && resp.status !== '1') {
   655→      throw new Error(this.extractError(resp))
   656→    }
   657→
   658→    // Leopard's API returns data as either:
   659→    // - An array of shippers (when fetching all)
   660→    // - A SINGLE shipper object (when filtering or only one shipper exists)
   661→    // We must handle BOTH shapes.
   662→    const data = resp.data
   663→    if (!data) {
   664→      return []
   665→    }
   666→
   667→    let shippers: LeopardShipper[] = []
   668→    if (Array.isArray(data)) {
   669→      shippers = data
   670→    } else if (typeof data === 'object') {
   671→      shippers = [data]
   672→    }
   673→
   674→    return shippers.map((s) => this.mapShipper(s))
   675→  }
   676→
   677→  // ──────────────────────────────────────────────────────────────
   678→  // 9b. fetchShipperById — GET getShipperDetails?request_param=shipment_id&request_value={id}
   679→  //
   680→  // Per Leopard API PDF (page 73-78): getShipperDetails supports filtering
   681→  // by a single shipper via request_param + request_value query params.
   682→  // This is the CORRECT way to import a single shipper — NOT fetch-all-filter.
   683→  //
   684→  // Returns a single shipper or null if not found.
   685→  // ──────────────────────────────────────────────────────────────
   686→
   687→  async fetchShipperById(shipmentId: string): Promise<{
   688→    providerAddressCode: string
   689→    label: string
   690→    address: string
   691→    cityName: string
   692→    contactPersonName: string
   693→    phone1: string
   694→    phone2?: string
   695→  } | null> {
   696→    const params = {
   697→      api_key: this.apiKey,
   698→      api_password: this.apiPassword,
   699→      request_param: 'shipment_id',
   700→      request_value: shipmentId,
   701→    }
   702→
   703→    const resp = await this.getWithParams<LeopardShipper[] | LeopardShipper>('getShipperDetails', params)
   704→
   705→    if (resp.status !== 1 && resp.status !== '1') {
   706→      throw new Error(this.extractError(resp))
   707→    }
   708→
   709→    // Leopard's API returns data as either:
   710→    // - An array of shippers (when fetching all)
   711→    // - A SINGLE shipper object (when filtering by request_param=request_value)
   712→    // We must handle BOTH shapes.
   713→    const data = resp.data
   714→    if (!data) {
   715→      return null
   716→    }
   717→
   718→    let shipper: LeopardShipper | null = null
   719→    if (Array.isArray(data)) {
   720→      // Array response — take the first match
   721→      if (data.length === 0) return null
   722→      shipper = data[0]
   723→    } else if (typeof data === 'object') {
   724→      // Single object response (the common case for request_param filtering)
   725→      shipper = data
   726→    }
   727→
   728→    if (!shipper) return null
   729→    return this.mapShipper(shipper)
   730→  }
   731→
   732→  // ──────────────────────────────────────────────────────────────
   733→  // Helper: map a raw LeopardShipper to the normalized pickup address shape
   734→  // ──────────────────────────────────────────────────────────────
   735→
   736→  private mapShipper(s: LeopardShipper): {
   737→    providerAddressCode: string
   738→    label: string
   739→    address: string
   740→    cityName: string
   741→    contactPersonName: string
   742→    phone1: string
   743→    phone2?: string
   744→  } {
   745→    return {
   746→      providerAddressCode: String(s.shipment_id),
   747→      label: s.shipment_name_eng ?? s.shipment_name ?? 'Shipper',
   748→      address: s.shipment_address ?? '',
   749→      // cityName: we don't have the name, only the city_id. The caller
   750→      // (sync/import/refresh action) will resolve the name from
   751→      // courier_operational_cities.
   752→      cityName: String(s.shipper_city_id ?? s.city_id ?? ''),
   753→      contactPersonName: s.shipment_contact_person ?? s.shipment_name_eng ?? s.shipment_name ?? '',
   754→      phone1: String(s.shipment_phone ?? ''),
   755→    }
   756→  }

   758→  // ──────────────────────────────────────────────────────────────
   759→  // 10. pingConnection — read-only connectivity check (test route)
   760→  //
   761→  // Per Leopard API PDF: getAllCities is the lightest read-only endpoint
   762→  // that requires valid credentials. If it returns ≥1 city, credentials
   763→  // are valid and the API is reachable. This is the real "Test" action.
   764→  // ──────────────────────────────────────────────────────────────
   765→
   766→  async pingConnection(): Promise<{ success: boolean; error?: string }> {
   767→    try {
   768→      const cities = await this.fetchOperationalCities()
   769→      if (cities.length === 0) {
   770→        return {
   771→          success: false,
   772→          error: 'Leopard accepted the credentials but returned 0 cities.',
   773→        }
   774→      }
   775→      return { success: true }
   776→    } catch (e) {
   777→      return {
   778→        success: false,
   779→        error: e instanceof Error ? e.message : 'Leopard connectivity check failed.',
   780→      }
   781→    }
   782→  }
