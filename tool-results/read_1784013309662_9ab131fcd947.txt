   745→   745→function PricingTab({
   746→   746→  productId,
   747→   747→  variants,
   748→   748→  onUpdate,
   749→   749→}: {
   750→   750→  productId: string
   751→   751→  variants: ProductVariant[]
   752→   752→  onUpdate: () => void
   753→   753→}) {
   754→   754→  const [drafts, setDrafts] = useState<Record<string, { salePrice: number | null; comparePrice: number | null }>>(
   755→   755→    () =>
   756→   756→      Object.fromEntries(
   757→   757→        variants.map((v) => [
   758→   758→          v.id,
   759→   759→          { salePrice: v.salePrice, comparePrice: v.comparePrice },
   760→   760→        ]),
   761→   761→      ),
   762→   762→  )
   763→   763→  const [saving, setSaving] = useState<string | null>(null)
   764→   764→
   765→   765→  function setDraft(
   766→   766→    variantId: string,
   767→   767→    field: 'salePrice' | 'comparePrice',
   768→   768→    value: number | null,
   769→   769→  ) {
   770→   770→    setDrafts((d) => ({
   771→   771→      ...d,
   772→   772→      [variantId]: { ...d[variantId], [field]: value },
   773→   773→    }))
   774→   774→  }
   775→   775→
   776→   776→  async function save(variantId: string) {
   777→   777→    const draft = drafts[variantId]
   778→   778→    if (!draft) return
   779→   779→    setSaving(variantId)
   780→   780→    try {
   781→   781→      await api.patch(`/api/products/${productId}/variants/${variantId}/pricing`, {
   782→   782→        sale_price: draft.salePrice,
   783→   783→        compare_price: draft.comparePrice,
   784→   784→      })
   785→   785→      toast.success('Pricing updated.')
   786→   786→      onUpdate()
   787→   787→    } catch (err) {
   788→   788→      toast.error(err instanceof FetchError ? err.message : 'Failed to update pricing.')
   789→   789→    } finally {
   790→   790→      setSaving(null)
   791→   791→    }
   792→   792→  }
   793→   793→
   794→   794→  return (
   795→   795→    <Card>
   796→   796→      <CardHeader>
   797→   797→        <CardTitle className="text-base flex items-center gap-2">
   798→   798→          <Tag className="h-4 w-4 text-primary" /> Your company pricing
   799→   799→        </CardTitle>
   800→   800→        <CardDescription>
   801→   801→          Set the sale and compare-at prices for your storefront.
   802→   802→        </CardDescription>
   803→   803→      </CardHeader>
   804→   804→      <CardContent className="p-0">
   805→   805→        <div className="overflow-x-auto">
   806→   806→          <Table>
   807→   807→            <TableHeader>
   808→   808→              <TableRow>
   809→   809→                <TableHead>SKU</TableHead>
   810→   810→                <TableHead className="text-right">Cost</TableHead>
   811→   811→                <TableHead>Sale price</TableHead>
   812→   812→                <TableHead>Compare price</TableHead>
   813→   813→                <TableHead className="text-right">Action</TableHead>
   814→   814→              </TableRow>
   815→   815→            </TableHeader>
   816→   816→            <TableBody>
   817→   817→              {variants.map((v) => {
   818→   818→                const draft = drafts[v.id] ?? { salePrice: v.salePrice, comparePrice: v.comparePrice }
   819→   819→                const dirty =
   820→   820→                  draft.salePrice !== v.salePrice ||
   821→   821→                  draft.comparePrice !== v.comparePrice
   822→   822→                return (
   823→   823→                  <TableRow key={v.id}>
   824→   824→                    <TableCell className="font-mono text-xs">{v.sku}</TableCell>
   825→   825→                    <TableCell className="text-right text-xs text-muted-foreground">
   826→   826→                      {formatMoney(v.costPrice)}
   827→   827→                    </TableCell>
   828→   828→                    <TableCell>
   829→   829→                      <Input
   830→   830→                        type="number"
   831→   831→                        min="0"
   832→   832→                        step="0.01"
   833→   833→                        value={draft.salePrice ?? ''}
   834→   834→                        onChange={(e) =>
   835→   835→                          setDraft(
   836→   836→                            v.id,
   837→   837→                            'salePrice',
   838→   838→                            e.target.value ? Number(e.target.value) : null,
   839→   839→                          )
   840→   840→                        }
   841→   841→                        className="h-8 w-32 text-xs"
   842→   842→                      />
   843→   843→                    </TableCell>
   844→   844→                    <TableCell>
   845→   845→                      <Input
   846→   846→                        type="number"
   847→   847→                        min="0"
   848→   848→                        step="0.01"
   849→   849→                        value={draft.comparePrice ?? ''}
   850→   850→                        onChange={(e) =>
   851→   851→                          setDraft(
   852→   852→                            v.id,
   853→   853→                            'comparePrice',
   854→   854→                            e.target.value ? Number(e.target.value) : null,
   855→   855→                          )
   856→   856→                        }
   857→   857→                        className="h-8 w-32 text-xs"
   858→   858→                      />
   859→   859→                    </TableCell>
   860→   860→                    <TableCell className="text-right">
   861→   861→                      <Button
   862→   862→                        size="sm"
   863→   863→                        variant={dirty ? 'default' : 'outline'}
   864→   864→                        disabled={!dirty || saving === v.id}
   865→   865→                        onClick={() => save(v.id)}
   866→   866→                      >
   867→   867→                        {saving === v.id ? (
   868→   868→                          <Loader2 className="h-3.5 w-3.5 animate-spin" />
   869→   869→                        ) : (
   870→   870→                          <Save className="h-3.5 w-3.5" />
   871→   871→                        )}
   872→   872→                        Save
   873→   873→                      </Button>
   874→   874→                    </TableCell>
   875→   875→                  </TableRow>
   876→   876→                )
   877→   877→              })}
   878→   878→            </TableBody>
   879→   879→          </Table>
   880→   880→        </div>
   881→   881→      </CardContent>
   882→   882→    </Card>
   883→   883→  )
   884→   884→}
