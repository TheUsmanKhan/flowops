   202→export function ProductCreateView({ onBack }: { onBack: () => void }) {
   203→  const navigate = useAppStore((s) => s.navigate)
   204→  const queryClient = useQueryClient()
   205→
   206→  const [step, setStep] = useState(0)
   207→  const [submitting, setSubmitting] = useState(false)
   208→  const [submitError, setSubmitError] = useState<string | null>(null)
   209→
   210→  // Step 1 state
   211→  const [title, setTitle] = useState('')
   212→  const [shortDescription, setShortDescription] = useState('')
   213→  const [description, setDescription] = useState('')
   214→  const [productType, setProductType] = useState<ProductType>('variable')
   215→  const [categoryId, setCategoryId] = useState('')
   216→  const [brandId, setBrandId] = useState('')
   217→  const [baseSku, setBaseSku] = useState('')
   218→  const [isFeatured, setIsFeatured] = useState(false)
   219→  const [isStitchable, setIsStitchable] = useState(false)
   220→
   221→  // Step 2 — simple mode
   222→  const [simpleVariant, setSimpleVariant] = useState<VariantDraft>(blankSimpleVariant())
   223→
   224→  // Step 2 — stitchable mode
   225→  const [includeUnstitched, setIncludeUnstitched] = useState(true)
   226→  const [baseFabricCost, setBaseFabricCost] = useState<number>(0)
   227→  const [includeSizes, setIncludeSizes] = useState(false)
   228→  const [selectedSizes, setSelectedSizes] = useState<string[]>([])
   229→  const [customSizes, setCustomSizes] = useState<string[]>([])
   230→  const [customSizeInput, setCustomSizeInput] = useState('')
   231→  const [stitchingTypes, setStitchingTypes] = useState<
   232→    Array<{
   233→      type: 'stitched_basic' | 'stitched_heavy' | 'custom_order'
   234→      enabled: boolean
   235→      charge: number
   236→      productionDays: number
   237→    }>
   238→  >([
   239→    { type: 'stitched_basic', enabled: true, charge: 0, productionDays: DEFAULT_PRODUCTION_DAYS.stitched_basic },
   240→    { type: 'stitched_heavy', enabled: false, charge: 0, productionDays: DEFAULT_PRODUCTION_DAYS.stitched_heavy },
   241→    { type: 'custom_order', enabled: false, charge: 0, productionDays: DEFAULT_PRODUCTION_DAYS.custom_order },
   242→  ])
   243→  const [generatedVariants, setGeneratedVariants] = useState<
   244→    Array<GeneratedVariant & { sale_price: number; is_active: boolean }>
   245→  >([])
   246→  const [generating, setGenerating] = useState(false)
   247→
   248→  // Step 2 — regular variable mode
   249→  const [regularVariants, setRegularVariants] = useState<VariantDraft[]>([
   250→    blankRegularVariant(1),
   251→  ])
   252→
   253→  // Step 3 state
   254→  const [productScope, setProductScope] = useState<ProductScope>('private')
   255→
   256→  // ---- Derived: slug for variant SKU generation
   257→  const slug = useMemo(() => slugify(title || 'product'), [title])
   258→
   259→  // ---- Queries: categories & brands
   260→  const { data: catData } = useQuery<{ categories: CategoryPublic[] }>({
   261→    queryKey: ['categories'],
   262→    queryFn: () => api.get<{ categories: CategoryPublic[] }>('/api/categories'),
   263→    staleTime: 60_000,
   264→  })
   265→  const { data: brandData } = useQuery<{ brands: BrandPublic[] }>({
   266→    queryKey: ['brands'],
   267→    queryFn: () => api.get<{ brands: BrandPublic[] }>('/api/brands'),
   268→    staleTime: 60_000,
   269→  })
   270→
   271→  // Reset stitching flag if product type changes away from variable.
   272→  useEffect(() => {
   273→    if (productType !== 'variable' && isStitchable) {
   274→      setIsStitchable(false)
   275→    }
   276→  }, [productType, isStitchable])
   277→
   278→  // ---- Helpers
   279→  function validateStep(s: number): string | null {
   280→    if (s === 0) {
   281→      if (title.trim().length < 2) return 'Product title must be at least 2 characters.'
   282→      if (shortDescription.length > 500) return 'Short description must be 500 characters or fewer.'
   283→    }
   284→    if (s === 1) {
   285→      const variants = collectVariantsForValidation()
   286→      if (variants.length === 0) {
   287→        if (isStitchable && productType === 'variable') {
   288→          return 'Generate at least one variant using the “Generate Variants” button.'
   289→        }
   290→        return 'Add at least one variant.'
   291→      }
   292→      for (const v of variants) {
   293→        if (!v.sku.trim()) return 'Every variant needs an SKU.'
   294→        if (typeof v.sale_price !== 'number' || v.sale_price < 0) {
   295→          return `Variant ${v.sku || '(unnamed)'} needs a valid sale price.`
   296→        }
   297→        if (typeof v.cost_price !== 'number' || v.cost_price < 0) {
   298→          return `Variant ${v.sku || '(unnamed)'} needs a valid cost price.`
   299→        }
   300→      }
   301→      if (isStitchable && productType === 'variable') {
   302→        if (stitchingTypes.every((s2) => !s2.enabled)) {
   303→          return 'Enable at least one stitching type.'
   304→        }
   305→        if (includeUnstitched && (!baseFabricCost || baseFabricCost < 0)) {
   306→          return 'Enter the unstitched fabric cost.'
   307→        }
   308→      }
   309→    }
   310→    return null
   311→  }
   312→
   313→  function collectVariantsForValidation(): VariantDraft[] {
   314→    if (productType === 'simple') return [simpleVariant]
   315→    if (productType === 'variable' && isStitchable) {
   316→      return generatedVariants.map((g) => ({
   317→        sku: g.sku,
   318→        barcode: '',
   319→        attribute_values: g.attribute_values,
   320→        cost_price: g.cost_price,
   321→        stitching_charges: g.stitching_charges,
   322→        compare_price: null,
   323→        weight_grams: 0,
   324→        fulfillment_type: g.fulfillment_type as 'stock_based' | 'made_to_order',
   325→        stitching_type: g.stitching_type as VariantDraft['stitching_type'],
   326→        production_days: g.production_days,
   327→        sale_price: g.sale_price,
   328→        is_active: g.is_active,
   329→        is_default: g.is_default,
   330→        has_opening_stock: false,
   331→        opening_stock_qty: 0,
   332→        opening_stock_cost: 0,
   333→        opening_stock_location_id: '',
   334→        fabric_source_variant_id: null,
   335→      }))
   336→    }
   337→    if (productType === 'variable') return regularVariants
   338→    // bundle / service — use simple variant form
   339→    return [simpleVariant]
   340→  }
   341→
   342→  function goNext() {
   343→    setSubmitError(null)
   344→    const err = validateStep(step)
   345→    if (err) {
   346→      setSubmitError(err)
   347→      return
   348→    }
   349→    setStep((s) => Math.min(s + 1, STEPS.length - 1))
   350→  }
   351→
   352→  // ---- Stitchable: generate variants via API
   353→  async function generateVariants() {
   354→    setSubmitError(null)
   355→    const enabledTypes = stitchingTypes.filter((s) => s.enabled).map((s) => s.type)
   356→    if (enabledTypes.length === 0) {
   357→      setSubmitError('Enable at least one stitching type.')
   358→      return
   359→    }
   360→    if (includeUnstitched && (!baseFabricCost || baseFabricCost < 0)) {
   361→      setSubmitError('Enter the unstitched fabric cost.')
   362→      return
   363→    }
   364→    setGenerating(true)
   365→    try {
   366→      const sizes = includeSizes ? [...selectedSizes, ...customSizes] : []
   367→      const res = await api.post<{ variants: GeneratedVariant[] }>(
   368→        '/api/products/generate-stitched',
   369→        {
   370→          product_slug: slug,
   371→          base_sku: baseSku.trim() || undefined,
   372→          sizes,
   373→          stitching_types: enabledTypes,
   374→          base_fabric_cost: Number(baseFabricCost) || 0,
   375→          base_stitching: stitchingTypes.find((s) => s.type === 'stitched_basic')?.charge ?? 0,
   376→          heavy_stitching: stitchingTypes.find((s) => s.type === 'stitched_heavy')?.charge ?? 0,
   377→          custom_stitching: stitchingTypes.find((s) => s.type === 'custom_order')?.charge ?? 0,
   378→          include_unstitched: includeUnstitched,
   379→        },
   380→      )
   381→      // Default sale price = cost + 30% markup so the user has a starting point
   382→      setGeneratedVariants(
   383→        res.variants.map((v) => ({
   384→          ...v,
   385→          sale_price: Math.ceil((v.cost_price * 1.3) * 100) / 100,
   386→          is_active: true,
   387→        })),
   388→      )
   389→      toast.success(`${res.variants.length} variants generated.`)
   390→    } catch (err) {
   391→      const msg =
   392→        err instanceof FetchError ? err.message : 'Failed to generate variants.'
   393→      setSubmitError(msg)
   394→      toast.error(msg)
   395→    } finally {
   396→      setGenerating(false)
   397→    }
   398→  }
   399→
   400→  // ---- Submit final product
   401→  async function submit() {
   402→    setSubmitError(null)
   403→    for (const s of [0, 1, 2]) {
   404→      const err = validateStep(s)
   405→      if (err) {
   406→        setSubmitError(err)
   407→        setStep(s)
   408→        return
   409→      }
   410→    }
   411→
   412→    // Clone the drafts so we can normalize the default flag without mutating state.
   413→    const variants = collectVariantsForValidation().map<VariantDraft>((v) => ({ ...v }))
   414→
   415→    // Ensure at least one variant is the default.
   416→    if (!variants.some((v) => v.is_default) && variants.length > 0) {
   417→      variants[0].is_default = true
   418→    }
   419→
   420→    const payload = {
   421→      title: title.trim(),
   422→      base_sku: baseSku.trim() || undefined,
   423→      description: description.trim() || undefined,
   424→      short_description: shortDescription.trim() || undefined,
   425→      product_type: productType,
   426→      category_id: categoryId || undefined,
   427→      brand_id: brandId || undefined,
   428→      product_scope: productScope,
   429→      is_stitchable: isStitchable,
   430→      stitching_base_price:
   431→        isStitchable && productType === 'variable' ? Number(baseFabricCost) || 0 : 0,
   432→      has_size_variants:
   433→        isStitchable && productType === 'variable'
   434→          ? includeSizes && (selectedSizes.length + customSizes.length) > 0
   435→          : false,
   436→      is_active: true,
   437→      is_featured: isFeatured,
   438→      variants: variants.map((v) => ({
   439→        sku: v.sku.trim(),
   440→        barcode: v.barcode?.trim() || undefined,
   441→        attribute_values: v.attribute_values,
   442→        cost_price: Number(v.cost_price) || 0,
   443→        stitching_charges: Number(v.stitching_charges) || 0,
   444→        compare_price: v.compare_price ?? undefined,
   445→        weight_grams: Number(v.weight_grams) || 0,
   446→        fulfillment_type: v.fulfillment_type,
   447→        stitching_type: v.stitching_type ?? undefined,
   448→        production_days: Number(v.production_days) || 0,
   449→        allow_backorder: false,
   450→        requires_shipping: true,
   451→        is_taxable: true,
   452→        is_active: v.is_active,
   453→        is_default: v.is_default,
   454→        sale_price: Number(v.sale_price) || 0,
   455→      })),
   456→    }
   457→
   458→    setSubmitting(true)
   459→    try {
   460→      const res = await api.post<{ id: string; slug: string; title: string }>(
   461→        '/api/products',
   462→        payload,
   463→      )
   464→
   465→      // After product creation, process opening stock for variants that have it
   466→      const variantsWithOpeningStock = variants.filter(
   467→        (v) => v.has_opening_stock && v.opening_stock_qty > 0 && v.opening_stock_location_id,
   468→      )
   469→      if (variantsWithOpeningStock.length > 0) {
   470→        try {
   471→          await api.post('/api/inventory/receive', {
   472→            location_id: variantsWithOpeningStock[0].opening_stock_location_id,
   473→            notes: `Opening stock for ${res.title}`,
   474→            items: variantsWithOpeningStock.map((v) => ({
   475→              org_variant_id: v.sku, // Note: this won't work — we need the actual variant ID
   476→              quantity: v.opening_stock_qty,
   477→              cost_per_unit: v.opening_stock_cost || v.cost_price,
   478→            })),
   479→          })
   480→        } catch {
   481→          // Opening stock failure shouldn't block product creation
   482→          toast.warning('Product created, but opening stock could not be set. Use Receive Stock manually.')
   483→        }
   484→      }
   485→
   486→      queryClient.invalidateQueries({ queryKey: ['products'] })
   487→      toast.success(`"${res.title}" created.`)
   488→      navigate({ name: 'product-detail', id: res.id })
   489→    } catch (err) {
   490→      const msg =
   491→        err instanceof FetchError ? err.message : 'Failed to create product.'
   492→      setSubmitError(msg)
   493→      toast.error(msg)
   494→    } finally {
   495→      setSubmitting(false)
   496→    }
   497→  }
   498→
   499→  // ---- Render
   500→  return (
   501→    <div className="space-y-6 max-w-3xl mx-auto">
   502→      <button
   503→        onClick={onBack}
   504→        className="flex items-center gap-1.5 text-sm text-muted-foreground hover:text-foreground"
   505→        disabled={submitting}
   506→      >
   507→        <ArrowLeft className="h-4 w-4" /> Back to products
   508→      </button>
   509→
   510→      <PageHeader
   511→        title="Create New Product"
   512→        description="Set up your product, its variants, and how it syncs to Shopify."
   513→      />
   514→
   515→      {/* Stepper */}
   516→      <div className="flex items-center gap-2">
   517→        {STEPS.map((label, i) => (
   518→          <div key={label} className="flex items-center gap-2 flex-1">
   519→            <div
   520→              className={cn(
   521→                'flex h-7 w-7 items-center justify-center rounded-full text-xs font-medium shrink-0 transition-colors',
   522→                i < step
   523→                  ? 'bg-primary text-primary-foreground'
   524→                  : i === step
   525→                    ? 'bg-primary text-primary-foreground ring-4 ring-primary/15'
   526→                    : 'bg-muted text-muted-foreground',
   527→              )}
   528→            >
   529→              {i < step ? <Check className="h-3.5 w-3.5" /> : i + 1}
   530→            </div>
   531→            <span
   532→              className={cn(
   533→                'text-sm font-medium hidden sm:block',
   534→                i === step ? 'text-foreground' : 'text-muted-foreground',
   535→              )}
   536→            >
   537→              {label}
   538→            </span>
   539→            {i < STEPS.length - 1 && (
   540→              <div
   541→                className={cn(
   542→                  'h-px flex-1',
   543→                  i < step ? 'bg-primary' : 'bg-border',
   544→                )}
   545→              />
   546→            )}
   547→          </div>
   548→        ))}
   549→      </div>
   550→
   551→      {submitError && (
   552→        <div className="flex items-start gap-2.5 rounded-lg border border-destructive/30 bg-destructive/5 p-3 text-sm text-destructive">
   553→          <AlertCircle className="h-4 w-4 shrink-0 mt-0.5" />
   554→          <div className="flex-1">
   555→            <p className="font-medium">Couldn&apos;t continue</p>
   556→            <p className="text-xs mt-0.5 opacity-90">{submitError}</p>
   557→          </div>
   558→          <button
   559→            onClick={() => setSubmitError(null)}
   560→            className="text-destructive/60 hover:text-destructive text-xs"
   561→          >
   562→            Dismiss
   563→          </button>
   564→        </div>
   565→      )}
   566→
   567→      {/* Step 1 */}
   568→      {step === 0 && (
   569→        <Card>
   570→          <CardHeader>
   571→            <CardTitle className="text-base flex items-center gap-2">
   572→              <Package className="h-5 w-5 text-primary" /> Basic details
   573→            </CardTitle>
   574→            <CardDescription>
   575→              Core product information your customers will see.
   576→            </CardDescription>
   577→          </CardHeader>
   578→          <CardContent className="space-y-5">
   579→            <div className="space-y-1.5">
   580→              <Label htmlFor="title">
   581→                Title <span className="text-destructive">*</span>
   582→              </Label>
   583→              <Input
   584→                id="title"
   585→                value={title}
   586→                onChange={(e) => setTitle(e.target.value)}
   587→                placeholder="e.g. Lawn Embroidered Kurta — Summer Collection"
   588→                autoFocus

  1820→function slugify(s: string): string {
  1821→  return s
  1822→    .toLowerCase()
  1823→    .trim()

  1825→    .replace(/[\s_-]+/g, '-')

  1834→function attrsToText(attrs: Record<string, string>): string {
  1835→  return Object.entries(attrs)
  1836→    .map(([k, v]) => `${k}: ${v}`)
  1837→    .join('\n')
  1838→}

  1840→function textToAttrs(text: string): Record<string, string> {
  1841→  const out: Record<string, string> = {}
  1842→  for (const line of text.split('\n')) {
  1843→    const m = line.match(/^\s*([^:]+):\s*(.+)\s*$/)
  1844→    if (m) {
  1845→      out[m[1].trim()] = m[2].trim()
  1846→    }
  1847→  }
  1848→  return out
  1849→}
