   225→const PKR = new Intl.NumberFormat('en-PK', { maximumFractionDigits: 0 })
   226→
   227→function formatPKR(n: number): string {
   228→  return `Rs. ${PKR.format(n)}`
   229→}
   230→
   231→function formatDate(iso: string | null): string {
   232→  if (!iso) return '—'
   233→  try {
   234→    return new Date(iso).toLocaleDateString('en-PK', {
   235→      day: '2-digit',
   236→      month: 'short',
   237→      year: 'numeric',
   238→    })
   239→  } catch {
   240→    return iso
   241→  }
   242→}
   243→
   244→function formatDateTime(iso: string | null): string {
   245→  if (!iso) return '—'
   246→  try {
   247→    return new Date(iso).toLocaleString('en-PK', {
   248→      day: '2-digit',
   249→      month: 'short',
   250→      hour: '2-digit',
   251→      minute: '2-digit',
   252→    })
   253→  } catch {
   254→    return iso
   255→  }
   256→}
   257→
   258→function getErrorMessage(err: unknown): string {
   259→  if (err instanceof FetchError) return err.message
   260→  if (err instanceof Error) return err.message
   261→  return 'Something went wrong.'
   262→}
   263→
   264→function formatAttributeValues(values: Record<string, string>): string {
   265→  if (!values || Object.keys(values).length === 0) return ''
   266→  return Object.entries(values)
   267→    .map(([k, v]) => `${k}: ${v}`)
   268→    .join(' · ')
   269→}

   338→    onError: (err) => toast.error(getErrorMessage(err)),
   339→  })
   340→  const deliveredMutation = useMutation({
   341→    mutationFn: () => api.post(`/api/orders/${orderId}/delivered`),

   358→  })

   361→    mutationFn: (vars: { cancellation_reason: string }) =>

   366→      invalidateAll()

   368→    onError: (err) => toast.error(getErrorMessage(err)),
   369→  })
   370→
   371→  const rtoMutation = useMutation({
   372→    mutationFn: (vars: { return_reason: string }) =>
   373→      api.post(`/api/orders/${orderId}/rto`, vars),
   374→    onSuccess: () => {

   379→    onError: (err) => toast.error(getErrorMessage(err)),

   507→
   508→  return (

   530→            </Button>

   532→        }
   533→      />
   534→
   535→      {/* ── External reference + source badge band ─────────────────────── */}
   536→      {(order.externalOrderReference || order.orderSource !== 'manual') && (

   540→              <span className="text-xs font-medium uppercase tracking-wide text-muted-foreground shrink-0">
   541→                External Ref
   542→              </span>
   543→              <span className="text-sm font-semibold font-mono truncate">
   544→                {order.externalOrderReference}
   545→              </span>
   546→            </div>
   547→          )}
   548→          {order.orderSource !== 'manual' && (
   549→            <Badge variant="outline" className={sourceBadge.className}>
   550→              {sourceBadge.label}
   551→            </Badge>
   552→          )}
   553→        </div>
   554→      )}
   555→
   556→      {/* ── Status badge row ─────────────────────────────────────────────── */}
   557→      <div className="flex flex-wrap items-center gap-2">
   558→        <Badge variant="outline" className={statusBadge.className}>
   559→          {statusBadge.label}
   560→        </Badge>
   561→        <Badge variant="outline" className={paymentBadge.className}>
   562→          {paymentBadge.label}
   563→        </Badge>
   564→        {order.paymentType === 'partial_advance' && (

   566→            Partial Advance
   567→          </Badge>
   568→        )}
   569→        {order.paymentType === 'fully_prepaid' && (

   571→            Prepaid
