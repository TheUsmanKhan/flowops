   104→   104→/**
   105→   105→ * Fetches live inventory per variant for this product and returns a map keyed
   106→   106→ * by variantId. Used by the parent-child table to display on-hand stock
   107→   107→ * alongside cost/price columns. Reads from the same inventory_pools data as
   108→   108→ * the Inventory tab and the Inventory Dashboard — no parallel data source.
   109→   109→ */
   110→   110→function useVariantInventoryMap(productId: string) {
   111→   111→  const { data } = useQuery<InventorySummaryResponse>({
   112→   112→    queryKey: ['product-inventory', productId],
   113→   113→    queryFn: () => api.get(`/api/inventory/summary?product_id=${productId}`),
   114→   114→    staleTime: 15_000,
   115→   115→  })
   116→   116→  const map = new Map<string, InventorySummaryVariant>()
   117→   117→  for (const v of data?.variants ?? []) {
   118→   118→    map.set(v.variantId, v)
   119→   119→  }
   120→   120→  return map
   121→   121→}

   123→   123→/** Compact stock cell — shows on_hand, with available in muted subtext. */
   124→   124→function StockCell({ inv }: { inv?: InventorySummaryVariant }) {
   125→   125→  if (!inv) {
   126→   126→    return <span className="text-xs text-muted-foreground">—</span>
   127→   127→  }
   128→   128→  const onHand = inv.totalOnHand
   129→   129→  const available = inv.totalAvailable
   130→   130→  if (onHand === 0) {
   131→   131→    return <span className="text-xs text-muted-foreground">0</span>
   132→   132→  }
   133→   133→  const lowStock = available > 0 && available <= 5
   134→   134→  return (
   135→   135→    <div className="text-xs leading-tight">
   136→   136→      <span className={cn('font-medium', lowStock ? 'text-amber-700' : 'text-foreground')}>{onHand}</span>
   137→   137→      {available !== onHand && (
   138→   138→        <div className="text-[10px] text-muted-foreground">{available} avail.</div>
   139→   139→      )}
   140→   140→    </div>
   141→   141→  )
   142→   142→}

   179→   179→export function ParentChildVariantTable({
   180→   180→  productId,

   182→   182→}: {
   183→   183→  productId: string
   184→   184→  mode?: 'cost' | 'pricing' | 'both'
   185→   185→}) {
   186→   186→  const can = useCan()
   187→   187→  const queryClient = useQueryClient()
   188→   188→  const [expandedGroups, setExpandedGroups] = useState<Set<string>>(new Set())
   189→   189→  const [editingVariant, setEditingVariant] = useState<ChildVariant | null>(null)
   190→   190→
   191→   191→  const canEditCost = can(PERMISSIONS.PRODUCTS_EDIT)
   192→   192→  const canEditPrice = can(PERMISSIONS.PRODUCTS_PRICING)
   193→   193→  const showCost = mode === 'cost' || mode === 'both'
   194→   194→  const showPricing = mode === 'pricing' || mode === 'both'
   195→   195→
   196→   196→  const { data, isLoading, isError, refetch } = useQuery<VariantGroupsResponse>({
   197→   197→    queryKey: ['variant-groups', productId],
   198→   198→    queryFn: () => api.get(`/api/products/${productId}/variant-groups`),
   199→   199→    staleTime: 15_000,
   200→   200→  })
   201→   201→
   202→   202→  // Live inventory per variant — powers the read-only Stock column.
   203→   203→  const inventoryMap = useVariantInventoryMap(productId)
   204→   204→
   205→   205→  function toggleGroup(parentValue: string) {
   206→   206→    setExpandedGroups((prev) => {
   207→   207→      const next = new Set(prev)
   208→   208→      if (next.has(parentValue)) next.delete(parentValue)
   209→   209→      else next.add(parentValue)
   210→   210→      return next
   211→   211→    })
   212→   212→  }
   213→   213→
   214→   214→  // Expand all groups by default on first load
   215→   215→  if (data && expandedGroups.size === 0 && data.groups.length > 0) {
   216→   216→    setExpandedGroups(new Set(data.groups.map((g) => g.parentValue)))
   217→   217→  }
   218→   218→
   219→   219→  if (isLoading) {
   220→   220→    return (
   221→   221→      <Card>
   222→   222→        <CardContent className="p-6 space-y-3">
   223→   223→          {[0, 1, 2].map((i) => (
   224→   224→            <div key={i} className="h-20 rounded bg-muted/50 animate-pulse" />
   225→   225→          ))}
   226→   226→        </CardContent>
   227→   227→      </Card>
   228→   228→    )
   229→   229→  }
   230→   230→
   231→   231→  if (isError) {
   232→   232→    return (
   233→   233→      <Card>
   234→   234→        <CardContent className="p-6 text-center">
   235→   235→          <AlertCircle className="h-8 w-8 mx-auto text-muted-foreground mb-2" />
   236→   236→          <p className="text-sm text-muted-foreground mb-3">Failed to load variant groups.</p>
   237→   237→          <Button variant="outline" size="sm" onClick={() => refetch()}>Retry</Button>
   238→   238→        </CardContent>
   239→   239→      </Card>
   240→   240→    )
   241→   241→  }
   242→   242→
   243→   243→  if (!data || data.groups.length === 0) {
   244→   244→    return (
   245→   245→      <Card>
   246→   246→        <CardContent className="p-6 text-center text-sm text-muted-foreground">
   247→   247→          No variants found for this product.
   248→   248→        </CardContent>
   249→   249→      </Card>
   250→   250→    )
   251→   251→  }
   252→   252→
   253→   253→  // If only one attribute, render flat table (no grouping)
   254→   254→  if (!data.hasMultipleAttributes) {
   255→   255→    return (
   256→   256→      <>
   257→   257→        <FlatVariantTable
   258→   258→          groups={data.groups}
   259→   259→          productId={productId}
   260→   260→          showCost={showCost}
   261→   261→          showPricing={showPricing}
   262→   262→          canEditCost={canEditCost}
   263→   263→          canEditPrice={canEditPrice}
   264→   264→          queryClient={queryClient}
   265→   265→          onEdit={(v) => setEditingVariant(v)}
   266→   266→          inventoryMap={inventoryMap}
   267→   267→        />
   268→   268→        {editingVariant && (
   269→   269→          <VariantEditDialog
   270→   270→            variant={editingVariant}
   271→   271→            productId={productId}
   272→   272→            onClose={() => setEditingVariant(null)}
   273→   273→            queryClient={queryClient}
   274→   274→          />
   275→   275→        )}
   276→   276→      </>
   277→   277→    )
   278→   278→  }
   279→   279→
   280→   280→  // Grouped table with parent-child cascade
   281→   281→  return (
   282→   282→    <div className="space-y-3">
   283→   283→      {data.parentAttributeName && (
   284→   284→        <p className="text-xs text-muted-foreground">
   285→   285→          Grouped by <span className="font-medium text-foreground">{data.parentAttributeDisplayName || data.parentAttributeName}</span> —
   286→   286→          set prices once per group, or override individual variants.
   287→   287→        </p>
   288→   288→      )}
   289→   289→      {data.groups.map((group) => (
   290→   290→        <GroupCard
   291→   291→          key={group.parentValue}
   292→   292→          group={group}
   293→   293→          productId={productId}
   294→   294→          parentAttributeName={data.parentAttributeName!}
   295→   295→          expanded={expandedGroups.has(group.parentValue)}
   296→   296→          onToggle={() => toggleGroup(group.parentValue)}
   297→   297→          showCost={showCost}
   298→   298→          showPricing={showPricing}
   299→   299→          canEditCost={canEditCost}
   300→   300→          canEditPrice={canEditPrice}
   301→   301→          queryClient={queryClient}
   302→   302→          onEditVariant={(v) => setEditingVariant(v)}
   303→   303→          inventoryMap={inventoryMap}
   304→   304→        />
   305→   305→      ))}
   306→   306→      {editingVariant && (
   307→   307→        <VariantEditDialog
   308→   308→          variant={editingVariant}
   309→   309→          productId={productId}
   310→   310→          onClose={() => setEditingVariant(null)}
   311→   311→          queryClient={queryClient}
   312→   312→        />
   313→   313→      )}
   314→   314→    </div>

   321→   321→function GroupCard({
   322→   322→  group,
   323→   323→  productId,
   324→   324→  parentAttributeName,
   325→   325→  expanded,
   326→   326→  onToggle,
   327→   327→  showCost,
   328→   328→  showPricing,
   329→   329→  canEditCost,
   330→   330→  canEditPrice,
   331→   331→  queryClient,
   332→   332→  onEditVariant,
   333→   333→  inventoryMap,
   334→   334→}: {
   335→   335→  group: VariantGroup
   336→   336→  productId: string
   337→   337→  parentAttributeName: string
   338→   338→  expanded: boolean
   339→   339→  onToggle: () => void
   340→   340→  showCost: boolean
   341→   341→  showPricing: boolean
   342→   342→  canEditCost: boolean
   343→   343→  canEditPrice: boolean
   344→   344→  queryClient: ReturnType<typeof useQueryClient>
   345→   345→  onEditVariant: (v: ChildVariant) => void
   346→   346→  inventoryMap: Map<string, InventorySummaryVariant>
   347→   347→}) {
   348→   348→  const [parentCost, setParentCost] = useState('')
   349→   349→  const [parentSale, setParentSale] = useState('')
   350→   350→  const [parentCompare, setParentCompare] = useState('')
   351→   351→  const [applying, setApplying] = useState(false)
   352→   352→
   353→   353→  const firstChild = group.children[0]
   354→   354→  if (parentCost === '' && firstChild) setParentCost(String(firstChild.costPrice))
   355→   355→  if (parentSale === '' && firstChild?.salePrice != null) setParentSale(String(firstChild.salePrice))
   356→   356→  if (parentCompare === '' && firstChild?.comparePrice != null) setParentCompare(String(firstChild.comparePrice))
   357→   357→
   358→   358→  // Bug 1 fix: ONE handler cascades ALL THREE fields (cost, sale, compare).
   359→   359→  // Calls both server endpoints (cost + sale-price) in sequence, then
   360→   360→  // invalidates the cache once. Each endpoint only updates children whose
   361→   361→  // relevant synced flag is true, so the three flags remain INDEPENDENT.
   362→   362→  async function applyAllToGroup() {
   363→   363→    const cost = Number(parentCost)
   364→   364→    const sale = Number(parentSale)
   365→   365→    const compare = parentCompare ? Number(parentCompare) : null
   366→   366→
   367→   367→    if (isNaN(cost) || cost < 0) { toast.error('Enter a valid cost price'); return }
   368→   368→    if (isNaN(sale) || sale < 0) { toast.error('Enter a valid sale price'); return }
   369→   369→
   370→   370→    setApplying(true)
   371→   371→    let costCount = 0
   372→   372→    let saleCount = 0
   373→   373→    try {
   374→   374→      // Cascade cost to cost-synced children
   375→   375→      const costRes = await api.post<{ success: boolean; updated_count: number }>(
   376→   376→        `/api/products/${productId}/variant-groups/dummy/cost`,
   377→   377→        { cost_price: cost, parent_attribute_name: parentAttributeName, parent_value: group.parentValue },
   378→   378→      )
   379→   379→      costCount = costRes.updated_count
   380→   380→
   381→   381→      // Cascade sale + compare to their respective synced children
   382→   382→      const saleRes = await api.post<{ success: boolean; updated_count: number }>(
   383→   383→        `/api/products/${productId}/variant-groups/dummy/sale-price`,
   384→   384→        { sale_price: sale, compare_price: compare, parent_attribute_name: parentAttributeName, parent_value: group.parentValue },
   385→   385→      )
   386→   386→      saleCount = saleRes.updated_count
   387→   387→
   388→   388→      toast.success(`Applied to group — Cost: ${costCount}, Sale+Compare: ${saleCount} variant(s)`)
   389→   389→      queryClient.invalidateQueries({ queryKey: ['variant-groups', productId] })
   390→   390→      queryClient.invalidateQueries({ queryKey: ['product', productId] })
   391→   391→    } catch (err) {
   392→   392→      toast.error(err instanceof FetchError ? err.message : 'Failed to apply to group')
   393→   393→    } finally { setApplying(false) }
   394→   394→  }
   395→   395→
   396→   396→  const childAttrKeys = group.children.length > 0
   397→   397→    ? Object.keys(group.children[0].attributeValues).filter((k) => k !== parentAttributeName)
   398→   398→    : []
   399→   399→
   400→   400→  return (
   401→   401→    <Card>
   402→   402→      <CardHeader className="pb-3 cursor-pointer" onClick={onToggle}>
   403→   403→        <div className="flex items-center justify-between">
   404→   404→          <div className="flex items-center gap-2">
   405→   405→            {expanded ? <ChevronDown className="h-4 w-4 text-muted-foreground" /> : <ChevronRight className="h-4 w-4 text-muted-foreground" />}
   406→   406→            <CardTitle className="text-base">{group.parentValue}</CardTitle>
   407→   407→            <Badge variant="secondary" className="text-[10px]">{group.childCount} variant{group.childCount === 1 ? '' : 's'}</Badge>
   408→   408→          </div>
   409→   409→        </div>
   410→   410→      </CardHeader>
   411→   411→
   412→   412→      {expanded && (
   413→   413→        <CardContent className="space-y-4">
   414→   414→          {/* Parent group inputs — uses the shared ParentGroupInputs component
   415→   415→              with a single "Apply to Group" button that cascades ALL THREE
   416→   416→              fields (cost, sale, compare) independently. Bug 1 fix. */}
   417→   417→          <ParentGroupInputs
   418→   418→            parentCost={parentCost}
   419→   419→            parentSale={parentSale}
   420→   420→            parentCompare={parentCompare}
   421→   421→            onCostChange={setParentCost}
   422→   422→            onSaleChange={setParentSale}
   423→   423→            onCompareChange={setParentCompare}
   424→   424→            onApplyAll={applyAllToGroup}
   425→   425→            showCost={showCost}
   426→   426→            showPricing={showPricing}
   427→   427→            canEditCost={canEditCost}
   428→   428→            canEditPrice={canEditPrice}
   429→   429→            applying={applying}
   430→   430→          />
   431→   431→
   432→   432→          {/* Children table */}
   433→   433→          <div className="rounded-md border overflow-x-auto scrollbar-thin">
   434→   434→            <table className="w-full text-sm">
   435→   435→              <thead className="bg-muted/30">
   436→   436→                <tr className="text-left text-xs text-muted-foreground">
   437→   437→                  {childAttrKeys.map((k) => (<th key={k} className="px-3 py-2 font-medium">{k}</th>))}
   438→   438→                  <th className="px-3 py-2 font-medium">SKU</th>
   439→   439→                  <th className="px-3 py-2 font-medium">Fulfillment</th>
   440→   440→                  <th className="px-3 py-2 font-medium text-center">Stock</th>
   441→   441→                  {showCost && <th className="px-3 py-2 font-medium text-right">Cost</th>}
   442→   442→                  {showPricing && <th className="px-3 py-2 font-medium text-right">Sale</th>}
   443→   443→                  {showPricing && <th className="px-3 py-2 font-medium text-right">Compare</th>}
   444→   444→                  <th className="px-3 py-2 font-medium text-center">Active</th>
   445→   445→                  <th className="px-3 py-2 font-medium">Actions</th>
   446→   446→                </tr>
   447→   447→              </thead>
   448→   448→              <tbody>
   449→   449→                {group.children.map((child) => (
   450→   450→                  <ChildRow
   451→   451→                    key={child.variantId}
   452→   452→                    child={child}
   453→   453→                    childAttrKeys={childAttrKeys}
   454→   454→                    productId={productId}
   455→   455→                    showCost={showCost}
   456→   456→                    showPricing={showPricing}
   457→   457→                    canEditCost={canEditCost}
   458→   458→                    canEditPrice={canEditPrice}
   459→   459→                    queryClient={queryClient}
   460→   460→                    onEdit={() => onEditVariant(child)}
   461→   461→                    inventory={inventoryMap.get(child.variantId)}
   462→   462→                  />
   463→   463→                ))}
   464→   464→              </tbody>
   465→   465→            </table>
   466→   466→          </div>
   467→   467→        </CardContent>
   468→   468→      )}
   469→   469→    </Card>
   470→   470→  )
   471→   471→}

   503→   503→  const [toggling, setToggling] = useState(false)
   504→   504→
   505→   505→  async function saveCost() {
   506→   506→    const newCost = Number(costValue)
   507→   507→    if (isNaN(newCost) || newCost < 0) return
   508→   508→    if (newCost === child.costPrice) return
   509→   509→    setSaving(true)
   510→   510→    try {
   511→   511→      await api.post(`/api/products/${productId}/variants/${child.variantId}/override-cost`, { cost_price: newCost })
   512→   512→      toast.success('Cost overridden — no longer synced with parent')
   513→   513→      queryClient.invalidateQueries({ queryKey: ['variant-groups', productId] })
   514→   514→      queryClient.invalidateQueries({ queryKey: ['product', productId] })
   515→   515→    } catch (err) {
   516→   516→      toast.error(err instanceof FetchError ? err.message : 'Failed to override cost')
   517→   517→      setCostValue(String(child.costPrice))
   518→   518→    } finally { setSaving(false) }
   519→   519→  }
   520→   520→
   521→   521→  async function savePrice() {
   522→   522→    const newSale = Number(saleValue)
   523→   523→    const newCompare = compareValue ? Number(compareValue) : null
   524→   524→    if (isNaN(newSale) || newSale < 0) return
   525→   525→    setSaving(true)
   526→   526→    try {
   527→   527→      await api.post(`/api/products/${productId}/variants/${child.variantId}/override-price`, { sale_price: newSale, compare_price: newCompare })
   528→   528→      toast.success('Price overridden — no longer synced with parent')
   529→   529→      queryClient.invalidateQueries({ queryKey: ['variant-groups', productId] })
   530→   530→      queryClient.invalidateQueries({ queryKey: ['product', productId] })
   531→   531→    } catch (err) {
   532→   532→      toast.error(err instanceof FetchError ? err.message : 'Failed to override price')
   533→   533→      setSaleValue(child.salePrice != null ? String(child.salePrice) : '')
   534→   534→      setCompareValue(child.comparePrice != null ? String(child.comparePrice) : '')
   535→   535→    } finally { setSaving(false) }
   536→   536→  }
   537→   537→
   538→   538→  async function resyncCost() {
   539→   539→    setSaving(true)
   540→   540→    try {
   541→   541→      const res = await api.post<{ success: boolean; cost_price: number }>(`/api/products/${productId}/variants/${child.variantId}/resync-cost`)
   542→   542→      toast.success(`Re-synced with parent (Rs. ${res.cost_price})`)
   543→   543→      queryClient.invalidateQueries({ queryKey: ['variant-groups', productId] })
   544→   544→      queryClient.invalidateQueries({ queryKey: ['product', productId] })
   545→   545→    } catch (err) {
   546→   546→      toast.error(err instanceof FetchError ? err.message : 'Failed to re-sync')
   547→   547→    } finally { setSaving(false) }
   548→   548→  }
   549→   549→
   550→   550→  async function resyncPrice(field: 'sale_price' | 'compare_price') {

   552→   552→    try {
   553→   553→      await api.post(`/api/products/${productId}/variants/${child.variantId}/resync-price`, { field })
   554→   554→      toast.success(`Re-synced ${field === 'sale_price' ? 'sale price' : 'compare price'} with parent`)
   555→   555→      queryClient.invalidateQueries({ queryKey: ['variant-groups', productId] })
   556→   556→      queryClient.invalidateQueries({ queryKey: ['product', productId] })
   557→   557→    } catch (err) {
   558→   558→      toast.error(err instanceof FetchError ? err.message : 'Failed to re-sync')
   559→   559→    } finally { setSaving(false) }
   560→   560→  }
   561→   561→
   562→   562→  async function toggleActive() {

   568→   568→      queryClient.invalidateQueries({ queryKey: ['product', productId] })
   569→   569→    } catch (err) {
   570→   570→      toast.error(err instanceof FetchError ? err.message : 'Failed to toggle')

   599→   599→        <td className="px-3 py-2">
   600→   600→          <div className="flex items-center gap-1 justify-end">
   601→   601→            {canEditPrice ? (
   602→   602→              <Input type="number" min="0" step="0.01" value={saleValue} onChange={(e) => setSaleValue(e.target.value)} onBlur={savePrice} className="h-7 w-20 text-xs text-right" />
   603→   603→            ) : (<span className="text-xs">{child.salePrice ?? '—'}</span>)}
   604→   604→            {child.salePriceSyncedWithParent ? <Link2 className="h-3 w-3 text-emerald-500" /> : <Unlink className="h-3 w-3 text-amber-500" />}

   673→   673→  const [confirmSkuChange, setConfirmSkuChange] = useState(false)
   674→   674→
   675→   675→  async function save() {
   676→   676→    // If SKU changed, show confirmation
   677→   677→    if (form.sku !== variant.sku && !confirmSkuChange) {
   678→   678→      setConfirmSkuChange(true)

   680→   680→    }

   701→   701→      queryClient.invalidateQueries({ queryKey: ['variant-groups', productId] })

   730→   730→          </div>
   731→   731→          <div className="space-y-1.5">
   732→   732→            <Label className="text-xs">Barcode</Label>
   733→   733→            <Input value={form.barcode} onChange={(e) => setForm({ ...form, barcode: e.target.value })} className="h-9 text-sm" placeholder="UPC/EAN" />

   887→   887→  async function savePrice() {
