   600→      orderDetailParts.push(
   601→        `${variant.product.title}${inner ? ` (${inner})` : ''} ×${item.quantity}`,
   602→      )
   603→    }
   604→
   605→    // 5. Compute totals (CPU-only)
   606→    const discountAmount = d.discount_amount ?? 0
   607→    const courierCharges = d.courier_charges ?? 0
   608→    const estimatedDeliveryCharge = d.estimated_delivery_charge ?? 0
   609→    const taxAmount = d.tax_amount ?? 0
   610→    const totalOrderValue = subtotal + courierCharges + estimatedDeliveryCharge + taxAmount - discountAmount
   611→
   612→    // 6. Determine payment status + source (CPU-only)
   613→    let paymentStatus: string = 'cod_pending'
   614→    let paymentSource: string = 'cod_native'
   615→    let advanceAmount: number | null = null
   616→    let advancePaymentMethod: string | null = null
   617→    let advancePaymentReference: string | null = null
   618→    let advancePaidAt: Date | null = null
   619→
   620→    if (d.payment_type === 'partial_advance') {
   621→      paymentStatus = 'advance_paid'
   622→      paymentSource = 'manual_conversion'
   623→      advanceAmount = d.advance_amount ?? 0
   624→      advancePaymentMethod = d.advance_payment_method || null
   625→      advancePaymentReference = d.advance_payment_reference || null
   626→      advancePaidAt = new Date()
   627→    } else if (d.payment_type === 'fully_prepaid') {
   628→      paymentStatus = 'fully_prepaid'
   629→      paymentSource = 'manual_conversion'
   630→      advanceAmount = totalOrderValue
   631→      advancePaymentMethod = d.advance_payment_method || null
   632→      advancePaymentReference = d.advance_payment_reference || null
   633→      advancePaidAt = new Date()
   634→    }
   635→
   636→    const remainingCodAmount = totalOrderValue - (advanceAmount ?? 0)
   637→
   638→    // 7. Determine initial order status — uses the ALREADY-FETCHED
   639→    //    companyOrderSetting (no separate query). Previously this did a
   640→    //    dedicated findUnique for requireOrderConfirmation.
   641→    let orderStatus: string = 'pending'
   642→    let confirmedAt: Date | null = null
   643→    let skippedConfirmation = false
   644→
   645→    if (paymentStatus === 'advance_paid' || paymentStatus === 'fully_prepaid') {
   646→      orderStatus = 'confirmed'
   647→      confirmedAt = new Date()
   648→    } else {
   649→      if (!orderSettings?.requireOrderConfirmation) {
   650→        orderStatus = 'confirmed'
   651→        confirmedAt = new Date()
   652→        skippedConfirmation = true
   653→      }
   654→    }
   655→
   656→    // 8. Universal courier reference fields (migration 015).
   657→    const orderRefNumber =
   658→      (d.order_ref_number && d.order_ref_number.trim()) || flowopsOrderNumber
   659→    const orderDetail =
   660→      (d.order_detail && d.order_detail.trim()) || orderDetailParts.join(', ')
   661→
   662→    // 8b. Self-fulfilled reference number (Phase B1). When fulfillmentChannel
   663→    //     is 'self_fulfilled', generate the per-company SF-YYYY-NNNNN reference
   664→    //     via the SQL function. For 'courier' orders, this stays null.
   665→    const isSelfFulfilled = d.fulfillment_channel === 'self_fulfilled'
   666→    const selfFulfilledReferenceNumber = isSelfFulfilled
   667→      ? await generateSelfFulfilledReference(ctx.company.id)
   668→      : null
   669→
   670→    // 9. Create the order (sequential — needs all above)
   671→    const order = await db.order.create({
   672→      data: {
   673→        organizationId: ctx.company.organizationId,
   674→        companyId: ctx.company.id,
   675→        flowopsOrderNumber,
   676→        orderSource: 'manual',
   677→        fulfillmentChannel: d.fulfillment_channel,
   678→        selfFulfilledReferenceNumber,
   679→        customerId,
   680→        recipientName,
   681→        usedCustomerAddressId,
   682→        usedCustomerPhoneId,
   683→        status: orderStatus,
   684→        paymentType: d.payment_type,
   685→        paymentStatus,
   686→        paymentSource,
   687→        subtotal,
   688→        discountAmount: discountAmount || null,
   689→        discountReason: d.discount_reason || null,
   690→        courierCharges: courierCharges || null,
   691→        estimatedDeliveryCharge: estimatedDeliveryCharge || null,
   692→        taxAmount: taxAmount || null,
   693→        taxLabel: d.tax_label?.trim() || null,
   694→        totalOrderValue,
   695→        advanceAmount,
   696→        advancePaymentMethod,
   697→        advancePaymentReference,
   698→        advancePaidAt,
   699→        remainingCodAmount,
   700→        deliveryAddress: d.delivery_address,
   701→        deliveryCity: d.delivery_city,
   702→        deliveryCountry: d.delivery_country?.trim() || 'PK',
   703→        courierName: d.courier_name || null,
   704→        courierCompanyIntegrationId: d.courier_company_integration_id || null,
   705→        recommendedCourierCompanyIntegrationId: d.courier_company_integration_id || null,
   706→        dispatchLocationId: d.dispatch_location_id,
   707→        notesForCourier: d.notes_for_courier || null,
   708→        pickupAddressId: d.pickup_address_id || null,
   709→        orderRefNumber,
   710→        orderDetail,
   711→        skippedConfirmation,
   712→        confirmedAt,
   713→        createdBy: ctx.employee.id,
   714→        // ─── Sales attribution (Phase 3) ───────────────────────────────
   715→        // Whoever creates the order owns it — automatic, no manual step.
   716→        // This is the single most important data-capture point for KPI,
   717→        // commission, and scoped-visibility features. Set once at creation,
   718→        // never changes afterward (represents "who sold this").
   719→        salesEmployeeId: ctx.employee.id,
   720→      },
   721→    })
   722→
   723→    // 10. Batch-create order items — SINGLE query via createManyAndReturn
   724→    //     (was N sequential inserts in a loop). Returns the created rows so
   725→    //     we can populate `createdItems` for the response.
   726→    const createdItemRows = await db.orderItem.createManyAndReturn({
   727→      data: orderItemsData.map((itemData) => ({
   728→        orderId: order.id,
   729→        orgVariantId: itemData.orgVariantId,
   730→        organizationId: ctx.company.organizationId,
   731→        quantity: itemData.quantity,
   732→        unitPrice: itemData.unitPrice,
   733→        originalUnitPrice: itemData.originalUnitPrice,
   734→        discountType: itemData.discountType,
   735→        discountValue: itemData.discountValue,
   736→        lineTotal: itemData.lineTotal,
   737→        fulfillmentStatus: 'pending',
   738→        fulfillmentTypeSnapshot: itemData.fulfillmentTypeSnapshot,
   739→        reservedLocationId: d.dispatch_location_id,
   740→      })),
   741→      select: { id: true, orgVariantId: true, quantity: true },
   742→    })
   743→    const createdItems = createdItemRows.map((r) => ({
   744→      id: r.id,
   745→      orgVariantId: r.orgVariantId,
   746→      quantity: r.quantity,
   747→    }))
   748→
   749→    // 11. Audit log (fire-and-forget — see src/lib/audit.ts)
   750→    insertAuditLog({
   751→      action: 'order.created',
   752→      entityType: 'order',
   753→      entityId: order.id,
   754→      companyId: ctx.company.id,
   755→      organizationId: ctx.company.organizationId,
   756→      userId: ctx.user.id,
   757→      employeeId: ctx.employee.id,
   758→      newValues: {
   759→        flowopsOrderNumber,
   760→        status: orderStatus,
   761→        paymentType: d.payment_type,
   762→        paymentStatus,
   763→        itemCount: orderItemsData.length,
   764→        totalOrderValue,
   765→        customerId,
   766→      },
   767→    })
   768→
   769→    // 12. Customer Management System integration — address lastUsedAt +
   770→    //     save-one-off-address + recompute cached stats. The
   771→    //     saveAddressForNextTime branch MUST be sequential (the order.update
   772→    //     needs the created address ID), but markAddressAsUsed (the other
   773→    //     branch) and updateCustomerStats are independent → run in parallel.
   774→    if (selectedSavedAddressId) {
   775→      // Bump lastUsedAt on the saved address + recompute customer stats
   776→      // in parallel (independent writes to different tables).
   777→      await Promise.all([
   778→        markAddressAsUsed(selectedSavedAddressId),
   779→        updateCustomerStats(customerId),
   780→      ])
   781→
   782→      // Propagate city correction: if the user modified the city for
   783→      // this order (e.g., corrected "lahore" → "Lahore"), update the saved
   784→      // CustomerAddress row so future orders see the corrected city.
   785→      // This is FIRE-AND-FORGET — a failure here must NOT cause order
   786→      // creation to fail. The order is already created; this is a
   787→      // convenience optimization for future orders, not a critical write.
   788→      if (d.delivery_city?.trim()) {
   789→        db.customerAddress.update({
   790→          where: { id: selectedSavedAddressId },
   791→          data: { city: d.delivery_city.trim() },
   792→        }).catch((err) => {
   793→          console.error(
   794→            `[createManualOrder] Failed to propagate city correction to CustomerAddress ${selectedSavedAddressId}:`,
   795→            err instanceof Error ? err.message : err,
   796→          )
   797→        })
   798→      }
   799→      // Propagate country correction the same fire-and-forget way (mirrors
   800→      // the city correction above). Only writes if a non-empty country was
   801→      // provided on the order — the default "Pakistan" still overwrites a
   802→      // saved different country, which is the intended correction behavior.
   803→      if (d.delivery_country?.trim()) {
   804→        db.customerAddress.update({
   805→          where: { id: selectedSavedAddressId },
   806→          data: { country: d.delivery_country.trim() },
   807→        }).catch((err) => {
   808→          console.error(
   809→            `[createManualOrder] Failed to propagate country correction to CustomerAddress ${selectedSavedAddressId}:`,
   810→            err instanceof Error ? err.message : err,
   811→          )
   812→        })
   813→      }
   814→    } else if (saveAddressForNextTime && d.delivery_address && d.delivery_city) {
   815→      // One-off address typed + user opted to save it for next time.
   816→      // Persist as a non-default customer_addresses row, then link the
   817→      // order to it + recompute customer stats in parallel.
   818→      const savedAddr = await db.customerAddress.create({
   819→        data: {
   820→          customerId,
   821→          organizationId: ctx.company.organizationId,
   822→          label: null,
   823→          address: d.delivery_address,
   824→          city: d.delivery_city,
   825→          // Country (Phase: Country System). Defaults to "PK" when
   826→          // absent — mirrors the order's own deliveryCountry default.
   827→          country: d.delivery_country?.trim() || 'PK',
   828→          isDefault: false,
   829→          lastUsedAt: new Date(),
   830→        },
   831→      })
   832→      await Promise.all([
   833→        db.order.update({
   834→          where: { id: order.id },
   835→          data: { usedCustomerAddressId: savedAddr.id },
   836→        }),
   837→        updateCustomerStats(customerId),
   838→      ])
   839→    } else {
   840→      // No address to save — just recompute stats.
   841→      await updateCustomerStats(customerId)
   842→    }
   843→
   844→    // Phase 6 — Fire-and-forget: recompute the sales employee's funnel stats
   845→    // (order was just created, affects their totalOrders count). Only when the
   846→    // order has a salesEmployeeId (manual orders always do; webhook orders don't).
   847→    updateEmployeeStats(ctx.employee.id).catch(() => {})
   848→
   849→    // 13. If the order auto-confirmed, run the stock reservation logic.
   850→    //     Sequential — reserveOrderStock reads the order items created above.
   851→    if (orderStatus === 'confirmed') {
   852→      await reserveOrderStock(order.id, ctx)
   853→    }
   854→
   855→    // 14. AUTO-BOOKING — uses the ALREADY-FETCHED orderSettings (no 2nd
   856→    //     findUnique). Fires in the background (PostEx can take 50-100s).
   857→    //     SKIP entirely for self-fulfilled orders (Phase B1) — no courier
   858→    //     to book, no city-courier-matching to run. The order is created
   859→    //     with courierBookingStatus='not_booked' + fulfillmentChannel=
   860→    //     'self_fulfilled' + a selfFulfilledReferenceNumber already set.
   861→    let bookingAttempted = false
   862→    let bookingSucceeded = false
   863→    let bookingError: string | undefined
   864→    let bookingTrackingNumber: string | undefined
   865→
   866→    if (orderStatus === 'confirmed' && !isSelfFulfilled) {
   867→      const shouldAutoBook =
   868→        orderSettings?.courierBookingMode === 'automatic' &&
   869→        (d.courier_company_integration_id || orderSettings?.defaultCourierCompanyIntegrationId)
   870→
   871→      if (shouldAutoBook) {
   872→        bookingAttempted = true
   873→        ;(async () => {
   874→          try {
   875→            const { maybeAutoBookOrder } = await import('./booking.actions')
   876→            const bookResult = await maybeAutoBookOrder(order.id, 'manual', orderStatus)
   877→            if (bookResult.success) {
   878→              console.log(`[createManualOrder] Background auto-booking succeeded for ${flowopsOrderNumber}: tracking=${bookResult.data?.trackingNumber}`)
   879→            } else {
   880→              console.warn(`[createManualOrder] Background auto-booking failed for ${flowopsOrderNumber}: ${bookResult.error}`)
   881→            }
   882→          } catch (err) {
   883→            console.error(`[createManualOrder] Background auto-booking threw for ${flowopsOrderNumber}:`, err)
   884→          }
   885→        })()
   886→      }
   887→    }
   888→
   889→    insertMetricEvent({
   890→      companyId: ctx.company.id,
   891→      entityType: 'order',
   892→      entityId: order.id,
   893→      metricKey: 'order.created',
   894→      numericValue: Number(totalOrderValue),
   895→      dimensions: { order_source: 'manual', payment_type: d.payment_type, company_id: ctx.company.id },
   896→    })
   897→
   898→    return {
   899→      success: true,
   900→      data: {
   901→        orderId: order.id,
   902→        flowopsOrderNumber,
   903→        orderItems: createdItems,
   904→        bookingAttempted,
   905→        bookingSucceeded,
   906→        bookingError,
   907→        bookingTrackingNumber,
   908→      },
   909→    }
   910→  } catch (err) {
   911→    return {
   912→      success: false,
   913→      error: err instanceof Error ? err.message : 'Failed to create order',
   914→    }
   915→  }
   916→}
   917→
   918→// ──────────────────────────────────────────────────────────────
   919→// createOrderFromShopifyWebhook — webhook-driven order ingestion
   920→// ──────────────────────────────────────────────────────────────
   921→
   922→export async function createOrderFromShopifyWebhook(
   923→  payload: ShopifyOrderWebhook,
   924→  companyId: string,
   925→  organizationId: string,
   926→  /**
   927→   * Optional injected workspace context. When provided (by the webhook
   928→   * route), the function uses it instead of calling getWorkspace() — which
   929→   * requires a logged-in user session that webhooks don't have. The
   930→   * injected context is built by the webhook route from the
   931→   * CompanyIntegration's connectedByEmployeeId (the employee who connected
   932→   * Shopify). When omitted (e.g. unit-test / future admin-tool callers),
   933→   * getWorkspace() is called as before — 100% backward-compatible.
   934→   */
   935→  injectedContext?: WorkspaceContext,
   936→): Promise<ActionResult<{ orderId: string; flowopsOrderNumber: string }>> {
   937→  try {
   938→    const ctx = injectedContext ?? (await getWorkspace())
   939→    // This function is called by the Shopify webhook route (with an injected
   940→    // context built from the integration's connectedByEmployeeId). It is also
   941→    // unit-testable with a mock payload + mock context.
   942→
   943→    const parsed = shopifyOrderWebhookSchema.safeParse(payload)
   944→    if (!parsed.success) {
   945→      return { success: false, error: 'Invalid Shopify webhook payload' }
   946→    }
   947→    const d = parsed.data
   948→
   949→    // Map Shopify financial_status → FlowOps payment_status
   950→    let paymentStatus: string
   951→    let paymentSource: string = 'shopify_gateway'
   952→    let paymentType: string
   953→    let orderStatus: string = 'pending'
   954→    let confirmedAt: Date | null = null
   955→    let advanceAmount: number | null = null
   956→    let advancePaidAt: Date | null = null
   957→
   958→    switch (d.financial_status) {
   959→      case 'paid':
   960→        paymentStatus = 'fully_prepaid'
   961→        paymentType = 'fully_prepaid'
   962→        orderStatus = 'confirmed'
   963→        confirmedAt = new Date()
   964→        advancePaidAt = new Date()
   965→        // advanceAmount will be set to totalOrderValue below (after total_price is parsed)
   966→        break
   967→      case 'partially_paid':
   968→        paymentStatus = 'advance_paid'
   969→        paymentType = 'partial_advance'
   970→        orderStatus = 'confirmed'
   971→        confirmedAt = new Date()
   972→        advancePaidAt = new Date()
   973→        // Shopify webhook does not carry the partial amount paid — leave advanceAmount null.
   974→        // remainingCodAmount will be computed as totalOrderValue - 0 = totalOrderValue.
   975→        break
   976→      default:
   977→        paymentStatus = 'cod_pending'
   978→        paymentType = 'full_cod'
   979→        paymentSource = 'cod_native'
   980→        // Check company settings for COD orders
   981→        const settings = await db.companyOrderSetting.findUnique({ where: { companyId } })
   982→        if (!settings?.requireOrderConfirmation) {
   983→          orderStatus = 'confirmed'
   984→          confirmedAt = new Date()
   985→        }
   986→    }
   987→
   988→    // Resolve customer via the cross-platform matching strategy (Step 1's
   989→    // match_or_create_customer() SQL function, wrapped by
   990→    // matchOrCreateExternalCustomer). This handles the layered:
   991→    //   1. exact_identity (existing Shopify customer mapping)
   992→    //   2. phone_match (normalized phone)
   993→    //   3. email_match
   994→    //   4. create new customer + primary phone + external identity mapping
   995→    //
   996→    // The customer's name comes from Shopify's first_name + last_name.
   997→    // Phone/email come from the Shopify payload (may be null for guest orders).
   998→    const shopifyCustomerName =
   999→      `${d.customer.first_name ?? ''} ${d.customer.last_name ?? ''}`.trim() ||
  1000→      'Unknown Shopify Customer'
  1001→
  1002→    const matchResult = await matchOrCreateExternalCustomer({
  1003→      platform: 'shopify',
  1004→      external_customer_id: String(d.customer.id ?? d.id),
  1005→      phone: d.customer.phone || undefined,
  1006→      email: d.customer.email || undefined,
  1007→      name: shopifyCustomerName,
  1008→      organizationId,
  1009→    })
  1010→    if (!matchResult.success || !matchResult.data) {
  1011→      return {
  1012→        success: false,
  1013→        error: matchResult.error ?? 'Could not resolve customer from Shopify payload',
  1014→      }
  1015→    }
  1016→    const customerId = matchResult.data.customerId
  1017→
  1018→    // Fetch the resolved customer (for recipient_name default + address lookup).
  1019→    const customer = await db.customer.findUnique({
  1020→      where: { id: customerId },
  1021→      include: {
  1022→        phones: { where: { isPrimary: true }, take: 1 },
  1023→        addresses: { where: { isDefault: true }, take: 1 },
  1024→      },
  1025→    })
  1026→    if (!customer) {
  1027→      return { success: false, error: 'Resolved customer not found' }
  1028→    }
  1029→
  1030→    // If Shopify sent a default_address AND the customer has no saved address
  1031→    // yet (newly created), persist it as their default customer_addresses row.
  1032→    // For existing customers, we leave their saved addresses alone — the
  1033→    // Shopify address becomes a one-off delivery snapshot on this order only.
  1034→    let usedCustomerAddressId: string | null = null
  1035→    let usedCustomerPhoneId: string | null = customer.phones[0]?.id ?? null
  1036→    const shopifyAddress1 = d.customer.default_address?.address1 || null
  1037→    const shopifyCity = d.customer.default_address?.city || null
  1038→    // Country (Phase: Country System). Shopify's default_address.country_code
  1039→    // is an alpha-2 code (e.g. "PK", "GB") — maps directly onto
  1040→    // CustomerAddress.country / Order.deliveryCountry with no translation.
  1041→    // Fallback chain: country_code (preferred) → country (name, normalized
  1042→    // to alpha-2 via countryNameToCode) → null (handled by caller). The
  1043→    // caller (createOrderFromShopify) falls back to Company.countryCode if
  1044→    // Shopify sends neither.
  1045→    const shopifyCountryCode = d.customer.default_address?.country_code || null
  1046→    const shopifyCountryName = d.customer.default_address?.country || null
  1047→    // Normalize: prefer country_code; if absent, convert the country NAME
  1048→    // to an alpha-2 code via the lookup table. If both are absent, null.
  1049→    const shopifyCountry = shopifyCountryCode || (shopifyCountryName ? (countryNameToCode(shopifyCountryName) ?? null) : null)
  1050→
  1051→    if (shopifyAddress1 && shopifyCity && customer.addresses.length === 0) {
  1052→      const newAddr = await db.customerAddress.create({
  1053→        data: {
  1054→          customerId: customer.id,
  1055→          organizationId,
  1056→          label: 'Shopify Default',
  1057→          address: shopifyAddress1,
  1058→          city: shopifyCity,
  1059→          // Capture the country Shopify sent instead of discarding it
  1060→          // (Phase 1 finding #4 fix). Falls back to "PK" if Shopify
  1061→          // didn't include one — non-blocking, the address is still saved.
  1062→          country: shopifyCountry || 'PK',
  1063→          isDefault: true,
  1064→          lastUsedAt: new Date(),
  1065→        },
  1066→      })
  1067→      usedCustomerAddressId = newAddr.id
  1068→    } else if (customer.addresses.length > 0) {
  1069→      // Existing customer with a saved default address — use it for this order.
  1070→      usedCustomerAddressId = customer.addresses[0].id
  1071→      await markAddressAsUsed(usedCustomerAddressId)
  1072→    }
  1073→
  1074→    const recipientName = shopifyCustomerName
  1075→
  1076→    // Resolve variants by SKU — include companyPricing for per-company price
  1077→    // resolution. No market gating: a variant is sellable as long as it's
  1078→    // active AND has a CompanyVariantPricing row for this company.
  1079→    const skus = d.line_items.map((li) => li.sku).filter(Boolean) as string[]
  1080→    const variants = await db.orgProductVariant.findMany({
  1081→      where: { sku: { in: skus }, organizationId },
  1082→      include: {
  1083→        product: { select: { id: true } },
  1084→        companyPricing: { where: { companyId } },
  1085→      },
  1086→    })
  1087→
  1088→    // Compute subtotal + build order items with gate flagging
  1089→    let subtotal = 0
  1090→    const orderItemsData: Array<{
  1091→      orgVariantId: string
  1092→      quantity: number
  1093→      unitPrice: number
  1094→      originalUnitPrice: number
  1095→      discountType: string | null
  1096→      discountValue: number | null
  1097→      fulfillmentTypeSnapshot: string
  1098→      needsReview: boolean
  1099→      needsReviewReason: string | null
  1100→    }> = []
  1101→
  1102→    for (const li of d.line_items) {
  1103→      const variant = variants.find((v) => v.sku === li.sku)
  1104→      if (!variant) continue // skip unmatched items for now
  1105→
  1106→      // ── Gate check (SOFT — flag, don't block) ──
  1107→      let gateReason: string | null = null
  1108→      // Gate 1: variant isActive
  1109→      if (!variant.isActive) {
  1110→        gateReason = 'Not enabled for your company'
  1111→      }
  1112→      // Gate 2: CompanyVariantPricing must exist for this variant
  1113→      else if (!variant.companyPricing[0]) {
  1114→        gateReason = 'No price set for this company'
  1115→      }
  1116→
  1117→      const needsReview = gateReason !== null
