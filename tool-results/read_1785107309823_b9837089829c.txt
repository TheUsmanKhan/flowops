    74→// Helper: reserveOrderStock — shared internal function
    75→// ──────────────────────────────────────────────────────────────
    76→// Processes ALL order_items for an order and attempts to reserve
    77→// stock for each based on its fulfillment_type_snapshot.
    78→//
    79→// For stock_based items:
    80→//   - If sufficient available stock → reserveStockForOrder(), fulfillment_status='reserved'
    81→//   - If insufficient + inventory_policy='continue' → fulfillment_status='backordered'
    82→//   - If insufficient + inventory_policy='deny' → outcome='failed'
    83→//
    84→// For made_to_order items:
    85→//   - checkAndFulfillMadeToOrderVariant() checks returned stock first
    86→//   - If existing_stock → fulfillment_status='reserved', returned_stitched_used=true
    87→//   - If fresh_production → fulfillment_status='reserved', production_order_id set
    88→//   - If error → outcome='failed'
    89→//
    90→// After processing: if ANY item is 'backordered', order status → 'partially_backordered'.
    91→// If ALL items are 'reserved', order stays 'confirmed'.
    92→//
    93→// Called from:
    94→//   - createManualOrder() IF the order auto-confirmed at creation
    95→//   - confirmOrder() when a 'pending' order is manually confirmed
    96→//   - dispatchOrderAction() if status is still 'pending' at dispatch time
    97→
    98→async function reserveOrderStock(
    99→  orderId: string,
   100→  ctx: { company: { id: string; organizationId: string }; user: { id: string }; employee: { id: string } },
   101→): Promise<{
   102→  success: boolean
   103→  results: Array<{ orderItemId: string; outcome: 'reserved' | 'backordered' | 'failed'; reason?: string }>
   104→}> {
   105→  const order = await db.order.findUnique({
   106→    where: { id: orderId },
   107→    select: { dispatchLocationId: true, companyId: true, organizationId: true },
   108→  })
   109→  if (!order) return { success: false, results: [] }
   110→
   111→  const items = await db.orderItem.findMany({
   112→    where: { orderId },
   113→    include: {
   114→      orgVariant: {
   115→        select: { id: true, fulfillmentType: true, inventoryPolicy: true, stitchingCharges: true },
   116→      },
   117→    },
   118→  })
   119→
   120→  const results: Array<{ orderItemId: string; outcome: 'reserved' | 'backordered' | 'failed'; reason?: string }> = []
   121→  let hasBackordered = false
   122→
   123→  for (const item of items) {
   124→    if (item.fulfillmentStatus === 'reserved' || item.fulfillmentStatus === 'dispatched') {
   125→      // Already processed (e.g. re-confirmation attempt) — skip
   126→      results.push({ orderItemId: item.id, outcome: 'reserved' })
   127→      continue
   128→    }
   129→
   130→    const locationId = item.reservedLocationId ?? order.dispatchLocationId
   131→    if (!locationId) {
   132→      results.push({ orderItemId: item.id, outcome: 'failed', reason: 'No dispatch location set' })
   133→      continue
   134→    }
   135→
   136→    if (item.fulfillmentTypeSnapshot === 'stock_based') {
   137→      // Check available stock at the dispatch location
   138→      const pool = await db.inventoryPool.findUnique({
   139→        where: {
   140→          orgVariantId_locationId: {
   141→            orgVariantId: item.orgVariantId,
   142→            locationId,
   143→          },
   144→        },
   145→      })
   146→
   147→      const available = pool ? pool.onHand - pool.reserved : 0
   148→
   149→      if (available >= item.quantity) {
   150→        // Sufficient stock — reserve it
   151→        const reserveResult = await reserveStockForOrder({
   152→          orgVariantId: item.orgVariantId,
   153→          locationId,
   154→          organizationId: order.organizationId,
   155→          companyId: order.companyId,
   156→          employeeId: ctx.employee.id,
   157→          quantity: item.quantity,
   158→          orderId,
   159→        })
   160→
   161→        if (reserveResult.success) {
   162→          await db.orderItem.update({
   163→            where: { id: item.id },
   164→            data: { fulfillmentStatus: 'reserved', reservedLocationId: locationId },
   165→          })
   166→          results.push({ orderItemId: item.id, outcome: 'reserved' })
   167→        } else {
   168→          results.push({ orderItemId: item.id, outcome: 'failed', reason: reserveResult.error })
   169→        }
   170→      } else if (item.orgVariant.inventoryPolicy === 'continue') {
   171→        // Insufficient stock but backordering allowed
   172→        await db.orderItem.update({
   173→          where: { id: item.id },
   174→          data: { fulfillmentStatus: 'backordered', backorderedAt: new Date() },
   175→        })
   176→        await insertMetricEvent({
   177→          companyId: order.companyId,
   178→          entityType: 'product',
   179→          entityId: item.orgVariantId,
   180→          metricKey: 'order.backordered',
   181→          numericValue: item.quantity,
   182→          dimensions: { order_id: orderId },
   183→        }).catch(() => {})
   184→        hasBackordered = true
   185→        results.push({
   186→          orderItemId: item.id,
   187→          outcome: 'backordered',
   188→          reason: `Available: ${available}, required: ${item.quantity}`,
   189→        })
   190→      } else {
   191→        // Insufficient stock + policy='deny' — fail
   192→        results.push({
   193→          orderItemId: item.id,
   194→          outcome: 'failed',
   195→          reason: `Insufficient stock (available: ${available}, required: ${item.quantity}) and inventory_policy='deny'`,
   196→        })
   197→      }
   198→    } else if (item.fulfillmentTypeSnapshot === 'made_to_order') {
   199→      // Made-to-order: check returned stock first, then trigger production
   200→      const mtoResult = await checkAndFulfillMadeToOrderVariant(
   201→        item.orgVariantId,
   202→        item.quantity,
   203→        order.companyId,
   204→        locationId,
   205→      )
   206→
   207→      if (mtoResult.source === 'existing_stock' && mtoResult.locationId) {
   208→        // Returned stock available — reserve it
   209→        const reserveResult = await reserveStockForOrder({
   210→          orgVariantId: item.orgVariantId,
   211→          locationId: mtoResult.locationId,
   212→          organizationId: order.organizationId,
   213→          companyId: order.companyId,
   214→          employeeId: ctx.employee.id,
   215→          quantity: item.quantity,
   216→          orderId,
   217→        })
   218→
   219→        if (reserveResult.success) {
   220→          await db.orderItem.update({
   221→            where: { id: item.id },
   222→            data: {
   223→              fulfillmentStatus: 'reserved',
   224→              returnedStitchedUsed: true,
   225→              reservedLocationId: mtoResult.locationId,
   226→            },
   227→          })
   228→          await insertMetricEvent({
   229→            companyId: order.companyId,
   230→            entityType: 'product',
   231→            entityId: item.orgVariantId,
   232→            metricKey: 'order.made_to_order_from_returned_stock',
   233→            numericValue: 1,
   234→            dimensions: { order_id: orderId, stitching_cost_saved: Number(item.orgVariant.stitchingCharges) || 0 },
   235→          }).catch(() => {})
   236→          results.push({ orderItemId: item.id, outcome: 'reserved' })
   237→        } else {
   238→          results.push({ orderItemId: item.id, outcome: 'failed', reason: reserveResult.error })
   239→        }
   240→      } else if (mtoResult.source === 'fresh_production' && mtoResult.productionOrderId) {
   241→        // Fresh production triggered — link the production order
   242→        await db.orderItem.update({
   243→          where: { id: item.id },
   244→          data: {
   245→            fulfillmentStatus: 'reserved',
   246→            productionOrderId: mtoResult.productionOrderId,
   247→          },
   248→        })
   249→        // Also link the order_item back to the production order
   250→        await db.productionOrder.update({
   251→          where: { id: mtoResult.productionOrderId },
   252→          data: { orderItemId: item.id },
   253→        })
   254→        await insertMetricEvent({
   255→          companyId: order.companyId,
   256→          entityType: 'product',
   257→          entityId: item.orgVariantId,
   258→          metricKey: 'order.made_to_order_production_triggered',
   259→          numericValue: 1,
   260→          dimensions: { order_id: orderId, production_order_id: mtoResult.productionOrderId },
   261→        }).catch(() => {})
   262→        results.push({ orderItemId: item.id, outcome: 'reserved' })
   263→      } else {
   264→        // Error (no fabric source, insufficient fabric, etc.)
   265→        results.push({
   266→          orderItemId: item.id,
   267→          outcome: 'failed',
   268→          reason: mtoResult.error ?? 'Made-to-order fulfillment failed',
   269→        })
   270→      }
   271→    }
   272→  }
   273→
   274→  // Update order status based on results
   275→  if (hasBackordered) {
   276→    await db.order.update({
   277→      where: { id: orderId },
   278→      data: { status: 'partially_backordered' },
   279→    })
   280→  }
   281→  // If no backordered items, order stays 'confirmed' (or whatever it was)
   282→
   283→  return { success: true, results }
   284→}

   286→// ──────────────────────────────────────────────────────────────
   287→// createManualOrder
   288→// ──────────────────────────────────────────────────────────────
   289→
   290→export async function createManualOrder(
   291→  input: CreateManualOrderInput,
   292→): Promise<ActionResult<{
   293→  orderId: string
   294→  flowopsOrderNumber: string
   295→  orderItems: Array<{ id: string; orgVariantId: string; quantity: number }>
   296→}>> {
   297→  try {
   298→    const ctx = await getWorkspace()
   299→    await requirePermission(ctx, PERMISSIONS.ORDERS_CREATE)
   300→
   301→    // 1. Validate
   302→    const parsed = createManualOrderSchema.safeParse(input)
   303→    if (!parsed.success) {
   304→      return {
   305→        success: false,
   306→        error: parsed.error.issues[0]?.message ?? 'Invalid order data',
   307→      }
   308→    }
   309→    const d = parsed.data
   310→
   311→    // 2. Find or create customer
   312→    let customerId: string
   313→    if (d.customer_id) {
   314→      // Verify existing customer belongs to this org
   315→      const existing = await db.customer.findFirst({
   316→        where: { id: d.customer_id, organizationId: ctx.company.organizationId },
   317→      })
   318→      if (!existing) return { success: false, error: 'Customer not found' }
   319→      customerId = existing.id
   320→    } else if (d.customer) {
   321→      const customerResult = await findOrCreateCustomer({
   322→        ...d.customer,
   323→        organizationId: ctx.company.organizationId,
   324→        companyId: ctx.company.id,
   325→      })
   326→      if (!customerResult.success || !customerResult.data) {
   327→        return { success: false, error: customerResult.error ?? 'Failed to create customer' }
   328→      }
   329→      customerId = customerResult.data.customerId
   330→    } else {
   331→      return { success: false, error: 'Either customer or customer_id is required' }
   332→    }
   333→
   334→    // 3. Fetch variants + their pricing for this company + fulfillment_type snapshot
   335→    const variantIds = d.items.map((i) => i.org_variant_id)
   336→    const variants = await db.orgProductVariant.findMany({
   337→      where: { id: { in: variantIds }, organizationId: ctx.company.organizationId },
   338→      include: {
   339→        companyPricing: { where: { companyId: ctx.company.id } },
   340→      },
   341→    })
   342→
   343→    if (variants.length !== variantIds.length) {
   344→      return { success: false, error: 'One or more variants not found in this organization' }
   345→    }
   346→
   347→    // 4. Compute subtotal + build order items data
   348→    let subtotal = 0
   349→    const orderItemsData: Array<{
   350→      orgVariantId: string
   351→      quantity: number
   352→      unitPrice: number
   353→      lineTotal: number
   354→      fulfillmentTypeSnapshot: string
   355→    }> = []
   356→
   357→    for (const item of d.items) {
   358→      const variant = variants.find((v) => v.id === item.org_variant_id)
   359→      if (!variant) continue
   360→
   361→      // Use provided unit_price OR fall back to company pricing OR variant cost
   362→      const unitPrice =
   363→        item.unit_price ??
   364→        (variant.companyPricing[0]?.salePrice
   365→          ? Number(variant.companyPricing[0].salePrice)
   366→          : Number(variant.costPrice))
   367→
   368→      const lineTotal = item.quantity * unitPrice
   369→      subtotal += lineTotal
   370→
   371→      orderItemsData.push({
   372→        orgVariantId: item.org_variant_id,
   373→        quantity: item.quantity,
   374→        unitPrice,
   375→        lineTotal,
   376→        fulfillmentTypeSnapshot: variant.fulfillmentType,
   377→      })
   378→    }
   379→
   380→    // 5. Compute totals
   381→    const discountAmount = d.discount_amount ?? 0
   382→    const courierCharges = d.courier_charges ?? 0
   383→    const totalOrderValue = subtotal + courierCharges - discountAmount
   384→
   385→    // 6. Determine payment status + source
   386→    let paymentStatus: string = 'cod_pending'
   387→    let paymentSource: string = 'cod_native'
   388→    let advanceAmount: number | null = null
   389→    let advancePaymentMethod: string | null = null
   390→    let advancePaymentReference: string | null = null
   391→    let advancePaidAt: Date | null = null
   392→
   393→    if (d.payment_type === 'partial_advance') {
   394→      paymentStatus = 'advance_paid'
   395→      paymentSource = 'manual_conversion'
   396→      advanceAmount = d.advance_amount ?? 0
   397→      advancePaymentMethod = d.advance_payment_method || null
   398→      advancePaymentReference = d.advance_payment_reference || null
   399→      advancePaidAt = new Date()
   400→    } else if (d.payment_type === 'fully_prepaid') {
   401→      paymentStatus = 'fully_prepaid'
   402→      paymentSource = 'manual_conversion'
   403→      advanceAmount = totalOrderValue
   404→      advancePaymentMethod = d.advance_payment_method || null
   405→      advancePaymentReference = d.advance_payment_reference || null
   406→      advancePaidAt = new Date()
   407→    }
   408→
   409→    // 7. Determine initial order status
   410→    let orderStatus: string = 'pending'
   411→    let confirmedAt: Date | null = null
   412→    let skippedConfirmation = false
   413→
   414→    if (paymentStatus === 'advance_paid' || paymentStatus === 'fully_prepaid') {
   415→      // Payment itself is the confirmation signal — bypasses require_order_confirmation
   416→      orderStatus = 'confirmed'
   417→      confirmedAt = new Date()
   418→    } else {
   419→      // COD order — check company_order_settings
   420→      const settings = await db.companyOrderSetting.findUnique({
   421→        where: { companyId: ctx.company.id },
   422→      })
   423→      if (!settings?.requireOrderConfirmation) {
   424→        orderStatus = 'confirmed'
   425→        confirmedAt = new Date()
   426→        skippedConfirmation = true
   427→      }
   428→    }
   429→
   430→    // 8. Generate order number
   431→    const flowopsOrderNumber = await generateOrderNumber(ctx.company.id)
   432→
   433→    // 9. Create the order
   434→    const order = await db.order.create({
   435→      data: {
   436→        organizationId: ctx.company.organizationId,
   437→        companyId: ctx.company.id,
   438→        flowopsOrderNumber,
   439→        orderSource: 'manual',
   440→        customerId,
   441→        status: orderStatus,
   442→        paymentType: d.payment_type,
   443→        paymentStatus,
   444→        paymentSource,
   445→        subtotal,
   446→        discountAmount: discountAmount || null,
   447→        discountReason: d.discount_reason || null,
   448→        courierCharges: courierCharges || null,
   449→        totalOrderValue,
   450→        advanceAmount,
   451→        advancePaymentMethod,
   452→        advancePaymentReference,
   453→        advancePaidAt,
   454→        deliveryAddress: d.delivery_address,
   455→        deliveryCity: d.delivery_city,
   456→        courierName: d.courier_name || null,
   457→        dispatchLocationId: d.dispatch_location_id,
   458→        notesForCourier: d.notes_for_courier || null,
   459→        skippedConfirmation,
   460→        confirmedAt,
   461→        createdBy: ctx.employee.id,
   462→      },
   463→    })
   464→
   465→    // 10. Create order items (fulfillment_status = 'reserved' as PLACEHOLDER)
   466→    const createdItems: Array<{ id: string; orgVariantId: string; quantity: number }> = []
   467→    for (const itemData of orderItemsData) {
   468→      const item = await db.orderItem.create({
   469→        data: {
   470→          orderId: order.id,
   471→          orgVariantId: itemData.orgVariantId,
   472→          organizationId: ctx.company.organizationId,
   473→          quantity: itemData.quantity,
   474→          unitPrice: itemData.unitPrice,
   475→          lineTotal: itemData.lineTotal,
   476→          fulfillmentStatus: 'reserved', // PLACEHOLDER — Step 3 will validate/adjust
   477→          fulfillmentTypeSnapshot: itemData.fulfillmentTypeSnapshot,
   478→          reservedLocationId: d.dispatch_location_id,
   479→        },
   480→      })
   481→      createdItems.push({
   482→        id: item.id,
   483→        orgVariantId: item.orgVariantId,
   484→        quantity: item.quantity,
   485→      })
   486→    }
   487→
   488→    // 11. Audit log
   489→    await insertAuditLog({
   490→      action: 'order.created',
   491→      entityType: 'order',
   492→      entityId: order.id,
   493→      companyId: ctx.company.id,
   494→      organizationId: ctx.company.organizationId,
   495→      userId: ctx.user.id,
   496→      employeeId: ctx.employee.id,
   497→      newValues: {
   498→        flowopsOrderNumber,
   499→        status: orderStatus,
   500→        paymentType: d.payment_type,
   501→        paymentStatus,
   502→        itemCount: orderItemsData.length,
   503→        totalOrderValue,
   504→        customerId,
   505→      },
   506→    })
   507→
   508→    // 12. Update customer stats
   509→    await updateCustomerStats(customerId)
   510→
   511→    // 13. If the order auto-confirmed (payment-driven or company setting),
   512→    // run the stock reservation logic immediately.
   513→    if (orderStatus === 'confirmed') {
   514→      await reserveOrderStock(order.id, ctx)
   515→    }
   516→
   517→    await insertMetricEvent({
   518→      companyId: ctx.company.id,
   519→      entityType: 'order',
   520→      entityId: order.id,
   521→      metricKey: 'order.created',
   522→      numericValue: Number(totalOrderValue),
   523→      dimensions: { order_source: 'manual', payment_type: d.payment_type, company_id: ctx.company.id },
   524→    }).catch(() => {})
   525→
   526→    return {
   527→      success: true,
   528→      data: { orderId: order.id, flowopsOrderNumber, orderItems: createdItems },
   529→    }
   530→  } catch (err) {
   531→    return {
   532→      success: false,
   533→      error: err instanceof Error ? err.message : 'Failed to create order',
   534→    }
   535→  }
   536→}

   542→export async function createOrderFromShopifyWebhook(
   543→  payload: ShopifyOrderWebhook,
   544→  companyId: string,
   545→  organizationId: string,
   546→): Promise<ActionResult<{ orderId: string; flowopsOrderNumber: string }>> {
   547→  try {
   548→    const ctx = await getWorkspace()
   549→    // This function would be called by a future webhook handler.
   550→    // For now it's structured and unit-testable with a mock payload.
   551→
   552→    const parsed = shopifyOrderWebhookSchema.safeParse(payload)
   553→    if (!parsed.success) {
   554→      return { success: false, error: 'Invalid Shopify webhook payload' }
   555→    }
   556→    const d = parsed.data
   557→
   558→    // Map Shopify financial_status → FlowOps payment_status
   559→    let paymentStatus: string
   560→    let paymentSource: string = 'shopify_gateway'
   561→    let paymentType: string
   562→    let orderStatus: string = 'pending'
   563→    let confirmedAt: Date | null = null
   564→
   565→    switch (d.financial_status) {
   566→      case 'paid':
   567→        paymentStatus = 'fully_prepaid'
   568→        paymentType = 'fully_prepaid'
   569→        orderStatus = 'confirmed'
   570→        confirmedAt = new Date()
   571→        break
   572→      case 'partially_paid':
   573→        paymentStatus = 'advance_paid'
   574→        paymentType = 'partial_advance'
   575→        orderStatus = 'confirmed'
   576→        confirmedAt = new Date()
   577→        break
   578→      default:
   579→        paymentStatus = 'cod_pending'
   580→        paymentType = 'full_cod'
   581→        paymentSource = 'cod_native'
   582→        // Check company settings for COD orders
   583→        const settings = await db.companyOrderSetting.findUnique({ where: { companyId } })
   584→        if (!settings?.requireOrderConfirmation) {
   585→          orderStatus = 'confirmed'
   586→          confirmedAt = new Date()
   587→        }
   588→    }
   589→
   590→    // Find or create customer from Shopify customer data
   591→    const customerPhone = d.customer.phone || '0000000000' // fallback
   592→    let customer = await db.customer.findFirst({
   593→      where: { organizationId, phone: customerPhone },
   594→    })
   595→
   596→    if (!customer && d.customer.first_name) {
   597→      customer = await db.customer.create({
   598→        data: {
   599→          organizationId,
   600→          name: `${d.customer.first_name} ${d.customer.last_name ?? ''}`.trim(),
   601→          phone: customerPhone,
   602→          email: d.customer.email || null,
   603→          addresses: JSON.stringify(
   604→            d.customer.default_address
   605→              ? [{
   606→                  address: d.customer.default_address.address1 || '',
   607→                  city: d.customer.default_address.city || '',
   608→                  province: d.customer.default_address.province || '',
   609→                  is_default: true,
   610→                }]
   611→              : [],
   612→          ),
   613→        },
   614→      })
   615→    }
   616→
   617→    if (!customer) {
   618→      return { success: false, error: 'Could not resolve customer from Shopify payload' }
   619→    }
   620→
   621→    // Resolve variants by SKU
   622→    const skus = d.line_items.map((li) => li.sku).filter(Boolean) as string[]
   623→    const variants = await db.orgProductVariant.findMany({
   624→      where: { sku: { in: skus }, organizationId },
   625→    })
   626→
   627→    // Compute subtotal
   628→    let subtotal = 0
   629→    const orderItemsData: Array<{
   630→      orgVariantId: string
   631→      quantity: number
   632→      unitPrice: number
   633→      fulfillmentTypeSnapshot: string
   634→    }> = []

   637→      const variant = variants.find((v) => v.sku === li.sku)
   638→      if (!variant) continue // skip unmatched items for now
   639→      const unitPrice = parseFloat(li.price)
   640→      subtotal += unitPrice * li.quantity
   641→      orderItemsData.push({
   642→        orgVariantId: variant.id,
   643→        quantity: li.quantity,
   644→        unitPrice,
   645→        fulfillmentTypeSnapshot: variant.fulfillmentType,
   646→      })
   647→    }
   648→
   649→    if (orderItemsData.length === 0) {
   650→      return { success: false, error: 'No matching variants found for Shopify line items' }
   651→    }
   652→
   653→    const totalOrderValue = parseFloat(d.total_price)
   654→    const flowopsOrderNumber = await generateOrderNumber(companyId)
   655→
   656→    const order = await db.order.create({
   657→      data: {
   658→        organizationId,
   659→        companyId,
   660→        flowopsOrderNumber,
   661→        orderSource: 'shopify',
   662→        externalOrderReference: d.name,
   663→        externalOrderId: String(d.id),
   664→        customerId: customer.id,
   665→        status: orderStatus,
   666→        paymentType,
   667→        paymentStatus,
   668→        paymentSource,
   669→        subtotal,
   670→        totalOrderValue,
   671→        confirmedAt,
   672→        deliveryAddress: d.customer.default_address?.address1 || null,
   673→        deliveryCity: d.customer.default_address?.city || null,
   674→      },
   675→    })
   676→
   677→    for (const itemData of orderItemsData) {
   678→      await db.orderItem.create({
   679→        data: {
   680→          orderId: order.id,
   681→          orgVariantId: itemData.orgVariantId,
   682→          organizationId,
   683→          quantity: itemData.quantity,
   684→          unitPrice: itemData.unitPrice,
   685→          lineTotal: itemData.quantity * itemData.unitPrice,
   686→          fulfillmentStatus: 'reserved',
   687→          fulfillmentTypeSnapshot: itemData.fulfillmentTypeSnapshot,
   688→        },
   689→      })
   690→    }
   691→
   692→    await insertAuditLog({
   693→      action: 'order.created_from_shopify',
   694→      entityType: 'order',
   695→      entityId: order.id,
   696→      companyId,
   697→      organizationId,
   698→      userId: ctx.user.id,
   699→      employeeId: ctx.employee.id,
   700→      newValues: {
   701→        flowopsOrderNumber,
   702→        externalOrderReference: d.name,
   703→        financialStatus: d.financial_status,
   704→        totalOrderValue,
   705→      },
   706→    })
   707→
   708→    await updateCustomerStats(customer.id)
   709→
   710→    await insertMetricEvent({
   711→      companyId,
   712→      entityType: 'order',
   713→      entityId: order.id,
   714→      metricKey: 'order.created',
   715→      numericValue: Number(totalOrderValue),
   716→      dimensions: { order_source: 'shopify', payment_type: paymentType, company_id: companyId },
   717→    }).catch(() => {})
   718→
   719→    return { success: true, data: { orderId: order.id, flowopsOrderNumber } }
   720→  } catch (err) {
   721→    return {
   722→      success: false,
   723→      error: err instanceof Error ? err.message : 'Failed to create order from Shopify webhook',
   724→    }
   725→  }

   853→      newValues: {

   855→        paymentStatus: newPaymentStatus,
   856→        paymentSource: 'manual_conversion',
   857→        advanceAmount: d.advance_amount,

   859→      },
   860→    })

   862→    await insertMetricEvent({

   874→      success: false,
   875→      error: err instanceof Error ? err.message : 'Failed to convert payment status',

   944→// ──────────────────────────────────────────────────────────────
   945→
   946→export async function cancelOrder(input: CancelOrderInput): Promise<ActionResult> {
   947→  try {
   948→    const ctx = await getWorkspace()
   949→    await requirePermission(ctx, PERMISSIONS.ORDERS_CANCEL)
   950→
   951→    const parsed = cancelOrderSchema.safeParse(input)
   952→    if (!parsed.success) {
   953→      return { success: false, error: parsed.error.issues[0]?.message ?? 'Invalid input' }
   954→    }
   955→    const d = parsed.data
   956→
   957→    const order = await db.order.findFirst({
   958→      where: { id: d.order_id, companyId: ctx.company.id },
   959→    })
   960→    if (!order) return { success: false, error: 'Order not found' }
   961→
   962→    const nonCancellableStatuses = ['dispatched', 'delivered', 'rto', 'cancelled', 'refunded']
   963→    if (nonCancellableStatuses.includes(order.status)) {
   964→      return {
   965→        success: false,
   966→        error: `Cannot cancel an order that is already ${order.status} (use the returns flow instead)`,
   967→      }
   968→    }
   969→
   970→    await db.order.update({
   971→      where: { id: d.order_id },
   972→      data: {
   973→        status: 'cancelled',
   974→        cancelledAt: new Date(),
   975→        cancellationReason: d.cancellation_reason,
   976→      },
   977→    })
   978→
   979→    // Step 3: Unreserve stock for any items with fulfillment_status='reserved'.
   980→    // Items with fulfillment_status='backordered' need no inventory action
   981→    // (no reservation ever existed for them) — they'll be orphaned since
   982→    // the order is now cancelled and won't be picked up by future
   983→    // checkAndFulfillBackorders() runs (which skip cancelled orders).
   984→    const reservedItems = await db.orderItem.findMany({
   985→      where: { orderId: d.order_id, fulfillmentStatus: 'reserved' },
   986→    })
   987→
   988→    for (const item of reservedItems) {
   989→      const locationId = item.reservedLocationId ?? order.dispatchLocationId
   990→      if (!locationId) continue
   991→
   992→      await unreserveStockForOrder({
   993→        orgVariantId: item.orgVariantId,
   994→        locationId,
   995→        organizationId: order.organizationId,
   996→        companyId: ctx.company.id,
   997→        employeeId: ctx.employee.id,
   998→        quantity: item.quantity,
   999→        orderId: d.order_id,
  1000→      })
  1001→    }
  1002→
  1003→    await insertAuditLog({
  1004→      action: 'order.cancelled',
  1005→      entityType: 'order',
  1006→      entityId: d.order_id,
  1007→      companyId: ctx.company.id,
  1008→      organizationId: ctx.company.organizationId,
  1009→      userId: ctx.user.id,
  1010→      employeeId: ctx.employee.id,
  1011→      oldValues: { status: order.status },
  1012→      newValues: { status: 'cancelled', reason: d.cancellation_reason, unreservedItems: reservedItems.length },
  1013→    })
  1014→
  1015→    await insertMetricEvent({
  1016→      companyId: ctx.company.id,
  1017→      entityType: 'order',
  1018→      entityId: d.order_id,
  1019→      metricKey: 'order.cancelled',
  1020→      numericValue: Number(order.totalOrderValue),
  1021→      dimensions: { cancellation_reason: d.cancellation_reason, had_reserved_items: reservedItems.length > 0 },
  1022→    }).catch(() => {})
  1023→
  1024→    return { success: true }
  1025→  } catch (err) {
  1026→    return {
  1027→      success: false,
  1028→      error: err instanceof Error ? err.message : 'Failed to cancel order',
  1029→    }
  1030→  }
  1031→}

  1208→          advanceAmount: order.advanceAmount ? Number(order.advanceAmount) : null,
  1209→          codCollectedAmount: order.codCollectedAmount ? Number(order.codCollectedAmount) : null,
  1210→          deliveryAddress: order.deliveryAddress,
  1211→        },
  1212→        customer: order.customer,
  1213→        items: order.items.map((item) => ({
  1214→          id: item.id,
  1215→          orgVariantId: item.orgVariantId,
  1216→          quantity: item.quantity,
  1217→          unitPrice: Number(item.unitPrice),
  1218→          lineTotal: Number(item.lineTotal),
  1219→          fulfillmentStatus: item.fulfillmentStatus,
  1220→          fulfillmentTypeSnapshot: item.fulfillmentTypeSnapshot,
  1221→          variant: {
  1222→            sku: item.orgVariant.sku,
  1223→            productTitle: item.orgVariant.product.title,
  1224→            attributeValues: JSON.parse(item.orgVariant.attributeValues),
  1225→          },
  1226→        })),
  1227→      },
  1228→    }
  1229→  } catch (err) {
  1230→    return {
  1231→      success: false,
  1232→      error: err instanceof Error ? err.message : 'Failed to get order detail',
  1233→    }
  1234→  }
  1235→}
  1236→
  1237→// ──────────────────────────────────────────────────────────────
  1238→// Step 3: Dispatch + Processing/Packing actions
  1239→// ──────────────────────────────────────────────────────────────
  1240→
  1241→/**
  1242→ * Dispatch an order — deducts stock from inventory, sets tracking info.
  1243→ *
  1244→ * Business rules:
  1245→ *   - If order.status = 'pending' (confirmation was skipped): run the
  1246→ *     full confirmation/reservation logic inline first, then proceed.
  1247→ *   - If any item is still 'backordered' after reservation: BLOCK dispatch
  1248→ *     with a clear error (hard rule — no split shipments).
  1249→ *   - For each reserved item: call dispatchOrder() (inventory) which
  1250→ *     deducts on_hand, releases the reservation, and locks COGS.
  1251→ *   - The backfill_order_timestamps() trigger auto-sets confirmed_at/
  1252→ *     packed_at if they were still NULL.
  1253→ *   - If company_order_settings.require_packing_step = TRUE, verify
  1254→ *     order.packed_at IS NOT NULL before proceeding.
  1255→ */
  1256→export async function dispatchOrderAction(
  1257→  orderId: string,
  1258→  trackingNumber: string,
  1259→  courierName?: string,
  1260→): Promise<ActionResult> {
  1261→  try {
  1262→    const ctx = await getWorkspace()
  1263→    await requirePermission(ctx, PERMISSIONS.ORDERS_FULFILL)
  1264→
  1265→    const order = await db.order.findFirst({
  1266→      where: { id: orderId, companyId: ctx.company.id },
  1267→    })
  1268→    if (!order) return { success: false, error: 'Order not found' }
  1269→
  1270→    // Cannot dispatch already-dispatched/delivered/cancelled orders
  1271→    if (['dispatched', 'delivered', 'rto', 'cancelled', 'refunded'].includes(order.status)) {
  1272→      return { success: false, error: `Cannot dispatch an order that is already ${order.status}` }
  1273→    }
  1274→
  1275→    // Check packing requirement
  1276→    const settings = await db.companyOrderSetting.findUnique({
  1277→      where: { companyId: ctx.company.id },
  1278→    })
  1279→    if (settings?.requirePackingStep && !order.packedAt) {
  1280→      return {
  1281→        success: false,
  1282→        error: 'This order must be marked as packed before dispatching. Use markOrderPacked() first.',
  1283→      }
  1284→    }
  1285→
  1286→    // If order is still 'pending' (confirmation was skipped), run the
  1287→    // full confirmation + reservation logic inline first.
  1288→    if (order.status === 'pending') {
  1289→      await db.order.update({
  1290→        where: { id: orderId },
  1291→        data: { status: 'confirmed', confirmedAt: new Date() },
  1292→      })
  1293→      await reserveOrderStock(orderId, ctx)
  1294→    }
  1295→
  1296→    // Re-fetch the order to get the updated status after reservation
  1297→    const updatedOrder = await db.order.findFirst({
  1298→      where: { id: orderId },
  1299→      select: { status: true, dispatchLocationId: true, organizationId: true },
  1300→    })
  1301→    if (!updatedOrder) return { success: false, error: 'Order not found after reservation' }
  1302→
  1303→    // Check for backordered items — BLOCK dispatch if any exist
  1304→    const backorderedItems = await db.orderItem.findMany({
  1305→      where: { orderId, fulfillmentStatus: 'backordered' },
  1306→      include: {
  1307→        orgVariant: { select: { sku: true } },
  1308→      },
  1309→    })
  1310→
  1311→    if (backorderedItems.length > 0) {
  1312→      const itemList = backorderedItems
  1313→        .map((i) => `${i.orgVariant.sku} (qty: ${i.quantity})`)
  1314→        .join(', ')
  1315→      return {
  1316→        success: false,
  1317→        error: `Cannot dispatch: ${backorderedItems.length} item(s) are still backordered: ${itemList}. Receive stock or cancel those items first.`,
  1318→      }
  1319→    }
  1320→
  1321→    // Dispatch each reserved item
  1322→    const itemsToDispatch = await db.orderItem.findMany({
  1323→      where: { orderId, fulfillmentStatus: 'reserved' },
  1324→    })
  1325→
  1326→    for (const item of itemsToDispatch) {
  1327→      const locationId = item.reservedLocationId ?? updatedOrder.dispatchLocationId
  1328→      if (!locationId) {
  1329→        return { success: false, error: `Order item ${item.id} has no dispatch location` }
  1330→      }
  1331→
  1332→      const dispatchResult = await dispatchInventory({
  1333→        orgVariantId: item.orgVariantId,
  1334→        locationId,
  1335→        organizationId: updatedOrder.organizationId,
  1336→        companyId: ctx.company.id,
  1337→        employeeId: ctx.employee.id,
  1338→        quantity: item.quantity,
  1339→        orderId,
  1340→      })
  1341→
  1342→      if (!dispatchResult.success) {
  1343→        return {
  1344→          success: false,
  1345→          error: `Failed to dispatch item: ${dispatchResult.error}`,
  1346→        }
  1347→      }
  1348→
  1349→      // Mark item as dispatched
  1350→      await db.orderItem.update({
  1351→        where: { id: item.id },
  1352→        data: { fulfillmentStatus: 'dispatched', fulfilledAt: new Date() },
  1353→      })
  1354→    }
  1355→
  1356→    // Update order status — the backfill_order_timestamps() trigger
  1357→    // will auto-set confirmedAt/packedAt if they were still NULL.
  1358→    await db.order.update({
  1359→      where: { id: orderId },
  1360→      data: {
  1361→        status: 'dispatched',
  1362→        dispatchedAt: new Date(),
  1363→        trackingNumber: trackingNumber || null,
  1364→        courierName: courierName || order.courierName,
  1365→      },
  1366→    })
  1367→
  1368→    await insertAuditLog({
  1369→      action: 'order.dispatched',
  1370→      entityType: 'order',
  1371→      entityId: orderId,
  1372→      companyId: ctx.company.id,
  1373→      organizationId: ctx.company.organizationId,
  1374→      userId: ctx.user.id,
  1375→      employeeId: ctx.employee.id,
  1376→      oldValues: { status: updatedOrder.status },
  1377→      newValues: {
  1378→        status: 'dispatched',
  1379→        trackingNumber,
  1380→        courierName,
  1381→        itemsDispatched: itemsToDispatch.length,
  1382→      },
  1383→    })
  1384→
  1385→    const timeToDispatchHours = order.createdAt
  1386→      ? Math.round((new Date().getTime() - new Date(order.createdAt).getTime()) / (1000 * 60 * 60))
  1387→      : 0
  1388→    await insertMetricEvent({
  1389→      companyId: ctx.company.id,
  1390→      entityType: 'order',
  1391→      entityId: orderId,
  1392→      metricKey: 'order.dispatched',
  1393→      numericValue: Number(order.totalOrderValue),
  1394→      dimensions: {
  1395→        courier_name: courierName || order.courierName,
  1396→        employee_id: ctx.employee.id,
  1397→        skipped_confirmation: updatedOrder.status === 'pending',
  1398→        skipped_packing: !order.packedAt,
  1399→        time_to_dispatch_hours: timeToDispatchHours,
  1400→      },
  1401→    }).catch(() => {})
  1402→
  1403→    // Update customer stats
  1404→    const orderWithCustomer = await db.order.findUnique({
  1405→      where: { id: orderId },
  1406→      select: { customerId: true },
  1407→    })
  1408→    if (orderWithCustomer) {
  1409→      await updateCustomerStats(orderWithCustomer.customerId)
  1410→    }
  1411→
  1412→    return { success: true }
  1413→  } catch (err) {
  1414→    return {
  1415→      success: false,
  1416→      error: err instanceof Error ? err.message : 'Failed to dispatch order',
  1417→    }
  1418→  }
  1419→}
