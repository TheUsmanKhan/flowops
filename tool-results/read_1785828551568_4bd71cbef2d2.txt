    23→} from '@/components/ui/select'

    50→
    51→interface ExchangeDetail {
    52→  id: string
    53→  exchangeMethod: 'courier_replacement' | 'customer_self_return'
    54→  status: string
    55→  reason: string
    56→  oldItemPrice: number
    57→  newItemPrice: number
    58→  priceDifference: number
    59→  priceDifferenceStatus: string
    60→  priceDifferenceSettledAmount: number | null
    61→  priceDifferenceSettledAt: string | null
    62→  requestedAt: string
    63→  completedAt: string | null
    64→  cancelledAt: string | null
    65→  cancellationReason: string | null
    66→  oldItemCondition: string | null
    67→  oldItemVerifiedAt: string | null
    68→  oldItemNotes: string | null
    69→  oldItemEvidenceUrls: string
    70→  customerReturnTrackingNumber: string | null
    71→  customerReturnCourier: string | null
    72→  customerConfirmedShippedAt: string | null
    73→  markedAsNotReturned: boolean
    74→  notReturnedReason: string | null
    75→  notReturnedRecoveryStatus: string | null
    76→  notReturnedRecoveryAmount: number | null
    77→  originalOrder: {
    78→    id: string
    79→    flowopsOrderNumber: string
    80→    status: string
    81→    customer: { id: string; name: string; phones: Array<{ phoneRaw: string }> }
    82→  }
    83→  originalOrderItem: {
    84→    id: string
    85→    quantity: number
    86→    unitPrice: number
    87→    fulfillmentTypeSnapshot: string
    88→    orgVariant: { sku: string; product: { title: string } }
    89→  }
    90→  newOrgVariant: {
    91→    id: string
    92→    sku: string
    93→    fulfillmentType: string
    94→    product: { title: string }
    95→  }
    96→  newOrder: {
    97→    id: string
    98→    flowopsOrderNumber: string
    99→    status: string
   100→    dispatchedAt: string | null
   101→    deliveredAt: string | null
   102→  } | null
   103→  newOrderItem: { id: string; fulfillmentStatus: string } | null
   104→  requestedByEmployee: { id: string; user: { fullName: string } }
   105→  oldItemVerifiedByEmployee: { id: string; user: { fullName: string } } | null
   106→}
   107→
   108→const STATUS_BADGE: Record<string, { label: string; className: string }> = {
   109→  requested: { label: 'Requested', className: 'bg-gray-100 text-gray-700 border-gray-200' },
   110→  new_item_dispatched: { label: 'New Item Dispatched', className: 'bg-violet-50 text-violet-700 border-violet-200' },
   111→  awaiting_old_item_return: { label: 'Awaiting Old Item', className: 'bg-amber-50 text-amber-700 border-amber-200' },
   112→  awaiting_customer_to_ship_old_item: { label: 'Awaiting Customer Ship', className: 'bg-amber-50 text-amber-700 border-amber-200' },
   113→  customer_confirmed_shipped: { label: 'Customer Shipped', className: 'bg-sky-50 text-sky-700 border-sky-200' },
   114→  old_item_manually_verified: { label: 'Old Item Verified', className: 'bg-emerald-50 text-emerald-700 border-emerald-200' },
   115→  completed: { label: 'Completed', className: 'bg-emerald-50 text-emerald-700 border-emerald-200' },
   116→  customer_did_not_return: { label: 'Not Returned', className: 'bg-rose-50 text-rose-700 border-rose-200' },
   117→  cancelled: { label: 'Cancelled', className: 'bg-slate-100 text-slate-500 border-slate-200' },
   118→}
   119→
   120→const METHOD_BADGE: Record<string, { label: string; className: string; icon: typeof Truck }> = {
   121→  courier_replacement: { label: 'Courier Replacement', className: 'bg-violet-50 text-violet-700 border-violet-200', icon: Truck },
   122→  customer_self_return: { label: 'Customer Self-Return', className: 'bg-sky-50 text-sky-700 border-sky-200', icon: UserCheck },
   123→}
   124→
   125→export function ExchangeDetailView({ exchangeId }: { exchangeId: string }) {
   126→  const navigate = useAppStore((s) => s.navigate)
   127→  const queryClient = useQueryClient()
   128→
   129→  const [verifyOpen, setVerifyOpen] = useState(false)
   130→  const [settleOpen, setSettleOpen] = useState(false)
   131→  const [notReturnedOpen, setNotReturnedOpen] = useState(false)
   132→
   133→  const query = useQuery<ExchangeDetail>({
   134→    queryKey: ['exchange', exchangeId],
   135→    queryFn: () => api.get<ExchangeDetail>(`/api/exchanges/${exchangeId}`),
   136→    staleTime: 10_000,
   137→  })
   138→
   139→  const invalidate = () => {
   140→    queryClient.invalidateQueries({ queryKey: ['exchanges'] })
   141→    queryClient.invalidateQueries({ queryKey: ['exchange', exchangeId] })
   142→    queryClient.invalidateQueries({ queryKey: ['inventory-pools'] })
   143→  }
   144→
   145→  const settleMutation = useMutation({
   146→    mutationFn: (data: { amount: number; type: string }) =>
   147→      api.post(`/api/exchanges/${exchangeId}/settle-price-difference`, {
   148→        settled_amount: data.amount,
   149→        settlement_type: data.type,
   150→      }),
   151→    onSuccess: () => { toast.success('Price difference settled.'); setSettleOpen(false); invalidate() },
   152→    onError: (err) => toast.error(getErrorMessage(err)),
   153→  })
   154→
   155→  const notReturnedMutation = useMutation({
   156→    mutationFn: (data: { reason: string; recoveryStatus: string; recoveryAmount: number }) =>
   157→      api.post(`/api/exchanges/${exchangeId}/mark-not-returned`, {
   158→        not_returned_reason: data.reason,
   159→        recovery_status: data.recoveryStatus,
   160→        recovery_amount: data.recoveryAmount || undefined,
   161→      }),
   162→    onSuccess: () => { toast.success('Marked as not returned.'); setNotReturnedOpen(false); invalidate() },
   163→    onError: (err) => toast.error(getErrorMessage(err)),
   164→  })
   165→
   166→  if (query.isLoading) {
   167→    return (
   168→      <div className="space-y-4">
   169→        <Skeleton className="h-8 w-48" />
   170→        <Skeleton className="h-64" />
   171→      </div>
   172→    )
   173→  }
   174→
   175→  if (query.isError || !query.data) {
   176→    return (
   177→      <div className="space-y-4">
   178→        <Button variant="ghost" onClick={() => navigate({ name: 'exchanges' })}>
   179→          <ArrowLeft className="h-4 w-4" /> Back to Exchanges
   180→        </Button>
   181→        <Card>
   182→          <CardContent className="p-10 text-center">
   183→            <p className="text-sm text-muted-foreground">{query.isError ? getErrorMessage(query.error) : 'Exchange not found'}</p>
   184→          </CardContent>
   185→        </Card>
   186→      </div>
   187→    )
   188→  }
   189→
   190→  const e = query.data
   191→  const statusBadge = STATUS_BADGE[e.status] ?? { label: e.status, className: 'bg-gray-100 text-gray-700 border-gray-200' }
   192→  const methodBadge = METHOD_BADGE[e.exchangeMethod]
   193→  const MethodIcon = methodBadge.icon
   194→  const isTerminal = ['completed', 'customer_did_not_return', 'cancelled'].includes(e.status)
   195→  const needsSettlement = e.priceDifferenceStatus === 'customer_owes' || e.priceDifferenceStatus === 'refund_due'
   196→
   197→  // Parse evidence URLs
   198→  let evidenceUrls: string[] = []
   199→  try { evidenceUrls = JSON.parse(e.oldItemEvidenceUrls || '[]') } catch { /* ignore */ }
   200→
   201→  // Timeline events
   202→  const timeline: Array<{ label: string; time: string | null; icon: typeof Clock }> = [
   203→    { label: 'Exchange Requested', time: e.requestedAt, icon: RefreshCw },
   204→    { label: 'Customer Confirmed Shipped', time: e.customerConfirmedShippedAt, icon: UserCheck },
   205→    { label: 'Old Item Verified', time: e.oldItemVerifiedAt, icon: Package },
   206→    { label: 'Completed', time: e.completedAt, icon: CheckCircle2 },
   207→  ].filter((t) => t.time !== null || t.label === 'Exchange Requested')
   208→
   209→  return (
   210→    <div className="space-y-6">
   211→      <PageHeader
   212→        title={`Exchange ${e.originalOrder.flowopsOrderNumber}`}
   213→        description={`${methodBadge.label} · ${statusBadge.label}`}
   214→        actions={
   215→          <Button variant="ghost" size="sm" onClick={() => navigate({ name: 'exchanges' })}>
   216→            <ArrowLeft className="h-4 w-4" /> Back
   217→          </Button>
   218→        }
   219→      />
   220→
   221→      <div className="grid gap-6 lg:grid-cols-3">
   222→        {/* Main column */}
   223→        <div className="lg:col-span-2 space-y-6">
   224→          {/* Items */}
   225→          <Card>
   226→            <CardHeader><CardTitle className="text-base">Items</CardTitle></CardHeader>
   227→            <CardContent className="space-y-3">
   228→              {/* Old item */}
   229→              <div className="rounded-md border bg-muted/30 p-3">
   230→                <p className="text-[10px] uppercase tracking-wide text-muted-foreground mb-1">Old Item (being exchanged)</p>
   231→                <div className="flex items-center justify-between">
   232→                  <div>
   233→                    <p className="text-sm font-medium">{e.originalOrderItem.orgVariant.product.title}</p>
   234→                    <p className="text-xs text-muted-foreground font-mono">{e.originalOrderItem.orgVariant.sku}</p>
   235→                    <Badge variant="outline" className="text-[10px] mt-1 bg-sky-50 text-sky-700 border-sky-200">
   236→                      {e.originalOrderItem.fulfillmentTypeSnapshot === 'made_to_order' ? 'MTO' : 'Stock'}
   237→                    </Badge>
   238→                  </div>
   239→                  <p className="text-sm font-medium tabular-nums">{formatPKR(e.oldItemPrice)}</p>
   240→                </div>
   241→              </div>
   242→              {/* Arrow */}
   243→              <div className="flex items-center justify-center">
   244→                <RefreshCw className="h-5 w-5 text-muted-foreground" />
   245→              </div>
   246→              {/* New item */}
   247→              <div className="rounded-md border border-primary/30 bg-primary/5 p-3">
   248→                <p className="text-[10px] uppercase tracking-wide text-muted-foreground mb-1">New Item (replacement)</p>
   249→                <div className="flex items-center justify-between">
   250→                  <div>
   251→                    <p className="text-sm font-medium">{e.newOrgVariant.product.title}</p>
   252→                    <p className="text-xs text-muted-foreground font-mono">{e.newOrgVariant.sku}</p>
   253→                    <Badge variant="outline" className="text-[10px] mt-1 bg-sky-50 text-sky-700 border-sky-200">
   254→                      {e.newOrgVariant.fulfillmentType === 'made_to_order' ? 'MTO' : 'Stock'}
   255→                    </Badge>
   256→                  </div>
   257→                  <div className="text-right">
   258→                    <p className="text-sm font-medium tabular-nums">{formatPKR(e.newItemPrice)}</p>
   259→                    {e.newOrder && (
   260→                      <button
   261→                        onClick={() => navigate({ name: 'order-detail', id: e.newOrder!.id })}
   262→                        className="text-[10px] text-primary hover:underline flex items-center gap-0.5 mt-1"
   263→                      >
   264→                        {e.newOrder.flowopsOrderNumber} <ExternalLink className="h-2.5 w-2.5" />
   265→                      </button>
   266→                    )}
   267→                  </div>
   268→                </div>
   269→              </div>
   270→            </CardContent>
   271→          </Card>
   272→
   273→          {/* Verification details */}
   274→          {e.oldItemVerifiedAt && (
   275→            <Card>
   276→              <CardHeader><CardTitle className="text-base flex items-center gap-2"><Package className="h-4 w-4" /> Old Item Verification</CardTitle></CardHeader>
   277→              <CardContent className="space-y-3">
   278→                <div className="grid sm:grid-cols-2 gap-3 text-sm">
   279→                  <div>
   280→                    <p className="text-xs text-muted-foreground">Condition</p>
   281→                    <Badge variant="outline" className={cn(
   282→                      'mt-1',
   283→                      e.oldItemCondition === 'damaged' ? 'bg-rose-50 text-rose-700 border-rose-200' : 'bg-emerald-50 text-emerald-700 border-emerald-200',
   284→                    )}>
   285→                      {e.oldItemCondition}
   286→                    </Badge>
   287→                  </div>
   288→                  <div>
   289→                    <p className="text-xs text-muted-foreground">Verified By</p>
   290→                    <p className="text-sm font-medium">{e.oldItemVerifiedByEmployee?.user.fullName ?? '—'}</p>
   291→                  </div>
   292→                  <div>
   293→                    <p className="text-xs text-muted-foreground">Verified At</p>
   294→                    <p className="text-sm">{formatDateTime(e.oldItemVerifiedAt)}</p>
   295→                  </div>
   296→                </div>
   297→                {e.oldItemNotes && (
   298→                  <div>
   299→                    <p className="text-xs text-muted-foreground">Notes</p>
   300→                    <p className="text-sm mt-0.5">{e.oldItemNotes}</p>
   301→                  </div>
   302→                )}
   303→                {evidenceUrls.length > 0 && (
   304→                  <div>
   305→                    <p className="text-xs text-muted-foreground flex items-center gap-1"><Camera className="h-3 w-3" /> Evidence Photos</p>
   306→                    <div className="flex flex-wrap gap-2 mt-1">
   307→                      {evidenceUrls.map((url, i) => (
   308→                        <img key={i} src={url} alt={`Evidence ${i + 1}`} className="h-16 w-16 rounded object-cover border" />
   309→                      ))}
   310→                    </div>
   311→                  </div>
   312→                )}
   313→              </CardContent>
   314→            </Card>
   315→          )}
   316→
   317→          {/* Timeline */}
   318→          <Card>
   319→            <CardHeader><CardTitle className="text-base">Timeline</CardTitle></CardHeader>
   320→            <CardContent>
   321→              <ol className="space-y-3">
   322→                {timeline.map((t, i) => {
   323→                  const Icon = t.icon
   324→                  return (
   325→                    <li key={i} className="flex items-start gap-2.5">
   326→                      <div className={cn(
   327→                        'flex h-6 w-6 items-center justify-center rounded-full shrink-0',
   328→                        t.time ? 'bg-primary/10 text-primary' : 'bg-muted text-muted-foreground',
   329→                      )}>
   330→                        <Icon className="h-3 w-3" />
   331→                      </div>
   332→                      <div>
   333→                        <p className={cn('text-sm', t.time ? 'font-medium' : 'text-muted-foreground')}>{t.label}</p>
   334→                        {t.time && <p className="text-xs text-muted-foreground">{formatDateTime(t.time)}</p>}
   335→                      </div>
   336→                    </li>
   337→                  )
   338→                })}
   339→              </ol>
   340→            </CardContent>
   341→          </Card>
   342→        </div>
   343→
   344→        {/* Sidebar column */}
   345→        <div className="space-y-6">
   346→          {/* Price difference */}
   347→          <Card>
   348→            <CardHeader><CardTitle className="text-base flex items-center gap-2"><Banknote className="h-4 w-4" /> Price Difference</CardTitle></CardHeader>
   349→            <CardContent className="space-y-2 text-sm">
   350→              <div className="flex justify-between">
   351→                <span className="text-muted-foreground">Old item</span>
   352→                <span className="tabular-nums">{formatPKR(e.oldItemPrice)}</span>
   353→              </div>
   354→              <div className="flex justify-between">
   355→                <span className="text-muted-foreground">New item</span>
   356→                <span className="tabular-nums">{formatPKR(e.newItemPrice)}</span>
   357→              </div>
   358→              <Separator className="my-2" />
   359→              <div className="flex justify-between font-medium">
   360→                <span>Difference</span>
   361→                <span className={cn(
   362→                  'tabular-nums',
   363→                  e.priceDifference > 0 ? 'text-amber-700' : e.priceDifference < 0 ? 'text-sky-700' : 'text-muted-foreground',
   364→                )}>
   365→                  {e.priceDifference > 0 ? '+' : ''}{formatPKR(e.priceDifference)}
   366→                </span>
   367→              </div>
   368→              <div className="flex justify-between">
   369→                <span className="text-muted-foreground">Status</span>
   370→                <Badge variant="outline" className={cn(
   371→                  'text-[10px]',
   372→                  e.priceDifferenceStatus === 'settled' ? 'bg-emerald-50 text-emerald-700 border-emerald-200' : 'bg-amber-50 text-amber-700 border-amber-200',
   373→                )}>
   374→                  {e.priceDifferenceStatus.replace(/_/g, ' ')}
   375→                </Badge>
   376→              </div>
   377→              {e.priceDifferenceSettledAmount !== null && (
   378→                <p className="text-xs text-muted-foreground pt-1">
   379→                  Settled: {formatPKR(e.priceDifferenceSettledAmount)} on {formatDateTime(e.priceDifferenceSettledAt)}
   380→                </p>
   381→              )}
   382→              {needsSettlement && (
   383→                <Button size="sm" className="w-full mt-2" onClick={() => setSettleOpen(true)}>
   384→                  <Banknote className="h-3.5 w-3.5" /> Settle Payment
   385→                </Button>
   386→              )}
   387→            </CardContent>
   388→          </Card>
   389→
   390→          {/* Customer self-return tracking */}
   391→          {e.exchangeMethod === 'customer_self_return' && (
   392→            <Card>
   393→              <CardHeader><CardTitle className="text-base flex items-center gap-2"><UserCheck className="h-4 w-4" /> Customer Return Tracking</CardTitle></CardHeader>
   394→              <CardContent className="space-y-2 text-sm">
   395→                <div className="flex justify-between">
   396→                  <span className="text-muted-foreground">Tracking #</span>
   397→                  <span className="font-mono text-xs">{e.customerReturnTrackingNumber ?? '—'}</span>
   398→                </div>
   399→                <div className="flex justify-between">
   400→                  <span className="text-muted-foreground">Courier</span>
   401→                  <span>{e.customerReturnCourier ?? '—'}</span>
   402→                </div>
   403→                <div className="flex justify-between">
   404→                  <span className="text-muted-foreground">Confirmed Shipped</span>
   405→                  <span>{e.customerConfirmedShippedAt ? formatDateTime(e.customerConfirmedShippedAt) : '—'}</span>
   406→                </div>
   407→              </CardContent>
   408→            </Card>
   409→          )}
   410→
   411→          {/* Not returned info */}
   412→          {e.markedAsNotReturned && (
   413→            <Card className="border-rose-200">
   414→              <CardHeader><CardTitle className="text-base flex items-center gap-2 text-rose-700"><XCircle className="h-4 w-4" /> Not Returned</CardTitle></CardHeader>
   415→              <CardContent className="space-y-2 text-sm">
   416→                <p><span className="text-muted-foreground">Reason:</span> {e.notReturnedReason}</p>
   417→                <p><span className="text-muted-foreground">Recovery:</span> {e.notReturnedRecoveryStatus}</p>
   418→                {e.notReturnedRecoveryAmount !== null && (
   419→                  <p><span className="text-muted-foreground">Amount:</span> {formatPKR(e.notReturnedRecoveryAmount)}</p>
   420→                )}
   421→              </CardContent>
   422→            </Card>
   423→          )}
   424→
   425→          {/* Actions */}
   426→          <Card>
   427→            <CardHeader><CardTitle className="text-base">Actions</CardTitle></CardHeader>
   428→            <CardContent className="space-y-2">
   429→              {e.status === 'awaiting_customer_to_ship_old_item' && (
   430→                <Button className="w-full" variant="outline" onClick={() => navigate({ name: 'exchanges' })}>
   431→                  <UserCheck className="h-3.5 w-3.5" /> Confirm Customer Shipped (from list)
   432→                </Button>
   433→              )}
   434→              {(e.status === 'customer_confirmed_shipped' || e.status === 'awaiting_old_item_return') && (
   435→                <Button className="w-full" onClick={() => setVerifyOpen(true)}>
   436→                  <Package className="h-3.5 w-3.5" /> Verify Old Item Received
   437→                </Button>
   438→              )}
   439→              {!isTerminal && (
   440→                <Button
   441→                  className="w-full"
   442→                  variant="outline"
   443→                  onClick={() => setNotReturnedOpen(true)}
   444→                >
   445→                  <XCircle className="h-3.5 w-3.5" /> Mark as Not Returned
   446→                </Button>
   447→              )}
   448→              {e.status === 'requested' && e.exchangeMethod === 'courier_replacement' && (
   449→                <Button className="w-full" variant="outline" onClick={() => navigate({ name: 'exchanges' })}>
   450→                  <Truck className="h-3.5 w-3.5" /> Dispatch New Item (from list)
   451→                </Button>
   452→              )}
   453→            </CardContent>
   454→          </Card>
   455→        </div>
   456→      </div>
   457→
   458→      {/* Dialogs */}
   459→      {verifyOpen && (
   460→        <VerifyOldItemDialog
   461→          open={verifyOpen}
   462→          onOpenChange={setVerifyOpen}
   463→          exchangeId={exchangeId}
   464→          exchangeMethod={e.exchangeMethod}
   465→          onVerified={() => invalidate()}
   466→        />
   467→      )}
   468→      {settleOpen && (
   469→        <SettleDialog
   470→          exchange={e}
   471→          open={settleOpen}
   472→          onOpenChange={setSettleOpen}
   473→          loading={settleMutation.isPending}
   474→          onConfirm={(amount, type) => settleMutation.mutate({ amount, type })}
   475→        />
   476→      )}
   477→      {notReturnedOpen && (
   478→        <NotReturnedDialog
   479→          exchange={e}
   480→          open={notReturnedOpen}
   481→          onOpenChange={setNotReturnedOpen}
   482→          loading={notReturnedMutation.isPending}
   483→          onConfirm={(reason, recoveryStatus, recoveryAmount) =>
   484→            notReturnedMutation.mutate({ reason, recoveryStatus, recoveryAmount })
   485→          }
   486→        />
   487→      )}
   488→    </div>
   489→  )

   496→function SettleDialog({
   497→  exchange,
   498→  open,
   499→  onOpenChange,
   500→  loading,
   501→  onConfirm,
   502→}: {
   503→  exchange: ExchangeDetail
   504→  open: boolean
   505→  onOpenChange: (open: boolean) => void
   506→  loading: boolean
   507→  onConfirm: (amount: number, type: string) => void
   508→}) {
   509→  const isCustomerOwes = exchange.priceDifference > 0
   510→  const [amount, setAmount] = useState(Math.abs(exchange.priceDifference).toString())
   511→  const [type, setType] = useState(isCustomerOwes ? 'collected_from_customer' : 'refunded_to_customer')
   512→
   513→  return (
   514→    <Dialog open={open} onOpenChange={onOpenChange}>
   515→      <DialogContent className="sm:max-w-md">
   516→        <DialogHeader>
   517→          <DialogTitle>Settle Price Difference</DialogTitle>
   518→          <DialogDescription>
   519→            {isCustomerOwes
   520→              ? `Customer owes Rs. ${Math.abs(exchange.priceDifference).toLocaleString('en-PK')}.`
   521→              : `Rs. ${Math.abs(exchange.priceDifference).toLocaleString('en-PK')} due as refund.`}
   522→          </DialogDescription>
   523→        </DialogHeader>
   524→        <div className="space-y-3 py-2">
   525→          <div className="space-y-1.5">
   526→            <Label className="text-xs">Amount (Rs.)</Label>
   527→            <Input type="number" min="0" value={amount} onChange={(e) => setAmount(e.target.value)} autoFocus />
   528→          </div>
   529→          <div className="space-y-1.5">
   530→            <Label className="text-xs">Type</Label>
   531→            <Select value={type} onValueChange={setType}>
   532→              <SelectTrigger><SelectValue /></SelectTrigger>
   533→              <SelectContent>
   534→                <SelectItem value="collected_from_customer">Collected from customer</SelectItem>
   535→                <SelectItem value="refunded_to_customer">Refunded to customer</SelectItem>
   536→              </SelectContent>
   537→            </Select>
   538→          </div>
   539→        </div>
   540→        <DialogFooter>
   541→          <Button variant="outline" onClick={() => onOpenChange(false)}>Cancel</Button>
   542→          <Button disabled={loading} onClick={() => onConfirm(Number(amount) || 0, type)}>
   543→            {loading ? <><Loader2 className="h-4 w-4 animate-spin" /> Settling…</> : 'Settle Payment'}
   544→          </Button>
   545→        </DialogFooter>
   546→      </DialogContent>
   547→    </Dialog>
   548→  )
   549→}

   555→function NotReturnedDialog({
   556→  exchange,
   557→  open,
   558→  onOpenChange,
   559→  loading,
   560→  onConfirm,
   561→}: {
   562→  exchange: ExchangeDetail
   563→  open: boolean
   564→  onOpenChange: (open: boolean) => void
   565→  loading: boolean
   566→  onConfirm: (reason: string, recoveryStatus: string, recoveryAmount: number) => void
   567→}) {

   569→  const [recoveryStatus, setRecoveryStatus] = useState('pending')
   570→  const [recoveryAmount, setRecoveryAmount] = useState(exchange.oldItemPrice.toString())
   571→

   584→          <div className="space-y-1.5">
