    50→-- ============================================================================
    51→-- ADAPTATIONS FROM SPEC (necessary to integrate with the live schema)
    52→-- ============================================================================
    53→-- The spec was written in idealized PostgreSQL terms (lowercase table names,
    54→-- UUID primary keys, snake_case columns, `organizations(id)` references).
    55→-- The LIVE FlowOps schema uses the opposite conventions because it was built
    56→-- with Prisma on an existing multi-tenant SaaS. To produce SQL that actually
    57→-- runs on this live instance WITHOUT breaking the 92 existing orders, the
    58→-- following adaptations are made (each is the correct production call):
    59→--
    60→--   1. Table name "Customer" (PascalCase) instead of `customers`.
    61→--      The table already exists with 5 rows + 92 FK references from "Order".
    62→--      Creating a parallel `customers` table would orphan every order. We
    63→--      ALTER the existing table in place. The three NEW child tables use
    64→--      lowercase names (customer_phones, customer_addresses,
    65→--      customer_external_identities) exactly as the spec names them.
    66→--
    67→--   2. All IDs are TEXT (cuid) instead of UUID.
    68→--      Every existing PK/FK in the schema (Organization, Employee, Customer,
    69→--      Order, OrderItem, ...) is TEXT. A UUID customer_phones.customerId FK
    70→--      could not reference the TEXT "Customer".id. New child tables use
    71→--      TEXT PKs with `DEFAULT gen_random_uuid()::text` so raw-SQL inserts
    72→--      get a DB-generated ID while Prisma client inserts override with a
    73→--      cuid (same pattern the rest of the app uses).
    74→--
    75→--   3. All columns are camelCase (Prisma convention) instead of snake_case,
    76→--      AND are double-quoted everywhere so PostgreSQL preserves the casing
    77→--      (unquoted identifiers are folded to lowercase). camelCase columns on
    78→--      the new tables: "phoneRaw", "phoneNormalized", "isPrimary",
    79→--      "isDefault", "lastUsedAt", "externalCustomerId", "matchedVia",
    80→--      "organizationId", "customerId", "createdAt", "updatedAt". This
    81→--      matches the existing "Customer"/"Order" tables which Prisma created
    82→--      with quoted camelCase columns.
    83→--
    84→--   4. FK references use PascalCase: REFERENCES "Organization"(id),
    85→--      "Employee"(id), "Customer"(id).
    86→--
    87→--   5. RLS uses ENABLE (not FORCE ROW LEVEL SECURITY). The application
    88→--      connects via the `postgres` role, which BYPASSES RLS by default.
    89→--      This means:
    90→--        - The Prisma app keeps working exactly as before (no GUC-setting
    91→--          middleware required for the app to function).
    92→--        - Direct Supabase API access (anon/authenticated roles) IS subject
    93→--          to these RLS policies -> defense-in-depth for any future direct
    94→--          client access, exactly as the spec intends.
    95→--      When Step 2 wires GUC-setting middleware (app.active_org_id etc.),
    96→--      the policies become actively enforced for non-bypass roles.
    97→--
    98→--   6. normalize_phone() and match_or_create_customer() accept/return TEXT
    99→--      (not UUID) for the same ID-compatibility reason.
   100→--
   101→-- All other semantics — table structure, constraints, indexes, the layered
   102→-- matching strategy, RLS policy logic — follow the spec exactly.
   103→-- ============================================================================

   107→-- ============================================================================
   108→-- PART 0: normalize_phone() FUNCTION
   109→-- ============================================================================
   110→-- Strips all non-digit characters, then canonicalizes Pakistani phone numbers
   111→-- to E.164-style "+92XXXXXXXXXX". Used both at storage time (so every
   112→-- customer_phones."phoneNormalized" is canonical) and at lookup time (so
   113→-- matching is format-independent).
   114→--
   115→-- Examples:
   116→--   normalize_phone('0300-1234567')    -> '+923001234567'
   117→--   normalize_phone('+92 300 1234567') -> '+923001234567'
   118→--   normalize_phone('923001234567')    -> '+923001234567'
   119→--   normalize_phone('3001234567')      -> '+923001234567'  (10-digit local)
   120→--   normalize_phone(NULL)              -> NULL
   121→--   normalize_phone('')                -> NULL
   122→--
   123→-- Marked IMMUTABLE so it can be used in indexes and CHECK constraints.
   124→
   125→CREATE OR REPLACE FUNCTION normalize_phone(p_raw_phone TEXT)
   126→RETURNS TEXT
   127→LANGUAGE plpgsql
   128→IMMUTABLE
   129→AS $$
   130→DECLARE
   131→  v_digits TEXT;
   132→  v_len    INT;
   133→BEGIN
   134→  IF p_raw_phone IS NULL OR BTRIM(p_raw_phone) = '' THEN
   135→    RETURN NULL;
   136→  END IF;
   137→
   138→  -- Keep digits only
   139→  v_digits := REGEXP_REPLACE(p_raw_phone, '[^0-9]', '', 'g');
   140→  v_len    := LENGTH(v_digits);
   141→
   142→  IF v_len = 0 THEN
   143→    RETURN NULL;
   144→  END IF;
   145→
   146→  -- 12 digits starting with '92' -> already has country code, just prefix '+'
   147→  IF v_len = 12 AND v_digits LIKE '92%' THEN
   148→    RETURN '+' || v_digits;
   149→  END IF;
   150→
   151→  -- 11 digits starting with '0' -> local Pakistani format (03XXXXXXXXX)
   152→  IF v_len = 11 AND v_digits LIKE '0%' THEN
   153→    RETURN '+92' || SUBSTRING(v_digits FROM 2);
   154→  END IF;
   155→
   156→  -- 10 digits without leading 0 (e.g. 3001234567) -> treat as local, prefix +92
   157→  IF v_len = 10 THEN
   158→    RETURN '+92' || v_digits;
   159→  END IF;
   160→
   161→  -- Fallback for unusual lengths (10-13 digits): strip a leading 0 if present
   162→  -- and prefix +92. Best-effort consistency for storage/matching.
   163→  IF v_len BETWEEN 10 AND 13 THEN
   164→    IF v_digits LIKE '0%' THEN
   165→      v_digits := SUBSTRING(v_digits FROM 2);
   166→    END IF;
   167→    RETURN '+92' || v_digits;
   168→  END IF;
   169→
   170→  -- Anything outside 10-13 digits: return +digits as-is (rare, best-effort)
   171→  RETURN '+' || v_digits;
   172→END;
   173→$$;

   175→-- ============================================================================
   176→-- PART 1: customer_phones TABLE
   177→-- ============================================================================
   178→-- A customer can have multiple phone numbers (own, spouse's, family shared
   179→-- line). Each row stores the raw phone (for display, since Pakistani staff
   180→-- are used to seeing "0300-1234567") AND a normalized E.164 form (for
   181→-- reliable matching/deduplication).
   182→--
   183→-- UNIQUE("organizationId", "phoneNormalized") is what prevents duplicate
   184→-- customer creation when the same number is reused — within one organization,
   185→-- one normalized phone maps to exactly one customer.
   186→--
   187→-- ALL camelCase columns are double-quoted to preserve casing (PostgreSQL
   188→-- folds unquoted identifiers to lowercase).
   189→
   190→CREATE TABLE IF NOT EXISTS customer_phones (
   191→  id                 TEXT PRIMARY KEY DEFAULT gen_random_uuid()::text,
   192→  "customerId"       TEXT NOT NULL REFERENCES "Customer"(id) ON DELETE CASCADE,
   193→  "organizationId"   TEXT NOT NULL REFERENCES "Organization"(id),
   194→
   195→  "phoneRaw"         TEXT NOT NULL,
   196→  -- as entered/received, e.g. "0300-1234567" — for display
   197→
   198→  "phoneNormalized"  TEXT NOT NULL,
   199→  -- canonical E.164 form, e.g. "+923001234567" — for all matching/search/dedup
   200→
   201→  label              TEXT,
   202→  -- optional: "Personal", "Husband's number", "Work" (lowercase ok — not camelCase)
   203→
   204→  "isPrimary"        BOOLEAN NOT NULL DEFAULT FALSE,

   206→  "createdAt"        TIMESTAMPTZ NOT NULL DEFAULT NOW(),

   258→-- ============================================================================
   259→-- PART 3: customer_external_identities TABLE
   260→-- ============================================================================
   261→-- Maps platform-specific customer IDs (Shopify, Daraz, Instagram) to internal
   262→-- FlowOps customer IDs, per organization. This is the FAST path of the layered
   263→-- matching strategy: an exact previous match is recognized instantly.
   264→--
   265→-- "matchedVia" records HOW each mapping was established (for audit/debugging):
   266→--   exact_identity — first time this external ID was seen (Layer 4 of matching)
   267→--   phone_match    — matched an existing customer by normalized phone (Layer 2)
   268→--   email_match    — matched an existing customer by email (Layer 3)
   269→--   manual         — an operator explicitly linked the accounts
   270→
   271→CREATE TABLE IF NOT EXISTS customer_external_identities (
   272→  id                    TEXT PRIMARY KEY DEFAULT gen_random_uuid()::text,
   273→  "customerId"          TEXT NOT NULL REFERENCES "Customer"(id) ON DELETE CASCADE,
   274→  "organizationId"      TEXT NOT NULL REFERENCES "Organization"(id),
   275→
   276→  platform              TEXT NOT NULL
   277→                          CHECK (platform IN ('shopify','daraz','instagram')),
   278→
   279→  "externalCustomerId"  TEXT NOT NULL,
   280→  -- the platform's own customer identifier
   281→
   282→  "matchedVia"          TEXT NOT NULL DEFAULT 'manual'
   283→                          CHECK ("matchedVia" IN
   284→                            ('exact_identity','phone_match','email_match','manual')),
   285→
   286→  "createdAt"           TIMESTAMPTZ NOT NULL DEFAULT NOW(),
   287→
   288→  CONSTRAINT customer_ext_identities_unique
   289→    UNIQUE ("organizationId", platform, "externalCustomerId")
   290→);
   291→
   292→CREATE INDEX IF NOT EXISTS customer_external_identities_customer_idx
   293→  ON customer_external_identities ("customerId");
   294→
   295→CREATE INDEX IF NOT EXISTS customer_external_identities_org_platform_idx
   296→  ON customer_external_identities ("organizationId", platform, "externalCustomerId");

   299→-- PART 4: ADD NEW COLUMNS TO "Customer" + DB-SIDE ID DEFAULT
   300→-- ============================================================================
   301→-- flaggedAt / flaggedBy: track WHEN and WHO flagged a customer for RTO fraud.
   302→-- createdBy: audit trail for manual customer creation.
   303→-- (The spec's cached stats — totalOrdersCount, totalOrderValue, totalRtoCount,
   304→--  isFlagged, flaggedReason — already exist on the live table and are kept.)
   305→--
   306→-- ID DEFAULT: Prisma's @default(cuid()) generates IDs CLIENT-SIDE, so the
   307→-- "Customer".id column has NO DB-side default. The match_or_create_customer()
   308→-- SQL function (Part 10) inserts into "Customer" directly (no Prisma), so it
   309→-- needs a DB-side default to generate an id. We set DEFAULT gen_random_uuid()::text
   310→-- — Prisma will continue to override this with cuids on its own inserts, so
   311→-- existing behavior is unchanged; only raw-SQL inserts (from the function)
   312→-- use the DB default. The resulting id is a TEXT value either way (cuid or
   313→-- uuid-as-text), which is what every FK in the schema expects.
   314→
   315→ALTER TABLE "Customer"
   316→  ADD COLUMN IF NOT EXISTS "flaggedAt" TIMESTAMPTZ;
   317→ALTER TABLE "Customer"
   318→  ADD COLUMN IF NOT EXISTS "flaggedBy" TEXT REFERENCES "Employee"(id) ON DELETE SET NULL;
   319→ALTER TABLE "Customer"
   320→  ADD COLUMN IF NOT EXISTS "createdBy" TEXT REFERENCES "Employee"(id) ON DELETE SET NULL;
   321→
   322→ALTER TABLE "Customer"
   323→  ALTER COLUMN id SET DEFAULT gen_random_uuid()::text;
   324→
   325→-- "updatedAt" is NOT NULL but Prisma's @updatedAt only sets it CLIENT-SIDE,
   326→-- so the DB column has no default. Raw SQL inserts (from the function) need
   327→-- a DB-side default so the NOT NULL constraint is satisfied on INSERT. Prisma
   328→-- will continue to set updatedAt explicitly on its own writes, so this is safe.
   329→ALTER TABLE "Customer"
   330→  ALTER COLUMN "updatedAt" SET DEFAULT NOW();
   331→
   332→-- Index for the "flagged customers" list view (per organization).
   333→CREATE INDEX IF NOT EXISTS "Customer_organizationId_isFlagged_idx"
   334→  ON "Customer" ("organizationId", "isFlagged");

   336→-- ============================================================================
   337→-- PART 5: BACKFILL customer_phones FROM LEGACY Customer.phone (IF PRESENT)
   338→-- ============================================================================
   339→-- Migrate every customer's primary phone + alternate phone into customer_phones
   340→-- BEFORE the legacy columns are dropped. The primary phone gets
   341→-- "isPrimary"=TRUE (the partial unique index guarantees one primary per
   342→-- customer). Alternate phones get "isPrimary"=FALSE.
   343→--
   344→-- Wrapped in DO blocks that first check information_schema.columns so the
   345→-- migration is SAFE TO RE-RUN from any state — if the legacy columns were
   346→-- already dropped by a prior partial run, the backfill is a no-op rather than
   347→-- a hard error.
   348→--
   349→-- ON CONFLICT ("organizationId", "phoneNormalized") DO NOTHING handles the
   350→-- rare case where a customer's alternatePhone collides with another
   351→-- customer's already-migrated primary phone — the primary wins, the duplicate
   352→-- alternate is silently skipped (it is, by definition, the same contact).
   353→
   354→DO $$
   355→BEGIN
   356→  IF EXISTS (
   357→    SELECT 1 FROM information_schema.columns
   358→    WHERE table_name = 'Customer' AND column_name = 'phone'
   359→  ) THEN
   360→    INSERT INTO customer_phones
   361→      ("customerId", "organizationId", "phoneRaw", "phoneNormalized", "isPrimary", label, "createdAt")
   362→    SELECT
   363→      c.id,
   364→      c."organizationId",
   365→      c.phone,
   366→      normalize_phone(c.phone),
   367→      TRUE,
   368→      'Primary',
   369→      c."createdAt"
   370→    FROM "Customer" c
   371→    WHERE c.phone IS NOT NULL AND BTRIM(c.phone) <> ''
   372→    ON CONFLICT ("organizationId", "phoneNormalized") DO NOTHING;
   373→  END IF;
   374→END $$;
   375→
   376→DO $$
   377→BEGIN
   378→  IF EXISTS (
   379→    SELECT 1 FROM information_schema.columns
   380→    WHERE table_name = 'Customer' AND column_name = 'alternatePhone'
   381→  )
   382→  AND EXISTS (
   383→    SELECT 1 FROM information_schema.columns
   384→    WHERE table_name = 'Customer' AND column_name = 'phone'
   385→  ) THEN
   386→    INSERT INTO customer_phones
   387→      ("customerId", "organizationId", "phoneRaw", "phoneNormalized", "isPrimary", label, "createdAt")
   388→    SELECT
   389→      c.id,
   390→      c."organizationId",
   391→      c."alternatePhone",
   392→      normalize_phone(c."alternatePhone"),
   393→      FALSE,
   394→      'Alternate',
   395→      c."createdAt"
   396→    FROM "Customer" c
   397→    WHERE c."alternatePhone" IS NOT NULL
   398→      AND BTRIM(c."alternatePhone") <> ''
   399→      AND normalize_phone(c."alternatePhone") IS DISTINCT FROM normalize_phone(c.phone)
   400→    ON CONFLICT ("organizationId", "phoneNormalized") DO NOTHING;
   401→  END IF;
   402→END $$;
   403→
   404→-- ============================================================================
   405→-- PART 6: BACKFILL customer_addresses FROM LEGACY JSON (IF PRESENT)
   406→-- ============================================================================
   407→-- Parse the flat JSONB { "address": "...", "city": "..." } that was stored in
   408→-- Customer.shippingAddress and Customer.billingAddress, and migrate each into
   409→-- a customer_addresses row.
   410→--
   411→--  - Shipping address -> "isDefault" = TRUE (it is the primary delivery address)
   412→--  - Billing address  -> "isDefault" = FALSE, and ONLY if it differs from the
   413→--    shipping address (no point storing an exact duplicate)
   414→--
   415→-- "lastUsedAt" is set to the customer's most recent order's createdAt, so the
   416→-- UI can sort addresses by recency immediately after migration.
   417→--
   418→-- Wrapped in DO blocks (same re-run-safety rationale as Part 5).
   419→
   420→DO $$
   421→BEGIN
   422→  IF EXISTS (
   423→    SELECT 1 FROM information_schema.columns
   424→    WHERE table_name = 'Customer' AND column_name = 'shippingAddress'
   425→  ) THEN
   426→    INSERT INTO customer_addresses
   427→      ("customerId", "organizationId", label, address, city, "isDefault", "lastUsedAt", "createdAt", "updatedAt")
   428→    SELECT
   429→      c.id,
   430→      c."organizationId",
   431→      'Shipping',
   432→      BTRIM(c."shippingAddress"::jsonb ->> 'address'),
   433→      BTRIM(c."shippingAddress"::jsonb ->> 'city'),
   434→      TRUE,
   435→      (SELECT MAX(o."createdAt") FROM "Order" o WHERE o."customerId" = c.id),
   436→      c."createdAt",
   437→      NOW()
   438→    FROM "Customer" c
   439→    WHERE c."shippingAddress" IS NOT NULL
   440→      AND c."shippingAddress"::text NOT IN ('', '{}', 'null')
   441→      AND c."shippingAddress"::jsonb ->> 'address' IS NOT NULL
   442→      AND BTRIM(c."shippingAddress"::jsonb ->> 'address') <> ''
   443→    ON CONFLICT DO NOTHING;
   444→  END IF;
   445→END $$;
   446→
   447→DO $$
   448→BEGIN
   449→  IF EXISTS (
   450→    SELECT 1 FROM information_schema.columns
   451→    WHERE table_name = 'Customer' AND column_name = 'billingAddress'
   452→  ) THEN
   453→    INSERT INTO customer_addresses
   454→      ("customerId", "organizationId", label, address, city, "isDefault", "lastUsedAt", "createdAt", "updatedAt")
   455→    SELECT
   456→      c.id,
   457→      c."organizationId",
   458→      'Billing',
   459→      BTRIM(c."billingAddress"::jsonb ->> 'address'),
   460→      BTRIM(c."billingAddress"::jsonb ->> 'city'),
   461→      FALSE,
   462→      (SELECT MAX(o."createdAt") FROM "Order" o WHERE o."customerId" = c.id),
   463→      c."createdAt",
   464→      NOW()
   465→    FROM "Customer" c
   466→    WHERE c."billingAddress" IS NOT NULL
   467→      AND c."billingAddress"::text NOT IN ('', '{}', 'null')
   468→      AND c."billingAddress"::jsonb ->> 'address' IS NOT NULL
   469→      AND BTRIM(c."billingAddress"::jsonb ->> 'address') <> ''
   470→      AND (
   471→        c."billingAddress"::text IS DISTINCT FROM c."shippingAddress"::text
   472→        OR c."shippingAddress" IS NULL
   473→        OR c."shippingAddress"::text IN ('', '{}', 'null')
   474→        OR c."shippingAddress"::jsonb ->> 'address' IS NULL
   475→        OR BTRIM(c."shippingAddress"::jsonb ->> 'address') = ''
   476→      )
   477→    ON CONFLICT DO NOTHING;
   478→  END IF;
   479→END $$;

   508→-- PART 8: BACKFILL Order.recipientName FROM Customer.name
