    91→// ──────────────────────────────────────────────────────────────
    92→// Processes ALL order_items for an order and attempts to reserve
    93→// stock for each based on its fulfillment_type_snapshot.
    94→//
    95→// For stock_based items:
    96→//   - If sufficient available stock → reserveStockForOrder(), fulfillment_status='reserved'
    97→//   - If insufficient + inventory_policy='continue' → fulfillment_status='backordered'
    98→//   - If insufficient + inventory_policy='deny' → outcome='failed'
    99→//
   100→// For made_to_order items:
   101→//   - checkAndFulfillMadeToOrderVariant() checks returned stock first
   102→//   - If existing_stock → fulfillment_status='reserved', returned_stitched_used=true
   103→//   - If fresh_production → fulfillment_status='reserved', production_order_id set
   104→//   - If error → outcome='failed'
   105→//
   106→// After processing: if ANY item is 'backordered', order status → 'partially_backordered'.
   107→// If ALL items are 'reserved', order stays 'confirmed'.
   108→//
   109→// Called from:
   110→//   - createManualOrder() IF the order auto-confirmed at creation
   111→//   - confirmOrder() when a 'pending' order is manually confirmed
   112→//   - dispatchOrderAction() if status is still 'pending' at dispatch time
   113→
   114→async function reserveOrderStock(
   115→  orderId: string,
   116→  ctx: { company: { id: string; organizationId: string }; user: { id: string }; employee: { id: string } },
   117→): Promise<{
   118→  success: boolean
   119→  results: Array<{ orderItemId: string; outcome: 'reserved' | 'backordered' | 'failed'; reason?: string }>
   120→}> {
   121→  const order = await db.order.findUnique({
   122→    where: { id: orderId },
   123→    select: { dispatchLocationId: true, companyId: true, organizationId: true },
   124→  })
   125→  if (!order) return { success: false, results: [] }
   126→
   127→  const items = await db.orderItem.findMany({
   128→    where: { orderId },
   129→    include: {
   130→      orgVariant: {
   131→        select: { id: true, fulfillmentType: true, inventoryPolicy: true, stitchingCharges: true },
   132→      },
   133→    },
   134→  })
   135→
   136→  const results: Array<{ orderItemId: string; outcome: 'reserved' | 'backordered' | 'failed'; reason?: string }> = []
   137→  let hasBackordered = false
   138→
   139→  for (const item of items) {
   140→    if (item.fulfillmentStatus === 'reserved' || item.fulfillmentStatus === 'dispatched') {
   141→      // Already processed (e.g. re-confirmation attempt) — skip
   142→      results.push({ orderItemId: item.id, outcome: 'reserved' })
   143→      continue
   144→    }
   145→
   146→    const locationId = item.reservedLocationId ?? order.dispatchLocationId
   147→    if (!locationId) {
   148→      results.push({ orderItemId: item.id, outcome: 'failed', reason: 'No dispatch location set' })
   149→      continue
   150→    }
   151→
   152→    if (item.fulfillmentTypeSnapshot === 'stock_based') {
   153→      // Check available stock at the dispatch location
   154→      const pool = await db.inventoryPool.findUnique({
   155→        where: {
   156→          orgVariantId_locationId: {
   157→            orgVariantId: item.orgVariantId,
   158→            locationId,
   159→          },
   160→        },
   161→      })
   162→
   163→      const available = pool ? pool.onHand - pool.reserved : 0
   164→
   165→      if (available >= item.quantity) {
   166→        // Sufficient stock — reserve it
   167→        const reserveResult = await reserveStockForOrder({
   168→          orgVariantId: item.orgVariantId,
   169→          locationId,
   170→          organizationId: order.organizationId,
   171→          companyId: order.companyId,
   172→          employeeId: ctx.employee.id,
   173→          quantity: item.quantity,
   174→          orderId,
   175→        })
   176→
   177→        if (reserveResult.success) {
   178→          await db.orderItem.update({
   179→            where: { id: item.id },
   180→            data: { fulfillmentStatus: 'reserved', reservedLocationId: locationId },
   181→          })
   182→          results.push({ orderItemId: item.id, outcome: 'reserved' })
   183→        } else {
   184→          results.push({ orderItemId: item.id, outcome: 'failed', reason: reserveResult.error })
   185→        }
   186→      } else if (item.orgVariant.inventoryPolicy === 'continue') {
   187→        // Insufficient stock but backordering allowed
   188→        await db.orderItem.update({
   189→          where: { id: item.id },
   190→          data: { fulfillmentStatus: 'backordered', backorderedAt: new Date() },
   191→        })
   192→        await insertMetricEvent({
   193→          companyId: order.companyId,
   194→          entityType: 'product',
   195→          entityId: item.orgVariantId,
   196→          metricKey: 'order.backordered',
   197→          numericValue: item.quantity,
   198→          dimensions: { order_id: orderId },
   199→        }).catch(() => {})
   200→        hasBackordered = true
   201→        results.push({
   202→          orderItemId: item.id,
   203→          outcome: 'backordered',
   204→          reason: `Available: ${available}, required: ${item.quantity}`,
   205→        })
   206→      } else {
   207→        // Insufficient stock + policy='deny' — fail
   208→        results.push({
   209→          orderItemId: item.id,
   210→          outcome: 'failed',
   211→          reason: `Insufficient stock (available: ${available}, required: ${item.quantity}) and inventory_policy='deny'`,
   212→        })
   213→      }
   214→    } else if (item.fulfillmentTypeSnapshot === 'made_to_order') {
   215→      // Made-to-order: check returned stock first, then trigger production
   216→      const mtoResult = await checkAndFulfillMadeToOrderVariant(
   217→        item.orgVariantId,
   218→        item.quantity,
   219→        order.companyId,
   220→        locationId,
   221→      )
   222→
   223→      if (mtoResult.source === 'existing_stock' && mtoResult.locationId) {
   224→        // Returned stock available — reserve it
   225→        const reserveResult = await reserveStockForOrder({
   226→          orgVariantId: item.orgVariantId,
   227→          locationId: mtoResult.locationId,
   228→          organizationId: order.organizationId,
   229→          companyId: order.companyId,
   230→          employeeId: ctx.employee.id,
   231→          quantity: item.quantity,
   232→          orderId,
   233→        })
   234→
   235→        if (reserveResult.success) {
   236→          await db.orderItem.update({
   237→            where: { id: item.id },
   238→            data: {
   239→              fulfillmentStatus: 'reserved',
   240→              returnedStitchedUsed: true,
   241→              reservedLocationId: mtoResult.locationId,
   242→            },
   243→          })
   244→          await insertMetricEvent({
   245→            companyId: order.companyId,
   246→            entityType: 'product',
   247→            entityId: item.orgVariantId,
   248→            metricKey: 'order.made_to_order_from_returned_stock',
   249→            numericValue: 1,
   250→            dimensions: { order_id: orderId, stitching_cost_saved: Number(item.orgVariant.stitchingCharges) || 0 },
   251→          }).catch(() => {})
   252→          results.push({ orderItemId: item.id, outcome: 'reserved' })
   253→        } else {
   254→          results.push({ orderItemId: item.id, outcome: 'failed', reason: reserveResult.error })
   255→        }
   256→      } else if (mtoResult.source === 'fresh_production' && mtoResult.productionOrderId) {
   257→        // Fresh production triggered — link the production order
   258→        await db.orderItem.update({
   259→          where: { id: item.id },
   260→          data: {
   261→            fulfillmentStatus: 'reserved',
   262→            productionOrderId: mtoResult.productionOrderId,
   263→          },
   264→        })
   265→        // Also link the order_item back to the production order
   266→        await db.productionOrder.update({
   267→          where: { id: mtoResult.productionOrderId },
   268→          data: { orderItemId: item.id },
   269→        })
   270→        await insertMetricEvent({
   271→          companyId: order.companyId,
   272→          entityType: 'product',
   273→          entityId: item.orgVariantId,
   274→          metricKey: 'order.made_to_order_production_triggered',
   275→          numericValue: 1,
   276→          dimensions: { order_id: orderId, production_order_id: mtoResult.productionOrderId },
   277→        }).catch(() => {})
   278→        results.push({ orderItemId: item.id, outcome: 'reserved' })
   279→      } else {
   280→        // Error (no fabric source, insufficient fabric, etc.)
   281→        results.push({
   282→          orderItemId: item.id,
   283→          outcome: 'failed',
   284→          reason: mtoResult.error ?? 'Made-to-order fulfillment failed',
   285→        })
   286→      }
   287→    }
   288→  }
   289→
   290→  // Update order status based on results
   291→  if (hasBackordered) {
   292→    await db.order.update({
   293→      where: { id: orderId },
   294→      data: { status: 'partially_backordered' },
   295→    })
   296→  }
   297→  // If no backordered items, order stays 'confirmed' (or whatever it was)
   298→
   299→  return { success: true, results }
   300→}

   302→// ──────────────────────────────────────────────────────────────

   305→
   306→export async function createManualOrder(
   307→  input: CreateManualOrderInput,
   308→): Promise<ActionResult<{
   309→  orderId: string
   310→  flowopsOrderNumber: string
   311→  orderItems: Array<{ id: string; orgVariantId: string; quantity: number }>
   312→}>> {
   313→  try {
   314→    const ctx = await getWorkspace()
   315→    await requirePermission(ctx, PERMISSIONS.ORDERS_CREATE)
   316→
   317→    // 1. Validate
   318→    const parsed = createManualOrderSchema.safeParse(input)
   319→    if (!parsed.success) {
   320→      return {
   321→        success: false,
   322→        error: parsed.error.issues[0]?.message ?? 'Invalid order data',
   323→      }
   324→    }
   325→    const d = parsed.data
   326→
   327→    // 2. Resolve customer (existing customer_id OR inline new_customer).
   328→    //    Also resolve optional saved address/phone selection and recipient_name.
   329→    let customerId: string
   330→    let customerName: string
   331→    let usedCustomerAddressId: string | null = null
   332→    let usedCustomerPhoneId: string | null = null
   333→    let saveAddressForNextTime = d.save_address_for_next_time ?? false
   334→    // Whether the delivery_address came from a saved customer_addresses row
   335→    // (vs. typed as a one-off). Used to decide whether to bump lastUsedAt.
   336→    let selectedSavedAddressId: string | null = null
   337→
   338→    if (d.customer_id) {
   339→      // Existing customer path — verify they belong to this org.
   340→      const existing = await db.customer.findFirst({
   341→        where: { id: d.customer_id, organizationId: ctx.company.organizationId },
   342→        select: { id: true, name: true },
   343→      })
   344→      if (!existing) return { success: false, error: 'Customer not found' }
   345→      customerId = existing.id
   346→      customerName = existing.name
   347→
   348→      // If a saved address was selected, verify it belongs to this customer.
   349→      if (d.used_customer_address_id) {
   350→        const addr = await db.customerAddress.findFirst({
   351→          where: {
   352→            id: d.used_customer_address_id,
   353→            customerId: existing.id,
   354→            organizationId: ctx.company.organizationId,
   355→          },
   356→          select: { id: true, address: true, city: true },
   357→        })
   358→        if (addr) {
   359→          usedCustomerAddressId = addr.id
   360→          selectedSavedAddressId = addr.id
   361→          // The delivery_address snapshot is the order's own editable copy.
   362→          // If the caller passed the same address text, we trust it; if they
   363→          // passed different text (edited it), we still record the link to
   364→          // the saved address for lastUsedAt tracking but keep the edited text
   365→          // as the order's snapshot. Either way, delivery_address comes from
   366→          // the input — we do NOT override it here.
   367→        }
   368→      }
   369→
   370→      // If a saved phone was selected, verify it belongs to this customer.
   371→      if (d.used_customer_phone_id) {
   372→        const phone = await db.customerPhone.findFirst({
   373→          where: {
   374→            id: d.used_customer_phone_id,
   375→            customerId: existing.id,
   376→            organizationId: ctx.company.organizationId,
   377→          },
   378→          select: { id: true },
   379→        })
   380→        if (phone) {
   381→          usedCustomerPhoneId = phone.id
   382→        }
   383→      }
   384→    } else if (d.new_customer) {
   385→      // Inline new-customer path — create the customer + their phones +
   386→      // addresses now via the Customer Management System. The first phone is
   387→      // primary and the first address is default (validated by createCustomerSchema).
   388→      const createResult = await createCustomer(d.new_customer)
   389→      if (!createResult.success || !createResult.data) {
   390→        return { success: false, error: createResult.error ?? 'Failed to create customer' }
   391→      }
   392→      customerId = createResult.data.customerId
   393→      customerName = d.new_customer.name.trim()
   394→      // For a brand-new customer, the first address they provided IS the
   395→      // saved default address. Mark it as the selected address so we bump
   396→      // its lastUsedAt after order creation.
   397→      const newAddr = await db.customerAddress.findFirst({
   398→        where: { customerId, isDefault: true },
   399→        select: { id: true },
   400→      })
   401→      if (newAddr) {
   402→        usedCustomerAddressId = newAddr.id
   403→        selectedSavedAddressId = newAddr.id
   404→      }
   405→      // Same for the primary phone.
   406→      const newPhone = await db.customerPhone.findFirst({
   407→        where: { customerId, isPrimary: true },
   408→        select: { id: true },
   409→      })
   410→      if (newPhone) {
   411→        usedCustomerPhoneId = newPhone.id
   412→      }
   413→      // For a brand-new customer, we already saved the address as part of
   414→      // createCustomer — don't double-save it.
   415→      saveAddressForNextTime = false
   416→    } else {
   417→      return { success: false, error: 'Either customer_id or new_customer is required' }
   418→    }
   419→
   420→    // 3. recipient_name defaults to the customer's name if not explicitly overridden.
   421→    const recipientName = (d.recipient_name && d.recipient_name.trim()) || customerName
   422→
   423→    // 3. Fetch variants + their pricing for this company + fulfillment_type snapshot
   424→    const variantIds = d.items.map((i) => i.org_variant_id)
   425→    const variants = await db.orgProductVariant.findMany({
   426→      where: { id: { in: variantIds }, organizationId: ctx.company.organizationId },
   427→      include: {
   428→        companyPricing: { where: { companyId: ctx.company.id } },
   429→      },
   430→    })
   431→
   432→    if (variants.length !== variantIds.length) {
   433→      return { success: false, error: 'One or more variants not found in this organization' }
   434→    }
   435→
   436→    // 4. Compute subtotal + build order items data
   437→    let subtotal = 0
   438→    const orderItemsData: Array<{
   439→      orgVariantId: string
   440→      quantity: number
   441→      unitPrice: number
   442→      lineTotal: number
   443→      fulfillmentTypeSnapshot: string
   444→    }> = []
   445→
   446→    for (const item of d.items) {
   447→      const variant = variants.find((v) => v.id === item.org_variant_id)
   448→      if (!variant) continue
   449→
   450→      // Use provided unit_price OR fall back to company pricing OR variant cost
   451→      const unitPrice =
   452→        item.unit_price ??
   453→        (variant.companyPricing[0]?.salePrice
   454→          ? Number(variant.companyPricing[0].salePrice)
   455→          : Number(variant.costPrice))
   456→
   457→      const lineTotal = item.quantity * unitPrice
   458→      subtotal += lineTotal
   459→
   460→      orderItemsData.push({
   461→        orgVariantId: item.org_variant_id,
   462→        quantity: item.quantity,
   463→        unitPrice,
   464→        lineTotal,
   465→        fulfillmentTypeSnapshot: variant.fulfillmentType,
   466→      })
   467→    }
   468→
   469→    // 5. Compute totals
   470→    const discountAmount = d.discount_amount ?? 0
   471→    const courierCharges = d.courier_charges ?? 0
   472→    const estimatedDeliveryCharge = d.estimated_delivery_charge ?? 0
   473→    const taxAmount = d.tax_amount ?? 0
   474→    // Total = subtotal + courier charges + delivery charge + tax - discount
   475→    // (delivery charge and tax are ADDITIVE — not absorbed into subtotal)
   476→    const totalOrderValue = subtotal + courierCharges + estimatedDeliveryCharge + taxAmount - discountAmount
   477→
   478→    // 6. Determine payment status + source
   479→    let paymentStatus: string = 'cod_pending'
   480→    let paymentSource: string = 'cod_native'
   481→    let advanceAmount: number | null = null
   482→    let advancePaymentMethod: string | null = null
   483→    let advancePaymentReference: string | null = null
   484→    let advancePaidAt: Date | null = null
   485→
   486→    if (d.payment_type === 'partial_advance') {
   487→      paymentStatus = 'advance_paid'
   488→      paymentSource = 'manual_conversion'
   489→      advanceAmount = d.advance_amount ?? 0
   490→      advancePaymentMethod = d.advance_payment_method || null
   491→      advancePaymentReference = d.advance_payment_reference || null
   492→      advancePaidAt = new Date()
   493→    } else if (d.payment_type === 'fully_prepaid') {
   494→      paymentStatus = 'fully_prepaid'
   495→      paymentSource = 'manual_conversion'
   496→      advanceAmount = totalOrderValue
   497→      advancePaymentMethod = d.advance_payment_method || null
   498→      advancePaymentReference = d.advance_payment_reference || null
   499→      advancePaidAt = new Date()
   500→    }
   501→
   502→    // remainingCodAmount is NOT a GENERATED column in the DB — the application
   503→    // MUST compute it as totalOrderValue - (advanceAmount ?? 0).
   504→    const remainingCodAmount = totalOrderValue - (advanceAmount ?? 0)
   505→
   506→    // 7. Determine initial order status
   507→    let orderStatus: string = 'pending'
   508→    let confirmedAt: Date | null = null
   509→    let skippedConfirmation = false
   510→
   511→    if (paymentStatus === 'advance_paid' || paymentStatus === 'fully_prepaid') {
   512→      // Payment itself is the confirmation signal — bypasses require_order_confirmation
   513→      orderStatus = 'confirmed'
   514→      confirmedAt = new Date()
   515→    } else {
   516→      // COD order — check company_order_settings
   517→      const settings = await db.companyOrderSetting.findUnique({
   518→        where: { companyId: ctx.company.id },
   519→      })
   520→      if (!settings?.requireOrderConfirmation) {
   521→        orderStatus = 'confirmed'
   522→        confirmedAt = new Date()
   523→        skippedConfirmation = true
   524→      }
   525→    }
   526→
   527→    // 8. Generate order number
   528→    const flowopsOrderNumber = await generateOrderNumber(ctx.company.id)
   529→
   530→    // 9. Create the order
   531→    const order = await db.order.create({
   532→      data: {
   533→        organizationId: ctx.company.organizationId,
   534→        companyId: ctx.company.id,
   535→        flowopsOrderNumber,
   536→        orderSource: 'manual',
   537→        customerId,
   538→        // Customer Management System integration (migration 002):
   539→        //   - recipientName: the name to deliver to (defaults to customer.name)
   540→        //   - usedCustomerAddressId / usedCustomerPhoneId: which saved
   541→        //     customer_addresses / customer_phones rows were used for this
   542→        //     order, for lastUsedAt tracking and reporting. Nullable when a
   543→        //     one-off address/phone was typed without saving.
   544→        recipientName,
   545→        usedCustomerAddressId,
   546→        usedCustomerPhoneId,
   547→        status: orderStatus,
   548→        paymentType: d.payment_type,
   549→        paymentStatus,
   550→        paymentSource,
   551→        subtotal,
   552→        discountAmount: discountAmount || null,
   553→        discountReason: d.discount_reason || null,
   554→        courierCharges: courierCharges || null,
   555→        estimatedDeliveryCharge: estimatedDeliveryCharge || null,
   556→        taxAmount: taxAmount || null,
   557→        taxLabel: d.tax_label?.trim() || null,
   558→        totalOrderValue,
   559→        advanceAmount,
   560→        advancePaymentMethod,
   561→        advancePaymentReference,
   562→        advancePaidAt,
   563→        remainingCodAmount,
   564→        deliveryAddress: d.delivery_address,
   565→        deliveryCity: d.delivery_city,
   566→        courierName: d.courier_name || null,
   567→        courierCompanyIntegrationId: d.courier_company_integration_id || null,
   568→        recommendedCourierCompanyIntegrationId: d.courier_company_integration_id || null,
   569→        dispatchLocationId: d.dispatch_location_id,
   570→        notesForCourier: d.notes_for_courier || null,
   571→        skippedConfirmation,

   574→      },
   575→    })
   576→
   577→    // 10. Create order items (fulfillment_status = 'reserved' as PLACEHOLDER)
   578→    const createdItems: Array<{ id: string; orgVariantId: string; quantity: number }> = []
   579→    for (const itemData of orderItemsData) {

   586→          unitPrice: itemData.unitPrice,

   589→          fulfillmentTypeSnapshot: itemData.fulfillmentTypeSnapshot,
   590→          reservedLocationId: d.dispatch_location_id,
   591→        },
   592→      })
   593→      createdItems.push({
   594→        id: item.id,
   595→        orgVariantId: item.orgVariantId,
   596→        quantity: item.quantity,
   597→      })
   598→    }
   599→
   600→    // 11. Audit log
   601→    await insertAuditLog({
   602→      action: 'order.created',
   603→      entityType: 'order',
   604→      entityId: order.id,
   605→      companyId: ctx.company.id,
   606→      organizationId: ctx.company.organizationId,
   607→      userId: ctx.user.id,
   608→      employeeId: ctx.employee.id,
   609→      newValues: {
   610→        flowopsOrderNumber,
   611→        status: orderStatus,
   612→        paymentType: d.payment_type,
   613→        paymentStatus,
   614→        itemCount: orderItemsData.length,
   615→        totalOrderValue,
   616→        customerId,
   617→      },
   618→    })
   619→
   620→    // 12. Customer Management System integration:
   621→    //   a. If a saved address was selected, bump its lastUsedAt to NOW() so
   622→    //      the UI can sort saved addresses by recency.
   623→    //   b. If a brand-new one-off address was typed AND the caller set
   624→    //      save_address_for_next_time=true, persist it as a new
   625→    //      customer_addresses row (and link it via usedCustomerAddressId
   626→    //      for future tracking). This is the "Save this address for next
   627→    //      time" UX from Step 3's frontend.
   628→    //   c. Recompute cached customer stats (total_orders_count, etc.) and
   629→    //      auto-flag if RTO threshold crossed.
   630→    if (selectedSavedAddressId) {
   631→      await markAddressAsUsed(selectedSavedAddressId)
   632→    } else if (saveAddressForNextTime && d.delivery_address && d.delivery_city) {
   633→      // One-off address typed + user opted to save it for next time.
   634→      // Persist as a non-default customer_addresses row (default stays as-is
   635→      // unless the user explicitly changes it via the customer profile).
   636→      const savedAddr = await db.customerAddress.create({
   637→        data: {
   638→          customerId,
   639→          organizationId: ctx.company.organizationId,
   640→          label: null,
   641→          address: d.delivery_address,
   642→          city: d.delivery_city,
   643→          isDefault: false,
   644→          lastUsedAt: new Date(),
   645→        },
   646→      })
   647→      // Link the order to the newly-saved address for future tracking.
   648→      await db.order.update({
   649→        where: { id: order.id },
   650→        data: { usedCustomerAddressId: savedAddr.id },
   651→      })
   652→    }
   653→
   654→    await updateCustomerStats(customerId)
   655→
   656→    // 13. If the order auto-confirmed (payment-driven or company setting),
   657→    // run the stock reservation logic immediately.
   658→    if (orderStatus === 'confirmed') {
   659→      await reserveOrderStock(order.id, ctx)
   660→    }
   661→
   662→    await insertMetricEvent({
   663→      companyId: ctx.company.id,
   664→      entityType: 'order',
   665→      entityId: order.id,
   666→      metricKey: 'order.created',
   667→      numericValue: Number(totalOrderValue),
   668→      dimensions: { order_source: 'manual', payment_type: d.payment_type, company_id: ctx.company.id },
   669→    }).catch(() => {})
   670→
   671→    return {
   672→      success: true,
   673→      data: { orderId: order.id, flowopsOrderNumber, orderItems: createdItems },
   674→    }
   675→  } catch (err) {
   676→    return {
   677→      success: false,
   678→      error: err instanceof Error ? err.message : 'Failed to create order',
   679→    }
   680→  }
   681→}

   945→export async function confirmOrder(orderId: string): Promise<ActionResult> {
   946→  try {
   947→    const ctx = await getWorkspace()
   948→    await requirePermission(ctx, PERMISSIONS.ORDERS_MANAGE)
   949→
   950→    const order = await db.order.findFirst({
   951→      where: { id: orderId, companyId: ctx.company.id },
   952→    })
   953→    if (!order) return { success: false, error: 'Order not found' }
   954→    if (order.status !== 'pending') {
   955→      return { success: false, error: `Order is already ${order.status} (cannot confirm)` }
   956→    }
   957→
   958→    await db.order.update({
   959→      where: { id: orderId },
   960→      data: { status: 'confirmed', confirmedAt: new Date() },
   961→    })
   962→
   963→    await insertAuditLog({
   964→      action: 'order.confirmed',
   965→      entityType: 'order',
   966→      entityId: orderId,
   967→      companyId: ctx.company.id,
   968→      organizationId: ctx.company.organizationId,
   969→      userId: ctx.user.id,
   970→      employeeId: ctx.employee.id,
   971→      oldValues: { status: 'pending' },
   972→      newValues: { status: 'confirmed' },
   973→    })
   974→
   975→    // Step 3: Run stock reservation for all items.
   976→    // This may flip the order to 'partially_backordered' if some items
   977→    // can't be fulfilled immediately.
   978→    const reserveResult = await reserveOrderStock(orderId, ctx)
   979→    if (!reserveResult.success) {
   980→      // Non-fatal — the order is confirmed, but reservation had issues.
   981→      // The results array has per-item details.
   982→    }
   983→
   984→    await insertMetricEvent({
   985→      companyId: ctx.company.id,
   986→      entityType: 'order',
   987→      entityId: orderId,
   988→      metricKey: 'order.confirmed',
   989→      numericValue: Number(order.totalOrderValue),
   990→      dimensions: { confirmation_method: 'manual' },
   991→    }).catch(() => {})
   992→
   993→    // Recompute cached customer stats (order count may change if this was a
   994→    // re-confirmation, and the status transition affects value/rto calcs).
   995→    // Non-fatal — never break the confirm action on a stats failure.
   996→    await updateCustomerStats(order.customerId).catch(() => {})
   997→
   998→    return { success: true }
   999→  } catch (err) {
  1000→    return {
  1001→      success: false,
  1002→      error: err instanceof Error ? err.message : 'Failed to confirm order',
  1003→    }
  1004→  }
  1005→}

  1248→export async function cancelOrder(input: CancelOrderInput): Promise<ActionResult> {
  1249→  try {
  1250→    const ctx = await getWorkspace()
  1251→    await requirePermission(ctx, PERMISSIONS.ORDERS_CANCEL)
  1252→
  1253→    const parsed = cancelOrderSchema.safeParse(input)
  1254→    if (!parsed.success) {
  1255→      return { success: false, error: parsed.error.issues[0]?.message ?? 'Invalid input' }
  1256→    }
  1257→    const d = parsed.data
  1258→
  1259→    const order = await db.order.findFirst({
  1260→      where: { id: d.order_id, companyId: ctx.company.id },
  1261→    })
  1262→    if (!order) return { success: false, error: 'Order not found' }
  1263→
  1264→    const nonCancellableStatuses = ['dispatched', 'delivered', 'rto', 'cancelled', 'refunded']
  1265→    if (nonCancellableStatuses.includes(order.status)) {
  1266→      return {
  1267→        success: false,
  1268→        error: `Cannot cancel an order that is already ${order.status} (use the returns flow instead)`,
  1269→      }
  1270→    }
  1271→
  1272→    await db.order.update({
  1273→      where: { id: d.order_id },
  1274→      data: {
  1275→        status: 'cancelled',
  1276→        cancelledAt: new Date(),
  1277→        cancellationReason: d.cancellation_reason,
  1278→      },
  1279→    })
  1280→
  1281→    // Step 3: Unreserve stock for any items with fulfillment_status='reserved'.
  1282→    // Items with fulfillment_status='backordered' need no inventory action
  1283→    // (no reservation ever existed for them) — they'll be orphaned since
  1284→    // the order is now cancelled and won't be picked up by future
  1285→    // checkAndFulfillBackorders() runs (which skip cancelled orders).
  1286→    const reservedItems = await db.orderItem.findMany({
  1287→      where: { orderId: d.order_id, fulfillmentStatus: 'reserved' },
  1288→    })
  1289→
  1290→    for (const item of reservedItems) {
  1291→      const locationId = item.reservedLocationId ?? order.dispatchLocationId
  1292→      if (!locationId) continue
  1293→
  1294→      await unreserveStockForOrder({
  1295→        orgVariantId: item.orgVariantId,
  1296→        locationId,
  1297→        organizationId: order.organizationId,
  1298→        companyId: ctx.company.id,
  1299→        employeeId: ctx.employee.id,
  1300→        quantity: item.quantity,
  1301→        orderId: d.order_id,
  1302→      })
  1303→    }
  1304→
  1305→    await insertAuditLog({
  1306→      action: 'order.cancelled',
  1307→      entityType: 'order',
  1308→      entityId: d.order_id,
  1309→      companyId: ctx.company.id,
  1310→      organizationId: ctx.company.organizationId,
  1311→      userId: ctx.user.id,
  1312→      employeeId: ctx.employee.id,
  1313→      oldValues: { status: order.status },
  1314→      newValues: { status: 'cancelled', reason: d.cancellation_reason, unreservedItems: reservedItems.length },
  1315→    })
  1316→
  1317→    await insertMetricEvent({
  1318→      companyId: ctx.company.id,
  1319→      entityType: 'order',
  1320→      entityId: d.order_id,
  1321→      metricKey: 'order.cancelled',
  1322→      numericValue: Number(order.totalOrderValue),
  1323→      dimensions: { cancellation_reason: d.cancellation_reason, had_reserved_items: reservedItems.length > 0 },
  1324→    }).catch(() => {})
  1325→
  1326→    // Recompute cached customer stats (cancelled orders are excluded from
  1327→    // total_orders_count, so this count drops). Non-fatal.
  1328→    await updateCustomerStats(order.customerId).catch(() => {})
  1329→
  1330→    return { success: true }
  1331→  } catch (err) {
  1332→    return {
  1333→      success: false,
  1334→      error: err instanceof Error ? err.message : 'Failed to cancel order',
  1335→    }
  1336→  }
  1337→}

  1705→// Step 3: Dispatch + Processing/Packing actions
  1706→// ──────────────────────────────────────────────────────────────
  1707→
  1708→/**
  1709→ * Dispatch an order — deducts stock from inventory, sets tracking info.
  1710→ *
  1711→ * Business rules:
  1712→ *   - If order.status = 'pending' (confirmation was skipped): run the
  1713→ *     full confirmation/reservation logic inline first, then proceed.
  1714→ *   - If any item is still 'backordered' after reservation: BLOCK dispatch
  1715→ *     with a clear error (hard rule — no split shipments).
  1716→ *   - For each reserved item: call dispatchOrder() (inventory) which
  1717→ *     deducts on_hand, releases the reservation, and locks COGS.
  1718→ *   - The backfill_order_timestamps() trigger auto-sets confirmed_at/
  1719→ *     packed_at if they were still NULL.
  1720→ *   - If company_order_settings.require_packing_step = TRUE, verify
  1721→ *     order.packed_at IS NOT NULL before proceeding.
  1722→ */
  1723→export async function dispatchOrderAction(
  1724→  orderId: string,
  1725→  trackingNumber: string,
  1726→  courierName?: string,
  1727→): Promise<ActionResult> {
  1728→  try {
  1729→    const ctx = await getWorkspace()
  1730→    await requirePermission(ctx, PERMISSIONS.ORDERS_FULFILL)
  1731→
  1732→    const order = await db.order.findFirst({
  1733→      where: { id: orderId, companyId: ctx.company.id },
  1734→    })
  1735→    if (!order) return { success: false, error: 'Order not found' }
  1736→
  1737→    // Cannot dispatch already-dispatched/delivered/cancelled orders
  1738→    if (['dispatched', 'delivered', 'rto', 'cancelled', 'refunded'].includes(order.status)) {
  1739→      return { success: false, error: `Cannot dispatch an order that is already ${order.status}` }
  1740→    }
  1741→
  1742→    // Check packing requirement
  1743→    const settings = await db.companyOrderSetting.findUnique({
  1744→      where: { companyId: ctx.company.id },
  1745→    })
  1746→    if (settings?.requirePackingStep && !order.packedAt) {
  1747→      return {
  1748→        success: false,
  1749→        error: 'This order must be marked as packed before dispatching. Use markOrderPacked() first.',
  1750→      }
  1751→    }
  1752→
  1753→    // If order is still 'pending' (confirmation was skipped), run the
  1754→    // full confirmation + reservation logic inline first.
  1755→    if (order.status === 'pending') {
  1756→      await db.order.update({
  1757→        where: { id: orderId },
  1758→        data: { status: 'confirmed', confirmedAt: new Date() },
  1759→      })
  1760→      await reserveOrderStock(orderId, ctx)
  1761→    }
  1762→
  1763→    // Re-fetch the order to get the updated status after reservation
  1764→    const updatedOrder = await db.order.findFirst({
  1765→      where: { id: orderId },
  1766→      select: { status: true, dispatchLocationId: true, organizationId: true },
  1767→    })
  1768→    if (!updatedOrder) return { success: false, error: 'Order not found after reservation' }
  1769→
  1770→    // Check for backordered items — BLOCK dispatch if any exist
  1771→    const backorderedItems = await db.orderItem.findMany({
  1772→      where: { orderId, fulfillmentStatus: 'backordered' },
  1773→      include: {
  1774→        orgVariant: { select: { sku: true } },
  1775→      },
  1776→    })
  1777→
  1778→    if (backorderedItems.length > 0) {
  1779→      const itemList = backorderedItems
  1780→        .map((i) => `${i.orgVariant.sku} (qty: ${i.quantity})`)
  1781→        .join(', ')
  1782→      return {
  1783→        success: false,
  1784→        error: `Cannot dispatch: ${backorderedItems.length} item(s) are still backordered: ${itemList}. Receive stock or cancel those items first.`,
  1785→      }
  1786→    }
  1787→
  1788→    // Dispatch each reserved item
  1789→    const itemsToDispatch = await db.orderItem.findMany({
  1790→      where: { orderId, fulfillmentStatus: 'reserved' },
  1791→    })
  1792→
  1793→    for (const item of itemsToDispatch) {
  1794→      const locationId = item.reservedLocationId ?? updatedOrder.dispatchLocationId
  1795→      if (!locationId) {
  1796→        return { success: false, error: `Order item ${item.id} has no dispatch location` }
  1797→      }
  1798→
  1799→      const dispatchResult = await dispatchInventory({
  1800→        orgVariantId: item.orgVariantId,
  1801→        locationId,
  1802→        organizationId: updatedOrder.organizationId,
  1803→        companyId: ctx.company.id,
  1804→        employeeId: ctx.employee.id,
  1805→        quantity: item.quantity,
  1806→        orderId,
  1807→      })
  1808→
  1809→      if (!dispatchResult.success) {
  1810→        return {
  1811→          success: false,
  1812→          error: `Failed to dispatch item: ${dispatchResult.error}`,
  1813→        }
  1814→      }
  1815→
  1816→      // Mark item as dispatched
  1817→      await db.orderItem.update({
  1818→        where: { id: item.id },
  1819→        data: { fulfillmentStatus: 'dispatched', fulfilledAt: new Date() },
  1820→      })
  1821→    }
  1822→
  1823→    // Update order status — the backfill_order_timestamps() trigger
  1824→    // will auto-set confirmedAt/packedAt if they were still NULL.
  1825→    await db.order.update({
  1826→      where: { id: orderId },
  1827→      data: {
  1828→        status: 'dispatched',
  1829→        dispatchedAt: new Date(),
  1830→        trackingNumber: trackingNumber || null,
  1831→        courierName: courierName || order.courierName,
  1832→      },
  1833→    })
  1834→
  1835→    await insertAuditLog({
  1836→      action: 'order.dispatched',
  1837→      entityType: 'order',
  1838→      entityId: orderId,
  1839→      companyId: ctx.company.id,
  1840→      organizationId: ctx.company.organizationId,
  1841→      userId: ctx.user.id,
  1842→      employeeId: ctx.employee.id,
  1843→      oldValues: { status: updatedOrder.status },
  1844→      newValues: {
  1845→        status: 'dispatched',
  1846→        trackingNumber,
  1847→        courierName,
  1848→        itemsDispatched: itemsToDispatch.length,
  1849→      },
  1850→    })
  1851→
  1852→    const timeToDispatchHours = order.createdAt
  1853→      ? Math.round((new Date().getTime() - new Date(order.createdAt).getTime()) / (1000 * 60 * 60))
  1854→      : 0
  1855→    await insertMetricEvent({
  1856→      companyId: ctx.company.id,
  1857→      entityType: 'order',
  1858→      entityId: orderId,
  1859→      metricKey: 'order.dispatched',
  1860→      numericValue: Number(order.totalOrderValue),
  1861→      dimensions: {
  1862→        courier_name: courierName || order.courierName,
  1863→        employee_id: ctx.employee.id,
  1864→        skipped_confirmation: updatedOrder.status === 'pending',
  1865→        skipped_packing: !order.packedAt,
  1866→        time_to_dispatch_hours: timeToDispatchHours,
  1867→      },
  1868→    }).catch(() => {})
  1869→
  1870→    // Update customer stats
  1871→    const orderWithCustomer = await db.order.findUnique({
  1872→      where: { id: orderId },
  1873→      select: { customerId: true },
  1874→    })
  1875→    if (orderWithCustomer) {
  1876→      await updateCustomerStats(orderWithCustomer.customerId)
  1877→    }
  1878→
  1879→    return { success: true }
  1880→  } catch (err) {
  1881→    return {
  1882→      success: false,
  1883→      error: err instanceof Error ? err.message : 'Failed to dispatch order',
  1884→    }
  1885→  }
  1886→}
