   134→const PKR = new Intl.NumberFormat('en-PK', { maximumFractionDigits: 0 })
   135→
   136→function formatPKR(n: number): string {
   137→  return `Rs. ${PKR.format(n)}`

   140→function getErrorMessage(err: unknown): string {
   141→  if (err instanceof FetchError) return err.message
   142→  if (err instanceof Error) return err.message
   143→  return 'Something went wrong. Please try again.'
   144→}

   146→function parseQty(v: string): number {
   147→  if (v === '') return 0
   148→  const n = Number(v)
   149→  return Number.isFinite(n) && n >= 0 ? Math.floor(n) : 0
   150→}
   151→
   152→function parsePrice(v: string): number {
   153→  if (v === '') return 0
   154→  const n = Number(v)
   155→  return Number.isFinite(n) && n >= 0 ? n : 0
   156→}

   175→/** Stock badge for a variant based on fulfillmentType. */
   176→function stockBadgeFor(
   177→  fulfillmentType: string,
   178→): { label: string; className: string } {
   179→  if (fulfillmentType === 'made_to_order') {
   180→    return {
   181→      label: 'Made to Order',
   182→      className: 'bg-purple-50 text-purple-700 border-purple-200',
   183→    }
   184→  }
   185→  if (fulfillmentType === 'backorder') {
   186→    return {
   187→      label: 'Backorder Available',
   188→      className: 'bg-orange-50 text-orange-700 border-orange-200',
   189→    }
   190→  }
   191→  return {
   192→    label: 'In Stock',
   193→    className: 'bg-emerald-50 text-emerald-700 border-emerald-200',
   194→  }
   195→}

   200→
   201→export function OrderCreateView() {
   202→  const navigate = useAppStore((s) => s.navigate)
   203→  const can = useCan()
   204→  const queryClient = useQueryClient()
   205→  const canCreate = can(PERMISSIONS.ORDERS_CREATE)
   206→
   207→  const [step, setStep] = useState(1)
   208→  // Tracks the furthest step the user has reached — used to gate the
   209→  // stepper's "click to go back" behaviour so users can only jump to
   210→  // previously-visited (or current) steps.
   211→  const [maxStep, setMaxStep] = useState(1)
   212→
   213→  // ── Form state ────────────────────────────────────────────────────────────
   214→  // Customer
   215→  const [customerMode, setCustomerMode] = useState<'existing' | 'new'>('existing')
   216→  const [phoneSearch, setPhoneSearch] = useState('')
   217→  const [debouncedPhone, setDebouncedPhone] = useState('')
   218→  const [selectedCustomer, setSelectedCustomer] = useState<CustomerRow | null>(null)
   219→  const [newCustomer, setNewCustomer] = useState({
   220→    name: '',
   221→    phone: '',
   222→    alternate_phone: '',
   223→    email: '',
   224→    address: '',
   225→    city: '',
   226→  })
   227→
   228→  // Items
   229→  const [cart, setCart] = useState<CartItem[]>([])
   230→  const [variantSearch, setVariantSearch] = useState('')
   231→
   232→  // Payment
   233→  const [paymentType, setPaymentType] = useState<PaymentType>('full_cod')
   234→  const [advanceAmount, setAdvanceAmount] = useState('')
   235→  const [advancePaymentMethod, setAdvancePaymentMethod] = useState('')
   236→  const [advancePaymentReference, setAdvancePaymentReference] = useState('')
   237→  const [advancePaymentScreenshotUrl, setAdvancePaymentScreenshotUrl] = useState('')
   238→
   239→  // Delivery
   240→  const [deliveryAddress, setDeliveryAddress] = useState('')
   241→  const [deliveryCity, setDeliveryCity] = useState('')
   242→  const [courierName, setCourierName] = useState('')
   243→  const [dispatchLocationId, setDispatchLocationId] = useState('')
   244→  const [notesForCourier, setNotesForCourier] = useState('')
   245→  const [discountAmount, setDiscountAmount] = useState('')
   246→  const [discountReason, setDiscountReason] = useState('')
   247→
   248→  // ── Debounce phone search ────────────────────────────────────────────────
   249→  useEffect(() => {
   250→    const t = setTimeout(() => setDebouncedPhone(phoneSearch.trim()), 350)
   251→    return () => clearTimeout(t)
   252→  }, [phoneSearch])
   253→
   254→  // ── Data queries ──────────────────────────────────────────────────────────
   255→  const locationsQuery = useQuery<LocationsResponse>({
   256→    queryKey: ['locations'],
   257→    queryFn: () => api.get<LocationsResponse>('/api/inventory-locations'),
   258→    staleTime: 30_000,
   259→  })
   260→
   261→  const productsQuery = useQuery<ProductsResponse>({
   262→    queryKey: ['products', 'for-order-create'],
   263→    queryFn: () => api.get<ProductsResponse>('/api/products?pageSize=100'),
   264→    staleTime: 60_000,
   265→  })
   266→
   267→  const trimmedPhone = debouncedPhone
   268→  const customersQuery = useQuery<CustomersSearchResponse>({
   269→    queryKey: ['customers', 'search', trimmedPhone],
   270→    queryFn: () =>
   271→      api.get<CustomersSearchResponse>(
   272→        `/api/customers?search=${encodeURIComponent(trimmedPhone)}&limit=10`,
   273→      ),
   274→    enabled: trimmedPhone.length >= 4,
   275→    staleTime: 10_000,
   276→  })
   277→
   278→  // ── Variant options (flatten products → variants) ─────────────────────────
   279→  const variantOptions: VariantOption[] = useMemo(() => {
   280→    const list: VariantOption[] = []
   281→    for (const p of productsQuery.data?.products ?? []) {
   282→      for (const v of p.variants) {
   283→        list.push({
   284→          variantId: v.id,
   285→          sku: v.sku,
   286→          productTitle: p.title,
   287→          costPrice: v.costPrice,
   288→          salePrice: v.salePrice,
   289→          fulfillmentType: v.fulfillmentType,
   290→        })
   291→      }
   292→    }
   293→    return list
   294→  }, [productsQuery.data])
   295→
   296→  const variantSearchResults = useMemo(() => {
   297→    if (!variantSearch.trim()) return []
   298→    const q = variantSearch.toLowerCase()
   299→    return variantOptions
   300→      .filter(
   301→        (v) => v.sku.toLowerCase().includes(q) || v.productTitle.toLowerCase().includes(q),
   302→      )
   303→      .slice(0, 8)
   304→  }, [variantOptions, variantSearch])
   305→
   306→  // ── Computed totals ───────────────────────────────────────────────────────
   307→  const subtotal = useMemo(
   308→    () => cart.reduce((s, i) => s + i.unitPrice * i.quantity, 0),
   309→    [cart],
   310→  )
   311→  const discount = parsePrice(discountAmount)
   312→  const total = Math.max(0, subtotal - discount)
   313→
   314→  // ── When switching to "new" customer, pre-fill delivery fields ────────────
   315→  useEffect(() => {
   316→    if (customerMode === 'new') {
   317→      if (newCustomer.address && !deliveryAddress) {
   318→        setDeliveryAddress(newCustomer.address)
   319→      }
   320→      if (newCustomer.city && !deliveryCity) {
   321→        setDeliveryCity(newCustomer.city)
   322→      }
   323→    }
   324→  }, [customerMode])
   325→
   326→  // ── Auto-select default dispatch location ─────────────────────────────────
   327→  useEffect(() => {
   328→    if (!dispatchLocationId && locationsQuery.data?.locations.length) {
   329→      const def = locationsQuery.data.locations.find((l) => l.isDefault)
   330→      setDispatchLocationId(def?.id ?? locationsQuery.data.locations[0].id)
   331→    }
   332→  }, [dispatchLocationId, locationsQuery.data])
   333→
   334→  // ── Item manipulation ─────────────────────────────────────────────────────
   335→  const addVariant = (v: VariantOption) => {
   336→    setCart((prev) => {
   337→      if (prev.some((i) => i.variantId === v.variantId)) {
   338→        toast.info('That variant is already in the cart.')
   339→        return prev
   340→      }
   341→      const unitPrice = v.salePrice ?? v.costPrice
   342→      return [
   343→        ...prev,
   344→        {
   345→          variantId: v.variantId,
   346→          sku: v.sku,
   347→          productTitle: v.productTitle,
   348→          unitPrice,
   349→          quantity: 1,
   350→          fulfillmentType: v.fulfillmentType,
   351→        },
   352→      ]
   353→    })
   354→    setVariantSearch('')
   355→  }
   356→
   357→  const removeItem = (variantId: string) => {
   358→    setCart((prev) => prev.filter((i) => i.variantId !== variantId))
   359→  }
   360→
   361→  const updateItem = (variantId: string, patch: Partial<CartItem>) => {
   362→    setCart((prev) =>
   363→      prev.map((i) => (i.variantId === variantId ? { ...i, ...patch } : i)),
   364→    )
   365→  }
   366→
   367→  // ── Step validation ───────────────────────────────────────────────────────
   368→  const customerStepValid =
   369→    customerMode === 'existing'
   370→      ? !!selectedCustomer
   371→      : newCustomer.name.trim().length >= 2 &&
   372→        newCustomer.phone.trim().length >= 7 &&
   373→        newCustomer.address.trim().length >= 2 &&
   374→        newCustomer.city.trim().length >= 2
   375→
   376→  const itemsStepValid = cart.length > 0 && cart.every((i) => i.quantity > 0 && i.unitPrice >= 0)
   377→
   378→  const paymentStepValid =
   379→    paymentType === 'full_cod' ||
   380→    (paymentType === 'partial_advance' && parsePrice(advanceAmount) > 0) ||
   381→    paymentType === 'fully_prepaid'
   382→
   383→  const deliveryStepValid =
   384→    deliveryAddress.trim().length >= 2 &&
   385→    deliveryCity.trim().length >= 2 &&
   386→    !!dispatchLocationId
   387→
   388→  const canGoNext = (): boolean => {
   389→    if (step === 1) return customerStepValid
   390→    if (step === 2) return itemsStepValid
   391→    if (step === 3) return paymentStepValid
   392→    if (step === 4) return deliveryStepValid
   393→    return true
   394→  }
   395→
   396→  const goToStep = (target: number) => {
   397→    if (target < 1 || target > STEPS.length) return
   398→    // Only allow jumping to a previously-visited (or current) step.
   399→    if (target <= maxStep) setStep(target)
   400→  }
   401→
   402→  const goNext = () => {
   403→    if (!canGoNext()) return
   404→    const next = step + 1
   405→    setStep(next)
   406→    setMaxStep((m) => Math.max(m, next))
   407→  }
   408→
   409→  // ── Mutation: create order ────────────────────────────────────────────────
   410→  const createMutation = useMutation({
   411→    mutationFn: async (payload: unknown) =>
   412→      api.post<CreateOrderResponse>('/api/orders', payload),
   413→    onSuccess: (data) => {
   414→      toast.success(`Order ${data.flowopsOrderNumber} created successfully.`)
   415→      void queryClient.invalidateQueries({ queryKey: ['orders'] })
   416→      navigate({ name: 'order-detail', id: data.orderId })
   417→    },
   418→    onError: (err) => toast.error(getErrorMessage(err)),
   419→  })
   420→
   421→  // ── Permission gate (after all hooks) ─────────────────────────────────────
   422→  if (!canCreate) {
   423→    return (
   424→      <div className="space-y-6">
   425→        <PageHeader
   426→          title="Create Order"
   427→          description="Manually create a customer order"
   428→          actions={
   429→            <Button variant="outline" size="sm" onClick={() => navigate({ name: 'orders' })}>
   430→              <ArrowLeft className="h-4 w-4" /> Back to Orders
   431→            </Button>
   432→          }
   433→        />
   434→        <Card>
   435→          <CardContent className="p-10 text-center">
   436→            <AlertCircle className="h-10 w-10 mx-auto text-muted-foreground/40 mb-3" />
   437→            <p className="text-sm text-muted-foreground">
   438→              You don&apos;t have permission to create orders.
   439→            </p>
   440→          </CardContent>
   441→        </Card>
   442→      </div>
   443→    )
   444→  }
   445→
   446→  // ── Build payload + submit ────────────────────────────────────────────────
   447→  const handleSubmit = () => {
   448→    if (!customerStepValid || !itemsStepValid || !paymentStepValid || !deliveryStepValid) {
   449→      toast.error('Please complete all required fields before creating the order.')
   450→      return
   451→    }
   452→
   453→    const payload: Record<string, unknown> = {
   454→      items: cart.map((i) => ({
   455→        org_variant_id: i.variantId,
   456→        quantity: i.quantity,
   457→        unit_price: i.unitPrice,
   458→      })),
   459→      payment_type: paymentType,
   460→      delivery_address: deliveryAddress.trim(),
   461→      delivery_city: deliveryCity.trim(),
   462→      courier_name: courierName.trim() || undefined,
   463→      dispatch_location_id: dispatchLocationId,
   464→      notes_for_courier: notesForCourier.trim() || undefined,
   465→      discount_amount: discount > 0 ? discount : undefined,
   466→      discount_reason: discountReason.trim() || undefined,
   467→    }
   468→
   469→    if (paymentType === 'partial_advance' || paymentType === 'fully_prepaid') {
   470→      payload.advance_amount =
   471→        paymentType === 'fully_prepaid' ? total : parsePrice(advanceAmount)
   472→      payload.advance_payment_method = advancePaymentMethod || undefined
   473→      payload.advance_payment_reference = advancePaymentReference.trim() || undefined
   474→      payload.advance_payment_screenshot_url = advancePaymentScreenshotUrl.trim() || undefined
   475→    }
   476→
   477→    if (customerMode === 'existing' && selectedCustomer) {
   478→      payload.customer_id = selectedCustomer.id
   479→    } else {
   480→      payload.customer = {
   481→        name: newCustomer.name.trim(),
   482→        phone: newCustomer.phone.trim(),
   483→        alternate_phone: newCustomer.alternate_phone.trim() || undefined,
   484→        email: newCustomer.email.trim() || undefined,
   485→        addresses: [
   486→          {
   487→            label: 'Home',
   488→            address: newCustomer.address.trim(),
   489→            city: newCustomer.city.trim(),
   490→            is_default: true,
   491→          },
   492→        ],
   493→      }
   494→    }
   495→
   496→    createMutation.mutate(payload)
   497→  }
   498→
   499→  return (
   500→    <div className="space-y-6">
   501→      <PageHeader
   502→        title="Create Order"
   503→        description="Manually create a customer order in 5 quick steps"
   504→        actions={
   505→          <Button variant="outline" size="sm" onClick={() => navigate({ name: 'orders' })}>
   506→            <ArrowLeft className="h-4 w-4" /> Back to Orders
   507→          </Button>
   508→        }
   509→      />
   510→
   511→      {/* ── Stepper ─────────────────────────────────────────────────────────── */}
   512→      <Stepper currentStep={step} maxStep={maxStep} onStepClick={goToStep} />
   513→
   514→      {/* ── Step content ────────────────────────────────────────────────────── */}
   515→      {step === 1 && (
   516→        <CustomerStep
   517→          customerMode={customerMode}
   518→          setCustomerMode={(m) => {
   519→            setCustomerMode(m)
   520→            setSelectedCustomer(null)
   521→          }}
   522→          phoneSearch={phoneSearch}
   523→          setPhoneSearch={setPhoneSearch}
   524→          isSearching={customersQuery.isFetching && trimmedPhone.length >= 4}
   525→          selectedCustomer={selectedCustomer}
   526→          setSelectedCustomer={setSelectedCustomer}
   527→          newCustomer={newCustomer}
   528→          setNewCustomer={setNewCustomer}
   529→          customers={customersQuery.data?.customers ?? []}
   530→          hasSearched={trimmedPhone.length >= 4}
   531→        />
   532→      )}
   533→
   534→      {step === 2 && (
   535→        <ItemsStep
   536→          cart={cart}
   537→          variantSearch={variantSearch}
   538→          setVariantSearch={setVariantSearch}
   539→          variantSearchResults={variantSearchResults}
   540→          addVariant={addVariant}
   541→          removeItem={removeItem}
   542→          updateItem={updateItem}
   543→          isLoadingProducts={productsQuery.isLoading}
   544→          subtotal={subtotal}
   545→        />
   546→      )}
   547→
   548→      {step === 3 && (
   549→        <PaymentStep
   550→          paymentType={paymentType}
   551→          setPaymentType={setPaymentType}
   552→          advanceAmount={advanceAmount}
   553→          setAdvanceAmount={setAdvanceAmount}
   554→          advancePaymentMethod={advancePaymentMethod}
   555→          setAdvancePaymentMethod={setAdvancePaymentMethod}
   556→          advancePaymentReference={advancePaymentReference}
   557→          setAdvancePaymentReference={setAdvancePaymentReference}
   558→          advancePaymentScreenshotUrl={advancePaymentScreenshotUrl}
   559→          setAdvancePaymentScreenshotUrl={setAdvancePaymentScreenshotUrl}
   560→          subtotal={subtotal}
   561→          total={total}
   562→          discount={discount}
   563→        />
   564→      )}
   565→
   566→      {step === 4 && (
   567→        <DeliveryStep
   568→          deliveryAddress={deliveryAddress}
   569→          setDeliveryAddress={setDeliveryAddress}
   570→          deliveryCity={deliveryCity}
   571→          setDeliveryCity={setDeliveryCity}
   572→          courierName={courierName}
   573→          setCourierName={setCourierName}
   574→          dispatchLocationId={dispatchLocationId}
   575→          setDispatchLocationId={setDispatchLocationId}
   576→          notesForCourier={notesForCourier}
   577→          setNotesForCourier={setNotesForCourier}
   578→          discountAmount={discountAmount}
   579→          setDiscountAmount={setDiscountAmount}
   580→          discountReason={discountReason}
   581→          setDiscountReason={setDiscountReason}
   582→          locations={locationsQuery.data?.locations ?? []}
   583→          isLoadingLocations={locationsQuery.isLoading}
   584→          subtotal={subtotal}
   585→          discount={discount}
   586→          total={total}
   587→        />
   588→      )}
   589→
   590→      {step === 5 && (
   591→        <ReviewStep
   592→          customerMode={customerMode}
   593→          selectedCustomer={selectedCustomer}
   594→          newCustomer={newCustomer}
   595→          cart={cart}
   596→          paymentType={paymentType}
   597→          advanceAmount={advanceAmount}
   598→          advancePaymentMethod={advancePaymentMethod}
   599→          advancePaymentReference={advancePaymentReference}
   600→          advancePaymentScreenshotUrl={advancePaymentScreenshotUrl}
   601→          deliveryAddress={deliveryAddress}
   602→          deliveryCity={deliveryCity}
   603→          courierName={courierName}
   604→          dispatchLocationId={dispatchLocationId}
   605→          locations={locationsQuery.data?.locations ?? []}
   606→          notesForCourier={notesForCourier}
   607→          discountAmount={discountAmount}
   608→          discountReason={discountReason}
   609→          subtotal={subtotal}
   610→          discount={discount}

  1481→/**
  1482→ * Screenshot upload — uploads to /api/upload?type=order_screenshot on file
  1483→ * select and stores the returned URL. Follows the logo-upload pattern.
  1484→ */
  1485→function ScreenshotUpload({
  1486→  url,
  1487→  onChange,
  1488→}: {
  1489→  url: string
  1490→  onChange: (url: string) => void
  1491→}) {
  1492→  const inputRef = useRef<HTMLInputElement>(null)
  1493→  const [uploading, setUploading] = useState(false)
  1494→  const [error, setError] = useState<string | null>(null)
  1495→
  1496→  async function handleFile(file: File) {
  1497→    setError(null)
  1498→    if (file.size > 5 * 1024 * 1024) {
  1499→      setError('File too large. Maximum 5 MB.')
  1500→      return
  1501→    }
  1502→    if (!['image/jpeg', 'image/png', 'image/webp'].includes(file.type)) {
  1503→      setError('Only JPG, PNG, and WebP images are allowed.')
  1504→      return
  1505→    }
  1506→    setUploading(true)
  1507→    try {
  1508→      const fd = new FormData()
  1509→      fd.append('file', file)
  1510→      const res = await fetch('/api/upload?type=order_screenshot', {
  1511→        method: 'POST',
  1512→        body: fd,
  1513→      })
  1514→      const text = await res.text()
  1515→      let body: unknown = null
  1516→      if (text) {
  1517→        try {
  1518→          body = JSON.parse(text)
  1519→        } catch {
  1520→          body = text
  1521→        }
  1522→      }
  1523→      if (!res.ok) {
  1524→        const message =
  1525→          body && typeof body === 'object' && 'error' in body
  1526→            ? String((body as { error: unknown }).error)
  1527→            : typeof body === 'string'
  1528→              ? body
  1529→              : `Upload failed (HTTP ${res.status})`
  1530→        throw new Error(message)
  1531→      }
  1532→      const { url: uploadedUrl } = body as { url: string }
  1533→      onChange(uploadedUrl)
  1534→      toast.success('Screenshot uploaded.')
  1535→    } catch (e) {
  1536→      setError(e instanceof Error ? e.message : 'Network error during upload')
  1537→    } finally {
  1538→      setUploading(false)
  1539→    }
  1540→  }
  1541→
  1542→  return (
  1543→    <div className="space-y-1.5">
  1544→      <Label>Payment screenshot (optional)</Label>
  1545→      <div className="flex items-center gap-3">
  1546→        {url ? (
  1547→          <div className="relative group">
  1548→            <button
  1549→              type="button"
  1550→              onClick={() => inputRef.current?.click()}
  1551→              className="block rounded-md border overflow-hidden"
  1552→            >
  1553→              <img
  1554→                src={url}
  1555→                alt="Payment screenshot"
  1556→                className="h-20 w-20 object-cover"
  1557→              />
  1558→            </button>
  1559→            <button
  1560→              type="button"
  1561→              onClick={() => {
  1562→                setError(null)
  1563→                onChange('')
  1564→              }}
  1565→              className="absolute -top-1 -right-1 h-5 w-5 rounded-full bg-destructive text-destructive-foreground flex items-center justify-center shadow-sm hover:scale-110 transition-transform"
  1566→              aria-label="Remove screenshot"
  1567→            >
  1568→              <X className="h-3 w-3" />
  1569→            </button>
  1570→          </div>
  1571→        ) : (
  1572→          <button
  1573→            type="button"
  1574→            onClick={() => inputRef.current?.click()}
  1575→            disabled={uploading}
  1576→            className={cn(
  1577→              'flex h-20 w-20 items-center justify-center rounded-md border-2 border-dashed text-muted-foreground hover:border-primary/50 hover:text-primary transition-colors',
  1578→              uploading && 'opacity-60 cursor-not-allowed',
  1579→            )}
  1580→          >
  1581→            {uploading ? (
  1582→              <Loader2 className="h-5 w-5 animate-spin" />
  1583→            ) : (
  1584→              <Upload className="h-5 w-5" />
  1585→            )}
  1586→          </button>
  1587→        )}
  1588→        <div className="text-xs text-muted-foreground">
  1589→          <p>{url ? 'Click image to replace' : 'Click to upload'}</p>
  1590→          <p>JPG, PNG, WebP — max 5 MB</p>
  1591→          {error && <p className="text-destructive mt-1">{error}</p>}
  1592→        </div>
  1593→        <input
  1594→          ref={inputRef}
  1595→          type="file"
  1596→          accept="image/jpeg,image/png,image/webp"
  1597→          className="hidden"
  1598→          onChange={(e) => {
  1599→            const f = e.target.files?.[0]
  1600→            if (f) handleFile(f)
  1601→            e.target.value = ''
  1602→          }}
  1603→        />
  1604→      </div>
