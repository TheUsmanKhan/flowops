   207→/// Company-scoped roles. System roles (Owner/Founder/Co-Founder/Investor) are
   208→/// elevated and bypass all permission checks. Custom roles use role_permissions.
   209→model Role {
   210→  id            String   @id @default(cuid())
   211→  companyId     String
   212→  company       Company  @relation(fields: [companyId], references: [id], onDelete: Cascade)
   213→  name          String
   214→  description   String?
   215→  roleTier      String   @default("standard") // elevated | standard
   216→  isSystemRole  Boolean  @default(false)
   217→  systemRoleKey String?  // owner | founder | co_founder | investor | null
   218→  // NOTE: systemRoleKey is unique PER COMPANY (not globally), so every
   219→  // company can seed its own owner/founder/co_founder/investor roles.
   220→  // ─── Data scope for the Orders module ───────────────────────────────
   221→  // "all"  → role sees ALL orders in the company (default, preserves
   222→  //          existing behavior for every row after migration)
   223→  // "own"  → role sees only orders where salesEmployeeId === the caller's
   224→  //          own Employee.id (enforced at the query layer, not in SQL)
   225→  ordersDataScope String  @default("all") // "own" | "all"
   226→  isActive      Boolean  @default(true)
   227→  createdById   String?
   228→  createdBy     Profile? @relation("RoleCreator", fields: [createdById], references: [id])
   229→  createdAt     DateTime @default(now())
   230→  updatedAt     DateTime @updatedAt
   231→
   232→  employees       Employee[]
   233→  rolePermissions RolePermission[]
   234→  invitations     Invitation[]
   235→
   236→  @@unique([companyId, name])
   237→  @@unique([companyId, systemRoleKey])
   238→}
