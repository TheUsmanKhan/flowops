   226→const PKR = new Intl.NumberFormat('en-PK', { maximumFractionDigits: 0 })
   227→
   228→function formatPKR(n: number): string {
   229→  return `Rs. ${PKR.format(n)}`
   230→}
   231→
   232→function formatDate(iso: string | null): string {
   233→  if (!iso) return '—'
   234→  try {
   235→    return new Date(iso).toLocaleDateString('en-PK', {
   236→      day: '2-digit',
   237→      month: 'short',
   238→      year: 'numeric',
   239→    })
   240→  } catch {
   241→    return iso
   242→  }
   243→}
   244→
   245→function formatDateTime(iso: string | null): string {
   246→  if (!iso) return '—'
   247→  try {
   248→    return new Date(iso).toLocaleString('en-PK', {
   249→      day: '2-digit',
   250→      month: 'short',
   251→      hour: '2-digit',
   252→      minute: '2-digit',
   253→    })
   254→  } catch {
   255→    return iso
   256→  }
   257→}
   258→
   259→function getErrorMessage(err: unknown): string {
   260→  if (err instanceof FetchError) return err.message
   261→  if (err instanceof Error) return err.message
   262→  return 'Something went wrong.'
   263→}
   264→
   265→function formatAttributeValues(values: Record<string, string>): string {
   266→  if (!values || Object.keys(values).length === 0) return ''
   267→  return Object.entries(values)
   268→    .map(([k, v]) => `${k}: ${v}`)
   269→    .join(' · ')
   270→}

   561→
   562→      {/* ── Status badge row ─────────────────────────────────────────────── */}
   563→      <div className="flex flex-wrap items-center gap-2">
   564→        <Badge variant="outline" className={statusBadge.className}>
   565→          {statusBadge.label}
   566→        </Badge>
   567→        <Badge variant="outline" className={paymentBadge.className}>
   568→          {paymentBadge.label}
