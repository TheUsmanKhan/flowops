   291→   291→ACTION 1: createOrganization()
   292→   292→─────────────────────────────
   293→   293→Creates a brand new organization AND its first company together.
   294→   294→Called when user wants a completely new business umbrella.
   295→   295→
   296→   296→  Input:
   297→   297→    org_name: string
   298→   298→    org_logo_file?: File (optional)
   299→   299→    company_name: string
   300→   300→    company_legal_name?: string
   301→   301→    company_logo_file?: File (optional)
   302→   302→    base_currency: string  (e.g. "PKR")
   303→   303→    country_code: string   (e.g. "PK")
   304→   304→    province?: string
   305→   305→    city?: string
   306→   306→    address?: string
   307→   307→    phone?: string
   308→   308→    email?: string
   309→   309→    website?: string
   310→   310→    ntn?: string
   311→   311→    strn?: string
   312→   312→    timezone: string       (default "Asia/Karachi")
   313→   313→    fiscal_year_start: number (1-12, default 1)
   314→   314→
   315→   315→  Server Action steps:
   316→   316→    1. Validate all inputs with Zod
   317→   317→    2. If org_logo_file provided:
   318→   318→       Upload to Supabase Storage:
   319→   319→       'organizations/{user_id}/{timestamp}-logo.jpg'
   320→   320→       Get public URL
   321→   321→    3. Generate unique slug from org_name
   322→   322→       (slugify + check uniqueness, append random suffix if taken)
   323→   323→    4. INSERT organizations row:
   324→   324→       { name, slug, logo_url, owner_id: current_user_id }
   325→   325→    5. If company_logo_file provided:
   326→   326→       Upload to Supabase Storage:
   327→   327→       'companies/{new_org_id}/{timestamp}-logo.jpg'
   328→   328→    6. Generate unique slug from company_name
   329→   329→    7. INSERT companies row:
   330→   330→       { organization_id: new_org.id, name, legal_name, logo_url,
   331→   331→         base_currency, country_code, province, city, address,
   332→   332→         phone, email, website, tax_id: ntn, tax_id_type: 'NTN',
   333→   333→         timezone, fiscal_year_start, created_by: current_user_id }
   334→   334→    8. INSERT 4 system roles for this company:
   335→   335→       - { name:'Owner', system_role_key:'owner', role_tier:'elevated', is_system_role:true }
   336→   336→       - { name:'Founder', system_role_key:'founder', role_tier:'elevated', is_system_role:true }
   337→   337→       - { name:'Co-Founder', system_role_key:'co_founder', role_tier:'elevated', is_system_role:true }
   338→   338→       - { name:'Investor', system_role_key:'investor', role_tier:'elevated', is_system_role:true }
   339→   339→    9. INSERT employees row:
   340→   340→       { company_id: new_company.id, user_id: current_user_id,
   341→   341→         role_id: owner_role.id, status: 'active' }
   342→   342→    10. UPDATE user_settings:
   343→   343→        { active_company_id: new_company.id,
   344→   344→          active_org_id: new_org.id }
   345→   345→    11. If STRN also provided: store separately in company metadata
   346→   346→        or as second tax record (metadata JSONB field)
   347→   347→    12. INSERT audit_log: action='organization.created'
   348→   348→    13. INSERT audit_log: action='company.created'
   349→   349→    14. Return { success: true, org_id, company_id }
