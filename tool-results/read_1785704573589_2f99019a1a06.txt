   250→function toISODate(d: Date): string {
   251→  const y = d.getFullYear()
   252→  const m = String(d.getMonth() + 1).padStart(2, '0')
   253→  const day = String(d.getDate()).padStart(2, '0')
   254→  return `${y}-${m}-${day}`
   255→}
   256→
   257→function resolveTimeSpan(value: TimeSpanValue, custom?: DateRange): DateRange {
   258→  const now = new Date()
   259→  const today = new Date(now.getFullYear(), now.getMonth(), now.getDate())
   260→
   261→  switch (value) {
   262→    case 'today':
   263→      return { from: toISODate(today), to: toISODate(today) }
   264→    case 'yesterday': {
   265→      const y = new Date(today)
   266→      y.setDate(y.getDate() - 1)
   267→      return { from: toISODate(y), to: toISODate(y) }
   268→    }
   269→    case 'last_7_days': {
   270→      const from = new Date(today)
   271→      from.setDate(from.getDate() - 6)
   272→      return { from: toISODate(from), to: toISODate(today) }
   273→    }
   274→    case 'last_30_days': {
   275→      const from = new Date(today)
   276→      from.setDate(from.getDate() - 29)
   277→      return { from: toISODate(from), to: toISODate(today) }
   278→    }
   279→    case 'this_month': {
   280→      const from = new Date(now.getFullYear(), now.getMonth(), 1)
   281→      return { from: toISODate(from), to: toISODate(today) }
   282→    }
   283→    case 'last_month': {
   284→      const firstOfThis = new Date(now.getFullYear(), now.getMonth(), 1)
   285→      const from = new Date(firstOfThis)
   286→      from.setMonth(from.getMonth() - 1)
   287→      const to = new Date(firstOfThis.getFullYear(), firstOfThis.getMonth(), 0)
   288→      return { from: toISODate(from), to: toISODate(to) }
   289→    }
   290→    case 'this_year': {
   291→      const from = new Date(now.getFullYear(), 0, 1)
   292→      return { from: toISODate(from), to: toISODate(today) }
   293→    }
   294→    case 'custom':
   295→      return {
   296→        from: custom?.from ?? toISODate(today),
   297→        to: custom?.to ?? toISODate(today),
   298→      }
   299→  }
   300→}

   302→function spanDayCount(range: DateRange): number {
   303→  const from = new Date(range.from + 'T00:00:00')
   304→  const to = new Date(range.to + 'T23:59:59')
   305→  const diff = to.getTime() - from.getTime()
   306→  return Math.max(1, Math.round(diff / (1000 * 60 * 60 * 24)) + 1)
   307→}
   308→
   309→function isMonthlyGranularity(range: DateRange): boolean {
   310→  return spanDayCount(range) > 30
   311→}

   421→function toggleArray(arr: string[], value: string): string[] {
   422→  return arr.includes(value) ? arr.filter((v) => v !== value) : [...arr, value]
   423→}

   425→function countActiveFilters(f: FilterState): number {
   426→  let n = 0
   427→  if (f.statuses.length) n++
   428→  if (f.paymentTypes.length) n++
   429→  if (f.paymentStatuses.length) n++
   430→  if (f.orderSources.length) n++
   431→  if (f.dateFrom || f.dateTo) n++
   432→  if (f.amountMin || f.amountMax) n++
   433→  if (f.orgVariantId) n++
   434→  if (f.customerId) n++
   435→  if (f.deliveryCity.trim()) n++
   436→  if (f.courier.trim()) n++
   437→  return n
   438→}

   453→function applyPreset(preset: DatePresetValue): { from: string; to: string } {
   454→  const now = new Date()
   455→  const to = new Date(now)
   456→  const from = new Date(now)
   457→  if (preset === 'today') {
   458→    // from = today
   459→  } else if (preset === '7d') {
   460→    from.setDate(now.getDate() - 6)
   461→  } else if (preset === '30d') {
   462→    from.setDate(now.getDate() - 29)
   463→  } else if (preset === 'this_month') {
   464→    from.setDate(1)
   465→  }
   466→  return { from: toISODate(from), to: toISODate(to) }
   467→}

  1135→function buildChartData(
  1136→  orders: OrderRow[],
  1137→  range: DateRange,
  1138→  monthly: boolean,
  1139→  metric: ChartMetric,
  1140→): ChartPoint[] {
  1141→  // Build the list of interval buckets
  1142→  const buckets = buildIntervalBuckets(range, monthly)
  1143→
  1144→  // For each order, compute its bucket key (yyyy-mm-dd or yyyy-mm)
  1145→  for (const o of orders) {
  1146→    const created = new Date(o.createdAt)
  1147→    if (Number.isNaN(created.getTime())) continue
  1148→    const key = monthly
  1149→      ? `${created.getFullYear()}-${String(created.getMonth() + 1).padStart(2, '0')}`
  1150→      : toISODate(created)
  1151→    const b = buckets.find((x) => x.key === key)
  1152→    if (!b) continue
  1153→    b.total += 1
  1154→    if (o.status === 'rto') b.rto += 1
  1155→    if (o.status === 'cancelled') b.cancelled += 1
  1156→    if (o.status !== 'cancelled') b.revenue += o.totalOrderValue
  1157→    if (o.status === 'pending') b.pending += 1
  1158→    if (o.status === 'partially_backordered') b.backordered += 1
  1159→  }
  1160→
  1161→  return buckets.map((b) => {
  1162→    let value = 0
  1163→    let count: number | undefined
  1164→    let total: number | undefined
  1165→    switch (metric) {
  1166→      case 'total_orders':
  1167→        value = b.total
  1168→        break
  1169→      case 'revenue':
  1170→        value = b.revenue
  1171→        break
  1172→      case 'rto_rate':
  1173→        value = b.total > 0 ? (b.rto / b.total) * 100 : 0
  1174→        count = b.rto
  1175→        total = b.total
  1176→        break
  1177→      case 'cancellation_rate':
  1178→        value = b.total > 0 ? (b.cancelled / b.total) * 100 : 0
  1179→        count = b.cancelled
  1180→        total = b.total
  1181→        break
  1182→      case 'pending_current':
  1183→        value = b.pending
  1184→        break
  1185→      case 'backordered_current':
  1186→        value = b.backordered
  1187→        break
  1188→    }
  1189→    return { label: b.label, value, count, total }
  1190→  })
  1191→}
  1192→
  1193→interface Bucket {
  1194→  key: string
  1195→  label: string
  1196→  total: number
  1197→  revenue: number
  1198→  rto: number
  1199→  cancelled: number
  1200→  pending: number
  1201→  backordered: number
  1202→}

  1204→function buildIntervalBuckets(range: DateRange, monthly: boolean): Bucket[] {
  1205→  const out: Bucket[] = []
  1206→  const start = new Date(range.from + 'T00:00:00')
  1207→  const end = new Date(range.to + 'T23:59:59')
  1208→
  1209→  if (monthly) {
  1210→    // Iterate by month from start month to end month (inclusive)
  1211→    const cursor = new Date(start.getFullYear(), start.getMonth(), 1)
  1212→    const endMonth = new Date(end.getFullYear(), end.getMonth(), 1)
  1213→    while (cursor <= endMonth) {
  1214→      const key = `${cursor.getFullYear()}-${String(cursor.getMonth() + 1).padStart(2, '0')}`
  1215→      const label = cursor.toLocaleDateString('en-PK', { month: 'short', year: '2-digit' })
  1216→      out.push({ key, label, total: 0, revenue: 0, rto: 0, cancelled: 0, pending: 0, backordered: 0 })
  1217→      cursor.setMonth(cursor.getMonth() + 1)
  1218→    }
  1219→  } else {
  1220→    // Iterate by day from start to end (inclusive)
  1221→    const cursor = new Date(start.getFullYear(), start.getMonth(), start.getDate())
  1222→    const endDay = new Date(end.getFullYear(), end.getMonth(), end.getDate())
  1223→    while (cursor <= endDay) {
  1224→      const key = toISODate(cursor)
  1225→      const label = cursor.toLocaleDateString('en-PK', { day: '2-digit', month: 'short' })
  1226→      out.push({ key, label, total: 0, revenue: 0, rto: 0, cancelled: 0, pending: 0, backordered: 0 })
  1227→      cursor.setDate(cursor.getDate() + 1)
  1228→    }
  1229→  }
  1230→  return out
  1231→}
