   157→function getErrorMessage(err: unknown): string {
   158→  if (err instanceof FetchError) return err.message
   159→  if (err instanceof Error) return err.message
   160→  return 'Something went wrong. Please try again.'

   163→function parseQty(v: string): number {
   164→  if (v === '') return 0
   165→  const n = Number(v)
   166→  return Number.isFinite(n) && n >= 0 ? Math.floor(n) : 0
   167→}
   168→
   169→function parsePrice(v: string): number {
   170→  if (v === '') return 0
   171→  const n = Number(v)
   172→  return Number.isFinite(n) && n >= 0 ? n : 0
   173→}

   185→function stockBadgeFor(
   186→  fulfillmentType: string,
   187→): { label: string; className: string } {
   188→  if (fulfillmentType === 'made_to_order') {
   189→    return {
   190→      label: 'Made to Order',
   191→      className: 'bg-purple-50 text-purple-700 border-purple-200',
   192→    }
   193→  }
   194→  if (fulfillmentType === 'backorder') {
   195→    return {
   196→      label: 'Backorder Available',
   197→      className: 'bg-orange-50 text-orange-700 border-orange-200',
   198→    }
   199→  }
   200→  return {
   201→    label: 'In Stock',
   202→    className: 'bg-emerald-50 text-emerald-700 border-emerald-200',
   203→  }

   804→      order_detail: orderDetailUserEdited ? (orderDetail.trim() || undefined) : undefined,

   808→      tax_amount: tax > 0 ? tax : undefined,

   815→      payload.advance_payment_method = advancePaymentMethod || undefined

   830→      payload.recipient_name = recipientName.trim() || undefined

   832→    }
   833→

   835→  }

   851→  // ── Upload payment proof + persist URL to the order ───────────────────────
   852→  // Returns true on success, false on any failure. NEVER throws.
   853→  async function uploadPaymentProof(orderId: string): Promise<boolean> {
   854→    if (!paymentProofFile) return true
   855→
   856→    setUploadingProof(true)
   857→    try {
   858→      const fd = new FormData()
   859→      fd.append('file', paymentProofFile)
   860→      const uploadRes = await fetch(
   861→        `/api/upload?type=payment-proofs&id=${orderId}`,
   862→        { method: 'POST', body: fd, credentials: 'include', headers: getSessionToken() ? { Authorization: `Bearer ${getSessionToken()}` } : {} },
   863→      )
   864→      const uploadText = await uploadRes.text()
   865→      let uploadBody: unknown = null
   866→      if (uploadText) {
   867→        try {
   868→          uploadBody = JSON.parse(uploadText)
   869→        } catch {
   870→          uploadBody = uploadText
   871→        }
   872→      }
   873→      if (!uploadRes.ok) {
   874→        const msg =
   875→          uploadBody && typeof uploadBody === 'object' && 'error' in uploadBody
   876→            ? String((uploadBody as { error: unknown }).error)
   877→            : typeof uploadBody === 'string'
   878→              ? uploadBody
   879→              : `Upload failed (HTTP ${uploadRes.status})`
   880→        throw new Error(msg)
   881→      }
   882→      const { url } = uploadBody as { url: string }
   883→
   884→      await api.post(`/api/orders/${orderId}/payment-proof`, {
   885→        advance_payment_screenshot_url: url,
   886→      })
   887→      return true
   888→    } catch {
   889→      return false
   890→    } finally {
   891→      setUploadingProof(false)
   892→    }
   893→  }

   921→  const handleSubmit = async () => {
   922→    // ── Phase D2: strict country blocking ──────────────────────────────
   923→    // If the delivery country isn't assigned to ANY market, block order
   924→    // creation with the correct message. This applies to MANUAL order
   925→    // creation only (Shopify/webhook orders bypass this).
   926→    if (isCountryBlocked) {
   927→      toast.error(marketContext?.blockReason ?? 'This country isn\'t assigned to any market.')
   928→      customerSectionRef.current?.scrollIntoView({ behavior: 'smooth', block: 'start' })
   929→      return
   930→    }
   931→
   932→    const errs = validate()
   933→    setErrors(errs)
   934→    if (Object.keys(errs).length > 0) {
   935→      toast.error('Please complete all required fields before creating the order.')
   936→      const firstErrPath = Object.keys(errs)[0]
   937→      const section = sectionForErrorPath(firstErrPath)
   938→      const ref =
   939→        section === 'customer'
   940→          ? customerSectionRef.current
   941→          : section === 'items'
   942→            ? itemsSectionRef.current
   943→            : paymentSectionRef.current
   944→      ref?.scrollIntoView({ behavior: 'smooth', block: 'start' })
   945→      return
   946→    }
   947→
   948→    const payload = buildPayload()
   949→
   950→    setSubmittingOrder(true)
   951→    try {
   952→      const data = await api.post<CreateOrderResponse>('/api/orders', payload, {
   953→        'Idempotency-Key': idempotencyKeyRef.current,
   954→      })
   955→
   956→      // ── Order creation is ALWAYS successful at this point ──
   957→      toast.success(`Order ${data.flowopsOrderNumber} created successfully.`)
   958→
   959→      // Auto-booking now runs ASYNCHRONOUSLY in the background (PostEx API
   960→      // can take 50-100s). The order is created with courierBookingStatus=
   961→      // 'not_booked', and the background task updates it when PostEx responds.
   962→      // We navigate to the All Orders page immediately — the user will see
   963→      // the new order in the list without waiting for booking to finish.
   964→      if (data.bookingAttempted) {
   965→        toast.info('Courier booking is in progress… You can track the status from the All Orders page.', { duration: 6000 })
   966→      }
   967→
   968→      setHasChanges(false)
   969→      if (draftId) {
   970→        await api.delete(`/api/drafts?id=${draftId}`).catch(() => {})
   971→        setDraftId(undefined)
   972→      }
   973→      // Invalidate ALL relevant query keys so the All Orders list refetches
   974→      // with the newly created order as soon as we land on that page.
   975→      void queryClient.invalidateQueries({ queryKey: ['orders'] })
   976→      void queryClient.invalidateQueries({ queryKey: ['booking-workbench-bookable'] })
   977→      void queryClient.invalidateQueries({ queryKey: ['booking-workbench-activity'] })
   978→
   979→      // ── Navigate FIRST, then upload the payment proof in the background. ──
   980→      // The order is already created + persisted at this point. Forcing the
   981→      // user to wait for the (potentially slow) image upload before seeing
   982→      // the All Orders page is unnecessary — the upload can finish async.
   983→      // If the upload fails, a warning toast is shown on the orders page.
   984→      navigate({ name: 'orders' })
   985→
   986→      if (paymentProofFile) {
   987→        // Fire-and-forget: don't block navigation. The toast.warning surfaces
   988→        // globally if the upload fails, so the user is still informed.
   989→        void (async () => {
   990→          const ok = await uploadPaymentProof(data.orderId)
   991→          if (!ok) {
   992→            toast.warning(
   993→              'Order created successfully, but the payment proof image failed to upload — you can add it from the order detail page.',
   994→            )
   995→          }
   996→        })()
   997→      }
   998→    } catch (err) {
   999→      toast.error(getErrorMessage(err))
  1000→    } finally {
  1001→      setSubmittingOrder(false)
  1002→    }
  1003→  }

  1005→  // ── Phase 5: Retry booking handler (uses state declared at top of component) ──
  1006→  async function handleRetryBooking() {
  1007→    if (!bookingFailure) return
  1008→    setRetryingBooking(true)
  1009→    try {
  1010→      const result = await api.post<{ success: boolean; trackingNumber?: string; error?: string }>(
  1011→        '/api/booking-workbench/book',
  1012→        {
  1013→          orderId: bookingFailure.orderId,
  1014→          companyIntegrationId: courierIntegrationId,
  1015→        },
  1016→      )
  1017→      if (result.success && result.trackingNumber) {
  1018→        toast.success(`Booking successful! Tracking #: ${result.trackingNumber}`)
  1019→        setBookingFailure(null)
  1020→        void queryClient.invalidateQueries({ queryKey: ['orders'] })
  1021→        void queryClient.invalidateQueries({ queryKey: ['booking-workbench-bookable'] })
  1022→        navigate({ name: 'order-detail', id: bookingFailure.orderId })
  1023→      } else {
  1024→        toast.error(`Retry failed: ${result.error ?? 'Unknown error'}`)
  1025→      }
  1026→    } catch (err) {
  1027→      toast.error(getErrorMessage(err))
  1028→    } finally {
  1029→      setRetryingBooking(false)
  1030→    }
  1031→  }

  1033→  // Helper: which section does an error path belong to?

  1675→/** Compact relative-time formatter for the inline address list. */
  1676→function formatLastUsedShort(iso: string | null): string {
  1677→  if (!iso) return 'Never used'
  1678→  const date = new Date(iso)
  1679→  const now = new Date()
  1680→  const diffMs = now.getTime() - date.getTime()
  1681→  const diffDays = Math.floor(diffMs / 86_400_000)
  1682→  if (diffDays < 1) return 'Today'
  1683→  if (diffDays === 1) return '1 day ago'
  1684→  if (diffDays < 30) return `${diffDays} days ago`
  1685→  return date.toLocaleDateString('en-PK', { day: '2-digit', month: 'short' })
  1686→}
