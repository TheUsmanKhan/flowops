   190→export default function Page() {
   191→  const { hydrated, route, user, activeCompany, employee, setSession, setHydrated, navigate } =
   192→    useAppStore()
   193→
   194→  // Track whether this is the first session fetch (cold start) vs a background
   195→  // refetch (tab refocus, reconnect, invalidation). On the first fetch we show
   196→  // a loading spinner; on background refetches we update the store silently.
   197→  const isFirstFetchRef = useRef(true)
   198→
   199→  // ─── Session hydration via TanStack Query ──────────────────────────
   200→  // This query intentionally overrides the app's global refetchOnWindowFocus:
   201→  // false default. Session validity (active employee status, permissions,
   202→  // platform-level access) is the one place where catching a change quickly
   203→  // after the user returns to a background tab actually matters — e.g., an
   204→  // employee terminated while their tab sat in the background should have
   205→  // their UI reflect that promptly on refocus, not only on their next manual
   206→  // page refresh. Every other view in this app is correctly left on the
   207→  // global default because those views display data, not gate
   208→  // security-sensitive UI. Do not "clean up" this override to match the
   209→  // global default in future refactors.
   210→  const sessionQuery = useQuery<SessionResponse>({
   211→    queryKey: ['session', 'me'],
   212→    queryFn: () => api.get<SessionResponse>('/api/auth/me'),
   213→    staleTime: 60_000,
   214→    refetchOnWindowFocus: true,
   215→    refetchOnReconnect: true,
   216→    retry: 1,
   217→  })
   218→
   219→  // Wire session data into the Zustand store.
   220→  useEffect(() => {
   221→    const session = sessionQuery.data
   222→    if (!session) return
   223→
   224→    // On the very first successful fetch, restore the URL route + set hydrated.
   225→    if (isFirstFetchRef.current) {
   226→      isFirstFetchRef.current = false
   227→      const urlRoute = queryToRoute()
   228→      const currentRoute = useAppStore.getState().route
   229→
   230→      setSession({
   231→        user: session.user,
   232→        activeCompany: session.activeCompany,
   233→        companies: session.companies,
   234→        employee: session.employee ?? undefined,
   235→      })
   236→
   237→      // If the URL has a route, restore it (overrides the default 'login')
   238→      if (urlRoute && urlRoute.name !== currentRoute.name) {
   239→        if (session.user) {
   240→          useAppStore.getState().navigate(urlRoute)
   241→        }
   242→      }
   243→    } else {
   244→      // Background refetch: update the store ONLY if the session data
   245→      // actually changed, so the UI updates without a visible loading flicker.
   246→      const currentUser = useAppStore.getState().user
   247→      const currentActiveCompanyId = useAppStore.getState().activeCompany?.id
   248→      const newActiveCompanyId = session.activeCompany?.id
   249→      const newPermissions = session.employee?.permissions ?? []
   250→      const currentPermissions = useAppStore.getState().employee?.permissions ?? []
   251→
   252→      const userChanged = currentUser?.id !== session.user?.id
   253→      const companyChanged = currentActiveCompanyId !== newActiveCompanyId
   254→      const permissionsChanged =
   255→        newPermissions.length !== currentPermissions.length ||
   256→        newPermissions.some((p, i) => p !== currentPermissions[i])
   257→
   258→      if (userChanged || companyChanged || permissionsChanged || !currentUser) {
   259→        setSession({
   260→          user: session.user,
   261→          activeCompany: session.activeCompany,
   262→          companies: session.companies,
   263→          employee: session.employee ?? undefined,
   264→        })
   265→      }
   266→    }
   267→  }, [sessionQuery.data, setSession])
   268→
   269→  // On query error (401, network failure), mark hydrated so the login screen shows.
   270→  useEffect(() => {
   271→    if (sessionQuery.error && !hydrated) {
   272→      setHydrated(true)
   273→    }
   274→  }, [sessionQuery.error, hydrated, setHydrated])
   275→
   276→  // Mark hydrated on first successful fetch (if not already set by setSession).
   277→  useEffect(() => {
   278→    if (sessionQuery.isSuccess && !hydrated) {
   279→      setHydrated(true)
   280→    }
   281→  }, [sessionQuery.isSuccess, hydrated, setHydrated])
   282→
   283→  // Browser back/forward: sync URL → Zustand state
   284→  // NOTE: The useFormGuard's use-browser-back-guard also listens to popstate.
   285→  // When the guard is active (form is dirty), it re-pushes the URL to cancel
   286→  // the back navigation. We need to make sure our handler doesn't fight it.
   287→  // We use a global flag that the guard sets when it's intercepting.
   288→  useEffect(() => {
   289→    function handlePopState() {
   290→      // If the form guard is intercepting, skip our sync — the guard will
   291→      // handle the modal, and if the user confirms, the guard calls
   292→      // window.history.back() which will trigger another popstate that we'll handle.
   293→      if (typeof window !== 'undefined' && window.__formGuardIntercepting) {
   294→        return
   295→      }
   296→      const urlRoute = queryToRoute()
   297→      if (urlRoute) {
   298→        // Use set() directly, NOT navigate(), to avoid pushing a new history entry
   299→        useAppStore.setState({ route: urlRoute })
   300→      }
   301→    }
   302→    window.addEventListener('popstate', handlePopState)
   303→    return () => window.removeEventListener('popstate', handlePopState)
   304→  }, [])
   305→
   306→  // URL sync: ensure the address bar always reflects the current route
   307→  // (covers the case where user lands on "/" with no query string)
   308→  // Must be before any early returns to satisfy React Hooks rules.
   309→  useEffect(() => {
   310→    if (hydrated) {
   311→      replaceRouteInURL(route)
   312→    }
   313→     
   314→  }, [hydrated, route.name, 'id' in route ? route.id : '', 'token' in route ? route.token : ''])
   315→
   316→  // Loading screen while hydrating (first session fetch only).
   317→  // On background refetches, sessionQuery.isFetching is true but we DON'T show
   318→  // a spinner — TanStack Query returns cached data instantly while revalidating.
   319→  if (!hydrated) {
   320→    return (
   321→      <div className="min-h-screen flex items-center justify-center bg-background">
   322→        <Loader2 className="h-6 w-6 animate-spin text-primary" />
   323→      </div>
   324→    )
   325→  }
   326→
   327→  // ---- Unauthenticated views ----
   328→  if (!user) {
   329→    if (route.name === 'register') {
   330→      return (
   331→        <AuthShell
   332→          title="Create your account"
   333→          subtitle="Start managing your e-commerce operations in minutes."
   334→        >
   335→          <RegisterForm />
   336→        </AuthShell>
   337→      )
   338→    }
   339→    if (route.name === 'forgot') {
   340→      return (
   341→        <AuthShell
   342→          title="Reset your password"
   343→          subtitle="We'll email you a secure recovery link."
   344→        >
   345→          <ForgotPasswordForm />
   346→        </AuthShell>
   347→      )
   348→    }
   349→    if (route.name === 'reset') {
   350→      return (
   351→        <AuthShell
   352→          title="Set a new password"
   353→          subtitle="Choose a strong password for your account."
   354→        >
   355→          <ResetPasswordForm token={route.token} />
   356→        </AuthShell>
   357→      )
   358→    }
   359→    // default: login
   360→    return (
   361→      <AuthShell
   362→        title="Welcome back"
   363→        subtitle="Sign in to your FlowOps workspace."
   364→      >
   365→        <LoginForm />
   366→      </AuthShell>
   367→    )
   368→  }
   369→
   370→  // ---- Authenticated but not onboarded ----
   371→  // Allow create-organization even before onboarding completes.
   372→  if (user && route.name === 'create-organization') {
   373→    return (
   374→      <DashboardShell>
   375→        <CreateOrganizationView onBack={() => navigate({ name: user.isOnboarded && activeCompany ? 'dashboard' : 'onboarding' })} />
   376→      </DashboardShell>
   377→    )
   378→  }
   379→  if (!user.isOnboarded || !activeCompany) {
   380→    if (route.name === 'create-company') {
   381→      return (
   382→        <DashboardShell>
   383→          <CreateCompanyView onBack={() => navigate({ name: 'onboarding' })} />
   384→        </DashboardShell>
   385→      )
   386→    }
   387→    return <OnboardingView />
   388→  }
   389→
   390→  // ---- Authenticated + onboarded: dashboard shell ----
   391→  return <DashboardShell>{renderRoute(route, employee)}</DashboardShell>
   392→}
