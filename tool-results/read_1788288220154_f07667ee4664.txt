   208→/// Company-scoped roles. System roles (Owner/Founder/Co-Founder/Investor) are
   209→/// elevated and bypass all permission checks. Custom roles use role_permissions.
   210→model Role {
   211→  id            String   @id @default(cuid())
   212→  companyId     String
   213→  company       Company  @relation(fields: [companyId], references: [id], onDelete: Cascade)
   214→  name          String
   215→  description   String?
   216→  roleTier      String   @default("standard") // elevated | standard
   217→  isSystemRole  Boolean  @default(false)
   218→  systemRoleKey String?  // owner | founder | co_founder | investor | null
   219→  // NOTE: systemRoleKey is unique PER COMPANY (not globally), so every
   220→  // company can seed its own owner/founder/co_founder/investor roles.
   221→  // ─── Data scope for the Orders module ───────────────────────────────
   222→  // "all"  → role sees ALL orders in the company (default, preserves
   223→  //          existing behavior for every row after migration)
   224→  // "own"  → role sees only orders where salesEmployeeId === the caller's
   225→  //          own Employee.id (enforced at the query layer, not in SQL)
   226→  ordersDataScope String  @default("all") // "own" | "all"
   227→  isActive      Boolean  @default(true)
   228→  createdById   String?
   229→  createdBy     Profile? @relation("RoleCreator", fields: [createdById], references: [id])
   230→  createdAt     DateTime @default(now())
   231→  updatedAt     DateTime @updatedAt
   232→
   233→  employees       Employee[]
   234→  rolePermissions RolePermission[]
   235→  invitations     Invitation[]
   236→
   237→  @@unique([companyId, name])
   238→  @@unique([companyId, systemRoleKey])
   239→}
