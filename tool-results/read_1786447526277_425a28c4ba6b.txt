    91→// ──────────────────────────────────────────────────────────────
    92→// Processes ALL order_items for an order and attempts to reserve
    93→// stock for each based on its fulfillment_type_snapshot.
    94→//
    95→// For stock_based items:
    96→//   - If sufficient available stock → reserveStockForOrder(), fulfillment_status='reserved'
    97→//   - If insufficient + inventory_policy='continue' → fulfillment_status='backordered'
    98→//   - If insufficient + inventory_policy='deny' → outcome='failed'
    99→//
   100→// For made_to_order items:
   101→//   - checkAndFulfillMadeToOrderVariant() checks returned stock first
   102→//   - If existing_stock → fulfillment_status='reserved', returned_stitched_used=true
   103→//   - If fresh_production → fulfillment_status='reserved', production_order_id set
   104→//   - If error → outcome='failed'
   105→//
   106→// After processing: if ANY item is 'backordered', order status → 'partially_backordered'.
   107→// If ALL items are 'reserved', order stays 'confirmed'.
   108→//
   109→// Called from:
   110→//   - createManualOrder() IF the order auto-confirmed at creation
   111→//   - confirmOrder() when a 'pending' order is manually confirmed
   112→//   - dispatchOrderAction() if status is still 'pending' at dispatch time
   113→
   114→async function reserveOrderStock(
   115→  orderId: string,
   116→  ctx: { company: { id: string; organizationId: string }; user: { id: string }; employee: { id: string } },
   117→): Promise<{
   118→  success: boolean
   119→  results: Array<{ orderItemId: string; outcome: 'reserved' | 'backordered' | 'failed'; reason?: string }>
   120→}> {
   121→  const order = await db.order.findUnique({
   122→    where: { id: orderId },
   123→    select: { dispatchLocationId: true, companyId: true, organizationId: true },
   124→  })
   125→  if (!order) return { success: false, results: [] }
   126→
   127→  const items = await db.orderItem.findMany({
   128→    where: { orderId },
   129→    include: {
   130→      orgVariant: {
   131→        select: { id: true, fulfillmentType: true, inventoryPolicy: true, stitchingCharges: true },
   132→      },
   133→    },
   134→  })
   135→
   136→  const results: Array<{ orderItemId: string; outcome: 'reserved' | 'backordered' | 'failed'; reason?: string }> = []
   137→  let hasBackordered = false
   138→
   139→  for (const item of items) {
   140→    if (item.fulfillmentStatus === 'reserved' || item.fulfillmentStatus === 'dispatched') {
   141→      // Already processed (e.g. re-confirmation attempt) — skip
   142→      results.push({ orderItemId: item.id, outcome: 'reserved' })
   143→      continue
   144→    }
   145→
   146→    const locationId = item.reservedLocationId ?? order.dispatchLocationId
   147→    if (!locationId) {
   148→      results.push({ orderItemId: item.id, outcome: 'failed', reason: 'No dispatch location set' })
   149→      continue
   150→    }
   151→
   152→    if (item.fulfillmentTypeSnapshot === 'stock_based') {
   153→      // Check available stock at the dispatch location
   154→      const pool = await db.inventoryPool.findUnique({
   155→        where: {
   156→          orgVariantId_locationId: {
   157→            orgVariantId: item.orgVariantId,
   158→            locationId,
   159→          },
   160→        },
   161→      })
   162→
   163→      const available = pool ? pool.onHand - pool.reserved : 0
   164→
   165→      if (available >= item.quantity) {
   166→        // Sufficient stock — reserve it
   167→        const reserveResult = await reserveStockForOrder({
   168→          orgVariantId: item.orgVariantId,
   169→          locationId,
   170→          organizationId: order.organizationId,
   171→          companyId: order.companyId,
   172→          employeeId: ctx.employee.id,
   173→          quantity: item.quantity,
   174→          orderId,
   175→        })
   176→
   177→        if (reserveResult.success) {
   178→          await db.orderItem.update({
   179→            where: { id: item.id },
   180→            data: { fulfillmentStatus: 'reserved', reservedLocationId: locationId },
   181→          })
   182→          results.push({ orderItemId: item.id, outcome: 'reserved' })
   183→        } else {
   184→          results.push({ orderItemId: item.id, outcome: 'failed', reason: reserveResult.error })
   185→        }
   186→      } else if (item.orgVariant.inventoryPolicy === 'continue') {
   187→        // Insufficient stock but backordering allowed
   188→        await db.orderItem.update({
   189→          where: { id: item.id },
   190→          data: { fulfillmentStatus: 'backordered', backorderedAt: new Date() },
   191→        })
   192→        insertMetricEvent({
   193→          companyId: order.companyId,
   194→          entityType: 'product',
   195→          entityId: item.orgVariantId,
   196→          metricKey: 'order.backordered',
   197→          numericValue: item.quantity,
   198→          dimensions: { order_id: orderId },
   199→        })
   200→        hasBackordered = true
   201→        results.push({
   202→          orderItemId: item.id,
   203→          outcome: 'backordered',
   204→          reason: `Available: ${available}, required: ${item.quantity}`,
   205→        })
   206→      } else {
   207→        // Insufficient stock + policy='deny' — fail
   208→        results.push({
   209→          orderItemId: item.id,
   210→          outcome: 'failed',
   211→          reason: `Insufficient stock (available: ${available}, required: ${item.quantity}) and inventory_policy='deny'`,
   212→        })
   213→      }
   214→    } else if (item.fulfillmentTypeSnapshot === 'made_to_order') {
   215→      // Made-to-order: check returned stock first, then trigger production
   216→      const mtoResult = await checkAndFulfillMadeToOrderVariant(
   217→        item.orgVariantId,
   218→        item.quantity,
   219→        order.companyId,
   220→        locationId,
   221→      )
   222→
   223→      if (mtoResult.source === 'existing_stock' && mtoResult.locationId) {
   224→        // Returned stock available — reserve it
   225→        const reserveResult = await reserveStockForOrder({
   226→          orgVariantId: item.orgVariantId,
   227→          locationId: mtoResult.locationId,
   228→          organizationId: order.organizationId,
   229→          companyId: order.companyId,
   230→          employeeId: ctx.employee.id,
   231→          quantity: item.quantity,
   232→          orderId,
   233→        })
   234→
   235→        if (reserveResult.success) {
   236→          await db.orderItem.update({
   237→            where: { id: item.id },
   238→            data: {
   239→              fulfillmentStatus: 'reserved',
   240→              returnedStitchedUsed: true,
   241→              reservedLocationId: mtoResult.locationId,
   242→            },
   243→          })
   244→          insertMetricEvent({
   245→            companyId: order.companyId,
   246→            entityType: 'product',
   247→            entityId: item.orgVariantId,
   248→            metricKey: 'order.made_to_order_from_returned_stock',
   249→            numericValue: 1,
   250→            dimensions: { order_id: orderId, stitching_cost_saved: Number(item.orgVariant.stitchingCharges) || 0 },
   251→          })
   252→          results.push({ orderItemId: item.id, outcome: 'reserved' })
   253→        } else {
   254→          results.push({ orderItemId: item.id, outcome: 'failed', reason: reserveResult.error })
   255→        }
   256→      } else if (mtoResult.source === 'fresh_production' && mtoResult.productionOrderId) {
   257→        // Fresh production triggered — link the production order
   258→        await db.orderItem.update({
   259→          where: { id: item.id },
   260→          data: {
   261→            fulfillmentStatus: 'reserved',
   262→            productionOrderId: mtoResult.productionOrderId,
   263→          },
   264→        })
   265→        // Also link the order_item back to the production order
   266→        await db.productionOrder.update({
   267→          where: { id: mtoResult.productionOrderId },
   268→          data: { orderItemId: item.id },
   269→        })
   270→        insertMetricEvent({
   271→          companyId: order.companyId,
   272→          entityType: 'product',
   273→          entityId: item.orgVariantId,
   274→          metricKey: 'order.made_to_order_production_triggered',
   275→          numericValue: 1,
   276→          dimensions: { order_id: orderId, production_order_id: mtoResult.productionOrderId },
   277→        })
   278→        results.push({ orderItemId: item.id, outcome: 'reserved' })
   279→      } else {
   280→        // Error (no fabric source, insufficient fabric, etc.)
   281→        results.push({
   282→          orderItemId: item.id,
   283→          outcome: 'failed',
   284→          reason: mtoResult.error ?? 'Made-to-order fulfillment failed',
   285→        })
   286→      }
   287→    }
   288→  }
   289→
   290→  // Update order status based on results
   291→  if (hasBackordered) {
   292→    await db.order.update({
   293→      where: { id: orderId },
   294→      data: { status: 'partially_backordered' },
   295→    })
   296→  }
   297→  // If no backordered items, order stays 'confirmed' (or whatever it was)
   298→
   299→  return { success: true, results }
   300→}

   302→// ──────────────────────────────────────────────────────────────

   305→
   306→export async function createManualOrder(
   307→  input: CreateManualOrderInput,
   308→): Promise<ActionResult<{
   309→  orderId: string
   310→  flowopsOrderNumber: string
   311→  orderItems: Array<{ id: string; orgVariantId: string; quantity: number }>
   312→  /** Auto-booking is now ASYNC — bookingAttempted=true means it's running in the background. */
   313→  bookingAttempted: boolean
   314→  bookingSucceeded?: boolean
   315→  bookingError?: string
   316→  bookingTrackingNumber?: string
   317→}>> {
   318→  try {
   319→    const ctx = await getWorkspace()
   320→    await requirePermission(ctx, PERMISSIONS.ORDERS_CREATE)
   321→
   322→    // 1. Validate
   323→    const parsed = createManualOrderSchema.safeParse(input)
   324→    if (!parsed.success) {
   325→      return {
   326→        success: false,
   327→        error: parsed.error.issues[0]?.message ?? 'Invalid order data',
   328→      }
   329→    }
   330→    const d = parsed.data
   331→
   332→    // ──────────────────────────────────────────────────────────────
   333→    // 2. PARALLEL GROUP 1 — all independent pre-create reads.
   334→    //    Previously these ran SEQUENTIALLY (customer → variants → settings →
   335→    //    order-number → …), costing ~4× the per-query network latency.
   336→    //    Now they run in a single Promise.all() batch. None of these depend
   337→    //    on each other's results:
   338→    //      - customer resolution: depends on ctx + d only
   339→    //      - variant fetch: depends on ctx.company.organizationId + d.items
   340→    //      - companyOrderSetting: depends on ctx.company.id (fetched ONCE here
   341→    //        — previously fetched TWICE, once for requireOrderConfirmation
   342→    //        and again for courierBookingMode)
   343→    //      - generateOrderNumber: SQL function, depends on ctx.company.id only
   344→    // ──────────────────────────────────────────────────────────────
   345→    const variantIds = d.items.map((i) => i.org_variant_id)
   346→    const [custResult, variants, orderSettings, flowopsOrderNumber] = await Promise.all([
   347→      // ── Customer resolution (existing lookup OR inline new-customer creation) ──
   348→      (async () => {
   349→        let customerId: string
   350→        let customerName: string
   351→        let usedCustomerAddressId: string | null = null
   352→        let usedCustomerPhoneId: string | null = null
   353→        let saveAddressForNextTime = d.save_address_for_next_time ?? false
   354→        let selectedSavedAddressId: string | null = null
   355→
   356→        if (d.customer_id) {
   357→          // Existing customer path — verify they belong to this org.
   358→          const existing = await db.customer.findFirst({
   359→            where: { id: d.customer_id, organizationId: ctx.company.organizationId },
   360→            select: { id: true, name: true },
   361→          })
   362→          if (!existing) return { ok: false as const, error: 'Customer not found' }
   363→          customerId = existing.id
   364→          customerName = existing.name
   365→
   366→          // If BOTH a saved address AND a saved phone were selected, verify
   367→          // them in PARALLEL (independent lookups). Otherwise do whichever
   368→          // is present.
   369→          if (d.used_customer_address_id && d.used_customer_phone_id) {
   370→            const [addr, phone] = await Promise.all([
   371→              db.customerAddress.findFirst({
   372→                where: { id: d.used_customer_address_id, customerId: existing.id, organizationId: ctx.company.organizationId },
   373→                select: { id: true },
   374→              }),
   375→              db.customerPhone.findFirst({
   376→                where: { id: d.used_customer_phone_id, customerId: existing.id, organizationId: ctx.company.organizationId },
   377→                select: { id: true },
   378→              }),
   379→            ])
   380→            if (addr) {
   381→              usedCustomerAddressId = addr.id
   382→              selectedSavedAddressId = addr.id
   383→            }
   384→            if (phone) {
   385→              usedCustomerPhoneId = phone.id
   386→            }
   387→          } else if (d.used_customer_address_id) {
   388→            const addr = await db.customerAddress.findFirst({
   389→              where: { id: d.used_customer_address_id, customerId: existing.id, organizationId: ctx.company.organizationId },
   390→              select: { id: true },
   391→            })
   392→            if (addr) {
   393→              usedCustomerAddressId = addr.id
   394→              selectedSavedAddressId = addr.id
   395→            }
   396→          } else if (d.used_customer_phone_id) {
   397→            const phone = await db.customerPhone.findFirst({
   398→              where: { id: d.used_customer_phone_id, customerId: existing.id, organizationId: ctx.company.organizationId },
   399→              select: { id: true },
   400→            })
   401→            if (phone) {
   402→              usedCustomerPhoneId = phone.id
   403→            }
   404→          }
   405→        } else if (d.new_customer) {
   406→          // Inline new-customer path — create the customer + phones +
   407→          // addresses via the INTERNAL variant (skips the redundant
   408→          // getWorkspace + requirePermission that createCustomer() would
   409→          // re-run). createCustomerInternal also returns the default
   410→          // address + primary phone IDs, saving two findFirst queries.
   411→          const createResult = await createCustomerInternal(ctx, d.new_customer)
   412→          if (!createResult.success || !createResult.data) {
   413→            return { ok: false as const, error: createResult.error ?? 'Failed to create customer' }
   414→          }
   415→          customerId = createResult.data.customerId
   416→          customerName = d.new_customer.name.trim()
   417→          // For a brand-new customer, the first address they provided IS the
   418→          // saved default address. The internal variant returns its ID
   419→          // directly — no extra query needed.
   420→          usedCustomerAddressId = createResult.data.defaultAddressId
   421→          selectedSavedAddressId = createResult.data.defaultAddressId
   422→          usedCustomerPhoneId = createResult.data.primaryPhoneId
   423→          // For a brand-new customer, we already saved the address as part of
   424→          // createCustomer — don't double-save it.
   425→          saveAddressForNextTime = false
   426→        } else {
   427→          return { ok: false as const, error: 'Either customer_id or new_customer is required' }
   428→        }
   429→
   430→        return {
   431→          ok: true as const,
   432→          customerId,
   433→          customerName,
   434→          usedCustomerAddressId,
   435→          usedCustomerPhoneId,
   436→          saveAddressForNextTime,
   437→          selectedSavedAddressId,
   438→        }
   439→      })(),
   440→      // ── Variant fetch (already batched via `in` clause) ──
   441→      db.orgProductVariant.findMany({
   442→        where: { id: { in: variantIds }, organizationId: ctx.company.organizationId },
   443→        include: {
   444→          product: { select: { title: true } },
   445→          companyPricing: { where: { companyId: ctx.company.id } },
   446→        },
   447→      }),
   448→      // ── Company order settings — fetched ONCE (was twice) ──
   449→      db.companyOrderSetting.findUnique({
   450→        where: { companyId: ctx.company.id },
   451→        select: {
   452→          requireOrderConfirmation: true,
   453→          courierBookingMode: true,
   454→          defaultCourierCompanyIntegrationId: true,
   455→        },
   456→      }),
   457→      // ── Order number (SQL function, independent) ──
   458→      generateOrderNumber(ctx.company.id),
   459→    ])
   460→
   461→    if (!custResult.ok) {
   462→      return { success: false, error: custResult.error }
   463→    }
   464→    const {
   465→      customerId,
   466→      customerName,
   467→      usedCustomerAddressId,
   468→      usedCustomerPhoneId,
   469→      saveAddressForNextTime,
   470→      selectedSavedAddressId,
   471→    } = custResult
   472→
   473→    if (variants.length !== variantIds.length) {
   474→      return { success: false, error: 'One or more variants not found in this organization' }
   475→    }
   476→
   477→    // 3. recipient_name defaults to the customer's name if not explicitly overridden.
   478→    const recipientName = (d.recipient_name && d.recipient_name.trim()) || customerName
   479→
   480→    // 4. Compute subtotal + build order items data (CPU-only, no DB)
   481→    let subtotal = 0
   482→    const orderItemsData: Array<{
   483→      orgVariantId: string
   484→      quantity: number
   485→      unitPrice: number
   486→      lineTotal: number
   487→      fulfillmentTypeSnapshot: string
   488→    }> = []
   489→
   490→    const orderDetailParts: string[] = []
   491→
   492→    for (const item of d.items) {
   493→      const variant = variants.find((v) => v.id === item.org_variant_id)
   494→      if (!variant) continue
   495→
   496→      const unitPrice =
   497→        item.unit_price ??
   498→        (variant.companyPricing[0]?.salePrice
   499→          ? Number(variant.companyPricing[0].salePrice)
   500→          : Number(variant.costPrice))
   501→
   502→      const lineTotal = item.quantity * unitPrice
   503→      subtotal += lineTotal
   504→
   505→      orderItemsData.push({
   506→        orgVariantId: item.org_variant_id,
   507→        quantity: item.quantity,
   508→        unitPrice,
   509→        lineTotal,
   510→        fulfillmentTypeSnapshot: variant.fulfillmentType,
   511→      })
   512→
   513→      const attrParts: string[] = []
   514→      try {
   515→        const attrs = JSON.parse(variant.attributeValues || '{}') as Record<string, string>
   516→        for (const [k, v] of Object.entries(attrs)) {
   517→          if (v) attrParts.push(`${k}: ${v}`)
   518→        }
   519→      } catch {
   520→        // ignore parse errors
   521→      }
   522→      const inner = [variant.sku, ...attrParts].filter(Boolean).join(', ')
   523→      orderDetailParts.push(
   524→        `${variant.product.title}${inner ? ` (${inner})` : ''} ×${item.quantity}`,
   525→      )
   526→    }
   527→
   528→    // 5. Compute totals (CPU-only)
   529→    const discountAmount = d.discount_amount ?? 0
   530→    const courierCharges = d.courier_charges ?? 0
   531→    const estimatedDeliveryCharge = d.estimated_delivery_charge ?? 0
   532→    const taxAmount = d.tax_amount ?? 0
   533→    const totalOrderValue = subtotal + courierCharges + estimatedDeliveryCharge + taxAmount - discountAmount
   534→
   535→    // 6. Determine payment status + source (CPU-only)
   536→    let paymentStatus: string = 'cod_pending'
   537→    let paymentSource: string = 'cod_native'
   538→    let advanceAmount: number | null = null
   539→    let advancePaymentMethod: string | null = null
   540→    let advancePaymentReference: string | null = null
   541→    let advancePaidAt: Date | null = null
   542→
   543→    if (d.payment_type === 'partial_advance') {
   544→      paymentStatus = 'advance_paid'
   545→      paymentSource = 'manual_conversion'
   546→      advanceAmount = d.advance_amount ?? 0
   547→      advancePaymentMethod = d.advance_payment_method || null
   548→      advancePaymentReference = d.advance_payment_reference || null
   549→      advancePaidAt = new Date()
   550→    } else if (d.payment_type === 'fully_prepaid') {
   551→      paymentStatus = 'fully_prepaid'
   552→      paymentSource = 'manual_conversion'
   553→      advanceAmount = totalOrderValue
   554→      advancePaymentMethod = d.advance_payment_method || null
   555→      advancePaymentReference = d.advance_payment_reference || null
   556→      advancePaidAt = new Date()
   557→    }
   558→
   559→    const remainingCodAmount = totalOrderValue - (advanceAmount ?? 0)
   560→
   561→    // 7. Determine initial order status — uses the ALREADY-FETCHED
   562→    //    companyOrderSetting (no separate query). Previously this did a
   563→    //    dedicated findUnique for requireOrderConfirmation.
   564→    let orderStatus: string = 'pending'
   565→    let confirmedAt: Date | null = null
   566→    let skippedConfirmation = false
   567→
   568→    if (paymentStatus === 'advance_paid' || paymentStatus === 'fully_prepaid') {
   569→      orderStatus = 'confirmed'
   570→      confirmedAt = new Date()
   571→    } else {
   572→      if (!orderSettings?.requireOrderConfirmation) {
   573→        orderStatus = 'confirmed'
   574→        confirmedAt = new Date()
   575→        skippedConfirmation = true
   576→      }
   577→    }
   578→
   579→    // 8. Universal courier reference fields (migration 015).
   580→    const orderRefNumber =
   581→      (d.order_ref_number && d.order_ref_number.trim()) || flowopsOrderNumber
   582→    const orderDetail =
   583→      (d.order_detail && d.order_detail.trim()) || orderDetailParts.join(', ')
   584→
   585→    // 9. Create the order (sequential — needs all above)
   586→    const order = await db.order.create({
   587→      data: {
   588→        organizationId: ctx.company.organizationId,
   589→        companyId: ctx.company.id,
   590→        flowopsOrderNumber,
   591→        orderSource: 'manual',
   592→        customerId,
   593→        recipientName,
   594→        usedCustomerAddressId,
   595→        usedCustomerPhoneId,
   596→        status: orderStatus,
   597→        paymentType: d.payment_type,
   598→        paymentStatus,
   599→        paymentSource,
   600→        subtotal,
   601→        discountAmount: discountAmount || null,
   602→        discountReason: d.discount_reason || null,
   603→        courierCharges: courierCharges || null,
   604→        estimatedDeliveryCharge: estimatedDeliveryCharge || null,
   605→        taxAmount: taxAmount || null,
   606→        taxLabel: d.tax_label?.trim() || null,
   607→        totalOrderValue,
   608→        advanceAmount,
   609→        advancePaymentMethod,
   610→        advancePaymentReference,
   611→        advancePaidAt,
   612→        remainingCodAmount,
   613→        deliveryAddress: d.delivery_address,
   614→        deliveryCity: d.delivery_city,
   615→        courierName: d.courier_name || null,
   616→        courierCompanyIntegrationId: d.courier_company_integration_id || null,
   617→        recommendedCourierCompanyIntegrationId: d.courier_company_integration_id || null,
   618→        dispatchLocationId: d.dispatch_location_id,
   619→        notesForCourier: d.notes_for_courier || null,
   620→        pickupAddressId: d.pickup_address_id || null,
   621→        orderRefNumber,
   622→        orderDetail,
   623→        skippedConfirmation,
   624→        confirmedAt,
   625→        createdBy: ctx.employee.id,
   626→      },
   627→    })
   628→
   629→    // 10. Batch-create order items — SINGLE query via createManyAndReturn
   630→    //     (was N sequential inserts in a loop). Returns the created rows so
   631→    //     we can populate `createdItems` for the response.
   632→    const createdItemRows = await db.orderItem.createManyAndReturn({
   633→      data: orderItemsData.map((itemData) => ({
   634→        orderId: order.id,
   635→        orgVariantId: itemData.orgVariantId,
   636→        organizationId: ctx.company.organizationId,
   637→        quantity: itemData.quantity,
   638→        unitPrice: itemData.unitPrice,
   639→        lineTotal: itemData.lineTotal,
   640→        fulfillmentStatus: 'reserved', // PLACEHOLDER — reserveOrderStock validates/adjusts
   641→        fulfillmentTypeSnapshot: itemData.fulfillmentTypeSnapshot,
   642→        reservedLocationId: d.dispatch_location_id,
   643→      })),
   644→      select: { id: true, orgVariantId: true, quantity: true },
   645→    })
   646→    const createdItems = createdItemRows.map((r) => ({
   647→      id: r.id,
   648→      orgVariantId: r.orgVariantId,
   649→      quantity: r.quantity,
   650→    }))
   651→
   652→    // 11. Audit log (fire-and-forget — see src/lib/audit.ts)
   653→    insertAuditLog({
   654→      action: 'order.created',
   655→      entityType: 'order',
   656→      entityId: order.id,
   657→      companyId: ctx.company.id,
   658→      organizationId: ctx.company.organizationId,
   659→      userId: ctx.user.id,
   660→      employeeId: ctx.employee.id,
   661→      newValues: {
   662→        flowopsOrderNumber,
   663→        status: orderStatus,
   664→        paymentType: d.payment_type,
   665→        paymentStatus,
   666→        itemCount: orderItemsData.length,
   667→        totalOrderValue,
   668→        customerId,
   669→      },
   670→    })
   671→
   672→    // 12. Customer Management System integration — address lastUsedAt +
   673→    //     save-one-off-address + recompute cached stats. The
   674→    //     saveAddressForNextTime branch MUST be sequential (the order.update
   675→    //     needs the created address ID), but markAddressAsUsed (the other
   676→    //     branch) and updateCustomerStats are independent → run in parallel.
   677→    if (selectedSavedAddressId) {
   678→      // Bump lastUsedAt on the saved address + recompute customer stats
   679→      // in parallel (independent writes to different tables).
   680→      await Promise.all([
   681→        markAddressAsUsed(selectedSavedAddressId),
   682→        updateCustomerStats(customerId),
   683→      ])
   684→    } else if (saveAddressForNextTime && d.delivery_address && d.delivery_city) {
   685→      // One-off address typed + user opted to save it for next time.
   686→      // Persist as a non-default customer_addresses row, then link the
   687→      // order to it + recompute customer stats in parallel.
   688→      const savedAddr = await db.customerAddress.create({
   689→        data: {
   690→          customerId,
   691→          organizationId: ctx.company.organizationId,
   692→          label: null,
   693→          address: d.delivery_address,
   694→          city: d.delivery_city,
   695→          isDefault: false,
   696→          lastUsedAt: new Date(),
   697→        },
   698→      })
   699→      await Promise.all([
   700→        db.order.update({
   701→          where: { id: order.id },
   702→          data: { usedCustomerAddressId: savedAddr.id },
   703→        }),
   704→        updateCustomerStats(customerId),
   705→      ])
   706→    } else {
   707→      // No address to save — just recompute stats.
   708→      await updateCustomerStats(customerId)
   709→    }
   710→
   711→    // 13. If the order auto-confirmed, run the stock reservation logic.
   712→    //     Sequential — reserveOrderStock reads the order items created above.
   713→    if (orderStatus === 'confirmed') {
   714→      await reserveOrderStock(order.id, ctx)
   715→    }
   716→
   717→    // 14. AUTO-BOOKING — uses the ALREADY-FETCHED orderSettings (no 2nd
   718→    //     findUnique). Fires in the background (PostEx can take 50-100s).
   719→    let bookingAttempted = false
   720→    let bookingSucceeded = false
   721→    let bookingError: string | undefined
   722→    let bookingTrackingNumber: string | undefined
   723→
   724→    if (orderStatus === 'confirmed') {
   725→      const shouldAutoBook =
   726→        orderSettings?.courierBookingMode === 'automatic' &&
   727→        (d.courier_company_integration_id || orderSettings?.defaultCourierCompanyIntegrationId)
   728→
   729→      if (shouldAutoBook) {
   730→        bookingAttempted = true
   731→        ;(async () => {
   732→          try {
   733→            const { maybeAutoBookOrder } = await import('./booking.actions')
   734→            const bookResult = await maybeAutoBookOrder(order.id, 'manual', orderStatus)
   735→            if (bookResult.success) {
   736→              console.log(`[createManualOrder] Background auto-booking succeeded for ${flowopsOrderNumber}: tracking=${bookResult.data?.trackingNumber}`)
   737→            } else {
   738→              console.warn(`[createManualOrder] Background auto-booking failed for ${flowopsOrderNumber}: ${bookResult.error}`)
   739→            }
   740→          } catch (err) {
   741→            console.error(`[createManualOrder] Background auto-booking threw for ${flowopsOrderNumber}:`, err)
   742→          }
   743→        })()
   744→      }
   745→    }
   746→
   747→    insertMetricEvent({
   748→      companyId: ctx.company.id,
   749→      entityType: 'order',
   750→      entityId: order.id,
   751→      metricKey: 'order.created',
   752→      numericValue: Number(totalOrderValue),
   753→      dimensions: { order_source: 'manual', payment_type: d.payment_type, company_id: ctx.company.id },
   754→    })
   755→
   756→    return {
   757→      success: true,
   758→      data: {
   759→        orderId: order.id,
   760→        flowopsOrderNumber,
   761→        orderItems: createdItems,
   762→        bookingAttempted,
   763→        bookingSucceeded,
   764→        bookingError,
   765→        bookingTrackingNumber,
   766→      },
   767→    }
   768→  } catch (err) {
   769→    return {
   770→      success: false,
   771→      error: err instanceof Error ? err.message : 'Failed to create order',
   772→    }
   773→  }
   774→}

   784→): Promise<ActionResult<{ orderId: string; flowopsOrderNumber: string }>> {

   787→    // This function would be called by a future webhook handler.

   928→        orgVariantId: variant.id,
   929→        quantity: li.quantity,
   930→        unitPrice,
   931→        fulfillmentTypeSnapshot: variant.fulfillmentType,

   940→

  1037→      entityType: 'order',
  1038→      entityId: order.id,

  1044→    return { success: true, data: { orderId: order.id, flowopsOrderNumber } }
  1045→  } catch (err) {
  1046→    return {
  1047→      success: false,
  1048→      error: err instanceof Error ? err.message : 'Failed to create order from Shopify webhook',
  1049→    }
  1050→  }
  1051→}

  1057→export async function confirmOrder(orderId: string): Promise<ActionResult> {
  1058→  try {
  1059→    const ctx = await getWorkspace()
  1060→    await requirePermission(ctx, PERMISSIONS.ORDERS_MANAGE)
  1061→
  1062→    const order = await db.order.findFirst({
  1063→      where: { id: orderId, companyId: ctx.company.id },
  1064→    })
  1065→    if (!order) return { success: false, error: 'Order not found' }
  1066→    if (order.status !== 'pending') {
  1067→      return { success: false, error: `Order is already ${order.status} (cannot confirm)` }
  1068→    }
  1069→
  1070→    await db.order.update({
  1071→      where: { id: orderId },
  1072→      data: { status: 'confirmed', confirmedAt: new Date() },
  1073→    })
  1074→
  1075→    insertAuditLog({
  1076→      action: 'order.confirmed',
  1077→      entityType: 'order',
  1078→      entityId: orderId,
  1079→      companyId: ctx.company.id,
  1080→      organizationId: ctx.company.organizationId,
  1081→      userId: ctx.user.id,
  1082→      employeeId: ctx.employee.id,
  1083→      oldValues: { status: 'pending' },
  1084→      newValues: { status: 'confirmed' },
  1085→    })
  1086→
  1087→    // Step 3: Run stock reservation for all items.
  1088→    // This may flip the order to 'partially_backordered' if some items
  1089→    // can't be fulfilled immediately.
  1090→    const reserveResult = await reserveOrderStock(orderId, ctx)
  1091→    if (!reserveResult.success) {
  1092→      // Non-fatal — the order is confirmed, but reservation had issues.
  1093→      // The results array has per-item details.
  1094→    }
  1095→
  1096→    insertMetricEvent({
  1097→      companyId: ctx.company.id,
  1098→      entityType: 'order',
  1099→      entityId: orderId,
  1100→      metricKey: 'order.confirmed',
  1101→      numericValue: Number(order.totalOrderValue),
  1102→      dimensions: { confirmation_method: 'manual' },
  1103→    })
  1104→
  1105→    // Recompute cached customer stats (order count may change if this was a
  1106→    // re-confirmation, and the status transition affects value/rto calcs).
  1107→    // Non-fatal — never break the confirm action on a stats failure.
  1108→    await updateCustomerStats(order.customerId).catch(() => {})
  1109→
  1110→    return { success: true }
  1111→  } catch (err) {
  1112→    return {
  1113→      success: false,
  1114→      error: err instanceof Error ? err.message : 'Failed to confirm order',
  1115→    }
  1116→  }
  1117→}

  1130→    const parsed = convertPaymentSchema.safeParse(input)

  1133→    }
  1134→    const d = parsed.data
  1135→
  1136→    const order = await db.order.findFirst({
  1137→      where: { id: d.order_id, companyId: ctx.company.id },
  1138→    })

  1140→    if (order.paymentStatus !== 'cod_pending') {
  1141→      return { success: false, error: 'Order payment is already converted' }
  1142→    }
  1143→

  1145→      paymentType: order.paymentType,

  1148→      status: order.status,
  1149→    }

  1152→

  1154→    // whenever the advance amount changes: totalOrderValue - (advanceAmount ?? 0).
  1155→    const newRemainingCod = Number(order.totalOrderValue) - (d.advance_amount ?? 0)
  1156→

  1164→      advancePaymentScreenshotUrl: d.advance_payment_screenshot_url || null,

  1166→      remainingCodAmount: newRemainingCod,
  1167→      convertedBy: ctx.employee.id,
  1168→      convertedAt: new Date(),
  1169→    }
  1170→
  1171→    // If order was pending (awaiting confirmation), payment conversion is
  1172→    // itself a confirmation signal

  1174→      updateData.status = 'confirmed'
  1175→      updateData.confirmedAt = new Date()
  1176→    }
  1177→
  1178→    await db.order.update({ where: { id: d.order_id }, data: updateData })
  1179→

  1182→      entityType: 'order',
  1183→      entityId: d.order_id,
  1184→      companyId: ctx.company.id,
  1185→      organizationId: ctx.company.organizationId,
  1186→      userId: ctx.user.id,
  1187→      employeeId: ctx.employee.id,
  1188→      oldValues,
  1189→      newValues: {
  1190→        paymentType: d.new_payment_type,
  1191→        paymentStatus: newPaymentStatus,
  1192→        paymentSource: 'manual_conversion',
  1193→        advanceAmount: d.advance_amount,
  1194→        remainingCodAmount: newRemainingCod,
  1195→        status: updateData.status ?? order.status,
  1196→      },
  1197→    })
  1198→
  1199→    insertMetricEvent({
  1200→      companyId: ctx.company.id,
  1201→      entityType: 'order',
  1202→      entityId: d.order_id,

  1360→export async function cancelOrder(input: CancelOrderInput): Promise<ActionResult> {
  1361→  try {
  1362→    const ctx = await getWorkspace()
  1363→    await requirePermission(ctx, PERMISSIONS.ORDERS_CANCEL)
  1364→
  1365→    const parsed = cancelOrderSchema.safeParse(input)
  1366→    if (!parsed.success) {
  1367→      return { success: false, error: parsed.error.issues[0]?.message ?? 'Invalid input' }
  1368→    }
  1369→    const d = parsed.data
  1370→
  1371→    const order = await db.order.findFirst({
  1372→      where: { id: d.order_id, companyId: ctx.company.id },
  1373→    })
  1374→    if (!order) return { success: false, error: 'Order not found' }
  1375→
  1376→    const nonCancellableStatuses = ['dispatched', 'delivered', 'rto', 'cancelled', 'refunded']
  1377→    if (nonCancellableStatuses.includes(order.status)) {
  1378→      return {
  1379→        success: false,
  1380→        error: `Cannot cancel an order that is already ${order.status} (use the returns flow instead)`,
  1381→      }
  1382→    }
  1383→
  1384→    // ── Phase 3: Set physicalUnpackRequired if the order was already
  1385→    // processing/packed (a physical parcel likely already exists) ──
  1386→    const physicalUnpackRequired =
  1387→      order.status === 'processing' ||
  1388→      order.packedAt !== null
  1389→
  1390→    await db.order.update({
  1391→      where: { id: d.order_id },
  1392→      data: {
  1393→        status: 'cancelled',
  1394→        cancelledAt: new Date(),
  1395→        cancellationReason: d.cancellation_reason,
  1396→        physicalUnpackRequired,
  1397→      },
  1398→    })
  1399→
  1400→    // Step 3: Unreserve stock for any items with fulfillment_status='reserved'.
  1401→    // Items with fulfillment_status='backordered' need no inventory action
  1402→    // (no reservation ever existed for them) — they'll be orphaned since
  1403→    // the order is now cancelled and won't be picked up by future
  1404→    // checkAndFulfillBackorders() runs (which skip cancelled orders).
  1405→    const reservedItems = await db.orderItem.findMany({
  1406→      where: { orderId: d.order_id, fulfillmentStatus: 'reserved' },
  1407→    })
  1408→
  1409→    for (const item of reservedItems) {
  1410→      const locationId = item.reservedLocationId ?? order.dispatchLocationId
  1411→      if (!locationId) continue
  1412→
  1413→      await unreserveStockForOrder({
  1414→        orgVariantId: item.orgVariantId,
  1415→        locationId,
  1416→        organizationId: order.organizationId,
  1417→        companyId: ctx.company.id,
  1418→        employeeId: ctx.employee.id,
  1419→        quantity: item.quantity,
  1420→        orderId: d.order_id,
  1421→      })
  1422→    }
  1423→
  1424→    insertAuditLog({
  1425→      action: 'order.cancelled',
  1426→      entityType: 'order',
  1427→      entityId: d.order_id,
  1428→      companyId: ctx.company.id,
  1429→      organizationId: ctx.company.organizationId,
  1430→      userId: ctx.user.id,
  1431→      employeeId: ctx.employee.id,
  1432→      oldValues: { status: order.status },
  1433→      newValues: { status: 'cancelled', reason: d.cancellation_reason, unreservedItems: reservedItems.length },
  1434→    })
  1435→
  1436→    insertMetricEvent({
  1437→      companyId: ctx.company.id,
  1438→      entityType: 'order',
  1439→      entityId: d.order_id,
  1440→      metricKey: 'order.cancelled',
  1441→      numericValue: Number(order.totalOrderValue),
  1442→      dimensions: { cancellation_reason: d.cancellation_reason, had_reserved_items: reservedItems.length > 0 },
  1443→    })
  1444→
  1445→    // Recompute cached customer stats (cancelled orders are excluded from
  1446→    // total_orders_count, so this count drops). Non-fatal.
  1447→    await updateCustomerStats(order.customerId).catch(() => {})
  1448→
  1449→    return { success: true }
  1450→  } catch (err) {
  1451→    return {
  1452→      success: false,
  1453→      error: err instanceof Error ? err.message : 'Failed to cancel order',
  1454→    }
  1455→  }
  1456→}

  1462→export async function listOrders(
  1463→  filters: OrderFilters = {},
  1464→): Promise<ActionResult<{
  1465→  orders: Array<{
  1466→    id: string
  1467→    flowopsOrderNumber: string
  1468→    externalOrderReference: string | null
  1469→    externalOrderId: string | null
  1470→    orderSource: string
  1471→    status: string
  1472→    paymentType: string
  1473→    paymentStatus: string
  1474→    paymentSource: string
  1475→    subtotal: number
  1476→    discountAmount: number | null
  1477→    courierCharges: number | null
  1478→    totalOrderValue: number
  1479→    advanceAmount: number | null
  1480→    remainingCodAmount: number | null
  1481→    codCollected: boolean
  1482→    courierName: string | null
  1483→    trackingNumber: string | null
  1484→    dispatchLocationId: string | null
  1485→    customerId: string
  1486→    deliveryAddress: string | null
  1487→    deliveryCity: string | null
  1488→    confirmedAt: Date | null
  1489→    dispatchedAt: Date | null
  1490→    deliveredAt: Date | null
  1491→    createdAt: Date
  1492→    customerName: string
  1493→    customerPhone: string
  1494→    itemCount: number
  1495→  }>
  1496→  total: number
  1497→}>> {
  1498→  try {
  1499→    const ctx = await getWorkspace()
  1500→    const limit = Math.min(filters.limit ?? 50, 100)
  1501→    const offset = filters.offset ?? 0
  1502→
  1503→    const where: Prisma.OrderWhereInput = {
  1504→      companyId: ctx.company.id,
  1505→    }
  1506→
  1507→    // Status filter — prefer multi-select `statuses`, fall back to single `status`
  1508→    if (filters.statuses && filters.statuses.length > 0) {
  1509→      where.status = { in: filters.statuses }
  1510→    } else if (filters.status) {
  1511→      where.status = filters.status
  1512→    }
  1513→
  1514→    // Payment type filter — prefer multi-select `paymentTypes`, fall back to single `paymentType`
  1515→    if (filters.paymentTypes && filters.paymentTypes.length > 0) {
  1516→      where.paymentType = { in: filters.paymentTypes }
  1517→    } else if (filters.paymentType) {
  1518→      where.paymentType = filters.paymentType
  1519→    }
  1520→
  1521→    // Payment status filter (multi-select preferred)
  1522→    if (filters.paymentStatuses && filters.paymentStatuses.length > 0) {
  1523→      where.paymentStatus = { in: filters.paymentStatuses }
  1524→    } else if (filters.paymentStatus) {
  1525→      where.paymentStatus = filters.paymentStatus
  1526→    }
  1527→
  1528→    // Order source filter — prefer multi-select `orderSources`, fall back to single `orderSource`
  1529→    if (filters.orderSources && filters.orderSources.length > 0) {
  1530→      where.orderSource = { in: filters.orderSources }
  1531→    } else if (filters.orderSource) {
  1532→      where.orderSource = filters.orderSource
  1533→    }
  1534→
  1535→    // Courier filter — prefer multi-select `courierNames`, fall back to single `courierName`
  1536→    if (filters.courierNames && filters.courierNames.length > 0) {
  1537→      where.courierName = { in: filters.courierNames }
  1538→    } else if (filters.courierName) {
  1539→      where.courierName = filters.courierName
  1540→    }
  1541→
  1542→    if (filters.customerId) where.customerId = filters.customerId
  1543→
  1544→    // Delivery city filter — case-insensitive contains on delivery_city
  1545→    if (filters.deliveryCity) {
  1546→      where.deliveryCity = { contains: filters.deliveryCity, mode: 'insensitive' }
  1547→    }
  1548→
  1549→    if (filters.dateFrom || filters.dateTo) {
  1550→      where.createdAt = {}
  1551→      if (filters.dateFrom) where.createdAt.gte = new Date(filters.dateFrom)
  1552→      if (filters.dateTo) {
  1553→        // date_to is inclusive of the whole day
  1554→        const end = new Date(filters.dateTo)
  1555→        end.setHours(23, 59, 59, 999)
  1556→        where.createdAt.lte = end
  1557→      }
  1558→    }
  1559→
  1560→    // Amount range filter on total_order_value
  1561→    if (filters.amountMin !== undefined || filters.amountMax !== undefined) {
  1562→      where.totalOrderValue = {}
  1563→      if (filters.amountMin !== undefined) where.totalOrderValue.gte = filters.amountMin
  1564→      if (filters.amountMax !== undefined) where.totalOrderValue.lte = filters.amountMax
  1565→    }
  1566→
  1567→    // Filter to orders containing a specific variant — uses Prisma's `some`
  1568→    // relation filter which compiles to an EXISTS subquery (avoids duplicate
  1569→    // rows that a join-based filter would introduce).
  1570→    if (filters.orgVariantId) {
  1571→      where.items = { some: { orgVariantId: filters.orgVariantId } }
  1572→    }
  1573→
  1574→    if (filters.search) {
  1575→      where.OR = [
  1576→        { flowopsOrderNumber: { contains: filters.search, mode: 'insensitive' } },
  1577→        { externalOrderReference: { contains: filters.search, mode: 'insensitive' } },
  1578→        // Customer Management System: phones live in customer_phones now.
  1579→        // Search across the customer's name + any of their phone numbers.
  1580→        { customer: { name: { contains: filters.search, mode: 'insensitive' } } },
  1581→        { customer: { phones: { some: { phoneRaw: { contains: filters.search, mode: 'insensitive' } } } } },
  1582→      ]
  1583→    }
  1584→

  1586→      db.order.findMany({
  1587→        where,
  1588→        include: {
  1589→          customer: {
  1590→            select: {
  1591→              name: true,
  1592→              phones: {
  1593→                where: { isPrimary: true },
  1594→                take: 1,
  1595→                select: { phoneRaw: true },
  1596→              },
  1597→            },
  1598→          },
  1599→          _count: { select: { items: true } },
  1600→        },
  1601→        orderBy: { createdAt: 'desc' },
  1602→        take: limit,
  1603→        skip: offset,
  1604→      }),
  1605→      db.order.count({ where }),
  1606→    ])
  1607→
  1608→    return {
  1609→      success: true,
  1610→      data: {
  1611→        orders: orders.map((o) => ({
  1612→          id: o.id,
  1613→          flowopsOrderNumber: o.flowopsOrderNumber,
  1614→          externalOrderReference: o.externalOrderReference,
  1615→          externalOrderId: o.externalOrderId,
  1616→          orderSource: o.orderSource,
  1617→          status: o.status,
  1618→          paymentType: o.paymentType,
  1619→          paymentStatus: o.paymentStatus,
  1620→          paymentSource: o.paymentSource,
  1621→          subtotal: Number(o.subtotal),
  1622→          discountAmount: o.discountAmount ? Number(o.discountAmount) : null,
  1623→          courierCharges: o.courierCharges ? Number(o.courierCharges) : null,
  1624→          estimatedDeliveryCharge: o.estimatedDeliveryCharge ? Number(o.estimatedDeliveryCharge) : null,
  1625→          actualDeliveryCharge: o.actualDeliveryCharge ? Number(o.actualDeliveryCharge) : null,
  1626→          taxAmount: o.taxAmount ? Number(o.taxAmount) : null,
  1627→          taxLabel: o.taxLabel ?? null,
  1628→          totalOrderValue: Number(o.totalOrderValue),
  1629→          advanceAmount: o.advanceAmount ? Number(o.advanceAmount) : null,
  1630→          remainingCodAmount: o.remainingCodAmount
  1631→            ? Number(o.remainingCodAmount)
  1632→            : Math.max(
  1633→                0,
  1634→                Number(o.totalOrderValue) - (o.advanceAmount ? Number(o.advanceAmount) : 0),
  1635→              ),
  1636→          codCollected: o.codCollected,
  1637→          courierName: o.courierName,
  1638→          trackingNumber: o.trackingNumber,
  1639→          courierCompanyIntegrationId: o.courierCompanyIntegrationId,
  1640→          courierBookingStatus: o.courierBookingStatus,
  1641→          courierBookingFailureReason: o.courierBookingFailureReason,
  1642→          courierCityStatus: o.courierCityStatus,
  1643→          courierSubStatus: o.courierSubStatus,
  1644→          needsShipperAdvice: o.needsShipperAdvice,
  1645→          dispatchLocationId: o.dispatchLocationId,
  1646→          customerId: o.customerId,
  1647→          deliveryAddress: o.deliveryAddress,
  1648→          deliveryCity: o.deliveryCity,
  1649→          // Universal courier reference fields (migration 015)
  1650→          orderRefNumber: o.orderRefNumber,
  1651→          orderDetail: o.orderDetail,
  1652→          notesForCourier: o.notesForCourier,
  1653→          confirmedAt: o.confirmedAt,

  1656→          createdAt: o.createdAt,
  1657→          customerName: o.customer.name,
  1658→          customerPhone: o.customer.phones[0]?.phoneRaw ?? null,
  1659→          itemCount: o._count.items,

  1663→    }
  1664→  } catch (err) {
  1665→    return {
  1666→      success: false,

  1668→    }
