     1→/**
     2→ * Item Exchange System — server actions.
     3→ *
     4→ * Implements the full exchange lifecycle with STRICT SEQUENCING for the two
     5→ * exchange methods:
     6→ *
     7→ *   courier_replacement:    request → dispatch new item → (courier collects old)
     8→ *                           → await return → manually verify old → complete
     9→ *
    10→ *   customer_self_return:   request (awaiting_customer_to_ship) → customer
    11→ *                           confirmed shipped → await physical arrival →
    12→ *                           manually verify old (THIS IS THE GATE) →
    13→ *                           create+dispatch new order → complete
    14→ *
    15→ * CRITICAL RULE (enforced throughout):
    16→ *   For customer_self_return, the new order/item is NEVER created before
    17→ *   verifyOldItemReceived() succeeds. This sequential gate is the core
    18→ *   business rule and must not be bypassed by any code path.
    19→ *
    20→ * INVENTORY INTEGRATION:
    21→ *   verifyOldItemReceived() is the ONLY function that processes the old item's
    22→ *   return in inventory. It calls processInventoryTransaction() directly (the
    23→ *   same function the /api/inventory/receive-returned-stitched and
    24→ *   /api/inventory/receive routes use) — we do NOT reimplement that logic.
    25→ *   For damaged items, a stock_loss_records entry is created directly.
    26→ *
    27→ * Every mutation calls insertAuditLog() AND insertMetricEvent(), and returns
    28→ * { success, data?, error? }.
    29→ */

    77→// ──────────────────────────────────────────────────────────────
    78→// Internal helper: create the linked EXCHANGE SHIPMENT + reserve stock +
    79→// dispatch. Used by BOTH:
    80→//   - dispatchExchangeNewItem() for courier_replacement (immediate)
    81→//   - verifyOldItemReceived() for customer_self_return (after verification)
    82→//
    83→// MIGRATION 008 CHANGE: this function now creates an ExchangeShipment
    84→// (EXCH-{YYYY}-{NNNNN}) instead of an Order row. The legacy newOrderId/
    85→// newOrderItemId columns on order_exchanges remain populated only on old,
    86→// pre-migration historical exchange records. NEW exchanges are linked via
    87→// the exchange_shipments.orderExchangeId FK.
    88→//
    89→// Returns the new exchange shipment ID + number.
    90→// ──────────────────────────────────────────────────────────────
    91→async function createAndReserveExchangeShipment(
    92→  exchangeId: string,
    93→  ctx: Awaited<ReturnType<typeof getWorkspace>>,
    94→  options?: {
    95→    orderRefNumber?: string
    96→    orderDetail?: string
    97→  },
    98→): Promise<{ newExchangeShipmentId: string; exchangeShipmentNumber: string }> {
    99→  // Fetch the exchange record (already validated by caller)
   100→  const exchange = await db.orderExchange.findUniqueOrThrow({
   101→    where: { id: exchangeId },
   102→    include: {
   103→      originalOrder: {
   104→        select: {
   105→          customerId: true,
   106→          dispatchLocationId: true,
   107→          usedCustomerAddressId: true,
   108→          usedCustomerPhoneId: true,
   109→        },
   110→      },
   111→      newOrgVariant: {
   112→        select: {
   113→          id: true,
   114→          fulfillmentType: true,
   115→          costPrice: true,
   116→          sku: true,
   117→          attributeValues: true,
   118→          product: { select: { title: true } },
   119→        },
   120→      },
   121→    },
   122→  })
   123→
   124→  const customerId = exchange.originalOrder.customerId
   125→
   126→  const shippingAddressId = exchange.originalOrder.usedCustomerAddressId
   127→  const shippingPhoneId = exchange.originalOrder.usedCustomerPhoneId
   128→
   129→  const priceDiff = Number(exchange.priceDifference)
   130→  const baseInvoiceAmount =
   131→    exchange.priceDifferenceStatus === 'customer_owes' && priceDiff > 0 ? priceDiff : 0
   132→  const invoiceAmount = baseInvoiceAmount
   133→
   134→  // 1. Generate the EXCH-{YYYY}-{NNNNN} number
   135→  const exchangeShipmentNumber = await generateExchangeShipmentNumber()
   136→
   137→  // 1a. Universal courier reference fields (migration 015)
   138→  const orderRefNumber =
   139→    (options?.orderRefNumber && options.orderRefNumber.trim()) || exchangeShipmentNumber
   140→  let orderDetail = (options?.orderDetail && options.orderDetail.trim()) || ''
   141→  if (!orderDetail) {
   142→    const attrParts: string[] = []
   143→    try {
   144→      const attrs = JSON.parse(exchange.newOrgVariant.attributeValues || '{}') as Record<string, string>
   145→      for (const [k, v] of Object.entries(attrs)) {
   146→        if (v) attrParts.push(`${k}: ${v}`)
   147→      }
   148→    } catch {
   149→      // ignore parse errors
   150→    }
   151→    const inner = [exchange.newOrgVariant.sku, ...attrParts].filter(Boolean).join(', ')
   152→    orderDetail = `${exchange.newOrgVariant.product.title}${inner ? ` (${inner})` : ''} ×1`
   153→  }
   154→
   155→  // 2. Create the exchange shipment — status='confirmed' (NOT dispatched!)
   156→  // The shipment will be booked via the Booking Workbench or SendExchangeShipmentModal,
   157→  // and only transition to 'dispatched' after a real courier booking succeeds.
   158→  const shipment = await db.exchangeShipment.create({
   159→    data: {
   160→      exchangeShipmentNumber,
   161→      organizationId: ctx.company.organizationId,
   162→      companyId: ctx.company.id,
   163→      orderExchangeId: exchangeId,
   164→      newOrgVariantId: exchange.newOrgVariantId,
   165→      quantity: 1,
   166→      fulfillmentTypeSnapshot: exchange.newOrgVariant.fulfillmentType,
   167→      customerId,
   168→      shippingAddressId,
   169→      shippingPhoneId,
   170→      status: 'confirmed',
   171→      isPriorityBackorder: true,
   172→      invoiceAmount,
   173→      orderRefNumber,
   174→      orderDetail,
   175→      confirmedAt: new Date(),
   176→      createdBy: ctx.employee.id,
   177→    },
   178→  })
   179→
   180→  // 3. Reserve stock (if stock_based) — mirrors reserveOrderStock()
   181→  const locationId = exchange.originalOrder.dispatchLocationId
   182→  if (locationId && exchange.newOrgVariant.fulfillmentType === 'stock_based') {
   183→    const pool = await db.inventoryPool.findUnique({
   184→      where: {
   185→        orgVariantId_locationId: {
   186→          orgVariantId: exchange.newOrgVariantId,
   187→          locationId,
   188→        },
   189→      },
   190→    })
   191→    const available = pool ? pool.onHand - pool.reserved : 0
   192→
   193→    if (available >= 1) {
   194→      await reserveStockForOrder({
   195→        orgVariantId: exchange.newOrgVariantId,
   196→        locationId,
   197→        organizationId: ctx.company.organizationId,
   198→        companyId: ctx.company.id,
   199→        employeeId: ctx.employee.id,
   200→        quantity: 1,
   201→      })
   202→    } else {
   203→      // Insufficient stock — mark as backordered (isPriorityBackorder=true)
   204→      await db.exchangeShipment.update({
   205→        where: { id: shipment.id },
   206→        data: { status: 'backordered', backorderedAt: new Date() },
   207→      })
   208→    }
   209→  }
   210→
   211→  // 4. Audit + metric
   212→  insertAuditLog({
   213→    action: 'exchange_shipment.created',
   214→    entityType: 'exchange_shipment',
   215→    entityId: shipment.id,
   216→    companyId: ctx.company.id,
   217→    organizationId: ctx.company.organizationId,
   218→    userId: ctx.user.id,
   219→    employeeId: ctx.employee.id,
   220→    newValues: {
   221→      exchangeId,
   222→      exchangeShipmentNumber,
   223→      newOrgVariantId: exchange.newOrgVariantId,
   224→      invoiceAmount,
   225→    },
   226→  })
   227→
   228→  insertMetricEvent({
   229→    companyId: ctx.company.id,
   230→    entityType: 'exchange_shipment',
   231→    entityId: shipment.id,
   232→    metricKey: 'exchange_shipment.created',
   233→    numericValue: 1,
   234→    dimensions: { exchange_method: exchange.exchangeMethod },
   235→  })
   236→
   237→  return { newExchangeShipmentId: shipment.id, exchangeShipmentNumber }
   238→}

   240→// ══════════════════════════════════════════════════════════════
   241→// PART 2: CREATE EXCHANGE REQUEST
   242→// ══════════════════════════════════════════════════════════════
   243→
   244→export async function createExchangeRequest(
   245→  input: CreateExchangeRequestInput,
   246→): Promise<ActionResult<{ exchangeId: string }>> {
   247→  try {
   248→    const ctx = await getWorkspace()
   249→    await requirePermission(ctx, PERMISSIONS.ORDERS_MANAGE)
   250→
   251→    const parsed = createExchangeRequestSchema.safeParse(input)
   252→    if (!parsed.success) {
   253→      return {
   254→        success: false,
   255→        error: parsed.error.issues[0]?.message ?? 'Invalid exchange request data',
   256→      }
   257→    }
   258→    const d = parsed.data
   259→
   260→    // 3. Fetch the original order_item + its parent order — VERIFY delivered
   261→    const orderItem = await db.orderItem.findFirst({
   262→      where: { id: d.original_order_item_id },
   263→      include: {
   264→        order: {
   265→          select: {
   266→            id: true,
   267→            status: true,
   268→            companyId: true,
   269→            organizationId: true,
   270→            customerId: true,
   271→          },
   272→        },
   273→      },
   274→    })
   275→    if (!orderItem) {
   276→      return { success: false, error: 'Original order item not found' }
   277→    }
   278→    // Verify the order belongs to this company
   279→    if (orderItem.order.companyId !== ctx.company.id) {
   280→      return { success: false, error: 'Original order item not found' }
   281→    }
   282→    // CRITICAL: only delivered orders can have exchanges
   283→    if (orderItem.order.status !== 'delivered') {
   284→      return {
   285→        success: false,
   286→        error: 'Items can only be exchanged after the order has been delivered to the customer.',
   287→      }
   288→    }
   289→
   290→    // 4. old_item_price from the order_item's unit_price
   291→    const oldItemPrice = Number(orderItem.unitPrice)
   292→
   293→    // 5. new_item_price from the target variant's company_variant_pricing
   294→    const variant = await db.orgProductVariant.findFirst({
   295→      where: {
   296→        id: d.new_org_variant_id,
   297→        organizationId: ctx.company.organizationId,
   298→      },
   299→      include: {
   300→        companyPricing: { where: { companyId: ctx.company.id }, take: 1 },
   301→      },
   302→    })
   303→    if (!variant) {
   304→      return { success: false, error: 'New variant not found in this organization' }
   305→    }
   306→    const newItemPrice = variant.companyPricing[0]?.salePrice
   307→      ? Number(variant.companyPricing[0].salePrice)
   308→      : Number(variant.costPrice)
   309→
   310→    // 6. Determine price_difference_status
   311→    const priceDifference = newItemPrice - oldItemPrice
   312→    const priceDifferenceStatus: 'customer_owes' | 'refund_due' | 'settled' =
   313→      priceDifference > 0 ? 'customer_owes' : priceDifference < 0 ? 'refund_due' : 'settled'
   314→
   315→    // 7. INSERT order_exchanges — status depends on exchange_method
   316→    const initialStatus =
   317→      d.exchange_method === 'courier_replacement'
   318→        ? 'requested' // ready to proceed to immediate new item dispatch
   319→        : 'awaiting_customer_to_ship_old_item' // new item dispatch is BLOCKED until verification
   320→
   321→    const exchange = await db.orderExchange.create({
   322→      data: {
   323→        organizationId: ctx.company.organizationId,
   324→        companyId: ctx.company.id,
   325→        originalOrderId: orderItem.order.id,
   326→        originalOrderItemId: orderItem.id,
   327→        newOrgVariantId: d.new_org_variant_id,
   328→        exchangeMethod: d.exchange_method,
   329→        status: initialStatus,
   330→        oldItemPrice,
   331→        newItemPrice,
   332→        priceDifference: newItemPrice - oldItemPrice,
   333→        priceDifferenceStatus,
   334→        reason: d.reason,
   335→        requestedBy: ctx.employee.id,
   336→      },
   337→    })
   338→
   339→    // 8. Audit log
   340→    insertAuditLog({
   341→      action: 'exchange.requested',
   342→      entityType: 'order_exchange',
   343→      entityId: exchange.id,
   344→      companyId: ctx.company.id,
   345→      organizationId: ctx.company.organizationId,
   346→      userId: ctx.user.id,
   347→      employeeId: ctx.employee.id,
   348→      newValues: {
   349→        originalOrderId: orderItem.order.id,
   350→        originalOrderItemId: orderItem.id,
   351→        newOrgVariantId: d.new_org_variant_id,
   352→        exchangeMethod: d.exchange_method,
   353→        status: initialStatus,
   354→        oldItemPrice,
   355→        newItemPrice,
   356→        priceDifference,
   357→        priceDifferenceStatus,
   358→        reason: d.reason,
   359→      },
   360→    })
   361→
   362→    // 9. Metric event
   363→    insertMetricEvent({
   364→      companyId: ctx.company.id,
   365→      entityType: 'order',
   366→      entityId: orderItem.order.id,
   367→      metricKey: 'exchange.requested',
   368→      numericValue: 1,
   369→      dimensions: {
   370→        exchange_id: exchange.id,
   371→        exchange_method: d.exchange_method,
   372→      },
   373→    })
   374→
   375→    return { success: true, data: { exchangeId: exchange.id } }
   376→  } catch (err) {
   377→    return {
   378→      success: false,
   379→      error: err instanceof Error ? err.message : 'Failed to create exchange request',
   380→    }
   381→  }
   382→}

   388→export async function dispatchExchangeNewItem(
   389→  exchangeId: string,
   390→  options?: {
   391→    /** Universal courier reference field override (migration 015) */
   392→    orderRefNumber?: string
   393→    /** Universal courier item-description override (migration 015) */
   394→    orderDetail?: string
   395→  },
   396→): Promise<ActionResult<{ newExchangeShipmentId: string; exchangeShipmentNumber: string }>> {
   397→  try {
   398→    const ctx = await getWorkspace()
   399→    await requirePermission(ctx, PERMISSIONS.ORDERS_MANAGE)
   400→
   401→    const exchange = await db.orderExchange.findFirst({
   402→      where: { id: exchangeId, companyId: ctx.company.id },
   403→    })
   404→    if (!exchange) return { success: false, error: 'Exchange not found' }
   405→
   406→    // Only valid for courier_replacement with status='requested'
   407→    if (exchange.exchangeMethod !== 'courier_replacement') {
   408→      return {
   409→        success: false,
   410→        error:
   411→          'This action is only valid for courier_replacement exchanges. For customer_self_return, the new item is dispatched automatically after old item verification.',
   412→      }
   413→    }
   414→    if (exchange.status !== 'requested') {
   415→      return {
   416→        success: false,
   417→        error: `Cannot dispatch new item for an exchange with status '${exchange.status}'. Expected 'requested'.`,
   418→      }
   419→    }
   420→
   421→    // Create + reserve the new EXCHANGE SHIPMENT (courier booking happens later via Workbench)
   422→    const { newExchangeShipmentId, exchangeShipmentNumber } = await createAndReserveExchangeShipment(
   423→      exchangeId,
   424→      ctx,
   425→      options,
   426→    )
   427→
   428→    // Update exchange status to 'awaiting_old_item_return' (old item collection
   429→    // happens during this delivery, but manual verification still needed)
   430→    await db.orderExchange.update({
   431→      where: { id: exchangeId },
   432→      data: { status: 'awaiting_old_item_return' },
   433→    })
   434→
   435→    return { success: true, data: { newExchangeShipmentId, exchangeShipmentNumber } }
   436→  } catch (err) {
   437→    return {
   438→      success: false,
   439→      error: err instanceof Error ? err.message : 'Failed to dispatch exchange new item',
   440→    }
   441→  }
   442→}

   510→// ══════════════════════════════════════════════════════════════
   511→// PART 5: MANUAL VERIFICATION — shared by both methods
   512→// This is the ONLY function that processes the old item's return in inventory.
   513→// ══════════════════════════════════════════════════════════════
   514→
   515→export async function verifyOldItemReceived(
   516→  input: VerifyOldItemReceivedInput,
   517→): Promise<ActionResult<{ exchangeStatus: string }>> {
   518→  try {
   519→    const ctx = await getWorkspace()
   520→    // GUARD: inventory.receive OR orders.manage
   521→    await requirePermission(ctx, PERMISSIONS.INVENTORY_RECEIVE)
   522→
   523→    const parsed = verifyOldItemReceivedSchema.safeParse(input)
   524→    if (!parsed.success) {
   525→      return {
   526→        success: false,
   527→        error: parsed.error.issues[0]?.message ?? 'Invalid input',
   528→      }
   529→    }
   530→    const d = parsed.data
   531→
   532→    const exchange = await db.orderExchange.findFirst({
   533→      where: { id: d.exchange_id, companyId: ctx.company.id },
   534→      include: {
   535→        originalOrderItem: {
   536→          select: {
   537→            id: true,
   538→            orgVariantId: true,
   539→            fulfillmentTypeSnapshot: true,
   540→            unitPrice: true,
   541→            reservedLocationId: true,
   542→          },
   543→        },
   544→        originalOrder: {
   545→          select: { id: true, dispatchLocationId: true },
   546→        },
   547→      },
   548→    })
   549→    if (!exchange) return { success: false, error: 'Exchange not found' }
   550→
   551→    // 3. Verify status is one of the awaiting-return states
   552→    const validStatuses = ['awaiting_old_item_return', 'customer_confirmed_shipped']
   553→    if (!validStatuses.includes(exchange.status)) {
   554→      return {
   555→        success: false,
   556→        error: `Cannot verify old item for an exchange with status '${exchange.status}'. Expected one of: ${validStatuses.join(', ')}.`,
   557→      }
   558→    }
   559→
   560→    const locationId =
   561→      exchange.originalOrderItem.reservedLocationId ?? exchange.originalOrder.dispatchLocationId
   562→    if (!locationId) {
   563→      return {
   564→        success: false,
   565→        error: 'Cannot receive returned item — no dispatch location set on the original order.',
   566→      }
   567→    }
   568→
   569→    let inventoryTxnId: string | null = null
   570→    let stockLossId: string | null = null
   571→    const oldItemCost = Number(exchange.oldItemPrice)
   572→
   573→    // 5. IF condition IN ('perfect','good','open_box'): receive into inventory
   574→    // 6. IF condition = 'damaged': create stock_loss_records entry directly
   575→    if (d.condition === 'damaged') {
   576→      // Damaged → straight to stock_loss, no inventory addition
   577→      const lossRecord = await db.stockLossRecord.create({
   578→        data: {
   579→          organizationId: ctx.company.organizationId,
   580→          companyId: ctx.company.id,
   581→          orgVariantId: exchange.originalOrderItem.orgVariantId,
   582→          locationId,
   583→          lossType: 'damaged',
   584→          subType: 'confirmed',
   585→          damageType: 'other',
   586→          quantity: 1,
   587→          costPerUnit: oldItemCost,
   588→          investigationStatus: 'none',
   589→          resolution: 'written_off',
   590→          responsibleParty: 'customer',
   591→          evidenceUrls: JSON.stringify(d.evidence_urls),
   592→          notes: `Damaged exchanged item. ${d.notes || ''}`.trim(),
   593→          reportedById: ctx.employee.id,
   594→          approvedById: ctx.employee.id,
   595→          resolvedById: ctx.employee.id,
   596→          resolvedAt: new Date(),
   597→        },
   598→      })
   599→      stockLossId = lossRecord.id
   600→    } else {
   601→      // Not damaged → add to stock via processInventoryTransaction
   602→      // (same function the /api/inventory/receive-returned-stitched route uses)
   603→      const txnType =
   604→        exchange.originalOrderItem.fulfillmentTypeSnapshot === 'made_to_order'
   605→          ? 'return_stitched_received'
   606→          : 'return_resellable'
   607→
   608→      const txnResult = await processInventoryTransaction({
   609→        orgVariantId: exchange.originalOrderItem.orgVariantId,
   610→        locationId,
   611→        organizationId: ctx.company.organizationId,
   612→        companyId: ctx.company.id,
   613→        employeeId: ctx.employee.id,
   614→        transactionType: txnType,
   615→        quantity: 1,
   616→        costPerUnit: oldItemCost,
   617→        referenceType: 'order',
   618→        referenceId: exchange.originalOrderId,
   619→        notes: `Exchanged item (${d.condition}). ${d.notes || ''}`.trim(),
   620→      })
   621→
   622→      if (!txnResult.success || !txnResult.transactionId) {
   623→        return {
   624→          success: false,
   625→          error: `Failed to receive returned item into inventory: ${txnResult.error}`,
   626→        }
   627→      }
   628→      inventoryTxnId = txnResult.transactionId
   629→    }
   630→
   631→    // 7. UPDATE order_exchanges with verification data
   632→    await db.orderExchange.update({
   633→      where: { id: d.exchange_id },
   634→      data: {
   635→        oldItemCondition: d.condition,
   636→        oldItemVerifiedAt: new Date(),
   637→        oldItemVerifiedBy: ctx.employee.id,
   638→        oldItemEvidenceUrls: JSON.stringify(d.evidence_urls),
   639→        oldItemNotes: d.notes?.trim() || null,
   640→        oldItemInventoryTxnId: inventoryTxnId,
   641→        oldItemStockLossId: stockLossId,
   642→        status: 'old_item_manually_verified',
   643→      },
   644→    })
   645→
   646→    // 8. IF customer_self_return: THIS IS THE GATING POINT.
   647→    //    PROMPT 5 CHANGE: verification now STOPS at 'old_item_manually_verified'.
   648→    //    The actual shipment creation + dispatch is a separate explicit step,
   649→    //    triggered by the "Send Replacement Order" button in the Exchange Detail UI.
   650→    //    This lets staff select a courier, address, and city before dispatching.
   651→    //    For courier_replacement, the new item was already dispatched in Part 3
   652→    //    (via dispatchExchangeNewItem), so we mark as completed here.
   653→    if (exchange.exchangeMethod === 'courier_replacement') {
   654→      // For courier_replacement: the new item was already dispatched earlier,
   655→      // so we can mark the exchange as completed now.
   656→      await db.orderExchange.update({
   657→        where: { id: d.exchange_id },
   658→        data: {
   659→          status: 'completed',
   660→          completedAt: new Date(),
   661→        },
   662→      })
   663→    }
   664→    // For customer_self_return: status stays at 'old_item_manually_verified'.
   665→    // The "Send Replacement Order" button (Prompt 5 Phase 1) calls
   666→    // dispatchReplacementForSelfReturnExchange() to create + dispatch the shipment.
   667→
   668→    // 10. Audit log
   669→    insertAuditLog({
   670→      action: 'exchange.old_item_verified',
   671→      entityType: 'order_exchange',
   672→      entityId: d.exchange_id,
   673→      companyId: ctx.company.id,
   674→      organizationId: ctx.company.organizationId,
   675→      userId: ctx.user.id,
   676→      employeeId: ctx.employee.id,
   677→      newValues: {
   678→        condition: d.condition,
   679→        inventoryTxnId,
   680→        stockLossId,
   681→        exchangeMethod: exchange.exchangeMethod,
   682→      },
   683→    })
   684→
   685→    // Metric event
   686→    insertMetricEvent({
   687→      companyId: ctx.company.id,
   688→      entityType: 'order_exchange',
   689→      entityId: d.exchange_id,
   690→      metricKey: 'exchange.old_item_verified',
   691→      numericValue: oldItemCost,
   692→      dimensions: {
   693→        condition: d.condition,
   694→        exchange_method: exchange.exchangeMethod,
   695→      },
   696→    })
   697→
   698→    // Also fire exchange.completed metric — ONLY for courier_replacement
   699→    // (customer_self_return no longer auto-completes; completion happens when
   700→    // the "Send Replacement Order" button is clicked and the shipment is dispatched)
   701→    if (exchange.exchangeMethod === 'courier_replacement') {
   702→      insertMetricEvent({
   703→        companyId: ctx.company.id,
   704→        entityType: 'order_exchange',
   705→        entityId: d.exchange_id,
   706→        metricKey: 'exchange.completed',
   707→        numericValue: Number(exchange.newItemPrice) - Number(exchange.oldItemPrice),
   708→        dimensions: {
   709→          exchange_method: exchange.exchangeMethod,
   710→          condition: d.condition,
   711→        },
   712→      })
   713→    }
   714→
   715→    return {
   716→      success: true,
   717→      data: {
   718→        exchangeStatus: exchange.exchangeMethod === 'courier_replacement'
   719→          ? 'completed'
   720→          : 'old_item_manually_verified',
   721→      },
   722→    }
   723→  } catch (err) {
   724→    return {
   725→      success: false,
   726→      error: err instanceof Error ? err.message : 'Failed to verify old item',
   727→    }
   728→  }
   729→}

   732→// PART 6: PRICE DIFFERENCE SETTLEMENT
   733→// ══════════════════════════════════════════════════════════════
   734→
   735→export async function settlePriceDifference(
   736→  input: SettlePriceDifferenceInput,
   737→): Promise<ActionResult> {
   738→  try {
   739→    const ctx = await getWorkspace()
   740→    await requirePermission(ctx, PERMISSIONS.ORDERS_MANAGE)
   741→
   742→    const parsed = settlePriceDifferenceSchema.safeParse(input)
   743→    if (!parsed.success) {
   744→      return {
   745→        success: false,
   746→        error: parsed.error.issues[0]?.message ?? 'Invalid input',
   747→      }
   748→    }
   749→    const d = parsed.data
   750→
   751→    const exchange = await db.orderExchange.findFirst({
   752→      where: { id: d.exchange_id, companyId: ctx.company.id },
   753→      include: {
   754→        exchangeShipments: {
   755→          select: { estimatedDeliveryCharge: true },
   756→          orderBy: { createdAt: 'desc' },
   757→          take: 1,
   758→        },
   759→      },
   760→    })
   761→    if (!exchange) return { success: false, error: 'Exchange not found' }
   762→
   763→    // ── Compute refundAmount for refund_due case ──
   764→    let refundAmount: number | null = null
   765→    if (d.settlement_type === 'refunded_to_customer') {
   766→      const rawPriceDiff = Math.abs(Number(exchange.priceDifference))
   767→      const estimatedDeliveryCharge = exchange.exchangeShipments[0]?.estimatedDeliveryCharge
   768→        ? Number(exchange.exchangeShipments[0].estimatedDeliveryCharge)
   769→        : 0
   770→
   771→      // Check company setting
   772→      const orderSettings = await db.companyOrderSetting.findUnique({
   773→        where: { companyId: ctx.company.id },
   774→        select: { deductDeliveryChargeFromRefund: true },
   775→      })
   776→      const deductDelivery = orderSettings?.deductDeliveryChargeFromRefund ?? false
   777→
   778→      if (deductDelivery) {
   779→        refundAmount = Math.max(0, rawPriceDiff - estimatedDeliveryCharge)
   780→      } else {
   781→        // Default: customer gets full price difference back; delivery charge is business-absorbed
   782→        refundAmount = rawPriceDiff
   783→      }
   784→    }
   785→
   786→    await db.orderExchange.update({
   787→      where: { id: d.exchange_id },
   788→      data: {
   789→        priceDifferenceSettledAmount: d.settled_amount,
   790→        priceDifferenceSettledAt: new Date(),
   791→        priceDifferenceSettledBy: ctx.employee.id,
   792→        priceDifferenceStatus: 'settled',
   793→        // Refund tracking (migration 014)
   794→        ...(d.settlement_type === 'refunded_to_customer'
   795→          ? {
   796→              refundMethod: d.refund_method ?? null,
   797→              refundReference: d.refund_reference?.trim() || null,
   798→              refundProcessedAt: new Date(),
   799→              refundProcessedBy: ctx.employee.id,
   800→              refundAmount,
   801→            }
   802→          : {}),
   803→      },
   804→    })
   805→
   806→    insertAuditLog({
   807→      action: 'exchange.price_difference_settled',
   808→      entityType: 'order_exchange',
   809→      entityId: d.exchange_id,
   810→      companyId: ctx.company.id,
   811→      organizationId: ctx.company.organizationId,
   812→      userId: ctx.user.id,
   813→      employeeId: ctx.employee.id,
   814→      newValues: {
   815→        settledAmount: d.settled_amount,
   816→        settlementType: d.settlement_type,
   817→        ...(refundAmount !== null ? { refundAmount, refundMethod: d.refund_method } : {}),
   818→      },
   819→    })
   820→
   821→    return { success: true }
   822→  } catch (err) {
   823→    return {
   824→      success: false,
   825→      error: err instanceof Error ? err.message : 'Failed to settle price difference',
   826→    }
   827→  }
   828→}

   834→export async function markExchangeAsNotReturned(
   835→  input: MarkNotReturnedInput,
   836→): Promise<ActionResult> {
   837→  try {
   838→    const ctx = await getWorkspace()
   839→    await requirePermission(ctx, PERMISSIONS.ORDERS_MANAGE)
   840→
   841→    const parsed = markNotReturnedSchema.safeParse(input)
   842→    if (!parsed.success) {
   843→      return {
   844→        success: false,
   845→        error: parsed.error.issues[0]?.message ?? 'Invalid input',
   846→      }
   847→    }

   849→
   850→    const exchange = await db.orderExchange.findFirst({
   851→      where: { id: d.exchange_id, companyId: ctx.company.id },
   852→      include: {
   853→        originalOrder: { select: { customerId: true } },
   854→      },
   855→    })
   856→    if (!exchange) return { success: false, error: 'Exchange not found' }
   857→
   858→    // 3. Verify not already in a terminal state
   859→    const terminalStates = ['completed', 'customer_did_not_return', 'cancelled']
   860→    if (terminalStates.includes(exchange.status)) {
   861→      return {
   862→        success: false,
   863→        error: `Cannot mark as not returned — exchange is already in terminal status '${exchange.status}'.`,
   864→      }
   865→    }
   866→
   867→    // 4. UPDATE order_exchanges (the DB CHECK constraint enforces status consistency)
   868→    await db.orderExchange.update({
   869→      where: { id: d.exchange_id },
   870→      data: {
   871→        markedAsNotReturned: true,
   872→        notReturnedReason: d.not_returned_reason,
   873→        notReturnedRecoveryStatus: d.recovery_status,
   874→        notReturnedRecoveryAmount: d.recovery_amount ?? null,
   875→        status: 'customer_did_not_return',
   876→      },
   877→    })
   878→
   879→    // 5. Call the existing flagCustomer() action with reason "Exchange item not returned"
   880→    if (exchange.originalOrder.customerId) {
   881→      await flagCustomer(
   882→        exchange.originalOrder.customerId,
   883→        'Exchange item not returned',
   884→      ).catch((e) => {
   885→        // Non-fatal — the exchange is still marked, just the flag failed
   886→        console.error('[exchange] flagCustomer failed:', e)
   887→      })
   888→    }
   889→
   890→    // 6. NOTE: for customer_self_return, the new item was NEVER dispatched
   891→    //    (sequential gating held), so nothing to reverse.
   892→    //    For courier_replacement, the new item WAS dispatched — this is now
   893→    //    an unrecovered loss tracked via not_returned_recovery_amount.
   894→
   895→    // 7. Audit log
   896→    insertAuditLog({
   897→      action: 'exchange.not_returned',
   898→      entityType: 'order_exchange',
   899→      entityId: d.exchange_id,
   900→      companyId: ctx.company.id,
   901→      organizationId: ctx.company.organizationId,
   902→      userId: ctx.user.id,
   903→      employeeId: ctx.employee.id,
   904→      newValues: {
   905→        reason: d.not_returned_reason,
   906→        recoveryStatus: d.recovery_status,
   907→        recoveryAmount: d.recovery_amount ?? null,
   908→        exchangeMethod: exchange.exchangeMethod,
   909→        customerId: exchange.originalOrder.customerId,
   910→      },
   911→    })
   912→
   913→    // Metric event
   914→    insertMetricEvent({
   915→      companyId: ctx.company.id,
   916→      entityType: 'order_exchange',
   917→      entityId: d.exchange_id,
   918→      metricKey: 'exchange.not_returned',
   919→      numericValue: Number(exchange.oldItemPrice), // loss value
   920→      dimensions: {
   921→        exchange_method: exchange.exchangeMethod,
   922→        recovery_status: d.recovery_status,
   923→      },
   924→    })
   925→
   926→    return { success: true }
   927→  } catch (err) {
   928→    return {
   929→      success: false,
   930→      error: err instanceof Error ? err.message : 'Failed to mark exchange as not returned',
   931→    }
   932→  }
   933→}

  1007→    dateTo?: string
  1008→    limit?: number
  1009→    offset?: number
  1010→  } = {},
  1011→): Promise<ActionResult<{
  1012→  exchanges: Array<{
  1013→    id: string
  1014→    exchangeMethod: string
  1015→    status: string

  1020→    priceDifferenceStatus: string

  1085→          originalOrderId: e.originalOrderId,
  1086→          originalOrder: e.originalOrder,
  1087→          newOrderId: e.newOrderId,
  1088→          newOrder: e.newOrder,
  1089→          exchangeShipments: e.exchangeShipments,
  1090→        })),
  1091→        total,
  1092→      },
  1093→    }
  1094→  } catch (err) {
  1095→    return {
  1096→      success: false,
  1097→      error: err instanceof Error ? err.message : 'Failed to list exchanges',
  1098→    }
  1099→  }

  1195→
  1196→export async function listOverdueExchanges(
  1197→  daysThreshold: number = 7,
  1198→): Promise<ActionResult<{
  1199→  exchanges: Array<{
  1200→    id: string
  1201→    exchangeMethod: string
  1202→    status: string
  1203→    reason: string
  1204→    requestedAt: Date
  1205→    customerConfirmedShippedAt: Date | null
  1206→    originalOrder: { flowopsOrderNumber: string; customer: { name: string } }
  1207→    daysWaiting: number
  1208→  }>
  1209→}>> {
  1210→  try {
  1211→    const ctx = await getWorkspace()
  1212→    const threshold = new Date(Date.now() - daysThreshold * 86400000)
  1213→
  1214→    // Exchanges in waiting states where the relevant latest timestamp is older than threshold
  1215→    const exchanges = await db.orderExchange.findMany({
  1216→      where: {
  1217→        companyId: ctx.company.id,
  1218→        status: {
  1219→          in: [
  1220→            'awaiting_old_item_return',
  1221→            'awaiting_customer_to_ship_old_item',
  1222→            'customer_confirmed_shipped',
  1223→          ],
  1224→        },
  1225→        // For awaiting_customer_to_ship_old_item: requestedAt is the relevant timestamp
  1226→        // For customer_confirmed_shipped: customerConfirmedShippedAt is the relevant timestamp
  1227→        // For awaiting_old_item_return: requestedAt is the relevant timestamp (courier_replacement)
  1228→        OR: [
  1229→          { status: 'awaiting_customer_to_ship_old_item', requestedAt: { lt: threshold } },
  1230→          {
  1231→            status: 'customer_confirmed_shipped',
  1232→            customerConfirmedShippedAt: { lt: threshold },
  1233→          },
  1234→          { status: 'awaiting_old_item_return', requestedAt: { lt: threshold } },
  1235→        ],
  1236→      },
  1237→      orderBy: { requestedAt: 'asc' },
  1238→      include: {
  1239→        originalOrder: {
  1240→          select: {
  1241→            flowopsOrderNumber: true,
  1242→            customer: { select: { name: true } },
  1243→          },
  1244→        },
  1245→      },
  1246→    })
  1247→
  1248→    const now = Date.now()
  1249→    return {
  1250→      success: true,
  1251→      data: {
  1252→        exchanges: exchanges.map((e) => {
  1253→          // The "waiting since" timestamp depends on the current status
  1254→          const waitingSince =
  1255→            e.status === 'customer_confirmed_shipped'
  1256→              ? e.customerConfirmedShippedAt ?? e.requestedAt
  1257→              : e.requestedAt
  1258→          const daysWaiting = Math.floor(
  1259→            (now - new Date(waitingSince).getTime()) / 86400000,
  1260→          )
  1261→          return {
  1262→            id: e.id,
  1263→            exchangeMethod: e.exchangeMethod,
  1264→            status: e.status,
  1265→            reason: e.reason,
  1266→            requestedAt: e.requestedAt,
  1267→            customerConfirmedShippedAt: e.customerConfirmedShippedAt,
  1268→            originalOrder: e.originalOrder,
  1269→            daysWaiting,
  1270→          }
  1271→        }),
  1272→      },
  1273→    }
  1274→  } catch (err) {
  1275→    return {
  1276→      success: false,
  1277→      error: err instanceof Error ? err.message : 'Failed to list overdue exchanges',
  1278→    }
  1279→  }
  1280→}
