   168→    queryKey: ['product', productId],

   476→function PricingTab({
   477→  productId,
   478→  variants,
   479→  onUpdate,
   480→}: {
   481→  productId: string
   482→  variants: ProductVariant[]
   483→  onUpdate: () => void
   484→}) {
   485→  const [drafts, setDrafts] = useState<Record<string, { salePrice: number | null; comparePrice: number | null }>>(
   486→    () =>
   487→      Object.fromEntries(
   488→        variants.map((v) => [
   489→          v.id,
   490→          { salePrice: v.salePrice, comparePrice: v.comparePrice },
   491→        ]),
   492→      ),
   493→  )
   494→  const [saving, setSaving] = useState<string | null>(null)
   495→
   496→  function setDraft(
   497→    variantId: string,
   498→    field: 'salePrice' | 'comparePrice',
   499→    value: number | null,
   500→  ) {
   501→    setDrafts((d) => ({
   502→      ...d,
   503→      [variantId]: { ...d[variantId], [field]: value },
   504→    }))
   505→  }
   506→
   507→  async function save(variantId: string) {
   508→    const draft = drafts[variantId]
   509→    if (!draft) return
   510→    if (draft.salePrice == null || draft.salePrice <= 0) {
   511→      toast.error('Sale price must be a positive number.')
   512→      return
   513→    }
   514→    setSaving(variantId)
   515→    try {
   516→      // POST /api/products/[id]/pricing expects { pricing: [{ org_variant_id, sale_price, compare_price? }] }
   517→      await api.post(`/api/products/${productId}/pricing`, {
   518→        pricing: [
   519→          {
   520→            org_variant_id: variantId,
   521→            sale_price: draft.salePrice,
   522→            ...(draft.comparePrice != null ? { compare_price: draft.comparePrice } : {}),
   523→          },
   524→        ],
   525→      })
   526→      toast.success('Pricing updated.')
   527→      onUpdate()
   528→    } catch (err) {
   529→      toast.error(err instanceof FetchError ? err.message : 'Failed to update pricing.')
   530→    } finally {
   531→      setSaving(null)
   532→    }
   533→  }
   534→
   535→  return (
   536→    <Card>
   537→      <CardHeader>
   538→        <CardTitle className="text-base flex items-center gap-2">
   539→          <Tag className="h-4 w-4 text-primary" /> Your company pricing
   540→        </CardTitle>
   541→        <CardDescription>
   542→          Set the sale and compare-at prices for your storefront.
   543→        </CardDescription>
   544→      </CardHeader>
   545→      <CardContent className="p-0">
   546→        <div className="overflow-x-auto">
   547→          <Table>
   548→            <TableHeader>
   549→              <TableRow>
   550→                <TableHead>SKU</TableHead>
   551→                <TableHead className="text-right">Cost</TableHead>
   552→                <TableHead>Sale price</TableHead>
   553→                <TableHead>Compare price</TableHead>
   554→                <TableHead className="text-right">Action</TableHead>
   555→              </TableRow>
   556→            </TableHeader>
   557→            <TableBody>
   558→              {variants.map((v) => {
   559→                const draft = drafts[v.id] ?? { salePrice: v.salePrice, comparePrice: v.comparePrice }
   560→                const dirty =
   561→                  draft.salePrice !== v.salePrice ||
   562→                  draft.comparePrice !== v.comparePrice
   563→                const canSave =
   564→                  dirty &&
   565→                  draft.salePrice != null &&
   566→                  draft.salePrice > 0 &&
   567→                  (draft.comparePrice == null || draft.comparePrice > draft.salePrice)
   568→                return (
   569→                  <TableRow key={v.id}>
   570→                    <TableCell className="font-mono text-xs">{v.sku}</TableCell>
   571→                    <TableCell className="text-right text-xs text-muted-foreground">
   572→                      {formatMoney(v.costPrice)}
   573→                    </TableCell>
   574→                    <TableCell>
   575→                      <Input
   576→                        type="number"
   577→                        min="0"
   578→                        step="0.01"
   579→                        value={draft.salePrice ?? ''}
   580→                        onChange={(e) =>
   581→                          setDraft(
   582→                            v.id,
   583→                            'salePrice',
   584→                            e.target.value ? Number(e.target.value) : null,
   585→                          )
   586→                        }
   587→                        className="h-8 w-32 text-xs"
   588→                      />
   589→                    </TableCell>
   590→                    <TableCell>
   591→                      <Input
   592→                        type="number"
   593→                        min="0"
   594→                        step="0.01"
   595→                        value={draft.comparePrice ?? ''}
   596→                        onChange={(e) =>
   597→                          setDraft(
   598→                            v.id,
   599→                            'comparePrice',
   600→                            e.target.value ? Number(e.target.value) : null,
   601→                          )
   602→                        }
   603→                        className="h-8 w-32 text-xs"
   604→                      />
   605→                    </TableCell>
   606→                    <TableCell className="text-right">
   607→                      <Button
   608→                        size="sm"
   609→                        variant={dirty ? 'default' : 'outline'}
   610→                        disabled={!canSave || saving === v.id}
   611→                        onClick={() => save(v.id)}
   612→                      >
   613→                        {saving === v.id ? (
   614→                          <Loader2 className="h-3.5 w-3.5 animate-spin" />
   615→                        ) : (
   616→                          <Save className="h-3.5 w-3.5" />
   617→                        )}
   618→                        Save
   619→                      </Button>

  1232→/** Interactive is_active switch with optimistic update + revert on error. */
  1233→function VariantActiveSwitch({
  1234→  productId,
  1235→  variantId,
  1236→  isActive,
  1237→  canEdit,
  1238→}: {
  1239→  productId: string
  1240→  variantId: string
  1241→  isActive: boolean
  1242→  canEdit: boolean
  1243→}) {
  1244→  const queryClient = useQueryClient()
  1245→
  1246→  const mutation = useMutation({
  1247→    mutationFn: (newVal: boolean) =>
  1248→      api.post(`/api/products/${productId}/variants/${variantId}/toggle`, {
  1249→        is_active: newVal,
  1250→      }),
  1251→    onMutate: async (newVal) => {
  1252→      // Optimistic update — flip the cached variant's isActive immediately.
  1253→      await queryClient.cancelQueries({ queryKey: ['product', productId] })
  1254→      const prev = queryClient.getQueryData<{ product: ProductDetail }>(['product', productId])
  1255→      if (prev) {
  1256→        queryClient.setQueryData<{ product: ProductDetail }>(['product', productId], {
  1257→          product: {
  1258→            ...prev.product,
  1259→            variants: prev.product.variants.map((v) =>
  1260→              v.id === variantId ? { ...v, isActive: newVal } : v
  1261→            ),
  1262→          },
  1263→        })
  1264→      }
  1265→      return { prev }
  1266→    },
  1267→    onError: (err, _newVal, context) => {
  1268→      // Revert the optimistic update.
  1269→      if (context?.prev) {
  1270→        queryClient.setQueryData(['product', productId], context.prev)
  1271→      }
  1272→      toast.error(err instanceof FetchError ? err.message : 'Failed to toggle variant.')
  1273→    },
  1274→    onSuccess: () => {
  1275→      queryClient.invalidateQueries({ queryKey: ['product', productId] })
  1276→      toast.success('Variant status updated.')
  1277→    },
  1278→  })
  1279→
  1280→  return (
  1281→    <Switch
  1282→      checked={isActive}
  1283→      disabled={!canEdit || mutation.isPending}
  1284→      onCheckedChange={(v) => mutation.mutate(v)}
  1285→      aria-label="Toggle variant active"
  1286→    />
  1287→  )
  1288→}
