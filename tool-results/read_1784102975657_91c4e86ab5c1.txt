   241→const PKR = new Intl.NumberFormat('en-PK', { maximumFractionDigits: 0 })
   242→function formatPKR(n: number): string {
   243→  return `Rs. ${PKR.format(n)}`
   244→}
   245→
   246→function formatDate(iso: string): string {
   247→  try {
   248→    return new Date(iso).toLocaleDateString('en-PK', {
   249→      day: '2-digit',
   250→      month: 'short',
   251→      year: 'numeric',
   252→    })
   253→  } catch {
   254→    return iso
   255→  }
   256→}
   257→
   258→function getErrorMessage(err: unknown): string {
   259→  if (err instanceof FetchError) return err.message
   260→  if (err instanceof Error) return err.message
   261→  return 'Something went wrong. Please try again.'
   262→}
   263→
   264→function isThisMonth(iso: string): boolean {
   265→  const d = new Date(iso)
   266→  const now = new Date()
   267→  return d.getFullYear() === now.getFullYear() && d.getMonth() === now.getMonth()
   268→}

   312→        const q = search.toLowerCase()
   313→        return (
   314→          r.productTitle.toLowerCase().includes(q) ||
   315→          r.sku.toLowerCase().includes(q) ||
   316→          r.location.toLowerCase().includes(q)
   317→        )

   319→      return true
   320→    })

   339→              Refresh

   354→          value={lossesQuery.isLoading ? undefined : String(stats.open)}

   356→          tone={stats.open > 0 ? 'amber' : 'gray'}
   357→          loading={lossesQuery.isLoading}
   358→        />
   359→        <StatCard
   360→          label="Loss value this month"
   361→          value={lossesQuery.isLoading ? undefined : formatPKR(stats.monthLossValue)}

   364→          loading={lossesQuery.isLoading}
   365→        />
   366→        <StatCard
   367→          label="Recovered amount"
   368→          value={lossesQuery.isLoading ? undefined : formatPKR(stats.recovered)}
   369→          icon={<CheckCircle2 className="h-5 w-5" />}
   370→          tone="emerald"
   371→          loading={lossesQuery.isLoading}
   372→        />
   373→      </div>
   374→
   375→      {/* ── Filter bar ─────────────────────────────────────────────────────── */}
   376→      <div className="flex flex-col sm:flex-row gap-2 sm:items-center">
   377→        <div className="relative flex-1">
   378→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   379→          <Input
   380→            placeholder="Search product, SKU, or location…"
   381→            className="pl-9"
   382→            value={search}
   383→            onChange={(e) => setSearch(e.target.value)}
   384→          />
   385→        </div>
   386→        <Select value={typeFilter} onValueChange={setTypeFilter}>
   387→          <SelectTrigger className="w-full sm:w-44">
   388→            <SelectValue placeholder="All types" />
   389→          </SelectTrigger>
   390→          <SelectContent>
   391→            <SelectItem value="all">All types</SelectItem>
   392→            <SelectItem value="damaged">Damaged</SelectItem>
   393→            <SelectItem value="theft">Theft</SelectItem>
   394→            <SelectItem value="missing">Missing</SelectItem>
   395→            <SelectItem value="transit_loss">Transit Loss</SelectItem>
   396→          </SelectContent>
   397→        </Select>
   398→        <Select value={statusFilter} onValueChange={setStatusFilter}>
   399→          <SelectTrigger className="w-full sm:w-40">
   400→            <SelectValue placeholder="All statuses" />
   401→          </SelectTrigger>
   402→          <SelectContent>
   403→            <SelectItem value="all">All statuses</SelectItem>
   404→            <SelectItem value="open">Open</SelectItem>

   508→                    )
   509→                  })}
   510→                </TableBody>
   511→              </Table>

   515→            </p>
   516→          </CardContent>
   517→        </Card>
   518→      )}
   519→
   520→      {/* ── Report dialog ─────────────────────────────────────────────────── */}
   521→      {canReport && (
   522→        <ReportLossDialog
   523→          open={reportOpen}
   524→          onOpenChange={setReportOpen}
   525→          onSuccess={() => {
   526→            void queryClient.invalidateQueries({ queryKey: ['stock-loss'] })
   527→            void queryClient.invalidateQueries({ queryKey: ['inventory-dashboard'] })
   528→            setReportOpen(false)
   529→          }}
   530→        />
   531→      )}
   532→
   533→      {/* ── Resolve dialog ────────────────────────────────────────────────── */}
   534→      {resolveTarget && (canReport || canManage) && (
   535→        <ResolveDialog
   536→          target={resolveTarget}
   537→          canManage={canManage}

   542→          onSuccess={() => {
   543→            void queryClient.invalidateQueries({ queryKey: ['stock-loss'] })
   544→            void queryClient.invalidateQueries({ queryKey: ['inventory-dashboard'] })
   545→            setResolveTarget(null)
   546→          }}
   547→        />
   548→      )}
   549→    </div>
   550→  )
   551→}

  1079→function ResolveDialog({
  1080→  target,
  1081→  canManage,
  1082→  open,
  1083→  onOpenChange,
  1084→  onSuccess,
  1085→}: {
  1086→  target: StockLossRow
  1087→  canManage: boolean
  1088→  open: boolean
  1089→  onOpenChange: (v: boolean) => void
  1090→  onSuccess: () => void
  1091→}) {
  1092→  const [resolution, setResolution] = useState<LossResolution>('written_off')
  1093→  const [investigationStatus, setInvestigationStatus] = useState<InvestigationStatus>('closed')
  1094→  const [responsibleParty, setResponsibleParty] = useState<ResponsibleParty | ''>(
  1095→    target.responsibleParty ?? '',
  1096→  )
  1097→  const [policeReportRef, setPoliceReportRef] = useState('')
  1098→  const [insuranceClaimRef, setInsuranceClaimRef] = useState('')
  1099→  const [insuranceRecovered, setInsuranceRecovered] = useState('0')
  1100→  const [courierClaimRef, setCourierClaimRef] = useState('')
  1101→  const [courierClaimStatus, setCourierClaimStatus] = useState<
  1102→    'not_filed' | 'filed' | 'accepted' | 'rejected'
  1103→  >('not_filed')
  1104→  const [courierRecovered, setCourierRecovered] = useState('0')
  1105→  const [notes, setNotes] = useState('')
  1106→
  1107→  // Reset state when target changes.
  1108→  useEffect(() => {
  1109→    if (open) {
  1110→      setResolution('written_off')
  1111→      setInvestigationStatus('closed')
  1112→      setResponsibleParty(target.responsibleParty ?? '')
  1113→      setPoliceReportRef('')
  1114→      setInsuranceClaimRef('')
  1115→      setInsuranceRecovered('0')
  1116→      setCourierClaimRef('')
  1117→      setCourierClaimStatus('not_filed')
  1118→      setCourierRecovered('0')
  1119→      setNotes('')
  1120→    }
  1121→  }, [open, target])
  1122→
  1123→  const showInsuranceFields =
  1124→    resolution === 'claim_accepted' || resolution === 'recovered'
  1125→  const showCourierFields = target.lossType === 'transit_loss' || target.lossType === 'damaged'
  1126→  const showPoliceFields = target.lossType === 'theft' || target.lossType === 'missing'
  1127→
  1128→  const resolveMutation = useMutation({
  1129→    mutationFn: async (payload: ResolvePayload) =>
  1130→      api.patch(`/api/stock-loss/${target.id}`, payload),
  1131→    onSuccess: () => {
  1132→      toast.success('Stock loss resolved.')
  1133→      onSuccess()
  1134→    },
  1135→    onError: (err) => toast.error(getErrorMessage(err)),
  1136→  })
  1137→
  1138→  const handleSubmit = () => {
  1139→    const payload: ResolvePayload = {
  1140→      resolution,
  1141→      investigation_status: investigationStatus,
  1142→      responsible_party: responsibleParty || undefined,
  1143→      police_report_ref: showPoliceFields && policeReportRef.trim() ? policeReportRef.trim() : undefined,
  1144→      insurance_claim_ref: showInsuranceFields && insuranceClaimRef.trim() ? insuranceClaimRef.trim() : undefined,
