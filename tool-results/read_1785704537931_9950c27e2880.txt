    91→// ──────────────────────────────────────────────────────────────
    92→// Processes ALL order_items for an order and attempts to reserve
    93→// stock for each based on its fulfillment_type_snapshot.
    94→//
    95→// For stock_based items:
    96→//   - If sufficient available stock → reserveStockForOrder(), fulfillment_status='reserved'
    97→//   - If insufficient + inventory_policy='continue' → fulfillment_status='backordered'
    98→//   - If insufficient + inventory_policy='deny' → outcome='failed'
    99→//
   100→// For made_to_order items:
   101→//   - checkAndFulfillMadeToOrderVariant() checks returned stock first
   102→//   - If existing_stock → fulfillment_status='reserved', returned_stitched_used=true
   103→//   - If fresh_production → fulfillment_status='reserved', production_order_id set
   104→//   - If error → outcome='failed'
   105→//
   106→// After processing: if ANY item is 'backordered', order status → 'partially_backordered'.
   107→// If ALL items are 'reserved', order stays 'confirmed'.
   108→//
   109→// Called from:
   110→//   - createManualOrder() IF the order auto-confirmed at creation
   111→//   - confirmOrder() when a 'pending' order is manually confirmed
   112→//   - dispatchOrderAction() if status is still 'pending' at dispatch time
   113→
   114→async function reserveOrderStock(
   115→  orderId: string,
   116→  ctx: { company: { id: string; organizationId: string }; user: { id: string }; employee: { id: string } },
   117→): Promise<{
   118→  success: boolean
   119→  results: Array<{ orderItemId: string; outcome: 'reserved' | 'backordered' | 'failed'; reason?: string }>
   120→}> {
   121→  const order = await db.order.findUnique({
   122→    where: { id: orderId },
   123→    select: { dispatchLocationId: true, companyId: true, organizationId: true },
   124→  })
   125→  if (!order) return { success: false, results: [] }
   126→
   127→  const items = await db.orderItem.findMany({
   128→    where: { orderId },
   129→    include: {
   130→      orgVariant: {
   131→        select: { id: true, fulfillmentType: true, inventoryPolicy: true, stitchingCharges: true },
   132→      },
   133→    },
   134→  })
   135→
   136→  const results: Array<{ orderItemId: string; outcome: 'reserved' | 'backordered' | 'failed'; reason?: string }> = []
   137→  let hasBackordered = false
   138→
   139→  for (const item of items) {
   140→    if (item.fulfillmentStatus === 'reserved' || item.fulfillmentStatus === 'dispatched') {
   141→      // Already processed (e.g. re-confirmation attempt) — skip
   142→      results.push({ orderItemId: item.id, outcome: 'reserved' })
   143→      continue
   144→    }
   145→
   146→    const locationId = item.reservedLocationId ?? order.dispatchLocationId
   147→    if (!locationId) {
   148→      results.push({ orderItemId: item.id, outcome: 'failed', reason: 'No dispatch location set' })
   149→      continue
   150→    }
   151→
   152→    if (item.fulfillmentTypeSnapshot === 'stock_based') {
   153→      // Check available stock at the dispatch location
   154→      const pool = await db.inventoryPool.findUnique({
   155→        where: {
   156→          orgVariantId_locationId: {
   157→            orgVariantId: item.orgVariantId,
   158→            locationId,
   159→          },
   160→        },
   161→      })
   162→
   163→      const available = pool ? pool.onHand - pool.reserved : 0
   164→
   165→      if (available >= item.quantity) {
   166→        // Sufficient stock — reserve it
   167→        const reserveResult = await reserveStockForOrder({
   168→          orgVariantId: item.orgVariantId,
   169→          locationId,
   170→          organizationId: order.organizationId,
   171→          companyId: order.companyId,
   172→          employeeId: ctx.employee.id,
   173→          quantity: item.quantity,
   174→          orderId,
   175→        })
   176→
   177→        if (reserveResult.success) {
   178→          await db.orderItem.update({
   179→            where: { id: item.id },
   180→            data: { fulfillmentStatus: 'reserved', reservedLocationId: locationId },
   181→          })
   182→          results.push({ orderItemId: item.id, outcome: 'reserved' })
   183→        } else {
   184→          results.push({ orderItemId: item.id, outcome: 'failed', reason: reserveResult.error })
   185→        }
   186→      } else if (item.orgVariant.inventoryPolicy === 'continue') {
   187→        // Insufficient stock but backordering allowed
   188→        await db.orderItem.update({
   189→          where: { id: item.id },
   190→          data: { fulfillmentStatus: 'backordered', backorderedAt: new Date() },
   191→        })
   192→        await insertMetricEvent({
   193→          companyId: order.companyId,
   194→          entityType: 'product',
   195→          entityId: item.orgVariantId,
   196→          metricKey: 'order.backordered',
   197→          numericValue: item.quantity,
   198→          dimensions: { order_id: orderId },
   199→        }).catch(() => {})
   200→        hasBackordered = true
   201→        results.push({
   202→          orderItemId: item.id,
   203→          outcome: 'backordered',
   204→          reason: `Available: ${available}, required: ${item.quantity}`,
   205→        })
   206→      } else {
   207→        // Insufficient stock + policy='deny' — fail
   208→        results.push({
   209→          orderItemId: item.id,
   210→          outcome: 'failed',
   211→          reason: `Insufficient stock (available: ${available}, required: ${item.quantity}) and inventory_policy='deny'`,
   212→        })
   213→      }
   214→    } else if (item.fulfillmentTypeSnapshot === 'made_to_order') {
   215→      // Made-to-order: check returned stock first, then trigger production
   216→      const mtoResult = await checkAndFulfillMadeToOrderVariant(
   217→        item.orgVariantId,
   218→        item.quantity,
   219→        order.companyId,
   220→        locationId,
   221→      )
   222→
   223→      if (mtoResult.source === 'existing_stock' && mtoResult.locationId) {
   224→        // Returned stock available — reserve it
   225→        const reserveResult = await reserveStockForOrder({
   226→          orgVariantId: item.orgVariantId,
   227→          locationId: mtoResult.locationId,
   228→          organizationId: order.organizationId,
   229→          companyId: order.companyId,
   230→          employeeId: ctx.employee.id,
   231→          quantity: item.quantity,
   232→          orderId,
   233→        })
   234→
   235→        if (reserveResult.success) {
   236→          await db.orderItem.update({
   237→            where: { id: item.id },
   238→            data: {
   239→              fulfillmentStatus: 'reserved',
   240→              returnedStitchedUsed: true,
   241→              reservedLocationId: mtoResult.locationId,
   242→            },
   243→          })
   244→          await insertMetricEvent({
   245→            companyId: order.companyId,
   246→            entityType: 'product',
   247→            entityId: item.orgVariantId,
   248→            metricKey: 'order.made_to_order_from_returned_stock',
   249→            numericValue: 1,
   250→            dimensions: { order_id: orderId, stitching_cost_saved: Number(item.orgVariant.stitchingCharges) || 0 },
   251→          }).catch(() => {})
   252→          results.push({ orderItemId: item.id, outcome: 'reserved' })
   253→        } else {
   254→          results.push({ orderItemId: item.id, outcome: 'failed', reason: reserveResult.error })
   255→        }
   256→      } else if (mtoResult.source === 'fresh_production' && mtoResult.productionOrderId) {
   257→        // Fresh production triggered — link the production order
   258→        await db.orderItem.update({
   259→          where: { id: item.id },
   260→          data: {
   261→            fulfillmentStatus: 'reserved',
   262→            productionOrderId: mtoResult.productionOrderId,
   263→          },
   264→        })
   265→        // Also link the order_item back to the production order
   266→        await db.productionOrder.update({
   267→          where: { id: mtoResult.productionOrderId },
   268→          data: { orderItemId: item.id },
   269→        })
   270→        await insertMetricEvent({
   271→          companyId: order.companyId,
   272→          entityType: 'product',
   273→          entityId: item.orgVariantId,
   274→          metricKey: 'order.made_to_order_production_triggered',
   275→          numericValue: 1,
   276→          dimensions: { order_id: orderId, production_order_id: mtoResult.productionOrderId },
   277→        }).catch(() => {})
   278→        results.push({ orderItemId: item.id, outcome: 'reserved' })
   279→      } else {
   280→        // Error (no fabric source, insufficient fabric, etc.)
   281→        results.push({
   282→          orderItemId: item.id,
   283→          outcome: 'failed',
   284→          reason: mtoResult.error ?? 'Made-to-order fulfillment failed',
   285→        })
   286→      }
   287→    }
   288→  }
   289→
   290→  // Update order status based on results
   291→  if (hasBackordered) {
   292→    await db.order.update({
   293→      where: { id: orderId },
   294→      data: { status: 'partially_backordered' },
   295→    })
   296→  }
   297→  // If no backordered items, order stays 'confirmed' (or whatever it was)
   298→
   299→  return { success: true, results }
   300→}

   302→// ──────────────────────────────────────────────────────────────

   305→
   306→export async function createManualOrder(
   307→  input: CreateManualOrderInput,
   308→): Promise<ActionResult<{
   309→  orderId: string
   310→  flowopsOrderNumber: string
   311→  orderItems: Array<{ id: string; orgVariantId: string; quantity: number }>
   312→}>> {
   313→  try {
   314→    const ctx = await getWorkspace()
   315→    await requirePermission(ctx, PERMISSIONS.ORDERS_CREATE)
   316→
   317→    // 1. Validate
   318→    const parsed = createManualOrderSchema.safeParse(input)
   319→    if (!parsed.success) {
   320→      return {
   321→        success: false,
   322→        error: parsed.error.issues[0]?.message ?? 'Invalid order data',
   323→      }
   324→    }
   325→    const d = parsed.data
   326→
   327→    // 2. Resolve customer (existing customer_id OR inline new_customer).
   328→    //    Also resolve optional saved address/phone selection and recipient_name.
   329→    let customerId: string
   330→    let customerName: string
   331→    let usedCustomerAddressId: string | null = null
   332→    let usedCustomerPhoneId: string | null = null
   333→    let saveAddressForNextTime = d.save_address_for_next_time ?? false
   334→    // Whether the delivery_address came from a saved customer_addresses row
   335→    // (vs. typed as a one-off). Used to decide whether to bump lastUsedAt.
   336→    let selectedSavedAddressId: string | null = null
   337→
   338→    if (d.customer_id) {
   339→      // Existing customer path — verify they belong to this org.
   340→      const existing = await db.customer.findFirst({
   341→        where: { id: d.customer_id, organizationId: ctx.company.organizationId },
   342→        select: { id: true, name: true },
   343→      })
   344→      if (!existing) return { success: false, error: 'Customer not found' }
   345→      customerId = existing.id
   346→      customerName = existing.name
   347→
   348→      // If a saved address was selected, verify it belongs to this customer.
   349→      if (d.used_customer_address_id) {
   350→        const addr = await db.customerAddress.findFirst({
   351→          where: {
   352→            id: d.used_customer_address_id,
   353→            customerId: existing.id,
   354→            organizationId: ctx.company.organizationId,
   355→          },
   356→          select: { id: true, address: true, city: true },
   357→        })
   358→        if (addr) {
   359→          usedCustomerAddressId = addr.id
   360→          selectedSavedAddressId = addr.id
   361→          // The delivery_address snapshot is the order's own editable copy.
   362→          // If the caller passed the same address text, we trust it; if they
   363→          // passed different text (edited it), we still record the link to
   364→          // the saved address for lastUsedAt tracking but keep the edited text
   365→          // as the order's snapshot. Either way, delivery_address comes from
   366→          // the input — we do NOT override it here.
   367→        }
   368→      }
   369→
   370→      // If a saved phone was selected, verify it belongs to this customer.
   371→      if (d.used_customer_phone_id) {
   372→        const phone = await db.customerPhone.findFirst({
   373→          where: {
   374→            id: d.used_customer_phone_id,
   375→            customerId: existing.id,
   376→            organizationId: ctx.company.organizationId,
   377→          },
   378→          select: { id: true },
   379→        })
   380→        if (phone) {
   381→          usedCustomerPhoneId = phone.id
   382→        }
   383→      }
   384→    } else if (d.new_customer) {
   385→      // Inline new-customer path — create the customer + their phones +
   386→      // addresses now via the Customer Management System. The first phone is
   387→      // primary and the first address is default (validated by createCustomerSchema).
   388→      const createResult = await createCustomer(d.new_customer)
   389→      if (!createResult.success || !createResult.data) {
   390→        return { success: false, error: createResult.error ?? 'Failed to create customer' }
   391→      }
   392→      customerId = createResult.data.customerId
   393→      customerName = d.new_customer.name.trim()
   394→      // For a brand-new customer, the first address they provided IS the
   395→      // saved default address. Mark it as the selected address so we bump
   396→      // its lastUsedAt after order creation.
   397→      const newAddr = await db.customerAddress.findFirst({
   398→        where: { customerId, isDefault: true },
   399→        select: { id: true },
   400→      })
   401→      if (newAddr) {
   402→        usedCustomerAddressId = newAddr.id
   403→        selectedSavedAddressId = newAddr.id
   404→      }
   405→      // Same for the primary phone.
   406→      const newPhone = await db.customerPhone.findFirst({
   407→        where: { customerId, isPrimary: true },
   408→        select: { id: true },
   409→      })
   410→      if (newPhone) {
   411→        usedCustomerPhoneId = newPhone.id
   412→      }
   413→      // For a brand-new customer, we already saved the address as part of
   414→      // createCustomer — don't double-save it.
   415→      saveAddressForNextTime = false
   416→    } else {
   417→      return { success: false, error: 'Either customer_id or new_customer is required' }
   418→    }
   419→
   420→    // 3. recipient_name defaults to the customer's name if not explicitly overridden.
   421→    const recipientName = (d.recipient_name && d.recipient_name.trim()) || customerName
   422→
   423→    // 3. Fetch variants + their pricing for this company + fulfillment_type snapshot
   424→    const variantIds = d.items.map((i) => i.org_variant_id)
   425→    const variants = await db.orgProductVariant.findMany({
   426→      where: { id: { in: variantIds }, organizationId: ctx.company.organizationId },
   427→      include: {
   428→        companyPricing: { where: { companyId: ctx.company.id } },
   429→      },
   430→    })
   431→
   432→    if (variants.length !== variantIds.length) {
   433→      return { success: false, error: 'One or more variants not found in this organization' }
   434→    }
   435→
   436→    // 4. Compute subtotal + build order items data
   437→    let subtotal = 0
   438→    const orderItemsData: Array<{
   439→      orgVariantId: string
   440→      quantity: number
   441→      unitPrice: number
   442→      lineTotal: number
   443→      fulfillmentTypeSnapshot: string
   444→    }> = []
   445→
   446→    for (const item of d.items) {
   447→      const variant = variants.find((v) => v.id === item.org_variant_id)
   448→      if (!variant) continue
   449→
   450→      // Use provided unit_price OR fall back to company pricing OR variant cost
   451→      const unitPrice =
   452→        item.unit_price ??
   453→        (variant.companyPricing[0]?.salePrice
   454→          ? Number(variant.companyPricing[0].salePrice)
   455→          : Number(variant.costPrice))
   456→
   457→      const lineTotal = item.quantity * unitPrice
   458→      subtotal += lineTotal
   459→
   460→      orderItemsData.push({
   461→        orgVariantId: item.org_variant_id,
   462→        quantity: item.quantity,
   463→        unitPrice,
   464→        lineTotal,
   465→        fulfillmentTypeSnapshot: variant.fulfillmentType,
   466→      })
   467→    }
   468→
   469→    // 5. Compute totals
   470→    const discountAmount = d.discount_amount ?? 0
   471→    const courierCharges = d.courier_charges ?? 0
   472→    const totalOrderValue = subtotal + courierCharges - discountAmount
   473→
   474→    // 6. Determine payment status + source
   475→    let paymentStatus: string = 'cod_pending'
   476→    let paymentSource: string = 'cod_native'
   477→    let advanceAmount: number | null = null
   478→    let advancePaymentMethod: string | null = null
   479→    let advancePaymentReference: string | null = null
   480→    let advancePaidAt: Date | null = null
   481→
   482→    if (d.payment_type === 'partial_advance') {
   483→      paymentStatus = 'advance_paid'
   484→      paymentSource = 'manual_conversion'
   485→      advanceAmount = d.advance_amount ?? 0
   486→      advancePaymentMethod = d.advance_payment_method || null
   487→      advancePaymentReference = d.advance_payment_reference || null
   488→      advancePaidAt = new Date()
   489→    } else if (d.payment_type === 'fully_prepaid') {
   490→      paymentStatus = 'fully_prepaid'
   491→      paymentSource = 'manual_conversion'
   492→      advanceAmount = totalOrderValue
   493→      advancePaymentMethod = d.advance_payment_method || null
   494→      advancePaymentReference = d.advance_payment_reference || null
   495→      advancePaidAt = new Date()
   496→    }
   497→
   498→    // remainingCodAmount is NOT a GENERATED column in the DB — the application
   499→    // MUST compute it as totalOrderValue - (advanceAmount ?? 0).
   500→    const remainingCodAmount = totalOrderValue - (advanceAmount ?? 0)
   501→
   502→    // 7. Determine initial order status
   503→    let orderStatus: string = 'pending'
   504→    let confirmedAt: Date | null = null
   505→    let skippedConfirmation = false
   506→
   507→    if (paymentStatus === 'advance_paid' || paymentStatus === 'fully_prepaid') {
   508→      // Payment itself is the confirmation signal — bypasses require_order_confirmation
   509→      orderStatus = 'confirmed'
   510→      confirmedAt = new Date()
   511→    } else {
   512→      // COD order — check company_order_settings
   513→      const settings = await db.companyOrderSetting.findUnique({
   514→        where: { companyId: ctx.company.id },
   515→      })
   516→      if (!settings?.requireOrderConfirmation) {
   517→        orderStatus = 'confirmed'
   518→        confirmedAt = new Date()
   519→        skippedConfirmation = true
   520→      }
   521→    }
   522→
   523→    // 8. Generate order number
   524→    const flowopsOrderNumber = await generateOrderNumber(ctx.company.id)
   525→
   526→    // 9. Create the order
   527→    const order = await db.order.create({
   528→      data: {
   529→        organizationId: ctx.company.organizationId,
   530→        companyId: ctx.company.id,
   531→        flowopsOrderNumber,
   532→        orderSource: 'manual',
   533→        customerId,
   534→        // Customer Management System integration (migration 002):
   535→        //   - recipientName: the name to deliver to (defaults to customer.name)
   536→        //   - usedCustomerAddressId / usedCustomerPhoneId: which saved
   537→        //     customer_addresses / customer_phones rows were used for this
   538→        //     order, for lastUsedAt tracking and reporting. Nullable when a
   539→        //     one-off address/phone was typed without saving.
   540→        recipientName,
   541→        usedCustomerAddressId,
   542→        usedCustomerPhoneId,
   543→        status: orderStatus,
   544→        paymentType: d.payment_type,
   545→        paymentStatus,
   546→        paymentSource,
   547→        subtotal,
   548→        discountAmount: discountAmount || null,
   549→        discountReason: d.discount_reason || null,
   550→        courierCharges: courierCharges || null,
   551→        totalOrderValue,
   552→        advanceAmount,
   553→        advancePaymentMethod,
   554→        advancePaymentReference,
   555→        advancePaidAt,
   556→        remainingCodAmount,
   557→        deliveryAddress: d.delivery_address,
   558→        deliveryCity: d.delivery_city,
   559→        courierName: d.courier_name || null,
   560→        dispatchLocationId: d.dispatch_location_id,
   561→        notesForCourier: d.notes_for_courier || null,
   562→        skippedConfirmation,
   563→        confirmedAt,
   564→        createdBy: ctx.employee.id,
   565→      },
   566→    })
   567→
   568→    // 10. Create order items (fulfillment_status = 'reserved' as PLACEHOLDER)
   569→    const createdItems: Array<{ id: string; orgVariantId: string; quantity: number }> = []
   570→    for (const itemData of orderItemsData) {
   571→      const item = await db.orderItem.create({
   572→        data: {
   573→          orderId: order.id,
   574→          orgVariantId: itemData.orgVariantId,

   580→          fulfillmentTypeSnapshot: itemData.fulfillmentTypeSnapshot,
   581→          reservedLocationId: d.dispatch_location_id,
   582→        },
   583→      })

   612→    //   a. If a saved address was selected, bump its lastUsedAt to NOW() so
   613→    //      the UI can sort saved addresses by recency.

   615→    //      save_address_for_next_time=true, persist it as a new
   616→    //      customer_addresses row (and link it via usedCustomerAddressId
   617→    //      for future tracking). This is the "Save this address for next
   618→    //      time" UX from Step 3's frontend.
   619→    //   c. Recompute cached customer stats (total_orders_count, etc.) and
   620→    //      auto-flag if RTO threshold crossed.
   621→    if (selectedSavedAddressId) {
   622→      await markAddressAsUsed(selectedSavedAddressId)
   623→    } else if (saveAddressForNextTime && d.delivery_address && d.delivery_city) {

   629→          customerId,

   633→          city: d.delivery_city,
   634→          isDefault: false,

   640→        where: { id: order.id },
   641→        data: { usedCustomerAddressId: savedAddr.id },
   642→      })
   643→    }
   644→
   645→    await updateCustomerStats(customerId)
   646→
   647→    // 13. If the order auto-confirmed (payment-driven or company setting),
   648→    // run the stock reservation logic immediately.
   649→    if (orderStatus === 'confirmed') {
   650→      await reserveOrderStock(order.id, ctx)
   651→    }
   652→
   653→    await insertMetricEvent({
   654→      companyId: ctx.company.id,
   655→      entityType: 'order',
   656→      entityId: order.id,
   657→      metricKey: 'order.created',
   658→      numericValue: Number(totalOrderValue),
   659→      dimensions: { order_source: 'manual', payment_type: d.payment_type, company_id: ctx.company.id },
   660→    }).catch(() => {})
   661→
   662→    return {
   663→      success: true,
   664→      data: { orderId: order.id, flowopsOrderNumber, orderItems: createdItems },
   665→    }
   666→  } catch (err) {
   667→    return {
   668→      success: false,
   669→      error: err instanceof Error ? err.message : 'Failed to create order',
   670→    }
   671→  }
   672→}

   936→export async function confirmOrder(orderId: string): Promise<ActionResult> {
   937→  try {
   938→    const ctx = await getWorkspace()
   939→    await requirePermission(ctx, PERMISSIONS.ORDERS_MANAGE)
   940→
   941→    const order = await db.order.findFirst({
   942→      where: { id: orderId, companyId: ctx.company.id },
   943→    })
   944→    if (!order) return { success: false, error: 'Order not found' }
   945→    if (order.status !== 'pending') {
   946→      return { success: false, error: `Order is already ${order.status} (cannot confirm)` }
   947→    }
   948→
   949→    await db.order.update({
   950→      where: { id: orderId },
   951→      data: { status: 'confirmed', confirmedAt: new Date() },

   959→      organizationId: ctx.company.organizationId,
   960→      userId: ctx.user.id,
   961→      employeeId: ctx.employee.id,
   962→      oldValues: { status: 'pending' },
   963→      newValues: { status: 'confirmed' },
   964→    })
   965→
   966→    // Step 3: Run stock reservation for all items.
   967→    // This may flip the order to 'partially_backordered' if some items
   968→    // can't be fulfilled immediately.
   969→    const reserveResult = await reserveOrderStock(orderId, ctx)
   970→    if (!reserveResult.success) {
   971→      // Non-fatal — the order is confirmed, but reservation had issues.
   972→      // The results array has per-item details.
   973→    }
   974→
   975→    await insertMetricEvent({
   976→      companyId: ctx.company.id,
   977→      entityType: 'order',
   978→      entityId: orderId,
   979→      metricKey: 'order.confirmed',
   980→      numericValue: Number(order.totalOrderValue),
   981→      dimensions: { confirmation_method: 'manual' },
   982→    }).catch(() => {})
   983→
   984→    // Recompute cached customer stats (order count may change if this was a
   985→    // re-confirmation, and the status transition affects value/rto calcs).
   986→    // Non-fatal — never break the confirm action on a stats failure.
   987→    await updateCustomerStats(order.customerId).catch(() => {})
   988→
   989→    return { success: true }
   990→  } catch (err) {
   991→    return {
   992→      success: false,
   993→      error: err instanceof Error ? err.message : 'Failed to confirm order',
   994→    }
   995→  }
   996→}

  1705→ *     order.packed_at IS NOT NULL before proceeding.
  1706→ */
  1707→export async function dispatchOrderAction(
  1708→  orderId: string,
  1709→  trackingNumber: string,
  1710→  courierName?: string,
  1711→): Promise<ActionResult> {
  1712→  try {
  1713→    const ctx = await getWorkspace()
  1714→    await requirePermission(ctx, PERMISSIONS.ORDERS_FULFILL)
  1715→
  1716→    const order = await db.order.findFirst({
  1717→      where: { id: orderId, companyId: ctx.company.id },
  1718→    })
  1719→    if (!order) return { success: false, error: 'Order not found' }
  1720→
  1721→    // Cannot dispatch already-dispatched/delivered/cancelled orders
  1722→    if (['dispatched', 'delivered', 'rto', 'cancelled', 'refunded'].includes(order.status)) {
  1723→      return { success: false, error: `Cannot dispatch an order that is already ${order.status}` }
  1724→    }
  1725→
  1726→    // Check packing requirement
  1727→    const settings = await db.companyOrderSetting.findUnique({
  1728→      where: { companyId: ctx.company.id },
  1729→    })
  1730→    if (settings?.requirePackingStep && !order.packedAt) {
  1731→      return {
  1732→        success: false,
  1733→        error: 'This order must be marked as packed before dispatching. Use markOrderPacked() first.',
  1734→      }
  1735→    }
  1736→
  1737→    // If order is still 'pending' (confirmation was skipped), run the
  1738→    // full confirmation + reservation logic inline first.
  1739→    if (order.status === 'pending') {
  1740→      await db.order.update({
  1741→        where: { id: orderId },
  1742→        data: { status: 'confirmed', confirmedAt: new Date() },
  1743→      })
  1744→      await reserveOrderStock(orderId, ctx)
  1745→    }
  1746→
  1747→    // Re-fetch the order to get the updated status after reservation
  1748→    const updatedOrder = await db.order.findFirst({
  1749→      where: { id: orderId },
  1750→      select: { status: true, dispatchLocationId: true, organizationId: true },
  1751→    })
  1752→    if (!updatedOrder) return { success: false, error: 'Order not found after reservation' }
  1753→
  1754→    // Check for backordered items — BLOCK dispatch if any exist
  1755→    const backorderedItems = await db.orderItem.findMany({
  1756→      where: { orderId, fulfillmentStatus: 'backordered' },
  1757→      include: {
  1758→        orgVariant: { select: { sku: true } },
  1759→      },
  1760→    })
  1761→
  1762→    if (backorderedItems.length > 0) {
  1763→      const itemList = backorderedItems
  1764→        .map((i) => `${i.orgVariant.sku} (qty: ${i.quantity})`)
  1765→        .join(', ')
  1766→      return {
  1767→        success: false,
  1768→        error: `Cannot dispatch: ${backorderedItems.length} item(s) are still backordered: ${itemList}. Receive stock or cancel those items first.`,
  1769→      }
  1770→    }
  1771→
  1772→    // Dispatch each reserved item
  1773→    const itemsToDispatch = await db.orderItem.findMany({
  1774→      where: { orderId, fulfillmentStatus: 'reserved' },
  1775→    })
  1776→
  1777→    for (const item of itemsToDispatch) {
  1778→      const locationId = item.reservedLocationId ?? updatedOrder.dispatchLocationId
  1779→      if (!locationId) {
  1780→        return { success: false, error: `Order item ${item.id} has no dispatch location` }
  1781→      }
  1782→
  1783→      const dispatchResult = await dispatchInventory({
  1784→        orgVariantId: item.orgVariantId,
  1785→        locationId,
  1786→        organizationId: updatedOrder.organizationId,
  1787→        companyId: ctx.company.id,
  1788→        employeeId: ctx.employee.id,
  1789→        quantity: item.quantity,
  1790→        orderId,
  1791→      })
  1792→
  1793→      if (!dispatchResult.success) {
  1794→        return {
  1795→          success: false,
  1796→          error: `Failed to dispatch item: ${dispatchResult.error}`,
  1797→        }
  1798→      }
  1799→
  1800→      // Mark item as dispatched
  1801→      await db.orderItem.update({
  1802→        where: { id: item.id },
  1803→        data: { fulfillmentStatus: 'dispatched', fulfilledAt: new Date() },
  1804→      })
  1805→    }
  1806→
  1807→    // Update order status — the backfill_order_timestamps() trigger
  1808→    // will auto-set confirmedAt/packedAt if they were still NULL.
  1809→    await db.order.update({
  1810→      where: { id: orderId },
  1811→      data: {
  1812→        status: 'dispatched',
  1813→        dispatchedAt: new Date(),
  1814→        trackingNumber: trackingNumber || null,
  1815→        courierName: courierName || order.courierName,
  1816→      },
  1817→    })
  1818→
  1819→    await insertAuditLog({
  1820→      action: 'order.dispatched',
  1821→      entityType: 'order',
  1822→      entityId: orderId,
  1823→      companyId: ctx.company.id,
  1824→      organizationId: ctx.company.organizationId,
  1825→      userId: ctx.user.id,
  1826→      employeeId: ctx.employee.id,
  1827→      oldValues: { status: updatedOrder.status },
  1828→      newValues: {
  1829→        status: 'dispatched',
  1830→        trackingNumber,
  1831→        courierName,
  1832→        itemsDispatched: itemsToDispatch.length,
  1833→      },
  1834→    })
  1835→
  1836→    const timeToDispatchHours = order.createdAt
  1837→      ? Math.round((new Date().getTime() - new Date(order.createdAt).getTime()) / (1000 * 60 * 60))
  1838→      : 0
  1839→    await insertMetricEvent({
  1840→      companyId: ctx.company.id,
  1841→      entityType: 'order',
  1842→      entityId: orderId,
  1843→      metricKey: 'order.dispatched',
  1844→      numericValue: Number(order.totalOrderValue),
  1845→      dimensions: {
  1846→        courier_name: courierName || order.courierName,
  1847→        employee_id: ctx.employee.id,
  1848→        skipped_confirmation: updatedOrder.status === 'pending',
  1849→        skipped_packing: !order.packedAt,
  1850→        time_to_dispatch_hours: timeToDispatchHours,
  1851→      },
  1852→    }).catch(() => {})
  1853→
  1854→    // Update customer stats
  1855→    const orderWithCustomer = await db.order.findUnique({
  1856→      where: { id: orderId },
  1857→      select: { customerId: true },
  1858→    })
  1859→    if (orderWithCustomer) {
  1860→      await updateCustomerStats(orderWithCustomer.customerId)
  1861→    }
  1862→
  1863→    return { success: true }
  1864→  } catch (err) {
  1865→    return {
  1866→      success: false,
  1867→      error: err instanceof Error ? err.message : 'Failed to dispatch order',
  1868→    }
  1869→  }
  1870→}
