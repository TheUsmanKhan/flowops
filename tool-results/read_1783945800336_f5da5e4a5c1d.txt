    98→const categoryFormSchema = z.object({
    99→  name: z.string().min(2, 'Name must be at least 2 characters').max(100, 'Name is too long'),
   100→  parentId: z.string().optional().or(z.literal('')),
   101→  imageUrl: z
   102→    .string()
   103→    .url('Enter a valid image URL')
   104→    .optional()
   105→    .or(z.literal('')),
   106→  displayOrder: z.number().int('Must be a whole number').min(0, 'Must be 0 or greater').optional(),
   107→})
   108→type CategoryFormValues = z.infer<typeof categoryFormSchema>
   109→
   110→const brandFormSchema = z.object({
   111→  name: z.string().min(2, 'Name must be at least 2 characters').max(100, 'Name is too long'),
   112→  logoUrl: z
   113→    .string()
   114→    .url('Enter a valid logo URL')
   115→    .optional()
   116→    .or(z.literal('')),
   117→  isActive: z.boolean(),
   118→})
   119→type BrandFormValues = z.infer<typeof brandFormSchema>
   120→
   121→const attributeFormSchema = z.object({
   122→  name: z
   123→    .string()
   124→    .min(2, 'Key must be at least 2 characters')
   125→    .max(50, 'Key is too long')
   126→    .regex(/^[a-z0-9_]+$/, 'Use lowercase letters, numbers and underscores'),
   127→  displayName: z
   128→    .string()
   129→    .min(2, 'Display name must be at least 2 characters')
   130→    .max(100, 'Display name is too long'),
   131→  attributeType: z.enum(['select', 'color']),
   132→  displayOrder: z.number().int().min(0).optional(),
   133→})
   134→type AttributeFormValues = z.infer<typeof attributeFormSchema>
   135→
   136→const attributeValueFormSchema = z.object({
   137→  value: z.string().min(1, 'Value is required').max(100),
   138→  displayValue: z.string().min(1, 'Display value is required').max(50),
   139→  colorHex: z
   140→    .string()
   141→    .regex(/^#[0-9a-fA-F]{6}$/, 'Use #RRGGBB format')
   142→    .optional()
   143→    .or(z.literal('')),
   144→  displayOrder: z.number().int().min(0).optional(),
   145→})
   146→type AttributeValueFormValues = z.infer<typeof attributeValueFormSchema>

   278→  title: string

   328→function CategoriesTab() {
   329→  const queryClient = useQueryClient()
   330→  const [addOpen, setAddOpen] = useState(false)
   331→  const [addSubParentId, setAddSubParentId] = useState<string | null>(null)
   332→  const [editTarget, setEditTarget] = useState<Category | null>(null)
   333→  const [deleteTarget, setDeleteTarget] = useState<Category | null>(null)
   334→  const [deleteError, setDeleteError] = useState<string | null>(null)
   335→  const [expanded, setExpanded] = useState<Set<string>>(new Set())
   336→
   337→  const { data, isLoading, isError, refetch, isFetching } = useQuery<{
   338→    categories: Category[]
   339→  }>({
   340→    queryKey: ['categories'],
   341→    queryFn: () => api.get<{ categories: Category[] }>('/api/categories'),
   342→    staleTime: 30_000,
   343→  })
   344→
   345→  const categories = data?.categories ?? []
   346→
   347→  // Build a 2-level tree: roots + their direct children.
   348→  const { roots, childrenByParent } = useMemo(() => {
   349→    const roots = categories.filter((c) => !c.parentId)
   350→    const childrenByParent = new Map<string, Category[]>()
   351→    for (const c of categories) {
   352→      if (c.parentId) {
   353→        const arr = childrenByParent.get(c.parentId) ?? []
   354→        arr.push(c)
   355→        childrenByParent.set(c.parentId, arr)
   356→      }
   357→    }
   358→    // Sort children by name for stable ordering.
   359→    childrenByParent.forEach((arr) => arr.sort((a, b) => a.name.localeCompare(b.name)))
   360→    roots.sort((a, b) => a.name.localeCompare(b.name))
   361→    return { roots, childrenByParent }
   362→  }, [categories])
   363→
   364→  const createMutation = useMutation({
   365→    mutationFn: (values: CategoryFormValues) =>
   366→      api.post('/api/categories', {
   367→        name: values.name,
   368→        parentId: values.parentId || undefined,
   369→        imageUrl: values.imageUrl || undefined,
   370→        displayOrder: values.displayOrder ?? 0,
   371→      }),
   372→    onSuccess: () => {
   373→      queryClient.invalidateQueries({ queryKey: ['categories'] })
   374→      toast.success('Category created')
   375→      setAddOpen(false)
   376→      setAddSubParentId(null)
   377→    },
   378→    onError: (err) => {
   379→      toast.error(
   380→        err instanceof FetchError ? err.message : 'Failed to create category.',
   381→      )
   382→    },
   383→  })
   384→
   385→  const updateMutation = useMutation({
   386→    mutationFn: ({ id, values }: { id: string; values: CategoryFormValues }) =>
   387→      api.patch(`/api/catalog/categories/${id}`, {
   388→        name: values.name,
   389→        parentId: values.parentId || null,
   390→        imageUrl: values.imageUrl || null,
   391→        displayOrder: values.displayOrder ?? 0,
   392→      }),
   393→    onSuccess: () => {
   394→      queryClient.invalidateQueries({ queryKey: ['categories'] })
   395→      toast.success('Category updated')
   396→      setEditTarget(null)
   397→    },
   398→    onError: (err) => {
   399→      toast.error(
   400→        err instanceof FetchError ? err.message : 'Failed to update category.',
   401→      )
   402→    },
   403→  })
   404→
   405→  const deleteMutation = useMutation({
   406→    mutationFn: (id: string) => api.delete(`/api/catalog/categories/${id}`),
   407→    onSuccess: () => {
   408→      queryClient.invalidateQueries({ queryKey: ['categories'] })
   409→      toast.success('Category deleted')
   410→      setDeleteTarget(null)
   411→      setDeleteError(null)
   412→    },
   413→    onError: (err) => {
   414→      if (err instanceof FetchError && err.status === 409) {
   415→        setDeleteError(err.message)
   416→      } else {
   417→        toast.error(
   418→          err instanceof FetchError ? err.message : 'Failed to delete category.',
   419→        )
   420→      }
   421→    },
   422→  })
   423→
   424→  function toggleExpand(id: string) {
   425→    setExpanded((prev) => {
   426→      const next = new Set(prev)
   427→      if (next.has(id)) next.delete(id)
   428→      else next.add(id)
   429→      return next
   430→    })
   431→  }
   432→
   433→  if (isLoading) return <CategoriesSkeleton />
   434→
   435→  if (isError) return <ErrorState onRetry={() => refetch()} />
   436→
   437→  if (categories.length === 0) {
   438→    return (
   439→      <>
   440→        <EmptyState
   441→          icon={Folder}
   442→          title="No categories yet"
   443→          description="Categories help you group products into a browsable hierarchy. Create your first one to get started."
   444→          actionLabel="Add Root Category"
   445→          onAction={() => setAddOpen(true)}
   446→        />
   447→        <CategoryDialog
   448→          open={addOpen}
   449→          onOpenChange={setAddOpen}
   450→          mode="create"
   451→          categories={categories}
   452→          isPending={createMutation.isPending}
   453→          onSubmit={(v) => createMutation.mutate(v)}
   454→        />
   455→      </>
   456→    )
   457→  }
   458→
   459→  return (
   460→    <div className="space-y-4">
   461→      <div className="flex items-center justify-between gap-3">
   462→        <p className="text-xs text-muted-foreground">
   463→          {isFetching ? 'Refreshing…' : `${categories.length} categor${categories.length === 1 ? 'y' : 'ies'}`}
   464→        </p>
   465→        <Button size="sm" onClick={() => setAddOpen(true)}>
   466→          <Plus className="h-4 w-4" /> Add Root Category
   467→        </Button>
   468→      </div>
   469→
   470→      <Card>
   471→        <CardContent className="p-0">
   472→          <ul className="divide-y">
   473→            {roots.map((root) => {
   474→              const kids = childrenByParent.get(root.id) ?? []
   475→              const isOpen = expanded.has(root.id)
   476→              return (
   477→                <li key={root.id}>
   478→                  <CategoryRow
   479→                    category={root}
   480→                    isRoot
   481→                    hasChildren={kids.length > 0}
   482→                    expanded={isOpen}
   483→                    onToggle={() => toggleExpand(root.id)}
   484→                    onEdit={() => setEditTarget(root)}
   485→                    onDelete={() => {
   486→                      setDeleteTarget(root)
   487→                      setDeleteError(null)
   488→                    }}
   489→                    onAddSub={() => setAddSubParentId(root.id)}
   490→                  />
   491→                  {isOpen && kids.length > 0 && (
   492→                    <ul className="divide-y bg-muted/20">
   493→                      {kids.map((kid) => (
   494→                        <li key={kid.id}>
   495→                          <CategoryRow
   496→                            category={kid}
   497→                            isRoot={false}
   498→                            onEdit={() => setEditTarget(kid)}
   499→                            onDelete={() => {
   500→                              setDeleteTarget(kid)
   501→                              setDeleteError(null)
   502→                            }}
   503→                          />
   504→                        </li>
   505→                      ))}
   506→                    </ul>
   507→                  )}
   508→                </li>
   509→              )
   510→            })}
   511→          </ul>
   512→        </CardContent>
   513→      </Card>
   514→
   515→      <CategoryDialog
   516→        open={addOpen}
   517→        onOpenChange={setAddOpen}
   518→        mode="create"
   519→        categories={categories}
   520→        isPending={createMutation.isPending}
   521→        onSubmit={(v) => createMutation.mutate(v)}
   522→      />
   523→
   524→      {editTarget && (
   525→        <CategoryDialog
   526→          open={!!editTarget}
   527→          onOpenChange={(v) => !v && setEditTarget(null)}
   528→          mode="edit"
   529→          category={editTarget}
   530→          categories={categories}
   531→          isPending={updateMutation.isPending}
   532→          onSubmit={(v) =>
   533→            updateMutation.mutate({ id: editTarget.id, values: v })
   534→          }
   535→        />
   536→      )}
   537→
   538→      <CategoryDialog
   539→        open={addSubParentId !== null}
   540→        onOpenChange={(v) => !v && setAddSubParentId(null)}
   541→        mode="create"
   542→        lockedParentId={addSubParentId}
   543→        categories={categories}
   544→        isPending={createMutation.isPending}
   545→        onSubmit={(v) => createMutation.mutate(v)}
   546→      />
   547→
   548→      <DeleteConfirmDialog
   549→        open={!!deleteTarget}
   550→        onOpenChange={(v) => {
   551→          if (!v) {
   552→            setDeleteTarget(null)
   553→            setDeleteError(null)
   554→          }
   555→        }}
   556→        title="Delete category"
   557→        description="This permanently removes the category. Products will not be deleted but will lose this category assignment."
   558→        itemName={deleteTarget?.name ?? ''}
   559→        isPending={deleteMutation.isPending}
   560→        error={deleteError}
   561→        onConfirm={() => deleteTarget && deleteMutation.mutate(deleteTarget.id)}
   562→      />
   563→    </div>
   564→  )
   565→}

   567→function CategoryRow({
   568→  category,
   569→  isRoot,
   570→  hasChildren,
   571→  expanded,
   572→  onToggle,
   573→  onEdit,
   574→  onDelete,
   575→  onAddSub,
   576→}: {
   577→  category: Category
   578→  isRoot: boolean
   579→  hasChildren?: boolean
   580→  expanded?: boolean
   581→  onToggle?: () => void
   582→  onEdit: () => void
   583→  onDelete: () => void
   584→  onAddSub?: () => void
   585→}) {
   586→  return (
   587→    <div
   588→      className={cn(
   589→        'flex items-center gap-2 px-3 py-2.5 sm:px-4 transition-colors hover:bg-muted/30',
   590→        !isRoot && 'pl-10 sm:pl-12',
   591→      )}
   592→    >
   593→      {/* Expand toggle (roots only) */}
   594→      {isRoot ? (
   595→        <button
   596→          onClick={onToggle}
   597→          className="flex h-6 w-6 shrink-0 items-center justify-center rounded text-muted-foreground hover:bg-muted"
   598→          aria-label={expanded ? 'Collapse' : 'Expand'}
   599→          disabled={!hasChildren}
   600→        >
   601→          {hasChildren ? (
   602→            <ChevronRight
   603→              className={cn(
   604→                'h-4 w-4 transition-transform',
   605→                expanded && 'rotate-90',
   606→              )}
   607→            />
   608→          ) : (
   609→            <span className="h-1 w-1 rounded-full bg-muted-foreground/30" />
   610→          )}
   611→        </button>
   612→      ) : (
   613→        <span className="flex h-6 w-6 shrink-0 items-center justify-center text-muted-foreground/40">
   614→          <ChevronRight className="h-3 w-3 rotate-90" />
   615→        </span>
   616→      )}
   617→
   618→      <Folder
   619→        className={cn(
   620→          'h-4 w-4 shrink-0',
   621→          isRoot ? 'text-primary' : 'text-muted-foreground',
   622→        )}
