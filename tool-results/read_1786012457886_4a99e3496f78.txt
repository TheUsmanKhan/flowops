   143→function getErrorMessage(err: unknown): string {

   149→function parseQty(v: string): number {
   150→  if (v === '') return 0
   151→  const n = Number(v)
   152→  return Number.isFinite(n) && n >= 0 ? Math.floor(n) : 0
   153→}
   154→
   155→function parsePrice(v: string): number {
   156→  if (v === '') return 0
   157→  const n = Number(v)
   158→  return Number.isFinite(n) && n >= 0 ? n : 0
   159→}

   669→  // ── Upload payment proof + persist URL to the order ───────────────────────
   670→  // Returns true on success, false on any failure. NEVER throws.
   671→  async function uploadPaymentProof(orderId: string): Promise<boolean> {
   672→    if (!paymentProofFile) return true
   673→
   674→    setUploadingProof(true)
   675→    try {
   676→      const fd = new FormData()
   677→      fd.append('file', paymentProofFile)
   678→      const uploadRes = await fetch(
   679→        `/api/upload?type=payment-proofs&id=${orderId}`,
   680→        { method: 'POST', body: fd, credentials: 'include', headers: getSessionToken() ? { Authorization: `Bearer ${getSessionToken()}` } : {} },
   681→      )
   682→      const uploadText = await uploadRes.text()
   683→      let uploadBody: unknown = null
   684→      if (uploadText) {
   685→        try {
   686→          uploadBody = JSON.parse(uploadText)
   687→        } catch {
   688→          uploadBody = uploadText
   689→        }
   690→      }
   691→      if (!uploadRes.ok) {
   692→        const msg =
   693→          uploadBody && typeof uploadBody === 'object' && 'error' in uploadBody
   694→            ? String((uploadBody as { error: unknown }).error)
   695→            : typeof uploadBody === 'string'
   696→              ? uploadBody
   697→              : `Upload failed (HTTP ${uploadRes.status})`
   698→        throw new Error(msg)
   699→      }
   700→      const { url } = uploadBody as { url: string }
   701→
   702→      await api.post(`/api/orders/${orderId}/payment-proof`, {
   703→        advance_payment_screenshot_url: url,
   704→      })
   705→      return true
   706→    } catch {
   707→      return false
   708→    } finally {
   709→      setUploadingProof(false)
   710→    }
   711→  }

  1411→/** Compact relative-time formatter for the inline address list. */
  1412→function formatLastUsedShort(iso: string | null): string {
  1413→  if (!iso) return 'Never used'
  1414→  const date = new Date(iso)
  1415→  const now = new Date()
  1416→  const diffMs = now.getTime() - date.getTime()
  1417→  const diffDays = Math.floor(diffMs / 86_400_000)
  1418→  if (diffDays < 1) return 'Today'
  1419→  if (diffDays === 1) return '1 day ago'
  1420→  if (diffDays < 30) return `${diffDays} days ago`
  1421→  return date.toLocaleDateString('en-PK', { day: '2-digit', month: 'short' })
  1422→}
