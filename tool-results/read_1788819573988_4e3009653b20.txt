   161→export function ProductDetailView({ productId }: { productId: string }) {
   162→  const navigate = useAppStore((s) => s.navigate)
   163→  const queryClient = useQueryClient()
   164→  const can = useCan()
   165→  const canEdit = can('products.edit')
   166→  const [scopeDialogOpen, setScopeDialogOpen] = useState(false)
   167→  const [changingScope, setChangingScope] = useState(false)
   168→
   169→  const { data, isLoading, isError, refetch } = useQuery<{ product: ProductDetail }>({
   170→    queryKey: ['product', productId],
   171→    queryFn: () => api.get<{ product: ProductDetail }>(`/api/products/${productId}`),
   172→    staleTime: 30_000,
   173→  })
   174→
   175→  const product = data?.product
   176→
   177→  async function changeScope(scope: 'private' | 'organization' | 'selective') {
   178→    if (!product) return
   179→    setChangingScope(true)
   180→    try {
   181→      // PATCH endpoint: scope update is part of the owner-only product patch route.
   182→      await api.patch(`/api/products/${productId}`, { product_scope: scope })
   183→      queryClient.invalidateQueries({ queryKey: ['product', productId] })
   184→      queryClient.invalidateQueries({ queryKey: ['products'] })
   185→      toast.success(`Product scope set to ${PRODUCT_SCOPE_LABELS[scope]}.`)
   186→      setScopeDialogOpen(false)
   187→    } catch (err) {
   188→      toast.error(err instanceof FetchError ? err.message : 'Failed to update scope.')
   189→    } finally {
   190→      setChangingScope(false)
   191→    }
   192→  }
   193→
   194→  // ---- Render
   195→  if (isLoading) {
   196→    return (
   197→      <div className="flex items-center justify-center py-20">
   198→        <Loader2 className="h-6 w-6 animate-spin text-muted-foreground" />
   199→      </div>
   200→    )
   201→  }
   202→
   203→  if (isError || !product) {
   204→    return (
   205→      <div className="space-y-6">
   206→        <button
   207→          onClick={() => navigate({ name: 'products' })}
   208→          className="flex items-center gap-1.5 text-sm text-muted-foreground hover:text-foreground"
   209→        >
   210→          <ArrowLeft className="h-4 w-4" /> Back to products
   211→        </button>
   212→        <Card>
   213→          <CardContent className="p-10 text-center">
   214→            <AlertCircle className="h-8 w-8 mx-auto text-muted-foreground mb-3" />
   215→            <p className="text-sm text-muted-foreground mb-4">
   216→              This product couldn&apos;t be loaded. It may have been removed or you may not have access.
   217→            </p>
   218→            <Button variant="outline" onClick={() => refetch()}>
   219→              Try again
   220→            </Button>
   221→          </CardContent>
   222→        </Card>
   223→      </div>
   224→    )
   225→  }
   226→
   227→  return (
   228→    <div className="space-y-6">
   229→      <button
   230→        onClick={() => navigate({ name: 'products' })}
   231→        className="flex items-center gap-1.5 text-sm text-muted-foreground hover:text-foreground"
   232→      >
   233→        <ArrowLeft className="h-4 w-4" /> Back to products
   234→      </button>
   235→
   236→      <PageHeader
   237→        title={product.title}
   238→        description={product.shortDescription ?? product.slug}
   239→        actions={
   240→          <div className="flex flex-wrap items-center gap-2">
   241→            {product.isOwner && (
   242→              <Button onClick={() => setScopeDialogOpen(true)}>
   243→                <ArrowUpCircle className="h-4 w-4" /> Promote to Org
   244→              </Button>
   245→            )}
   246→          </div>
   247→        }
   248→      />
   249→
   250→      {/* Header badges */}
   251→      <div className="flex flex-wrap items-center gap-2">
   252→        <Badge variant="secondary">
   253→          {PRODUCT_TYPE_LABELS[product.productType] ?? product.productType}
   254→        </Badge>
   255→        <Badge variant="outline">
   256→          {PRODUCT_SCOPE_LABELS[product.productScope] ?? product.productScope}
   257→        </Badge>
   258→        {product.isStitchable && (
   259→          <Badge className="bg-emerald-100 text-emerald-700 hover:bg-emerald-100 border-transparent gap-1">
   260→            <Shirt className="h-3 w-3" /> Stitchable
   261→          </Badge>
   262→        )}
   263→        {product.isActive ? (
   264→          <Badge className="bg-emerald-100 text-emerald-700 hover:bg-emerald-100 border-transparent">
   265→            Active
   266→          </Badge>
   267→        ) : (
   268→          <Badge variant="outline">Inactive</Badge>
   269→        )}
   270→        {product.isFeatured && (
   271→          <Badge className="bg-amber-500/90 hover:bg-amber-500 text-white border-transparent">
   272→            Featured
   273→          </Badge>
   274→        )}
   275→        {product.isOwner && (
   276→          <Badge variant="outline" className="text-xs">You own this</Badge>
   277→        )}
   278→      </div>
   279→
   280→      <Tabs defaultValue="overview">
   281→        <TabsList className="w-full sm:w-auto overflow-x-auto justify-start">
   282→          <TabsTrigger value="overview">Overview</TabsTrigger>

   287→            Images ({product.images.length})

   290→          <TabsTrigger value="inventory">Inventory</TabsTrigger>
   291→          {product.subscription && (
   292→            <TabsTrigger value="pricing">Pricing</TabsTrigger>

   300→
   301→        {/* Variants (parent-child grouped table) */}
   302→        <TabsContent value="variants">
   303→          <ParentChildVariantTable productId={productId} mode="cost" />
   304→        </TabsContent>
   305→
   306→        {/* Images (upload + delete) */}
   307→        <TabsContent value="images">
   308→          <ImagesTab product={product} productId={productId} canEdit={canEdit} />
   309→        </TabsContent>
   310→
   311→        {/* Shopify Sync */}
   312→        <TabsContent value="shopify">
   313→          <Card>
   314→            <CardHeader>
   315→              <CardTitle className="text-base flex items-center gap-2">
   316→                <Store className="h-4 w-4 text-primary" /> Shopify Sync
   317→              </CardTitle>
   318→              <CardDescription>
   319→                A summary of the payload that will be sent to Shopify when this product syncs.
   320→              </CardDescription>
   321→            </CardHeader>
   322→            <CardContent className="space-y-4">
   323→              <div className="rounded-lg border p-4 space-y-3">
   324→                <p className="text-xs font-medium uppercase text-muted-foreground">
   325→                  Product payload
   326→                </p>
   327→                <pre className="text-xs bg-muted/50 rounded-md p-3 overflow-x-auto">
   328→{JSON.stringify(
   329→  {
   330→    title: product.title,
   331→    product_type: product.productType,
   332→    vendor: product.brand?.name ?? null,
   333→    status: product.isActive ? 'active' : 'draft',
   334→    variants: product.variants.map((v) => ({
   335→      sku: v.sku,
   336→      barcode: v.barcode ?? null,
   337→      price: v.salePrice,
   338→      compare_at_price: v.comparePrice,
   339→      grams: v.weightGrams,
   340→      weight_kg: v.weightKg ?? null,
   341→      requires_shipping: v.requiresShipping,
   342→      taxable: v.isTaxable,
   343→      inventory_management: v.fulfillmentType === 'stock_based' ? 'shopify' : null,
   344→      inventory_policy: v.inventoryPolicy,

   346→  },
   347→  null,
   348→  2,
   349→)}
   350→                </pre>
   351→              </div>
   352→              <div className="rounded-lg border border-emerald-200 bg-emerald-50 p-3 text-xs text-emerald-800">
   353→                <p className="font-medium">Sync notes</p>
   354→                <ul className="mt-1 space-y-0.5 list-disc list-inside opacity-90">

   356→                    <strong>stock_based</strong> variants track inventory in Shopify
   357→                    (<code>inventory_management: shopify</code>).
   358→                  </li>
   359→                  <li>
   360→                    <strong>made_to_order</strong> variants skip inventory tracking and
   361→                    always allow oversells (<code>inventory_policy: continue</code>).
   362→                  </li>
   363→                  <li>Stitching charges are bundled into <code>cost_price</code> for accounting, not shown to customers.</li>
   364→                </ul>
   365→              </div>
   366→            </CardContent>
   367→          </Card>
   368→        </TabsContent>
   369→
   370→        {/* Inventory */}
   371→        <TabsContent value="inventory">
   372→          <InventoryTab productId={productId} variants={product.variants} isStitchable={product.isStitchable} />
   373→        </TabsContent>
   374→
   375→        {/* Pricing (per-company sale price via CompanyVariantPricing) */}
   376→        {product.subscription && (
   377→          <TabsContent value="pricing">
   378→            <ParentChildVariantTable productId={productId} mode="pricing" />
   379→          </TabsContent>
   380→        )}
   381→      </Tabs>
   382→
   383→      {/* Promote-to-org dialog */}
   384→      {scopeDialogOpen && (
   385→        <PromoteDialog
   386→          currentScope={product.productScope}
   387→          loading={changingScope}
   388→          onClose={() => setScopeDialogOpen(false)}
   389→          onConfirm={(scope) => changeScope(scope)}
   390→        />
   391→      )}
   392→    </div>
   393→  )
   394→}

   479→// ----------------------------------------------------------------------------
   480→// DetailsTab — inline edit mode for the Overview tab
   481→// ----------------------------------------------------------------------------
   482→function DetailsTab({
   483→  product,
   484→  productId,
   485→  canEdit,
   486→}: {
   487→  product: ProductDetail
   488→  productId: string
   489→  canEdit: boolean
   490→}) {
   491→  const queryClient = useQueryClient()
   492→  const [editing, setEditing] = useState(false)
   493→  const [form, setForm] = useState(() => ({
   494→    title: product.title,
   495→    description: product.description ?? '',
   496→    shortDescription: product.shortDescription ?? '',
   497→    categoryId: product.category?.id ?? '',
   498→    brandId: product.brand?.id ?? '',
   499→    isFeatured: product.isFeatured,
   500→    isActive: product.isActive,
   501→    isStitchable: product.isStitchable,
   502→    stitchingBasePrice: product.stitchingBasePrice,
   503→    hasSizeVariants: product.hasSizeVariants,
   504→  }))
   505→
   506→  function resetForm() {
   507→    setForm({
   508→      title: product.title,
   509→      description: product.description ?? '',
   510→      shortDescription: product.shortDescription ?? '',
   511→      categoryId: product.category?.id ?? '',
   512→      brandId: product.brand?.id ?? '',
   513→      isFeatured: product.isFeatured,
   514→      isActive: product.isActive,
   515→      isStitchable: product.isStitchable,
   516→      stitchingBasePrice: product.stitchingBasePrice,
   517→      hasSizeVariants: product.hasSizeVariants,
   518→    })
   519→  }
   520→
   521→  // Lazy-load category/brand options only when entering edit mode.
   522→  const { data: catData } = useQuery<{ categories: Array<{ id: string; name: string }> }>({
   523→    queryKey: ['categories'],
   524→    queryFn: () => api.get<{ categories: Array<{ id: string; name: string }> }>('/api/categories'),
   525→    enabled: editing,
   526→    staleTime: 60_000,
   527→  })
   528→  const { data: brandData } = useQuery<{ brands: Array<{ id: string; name: string }> }>({
   529→    queryKey: ['brands'],
   530→    queryFn: () => api.get<{ brands: Array<{ id: string; name: string }> }>('/api/brands'),
   531→    enabled: editing,
   532→    staleTime: 60_000,
   533→  })
   534→
   535→  const mutation = useMutation({
   536→    mutationFn: (patch: Record<string, unknown>) =>
   537→      api.patch(`/api/products/${productId}`, patch),
   538→    onSuccess: () => {
   539→      queryClient.invalidateQueries({ queryKey: ['product', productId] })
   540→      queryClient.invalidateQueries({ queryKey: ['products'] })
   541→      toast.success('Product updated.')
   542→      setEditing(false)
   543→    },
   544→    onError: (err) => {
   545→      // Stay in edit mode so the user can fix and retry.
   546→      toast.error(err instanceof FetchError ? err.message : 'Failed to update product.')
   547→    },
   548→  })
   549→
   550→  function startEdit() {
   551→    resetForm()
   552→    setEditing(true)
   553→  }
   554→
   555→  function cancelEdit() {
   556→    setEditing(false)
   557→  }
   558→
   559→  function save() {
   560→    const patch: Record<string, unknown> = {}
   561→    if (form.title !== product.title) patch.title = form.title
   562→    if (form.description !== (product.description ?? '')) patch.description = form.description
   563→    if (form.shortDescription !== (product.shortDescription ?? '')) patch.short_description = form.shortDescription
   564→    if (form.categoryId !== (product.category?.id ?? '')) patch.category_id = form.categoryId || null
   565→    if (form.brandId !== (product.brand?.id ?? '')) patch.brand_id = form.brandId || null
   566→    if (form.isFeatured !== product.isFeatured) patch.is_featured = form.isFeatured
   567→    if (form.isActive !== product.isActive) patch.is_active = form.isActive
   568→    if (form.isStitchable !== product.isStitchable) patch.is_stitchable = form.isStitchable
   569→    if (form.stitchingBasePrice !== product.stitchingBasePrice) patch.stitching_base_price = form.stitchingBasePrice
   570→    if (form.hasSizeVariants !== product.hasSizeVariants) patch.has_size_variants = form.hasSizeVariants
   571→
   572→    if (Object.keys(patch).length === 0) {
   573→      toast.info('No changes to save.')
   574→      setEditing(false)
   575→      return
   576→    }
   577→    mutation.mutate(patch)
   578→  }
   579→
   580→  // ---- Edit mode
   581→  if (editing) {
   582→    return (
   583→      <div className="grid gap-6 lg:grid-cols-3">
   584→        <Card className="lg:col-span-2">
   585→          <CardHeader>
   586→            <CardTitle className="text-base">Edit product</CardTitle>
   587→            <CardDescription>
   588→              Update the storefront-visible details. Changes are live after save.
   589→            </CardDescription>
   590→            <CardAction>
   591→              <div className="flex items-center gap-2">
   592→                <Button
   593→                  variant="ghost"
   594→                  size="sm"
   595→                  onClick={cancelEdit}
   596→                  disabled={mutation.isPending}
   597→                >
   598→                  <X className="h-4 w-4" /> Cancel
   599→                </Button>
   600→                <Button size="sm" onClick={save} disabled={mutation.isPending}>
   601→                  {mutation.isPending ? (

  1098→/** Interactive is_active switch with optimistic update + revert on error. */
  1099→function VariantActiveSwitch({
  1100→  productId,
  1101→  variantId,
  1102→  isActive,
  1103→  canEdit,
  1104→}: {
  1105→  productId: string
  1106→  variantId: string
  1107→  isActive: boolean
  1108→  canEdit: boolean
  1109→}) {
  1110→  const queryClient = useQueryClient()
  1111→
  1112→  const mutation = useMutation({
  1113→    mutationFn: (newVal: boolean) =>
  1114→      api.post(`/api/products/${productId}/variants/${variantId}/toggle`, {
  1115→        is_active: newVal,
  1116→      }),
  1117→    onMutate: async (newVal) => {
  1118→      // Optimistic update — flip the cached variant's isActive immediately.
  1119→      await queryClient.cancelQueries({ queryKey: ['product', productId] })
  1120→      const prev = queryClient.getQueryData<{ product: ProductDetail }>(['product', productId])
  1121→      if (prev) {
  1122→        queryClient.setQueryData<{ product: ProductDetail }>(['product', productId], {
  1123→          product: {
  1124→            ...prev.product,
  1125→            variants: prev.product.variants.map((v) =>
  1126→              v.id === variantId ? { ...v, isActive: newVal } : v
  1127→            ),
  1128→          },
  1129→        })
  1130→      }
  1131→      return { prev }
  1132→    },
  1133→    onError: (err, _newVal, context) => {
  1134→      // Revert the optimistic update.
  1135→      if (context?.prev) {
  1136→        queryClient.setQueryData(['product', productId], context.prev)
  1137→      }
  1138→      toast.error(err instanceof FetchError ? err.message : 'Failed to toggle variant.')
  1139→    },
  1140→    onSuccess: () => {
  1141→      queryClient.invalidateQueries({ queryKey: ['product', productId] })
  1142→      toast.success('Variant status updated.')
  1143→    },
  1144→  })
  1145→
  1146→  return (

  1415→  const uploadMutation = useMutation({
  1416→    mutationFn: async (file: File) => {
  1417→      // Use raw fetch — multipart FormData must NOT have a JSON Content-Type.
  1418→      const fd = new FormData()

  1421→        method: 'POST',

  1425→        cache: 'no-store',

  1427→      const text = await res.text()

  1429→      if (text) {
  1430→        try {
  1431→          body = JSON.parse(text)

  1433→          body = text

  1436→      if (!res.ok) {
  1437→        const msg =
  1438→          body && typeof body === 'object' && 'error' in body
  1439→            ? String((body as { error: unknown }).error)

  1441→              ? body

  1443→        throw new FetchError(res.status, msg)
