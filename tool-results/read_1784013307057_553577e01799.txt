   131→  const navigate = useAppStore((s) => s.navigate)
   132→  const queryClient = useQueryClient()

   134→  const [changingScope, setChangingScope] = useState(false)
   135→
   136→  const { data, isLoading, isError, refetch } = useQuery<{ product: ProductDetail }>({
   137→    queryKey: ['product', productId],
   138→    queryFn: () => api.get<{ product: ProductDetail }>(`/api/products/${productId}`),
   139→    staleTime: 30_000,

   144→  async function changeScope(scope: 'private' | 'organization' | 'selective') {

   148→      // PATCH endpoint: scope update is part of the owner-only product patch route.
   149→      await api.patch(`/api/products/${productId}`, { product_scope: scope })

   151→      queryClient.invalidateQueries({ queryKey: ['products'] })
   152→      toast.success(`Product scope set to ${PRODUCT_SCOPE_LABELS[scope]}.`)

   155→      toast.error(err instanceof FetchError ? err.message : 'Failed to update scope.')
   156→    } finally {
   157→      setChangingScope(false)
   158→    }
   159→  }
   160→
   161→  // ---- Render
   162→  if (isLoading) {
   163→    return (
   164→      <div className="flex items-center justify-center py-20">
   165→        <Loader2 className="h-6 w-6 animate-spin text-muted-foreground" />
   166→      </div>
   167→    )
   168→  }
   169→
   170→  if (isError || !product) {
   171→    return (
   172→      <div className="space-y-6">
   173→        <button
   174→          onClick={() => navigate({ name: 'products' })}
   175→          className="flex items-center gap-1.5 text-sm text-muted-foreground hover:text-foreground"
   176→        >
   177→          <ArrowLeft className="h-4 w-4" /> Back to products

   180→          <CardContent className="p-10 text-center">

   182→            <p className="text-sm text-muted-foreground mb-4">
   183→              This product couldn&apos;t be loaded. It may have been removed or you may not have access.
   184→            </p>
   185→            <Button variant="outline" onClick={() => refetch()}>
   186→              Try again
   187→            </Button>
   188→          </CardContent>
   189→        </Card>
   190→      </div>
   191→    )
   192→  }
   193→
   194→  return (
   195→    <div className="space-y-6">
   196→      <button
   197→        onClick={() => navigate({ name: 'products' })}

   212→                </Button>
   213→                <Button onClick={() => setScopeDialogOpen(true)}>
   214→                  <ArrowUpCircle className="h-4 w-4" /> Promote to Org
   215→                </Button>
   216→              </>
   217→            )}
   218→          </div>
   219→        }
   220→      />
   221→
   222→      {/* Header badges */}
   223→      <div className="flex flex-wrap items-center gap-2">

   225→          {PRODUCT_TYPE_LABELS[product.productType] ?? product.productType}
   226→        </Badge>

   229→        </Badge>
   230→        {product.isStitchable && (
   231→          <Badge className="bg-emerald-100 text-emerald-700 hover:bg-emerald-100 border-transparent gap-1">

   588→      grams: v.weightGrams,
   589→      requires_shipping: v.requiresShipping,
   590→      taxable: v.isTaxable,
   591→      inventory_management: v.fulfillmentType === 'stock_based' ? 'shopify' : null,
   592→      inventory_policy: v.inventoryPolicy,
   593→    })),
   594→  },
   595→  null,
   596→  2,
   597→)}
   598→                </pre>
   599→              </div>
   600→              <div className="rounded-lg border border-emerald-200 bg-emerald-50 p-3 text-xs text-emerald-800">
   601→                <p className="font-medium">Sync notes</p>
   602→                <ul className="mt-1 space-y-0.5 list-disc list-inside opacity-90">
   603→                  <li>
   604→                    <strong>stock_based</strong> variants track inventory in Shopify
   605→                    (<code>inventory_management: shopify</code>).
   606→                  </li>
   607→                  <li>
   608→                    <strong>made_to_order</strong> variants skip inventory tracking and
   609→                    always allow oversells (<code>inventory_policy: continue</code>).
   610→                  </li>
   611→                  <li>Stitching charges are bundled into <code>cost_price</code> for accounting, not shown to customers.</li>
   612→                </ul>
   613→              </div>
   614→            </CardContent>
   615→          </Card>
   616→        </TabsContent>
   617→
   618→        {/* Pricing (non-owner with subscription) */}
   619→        {!product.isOwner && product.subscription && (
   620→          <TabsContent value="pricing">
   621→            <PricingTab
   622→              productId={productId}
   623→              variants={product.variants}
   624→              onUpdate={() => {
   625→                queryClient.invalidateQueries({ queryKey: ['product', productId] })
   626→              }}
   627→            />
   628→          </TabsContent>
   629→        )}
   630→      </Tabs>
   631→
   632→      {/* Promote-to-org dialog */}
   633→      {scopeDialogOpen && (
   634→        <PromoteDialog
   635→          currentScope={product.productScope}
   636→          loading={changingScope}
   637→          onClose={() => setScopeDialogOpen(false)}
   638→          onConfirm={(scope) => changeScope(scope)}
   639→        />
   640→      )}
   641→    </div>
   642→  )
   643→}

   745→function PricingTab({
   746→  productId,
   747→  variants,
   748→  onUpdate,
   749→}: {
   750→  productId: string
   751→  variants: ProductVariant[]
   752→  onUpdate: () => void
   753→}) {
   754→  const [drafts, setDrafts] = useState<Record<string, { salePrice: number | null; comparePrice: number | null }>>(
   755→    () =>
   756→      Object.fromEntries(
   757→        variants.map((v) => [
   758→          v.id,
   759→          { salePrice: v.salePrice, comparePrice: v.comparePrice },
   760→        ]),
   761→      ),
   762→  )
   763→  const [saving, setSaving] = useState<string | null>(null)
   764→
   765→  function setDraft(
   766→    variantId: string,
   767→    field: 'salePrice' | 'comparePrice',
   768→    value: number | null,
   769→  ) {
   770→    setDrafts((d) => ({
   771→      ...d,
   772→      [variantId]: { ...d[variantId], [field]: value },
   773→    }))
   774→  }
   775→
   776→  async function save(variantId: string) {
   777→    const draft = drafts[variantId]
   778→    if (!draft) return
   779→    setSaving(variantId)
   780→    try {
   781→      await api.patch(`/api/products/${productId}/variants/${variantId}/pricing`, {
   782→        sale_price: draft.salePrice,
   783→        compare_price: draft.comparePrice,
   784→      })
   785→      toast.success('Pricing updated.')
   786→      onUpdate()
   787→    } catch (err) {
   788→      toast.error(err instanceof FetchError ? err.message : 'Failed to update pricing.')
   789→    } finally {
   790→      setSaving(null)
   791→    }
   792→  }
   793→
   794→  return (
   795→    <Card>
   796→      <CardHeader>
   797→        <CardTitle className="text-base flex items-center gap-2">
   798→          <Tag className="h-4 w-4 text-primary" /> Your company pricing
   799→        </CardTitle>
   800→        <CardDescription>
   801→          Set the sale and compare-at prices for your storefront.
   802→        </CardDescription>
   803→      </CardHeader>
   804→      <CardContent className="p-0">
   805→        <div className="overflow-x-auto">
   806→          <Table>
   807→            <TableHeader>
   808→              <TableRow>
   809→                <TableHead>SKU</TableHead>
   810→                <TableHead className="text-right">Cost</TableHead>
   811→                <TableHead>Sale price</TableHead>
   812→                <TableHead>Compare price</TableHead>
   813→                <TableHead className="text-right">Action</TableHead>
   814→              </TableRow>
   815→            </TableHeader>
   816→            <TableBody>
   817→              {variants.map((v) => {
   818→                const draft = drafts[v.id] ?? { salePrice: v.salePrice, comparePrice: v.comparePrice }
   819→                const dirty =
   820→                  draft.salePrice !== v.salePrice ||
   821→                  draft.comparePrice !== v.comparePrice
   822→                return (
   823→                  <TableRow key={v.id}>
   824→                    <TableCell className="font-mono text-xs">{v.sku}</TableCell>
   825→                    <TableCell className="text-right text-xs text-muted-foreground">
   826→                      {formatMoney(v.costPrice)}
   827→                    </TableCell>
   828→                    <TableCell>
   829→                      <Input
   830→                        type="number"
   831→                        min="0"
   832→                        step="0.01"
   833→                        value={draft.salePrice ?? ''}
   834→                        onChange={(e) =>
   835→                          setDraft(
   836→                            v.id,
   837→                            'salePrice',
   838→                            e.target.value ? Number(e.target.value) : null,
   839→                          )
   840→                        }
   841→                        className="h-8 w-32 text-xs"
   842→                      />
   843→                    </TableCell>
   844→                    <TableCell>
   845→                      <Input
   846→                        type="number"
   847→                        min="0"
   848→                        step="0.01"
   849→                        value={draft.comparePrice ?? ''}
   850→                        onChange={(e) =>
   851→                          setDraft(
   852→                            v.id,
   853→                            'comparePrice',
   854→                            e.target.value ? Number(e.target.value) : null,
   855→                          )
   856→                        }
   857→                        className="h-8 w-32 text-xs"
   858→                      />
   859→                    </TableCell>
   860→                    <TableCell className="text-right">
   861→                      <Button
   862→                        size="sm"
   863→                        variant={dirty ? 'default' : 'outline'}
   864→                        disabled={!dirty || saving === v.id}
   865→                        onClick={() => save(v.id)}
   866→                      >
   867→                        {saving === v.id ? (
   868→                          <Loader2 className="h-3.5 w-3.5 animate-spin" />
   869→                        ) : (
   870→                          <Save className="h-3.5 w-3.5" />
   871→                        )}
   872→                        Save
   873→                      </Button>
   874→                    </TableCell>
   875→                  </TableRow>
   876→                )
   877→              })}
   878→            </TableBody>
   879→          </Table>
   880→        </div>
   881→      </CardContent>
   882→    </Card>
   883→  )
   884→}
