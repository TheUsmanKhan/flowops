  1955→function OpeningStockTable({
  1956→  variants,
  1957→  locations,
  1958→  noLocations,
  1959→  onChange,
  1960→  onNavigateLocations,
  1961→}: {
  1962→  variants: GeneratedVariant[]
  1963→  locations: Array<{ id: string; name: string; isDefault?: boolean }>
  1964→  noLocations: boolean
  1965→  onChange: (v: GeneratedVariant[]) => void
  1966→  onNavigateLocations: () => void
  1967→}) {
  1968→  // Track which MTO rows are expanded (by SKU). Stock_based rows are always
  1969→  // "expanded" (inputs always visible) since they're the primary use case.
  1970→  const [expandedMtoSkus, setExpandedMtoSkus] = useState<Set<string>>(new Set())
  1971→
  1972→  function toggleMtoExpand(sku: string) {
  1973→    setExpandedMtoSkus((prev) => {
  1974→      const next = new Set(prev)
  1975→      if (next.has(sku)) {
  1976→        next.delete(sku)
  1977→        // Discard any unsaved entry on collapse — also revert track_inventory
  1978→        // to false since no stock will be recorded for this MTO variant.
  1979→        onChange(
  1980→          variants.map((v) =>
  1981→            v.sku === sku
  1982→              ? {
  1983→                  ...v,
  1984→                  opening_stock_qty: 0,
  1985→                  opening_stock_location_id: '',
  1986→                  track_inventory: v.fulfillment_type === 'stock_based',
  1987→                }
  1988→              : v,
  1989→          ),
  1990→        )
  1991→      } else {
  1992→        next.add(sku)
  1993→        // Pre-fill the default location on expand
  1994→        const defaultLoc = locations.find((l) => l.isDefault)?.id ?? locations[0]?.id ?? ''
  1995→        onChange(
  1996→          variants.map((v) =>
  1997→            v.sku === sku && !v.opening_stock_location_id
  1998→              ? { ...v, opening_stock_location_id: defaultLoc }
  1999→              : v,
  2000→          ),
  2001→        )
  2002→      }
  2003→      return next
  2004→    })
  2005→  }

  2007→  function updateVariant(sku: string, patch: Partial<GeneratedVariant>) {
  2008→    // Bug 2 fix: when opening_stock_qty changes for a made_to_order variant,
  2009→    // optimistically flip track_inventory in local state so the badge updates
  2010→    // live in the wizard. If Qty is cleared back to 0, revert track_inventory
  2011→    // to false (since no stock will actually be recorded for it).
  2012→    const target = variants.find((v) => v.sku === sku)
  2013→    const isMto = target?.fulfillment_type === 'made_to_order'
  2014→    const finalPatch = { ...patch }
  2015→    if (isMto && 'opening_stock_qty' in patch) {
  2016→      const newQty = Number(patch.opening_stock_qty ?? 0)
  2017→      finalPatch.track_inventory = newQty > 0
  2018→    }
  2019→    onChange(variants.map((v) => (v.sku === sku ? { ...v, ...finalPatch } : v)))
  2020→  }
  2021→
  2022→  function applyDefaultLocationToAll() {
  2023→    const defaultLoc = locations.find((l) => l.isDefault)?.id ?? locations[0]?.id
  2024→    if (!defaultLoc) return
  2025→    // Apply to every row that currently has an active Qty entry OR is a
  2026→    // stock_based variant (whose inputs are always visible).
  2027→    onChange(
  2028→      variants.map((v) => {
  2029→        const hasActiveQty = Number(v.opening_stock_qty ?? 0) > 0
  2030→        if (v.fulfillment_type === 'stock_based' || hasActiveQty) {
  2031→          return { ...v, opening_stock_location_id: defaultLoc }
  2032→        }
  2033→        return v
  2034→      }),
  2035→    )
  2036→  }
  2037→

  2281→function textToAttrs(text: string): Record<string, string> {
  2282→  const out: Record<string, string> = {}
  2283→  for (const line of text.split('\n')) {
  2284→    const m = line.match(/^\s*([^:]+):\s*(.+)\s*$/)
  2285→    if (m) {
  2286→      out[m[1].trim()] = m[2].trim()
  2287→    }
  2288→  }
  2289→  return out
