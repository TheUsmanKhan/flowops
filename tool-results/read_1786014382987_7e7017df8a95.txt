   261→function toISODate(d: Date): string {
   262→  const y = d.getFullYear()
   263→  const m = String(d.getMonth() + 1).padStart(2, '0')
   264→  const day = String(d.getDate()).padStart(2, '0')
   265→  return `${y}-${m}-${day}`
   266→}
   267→
   268→function resolveTimeSpan(value: TimeSpanValue, custom?: DateRange): DateRange {
   269→  const now = new Date()
   270→  const today = new Date(now.getFullYear(), now.getMonth(), now.getDate())
   271→
   272→  switch (value) {
   273→    case 'today':
   274→      return { from: toISODate(today), to: toISODate(today) }
   275→    case 'yesterday': {
   276→      const y = new Date(today)
   277→      y.setDate(y.getDate() - 1)
   278→      return { from: toISODate(y), to: toISODate(y) }
   279→    }
   280→    case 'last_7_days': {
   281→      const from = new Date(today)
   282→      from.setDate(from.getDate() - 6)
   283→      return { from: toISODate(from), to: toISODate(today) }
   284→    }
   285→    case 'last_30_days': {
   286→      const from = new Date(today)
   287→      from.setDate(from.getDate() - 29)
   288→      return { from: toISODate(from), to: toISODate(today) }
   289→    }
   290→    case 'this_month': {
   291→      const from = new Date(now.getFullYear(), now.getMonth(), 1)
   292→      return { from: toISODate(from), to: toISODate(today) }
   293→    }
   294→    case 'last_month': {
   295→      const firstOfThis = new Date(now.getFullYear(), now.getMonth(), 1)
   296→      const from = new Date(firstOfThis)
   297→      from.setMonth(from.getMonth() - 1)
   298→      const to = new Date(firstOfThis.getFullYear(), firstOfThis.getMonth(), 0)
   299→      return { from: toISODate(from), to: toISODate(to) }
   300→    }
   301→    case 'this_year': {
   302→      const from = new Date(now.getFullYear(), 0, 1)
   303→      return { from: toISODate(from), to: toISODate(today) }
   304→    }
   305→    case 'custom':
   306→      return {
   307→        from: custom?.from ?? toISODate(today),
   308→        to: custom?.to ?? toISODate(today),
   309→      }
   310→  }
   311→}

   313→function spanDayCount(range: DateRange): number {
   314→  const from = new Date(range.from + 'T00:00:00')
   315→  const to = new Date(range.to + 'T23:59:59')
   316→  const diff = to.getTime() - from.getTime()
   317→  return Math.max(1, Math.round(diff / (1000 * 60 * 60 * 24)) + 1)
   318→}
   319→
   320→function isMonthlyGranularity(range: DateRange): boolean {
   321→  return spanDayCount(range) > 30
   322→}

   432→function toggleArray(arr: string[], value: string): string[] {
   433→  return arr.includes(value) ? arr.filter((v) => v !== value) : [...arr, value]

   436→function countActiveFilters(f: FilterState): number {
   437→  let n = 0
   438→  if (f.statuses.length) n++
   439→  if (f.paymentTypes.length) n++
   440→  if (f.paymentStatuses.length) n++
   441→  if (f.orderSources.length) n++
   442→  if (f.dateFrom || f.dateTo) n++
   443→  if (f.amountMin || f.amountMax) n++
   444→  if (f.orgVariantId) n++
   445→  if (f.customerId) n++
   446→  if (f.deliveryCity.trim()) n++
   447→  if (f.courier.trim()) n++
   448→  return n

   464→function applyPreset(preset: DatePresetValue): { from: string; to: string } {
   465→  const now = new Date()
   466→  const to = new Date(now)
   467→  const from = new Date(now)
   468→  if (preset === 'today') {
   469→    // from = today
   470→  } else if (preset === '7d') {
   471→    from.setDate(now.getDate() - 6)
   472→  } else if (preset === '30d') {
   473→    from.setDate(now.getDate() - 29)
   474→  } else if (preset === 'this_month') {
   475→    from.setDate(1)
   476→  }
   477→  return { from: toISODate(from), to: toISODate(to) }
   478→}

  1183→function buildChartData(
  1184→  orders: OrderRow[],
  1185→  range: DateRange,
  1186→  monthly: boolean,
  1187→  metric: ChartMetric,
  1188→): ChartPoint[] {
  1189→  // Build the list of interval buckets
  1190→  const buckets = buildIntervalBuckets(range, monthly)
  1191→
  1192→  // For each order, compute its bucket key (yyyy-mm-dd or yyyy-mm)
  1193→  for (const o of orders) {
  1194→    const created = new Date(o.createdAt)
  1195→    if (Number.isNaN(created.getTime())) continue
  1196→    const key = monthly
  1197→      ? `${created.getFullYear()}-${String(created.getMonth() + 1).padStart(2, '0')}`
  1198→      : toISODate(created)
  1199→    const b = buckets.find((x) => x.key === key)
  1200→    if (!b) continue
  1201→    b.total += 1
  1202→    if (o.status === 'rto') b.rto += 1
  1203→    if (o.status === 'cancelled') b.cancelled += 1
  1204→    if (o.status !== 'cancelled') b.revenue += o.totalOrderValue
  1205→    if (o.status === 'pending') b.pending += 1
  1206→    if (o.status === 'partially_backordered') b.backordered += 1
  1207→  }
  1208→
  1209→  return buckets.map((b) => {
  1210→    let value = 0
  1211→    let count: number | undefined
  1212→    let total: number | undefined
  1213→    switch (metric) {
  1214→      case 'total_orders':
  1215→        value = b.total
  1216→        break
  1217→      case 'revenue':
  1218→        value = b.revenue
  1219→        break
  1220→      case 'rto_rate':
  1221→        value = b.total > 0 ? (b.rto / b.total) * 100 : 0
  1222→        count = b.rto
  1223→        total = b.total
  1224→        break
  1225→      case 'cancellation_rate':
  1226→        value = b.total > 0 ? (b.cancelled / b.total) * 100 : 0
  1227→        count = b.cancelled
  1228→        total = b.total
  1229→        break
  1230→      case 'pending_current':
  1231→        value = b.pending
  1232→        break
  1233→      case 'backordered_current':
  1234→        value = b.backordered
  1235→        break
  1236→    }
  1237→    return { label: b.label, value, count, total }
  1238→  })
  1239→}
  1240→
  1241→interface Bucket {
  1242→  key: string
  1243→  label: string
  1244→  total: number
  1245→  revenue: number
  1246→  rto: number
  1247→  cancelled: number
  1248→  pending: number
  1249→  backordered: number
  1250→}

  1252→function buildIntervalBuckets(range: DateRange, monthly: boolean): Bucket[] {
  1253→  const out: Bucket[] = []
  1254→  const start = new Date(range.from + 'T00:00:00')
  1255→  const end = new Date(range.to + 'T23:59:59')
  1256→
  1257→  if (monthly) {
  1258→    // Iterate by month from start month to end month (inclusive)
  1259→    const cursor = new Date(start.getFullYear(), start.getMonth(), 1)
  1260→    const endMonth = new Date(end.getFullYear(), end.getMonth(), 1)
  1261→    while (cursor <= endMonth) {
  1262→      const key = `${cursor.getFullYear()}-${String(cursor.getMonth() + 1).padStart(2, '0')}`
  1263→      const label = cursor.toLocaleDateString('en-PK', { month: 'short', year: '2-digit' })
  1264→      out.push({ key, label, total: 0, revenue: 0, rto: 0, cancelled: 0, pending: 0, backordered: 0 })
  1265→      cursor.setMonth(cursor.getMonth() + 1)
  1266→    }
  1267→  } else {
  1268→    // Iterate by day from start to end (inclusive)
  1269→    const cursor = new Date(start.getFullYear(), start.getMonth(), start.getDate())
  1270→    const endDay = new Date(end.getFullYear(), end.getMonth(), end.getDate())
  1271→    while (cursor <= endDay) {
  1272→      const key = toISODate(cursor)
  1273→      const label = cursor.toLocaleDateString('en-PK', { day: '2-digit', month: 'short' })
  1274→      out.push({ key, label, total: 0, revenue: 0, rto: 0, cancelled: 0, pending: 0, backordered: 0 })
  1275→      cursor.setDate(cursor.getDate() + 1)
  1276→    }
  1277→  }
  1278→  return out
  1279→}

  1642→        onFocus={() => setOpen(true)}
  1643→        onKeyDown={onKeyDown}
