   193→/**
   194→ * Discrepancy reason options for cycle count items.
   195→ * These values are sent to the backend and drive the approve-time behavior:
   196→ *   - theft_suspected / unknown → shortage is QUARANTINED + a missing
   197→ *     stock_loss_record is auto-created (investigation opens)
   198→ *   - recording_error / transfer_not_recorded / damage_not_recorded /
   199→ *     no_discrepancy → standard cycle_count_adjust (on_hand set to counted)
   200→ *   - surplus (positive discrepancy) → standard adjust (on_hand set to counted)
   201→ *
   202→ * The backend reads these from item.discrepancyReason in the approve action.
   203→ */
   204→type DiscrepancyReason =
   205→  | 'recording_error'
   206→  | 'theft_suspected'
   207→  | 'damage_not_recorded'
   208→  | 'transfer_not_recorded'
   209→  | 'unknown'
   210→  | 'no_discrepancy'
   211→
   212→const DISCREPANCY_REASONS: Array<{
   213→  value: DiscrepancyReason
   214→  label: string
   215→  hint: string
   216→}> = [
   217→  { value: 'no_discrepancy', label: 'No discrepancy', hint: 'Count matches — no action needed.' },
   218→  { value: 'recording_error', label: 'Recording error', hint: 'Data-entry mistake in a prior receive/adjust/transfer.' },
   219→  { value: 'transfer_not_recorded', label: 'Transfer not recorded', hint: 'Stock moved to/from another location but the transfer was not logged.' },
   220→  { value: 'damage_not_recorded', label: 'Damage not recorded', hint: 'Units were damaged but no stock-loss report was filed.' },
   221→  { value: 'theft_suspected', label: 'Theft suspected', hint: 'Shortage appears stolen — will quarantine + open a missing-stock investigation on approval.' },
   222→  { value: 'unknown', label: 'Unknown', hint: 'Cause cannot be determined — will quarantine + open a missing-stock investigation on approval.' },
   223→]
   224→
   225→const DISCREPANCY_REASON_LABEL: Record<DiscrepancyReason, string> = {
   226→  recording_error: 'Recording error',
   227→  theft_suspected: 'Theft suspected',
   228→  damage_not_recorded: 'Damage not recorded',
   229→  transfer_not_recorded: 'Transfer not recorded',
   230→  unknown: 'Unknown',
   231→  no_discrepancy: 'No discrepancy',
   232→}
   233→
   234→const PKR = new Intl.NumberFormat('en-PK', { maximumFractionDigits: 0 })
   235→function formatPKR(n: number): string {
   236→  return `Rs. ${PKR.format(n)}`
   237→}
   238→
   239→function formatDate(iso: string | null): string {
   240→  if (!iso) return '—'
   241→  try {
   242→    return new Date(iso).toLocaleDateString('en-PK', {
   243→      day: '2-digit',
   244→      month: 'short',
   245→      year: 'numeric',
   246→    })
   247→  } catch {
   248→    return iso
   249→  }
   250→}
   251→
   252→function getErrorMessage(err: unknown): string {
   253→  if (err instanceof FetchError) return err.message
   254→  if (err instanceof Error) return err.message
   255→  return 'Something went wrong. Please try again.'
   256→}

   262→export function CycleCountsView() {
   263→  const can = useCan()
   264→  const queryClient = useQueryClient()
   265→
   266→  const [search, setSearch] = useState('')
   267→  const [statusFilter, setStatusFilter] = useState<string>('all')
   268→  const [createOpen, setCreateOpen] = useState(false)
   269→  const [selectedId, setSelectedId] = useState<string | null>(null)
   270→  const [cancelTarget, setCancelTarget] = useState<CycleCountRow | null>(null)
   271→
   272→  const canManage = can(PERMISSIONS.INVENTORY_CYCLE_COUNT)
   273→
   274→  const countsQuery = useQuery<CycleCountsResponse>({
   275→    queryKey: ['cycle-counts'],
   276→    queryFn: () => api.get<CycleCountsResponse>('/api/cycle-counts'),
   277→    staleTime: 15_000,
   278→  })
   279→
   280→  const counts = countsQuery.data?.counts ?? []
   281→
   282→  const filtered = useMemo(() => {
   283→    return counts.filter((c) => {
   284→      if (statusFilter !== 'all' && c.status !== statusFilter) return false
   285→      if (search) {
   286→        const q = search.toLowerCase()
   287→        return (
   288→          c.countName.toLowerCase().includes(q) ||
   289→          c.location.toLowerCase().includes(q)
   290→        )
   291→      }
   292→      return true
   293→    })
   294→  }, [counts, statusFilter, search])
   295→
   296→  const invalidateAll = (id?: string) => {
   297→    void queryClient.invalidateQueries({ queryKey: ['cycle-counts'] })
   298→    if (id) {
   299→      void queryClient.invalidateQueries({ queryKey: ['cycle-count', id] })
   300→    }
   301→    void queryClient.invalidateQueries({ queryKey: ['inventory-dashboard'] })
   302→  }
   303→
   304→  // ── Action mutations ─────────────────────────────────────────────────────
   305→  const startMutation = useMutation({
   306→    mutationFn: async (id: string) =>
   307→      api.patch(`/api/cycle-counts/${id}`, { action: 'start' }),
   308→    onSuccess: () => {
   309→      toast.success('Count started. Inventory snapshot captured.')
   310→      if (selectedId) invalidateAll(selectedId)
   311→      else invalidateAll()
   312→    },
   313→    onError: (err) => toast.error(getErrorMessage(err)),
   314→  })
   315→
   316→  const approveMutation = useMutation({
   317→    mutationFn: async (id: string) =>
   318→      api.patch(`/api/cycle-counts/${id}`, { action: 'approve' }),
   319→    onSuccess: () => {
   320→      toast.success('Count approved. Adjustments applied to inventory.')
   321→      if (selectedId) invalidateAll(selectedId)
   322→    },
   323→    onError: (err) => toast.error(getErrorMessage(err)),
   324→  })
   325→
   326→  const cancelMutation = useMutation({
   327→    mutationFn: async (id: string) =>
   328→      api.patch(`/api/cycle-counts/${id}`, { action: 'cancel' }),
   329→    onSuccess: () => {
   330→      toast.success('Count cancelled.')
   331→      setCancelTarget(null)
   332→      if (selectedId) invalidateAll(selectedId)
   333→    },
   334→    onError: (err) => toast.error(getErrorMessage(err)),
   335→  })
   336→
   337→  return (
   338→    <div className="space-y-6">
   339→      <PageHeader
   340→        title="Cycle Counts"
   341→        description="Reconcile system stock against physical counts. Discrepancies become inventory adjustments on approval."
   342→        actions={
   343→          <div className="flex items-center gap-2">
   344→            <Button
   345→              variant="outline"
   346→              size="sm"
   347→              onClick={() => countsQuery.refetch()}
   348→              disabled={countsQuery.isFetching}
   349→            >
   350→              <RefreshCw
   351→                className={countsQuery.isFetching ? 'h-4 w-4 animate-spin' : 'h-4 w-4'}
   352→              />
   353→              Refresh
   354→            </Button>
   355→            {canManage && (
   356→              <Button onClick={() => setCreateOpen(true)}>
   357→                <Plus className="h-4 w-4" /> New Cycle Count
   358→              </Button>
   359→            )}
   360→          </div>
   361→        }
   362→      />
   363→
   364→      {/* ── Filter bar ─────────────────────────────────────────────────────── */}
   365→      <div className="flex flex-col sm:flex-row gap-2 sm:items-center">
   366→        <div className="relative flex-1">
   367→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   368→          <Input
   369→            placeholder="Search count name or location…"
   370→            className="pl-9"
   371→            value={search}
   372→            onChange={(e) => setSearch(e.target.value)}
   373→          />
   374→        </div>
   375→        <Select value={statusFilter} onValueChange={setStatusFilter}>
   376→          <SelectTrigger className="w-full sm:w-48">
   377→            <SelectValue placeholder="All statuses" />
   378→          </SelectTrigger>
   379→          <SelectContent>
   380→            <SelectItem value="all">All statuses</SelectItem>
   381→            <SelectItem value="scheduled">Scheduled</SelectItem>
   382→            <SelectItem value="in_progress">In Progress</SelectItem>
   383→            <SelectItem value="pending_review">Pending Review</SelectItem>
   384→            <SelectItem value="approved">Approved</SelectItem>
   385→            <SelectItem value="cancelled">Cancelled</SelectItem>
   386→          </SelectContent>
   387→        </Select>
   388→      </div>
   389→
   390→      <div className="grid gap-6 lg:grid-cols-5">
   391→        {/* ── Left: counts list ─────────────────────────────────────────── */}
   392→        <div className="lg:col-span-2 space-y-3">
   393→          {countsQuery.isLoading ? (
   394→            <Card>
   395→              <CardContent className="p-3 space-y-2">
   396→                {Array.from({ length: 5 }).map((_, i) => (
   397→                  <Skeleton key={i} className="h-16" />
   398→                ))}
   399→              </CardContent>
   400→            </Card>
   401→          ) : countsQuery.isError ? (
   402→            <Card>
   403→              <CardContent className="p-10 text-center">
   404→                <p className="text-sm text-muted-foreground mb-4">
   405→                  Couldn&apos;t load cycle counts. {getErrorMessage(countsQuery.error)}
   406→                </p>
   407→                <Button variant="outline" onClick={() => countsQuery.refetch()}>
   408→                  Try again
   409→                </Button>
   410→              </CardContent>
   411→            </Card>
   412→          ) : filtered.length === 0 ? (
   413→            <EmptyState
   414→              hasCounts={counts.length > 0}
   415→              canManage={canManage}
   416→              onCreate={() => setCreateOpen(true)}
   417→            />
   418→          ) : (
   419→            <div className="space-y-2 max-h-[70vh] overflow-y-auto scrollbar-thin pr-1">
   420→              {filtered.map((c) => {
   421→                const badge = STATUS_BADGE[c.status]
   422→                const isSelected = c.id === selectedId
   423→                return (
   424→                  <button
   425→                    type="button"
   426→                    key={c.id}
   427→                    onClick={() => setSelectedId(c.id)}
   428→                    className={`w-full text-left rounded-md border p-3 transition-colors ${
   429→                      isSelected
   430→                        ? 'border-primary bg-primary/5'
   431→                        : 'hover:bg-muted/50 border-border'
   432→                    }`}
   433→                  >
   434→                    <div className="flex items-start justify-between gap-2 mb-1">
   435→                      <div className="min-w-0">
   436→                        <p className="font-medium text-sm truncate">{c.countName}</p>
   437→                        <p className="text-xs text-muted-foreground">{c.location}</p>
   438→                      </div>
   439→                      <Badge variant="outline" className={badge.className}>
   440→                        {badge.label}
   441→                      </Badge>
   442→                    </div>
   443→                    <div className="flex items-center justify-between text-xs text-muted-foreground mt-2">
   444→                      <span>
   445→                        {c.itemCount} item{c.itemCount === 1 ? '' : 's'} ·{' '}
   446→                        {COUNT_TYPE_LABEL[c.countType]}
   447→                      </span>
   448→                      {c.status === 'approved' || c.status === 'pending_review' ? (
   449→                        <span
   450→                          className={`tabular-nums font-medium ${
   451→                            c.totalDiscrepancies > 0
   452→                              ? 'text-amber-700'
   453→                              : 'text-emerald-700'
   454→                          }`}
   455→                        >
   456→                          {c.totalDiscrepancies} discrepancies · {formatPKR(c.totalVarianceValue)}
   457→                        </span>
   458→                      ) : (
   459→                        <span className="whitespace-nowrap">{formatDate(c.scheduledAt)}</span>
   460→                      )}
   461→                    </div>
   462→                  </button>
   463→                )
   464→              })}
   465→            </div>
   466→          )}
   467→        </div>
   468→
   469→        {/* ── Right: detail or empty ────────────────────────────────────── */}
   470→        <div className="lg:col-span-3">
   471→          {selectedId ? (
   472→            <CycleCountDetailPanel
   473→              countId={selectedId}
   474→              onBack={() => setSelectedId(null)}
   475→              onStart={() => startMutation.mutate(selectedId)}
   476→              onApprove={() => approveMutation.mutate(selectedId)}
   477→              onCancel={() => {
   478→                const target = counts.find((c) => c.id === selectedId)
   479→                if (target) setCancelTarget(target)
   480→              }}
   481→              starting={startMutation.isPending}
   482→              approving={approveMutation.isPending}
   483→              canManage={canManage}
   484→            />
   485→          ) : (
   486→            <Card>
   487→              <CardContent className="p-10 text-center">
   488→                <ClipboardList className="h-10 w-10 mx-auto text-muted-foreground/40 mb-3" />
   489→                <p className="text-sm text-muted-foreground">
   490→                  Select a cycle count on the left to see its items and take action.
   491→                </p>
   492→              </CardContent>
   493→            </Card>
   494→          )}
   495→        </div>
   496→      </div>
   497→
   498→      {/* ── Create dialog ─────────────────────────────────────────────────── */}
   499→      {canManage && (
   500→        <CreateCountDialog
   501→          open={createOpen}
   502→          onOpenChange={setCreateOpen}
   503→          onSuccess={() => {
   504→            void queryClient.invalidateQueries({ queryKey: ['cycle-counts'] })
   505→            setCreateOpen(false)
   506→          }}
   507→        />
   508→      )}
   509→
   510→      {/* ── Cancel dialog ─────────────────────────────────────────────────── */}
   511→      <AlertDialog
   512→        open={!!cancelTarget}
   513→        onOpenChange={(open) => {
   514→          if (!open) setCancelTarget(null)
   515→        }}
   516→      >
   517→        <AlertDialogContent>
   518→          <AlertDialogHeader>
   519→            <AlertDialogTitle>Reject for recount?</AlertDialogTitle>
   520→            <AlertDialogDescription>
   521→              {cancelTarget && (
   522→                <>
   523→                  <strong>{cancelTarget.countName}</strong> will be cancelled. You can create a new
   524→                  count if you need to recount this location.
   525→                </>
   526→              )}
   527→            </AlertDialogDescription>
   528→          </AlertDialogHeader>
   529→          <AlertDialogFooter>
   530→            <AlertDialogCancel disabled={cancelMutation.isPending}>Keep</AlertDialogCancel>
   531→            <AlertDialogAction
   532→              className="bg-rose-600 hover:bg-rose-700"
   533→              disabled={cancelMutation.isPending}
   534→              onClick={(e) => {
   535→                e.preventDefault()
   536→                if (cancelTarget) cancelMutation.mutate(cancelTarget.id)
   537→              }}
   538→            >
   539→              {cancelMutation.isPending ? (
   540→                <>
   541→                  <Loader2 className="h-4 w-4 animate-spin" /> Cancelling…
   542→                </>
   543→              ) : (
   544→                <>
   545→                  <XCircle className="h-4 w-4" /> Reject &amp; Cancel
   546→                </>
   547→              )}
   548→            </AlertDialogAction>
   549→          </AlertDialogFooter>
   550→        </AlertDialogContent>
   551→      </AlertDialog>
   552→    </div>
   553→  )
   554→}

   560→function CycleCountDetailPanel({
   561→  countId,
   562→  onBack,
   563→  onStart,
   564→  onApprove,
   565→  onCancel,
   566→  starting,
   567→  approving,
   568→  canManage,
   569→}: {
   570→  countId: string
   571→  onBack: () => void
   572→  onStart: () => void
   573→  onApprove: () => void
   574→  onCancel: () => void
   575→  starting: boolean
   576→  approving: boolean
   577→  canManage: boolean
   578→}) {
   579→  const detailQuery = useQuery<CycleCountDetailResponse>({
   580→    queryKey: ['cycle-count', countId],
   581→    queryFn: () => api.get<CycleCountDetailResponse>(`/api/cycle-counts/${countId}`),
   582→    staleTime: 10_000,
   583→  })
   584→
   585→  const count = detailQuery.data?.count
   586→
   587→  if (detailQuery.isLoading) {
   588→    return (
   589→      <Card>
   590→        <CardContent className="p-5 space-y-3">
   591→          <Skeleton className="h-8 w-1/2" />
   592→          <Skeleton className="h-32" />
   593→        </CardContent>
   594→      </Card>
   595→    )
   596→  }
   597→
   598→  if (detailQuery.isError || !count) {
   599→    return (
   600→      <Card>
   601→        <CardContent className="p-10 text-center">
   602→          <p className="text-sm text-muted-foreground mb-4">
   603→            {detailQuery.isError ? getErrorMessage(detailQuery.error) : 'Count not found.'}
   604→          </p>
   605→          <Button variant="outline" onClick={() => detailQuery.refetch()}>
   606→            Try again
   607→          </Button>
   608→        </CardContent>
   609→      </Card>
