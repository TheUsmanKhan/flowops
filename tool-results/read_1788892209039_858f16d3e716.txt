   133→const PKR = new Intl.NumberFormat('en-PK', { maximumFractionDigits: 0 })
   134→function formatPKR(n: number): string {
   135→  return `Rs. ${PKR.format(n)}`
   136→}
   137→
   138→function getErrorMessage(err: unknown): string {
   139→  if (err instanceof FetchError) return err.message
   140→  if (err instanceof Error) return err.message
   141→  return 'Something went wrong. Please try again.'
   142→}
   143→
   144→function parseQty(v: string): number {
   145→  if (v === '') return 0
   146→  const n = Number(v)
   147→  return Number.isFinite(n) && n >= 0 ? Math.floor(n) : 0
   148→}
   149→
   150→function parseCost(v: string): number {
   151→  if (v === '') return 0
   152→  const n = Number(v)
   153→  return Number.isFinite(n) && n >= 0 ? n : 0
   154→}
   155→
   156→function todayISO(): string {
   157→  return new Date().toISOString().split('T')[0]
   158→}

   237→  )

   278→        ...prev,
   279→        {

   292→  const removeItem = (variantId: string) => {
   293→    setItems((prev) => prev.filter((i) => i.variantId !== variantId))
   294→  }
   295→

   298→      prev.map((i) => (i.variantId === variantId ? { ...i, ...patch } : i)),

   328→        org_variant_id: i.variantId,
   329→        ordered_quantity: i.quantity,

   364→                  className="font-medium underline underline-offset-4 hover:text-primary"
   365→                  onClick={() => setCreateSupplierOpen(true)}
   366→                >
   367→                  Add a supplier
   368→                </button>{' '}
   369→                or visit the Suppliers page.
   370→              </span>
   371→            )}
   372→            {noLocations && (
   373→              <span className="block mt-1">
   374→                <button
   375→                  type="button"
   376→                  className="font-medium underline underline-offset-4 hover:text-primary"
   377→                  onClick={() => navigate({ name: 'inventory-locations' })}
   378→                >
   379→                  Create a location
   380→                </button>{' '}
   381→                first.
   382→              </span>
   383→            )}
   384→          </AlertDescription>

   386→      )}
   387→
   388→      {!canManage && (
   389→        <Alert variant="destructive">
   390→          <AlertCircle className="h-4 w-4" />
   391→          <AlertTitle>Permission required</AlertTitle>
   392→          <AlertDescription>
   393→            You don&apos;t have permission to manage purchase orders. Save your work elsewhere.
   394→          </AlertDescription>
   395→        </Alert>
   396→      )}
   397→
   398→      <div className="grid gap-6 lg:grid-cols-3">
   399→        {/* ── Left: form + items ────────────────────────────────────────── */}
   400→        <div className="lg:col-span-2 space-y-4">

   402→          <Card>
   403→            <CardHeader className="pb-3">
   404→              <CardTitle className="text-base flex items-center gap-2">
   405→                <ShoppingCart className="h-4 w-4" /> Order details
   406→              </CardTitle>
   407→            </CardHeader>
   408→            <CardContent className="space-y-4">
   409→              <div className="grid sm:grid-cols-2 gap-4">
   410→                <div className="space-y-1.5">
   411→                  <Label htmlFor="po-supplier">
   412→                    <span className="flex items-center justify-between">
   413→                      <span className="flex items-center gap-1.5">

   417→                        type="button"

   421→                        + Create new

   423→                    </span>

   425→                  {suppliersQuery.isLoading ? (
   426→                    <Skeleton className="h-9" />

   428→                    <Select value={supplierId} onValueChange={setSupplierId}>
   429→                      <SelectTrigger id="po-supplier">
   430→                        <SelectValue placeholder="Select supplier" />

   436→                            {s.contactPerson ? ` · ${s.contactPerson}` : ''}
   437→                          </SelectItem>
   438→                        ))}
   439→                      </SelectContent>
   440→                    </Select>

   443→                    <p className="text-xs text-muted-foreground">

   447→                </div>

   452→                      <Warehouse className="h-3.5 w-3.5" /> Delivery location

   488→                <div className="space-y-1.5">

   490→                    <span className="flex items-center gap-1.5">

   504→

   572→                      <TableHead className="text-right w-28">Line total</TableHead>

   574→                    </TableRow>

   576→                  <TableBody>

   578→                      <TableRow>
   579→                        <TableCell

   587→                      items.map((item) => (
   588→                        <TableRow key={item.variantId}>
   589→                          <TableCell>
   590→                            <div className="flex flex-col">
   591→                              <span className="text-sm font-medium">{item.productTitle}</span>
   592→                              <span className="text-xs text-muted-foreground font-mono">
   593→                                {item.sku}
   594→                              </span>

   596→                          </TableCell>
   597→                          <TableCell>
   598→                            <Input
   599→                              type="number"

   602→                              className="h-8 tabular-nums"
   603→                              value={item.quantity}
   604→                              onChange={(e) =>
   605→                                updateItem(item.variantId, {

   639→                        </TableRow>
   640→                      ))
   641→                    )}

   819→function QuickCreateSupplierDialog({
   820→  open,
   821→  onOpenChange,
   822→  onCreated,
   823→}: {
   824→  open: boolean
   825→  onOpenChange: (v: boolean) => void
   826→  onCreated: (id: string) => void
   827→}) {
   828→  const [name, setName] = useState('')
   829→  const [contactPerson, setContactPerson] = useState('')
   830→  const [phone, setPhone] = useState('')
   831→
   832→  const createMutation = useIdempotentMutation<{ id: string; name: string }, {
   833→    name: string
   834→    contactPerson?: string
   835→    phone?: string
   836→    isOrgLevel: boolean
   837→  }>({
   838→    url: '/api/suppliers',
   839→    mutationOptions: {
   840→      onSuccess: (data) => {
   841→        toast.success(`Supplier "${data.name}" created.`)
   842→        onCreated(data.id)
   843→        setName('')
   844→        setContactPerson('')
   845→        setPhone('')
   846→        onOpenChange(false)
   847→      },
   848→      onError: (err) => toast.error(getErrorMessage(err)),
   849→    },
   850→  })
   851→
   852→  return (
   853→    <Dialog open={open} onOpenChange={onOpenChange}>
   854→      <DialogContent className="sm:max-w-md">
   855→        <DialogHeader>
   856→          <DialogTitle>Create new supplier</DialogTitle>
   857→          <DialogDescription>
   858→            Quick-add a supplier. You can fill in payment terms and more details from the
   859→            Suppliers page later.
   860→          </DialogDescription>
   861→        </DialogHeader>
   862→        <form
   863→          onSubmit={(e) => {
   864→            e.preventDefault()
   865→            if (name.trim().length < 2) {
   866→              toast.error('Name must be at least 2 characters.')
   867→              return
   868→            }
   869→            createMutation.mutate({
   870→              name: name.trim(),
   871→              contactPerson: contactPerson.trim() || undefined,
   872→              phone: phone.trim() || undefined,
   873→              isOrgLevel: false,
   874→            })
   875→          }}
   876→          className="space-y-3"
   877→        >
   878→          <div className="space-y-1.5">
   879→            <Label htmlFor="qs-name">Name</Label>
   880→            <Input
   881→              id="qs-name"
   882→              autoFocus
   883→              placeholder="e.g. ABC Fabrics"
   884→              value={name}
   885→              onChange={(e) => setName(e.target.value)}
   886→            />
   887→          </div>
   888→          <div className="grid grid-cols-2 gap-3">
   889→            <div className="space-y-1.5">
   890→              <Label htmlFor="qs-contact">Contact person</Label>
   891→              <Input
   892→                id="qs-contact"
   893→                placeholder="Optional"
   894→                value={contactPerson}
   895→                onChange={(e) => setContactPerson(e.target.value)}
   896→              />
   897→            </div>
   898→            <div className="space-y-1.5">
   899→              <Label htmlFor="qs-phone">Phone</Label>
   900→              <Input
   901→                id="qs-phone"
   902→                placeholder="Optional"
   903→                value={phone}
   904→                onChange={(e) => setPhone(e.target.value)}
   905→              />
   906→            </div>
   907→          </div>
   908→          <DialogFooter>
   909→            <Button type="button" variant="outline" onClick={() => onOpenChange(false)}>
   910→              Cancel
   911→            </Button>
   912→            <Button type="submit" disabled={createMutation.isPending}>
   913→              {createMutation.isPending ? (
   914→                <>
   915→                  <Loader2 className="h-4 w-4 animate-spin" /> Creating…
   916→                </>
   917→              ) : (
   918→                'Create supplier'
   919→              )}
   920→            </Button>
   921→          </DialogFooter>
   922→        </form>
   923→      </DialogContent>
   924→    </Dialog>
   925→  )
   926→}
