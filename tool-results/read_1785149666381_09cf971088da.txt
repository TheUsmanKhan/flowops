   176→function parseQty(v: string): number {
   177→  if (v === '') return 0
   178→  const n = Number(v)
   179→  return Number.isFinite(n) && n >= 0 ? Math.floor(n) : 0

   197→/** Pick the shipping address from a customer's saved addresses array.
   198→ *  Honors the legacy fallback: untyped addresses are treated as shipping. */
   199→function pickShippingAddress(addresses: CustomerAddress[] | undefined): CustomerAddress | null {
   200→  if (!addresses || addresses.length === 0) return null
   201→  return (
   202→    addresses.find((a) => a.type === 'shipping') ??
   203→    addresses.find((a) => !a.type) ??
   204→    addresses[0]
   205→  )
   206→}

   555→  }

   557→  // ── Validate the full form using the Zod schema ───────────────────────────
   558→  function validate(): Record<string, string> {
   559→    const errs: Record<string, string> = {}
   560→    const payload = buildPayload()
   561→    const result = createManualOrderSchema.safeParse(payload)
   562→    if (!result.success) {
   563→      for (const issue of result.error.issues) {
   564→        const key = issue.path.join('.')
   565→        if (!errs[key]) errs[key] = issue.message
   566→      }
   567→    }
   568→    return errs
   569→  }

   571→  // ── Upload payment proof + persist URL to the order ───────────────────────
   572→  // Returns true on success, false on any failure. NEVER throws.
   573→  async function uploadPaymentProof(orderId: string): Promise<boolean> {
   574→    if (!paymentProofFile) return true // nothing to upload — trivially "success"
   575→
   576→    setUploadingProof(true)
   577→    try {
   578→      // Step 1: upload the raw file via /api/upload
   579→      const fd = new FormData()
   580→      fd.append('file', paymentProofFile)
   581→      const uploadRes = await fetch(
   582→        `/api/upload?type=payment-proofs&id=${orderId}`,
   583→        { method: 'POST', body: fd },
   584→      )
   585→      const uploadText = await uploadRes.text()
   586→      let uploadBody: unknown = null
   587→      if (uploadText) {
   588→        try {
   589→          uploadBody = JSON.parse(uploadText)
   590→        } catch {
   591→          uploadBody = uploadText
   592→        }
   593→      }
   594→      if (!uploadRes.ok) {
   595→        const msg =
   596→          uploadBody && typeof uploadBody === 'object' && 'error' in uploadBody
   597→            ? String((uploadBody as { error: unknown }).error)
