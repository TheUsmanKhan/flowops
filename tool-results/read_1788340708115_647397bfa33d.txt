     1→/**
     2→ * Customer Management System — server actions.
     3→ *
     4→ * This is the standalone Customer Management System (Step 2). It owns all
     5→ * CRUD for customers / phones / addresses, the cross-platform identity
     6→ * matching wrapper, cached-stats recomputation with auto-flagging, and the
     7→ * listing/detail endpoints.
     8→ *
     9→ * The Order system (order.actions.ts) CONSUMES this module's data and the
    10→ * `markAddressAsUsed` / `updateCustomerStats` helpers — it does not touch
    11→ * customer_phones / customer_addresses / customer_external_identities
    12→ * directly.
    13→ *
    14→ * DESIGN RULES (enforced throughout):
    15→ *  1. Phone normalization is ALWAYS done via the normalize_phone() SQL
    16→ *     function (single source of truth). Raw phone strings are never
    17→ *     compared directly.
    18→ *  2. A customer must always have >= 1 phone and >= 1 address. The
    19→ *     remove-actions refuse to delete the last remaining phone/address.
    20→ *  3. At most one is_primary phone and one is_default address per customer
    21→ *     (DB partial unique indexes + action-layer unsetting).
    22→ *  4. Every mutation calls insertAuditLog() and returns { success, data?, error? }.
    23→ *  5. All reads are organization-scoped via getWorkspace().

   141→/**
   142→ * Call the normalize_phone() SQL function. Returns the E.164 canonical form
   143→ * (e.g. "+923001234567") or null for empty/invalid input.
   144→ *
   145→ * Why a DB call instead of a TS port: the SQL function is the authoritative
   146→ * normalizer used by customer_phones indexes, match_or_create_customer(),
   147→ * and all matching logic. Keeping a single implementation guarantees the
   148→ * client debounce-preview and the server agree on what "matches".
   149→ */
   150→async function normalizePhone(raw: string): Promise<string | null> {
   151→  const rows = await db.$queryRaw<{ normalized: string | null }[]>`
   152→    SELECT normalize_phone(${raw}::TEXT) AS normalized
   153→  `
   154→  return rows[0]?.normalized ?? null
   155→}

   213→// ──────────────────────────────────────────────────────────────
   214→// searchCustomerByPhone
   215→// ──────────────────────────────────────────────────────────────
   216→/**
   217→ * Used by the Order creation page's live search. Normalizes the input via
   218→ * the SQL function, then looks up customer_phones within the active org.
   219→ *
   220→ * If found, returns the FULL customer record with ALL their phones and
   221→ * ALL their addresses (so the order form can populate the address dropdown
   222→ * and let the user pick which saved address to ship to).
   223→ *
   224→ * Returns { found: false } when no match — the caller decides whether to
   225→ * show an inline "create new customer" form.
   226→ */
   227→export async function searchCustomerByPhone(
   228→  phone: string,
   229→): Promise<ActionResult<{
   230→  found: boolean
   231→  customer?: {
   232→    id: string
   233→    name: string
   234→    email: string | null
   235→    totalOrdersCount: number
   236→    totalRtoCount: number
   237→    isFlagged: boolean
   238→    flaggedReason: string | null
   239→    phones: PhoneDTO[]
   240→    addresses: AddressDTO[]
   241→  }
   242→}>> {
   243→  try {
   244→    const ctx = await getWorkspace()
   245→
   246→    if (!phone || !phone.trim()) {
   247→      return { success: true, data: { found: false } }
   248→    }
   249→
   250→    const normalized = await normalizePhone(phone.trim())
   251→    if (!normalized) {
   252→      return { success: true, data: { found: false } }
   253→    }
   254→
   255→    // Find the customer_phones row for this normalized phone in this org.
   256→    const phoneRow = await db.customerPhone.findFirst({
   257→      where: {
   258→        organizationId: ctx.company.organizationId,
   259→        phoneNormalized: normalized,
   260→      },
   261→      select: { customerId: true },
   262→    })
   263→
   264→    if (!phoneRow) {
   265→      return { success: true, data: { found: false } }
   266→    }
   267→
   268→    // Fetch the full customer + all phones + all addresses.
   269→    const customer = await db.customer.findFirst({
   270→      where: {
   271→        id: phoneRow.customerId,
   272→        organizationId: ctx.company.organizationId,
   273→      },
   274→      include: {
   275→        phones: { orderBy: { isPrimary: 'desc' } },
   276→        addresses: {
   277→          orderBy: [
   278→            { isDefault: 'desc' },
   279→            { lastUsedAt: { sort: 'desc', nulls: 'last' } },
   280→          ],
   281→        },
   282→      },
   283→    })
   284→
   285→    if (!customer) {
   286→      return { success: true, data: { found: false } }
   287→    }
   288→
   289→    return {
   290→      success: true,
   291→      data: {
   292→        found: true,
   293→        customer: {
   294→          id: customer.id,
   295→          name: customer.name,
   296→          email: customer.email,
   297→          totalOrdersCount: customer.totalOrdersCount,
   298→          totalRtoCount: customer.totalRtoCount,
   299→          isFlagged: customer.isFlagged,
   300→          flaggedReason: customer.flaggedReason,
   301→          phones: customer.phones.map(toPhoneDTO),
   302→          addresses: customer.addresses.map(toAddressDTO),
   303→        },
   304→      },
   305→    }
   306→  } catch (err) {
   307→    return {
   308→      success: false,
   309→      error: err instanceof Error ? err.message : 'Failed to search customer by phone',
   310→    }
   311→  }
   312→}

   314→// ──────────────────────────────────────────────────────────────
   315→// createCustomer
   316→// ──────────────────────────────────────────────────────────────
   317→/**
   318→ * Create a new customer with at least one phone and one address.
   319→ *
   320→ * Before inserting, normalizes each phone and checks for org-wide
   321→ * phone_normalized conflicts. If any phone already belongs to another
   322→ * customer, returns a clear error suggesting the caller search for the
   323→ * existing customer instead of creating a duplicate.
   324→ *
   325→ * All inserts run in a single transaction so a partial failure (e.g. an
   326→ * address insert fails) rolls back the customer + phones too.
   327→ */
   328→export async function createCustomer(
   329→  input: CreateCustomerInput,
   330→): Promise<ActionResult<{ customerId: string }>> {
   331→  try {
   332→    const ctx = await getWorkspace()
   333→    await requirePermission(ctx, PERMISSIONS.ORDERS_CREATE)
   334→    // Delegate to the internal variant that skips the (now-redundant)
   335→    // getWorkspace + requirePermission resolution.
   336→    return createCustomerInternal(ctx, input)
   337→  } catch (err) {
   338→    return {
   339→      success: false,
   340→      error: err instanceof Error ? err.message : 'Failed to create customer',
   341→    }
   342→  }
   343→}
   344→
   345→/**
   346→ * Internal customer-creation variant that accepts a PRE-RESOLVED workspace
   347→ * context. Used by createManualOrder() which has already called
   348→ * getWorkspace() + requirePermission() — calling the public createCustomer()
   349→ * from there would redundantly re-resolve the context + re-check permissions
   350→ * (~3 extra DB round-trips on the new-customer order path).
   351→ *
   352→ * Returns the created customerId PLUS the default address + primary phone IDs
   353→ * so the caller (createManualOrder) can set usedCustomerAddressId /
   354→ * usedCustomerPhoneId WITHOUT two extra findFirst queries.
   355→ *
   356→ * Behavior is IDENTICAL to createCustomer() minus the redundant resolution.
   357→ */
   358→export async function createCustomerInternal(
   359→  ctx: { company: { id: string; organizationId: string }; user: { id: string }; employee: { id: string } },
   360→  input: CreateCustomerInput,
   361→): Promise<ActionResult<{ customerId: string; defaultAddressId: string | null; primaryPhoneId: string | null }>> {
   362→  try {
   363→    const parsed = createCustomerSchema.safeParse(input)
   364→    if (!parsed.success) {
   365→      return {
   366→        success: false,
   367→        error: parsed.error.issues[0]?.message ?? 'Invalid customer data',
   368→      }
   369→    }
   370→    const d = parsed.data
   371→
   372→    // Phone validation country hint (Phase 3 — Country System): use the
   373→    // customer's default (or first) address's country as the defaultCountry
   374→    // for interpreting AMBIGUOUS phone numbers (those without a + prefix).
   375→    // This is a HINT only — numbers WITH a + prefix are always validated
   376→    // internationally regardless of this setting (libphonenumber-js's
   377→    // defaultCountry only applies to no-+-prefix numbers). So a Pakistani
   378→    // phone (+92...) on a UK-delivery order still validates fine, and a UK
   379→    // local number (no +) on a UK address validates against UK rules.
   380→    // Falls back to 'PK' (the existing default) when no address/country.
   381→    // NOTE: country is now stored as an alpha-2 code directly — no
   382→    // name→code translation needed.
   383→    const phoneHintAddress = d.addresses.find((a) => a.is_default) ?? d.addresses[0]
   384→    const phoneCountryCode = phoneHintAddress?.country || 'PK'
   385→
   386→    // 1. Validate + normalize every phone using libphonenumber-js (international).
   387→    // This replaces the SQL normalize_phone() for the application-layer path.
   388→    // The SQL function remains for DB-level triggers/matching.
   389→    const phonesWithNormalized: Array<{
   390→      phoneRaw: string
   391→      phoneNormalized: string
   392→      label: string | null
   393→      isPrimary: boolean
   394→    }> = []
   395→    for (const p of d.phones) {
   396→      const { isValid, normalized } = validateAndNormalizePhone(p.phone.trim(), phoneCountryCode)
   397→      if (!isValid || !normalized) {
   398→        return {
   399→          success: false,
   400→          error: `Phone "${p.phone}" is not a valid phone number. Use format like 03001234567 or +923001234567.`,
   401→        }
   402→      }
   403→      phonesWithNormalized.push({
   404→        phoneRaw: p.phone.trim(),
   405→        phoneNormalized: normalized,
   406→        label: p.label?.trim() || null,
   407→        isPrimary: p.is_primary,
   408→      })
   409→    }
   410→
   411→    // 2. Check for org-wide phone conflicts BEFORE inserting.
   412→    //    A normalized phone maps to exactly one customer per org (DB unique
   413→    //    constraint). If any of the provided phones already belongs to
   414→    //    another customer, we refuse and point the caller to that customer.
   415→    if (phonesWithNormalized.length > 0) {
   416→      const conflicts = await db.customerPhone.findMany({
   417→        where: {
   418→          organizationId: ctx.company.organizationId,
   419→          phoneNormalized: { in: phonesWithNormalized.map((p) => p.phoneNormalized) },
   420→        },
   421→        select: {
   422→          phoneNormalized: true,
   423→          customerId: true,
   424→          customer: { select: { name: true } },
   425→        },
   426→      })
   427→      if (conflicts.length > 0) {
   428→        const c = conflicts[0]
   429→        return {
   430→          success: false,
   431→          error: `Phone ${c.phoneNormalized} already belongs to customer "${c.customer.name}". Use search to find this existing customer instead of creating a duplicate.`,
   432→        }
   433→      }
   434→    }
   435→
   436→    // 3. Normalize addresses (no province — only address + city + country).
   437→    const addressesData = d.addresses.map((a) => ({
   438→      label: a.label?.trim() || null,
   439→      address: a.address.trim(),
   440→      city: a.city.trim(),
   441→      // Country (Phase: Country System). Defaults to "PK" when the
   442→      // caller didn't send one (addressInputSchema default).
   443→      country: a.country?.trim() || 'PK',
   444→      isDefault: a.is_default,
   445→      lastUsedAt: null,
   446→    }))
   447→
   448→    // 4. Insert customer + phones + addresses in a single transaction.
   449→    //    Uses createManyAndReturn so we capture the created row IDs
   450→    //    (default address + primary phone) WITHOUT two extra findFirst
   451→    //    queries afterward — createManualOrder needs these IDs to set
   452→    //    usedCustomerAddressId / usedCustomerPhoneId on the order.
   453→    const { customer, defaultAddressId, primaryPhoneId } = await db.$transaction(async (tx) => {
   454→      const created = await tx.customer.create({
   455→        data: {
   456→          organizationId: ctx.company.organizationId,
   457→          name: d.name.trim(),
   458→          email: d.email?.trim() || null,
   459→          createdBy: ctx.employee.id,
   460→        },
   461→      })
   462→
   463→      const phones = await tx.customerPhone.createManyAndReturn({
   464→        data: phonesWithNormalized.map((p) => ({
   465→          customerId: created.id,
   466→          organizationId: ctx.company.organizationId,
   467→          phoneRaw: p.phoneRaw,
   468→          phoneNormalized: p.phoneNormalized,
   469→          label: p.label,
   470→          isPrimary: p.isPrimary,
   471→        })),
   472→        select: { id: true, isPrimary: true },
   473→      })
   474→
   475→      const addresses = await tx.customerAddress.createManyAndReturn({
   476→        data: addressesData.map((a) => ({
   477→          customerId: created.id,
   478→          organizationId: ctx.company.organizationId,
   479→          label: a.label,
   480→          address: a.address,
   481→          city: a.city,
   482→          country: a.country,
   483→          isDefault: a.isDefault,
   484→          lastUsedAt: a.lastUsedAt,
   485→        })),
   486→        select: { id: true, isDefault: true },
   487→      })
   488→
   489→      const defaultAddrId = addresses.find((a) => a.isDefault)?.id ?? null
   490→      const primaryPhId = phones.find((p) => p.isPrimary)?.id ?? null
   491→
   492→      return { customer: created, defaultAddressId: defaultAddrId, primaryPhoneId: primaryPhId }
   493→    })
   494→
   495→    // 5. Audit log.
   496→    insertAuditLog({
   497→      action: 'customer.created',
   498→      entityType: 'customer',
   499→      entityId: customer.id,
   500→      companyId: ctx.company.id,
   501→      organizationId: ctx.company.organizationId,
   502→      userId: ctx.user.id,
   503→      employeeId: ctx.employee.id,
   504→      newValues: {
   505→        name: customer.name,
   506→        email: customer.email,
   507→        phoneCount: phonesWithNormalized.length,
   508→        addressCount: addressesData.length,
   509→      },
   510→    })
   511→
   512→    return { success: true, data: { customerId: customer.id, defaultAddressId, primaryPhoneId } }
   513→  } catch (err) {
   514→    return {
   515→      success: false,
   516→      error: err instanceof Error ? err.message : 'Failed to create customer',
   517→    }
   518→  }
   519→}

   522→// updateCustomer — name/email only
   523→// ──────────────────────────────────────────────────────────────
   524→export async function updateCustomer(
   525→  input: UpdateCustomerInput,
   526→): Promise<ActionResult<{ customerId: string }>> {
   527→  try {
   528→    const ctx = await getWorkspace()
   529→    await requirePermission(ctx, PERMISSIONS.ORDERS_CREATE)
   530→
   531→    const parsed = updateCustomerSchema.safeParse(input)
   532→    if (!parsed.success) {
   533→      return {
   534→        success: false,
   535→        error: parsed.error.issues[0]?.message ?? 'Invalid customer data',
   536→      }
   537→    }
   538→    const d = parsed.data
   539→
   540→    const existing = await db.customer.findFirst({
   541→      where: { id: d.customer_id, organizationId: ctx.company.organizationId },
   542→    })
   543→    if (!existing) return { success: false, error: 'Customer not found' }
   544→
   545→    const updateData: Prisma.CustomerUncheckedUpdateInput = {}
   546→    if (d.name !== undefined && d.name.trim() !== existing.name) {
   547→      updateData.name = d.name.trim()
   548→    }
   549→    if (d.email !== undefined) {
   550→      const newEmail = d.email.trim() || null
   551→      if (newEmail !== existing.email) updateData.email = newEmail
   552→    }
   553→
   554→    if (Object.keys(updateData).length === 0) {
   555→      return { success: true, data: { customerId: existing.id } }
   556→    }
   557→
   558→    await db.customer.update({ where: { id: existing.id }, data: updateData })
   559→
   560→    insertAuditLog({
   561→      action: 'customer.updated',
   562→      entityType: 'customer',
   563→      entityId: existing.id,
   564→      companyId: ctx.company.id,
   565→      organizationId: ctx.company.organizationId,
   566→      userId: ctx.user.id,
   567→      employeeId: ctx.employee.id,
   568→      oldValues: { name: existing.name, email: existing.email },
   569→      newValues: updateData,
   570→    })
   571→
   572→    return { success: true, data: { customerId: existing.id } }
   573→  } catch (err) {
   574→    return {
   575→      success: false,
   576→      error: err instanceof Error ? err.message : 'Failed to update customer',
   577→    }
   578→  }
   579→}

   584→export async function addCustomerPhone(
   585→  customerId: string,
   586→  input: PhoneInput,
   587→): Promise<ActionResult<{ phoneId: string }>> {
   588→  try {
   589→    const ctx = await getWorkspace()
   590→    await requirePermission(ctx, PERMISSIONS.ORDERS_CREATE)
   591→
   592→    const parsed = phoneInputSchema.safeParse(input)
   593→    if (!parsed.success) {
   594→      return {
   595→        success: false,
   596→        error: parsed.error.issues[0]?.message ?? 'Invalid phone data',
   597→      }
   598→    }
   599→    const d = parsed.data
   600→
   601→    const customer = await db.customer.findFirst({
   602→      where: { id: customerId, organizationId: ctx.company.organizationId },
   603→    })
   604→    if (!customer) return { success: false, error: 'Customer not found' }
   605→
   606→    const normalized = await normalizePhone(d.phone.trim())
   607→    if (!normalized) {
   608→      return {
   609→        success: false,
   610→        error: `Could not normalize phone "${d.phone}" — please enter a valid phone number`,
   611→      }
   612→    }
   613→
   614→    // Org-wide uniqueness: a normalized phone maps to exactly one customer.

   723→// ──────────────────────────────────────────────────────────────
   724→// Phase 7: validateCustomerAddressCity — non-blocking early warning
   725→// ──────────────────────────────────────────────────────────────
   726→/**
   727→ * Check a customer address's city against ALL of the company's connected
   728→ * courier integrations' cached operational cities. Updates the
   729→ * `cityMatchedCouriers` array and `cityValidatedAt` timestamp on the
   730→ * CustomerAddress row.
   731→ *
   732→ * This is INFORMATIONAL ONLY — an early warning mechanism, NOT a guarantee.
   733→ * The authoritative check always remains `revalidateCityAtBookingTime()` at
   734→ * booking time (with its live-fallback + staleness protection).
   735→ *
   736→ * Uses exact case-insensitive match only — no fuzzy matching here (this is a
   737→ * background check, not an interactive resolution flow).
   738→ *
   739→ * Non-blocking: called fire-and-forget after address create/update. If the
   740→ * city matches zero couriers, the address is still saved successfully — the
   741→ * empty `cityMatchedCouriers` array surfaces a soft warning in the UI.
   742→ */
   743→async function validateCustomerAddressCity(
   744→  addressId: string,
   745→  city: string,
   746→  companyId: string,
   747→  country?: string,
   748→): Promise<void> {
   749→  if (!city.trim()) return
   750→
   751→  // ── Country guard (Phase 3): skip for non-Pakistan addresses ──
   752→  // courier_operational_cities is 100% Pakistan-sourced (PostEx/TCS/Leopard
   753→  // only operate in Pakistan). A foreign city will never match, and running
   754→  // the check would just set an empty cityMatchedCouriers array + a misleading
   755→  // cityValidatedAt timestamp. For non-Pakistan, leave the cached fields
   756→  // untouched (they default to empty/null) — the UI treats the city as plain
   757→  // free text with no courier-matching UI either.
   758→  if (country && country !== 'PK') {
   759→    return
   760→  }
   761→
   762→  // Fetch all active courier integrations for this company
   763→  const integrations = await db.companyIntegration.findMany({
   764→    where: { companyId, isActive: true, provider: { category: 'courier' } },
   765→    select: { provider: { select: { providerKey: true } } },
   766→  })
   767→  const providerKeys = [...new Set(integrations.map((i) => i.provider.providerKey))]
   768→
   769→  if (providerKeys.length === 0) {
   770→    // No couriers connected — set empty array + timestamp
   771→    await db.customerAddress.update({
   772→      where: { id: addressId },
   773→      data: { cityMatchedCouriers: [], cityValidatedAt: new Date() },
   774→    })
   775→    return
   776→  }
   777→
   778→  // Check each provider's cached cities for an exact case-insensitive match
   779→  const matchedProviders: string[] = []
   780→  for (const providerKey of providerKeys) {
   781→    const matched = await db.courierOperationalCity.findFirst({
   782→      where: {
   783→        providerKey,
   784→        cityName: { equals: city, mode: 'insensitive' },
   785→        isDeliveryCity: true,
   786→      },
   787→      select: { id: true },
   788→    })
   789→    if (matched) {
   790→      matchedProviders.push(providerKey)
   791→    }
   792→  }
   793→
   794→  await db.customerAddress.update({
   795→    where: { id: addressId },
   796→    data: { cityMatchedCouriers: matchedProviders, cityValidatedAt: new Date() },
   797→  })
   798→}

  1045→// ──────────────────────────────────────────────────────────────
  1046→// matchOrCreateExternalCustomer
  1047→// ──────────────────────────────────────────────────────────────
  1048→/**
  1049→ * Thin wrapper around the match_or_create_customer() SQL function built in
  1050→ * Step 1. Implements the layered matching strategy:
  1051→ *   1. exact_identity  — existing (org, platform, external_customer_id) mapping
  1052→ *   2. phone_match     — normalize phone, match customer_phones
  1053→ *   3. email_match     — match Customer.email within org
  1054→ *   4. create          — create new customer + primary phone + external identity
  1055→ *
  1056→ * A future Shopify/Daraz/Instagram webhook handler will call this directly.
  1057→ * No live webhook exists in this step — this is built so integration work
  1058→ * later is a drop-in call rather than new logic.
  1059→ *
  1060→ * NOTE: this function does NOT require a workspace session — it's intended
  1061→ * to be called from a webhook context where there's no logged-in user.
  1062→ * The SQL function is SECURITY DEFINER so it can insert into Customer /
  1063→ * customer_phones / customer_external_identities regardless of caller
  1064→ * permissions. The caller MUST supply organizationId (resolved from the
  1065→ * webhook's target company).
  1066→ */
  1067→export async function matchOrCreateExternalCustomer(
  1068→  input: MatchExternalCustomerInput & { organizationId: string },
  1069→): Promise<ActionResult<{
  1070→  customerId: string
  1071→  wasNewlyCreated: boolean
  1072→  matchedVia: string
  1073→}>> {
  1074→  try {
  1075→    const parsed = matchExternalCustomerSchema.safeParse(input)
  1076→    if (!parsed.success) {
  1077→      return {
  1078→        success: false,
  1079→        error: parsed.error.issues[0]?.message ?? 'Invalid external customer input',
  1080→      }
  1081→    }
  1082→    const d = parsed.data
  1083→
  1084→    // ── Pre-normalize international phones (Phase: Country System) ──
  1085→    // normalize_phone() (SQL) historically mangled international numbers by
  1086→    // blindly prepending +92 (e.g. +447911123456 → +92447911123456). The SQL
  1087→    // function now passes through + prefixed non-Pakistani numbers unchanged,
  1088→    // BUT it doesn't validate/canonicalize them. For a phone like "+44 7911
  1089→    // 123456" (with spaces) or "+44(0)7911112345", the SQL function would
  1090→    // pass the raw string through with spaces intact — which wouldn't match
  1091→    // a stored "+447911123456" for dedup.
  1092→    //
  1093→    // So: if the phone looks international (+ prefix that isn't +92, OR a
  1094→    // local-format number that validateAndNormalizePhone can parse with a
  1095→    // non-PK country hint), normalize it HERE via libphonenumber-js to a
  1096→    // clean E.164 string BEFORE passing to the SQL function. The SQL function
  1097→    // then passes it through (it sees a + non-92 prefix) and stores/matches
  1098→    // the canonical E.164 value. Pakistani-shaped numbers (0300..., +92...)
  1099→    // are passed through to the SQL function's normalize_phone() unchanged —
  1100→    // no behavior change for the working case.
  1101→    let phoneForSql = d.phone || null
  1102→    if (phoneForSql) {
  1103→      const trimmed = phoneForSql.trim()
  1104→      const looksInternational = trimmed.startsWith('+') && !trimmed.startsWith('+92')
  1105→      if (looksInternational) {
  1106→        const { normalized } = validateAndNormalizePhone(trimmed, 'GB')
  1107→        // Use the libphonenumber-js E.164 output if it parsed successfully.
  1108→        // If it didn't parse (truly invalid number), fall back to the raw
  1109→        // trimmed input — the SQL function's pass-through will store it as-is,
  1110→        // and the isValidFormat flagging below will mark it for correction.
  1111→        phoneForSql = normalized ?? trimmed
  1112→      }
  1113→      // Pakistani-shaped numbers (+92... or 0300... or 10-11 digit local)
  1114→      // are left as-is — the SQL function's normalize_phone() handles them
  1115→      // correctly (proven by existing behavior).
  1116→    }
  1117→
  1118→    // Check whether an external identity mapping already exists BEFORE
  1119→    // calling the SQL function, so we can report wasNewlyCreated accurately.
  1120→    const existingMapping = await db.customerExternalIdentity.findUnique({
  1121→      where: {
  1122→        organizationId_platform_externalCustomerId: {
  1123→          organizationId: input.organizationId,
  1124→          platform: d.platform,
  1125→          externalCustomerId: d.external_customer_id,
  1126→        },
  1127→      },
  1128→      select: { customerId: true, matchedVia: true },
  1129→    })
  1130→
  1131→    // Call the SQL function — it handles all 4 layers race-safely.
  1132→    // phoneForSql is pre-normalized for international numbers (E.164) so
  1133→    // normalize_phone()'s pass-through stores/matches the canonical value.
  1134→    // d.country (alpha-2 code, optional) is passed through so the SQL function
  1135→    // can persist it onto the customer_addresses row for newly-created customers.
  1136→    // Defaults to 'PK' if absent.
  1137→    const countryForSql = d.country?.trim() || 'PK'
  1138→    const rows = await db.$queryRaw<{ customer_id: string }[]>`
  1139→      SELECT match_or_create_customer(
  1140→        ${input.organizationId}::TEXT,
  1141→        ${d.platform}::TEXT,
  1142→        ${d.external_customer_id}::TEXT,
  1143→        ${phoneForSql}::TEXT,
  1144→        ${d.email || null}::TEXT,
  1145→        ${d.name || null}::TEXT,
  1146→        ${countryForSql}::TEXT
  1147→      ) AS customer_id
  1148→    `
  1149→    const customerId = rows[0]?.customer_id
  1150→    if (!customerId) {
  1151→      return { success: false, error: 'match_or_create_customer() returned no customer_id' }
  1152→    }
  1153→
  1154→    const wasNewlyCreated = !existingMapping
  1155→    const matchedVia = existingMapping?.matchedVia ?? 'exact_identity'
  1156→
  1157→    // If this customer was newly created AND a phone was provided, check if
  1158→    // the phone is valid. If not, flag the CustomerPhone row with
  1159→    // isValidFormat=false so it can be surfaced for correction later.
  1160→    // We do NOT block creation — external platforms may send unformatted
  1161→    // or invalid phones, and the customer still needs to be importable.
  1162→    //
  1163→    // Phase: Country System — use the pre-normalized phone (phoneForSql) for
  1164→    // the validity check. For international numbers, isValidPhoneFormat is
  1165→    // called with the default PK country but libphonenumber-js validates +
  1166→    // prefixed numbers internationally regardless of defaultCountry, so a
  1167→    // valid +44... number passes. For Pakistani local numbers, the default
  1168→    // PK country is correct. The phoneRaw match uses phoneForSql (the
  1169→    // pre-normalized value the SQL function stored as phoneRaw) — note the
  1170→    // SQL function stores p_phone (the raw arg) as phoneRaw, which is now
  1171→    // the pre-normalized E.164 value for international numbers.
  1172→    if (wasNewlyCreated && phoneForSql) {
  1173→      const phoneValid = isValidPhoneFormat(phoneForSql)
  1174→      if (!phoneValid) {
  1175→        // Find the phone row that was just created by the SQL function and
  1176→        // flag it. This is fire-and-forget — the customer is already created.
  1177→        db.customerPhone.updateMany({
  1178→          where: {
  1179→            customerId,
  1180→            phoneRaw: phoneForSql,
  1181→          },
  1182→          data: { isValidFormat: false },
  1183→        }).catch(() => {
  1184→          // Non-fatal: if the update fails, the customer is still created.
  1185→          // The phone will default to isValidFormat=true, which is the
  1186→          // less-restrictive option (won't block future orders).
  1187→        })
  1188→      }
  1189→    }
  1190→
  1191→    return {
  1192→      success: true,
  1193→      data: { customerId, wasNewlyCreated, matchedVia },
  1194→    }
  1195→  } catch (err) {
  1196→    return {
  1197→      success: false,
  1198→      error: err instanceof Error ? err.message : 'Failed to match or create external customer',
  1199→    }
  1200→  }
  1201→}

  1255→// ──────────────────────────────────────────────────────────────
  1256→// updateCustomerStats — recomputes cached counts from the orders table
  1257→// ──────────────────────────────────────────────────────────────
  1258→/**
  1259→ * Recompute total_orders_count, total_order_value (sum of delivered orders'
  1260→ * total_order_value), and total_rto_count from the orders table for this
  1261→ * customer, then persist them to the customer row.
  1262→ *
  1263→ * Auto-flagging: if total_rto_count >= 3 AND the customer is not already
  1264→ * flagged for the high-RTO reason, call flagCustomer() with reason
  1265→ * 'High RTO rate (3+ returns)'. Idempotent — won't re-flag if already
  1266→ * flagged for this exact reason.
  1267→ *
  1268→ * This function is called after every order creation and every order status
  1269→ * change (delivered / rto / cancelled) to keep the cached counts current.
  1270→ */
  1271→export async function updateCustomerStats(customerId: string): Promise<ActionResult> {
  1272→  try {
  1273→    // Note: this is an internal helper often called from order.actions.ts
  1274→    // which already has a workspace context. We fetch orders without a
  1275→    // workspace check here (the caller is trusted internal code). The
  1276→    // cached stats are org-scoped by virtue of the customer row itself
  1277→    // being org-scoped.
  1278→    //
  1279→    // DEFINITIONS (applied consistently between cached writes here and
  1280→    // the read-time percentage calculations in customer-detail-view.tsx):
  1281→    //   total_orders_count = COUNT of non-cancelled orders
  1282→    //   total_order_value  = SUM(total_order_value) for delivered + dispatched
  1283→    //                        (orders that have left the warehouse — excludes
  1284→    //                        cancelled, refunded, pending, confirmed, processing)
  1285→    //   total_rto_count    = COUNT where status = 'rto'
  1286→    //   rto_rate / delivery_rate = computed at read-time in the frontend from
  1287→    //                        these cached values (no separate cached columns)
  1288→    const orders = await db.order.findMany({
  1289→      where: { customerId, status: { not: 'cancelled' } },
  1290→      select: { totalOrderValue: true, status: true, deliveryCountry: true },
  1291→    })
  1292→
  1293→    const totalOrdersCount = orders.length
  1294→    // total_order_value = sum of delivered + dispatched orders' total_order_value
  1295→    // (orders that have actually shipped — reflects real revenue, excludes
  1296→    // pending/confirmed orders that may still be cancelled)
  1297→    // Phase F1: use the shared currency-aware revenue function.
  1298→    const revenueOrders = orders
  1299→      .filter((o) => o.status === 'delivered' || o.status === 'dispatched')
  1300→      .map((o) => ({ totalOrderValue: Number(o.totalOrderValue), deliveryCountry: o.deliveryCountry }))
  1301→
  1302→    // Fetch the customer's companyId for market resolution
  1303→    const customerRow = await db.customer.findUnique({
  1304→      where: { id: customerId },
  1305→      select: { organizationId: true },
  1306→    })
  1307→    // Find a company in this org that has this customer's orders
  1308→    const firstOrder = await db.order.findFirst({
  1309→      where: { customerId },
  1310→      select: { companyId: true },
  1311→    })
  1312→    const companyForRevenue = firstOrder?.companyId
  1313→    const companyForBaseCurrency = companyForRevenue
  1314→      ? await db.company.findUnique({ where: { id: companyForRevenue }, select: { baseCurrency: true } })
  1315→      : null
  1316→
  1317→    let totalOrderValue: number
  1318→    if (companyForRevenue && companyForBaseCurrency) {
  1319→      const { computeRevenueWithCurrencies } = await import('@/lib/analytics/revenue')
  1320→      const revenueResult = await computeRevenueWithCurrencies(
  1321→        companyForRevenue,
  1322→        revenueOrders,
  1323→        companyForBaseCurrency.baseCurrency,
  1324→      )
  1325→      totalOrderValue = revenueResult.estimatedTotalBase ?? 0
  1326→    } else {
  1327→      // Fallback: no company found (shouldn't happen — all orders belong to a company)
  1328→      totalOrderValue = revenueOrders.reduce((sum, o) => sum + o.totalOrderValue, 0)
  1329→    }
  1330→
  1331→    const totalRtoCount = orders.filter((o) => o.status === 'rto').length
  1332→
  1333→    await db.customer.update({
  1334→      where: { id: customerId },
  1335→      data: { totalOrdersCount, totalOrderValue, totalRtoCount },
  1336→    })
  1337→
  1338→    // Auto-flag at 3+ RTO (idempotent — only flags if not already flagged
  1339→    // for the high-RTO reason).
  1340→    const RTO_FLAG_REASON = 'High RTO rate (3+ returns)'
  1341→    if (totalRtoCount >= 3) {
  1342→      const customer = await db.customer.findUnique({
  1343→        where: { id: customerId },
  1344→        select: { isFlagged: true, flaggedReason: true },
  1345→      })
  1346→      if (customer && (!customer.isFlagged || customer.flaggedReason !== RTO_FLAG_REASON)) {
  1347→        await flagCustomerInternal(customerId, RTO_FLAG_REASON, /* auto */ true)
  1348→      }
  1349→    }
  1350→
  1351→    return { success: true }
  1352→  } catch (err) {
  1353→    // CRITICAL: never let a stats-update failure break the calling order
  1354→    // action. Log and return failure — the caller continues regardless.
  1355→    console.error('[customer] updateCustomerStats failed:', err)
  1356→    return {
  1357→      success: false,
  1358→      error: err instanceof Error ? err.message : 'Failed to update customer stats',
  1359→    }
  1360→  }
  1361→}

  1386→ * the metric event (to avoid metric spam during bulk recomputes).

  1388→async function flagCustomerInternal(
  1389→  customerId: string,
  1390→  reason: string,
  1391→  auto: boolean,
  1392→  ctx?: {
  1393→    company: { id: string; organizationId: string }

  1399→  const customer = ctx
  1400→    ? await db.customer.findFirst({
  1401→        where: { id: customerId, organizationId },
  1402→      })
  1403→    : await db.customer.findUnique({ where: { id: customerId } })
  1404→  if (!customer) return { success: false, error: 'Customer not found' }
  1405→
  1406→  // Idempotent: skip if already flagged for this exact reason.
  1407→  if (customer.isFlagged && customer.flaggedReason === reason) {
  1408→    return { success: true }
  1409→  }
  1410→
  1411→  await db.customer.update({
  1412→    where: { id: customerId },
  1413→    data: {
  1414→      isFlagged: true,
  1415→      flaggedReason: reason,
  1416→      flaggedAt: new Date(),
  1417→      flaggedBy: ctx?.employee.id ?? null,
  1418→    },
  1419→  })
  1420→
  1421→  if (ctx) {
  1422→    insertAuditLog({
  1423→      action: auto ? 'customer.auto_flagged' : 'customer.flagged',
  1424→      entityType: 'customer',
  1425→      entityId: customerId,
  1426→      companyId: ctx.company.id,
  1427→      organizationId: ctx.company.organizationId,
  1428→      userId: ctx.user.id,
  1429→      employeeId: ctx.employee.id,
  1430→      newValues: { reason, auto },
  1431→    })
  1432→
  1433→    if (!auto) {
  1434→      insertMetricEvent({
  1435→        companyId: ctx.company.id,
  1436→        entityType: 'customer',
  1437→        entityId: customerId,
  1438→        metricKey: 'customer.flagged',
  1439→        numericValue: 1,
  1440→        dimensions: { reason },
  1441→      })
  1442→    }
  1443→  }
  1444→
  1445→  return { success: true }
  1446→}

  1496→// listCustomers
  1497→// ──────────────────────────────────────────────────────────────
  1498→/**
  1499→ * Paginated customer list. `search` matches customer name OR any
  1500→ * associated phone (raw or normalized). Each row includes the primary
  1501→ * phone and default address summary joined in.
  1502→ */
  1503→export async function listCustomers(filters: {
  1504→  search?: string
  1505→  isFlagged?: boolean
  1506→  dateFrom?: string
  1507→  dateTo?: string
  1508→  limit?: number
  1509→  offset?: number
  1510→} = {}): Promise<ActionResult<{ customers: CustomerSummaryDTO[]; total: number }>> {
  1511→  try {
  1512→    const ctx = await getWorkspace()
  1513→    const limit = Math.min(filters.limit ?? 50, 100)
  1514→    const offset = filters.offset ?? 0
  1515→
  1516→    const where: Prisma.CustomerWhereInput = {
  1517→      organizationId: ctx.company.organizationId,
  1518→    }
  1519→    if (filters.isFlagged !== undefined) {
  1520→      where.isFlagged = filters.isFlagged
  1521→    }
  1522→    if (filters.dateFrom || filters.dateTo) {
  1523→      where.createdAt = {}
  1524→      if (filters.dateFrom) where.createdAt.gte = new Date(filters.dateFrom)
  1525→      if (filters.dateTo) where.createdAt.lte = new Date(filters.dateTo)
  1526→    }
  1527→
  1528→    // Search across name + any associated phone (raw or normalized).
  1529→    // We resolve matching customer IDs via a customer_phones query first,
  1530→    // then OR them into the customer name search.
  1531→    let searchCustomerIds: string[] | null = null
  1532→    if (filters.search && filters.search.trim()) {
  1533→      const q = filters.search.trim()
  1534→      // Try normalizing the search term as a phone — if it normalizes to
  1535→      // something, search customer_phones by the normalized form too.
  1536→      const normalizedSearch = await normalizePhone(q)
  1537→      const phoneWhere: Prisma.CustomerPhoneWhereInput = {
  1538→        organizationId: ctx.company.organizationId,
  1539→        OR: [
  1540→          { phoneRaw: { contains: q, mode: 'insensitive' } },
  1541→          ...(normalizedSearch ? [{ phoneNormalized: { contains: normalizedSearch } }] : []),
  1542→        ],
  1543→      }
  1544→      const phoneMatches = await db.customerPhone.findMany({
  1545→        where: phoneWhere,
  1546→        select: { customerId: true },
  1547→        distinct: ['customerId'],
  1548→      })
  1549→      searchCustomerIds = phoneMatches.map((p) => p.customerId)
  1550→
  1551→      where.OR = [
  1552→        { name: { contains: q, mode: 'insensitive' } },
  1553→        { email: { contains: q, mode: 'insensitive' } },
  1554→        ...(searchCustomerIds.length > 0 ? [{ id: { in: searchCustomerIds } }] : []),
  1555→      ]
  1556→    }
  1557→
  1558→    const [customers, total] = await Promise.all([
  1559→      db.customer.findMany({
  1560→        where,
  1561→        orderBy: { createdAt: 'desc' },
  1562→        take: limit,
  1563→        skip: offset,
  1564→        include: {
  1565→          phones: {
  1566→            where: { isPrimary: true },
  1567→            take: 1,
  1568→          },
  1569→          addresses: {
  1570→            where: { isDefault: true },
  1571→            take: 1,
  1572→          },
  1573→        },
  1574→      }),
  1575→      db.customer.count({ where }),
  1576→    ])
  1577→
  1578→    const customerDTOs: CustomerSummaryDTO[] = customers.map((c) => ({
  1579→      id: c.id,
  1580→      name: c.name,
  1581→      email: c.email,
  1582→      primaryPhone: c.phones[0]?.phoneRaw ?? null,
  1583→      defaultAddress: c.addresses[0]
  1584→        ? { address: c.addresses[0].address, city: c.addresses[0].city, country: c.addresses[0].country }
  1585→        : null,
  1586→      totalOrdersCount: c.totalOrdersCount,
  1587→      totalOrderValue: Number(c.totalOrderValue),
  1588→      totalRtoCount: c.totalRtoCount,
  1589→      isFlagged: c.isFlagged,
  1590→      flaggedReason: c.flaggedReason,
  1591→      createdAt: c.createdAt,
  1592→    }))
  1593→
  1594→    return { success: true, data: { customers: customerDTOs, total } }
  1595→  } catch (err) {
  1596→    return {
  1597→      success: false,
  1598→      error: err instanceof Error ? err.message : 'Failed to list customers',
  1599→    }
  1600→  }
  1601→}

  1606→/**

  1682→    const viewerScope = getOrdersDataScope(ctx)
  1683→    const viewerEmployeeId = ctx.employee.id
  1684→

  1687→      data: {
  1688→        id: customer.id,
  1689→        name: customer.name,
  1690→        email: customer.email,
  1691→        totalOrdersCount: customer.totalOrdersCount,
  1692→        totalOrderValue: Number(customer.totalOrderValue),
  1693→        totalRtoCount: customer.totalRtoCount,
  1694→        rtoRate,
  1695→        deliveryRate,
  1696→        isFlagged: customer.isFlagged,
  1697→        flaggedReason: customer.flaggedReason,
  1698→        flaggedAt: customer.flaggedAt,
  1699→        createdAt: customer.createdAt,
  1700→        updatedAt: customer.updatedAt,
  1701→        phones: customer.phones.map(toPhoneDTO),
  1702→        addresses: customer.addresses.map(toAddressDTO),
  1703→        externalIdentities: customer.externalIdentities.map((e) => ({
  1704→          id: e.id,
