    91→    91→// ──────────────────────────────────────────────────────────────
    92→    92→// Processes ALL order_items for an order and attempts to reserve
    93→    93→// stock for each based on its fulfillment_type_snapshot.
    94→    94→//
    95→    95→// For stock_based items:
    96→    96→//   - If sufficient available stock → reserveStockForOrder(), fulfillment_status='reserved'
    97→    97→//   - If insufficient + inventory_policy='continue' → fulfillment_status='backordered'
    98→    98→//   - If insufficient + inventory_policy='deny' → outcome='failed'
    99→    99→//
   100→   100→// For made_to_order items:
   101→   101→//   - checkAndFulfillMadeToOrderVariant() checks returned stock first
   102→   102→//   - If existing_stock → fulfillment_status='reserved', returned_stitched_used=true
   103→   103→//   - If fresh_production → fulfillment_status='reserved', production_order_id set
   104→   104→//   - If error → outcome='failed'
   105→   105→//
   106→   106→// After processing: if ANY item is 'backordered', order status → 'partially_backordered'.
   107→   107→// If ALL items are 'reserved', order stays 'confirmed'.
   108→   108→//
   109→   109→// Called from:
   110→   110→//   - createManualOrder() IF the order auto-confirmed at creation
   111→   111→//   - confirmOrder() when a 'pending' order is manually confirmed
   112→   112→//   - dispatchOrderAction() if status is still 'pending' at dispatch time
   113→   113→
   114→   114→async function reserveOrderStock(
   115→   115→  orderId: string,
   116→   116→  ctx: { company: { id: string; organizationId: string }; user: { id: string }; employee: { id: string } },
   117→   117→): Promise<{
   118→   118→  success: boolean
   119→   119→  results: Array<{ orderItemId: string; outcome: 'reserved' | 'backordered' | 'failed'; reason?: string }>
   120→   120→}> {
   121→   121→  const order = await db.order.findUnique({
   122→   122→    where: { id: orderId },
   123→   123→    select: { dispatchLocationId: true, companyId: true, organizationId: true },
   124→   124→  })
   125→   125→  if (!order) return { success: false, results: [] }
   126→   126→
   127→   127→  const items = await db.orderItem.findMany({
   128→   128→    where: { orderId },
   129→   129→    include: {
   130→   130→      orgVariant: {
   131→   131→        select: { id: true, fulfillmentType: true, inventoryPolicy: true, stitchingCharges: true },
   132→   132→      },
   133→   133→    },
   134→   134→  })
   135→   135→
   136→   136→  const results: Array<{ orderItemId: string; outcome: 'reserved' | 'backordered' | 'failed'; reason?: string }> = []
   137→   137→  let hasBackordered = false
   138→   138→
   139→   139→  for (const item of items) {
   140→   140→    if (item.fulfillmentStatus === 'reserved' || item.fulfillmentStatus === 'dispatched') {
   141→   141→      // Already processed (e.g. re-confirmation attempt) — skip
   142→   142→      results.push({ orderItemId: item.id, outcome: 'reserved' })
   143→   143→      continue
   144→   144→    }
   145→   145→
   146→   146→    const locationId = item.reservedLocationId ?? order.dispatchLocationId
   147→   147→    if (!locationId) {
   148→   148→      results.push({ orderItemId: item.id, outcome: 'failed', reason: 'No dispatch location set' })
   149→   149→      continue
   150→   150→    }
   151→   151→
   152→   152→    if (item.fulfillmentTypeSnapshot === 'stock_based') {
   153→   153→      // Check available stock at the dispatch location
   154→   154→      const pool = await db.inventoryPool.findUnique({
   155→   155→        where: {
   156→   156→          orgVariantId_locationId: {
   157→   157→            orgVariantId: item.orgVariantId,
   158→   158→            locationId,
   159→   159→          },
   160→   160→        },
   161→   161→      })
   162→   162→
   163→   163→      const available = pool ? pool.onHand - pool.reserved : 0
   164→   164→
   165→   165→      if (available >= item.quantity) {
   166→   166→        // Sufficient stock — reserve it
   167→   167→        const reserveResult = await reserveStockForOrder({
   168→   168→          orgVariantId: item.orgVariantId,
   169→   169→          locationId,
   170→   170→          organizationId: order.organizationId,
   171→   171→          companyId: order.companyId,
   172→   172→          employeeId: ctx.employee.id,
   173→   173→          quantity: item.quantity,
   174→   174→          orderId,
   175→   175→        })
   176→   176→
   177→   177→        if (reserveResult.success) {
   178→   178→          await db.orderItem.update({
   179→   179→            where: { id: item.id },
   180→   180→            data: { fulfillmentStatus: 'reserved', reservedLocationId: locationId },
   181→   181→          })
   182→   182→          results.push({ orderItemId: item.id, outcome: 'reserved' })
   183→   183→        } else {
   184→   184→          results.push({ orderItemId: item.id, outcome: 'failed', reason: reserveResult.error })
   185→   185→        }
   186→   186→      } else if (item.orgVariant.inventoryPolicy === 'continue') {
   187→   187→        // Insufficient stock but backordering allowed
   188→   188→        await db.orderItem.update({
   189→   189→          where: { id: item.id },
   190→   190→          data: { fulfillmentStatus: 'backordered', backorderedAt: new Date() },
   191→   191→        })
   192→   192→        await insertMetricEvent({
   193→   193→          companyId: order.companyId,
   194→   194→          entityType: 'product',
   195→   195→          entityId: item.orgVariantId,
   196→   196→          metricKey: 'order.backordered',
   197→   197→          numericValue: item.quantity,
   198→   198→          dimensions: { order_id: orderId },
   199→   199→        }).catch(() => {})
   200→   200→        hasBackordered = true
   201→   201→        results.push({
   202→   202→          orderItemId: item.id,
   203→   203→          outcome: 'backordered',
   204→   204→          reason: `Available: ${available}, required: ${item.quantity}`,
   205→   205→        })
   206→   206→      } else {
   207→   207→        // Insufficient stock + policy='deny' — fail
   208→   208→        results.push({
   209→   209→          orderItemId: item.id,
   210→   210→          outcome: 'failed',
   211→   211→          reason: `Insufficient stock (available: ${available}, required: ${item.quantity}) and inventory_policy='deny'`,
   212→   212→        })
   213→   213→      }
   214→   214→    } else if (item.fulfillmentTypeSnapshot === 'made_to_order') {
   215→   215→      // Made-to-order: check returned stock first, then trigger production
   216→   216→      const mtoResult = await checkAndFulfillMadeToOrderVariant(
   217→   217→        item.orgVariantId,
   218→   218→        item.quantity,
   219→   219→        order.companyId,
   220→   220→        locationId,
   221→   221→      )
   222→   222→
   223→   223→      if (mtoResult.source === 'existing_stock' && mtoResult.locationId) {
   224→   224→        // Returned stock available — reserve it
   225→   225→        const reserveResult = await reserveStockForOrder({
   226→   226→          orgVariantId: item.orgVariantId,
   227→   227→          locationId: mtoResult.locationId,
   228→   228→          organizationId: order.organizationId,
   229→   229→          companyId: order.companyId,
   230→   230→          employeeId: ctx.employee.id,
   231→   231→          quantity: item.quantity,
   232→   232→          orderId,
   233→   233→        })
   234→   234→
   235→   235→        if (reserveResult.success) {
   236→   236→          await db.orderItem.update({
   237→   237→            where: { id: item.id },
   238→   238→            data: {
   239→   239→              fulfillmentStatus: 'reserved',
   240→   240→              returnedStitchedUsed: true,
   241→   241→              reservedLocationId: mtoResult.locationId,
   242→   242→            },
   243→   243→          })
   244→   244→          await insertMetricEvent({
   245→   245→            companyId: order.companyId,
   246→   246→            entityType: 'product',
   247→   247→            entityId: item.orgVariantId,
   248→   248→            metricKey: 'order.made_to_order_from_returned_stock',
   249→   249→            numericValue: 1,
   250→   250→            dimensions: { order_id: orderId, stitching_cost_saved: Number(item.orgVariant.stitchingCharges) || 0 },
   251→   251→          }).catch(() => {})
   252→   252→          results.push({ orderItemId: item.id, outcome: 'reserved' })
   253→   253→        } else {
   254→   254→          results.push({ orderItemId: item.id, outcome: 'failed', reason: reserveResult.error })
   255→   255→        }
   256→   256→      } else if (mtoResult.source === 'fresh_production' && mtoResult.productionOrderId) {
   257→   257→        // Fresh production triggered — link the production order
   258→   258→        await db.orderItem.update({
   259→   259→          where: { id: item.id },
   260→   260→          data: {
   261→   261→            fulfillmentStatus: 'reserved',
   262→   262→            productionOrderId: mtoResult.productionOrderId,
   263→   263→          },
   264→   264→        })
   265→   265→        // Also link the order_item back to the production order
   266→   266→        await db.productionOrder.update({
   267→   267→          where: { id: mtoResult.productionOrderId },
   268→   268→          data: { orderItemId: item.id },
   269→   269→        })
   270→   270→        await insertMetricEvent({
   271→   271→          companyId: order.companyId,
   272→   272→          entityType: 'product',
   273→   273→          entityId: item.orgVariantId,
   274→   274→          metricKey: 'order.made_to_order_production_triggered',
   275→   275→          numericValue: 1,
   276→   276→          dimensions: { order_id: orderId, production_order_id: mtoResult.productionOrderId },
   277→   277→        }).catch(() => {})
   278→   278→        results.push({ orderItemId: item.id, outcome: 'reserved' })
   279→   279→      } else {
   280→   280→        // Error (no fabric source, insufficient fabric, etc.)
   281→   281→        results.push({
   282→   282→          orderItemId: item.id,
   283→   283→          outcome: 'failed',
   284→   284→          reason: mtoResult.error ?? 'Made-to-order fulfillment failed',
   285→   285→        })
   286→   286→      }
   287→   287→    }
   288→   288→  }
   289→   289→
   290→   290→  // Update order status based on results
   291→   291→  if (hasBackordered) {
   292→   292→    await db.order.update({
   293→   293→      where: { id: orderId },
   294→   294→      data: { status: 'partially_backordered' },
   295→   295→    })
   296→   296→  }
   297→   297→  // If no backordered items, order stays 'confirmed' (or whatever it was)
   298→   298→
   299→   299→  return { success: true, results }
   300→   300→}
   301→   301→

   305→   305→
   306→   306→export async function createManualOrder(
   307→   307→  input: CreateManualOrderInput,
   308→   308→): Promise<ActionResult<{
   309→   309→  orderId: string
   310→   310→  flowopsOrderNumber: string
   311→   311→  orderItems: Array<{ id: string; orgVariantId: string; quantity: number }>
   312→   312→  /** Auto-booking is now ASYNC — bookingAttempted=true means it's running in the background. */
   313→   313→  bookingAttempted: boolean
   314→   314→  bookingSucceeded?: boolean
   315→   315→  bookingError?: string
   316→   316→  bookingTrackingNumber?: string
   317→   317→}>> {
   318→   318→  try {
   319→   319→    const ctx = await getWorkspace()
   320→   320→    await requirePermission(ctx, PERMISSIONS.ORDERS_CREATE)
   321→   321→
   322→   322→    // 1. Validate
   323→   323→    const parsed = createManualOrderSchema.safeParse(input)
   324→   324→    if (!parsed.success) {
   325→   325→      return {
   326→   326→        success: false,
   327→   327→        error: parsed.error.issues[0]?.message ?? 'Invalid order data',
   328→   328→      }
   329→   329→    }
   330→   330→    const d = parsed.data
   331→   331→
   332→   332→    // 2. Resolve customer (existing customer_id OR inline new_customer).
   333→   333→    //    Also resolve optional saved address/phone selection and recipient_name.
   334→   334→    let customerId: string
   335→   335→    let customerName: string
   336→   336→    let usedCustomerAddressId: string | null = null
   337→   337→    let usedCustomerPhoneId: string | null = null
   338→   338→    let saveAddressForNextTime = d.save_address_for_next_time ?? false
   339→   339→    // Whether the delivery_address came from a saved customer_addresses row
   340→   340→    // (vs. typed as a one-off). Used to decide whether to bump lastUsedAt.
   341→   341→    let selectedSavedAddressId: string | null = null
   342→   342→
   343→   343→    if (d.customer_id) {
   344→   344→      // Existing customer path — verify they belong to this org.
   345→   345→      const existing = await db.customer.findFirst({
   346→   346→        where: { id: d.customer_id, organizationId: ctx.company.organizationId },
   347→   347→        select: { id: true, name: true },
   348→   348→      })
   349→   349→      if (!existing) return { success: false, error: 'Customer not found' }
   350→   350→      customerId = existing.id
   351→   351→      customerName = existing.name
   352→   352→
   353→   353→      // If a saved address was selected, verify it belongs to this customer.
   354→   354→      if (d.used_customer_address_id) {
   355→   355→        const addr = await db.customerAddress.findFirst({
   356→   356→          where: {
   357→   357→            id: d.used_customer_address_id,
   358→   358→            customerId: existing.id,
   359→   359→            organizationId: ctx.company.organizationId,
   360→   360→          },
   361→   361→          select: { id: true, address: true, city: true },
   362→   362→        })
   363→   363→        if (addr) {
   364→   364→          usedCustomerAddressId = addr.id
   365→   365→          selectedSavedAddressId = addr.id
   366→   366→          // The delivery_address snapshot is the order's own editable copy.
   367→   367→          // If the caller passed the same address text, we trust it; if they
   368→   368→          // passed different text (edited it), we still record the link to
   369→   369→          // the saved address for lastUsedAt tracking but keep the edited text
   370→   370→          // as the order's snapshot. Either way, delivery_address comes from
   371→   371→          // the input — we do NOT override it here.
   372→   372→        }
   373→   373→      }
   374→   374→
   375→   375→      // If a saved phone was selected, verify it belongs to this customer.
   376→   376→      if (d.used_customer_phone_id) {
   377→   377→        const phone = await db.customerPhone.findFirst({
   378→   378→          where: {
   379→   379→            id: d.used_customer_phone_id,
   380→   380→            customerId: existing.id,
   381→   381→            organizationId: ctx.company.organizationId,
   382→   382→          },
   383→   383→          select: { id: true },
   384→   384→        })
   385→   385→        if (phone) {
   386→   386→          usedCustomerPhoneId = phone.id
   387→   387→        }
   388→   388→      }
   389→   389→    } else if (d.new_customer) {
   390→   390→      // Inline new-customer path — create the customer + their phones +
   391→   391→      // addresses now via the Customer Management System. The first phone is
   392→   392→      // primary and the first address is default (validated by createCustomerSchema).
   393→   393→      const createResult = await createCustomer(d.new_customer)
   394→   394→      if (!createResult.success || !createResult.data) {
   395→   395→        return { success: false, error: createResult.error ?? 'Failed to create customer' }
   396→   396→      }
   397→   397→      customerId = createResult.data.customerId
   398→   398→      customerName = d.new_customer.name.trim()
   399→   399→      // For a brand-new customer, the first address they provided IS the
   400→   400→      // saved default address. Mark it as the selected address so we bump
   401→   401→      // its lastUsedAt after order creation.
   402→   402→      const newAddr = await db.customerAddress.findFirst({
   403→   403→        where: { customerId, isDefault: true },
   404→   404→        select: { id: true },
   405→   405→      })
   406→   406→      if (newAddr) {
   407→   407→        usedCustomerAddressId = newAddr.id
   408→   408→        selectedSavedAddressId = newAddr.id
   409→   409→      }
   410→   410→      // Same for the primary phone.
   411→   411→      const newPhone = await db.customerPhone.findFirst({
   412→   412→        where: { customerId, isPrimary: true },
   413→   413→        select: { id: true },
   414→   414→      })
   415→   415→      if (newPhone) {
   416→   416→        usedCustomerPhoneId = newPhone.id
   417→   417→      }
   418→   418→      // For a brand-new customer, we already saved the address as part of
   419→   419→      // createCustomer — don't double-save it.
   420→   420→      saveAddressForNextTime = false
   421→   421→    } else {
   422→   422→      return { success: false, error: 'Either customer_id or new_customer is required' }
   423→   423→    }
   424→   424→
   425→   425→    // 3. recipient_name defaults to the customer's name if not explicitly overridden.
   426→   426→    const recipientName = (d.recipient_name && d.recipient_name.trim()) || customerName
   427→   427→
   428→   428→    // 3. Fetch variants + their pricing for this company + fulfillment_type snapshot
   429→   429→    const variantIds = d.items.map((i) => i.org_variant_id)
   430→   430→    const variants = await db.orgProductVariant.findMany({
   431→   431→      where: { id: { in: variantIds }, organizationId: ctx.company.organizationId },
   432→   432→      include: {
   433→   433→        product: { select: { title: true } },
   434→   434→        companyPricing: { where: { companyId: ctx.company.id } },
   435→   435→      },
   436→   436→    })
   437→   437→
   438→   438→    if (variants.length !== variantIds.length) {
   439→   439→      return { success: false, error: 'One or more variants not found in this organization' }
   440→   440→    }
   441→   441→
   442→   442→    // 4. Compute subtotal + build order items data
   443→   443→    let subtotal = 0
   444→   444→    const orderItemsData: Array<{
   445→   445→      orgVariantId: string
   446→   446→      quantity: number
   447→   447→      unitPrice: number
   448→   448→      lineTotal: number
   449→   449→      fulfillmentTypeSnapshot: string
   450→   450→    }> = []
   451→   451→
   452→   452→    // Build the order_detail string as we iterate — universal courier
   453→   453→    // reference field (migration 015). Format:
   454→   454→    //   "Product Title (SKU-001, Size: M, Color: Blue) ×2, ..."
   455→   455→    // Auto-generated but editable from the order-create form. If the caller
   456→   456→    // supplies an explicit order_detail, we honour it verbatim.
   457→   457→    const orderDetailParts: string[] = []
   458→   458→
   459→   459→    for (const item of d.items) {
   460→   460→      const variant = variants.find((v) => v.id === item.org_variant_id)
   461→   461→      if (!variant) continue
   462→   462→
   463→   463→      // Use provided unit_price OR fall back to company pricing OR variant cost
   464→   464→      const unitPrice =
   465→   465→        item.unit_price ??
   466→   466→        (variant.companyPricing[0]?.salePrice
   467→   467→          ? Number(variant.companyPricing[0].salePrice)
   468→   468→          : Number(variant.costPrice))
   469→   469→
   470→   470→      const lineTotal = item.quantity * unitPrice
   471→   471→      subtotal += lineTotal
   472→   472→
   473→   473→      orderItemsData.push({
   474→   474→        orgVariantId: item.org_variant_id,
   475→   475→        quantity: item.quantity,
   476→   476→        unitPrice,
   477→   477→        lineTotal,
   478→   478→        fulfillmentTypeSnapshot: variant.fulfillmentType,
   479→   479→      })
   480→   480→
   481→   481→      // Append to orderDetail: "Product Title (SKU-001, Size: M) ×2"
   482→   482→      const attrParts: string[] = []
   483→   483→      try {
   484→   484→        const attrs = JSON.parse(variant.attributeValues || '{}') as Record<string, string>
   485→   485→        for (const [k, v] of Object.entries(attrs)) {
   486→   486→          if (v) attrParts.push(`${k}: ${v}`)
   487→   487→        }
   488→   488→      } catch {
   489→   489→        // ignore parse errors
   490→   490→      }
   491→   491→      const inner = [variant.sku, ...attrParts].filter(Boolean).join(', ')
   492→   492→      orderDetailParts.push(
   493→   493→        `${variant.product.title}${inner ? ` (${inner})` : ''} ×${item.quantity}`,
   494→   494→      )
   495→   495→    }
   496→   496→
   497→   497→    // 5. Compute totals
   498→   498→    const discountAmount = d.discount_amount ?? 0
   499→   499→    const courierCharges = d.courier_charges ?? 0
   500→   500→    const estimatedDeliveryCharge = d.estimated_delivery_charge ?? 0
   501→   501→    const taxAmount = d.tax_amount ?? 0
   502→   502→    // Total = subtotal + courier charges + delivery charge + tax - discount
   503→   503→    // (delivery charge and tax are ADDITIVE — not absorbed into subtotal)
   504→   504→    const totalOrderValue = subtotal + courierCharges + estimatedDeliveryCharge + taxAmount - discountAmount
   505→   505→
   506→   506→    // 6. Determine payment status + source
   507→   507→    let paymentStatus: string = 'cod_pending'
   508→   508→    let paymentSource: string = 'cod_native'
   509→   509→    let advanceAmount: number | null = null
   510→   510→    let advancePaymentMethod: string | null = null
   511→   511→    let advancePaymentReference: string | null = null
   512→   512→    let advancePaidAt: Date | null = null
   513→   513→
   514→   514→    if (d.payment_type === 'partial_advance') {
   515→   515→      paymentStatus = 'advance_paid'
   516→   516→      paymentSource = 'manual_conversion'
   517→   517→      advanceAmount = d.advance_amount ?? 0
   518→   518→      advancePaymentMethod = d.advance_payment_method || null
   519→   519→      advancePaymentReference = d.advance_payment_reference || null
   520→   520→      advancePaidAt = new Date()
   521→   521→    } else if (d.payment_type === 'fully_prepaid') {
   522→   522→      paymentStatus = 'fully_prepaid'
   523→   523→      paymentSource = 'manual_conversion'
   524→   524→      advanceAmount = totalOrderValue
   525→   525→      advancePaymentMethod = d.advance_payment_method || null
   526→   526→      advancePaymentReference = d.advance_payment_reference || null
   527→   527→      advancePaidAt = new Date()
   528→   528→    }
   529→   529→
   530→   530→    // remainingCodAmount is NOT a GENERATED column in the DB — the application
   531→   531→    // MUST compute it as totalOrderValue - (advanceAmount ?? 0).
   532→   532→    const remainingCodAmount = totalOrderValue - (advanceAmount ?? 0)
   533→   533→
   534→   534→    // 7. Determine initial order status
   535→   535→    let orderStatus: string = 'pending'
   536→   536→    let confirmedAt: Date | null = null
   537→   537→    let skippedConfirmation = false
   538→   538→
   539→   539→    if (paymentStatus === 'advance_paid' || paymentStatus === 'fully_prepaid') {
   540→   540→      // Payment itself is the confirmation signal — bypasses require_order_confirmation
   541→   541→      orderStatus = 'confirmed'
   542→   542→      confirmedAt = new Date()
   543→   543→    } else {
   544→   544→      // COD order — check company_order_settings
   545→   545→      const settings = await db.companyOrderSetting.findUnique({
   546→   546→        where: { companyId: ctx.company.id },
   547→   547→      })
   548→   548→      if (!settings?.requireOrderConfirmation) {
   549→   549→        orderStatus = 'confirmed'
   550→   550→        confirmedAt = new Date()
   551→   551→        skippedConfirmation = true
   552→   552→      }
   553→   553→    }
   554→   554→
   555→   555→    // 8. Generate order number
   556→   556→    const flowopsOrderNumber = await generateOrderNumber(ctx.company.id)
   557→   557→
   558→   558→    // 8a. Universal courier reference fields (migration 015).
   559→   559→    // orderRefNumber: defaults to flowopsOrderNumber if the caller left it
   560→   560→    // blank — every courier has a "reference" field so this is a core OMS
   561→   561→    // field, mapped to the courier's own ref at booking time.
   562→   562→    const orderRefNumber =
   563→   563→      (d.order_ref_number && d.order_ref_number.trim()) || flowopsOrderNumber
   564→   564→    // orderDetail: caller-provided takes precedence, otherwise use the
   565→   565→    // auto-generated string from the cart items (product title + SKU +
   566→   566→    // variant attributes + quantity).
   567→   567→    const orderDetail =
   568→   568→      (d.order_detail && d.order_detail.trim()) || orderDetailParts.join(', ')
   569→   569→
   570→   570→    // 9. Create the order
   571→   571→    const order = await db.order.create({
   572→   572→      data: {
   573→   573→        organizationId: ctx.company.organizationId,
   574→   574→        companyId: ctx.company.id,
   575→   575→        flowopsOrderNumber,
   576→   576→        orderSource: 'manual',
   577→   577→        customerId,
   578→   578→        // Customer Management System integration (migration 002):
   579→   579→        //   - recipientName: the name to deliver to (defaults to customer.name)
   580→   580→        //   - usedCustomerAddressId / usedCustomerPhoneId: which saved
   581→   581→        //     customer_addresses / customer_phones rows were used for this
   582→   582→        //     order, for lastUsedAt tracking and reporting. Nullable when a
   583→   583→        //     one-off address/phone was typed without saving.
   584→   584→        recipientName,
   585→   585→        usedCustomerAddressId,
   586→   586→        usedCustomerPhoneId,
   587→   587→        status: orderStatus,
   588→   588→        paymentType: d.payment_type,
   589→   589→        paymentStatus,
   590→   590→        paymentSource,
   591→   591→        subtotal,
   592→   592→        discountAmount: discountAmount || null,
   593→   593→        discountReason: d.discount_reason || null,
   594→   594→        courierCharges: courierCharges || null,
   595→   595→        estimatedDeliveryCharge: estimatedDeliveryCharge || null,
   596→   596→        taxAmount: taxAmount || null,
   597→   597→        taxLabel: d.tax_label?.trim() || null,
   598→   598→        totalOrderValue,
   599→   599→        advanceAmount,
   600→   600→        advancePaymentMethod,
   601→   601→        advancePaymentReference,
   602→   602→        advancePaidAt,
   603→   603→        remainingCodAmount,
   604→   604→        deliveryAddress: d.delivery_address,
   605→   605→        deliveryCity: d.delivery_city,
   606→   606→        courierName: d.courier_name || null,
   607→   607→        courierCompanyIntegrationId: d.courier_company_integration_id || null,
   608→   608→        recommendedCourierCompanyIntegrationId: d.courier_company_integration_id || null,
   609→   609→        dispatchLocationId: d.dispatch_location_id,
   610→   610→        notesForCourier: d.notes_for_courier || null,
   611→   611→        // Per-order pickup address override (null = use integration default at booking time)
   612→   612→        pickupAddressId: d.pickup_address_id || null,
   613→   613→        // Universal courier reference fields (migration 015)
   614→   614→        orderRefNumber,
   615→   615→        orderDetail,
   616→   616→        skippedConfirmation,
   617→   617→        confirmedAt,
   618→   618→        createdBy: ctx.employee.id,
   619→   619→      },
   620→   620→    })
   621→   621→
   622→   622→    // 10. Create order items (fulfillment_status = 'reserved' as PLACEHOLDER)
   623→   623→    const createdItems: Array<{ id: string; orgVariantId: string; quantity: number }> = []
   624→   624→    for (const itemData of orderItemsData) {
   625→   625→      const item = await db.orderItem.create({
   626→   626→        data: {
   627→   627→          orderId: order.id,
   628→   628→          orgVariantId: itemData.orgVariantId,
   629→   629→          organizationId: ctx.company.organizationId,
   630→   630→          quantity: itemData.quantity,
   631→   631→          unitPrice: itemData.unitPrice,
   632→   632→          lineTotal: itemData.lineTotal,
   633→   633→          fulfillmentStatus: 'reserved', // PLACEHOLDER — Step 3 will validate/adjust
   634→   634→          fulfillmentTypeSnapshot: itemData.fulfillmentTypeSnapshot,
   635→   635→          reservedLocationId: d.dispatch_location_id,
   636→   636→        },
   637→   637→      })
   638→   638→      createdItems.push({
   639→   639→        id: item.id,
   640→   640→        orgVariantId: item.orgVariantId,
   641→   641→        quantity: item.quantity,
   642→   642→      })
   643→   643→    }
   644→   644→
   645→   645→    // 11. Audit log

   648→   648→      entityType: 'order',
   649→   649→      entityId: order.id,
   650→   650→      companyId: ctx.company.id,
   651→   651→      organizationId: ctx.company.organizationId,
   652→   652→      userId: ctx.user.id,
   653→   653→      employeeId: ctx.employee.id,
   654→   654→      newValues: {
   655→   655→        flowopsOrderNumber,
   656→   656→        status: orderStatus,
   657→   657→        paymentType: d.payment_type,
   658→   658→        paymentStatus,
   659→   659→        itemCount: orderItemsData.length,
   660→   660→        totalOrderValue,
   661→   661→        customerId,
   662→   662→      },
   663→   663→    })
   664→   664→
   665→   665→    // 12. Customer Management System integration:
   666→   666→    //   a. If a saved address was selected, bump its lastUsedAt to NOW() so
   667→   667→    //      the UI can sort saved addresses by recency.
   668→   668→    //   b. If a brand-new one-off address was typed AND the caller set
   669→   669→    //      save_address_for_next_time=true, persist it as a new
   670→   670→    //      customer_addresses row (and link it via usedCustomerAddressId
   671→   671→    //      for future tracking). This is the "Save this address for next
   672→   672→    //      time" UX from Step 3's frontend.
   673→   673→    //   c. Recompute cached customer stats (total_orders_count, etc.) and
   674→   674→    //      auto-flag if RTO threshold crossed.
   675→   675→    if (selectedSavedAddressId) {
   676→   676→      await markAddressAsUsed(selectedSavedAddressId)
   677→   677→    } else if (saveAddressForNextTime && d.delivery_address && d.delivery_city) {
   678→   678→      // One-off address typed + user opted to save it for next time.
   679→   679→      // Persist as a non-default customer_addresses row (default stays as-is

   681→   681→      const savedAddr = await db.customerAddress.create({
   682→   682→        data: {
   683→   683→          customerId,
   684→   684→          organizationId: ctx.company.organizationId,

   686→   686→          address: d.delivery_address,
   687→   687→          city: d.delivery_city,
   688→   688→          isDefault: false,
   689→   689→          lastUsedAt: new Date(),
   690→   690→        },
   691→   691→      })
   692→   692→      // Link the order to the newly-saved address for future tracking.
   693→   693→      await db.order.update({
   694→   694→        where: { id: order.id },
   695→   695→        data: { usedCustomerAddressId: savedAddr.id },
   696→   696→      })
   697→   697→    }
   698→   698→
   699→   699→    await updateCustomerStats(customerId)
   700→   700→
   701→   701→    // 13. If the order auto-confirmed (payment-driven or company setting),
   702→   702→    // run the stock reservation logic immediately.
   703→   703→    if (orderStatus === 'confirmed') {
   704→   704→      await reserveOrderStock(order.id, ctx)
   705→   705→    }
   706→   706→
   707→   707→    // 14. AUTO-BOOKING (Phase 3): if the company's courierBookingMode is
   708→   708→    // 'automatic' AND the order is confirmed, automatically book the courier.
   709→   709→    //
   710→   710→    // ASYNCHRONOUS: PostEx's API can take 50-100 seconds to respond. Running
   711→   711→    // the booking synchronously would block the order creation response for
   712→   712→    // that long, causing a 502 Bad Gateway from the ALB. Instead, we fire the
   713→   713→    // booking in the background and return immediately. The order is created
   714→   714→    // with courierBookingStatus='not_booked', and the background task updates
   715→   715→    // it to 'booked' or 'failed' when PostEx responds.
   716→   716→    //
   717→   717→    // The frontend shows "Order created. Booking in progress…" and can poll
   718→   718→    // the order detail to see when the tracking number appears.
   719→   719→    let bookingAttempted = false
   720→   720→    let bookingSucceeded = false
   721→   721→    let bookingError: string | undefined
   722→   722→    let bookingTrackingNumber: string | undefined
   723→   723→
   724→   724→    if (orderStatus === 'confirmed') {
   725→   725→      // Check if auto-booking should fire (quick synchronous check — no API call)
   726→   726→      try {
   727→   727→        const settings = await db.companyOrderSetting.findUnique({
   728→   728→          where: { companyId: ctx.company.id },
   729→   729→          select: { courierBookingMode: true, defaultCourierCompanyIntegrationId: true },
   730→   730→        })
   731→   731→        const shouldAutoBook =
   732→   732→          settings?.courierBookingMode === 'automatic' &&
   733→   733→          (d.courier_company_integration_id || settings?.defaultCourierCompanyIntegrationId)
   734→   734→
   735→   735→        if (shouldAutoBook) {
   736→   736→          bookingAttempted = true // we're going to attempt it in the background
   737→   737→          // Fire-and-forget: the booking runs in the background. The order
   738→   738→          // creation returns immediately with bookingSucceeded=undefined
   739→   739→          // (meaning "in progress"). When the background task completes, it
   740→   740→          // updates the order's courierBookingStatus + trackingNumber.
   741→   741→          ;(async () => {
   742→   742→            try {
   743→   743→              const { maybeAutoBookOrder } = await import('./booking.actions')
   744→   744→              const bookResult = await maybeAutoBookOrder(order.id, 'manual', orderStatus)
   745→   745→              if (bookResult.success) {
   746→   746→                console.log(`[createManualOrder] Background auto-booking succeeded for ${flowopsOrderNumber}: tracking=${bookResult.data?.trackingNumber}`)
   747→   747→              } else {
   748→   748→                console.warn(`[createManualOrder] Background auto-booking failed for ${flowopsOrderNumber}: ${bookResult.error}`)
   749→   749→              }
   750→   750→            } catch (err) {
   751→   751→              console.error(`[createManualOrder] Background auto-booking threw for ${flowopsOrderNumber}:`, err)
   752→   752→            }
   753→   753→          })()
   754→   754→        }
   755→   755→      } catch {
   756→   756→        // Settings check failed — non-fatal, just skip auto-booking
   757→   757→      }
   758→   758→    }
   759→   759→
   760→   760→    await insertMetricEvent({
   761→   761→      companyId: ctx.company.id,
   762→   762→      entityType: 'order',
   763→   763→      entityId: order.id,
   764→   764→      metricKey: 'order.created',
   765→   765→      numericValue: Number(totalOrderValue),
   766→   766→      dimensions: { order_source: 'manual', payment_type: d.payment_type, company_id: ctx.company.id },
   767→   767→    }).catch(() => {})
   768→   768→
   769→   769→    return {
   770→   770→      success: true,
   771→   771→      data: {
   772→   772→        orderId: order.id,
   773→   773→        flowopsOrderNumber,
   774→   774→        orderItems: createdItems,
   775→   775→        // Phase 5: booking result — distinct from order creation success.
   776→   776→        // The frontend uses bookingAttempted + bookingSucceeded to show TWO
   777→   777→        // separate messages: "Order created" + (if applicable) "Booking failed".
   778→   778→        bookingAttempted,
   779→   779→        bookingSucceeded,
   780→   780→        bookingError,
   781→   781→        bookingTrackingNumber,
   782→   782→      },
   783→   783→    }
   784→   784→  } catch (err) {
   785→   785→    return {
   786→   786→      success: false,
   787→   787→      error: err instanceof Error ? err.message : 'Failed to create order',
   788→   788→    }
   789→   789→  }
   790→   790→}
