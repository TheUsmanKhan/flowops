   129→const PKR = new Intl.NumberFormat('en-PK', { maximumFractionDigits: 0 })
   130→
   131→function formatPKR(n: number): string {
   132→  return `Rs. ${PKR.format(n)}`
   133→}
   134→
   135→function getErrorMessage(err: unknown): string {
   136→  if (err instanceof FetchError) return err.message
   137→  if (err instanceof Error) return err.message
   138→  return 'Something went wrong. Please try again.'
   139→}
   140→
   141→function parseQty(v: string): number {
   142→  if (v === '') return 0
   143→  const n = Number(v)
   144→  return Number.isFinite(n) && n >= 0 ? Math.floor(n) : 0
   145→}
   146→
   147→function parsePrice(v: string): number {
   148→  if (v === '') return 0
   149→  const n = Number(v)
   150→  return Number.isFinite(n) && n >= 0 ? n : 0
   151→}

   173→
   174→export function OrderCreateView() {
   175→  const navigate = useAppStore((s) => s.navigate)
   176→  const can = useCan()
   177→  const queryClient = useQueryClient()
   178→  const canCreate = can(PERMISSIONS.ORDERS_CREATE)
   179→
   180→  const [step, setStep] = useState(1)
   181→
   182→  // ── Form state ────────────────────────────────────────────────────────────
   183→  // Customer
   184→  const [customerMode, setCustomerMode] = useState<'existing' | 'new'>('existing')
   185→  const [phoneSearch, setPhoneSearch] = useState('')
   186→  const [selectedCustomer, setSelectedCustomer] = useState<CustomerRow | null>(null)
   187→  const [newCustomer, setNewCustomer] = useState({
   188→    name: '',
   189→    phone: '',
   190→    alternate_phone: '',
   191→    email: '',
   192→    address: '',
   193→    city: '',
   194→  })
   195→
   196→  // Items
   197→  const [cart, setCart] = useState<CartItem[]>([])
   198→  const [variantSearch, setVariantSearch] = useState('')
   199→
   200→  // Payment
   201→  const [paymentType, setPaymentType] = useState<PaymentType>('full_cod')
   202→  const [advanceAmount, setAdvanceAmount] = useState('')
   203→  const [advancePaymentMethod, setAdvancePaymentMethod] = useState('')
   204→  const [advancePaymentReference, setAdvancePaymentReference] = useState('')
   205→  const [advancePaymentScreenshotUrl, setAdvancePaymentScreenshotUrl] = useState('')
   206→
   207→  // Delivery
   208→  const [deliveryAddress, setDeliveryAddress] = useState('')
   209→  const [deliveryCity, setDeliveryCity] = useState('')
   210→  const [courierName, setCourierName] = useState('')
   211→  const [dispatchLocationId, setDispatchLocationId] = useState('')
   212→  const [notesForCourier, setNotesForCourier] = useState('')
   213→  const [discountAmount, setDiscountAmount] = useState('')
   214→  const [discountReason, setDiscountReason] = useState('')
   215→
   216→  // ── Data queries ──────────────────────────────────────────────────────────
   217→  const locationsQuery = useQuery<LocationsResponse>({
   218→    queryKey: ['locations'],
   219→    queryFn: () => api.get<LocationsResponse>('/api/inventory-locations'),
   220→    staleTime: 30_000,
   221→  })
   222→
   223→  const productsQuery = useQuery<ProductsResponse>({
   224→    queryKey: ['products', 'for-order-create'],
   225→    queryFn: () => api.get<ProductsResponse>('/api/products?pageSize=100'),
   226→    staleTime: 60_000,
   227→  })
   228→
   229→  // Customer phone search — debounced via staleTime + query key
   230→  const trimmedPhone = phoneSearch.trim()
   231→  const customersQuery = useQuery<CustomersSearchResponse>({
   232→    queryKey: ['customers', 'search', trimmedPhone],
   233→    queryFn: () =>
   234→      api.get<CustomersSearchResponse>(
   235→        `/api/customers?search=${encodeURIComponent(trimmedPhone)}&limit=10`,
   236→      ),
   237→    enabled: trimmedPhone.length >= 4,
   238→    staleTime: 10_000,
   239→  })
   240→
   241→  // ── Variant options (flatten products → variants) ─────────────────────────
   242→  const variantOptions: VariantOption[] = useMemo(() => {
   243→    const list: VariantOption[] = []
   244→    for (const p of productsQuery.data?.products ?? []) {
   245→      for (const v of p.variants) {
   246→        list.push({
   247→          variantId: v.id,
   248→          sku: v.sku,
   249→          productTitle: p.title,
   250→          costPrice: v.costPrice,
   251→          salePrice: v.salePrice,
   252→          fulfillmentType: v.fulfillmentType,
   253→        })
   254→      }
   255→    }
   256→    return list
   257→  }, [productsQuery.data])
   258→
   259→  const variantSearchResults = useMemo(() => {
   260→    if (!variantSearch.trim()) return []
   261→    const q = variantSearch.toLowerCase()
   262→    return variantOptions
   263→      .filter(
   264→        (v) => v.sku.toLowerCase().includes(q) || v.productTitle.toLowerCase().includes(q),
   265→      )
   266→      .slice(0, 8)
   267→  }, [variantOptions, variantSearch])
   268→
   269→  // ── Computed totals ───────────────────────────────────────────────────────
   270→  const subtotal = useMemo(
   271→    () => cart.reduce((s, i) => s + i.unitPrice * i.quantity, 0),
   272→    [cart],
   273→  )
   274→  const discount = parsePrice(discountAmount)
   275→  const total = Math.max(0, subtotal - discount)
   276→
   277→  // ── Auto-fill delivery from customer ──────────────────────────────────────
   278→  useEffect(() => {
   279→    if (customerMode === 'existing' && selectedCustomer) {
   280→      // No addresses returned by customer search — let user type.
   281→    }
   282→  }, [customerMode, selectedCustomer])
   283→
   284→  // When switching to "new" customer, pre-fill delivery fields from their input.
   285→  useEffect(() => {
   286→    if (customerMode === 'new') {
   287→      if (newCustomer.address && !deliveryAddress) {
   288→        setDeliveryAddress(newCustomer.address)
   289→      }
   290→      if (newCustomer.city && !deliveryCity) {
   291→        setDeliveryCity(newCustomer.city)
   292→      }
   293→      if (newCustomer.phone && !courierName) {
   294→        // leave courier blank — not derivable
   295→      }
   296→    }
   297→  }, [customerMode, newCustomer.address, newCustomer.city, newCustomer.phone, courierName, deliveryAddress, deliveryCity])
   298→
   299→  // ── Auto-select default dispatch location ─────────────────────────────────
   300→  useEffect(() => {
   301→    if (!dispatchLocationId && locationsQuery.data?.locations.length) {
   302→      const def = locationsQuery.data.locations.find((l) => l.isDefault)
   303→      setDispatchLocationId(def?.id ?? locationsQuery.data.locations[0].id)
   304→    }
   305→  }, [dispatchLocationId, locationsQuery.data])
   306→
   307→  // ── Item manipulation ─────────────────────────────────────────────────────
   308→  const addVariant = (v: VariantOption) => {
   309→    setCart((prev) => {
   310→      if (prev.some((i) => i.variantId === v.variantId)) {
   311→        toast.info('That variant is already in the cart.')
   312→        return prev
   313→      }
   314→      const unitPrice = v.salePrice ?? v.costPrice
   315→      return [
   316→        ...prev,
   317→        {
   318→          variantId: v.variantId,
   319→          sku: v.sku,
   320→          productTitle: v.productTitle,
   321→          unitPrice,
   322→          quantity: 1,
   323→          fulfillmentType: v.fulfillmentType,
   324→        },
   325→      ]
   326→    })
   327→    setVariantSearch('')
   328→  }
   329→
   330→  const removeItem = (variantId: string) => {
   331→    setCart((prev) => prev.filter((i) => i.variantId !== variantId))
   332→  }
   333→
   334→  const updateItem = (variantId: string, patch: Partial<CartItem>) => {
   335→    setCart((prev) =>
   336→      prev.map((i) => (i.variantId === variantId ? { ...i, ...patch } : i)),
   337→    )
   338→  }
   339→
   340→  // ── Step validation ───────────────────────────────────────────────────────
   341→  const customerStepValid =
   342→    customerMode === 'existing'
   343→      ? !!selectedCustomer
   344→      : newCustomer.name.trim().length >= 2 &&
   345→        newCustomer.phone.trim().length >= 7 &&
   346→        newCustomer.address.trim().length >= 2 &&
   347→        newCustomer.city.trim().length >= 2
   348→
   349→  const itemsStepValid = cart.length > 0 && cart.every((i) => i.quantity > 0 && i.unitPrice >= 0)
   350→
   351→  const paymentStepValid =
   352→    paymentType === 'full_cod' ||
   353→    (paymentType === 'partial_advance' && parsePrice(advanceAmount) > 0) ||
   354→    paymentType === 'fully_prepaid'
   355→
   356→  const deliveryStepValid =
   357→    deliveryAddress.trim().length >= 2 &&
   358→    deliveryCity.trim().length >= 2 &&
   359→    !!dispatchLocationId
   360→
   361→  const canGoNext = (): boolean => {
   362→    if (step === 1) return customerStepValid
   363→    if (step === 2) return itemsStepValid
   364→    if (step === 3) return paymentStepValid
   365→    if (step === 4) return deliveryStepValid
   366→    return true
   367→  }
   368→
   369→  // ── Mutation: create order ────────────────────────────────────────────────
   370→  const createMutation = useMutation({
   371→    mutationFn: async (payload: unknown) =>
   372→      api.post<CreateOrderResponse>('/api/orders', payload),
   373→    onSuccess: (data) => {
   374→      toast.success(`Order ${data.flowopsOrderNumber} created successfully.`)
   375→      void queryClient.invalidateQueries({ queryKey: ['orders'] })
   376→      navigate({ name: 'order-detail', id: data.orderId })
   377→    },
   378→    onError: (err) => toast.error(getErrorMessage(err)),
   379→  })
   380→
   381→  // ── Permission gate (after all hooks) ─────────────────────────────────────
   382→  if (!canCreate) {
   383→    return (
   384→      <div className="space-y-6">
   385→        <PageHeader
   386→          title="Create Order"
   387→          description="Manually create a customer order"
   388→          actions={
   389→            <Button variant="outline" size="sm" onClick={() => navigate({ name: 'orders' })}>
   390→              <ArrowLeft className="h-4 w-4" /> Back to Orders
   391→            </Button>
   392→          }
   393→        />
   394→        <Card>
   395→          <CardContent className="p-10 text-center">
   396→            <AlertCircle className="h-10 w-10 mx-auto text-muted-foreground/40 mb-3" />
   397→            <p className="text-sm text-muted-foreground">
   398→              You don&apos;t have permission to create orders.
   399→            </p>
   400→          </CardContent>
   401→        </Card>
   402→      </div>
   403→    )
   404→  }
   405→
   406→  // ── Build payload + submit ────────────────────────────────────────────────
   407→  const handleSubmit = () => {
   408→    if (!customerStepValid || !itemsStepValid || !paymentStepValid || !deliveryStepValid) {
   409→      toast.error('Please complete all required fields before creating the order.')
   410→      return
   411→    }
   412→
   413→    const payload: Record<string, unknown> = {
   414→      items: cart.map((i) => ({
   415→        org_variant_id: i.variantId,
   416→        quantity: i.quantity,
   417→        unit_price: i.unitPrice,
   418→      })),
   419→      payment_type: paymentType,
   420→      delivery_address: deliveryAddress.trim(),
   421→      delivery_city: deliveryCity.trim(),
   422→      courier_name: courierName.trim() || undefined,
   423→      dispatch_location_id: dispatchLocationId,
   424→      notes_for_courier: notesForCourier.trim() || undefined,
   425→      discount_amount: discount > 0 ? discount : undefined,
   426→      discount_reason: discountReason.trim() || undefined,
   427→    }
   428→
   429→    if (paymentType === 'partial_advance' || paymentType === 'fully_prepaid') {
   430→      payload.advance_amount =
   431→        paymentType === 'fully_prepaid' ? total : parsePrice(advanceAmount)
   432→      payload.advance_payment_method = advancePaymentMethod || undefined
   433→      payload.advance_payment_reference = advancePaymentReference.trim() || undefined
   434→      payload.advance_payment_screenshot_url = advancePaymentScreenshotUrl.trim() || undefined
   435→    }
   436→
   437→    if (customerMode === 'existing' && selectedCustomer) {
   438→      payload.customer_id = selectedCustomer.id
   439→    } else {
   440→      payload.customer = {
   441→        name: newCustomer.name.trim(),
   442→        phone: newCustomer.phone.trim(),
   443→        alternate_phone: newCustomer.alternate_phone.trim() || undefined,
   444→        email: newCustomer.email.trim() || undefined,
   445→        addresses: [
   446→          {
   447→            label: 'Home',
   448→            address: newCustomer.address.trim(),
   449→            city: newCustomer.city.trim(),
   450→            is_default: true,
   451→          },
   452→        ],
   453→      }
   454→    }
   455→
   456→    createMutation.mutate(payload)
   457→  }
   458→
   459→  return (
   460→    <div className="space-y-6">
   461→      <PageHeader
   462→        title="Create Order"
   463→        description="Manually create a customer order"
   464→        actions={
   465→          <Button variant="outline" size="sm" onClick={() => navigate({ name: 'orders' })}>
   466→            <ArrowLeft className="h-4 w-4" /> Back to Orders
   467→          </Button>
   468→        }
   469→      />
   470→
   471→      {/* ── Stepper ─────────────────────────────────────────────────────────── */}
   472→      <Stepper currentStep={step} />
   473→
   474→      {/* ── Step content ────────────────────────────────────────────────────── */}
   475→      {step === 1 && (
   476→        <CustomerStep
   477→          customerMode={customerMode}
   478→          setCustomerMode={setCustomerMode}
   479→          phoneSearch={phoneSearch}
   480→          setPhoneSearch={setPhoneSearch}
   481→          selectedCustomer={selectedCustomer}
   482→          setSelectedCustomer={setSelectedCustomer}
   483→          newCustomer={newCustomer}
   484→          setNewCustomer={setNewCustomer}
   485→          customers={customersQuery.data?.customers ?? []}
   486→          isSearching={customersQuery.isFetching}
   487→        />
   488→      )}
   489→
   490→      {step === 2 && (
   491→        <ItemsStep
   492→          cart={cart}
   493→          variantSearch={variantSearch}
   494→          setVariantSearch={setVariantSearch}
   495→          variantSearchResults={variantSearchResults}
   496→          addVariant={addVariant}
   497→          removeItem={removeItem}
   498→          updateItem={updateItem}
   499→          isLoadingProducts={productsQuery.isLoading}
   500→          subtotal={subtotal}
   501→        />
   502→      )}
   503→
   504→      {step === 3 && (
   505→        <PaymentStep
   506→          paymentType={paymentType}
   507→          setPaymentType={setPaymentType}
   508→          advanceAmount={advanceAmount}
   509→          setAdvanceAmount={setAdvanceAmount}
   510→          advancePaymentMethod={advancePaymentMethod}
   511→          setAdvancePaymentMethod={setAdvancePaymentMethod}
   512→          advancePaymentReference={advancePaymentReference}
   513→          setAdvancePaymentReference={setAdvancePaymentReference}
   514→          advancePaymentScreenshotUrl={advancePaymentScreenshotUrl}
   515→          setAdvancePaymentScreenshotUrl={setAdvancePaymentScreenshotUrl}
   516→          subtotal={subtotal}
   517→          total={total}
   518→          discount={discount}
   519→        />
   520→      )}
   521→
   522→      {step === 4 && (
   523→        <DeliveryStep
   524→          deliveryAddress={deliveryAddress}
   525→          setDeliveryAddress={setDeliveryAddress}
   526→          deliveryCity={deliveryCity}
   527→          setDeliveryCity={setDeliveryCity}
   528→          courierName={courierName}
   529→          setCourierName={setCourierName}
   530→          dispatchLocationId={dispatchLocationId}
   531→          setDispatchLocationId={setDispatchLocationId}
   532→          notesForCourier={notesForCourier}
   533→          setNotesForCourier={setNotesForCourier}
   534→          discountAmount={discountAmount}
   535→          setDiscountAmount={setDiscountAmount}
   536→          discountReason={discountReason}
   537→          setDiscountReason={setDiscountReason}
   538→          locations={locationsQuery.data?.locations ?? []}
   539→          isLoadingLocations={locationsQuery.isLoading}
   540→          subtotal={subtotal}
   541→          discount={discount}
   542→          total={total}
   543→        />
   544→      )}
   545→
   546→      {step === 5 && (
   547→        <ReviewStep
   548→          customerMode={customerMode}
   549→          selectedCustomer={selectedCustomer}
   550→          newCustomer={newCustomer}
   551→          cart={cart}
   552→          paymentType={paymentType}
   553→          advanceAmount={advanceAmount}
   554→          advancePaymentMethod={advancePaymentMethod}
   555→          advancePaymentReference={advancePaymentReference}
   556→          deliveryAddress={deliveryAddress}
   557→          deliveryCity={deliveryCity}
   558→          courierName={courierName}
   559→          dispatchLocationId={dispatchLocationId}
   560→          locations={locationsQuery.data?.locations ?? []}
   561→          notesForCourier={notesForCourier}
   562→          discountAmount={discountAmount}
   563→          discountReason={discountReason}
   564→          subtotal={subtotal}
   565→          discount={discount}
   566→          total={total}
   567→        />
   568→      )}
   569→
   570→      {/* ── Step navigation ─────────────────────────────────────────────────── */}
   571→      <div className="flex items-center justify-between gap-3">
   572→        <Button
   573→          variant="outline"
   574→          onClick={() => (step === 1 ? navigate({ name: 'orders' }) : setStep(step - 1))}
   575→          disabled={createMutation.isPending}
   576→        >
   577→          <ArrowLeft className="h-4 w-4" />
   578→          {step === 1 ? 'Cancel' : 'Back'}
   579→        </Button>
   580→
   581→        <div className="flex items-center gap-2 text-xs text-muted-foreground">
   582→          Step {step} of {STEPS.length}
   583→        </div>
   584→
   585→        {step < 5 ? (
   586→          <Button onClick={() => setStep(step + 1)} disabled={!canGoNext()}>
   587→            Continue
   588→            <ArrowRight className="h-4 w-4" />
   589→          </Button>
   590→        ) : (
   591→          <Button
   592→            onClick={handleSubmit}
   593→            disabled={
   594→              createMutation.isPending ||
   595→              !customerStepValid ||
   596→              !itemsStepValid ||
   597→              !paymentStepValid ||
   598→              !deliveryStepValid
   599→            }
   600→          >
   601→            {createMutation.isPending ? (
   602→              <>
   603→                <Loader2 className="h-4 w-4 animate-spin" /> Creating…
   604→              </>
   605→            ) : (
   606→              <>
   607→                <CheckCircle2 className="h-4 w-4" /> Create Order
   608→              </>
   609→            )}
   610→          </Button>
   611→        )}
   612→      </div>
   613→    </div>
   614→  )
   615→}
