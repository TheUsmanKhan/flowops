   211→/// Company-scoped roles. System roles (Owner/Founder/Co-Founder/Investor) are
   212→/// elevated and bypass all permission checks. Custom roles use role_permissions.
   213→model Role {
   214→  id            String   @id @default(cuid())
   215→  companyId     String
   216→  company       Company  @relation(fields: [companyId], references: [id], onDelete: Cascade)
   217→  name          String
   218→  description   String?
   219→  roleTier      String   @default("standard") // elevated | standard
   220→  isSystemRole  Boolean  @default(false)
   221→  systemRoleKey String?  // owner | founder | co_founder | investor | null
   222→  // NOTE: systemRoleKey is unique PER COMPANY (not globally), so every
   223→  // company can seed its own owner/founder/co_founder/investor roles.
   224→  // ─── Data scope for the Orders module ───────────────────────────────
   225→  // "all"  → role sees ALL orders in the company (default, preserves
   226→  //          existing behavior for every row after migration)
   227→  // "own"  → role sees only orders where salesEmployeeId === the caller's
   228→  //          own Employee.id (enforced at the query layer, not in SQL)
   229→  ordersDataScope String  @default("all") // "own" | "all"
   230→  isActive      Boolean  @default(true)
   231→  createdById   String?
   232→  createdBy     Profile? @relation("RoleCreator", fields: [createdById], references: [id])
   233→  createdAt     DateTime @default(now())
   234→  updatedAt     DateTime @updatedAt
   235→
   236→  employees       Employee[]
   237→  rolePermissions RolePermission[]
   238→  invitations     Invitation[]
   239→
   240→  @@unique([companyId, name])
   241→  @@unique([companyId, systemRoleKey])
   242→}
