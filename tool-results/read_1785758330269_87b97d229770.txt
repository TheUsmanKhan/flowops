   104→/**
   105→ * Fetches live inventory per variant for this product and returns a map keyed
   106→ * by variantId. Used by the parent-child table to display on-hand stock
   107→ * alongside cost/price columns. Reads from the same inventory_pools data as
   108→ * the Inventory tab and the Inventory Dashboard — no parallel data source.
   109→ */
   110→function useVariantInventoryMap(productId: string) {
   111→  const { data } = useQuery<InventorySummaryResponse>({
   112→    queryKey: ['product-inventory', productId],
   113→    queryFn: () => api.get(`/api/inventory/summary?product_id=${productId}`),
   114→    staleTime: 15_000,
   115→  })
   116→  const map = new Map<string, InventorySummaryVariant>()
   117→  for (const v of data?.variants ?? []) {
   118→    map.set(v.variantId, v)
   119→  }
   120→  return map
   121→}

   123→/** Compact stock cell — shows on_hand, with available in muted subtext. */
   124→function StockCell({ inv }: { inv?: InventorySummaryVariant }) {
   125→  if (!inv) {
   126→    return <span className="text-xs text-muted-foreground">—</span>
   127→  }
   128→  const onHand = inv.totalOnHand
   129→  const available = inv.totalAvailable
   130→  if (onHand === 0) {
   131→    return <span className="text-xs text-muted-foreground">0</span>
   132→  }
   133→  const lowStock = available > 0 && available <= 5
   134→  return (
   135→    <div className="text-xs leading-tight">
   136→      <span className={cn('font-medium', lowStock ? 'text-amber-700' : 'text-foreground')}>{onHand}</span>
   137→      {available !== onHand && (
   138→        <div className="text-[10px] text-muted-foreground">{available} avail.</div>
   139→      )}
   140→    </div>
   141→  )
   142→}

   144→/**
   145→ * Bug 2 fix: badge that reads trackInventory (mutable), NOT fulfillmentType
   146→ * (immutable). A made_to_order variant with opening stock added shows
   147→ * "Stock Tracked" because trackInventory is true. Falls back to
   148→ * fulfillmentType-based display if trackInventory is undefined (e.g. older

   179→export function ParentChildVariantTable({
   180→  productId,
   181→  mode = 'both',
   182→}: {
   183→  productId: string
   184→  mode?: 'cost' | 'pricing' | 'both'
   185→}) {
   186→  const can = useCan()
   187→  const queryClient = useQueryClient()
   188→  const [expandedGroups, setExpandedGroups] = useState<Set<string>>(new Set())
   189→  const [editingVariant, setEditingVariant] = useState<ChildVariant | null>(null)
   190→
   191→  const canEditCost = can(PERMISSIONS.PRODUCTS_EDIT)
   192→  const canEditPrice = can(PERMISSIONS.PRODUCTS_PRICING)
   193→  const showCost = mode === 'cost' || mode === 'both'
   194→  const showPricing = mode === 'pricing' || mode === 'both'
   195→
   196→  const { data, isLoading, isError, refetch } = useQuery<VariantGroupsResponse>({
   197→    queryKey: ['variant-groups', productId],
   198→    queryFn: () => api.get(`/api/products/${productId}/variant-groups`),
   199→    staleTime: 15_000,
   200→  })
   201→
   202→  // Live inventory per variant — powers the read-only Stock column.
   203→  const inventoryMap = useVariantInventoryMap(productId)
   204→
   205→  function toggleGroup(parentValue: string) {
   206→    setExpandedGroups((prev) => {
   207→      const next = new Set(prev)
   208→      if (next.has(parentValue)) next.delete(parentValue)
   209→      else next.add(parentValue)
   210→      return next
   211→    })
   212→  }
   213→
   214→  // Expand all groups by default on first load
   215→  if (data && expandedGroups.size === 0 && data.groups.length > 0) {
   216→    setExpandedGroups(new Set(data.groups.map((g) => g.parentValue)))
   217→  }
   218→
   219→  if (isLoading) {
   220→    return (
   221→      <Card>
   222→        <CardContent className="p-6 space-y-3">
   223→          {[0, 1, 2].map((i) => (
   224→            <div key={i} className="h-20 rounded bg-muted/50 animate-pulse" />
   225→          ))}
   226→        </CardContent>
   227→      </Card>
   228→    )
   229→  }
   230→
   231→  if (isError) {
   232→    return (
   233→      <Card>
   234→        <CardContent className="p-6 text-center">
   235→          <AlertCircle className="h-8 w-8 mx-auto text-muted-foreground mb-2" />
   236→          <p className="text-sm text-muted-foreground mb-3">Failed to load variant groups.</p>
   237→          <Button variant="outline" size="sm" onClick={() => refetch()}>Retry</Button>
   238→        </CardContent>
   239→      </Card>
   240→    )
   241→  }
   242→
   243→  if (!data || data.groups.length === 0) {
   244→    return (
   245→      <Card>
   246→        <CardContent className="p-6 text-center text-sm text-muted-foreground">
   247→          No variants found for this product.
   248→        </CardContent>
   249→      </Card>
   250→    )
   251→  }
   252→
   253→  // If only one attribute, render flat table (no grouping)
   254→  if (!data.hasMultipleAttributes) {
   255→    return (
   256→      <>
   257→        <FlatVariantTable
   258→          groups={data.groups}
   259→          productId={productId}
   260→          showCost={showCost}
   261→          showPricing={showPricing}
   262→          canEditCost={canEditCost}
   263→          canEditPrice={canEditPrice}
   264→          queryClient={queryClient}
   265→          onEdit={(v) => setEditingVariant(v)}
   266→          inventoryMap={inventoryMap}
   267→        />
   268→        {editingVariant && (
   269→          <VariantEditDialog
   270→            variant={editingVariant}
   271→            productId={productId}
   272→            onClose={() => setEditingVariant(null)}
   273→            queryClient={queryClient}
   274→          />
   275→        )}
   276→      </>
   277→    )
   278→  }
   279→
   280→  // Grouped table with parent-child cascade
   281→  return (
   282→    <div className="space-y-3">
   283→      {data.parentAttributeName && (
   284→        <p className="text-xs text-muted-foreground">
   285→          Grouped by <span className="font-medium text-foreground">{data.parentAttributeDisplayName || data.parentAttributeName}</span> —
   286→          set prices once per group, or override individual variants.
   287→        </p>
   288→      )}
   289→      {data.groups.map((group) => (
   290→        <GroupCard
   291→          key={group.parentValue}
   292→          group={group}
   293→          productId={productId}
   294→          parentAttributeName={data.parentAttributeName!}
   295→          expanded={expandedGroups.has(group.parentValue)}
   296→          onToggle={() => toggleGroup(group.parentValue)}
   297→          showCost={showCost}
   298→          showPricing={showPricing}
   299→          canEditCost={canEditCost}
   300→          canEditPrice={canEditPrice}
   301→          queryClient={queryClient}
   302→          onEditVariant={(v) => setEditingVariant(v)}
   303→          inventoryMap={inventoryMap}
   304→        />
   305→      ))}
   306→      {editingVariant && (
   307→        <VariantEditDialog
   308→          variant={editingVariant}
   309→          productId={productId}
   310→          onClose={() => setEditingVariant(null)}
   311→          queryClient={queryClient}
   312→        />
   313→      )}
   314→    </div>
   315→  )
   316→}

   321→function GroupCard({
   322→  group,
   323→  productId,
   324→  parentAttributeName,
   325→  expanded,
   326→  onToggle,
   327→  showCost,
   328→  showPricing,
   329→  canEditCost,
   330→  canEditPrice,
   331→  queryClient,
   332→  onEditVariant,
   333→  inventoryMap,
   334→}: {
   335→  group: VariantGroup
   336→  productId: string
   337→  parentAttributeName: string
   338→  expanded: boolean
   339→  onToggle: () => void
   340→  showCost: boolean
   341→  showPricing: boolean
   342→  canEditCost: boolean
   343→  canEditPrice: boolean
   344→  queryClient: ReturnType<typeof useQueryClient>
   345→  onEditVariant: (v: ChildVariant) => void
   346→  inventoryMap: Map<string, InventorySummaryVariant>
   347→}) {
   348→  const [parentCost, setParentCost] = useState('')
   349→  const [parentSale, setParentSale] = useState('')
   350→  const [parentCompare, setParentCompare] = useState('')
   351→  const [applying, setApplying] = useState(false)
   352→
   353→  const firstChild = group.children[0]
   354→  if (parentCost === '' && firstChild) setParentCost(String(firstChild.costPrice))
   355→  if (parentSale === '' && firstChild?.salePrice != null) setParentSale(String(firstChild.salePrice))
   356→  if (parentCompare === '' && firstChild?.comparePrice != null) setParentCompare(String(firstChild.comparePrice))
   357→
   358→  // Bug 1 fix: ONE handler cascades ALL THREE fields (cost, sale, compare).
   359→  // Calls both server endpoints (cost + sale-price) in sequence, then
   360→  // invalidates the cache once. Each endpoint only updates children whose
   361→  // relevant synced flag is true, so the three flags remain INDEPENDENT.
   362→  async function applyAllToGroup() {
   363→    const cost = Number(parentCost)
   364→    const sale = Number(parentSale)
   365→    const compare = parentCompare ? Number(parentCompare) : null
   366→
   367→    if (isNaN(cost) || cost < 0) { toast.error('Enter a valid cost price'); return }
   368→    if (isNaN(sale) || sale < 0) { toast.error('Enter a valid sale price'); return }
   369→
   370→    setApplying(true)
   371→    let costCount = 0
   372→    let saleCount = 0
   373→    try {
   374→      // Cascade cost to cost-synced children
   375→      const costRes = await api.post<{ success: boolean; updated_count: number }>(
   376→        `/api/products/${productId}/variant-groups/dummy/cost`,
   377→        { cost_price: cost, parent_attribute_name: parentAttributeName, parent_value: group.parentValue },
   378→      )
   379→      costCount = costRes.updated_count
   380→
   381→      // Cascade sale + compare to their respective synced children
   382→      const saleRes = await api.post<{ success: boolean; updated_count: number }>(
   383→        `/api/products/${productId}/variant-groups/dummy/sale-price`,
   384→        { sale_price: sale, compare_price: compare, parent_attribute_name: parentAttributeName, parent_value: group.parentValue },
   385→      )
   386→      saleCount = saleRes.updated_count
   387→
   388→      toast.success(`Applied to group — Cost: ${costCount}, Sale+Compare: ${saleCount} variant(s)`)
   389→      queryClient.invalidateQueries({ queryKey: ['variant-groups', productId] })
   390→      queryClient.invalidateQueries({ queryKey: ['product', productId] })
   391→    } catch (err) {
   392→      toast.error(err instanceof FetchError ? err.message : 'Failed to apply to group')
   393→    } finally { setApplying(false) }
   394→  }
   395→
   396→  const childAttrKeys = group.children.length > 0
   397→    ? Object.keys(group.children[0].attributeValues).filter((k) => k !== parentAttributeName)
   398→    : []
   399→
   400→  return (
   401→    <Card>
   402→      <CardHeader className="pb-3 cursor-pointer" onClick={onToggle}>
   403→        <div className="flex items-center justify-between">
   404→          <div className="flex items-center gap-2">
   405→            {expanded ? <ChevronDown className="h-4 w-4 text-muted-foreground" /> : <ChevronRight className="h-4 w-4 text-muted-foreground" />}
   406→            <CardTitle className="text-base">{group.parentValue}</CardTitle>
   407→            <Badge variant="secondary" className="text-[10px]">{group.childCount} variant{group.childCount === 1 ? '' : 's'}</Badge>
   408→          </div>
   409→        </div>
   410→      </CardHeader>
   411→
   412→      {expanded && (
   413→        <CardContent className="space-y-4">
   414→          {/* Parent group inputs — uses the shared ParentGroupInputs component
   415→              with a single "Apply to Group" button that cascades ALL THREE
   416→              fields (cost, sale, compare) independently. Bug 1 fix. */}
   417→          <ParentGroupInputs
   418→            parentCost={parentCost}
   419→            parentSale={parentSale}
   420→            parentCompare={parentCompare}
   421→            onCostChange={setParentCost}
   422→            onSaleChange={setParentSale}
   423→            onCompareChange={setParentCompare}
   424→            onApplyAll={applyAllToGroup}
   425→            showCost={showCost}
   426→            showPricing={showPricing}
   427→            canEditCost={canEditCost}
   428→            canEditPrice={canEditPrice}
   429→            applying={applying}
   430→          />
   431→
   432→          {/* Children table */}
   433→          <div className="rounded-md border overflow-x-auto scrollbar-thin">
   434→            <table className="w-full text-sm">
   435→              <thead className="bg-muted/30">
   436→                <tr className="text-left text-xs text-muted-foreground">
   437→                  {childAttrKeys.map((k) => (<th key={k} className="px-3 py-2 font-medium">{k}</th>))}
   438→                  <th className="px-3 py-2 font-medium">SKU</th>
   439→                  <th className="px-3 py-2 font-medium">Fulfillment</th>
   440→                  <th className="px-3 py-2 font-medium text-center">Stock</th>
   441→                  {showCost && <th className="px-3 py-2 font-medium text-right">Cost</th>}
   442→                  {showPricing && <th className="px-3 py-2 font-medium text-right">Sale</th>}
   443→                  {showPricing && <th className="px-3 py-2 font-medium text-right">Compare</th>}
   444→                  <th className="px-3 py-2 font-medium text-center">Active</th>
   445→                  <th className="px-3 py-2 font-medium">Actions</th>
   446→                </tr>
   447→              </thead>
   448→              <tbody>
   449→                {group.children.map((child) => (
   450→                  <ChildRow
   451→                    key={child.variantId}
   452→                    child={child}
   453→                    childAttrKeys={childAttrKeys}
   454→                    productId={productId}
   455→                    showCost={showCost}
   456→                    showPricing={showPricing}
   457→                    canEditCost={canEditCost}
   458→                    canEditPrice={canEditPrice}
   459→                    queryClient={queryClient}
   460→                    onEdit={() => onEditVariant(child)}
   461→                    inventory={inventoryMap.get(child.variantId)}
   462→                  />
   463→                ))}
   464→              </tbody>
   465→            </table>
   466→          </div>
   467→        </CardContent>
   468→      )}
   469→    </Card>
   470→  )
   471→}

   473→// ──────────────────────────────────────────────────────────────

   476→function ChildRow({
   477→  child,
   478→  childAttrKeys,
   479→  productId,
   480→  showCost,
   481→  showPricing,
   482→  canEditCost,
   483→  canEditPrice,
   484→  queryClient,
   485→  onEdit,
   486→  inventory,
   487→}: {
   488→  child: ChildVariant
   489→  childAttrKeys: string[]
   490→  productId: string
   491→  showCost: boolean
   492→  showPricing: boolean
   493→  canEditCost: boolean
   494→  canEditPrice: boolean
   495→  queryClient: ReturnType<typeof useQueryClient>
   496→  onEdit: () => void
   497→  inventory?: InventorySummaryVariant
   498→}) {
   499→  const [costValue, setCostValue] = useState(String(child.costPrice))
   500→  const [saleValue, setSaleValue] = useState(child.salePrice != null ? String(child.salePrice) : '')
   501→  const [compareValue, setCompareValue] = useState(child.comparePrice != null ? String(child.comparePrice) : '')
   502→  const [saving, setSaving] = useState(false)
   503→  const [toggling, setToggling] = useState(false)
   504→
   505→  async function saveCost() {
   506→    const newCost = Number(costValue)
   507→    if (isNaN(newCost) || newCost < 0) return
   508→    if (newCost === child.costPrice) return
   509→    setSaving(true)
   510→    try {
   511→      await api.post(`/api/products/${productId}/variants/${child.variantId}/override-cost`, { cost_price: newCost })
   512→      toast.success('Cost overridden — no longer synced with parent')
   513→      queryClient.invalidateQueries({ queryKey: ['variant-groups', productId] })
   514→      queryClient.invalidateQueries({ queryKey: ['product', productId] })
   515→    } catch (err) {
   516→      toast.error(err instanceof FetchError ? err.message : 'Failed to override cost')
   517→      setCostValue(String(child.costPrice))
   518→    } finally { setSaving(false) }
   519→  }
   520→
   521→  async function savePrice() {
   522→    const newSale = Number(saleValue)
   523→    const newCompare = compareValue ? Number(compareValue) : null
   524→    if (isNaN(newSale) || newSale < 0) return
   525→    setSaving(true)
   526→    try {
   527→      await api.post(`/api/products/${productId}/variants/${child.variantId}/override-price`, { sale_price: newSale, compare_price: newCompare })
   528→      toast.success('Price overridden — no longer synced with parent')
   529→      queryClient.invalidateQueries({ queryKey: ['variant-groups', productId] })
   530→      queryClient.invalidateQueries({ queryKey: ['product', productId] })
   531→    } catch (err) {
   532→      toast.error(err instanceof FetchError ? err.message : 'Failed to override price')
   533→      setSaleValue(child.salePrice != null ? String(child.salePrice) : '')
   534→      setCompareValue(child.comparePrice != null ? String(child.comparePrice) : '')
   535→    } finally { setSaving(false) }
   536→  }
   537→
   538→  async function resyncCost() {
   539→    setSaving(true)
   540→    try {
   541→      const res = await api.post<{ success: boolean; cost_price: number }>(`/api/products/${productId}/variants/${child.variantId}/resync-cost`)
   542→      toast.success(`Re-synced with parent (Rs. ${res.cost_price})`)
   543→      queryClient.invalidateQueries({ queryKey: ['variant-groups', productId] })
   544→      queryClient.invalidateQueries({ queryKey: ['product', productId] })
   545→    } catch (err) {
   546→      toast.error(err instanceof FetchError ? err.message : 'Failed to re-sync')
   547→    } finally { setSaving(false) }
   548→  }
   549→
   550→  async function resyncPrice(field: 'sale_price' | 'compare_price') {
   551→    setSaving(true)
   552→    try {
   553→      await api.post(`/api/products/${productId}/variants/${child.variantId}/resync-price`, { field })
   554→      toast.success(`Re-synced ${field === 'sale_price' ? 'sale price' : 'compare price'} with parent`)
   555→      queryClient.invalidateQueries({ queryKey: ['variant-groups', productId] })
   556→      queryClient.invalidateQueries({ queryKey: ['product', productId] })
   557→    } catch (err) {
   558→      toast.error(err instanceof FetchError ? err.message : 'Failed to re-sync')
   559→    } finally { setSaving(false) }
   560→  }
   561→
   562→  async function toggleActive() {
   563→    setToggling(true)

   649→// Variant Edit Dialog — full editing of barcode, weight, stitching, etc.

   651→function VariantEditDialog({
   652→  variant,
   653→  productId,
   654→  onClose,
   655→  queryClient,
   656→}: {
   657→  variant: ChildVariant
   658→  productId: string
   659→  onClose: () => void
   660→  queryClient: ReturnType<typeof useQueryClient>
   661→}) {
   662→  const [form, setForm] = useState({
   663→    sku: variant.sku,
   664→    barcode: variant.barcode ?? '',
   665→    costPrice: String(variant.costPrice),
   666→    weightGrams: String(variant.weightGrams ?? 0),
   667→    stitchingCharges: String(variant.stitchingCharges ?? 0),
   668→    productionDays: String(variant.productionDays ?? 0),
   669→    isTaxable: variant.isTaxable ?? true,
   670→    requiresShipping: variant.requiresShipping ?? true,
   671→  })
   672→  const [saving, setSaving] = useState(false)
   673→  const [confirmSkuChange, setConfirmSkuChange] = useState(false)
   674→
   675→  async function save() {
   676→    // If SKU changed, show confirmation
   677→    if (form.sku !== variant.sku && !confirmSkuChange) {
   678→      setConfirmSkuChange(true)
   679→      return
   680→    }
   681→
   682→    setSaving(true)
   683→    try {
   684→      const patch: Record<string, unknown> = {}
   685→      if (form.sku !== variant.sku) patch.sku = form.sku
   686→      if (form.barcode !== (variant.barcode ?? '')) patch.barcode = form.barcode
   687→      if (Number(form.costPrice) !== variant.costPrice) patch.cost_price = Number(form.costPrice)
   688→      if (Number(form.weightGrams) !== (variant.weightGrams ?? 0)) patch.weight_grams = Number(form.weightGrams)
   689→      if (Number(form.stitchingCharges) !== (variant.stitchingCharges ?? 0)) patch.stitching_charges = Number(form.stitchingCharges)
   690→      if (Number(form.productionDays) !== (variant.productionDays ?? 0)) patch.production_days = Number(form.productionDays)
   691→      if (form.isTaxable !== (variant.isTaxable ?? true)) patch.is_taxable = form.isTaxable
   692→      if (form.requiresShipping !== (variant.requiresShipping ?? true)) patch.requires_shipping = form.requiresShipping
   693→
   694→      if (Object.keys(patch).length === 0) {
   695→        onClose()
   696→        return
   697→      }
   698→
   699→      await api.patch(`/api/products/${productId}/variants/${variant.variantId}`, patch)
   700→      toast.success('Variant updated')
   701→      queryClient.invalidateQueries({ queryKey: ['variant-groups', productId] })
   702→      queryClient.invalidateQueries({ queryKey: ['product', productId] })
   703→      onClose()
   704→    } catch (err) {
   705→      toast.error(err instanceof FetchError ? err.message : 'Failed to update variant')
   706→    } finally {
   707→      setSaving(false)
   708→      setConfirmSkuChange(false)
   709→    }
   710→  }
   711→
   712→  return (
   713→    <Dialog open onOpenChange={(v) => !v && onClose()}>
   714→      <DialogContent className="max-w-lg">
   715→        <DialogHeader>
   716→          <DialogTitle>Edit Variant</DialogTitle>
   717→        </DialogHeader>
   718→
   719→        {confirmSkuChange && (
   720→          <div className="rounded-lg border border-amber-300 bg-amber-50 p-3 text-sm text-amber-800">
   721→            <p className="font-medium">Change SKU?</p>
   722→            <p className="text-xs mt-1">Changing the SKU won&apos;t affect history but may cause confusion with existing labels. Continue?</p>
   723→          </div>
   724→        )}
   725→
   726→        <div className="grid grid-cols-2 gap-4">
   727→          <div className="space-y-1.5">
   728→            <Label className="text-xs">SKU</Label>
   729→            <Input value={form.sku} onChange={(e) => setForm({ ...form, sku: e.target.value })} className="h-9 text-sm font-mono" />
   730→          </div>
   731→          <div className="space-y-1.5">
   732→            <Label className="text-xs">Barcode</Label>
   733→            <Input value={form.barcode} onChange={(e) => setForm({ ...form, barcode: e.target.value })} className="h-9 text-sm" placeholder="UPC/EAN" />
   734→          </div>
   735→          <div className="space-y-1.5">
   736→            <Label className="text-xs">Cost Price</Label>
   737→            <Input type="number" min="0" step="0.01" value={form.costPrice} onChange={(e) => setForm({ ...form, costPrice: e.target.value })} className="h-9 text-sm" />
   738→          </div>
   739→          <div className="space-y-1.5">
   740→            <Label className="text-xs">Weight (grams)</Label>
   741→            <Input type="number" min="0" step="1" value={form.weightGrams} onChange={(e) => setForm({ ...form, weightGrams: e.target.value })} className="h-9 text-sm" />
   742→          </div>
   743→          <div className="space-y-1.5">
   744→            <Label className="text-xs">Stitching Charges</Label>
   745→            <Input type="number" min="0" step="0.01" value={form.stitchingCharges} onChange={(e) => setForm({ ...form, stitchingCharges: e.target.value })} className="h-9 text-sm" />
   746→          </div>
   747→          <div className="space-y-1.5">
   748→            <Label className="text-xs">Production Days</Label>
   749→            <Input type="number" min="0" step="1" value={form.productionDays} onChange={(e) => setForm({ ...form, productionDays: e.target.value })} className="h-9 text-sm" />
   750→          </div>
   751→          <div className="flex items-center gap-2">
   752→            <Switch checked={form.isTaxable} onCheckedChange={(v) => setForm({ ...form, isTaxable: v })} />
   753→            <Label className="text-xs">Taxable</Label>
   754→          </div>
   755→          <div className="flex items-center gap-2">
   756→            <Switch checked={form.requiresShipping} onCheckedChange={(v) => setForm({ ...form, requiresShipping: v })} />
   757→            <Label className="text-xs">Requires Shipping</Label>
   758→          </div>
   759→        </div>
   760→
   761→        <div className="flex items-center gap-2 pt-2 border-t">
   762→          <FulfillmentTypeBadge type={variant.fulfillmentType} />
   763→          <span className="text-xs text-muted-foreground">Fulfillment type cannot be changed after creation</span>
   764→        </div>
   765→
   766→        <DialogFooter>
   767→          <Button variant="ghost" onClick={onClose}>Cancel</Button>
   768→          <Button onClick={save} disabled={saving}>
   769→            {saving ? <Loader2 className="h-4 w-4 animate-spin" /> : <Check className="h-4 w-4" />}
   770→            {confirmSkuChange ? 'Confirm SKU Change & Save' : 'Save Changes'}
   771→          </Button>
   772→        </DialogFooter>
   773→      </DialogContent>
   774→    </Dialog>
   775→  )
   776→}

   849→function FlatRow({
   850→  child,
   851→  attrValues,
   852→  productId,
   853→  showCost,
   854→  showPricing,
   855→  canEditCost,
   856→  canEditPrice,
   857→  queryClient,
   858→  onEdit,
   859→  inventory,
   860→}: {
   861→  child: ChildVariant
   862→  attrValues: string
   863→  productId: string
   864→  showCost: boolean
   865→  showPricing: boolean
   866→  canEditCost: boolean
   867→  canEditPrice: boolean
   868→  queryClient: ReturnType<typeof useQueryClient>
   869→  onEdit: () => void
   870→  inventory?: InventorySummaryVariant
   871→}) {
   872→  const [costValue, setCostValue] = useState(String(child.costPrice))
   873→  const [saleValue, setSaleValue] = useState(child.salePrice != null ? String(child.salePrice) : '')
   874→  const [compareValue, setCompareValue] = useState(child.comparePrice != null ? String(child.comparePrice) : '')
   875→  const [toggling, setToggling] = useState(false)
   876→
   877→  async function saveCost() {
   878→    const newCost = Number(costValue)
   879→    if (isNaN(newCost) || newCost < 0 || newCost === child.costPrice) return
   880→    try {
   881→      await api.post(`/api/products/${productId}/variants/${child.variantId}/override-cost`, { cost_price: newCost })
   882→      toast.success('Cost updated')
   883→      queryClient.invalidateQueries({ queryKey: ['variant-groups', productId] })
   884→    } catch (err) { toast.error(err instanceof FetchError ? err.message : 'Failed') }
   885→  }
   886→
   887→  async function savePrice() {
   888→    const newSale = Number(saleValue)
   889→    if (isNaN(newSale) || newSale < 0) return
   890→    try {
   891→      await api.post(`/api/products/${productId}/variants/${child.variantId}/override-price`, { sale_price: newSale, compare_price: compareValue ? Number(compareValue) : null })
   892→      toast.success('Price updated')
   893→      queryClient.invalidateQueries({ queryKey: ['variant-groups', productId] })
   894→    } catch (err) { toast.error(err instanceof FetchError ? err.message : 'Failed') }
   895→  }
   896→
   897→  async function toggleActive() {
   898→    setToggling(true)
   899→    try {
   900→      await api.post(`/api/products/${productId}/variants/${child.variantId}/toggle`, { is_active: !child.isActive })
   901→      toast.success(child.isActive ? 'Deactivated' : 'Activated')
   902→      queryClient.invalidateQueries({ queryKey: ['variant-groups', productId] })
   903→      queryClient.invalidateQueries({ queryKey: ['product', productId] })
   904→    } catch (err) { toast.error(err instanceof FetchError ? err.message : 'Failed') }
   905→    finally { setToggling(false) }
   906→  }
   907→
   908→  return (
   909→    <tr className="border-t hover:bg-muted/20">
   910→      <td className="px-3 py-2 text-xs">
   911→        {attrValues || '—'}
   912→        {child.isDefault && <Badge variant="secondary" className="ml-1 text-[9px]">DEFAULT</Badge>}
   913→      </td>
   914→      <td className="px-3 py-2">
   915→        <span className="font-mono text-xs">{child.sku}</span>
   916→        {child.barcode && <div className="text-[10px] text-muted-foreground">{child.barcode}</div>}
   917→      </td>
   918→      <td className="px-3 py-2"><TrackingBadge trackInventory={child.trackInventory} fulfillmentType={child.fulfillmentType} /></td>
   919→      <td className="px-3 py-2 text-center"><StockCell inv={inventory} /></td>
   920→      {showCost && (
   921→        <td className="px-3 py-2 text-right">
   922→          {canEditCost ? (
   923→            <Input type="number" min="0" step="0.01" value={costValue} onChange={(e) => setCostValue(e.target.value)} onBlur={saveCost} className="h-7 w-20 text-xs text-right" />
   924→          ) : <span className="text-xs">{child.costPrice}</span>}
   925→        </td>
   926→      )}
   927→      {showPricing && (
   928→        <td className="px-3 py-2 text-right">
   929→          {canEditPrice ? (
   930→            <Input type="number" min="0" step="0.01" value={saleValue} onChange={(e) => setSaleValue(e.target.value)} onBlur={savePrice} className="h-7 w-20 text-xs text-right" />
   931→          ) : <span className="text-xs">{child.salePrice ?? '—'}</span>}
   932→        </td>
   933→      )}
   934→      {showPricing && (
   935→        <td className="px-3 py-2 text-right text-xs text-muted-foreground">{child.comparePrice ?? '—'}</td>
   936→      )}
   937→      <td className="px-3 py-2 text-center">
   938→        {canEditCost ? (
   939→          <Switch checked={child.isActive} onCheckedChange={toggleActive} disabled={toggling} />
   940→        ) : (
   941→          <Badge variant={child.isActive ? 'secondary' : 'outline'} className="text-[10px]">{child.isActive ? 'Active' : 'Inactive'}</Badge>

   943→      </td>
