   151→function getErrorMessage(err: unknown): string {
   152→  if (err instanceof FetchError) return err.message
   153→  if (err instanceof Error) return err.message
   154→  return 'Something went wrong. Please try again.'
   155→}

   157→function parseQty(v: string): number {
   158→  if (v === '') return 0
   159→  const n = Number(v)
   160→  return Number.isFinite(n) && n >= 0 ? Math.floor(n) : 0
   161→}
   162→
   163→function parsePrice(v: string): number {
   164→  if (v === '') return 0
   165→  const n = Number(v)
   166→  return Number.isFinite(n) && n >= 0 ? n : 0
   167→}

   691→      order_ref_number: orderRefNumber.trim() || undefined,

   707→      payload.advance_payment_reference = advancePaymentReference.trim() || undefined

   709→      // The file is uploaded and persisted AFTER the order is created
   710→      // (see handleSubmit).
   711→    }
   712→

   714→    // selectedCustomer (existing or just-created). The new schema sends

   742→  // ── Upload payment proof + persist URL to the order ───────────────────────
   743→  // Returns true on success, false on any failure. NEVER throws.
   744→  async function uploadPaymentProof(orderId: string): Promise<boolean> {
   745→    if (!paymentProofFile) return true
   746→
   747→    setUploadingProof(true)
   748→    try {
   749→      const fd = new FormData()
   750→      fd.append('file', paymentProofFile)
   751→      const uploadRes = await fetch(
   752→        `/api/upload?type=payment-proofs&id=${orderId}`,
   753→        { method: 'POST', body: fd, credentials: 'include', headers: getSessionToken() ? { Authorization: `Bearer ${getSessionToken()}` } : {} },
   754→      )
   755→      const uploadText = await uploadRes.text()
   756→      let uploadBody: unknown = null
   757→      if (uploadText) {
   758→        try {
   759→          uploadBody = JSON.parse(uploadText)
   760→        } catch {
   761→          uploadBody = uploadText
   762→        }
   763→      }
   764→      if (!uploadRes.ok) {
   765→        const msg =
   766→          uploadBody && typeof uploadBody === 'object' && 'error' in uploadBody
   767→            ? String((uploadBody as { error: unknown }).error)
   768→            : typeof uploadBody === 'string'
   769→              ? uploadBody
   770→              : `Upload failed (HTTP ${uploadRes.status})`
   771→        throw new Error(msg)
   772→      }
   773→      const { url } = uploadBody as { url: string }
   774→
   775→      await api.post(`/api/orders/${orderId}/payment-proof`, {
   776→        advance_payment_screenshot_url: url,
   777→      })
   778→      return true
   779→    } catch {
   780→      return false
   781→    } finally {
   782→      setUploadingProof(false)
   783→    }
   784→  }

  1599→// CRM Stats widget — shows when a customer is selected
  1600→// ─────────────────────────────────────────────────────────────────────────────
  1601→
  1602→function CrmStatsWidget({ customer }: { customer: SelectedCustomer }) {
  1603→  const totalOrders = customer.totalOrdersCount
  1604→  const totalRto = customer.totalRtoCount
  1605→  // Derive a delivery rate from the cached totals (returned orders / total).
  1606→  // This is informational — not the source of truth (the live query is).
  1607→  const deliveryRate =
  1608→    totalOrders > 0 ? ((totalOrders - totalRto) / totalOrders) * 100 : 100
  1609→  const rtoRate = totalOrders > 0 ? (totalRto / totalOrders) * 100 : 0
  1610→  const deliveredCount = Math.max(0, totalOrders - totalRto)

  1677→function formatLastUsedShort(iso: string | null): string {
  1678→  if (!iso) return 'Never used'
  1679→  const date = new Date(iso)
  1680→  const now = new Date()
  1681→  const diffMs = now.getTime() - date.getTime()
  1682→  const diffDays = Math.floor(diffMs / 86_400_000)
  1683→  if (diffDays < 1) return 'Today'
  1684→  if (diffDays === 1) return '1 day ago'
  1685→  if (diffDays < 30) return `${diffDays} days ago`
  1686→  return date.toLocaleDateString('en-PK', { day: '2-digit', month: 'short' })
  1687→}
