    74→    74→// Helper: reserveOrderStock — shared internal function
    75→    75→// ──────────────────────────────────────────────────────────────
    76→    76→// Processes ALL order_items for an order and attempts to reserve
    77→    77→// stock for each based on its fulfillment_type_snapshot.
    78→    78→//
    79→    79→// For stock_based items:
    80→    80→//   - If sufficient available stock → reserveStockForOrder(), fulfillment_status='reserved'
    81→    81→//   - If insufficient + inventory_policy='continue' → fulfillment_status='backordered'
    82→    82→//   - If insufficient + inventory_policy='deny' → outcome='failed'
    83→    83→//
    84→    84→// For made_to_order items:
    85→    85→//   - checkAndFulfillMadeToOrderVariant() checks returned stock first
    86→    86→//   - If existing_stock → fulfillment_status='reserved', returned_stitched_used=true
    87→    87→//   - If fresh_production → fulfillment_status='reserved', production_order_id set
    88→    88→//   - If error → outcome='failed'
    89→    89→//
    90→    90→// After processing: if ANY item is 'backordered', order status → 'partially_backordered'.
    91→    91→// If ALL items are 'reserved', order stays 'confirmed'.
    92→    92→//
    93→    93→// Called from:
    94→    94→//   - createManualOrder() IF the order auto-confirmed at creation
    95→    95→//   - confirmOrder() when a 'pending' order is manually confirmed
    96→    96→//   - dispatchOrderAction() if status is still 'pending' at dispatch time
    97→    97→
    98→    98→async function reserveOrderStock(
    99→    99→  orderId: string,
   100→   100→  ctx: { company: { id: string; organizationId: string }; user: { id: string }; employee: { id: string } },
   101→   101→): Promise<{
   102→   102→  success: boolean
   103→   103→  results: Array<{ orderItemId: string; outcome: 'reserved' | 'backordered' | 'failed'; reason?: string }>
   104→   104→}> {
   105→   105→  const order = await db.order.findUnique({
   106→   106→    where: { id: orderId },
   107→   107→    select: { dispatchLocationId: true, companyId: true, organizationId: true },
   108→   108→  })
   109→   109→  if (!order) return { success: false, results: [] }
   110→   110→
   111→   111→  const items = await db.orderItem.findMany({
   112→   112→    where: { orderId },
   113→   113→    include: {
   114→   114→      orgVariant: {
   115→   115→        select: { id: true, fulfillmentType: true, inventoryPolicy: true, stitchingCharges: true },
   116→   116→      },
   117→   117→    },
   118→   118→  })
   119→   119→
   120→   120→  const results: Array<{ orderItemId: string; outcome: 'reserved' | 'backordered' | 'failed'; reason?: string }> = []
   121→   121→  let hasBackordered = false
   122→   122→
   123→   123→  for (const item of items) {
   124→   124→    if (item.fulfillmentStatus === 'reserved' || item.fulfillmentStatus === 'dispatched') {
   125→   125→      // Already processed (e.g. re-confirmation attempt) — skip
   126→   126→      results.push({ orderItemId: item.id, outcome: 'reserved' })
   127→   127→      continue
   128→   128→    }
   129→   129→
   130→   130→    const locationId = item.reservedLocationId ?? order.dispatchLocationId
   131→   131→    if (!locationId) {
   132→   132→      results.push({ orderItemId: item.id, outcome: 'failed', reason: 'No dispatch location set' })
   133→   133→      continue
   134→   134→    }
   135→   135→
   136→   136→    if (item.fulfillmentTypeSnapshot === 'stock_based') {
   137→   137→      // Check available stock at the dispatch location
   138→   138→      const pool = await db.inventoryPool.findUnique({
   139→   139→        where: {
   140→   140→          orgVariantId_locationId: {
   141→   141→            orgVariantId: item.orgVariantId,
   142→   142→            locationId,
   143→   143→          },
   144→   144→        },
   145→   145→      })
   146→   146→
   147→   147→      const available = pool ? pool.onHand - pool.reserved : 0
   148→   148→
   149→   149→      if (available >= item.quantity) {
   150→   150→        // Sufficient stock — reserve it
   151→   151→        const reserveResult = await reserveStockForOrder({
   152→   152→          orgVariantId: item.orgVariantId,
   153→   153→          locationId,
   154→   154→          organizationId: order.organizationId,
   155→   155→          companyId: order.companyId,
   156→   156→          employeeId: ctx.employee.id,
   157→   157→          quantity: item.quantity,
   158→   158→          orderId,
   159→   159→        })
   160→   160→
   161→   161→        if (reserveResult.success) {
   162→   162→          await db.orderItem.update({
   163→   163→            where: { id: item.id },
   164→   164→            data: { fulfillmentStatus: 'reserved', reservedLocationId: locationId },
   165→   165→          })
   166→   166→          results.push({ orderItemId: item.id, outcome: 'reserved' })
   167→   167→        } else {
   168→   168→          results.push({ orderItemId: item.id, outcome: 'failed', reason: reserveResult.error })
   169→   169→        }
   170→   170→      } else if (item.orgVariant.inventoryPolicy === 'continue') {
   171→   171→        // Insufficient stock but backordering allowed
   172→   172→        await db.orderItem.update({
   173→   173→          where: { id: item.id },
   174→   174→          data: { fulfillmentStatus: 'backordered', backorderedAt: new Date() },
   175→   175→        })
   176→   176→        await insertMetricEvent({
   177→   177→          companyId: order.companyId,
   178→   178→          entityType: 'product',
   179→   179→          entityId: item.orgVariantId,
   180→   180→          metricKey: 'order.backordered',
   181→   181→          numericValue: item.quantity,
   182→   182→          dimensions: { order_id: orderId },
   183→   183→        }).catch(() => {})
   184→   184→        hasBackordered = true
   185→   185→        results.push({
   186→   186→          orderItemId: item.id,
   187→   187→          outcome: 'backordered',
   188→   188→          reason: `Available: ${available}, required: ${item.quantity}`,
   189→   189→        })
   190→   190→      } else {
   191→   191→        // Insufficient stock + policy='deny' — fail
   192→   192→        results.push({
   193→   193→          orderItemId: item.id,
   194→   194→          outcome: 'failed',
   195→   195→          reason: `Insufficient stock (available: ${available}, required: ${item.quantity}) and inventory_policy='deny'`,
   196→   196→        })
   197→   197→      }
   198→   198→    } else if (item.fulfillmentTypeSnapshot === 'made_to_order') {
   199→   199→      // Made-to-order: check returned stock first, then trigger production
   200→   200→      const mtoResult = await checkAndFulfillMadeToOrderVariant(
   201→   201→        item.orgVariantId,
   202→   202→        item.quantity,
   203→   203→        order.companyId,
   204→   204→        locationId,
   205→   205→      )
   206→   206→
   207→   207→      if (mtoResult.source === 'existing_stock' && mtoResult.locationId) {
   208→   208→        // Returned stock available — reserve it
   209→   209→        const reserveResult = await reserveStockForOrder({
   210→   210→          orgVariantId: item.orgVariantId,
   211→   211→          locationId: mtoResult.locationId,
   212→   212→          organizationId: order.organizationId,
   213→   213→          companyId: order.companyId,
   214→   214→          employeeId: ctx.employee.id,
   215→   215→          quantity: item.quantity,
   216→   216→          orderId,
   217→   217→        })
   218→   218→
   219→   219→        if (reserveResult.success) {
   220→   220→          await db.orderItem.update({
   221→   221→            where: { id: item.id },
   222→   222→            data: {
   223→   223→              fulfillmentStatus: 'reserved',
   224→   224→              returnedStitchedUsed: true,
   225→   225→              reservedLocationId: mtoResult.locationId,
   226→   226→            },
   227→   227→          })
   228→   228→          await insertMetricEvent({
   229→   229→            companyId: order.companyId,
   230→   230→            entityType: 'product',
   231→   231→            entityId: item.orgVariantId,
   232→   232→            metricKey: 'order.made_to_order_from_returned_stock',
   233→   233→            numericValue: 1,
   234→   234→            dimensions: { order_id: orderId, stitching_cost_saved: Number(item.orgVariant.stitchingCharges) || 0 },
   235→   235→          }).catch(() => {})
   236→   236→          results.push({ orderItemId: item.id, outcome: 'reserved' })
   237→   237→        } else {
   238→   238→          results.push({ orderItemId: item.id, outcome: 'failed', reason: reserveResult.error })
   239→   239→        }
   240→   240→      } else if (mtoResult.source === 'fresh_production' && mtoResult.productionOrderId) {
   241→   241→        // Fresh production triggered — link the production order
   242→   242→        await db.orderItem.update({
   243→   243→          where: { id: item.id },
   244→   244→          data: {
   245→   245→            fulfillmentStatus: 'reserved',
   246→   246→            productionOrderId: mtoResult.productionOrderId,
   247→   247→          },
   248→   248→        })
   249→   249→        // Also link the order_item back to the production order
   250→   250→        await db.productionOrder.update({
   251→   251→          where: { id: mtoResult.productionOrderId },
   252→   252→          data: { orderItemId: item.id },
   253→   253→        })
   254→   254→        await insertMetricEvent({
   255→   255→          companyId: order.companyId,
   256→   256→          entityType: 'product',
   257→   257→          entityId: item.orgVariantId,
   258→   258→          metricKey: 'order.made_to_order_production_triggered',
   259→   259→          numericValue: 1,
   260→   260→          dimensions: { order_id: orderId, production_order_id: mtoResult.productionOrderId },
   261→   261→        }).catch(() => {})
   262→   262→        results.push({ orderItemId: item.id, outcome: 'reserved' })
   263→   263→      } else {
   264→   264→        // Error (no fabric source, insufficient fabric, etc.)
   265→   265→        results.push({
   266→   266→          orderItemId: item.id,
   267→   267→          outcome: 'failed',
   268→   268→          reason: mtoResult.error ?? 'Made-to-order fulfillment failed',
   269→   269→        })
   270→   270→      }
   271→   271→    }
   272→   272→  }
   273→   273→
   274→   274→  // Update order status based on results
   275→   275→  if (hasBackordered) {
   276→   276→    await db.order.update({
   277→   277→      where: { id: orderId },
   278→   278→      data: { status: 'partially_backordered' },
   279→   279→    })
   280→   280→  }
   281→   281→  // If no backordered items, order stays 'confirmed' (or whatever it was)
   282→   282→
   283→   283→  return { success: true, results }
   284→   284→}

   288→   288→// ──────────────────────────────────────────────────────────────
   289→   289→
   290→   290→export async function createManualOrder(
   291→   291→  input: CreateManualOrderInput,
   292→   292→): Promise<ActionResult<{
   293→   293→  orderId: string
   294→   294→  flowopsOrderNumber: string
   295→   295→  orderItems: Array<{ id: string; orgVariantId: string; quantity: number }>
   296→   296→}>> {
   297→   297→  try {
   298→   298→    const ctx = await getWorkspace()
   299→   299→    await requirePermission(ctx, PERMISSIONS.ORDERS_CREATE)
   300→   300→
   301→   301→    // 1. Validate
   302→   302→    const parsed = createManualOrderSchema.safeParse(input)
   303→   303→    if (!parsed.success) {
   304→   304→      return {
   305→   305→        success: false,
   306→   306→        error: parsed.error.issues[0]?.message ?? 'Invalid order data',
   307→   307→      }
   308→   308→    }
   309→   309→    const d = parsed.data
   310→   310→
   311→   311→    // 2. Find or create customer
   312→   312→    let customerId: string
   313→   313→    if (d.customer_id) {
   314→   314→      // Verify existing customer belongs to this org
   315→   315→      const existing = await db.customer.findFirst({
   316→   316→        where: { id: d.customer_id, organizationId: ctx.company.organizationId },
   317→   317→      })
   318→   318→      if (!existing) return { success: false, error: 'Customer not found' }
   319→   319→      customerId = existing.id
   320→   320→    } else if (d.customer) {
   321→   321→      const customerResult = await findOrCreateCustomer({
   322→   322→        ...d.customer,
   323→   323→        organizationId: ctx.company.organizationId,
   324→   324→        companyId: ctx.company.id,
   325→   325→      })
   326→   326→      if (!customerResult.success || !customerResult.data) {
   327→   327→        return { success: false, error: customerResult.error ?? 'Failed to create customer' }
   328→   328→      }
   329→   329→      customerId = customerResult.data.customerId
   330→   330→    } else {
   331→   331→      return { success: false, error: 'Either customer or customer_id is required' }
   332→   332→    }
   333→   333→
   334→   334→    // 3. Fetch variants + their pricing for this company + fulfillment_type snapshot
   335→   335→    const variantIds = d.items.map((i) => i.org_variant_id)
   336→   336→    const variants = await db.orgProductVariant.findMany({
   337→   337→      where: { id: { in: variantIds }, organizationId: ctx.company.organizationId },
   338→   338→      include: {
   339→   339→        companyPricing: { where: { companyId: ctx.company.id } },
   340→   340→      },
   341→   341→    })
   342→   342→
   343→   343→    if (variants.length !== variantIds.length) {
   344→   344→      return { success: false, error: 'One or more variants not found in this organization' }
   345→   345→    }
   346→   346→
   347→   347→    // 4. Compute subtotal + build order items data
   348→   348→    let subtotal = 0
   349→   349→    const orderItemsData: Array<{
   350→   350→      orgVariantId: string
   351→   351→      quantity: number
   352→   352→      unitPrice: number
   353→   353→      lineTotal: number
   354→   354→      fulfillmentTypeSnapshot: string
   355→   355→    }> = []
   356→   356→
   357→   357→    for (const item of d.items) {
   358→   358→      const variant = variants.find((v) => v.id === item.org_variant_id)
   359→   359→      if (!variant) continue
   360→   360→
   361→   361→      // Use provided unit_price OR fall back to company pricing OR variant cost
   362→   362→      const unitPrice =
   363→   363→        item.unit_price ??
   364→   364→        (variant.companyPricing[0]?.salePrice
   365→   365→          ? Number(variant.companyPricing[0].salePrice)
   366→   366→          : Number(variant.costPrice))
   367→   367→
   368→   368→      const lineTotal = item.quantity * unitPrice
   369→   369→      subtotal += lineTotal
   370→   370→
   371→   371→      orderItemsData.push({
   372→   372→        orgVariantId: item.org_variant_id,
   373→   373→        quantity: item.quantity,
   374→   374→        unitPrice,
   375→   375→        lineTotal,
   376→   376→        fulfillmentTypeSnapshot: variant.fulfillmentType,
   377→   377→      })
   378→   378→    }
   379→   379→
   380→   380→    // 5. Compute totals
   381→   381→    const discountAmount = d.discount_amount ?? 0
   382→   382→    const courierCharges = d.courier_charges ?? 0
   383→   383→    const totalOrderValue = subtotal + courierCharges - discountAmount
   384→   384→
   385→   385→    // 6. Determine payment status + source
   386→   386→    let paymentStatus: string = 'cod_pending'
   387→   387→    let paymentSource: string = 'cod_native'
   388→   388→    let advanceAmount: number | null = null
   389→   389→    let advancePaymentMethod: string | null = null
   390→   390→    let advancePaymentReference: string | null = null
   391→   391→    let advancePaidAt: Date | null = null
   392→   392→
   393→   393→    if (d.payment_type === 'partial_advance') {
   394→   394→      paymentStatus = 'advance_paid'
   395→   395→      paymentSource = 'manual_conversion'
   396→   396→      advanceAmount = d.advance_amount ?? 0
   397→   397→      advancePaymentMethod = d.advance_payment_method || null
   398→   398→      advancePaymentReference = d.advance_payment_reference || null
   399→   399→      advancePaidAt = new Date()
   400→   400→    } else if (d.payment_type === 'fully_prepaid') {
   401→   401→      paymentStatus = 'fully_prepaid'
   402→   402→      paymentSource = 'manual_conversion'
   403→   403→      advanceAmount = totalOrderValue
   404→   404→      advancePaymentMethod = d.advance_payment_method || null
   405→   405→      advancePaymentReference = d.advance_payment_reference || null
   406→   406→      advancePaidAt = new Date()
   407→   407→    }
   408→   408→
   409→   409→    // 7. Determine initial order status
   410→   410→    let orderStatus: string = 'pending'
   411→   411→    let confirmedAt: Date | null = null
   412→   412→    let skippedConfirmation = false
   413→   413→
   414→   414→    if (paymentStatus === 'advance_paid' || paymentStatus === 'fully_prepaid') {
   415→   415→      // Payment itself is the confirmation signal — bypasses require_order_confirmation
   416→   416→      orderStatus = 'confirmed'
   417→   417→      confirmedAt = new Date()
   418→   418→    } else {
   419→   419→      // COD order — check company_order_settings
   420→   420→      const settings = await db.companyOrderSetting.findUnique({
   421→   421→        where: { companyId: ctx.company.id },
   422→   422→      })
   423→   423→      if (!settings?.requireOrderConfirmation) {
   424→   424→        orderStatus = 'confirmed'
   425→   425→        confirmedAt = new Date()
   426→   426→        skippedConfirmation = true
   427→   427→      }
   428→   428→    }
   429→   429→
   430→   430→    // 8. Generate order number
   431→   431→    const flowopsOrderNumber = await generateOrderNumber(ctx.company.id)
   432→   432→
   433→   433→    // 9. Create the order
   434→   434→    const order = await db.order.create({
   435→   435→      data: {
   436→   436→        organizationId: ctx.company.organizationId,
   437→   437→        companyId: ctx.company.id,
   438→   438→        flowopsOrderNumber,
   439→   439→        orderSource: 'manual',
   440→   440→        customerId,
   441→   441→        status: orderStatus,
   442→   442→        paymentType: d.payment_type,
   443→   443→        paymentStatus,
   444→   444→        paymentSource,
   445→   445→        subtotal,
   446→   446→        discountAmount: discountAmount || null,
   447→   447→        discountReason: d.discount_reason || null,
   448→   448→        courierCharges: courierCharges || null,
   449→   449→        totalOrderValue,
   450→   450→        advanceAmount,
   451→   451→        advancePaymentMethod,
   452→   452→        advancePaymentReference,
   453→   453→        advancePaidAt,
   454→   454→        deliveryAddress: d.delivery_address,
   455→   455→        deliveryCity: d.delivery_city,
   456→   456→        courierName: d.courier_name || null,
   457→   457→        dispatchLocationId: d.dispatch_location_id,
   458→   458→        notesForCourier: d.notes_for_courier || null,
   459→   459→        skippedConfirmation,
   460→   460→        confirmedAt,
   461→   461→        createdBy: ctx.employee.id,
   462→   462→      },
   463→   463→    })
   464→   464→
   465→   465→    // 10. Create order items (fulfillment_status = 'reserved' as PLACEHOLDER)
   466→   466→    const createdItems: Array<{ id: string; orgVariantId: string; quantity: number }> = []
   467→   467→    for (const itemData of orderItemsData) {
   468→   468→      const item = await db.orderItem.create({
   469→   469→        data: {
   470→   470→          orderId: order.id,
   471→   471→          orgVariantId: itemData.orgVariantId,
   472→   472→          organizationId: ctx.company.organizationId,
   473→   473→          quantity: itemData.quantity,
   474→   474→          unitPrice: itemData.unitPrice,
   475→   475→          lineTotal: itemData.lineTotal,
   476→   476→          fulfillmentStatus: 'reserved', // PLACEHOLDER — Step 3 will validate/adjust
   477→   477→          fulfillmentTypeSnapshot: itemData.fulfillmentTypeSnapshot,
   478→   478→          reservedLocationId: d.dispatch_location_id,
   479→   479→        },
   480→   480→      })
   481→   481→      createdItems.push({
   482→   482→        id: item.id,
   483→   483→        orgVariantId: item.orgVariantId,
   484→   484→        quantity: item.quantity,
   485→   485→      })
   486→   486→    }

   732→   732→export async function confirmOrder(orderId: string): Promise<ActionResult> {
   733→   733→  try {
   734→   734→    const ctx = await getWorkspace()
   735→   735→    await requirePermission(ctx, PERMISSIONS.ORDERS_MANAGE)
   736→   736→
   737→   737→    const order = await db.order.findFirst({
   738→   738→      where: { id: orderId, companyId: ctx.company.id },
   739→   739→    })
   740→   740→    if (!order) return { success: false, error: 'Order not found' }
   741→   741→    if (order.status !== 'pending') {
   742→   742→      return { success: false, error: `Order is already ${order.status} (cannot confirm)` }
   743→   743→    }
   744→   744→
   745→   745→    await db.order.update({
   746→   746→      where: { id: orderId },
   747→   747→      data: { status: 'confirmed', confirmedAt: new Date() },
   748→   748→    })
   749→   749→
   750→   750→    await insertAuditLog({
   751→   751→      action: 'order.confirmed',
   752→   752→      entityType: 'order',
   753→   753→      entityId: orderId,
   754→   754→      companyId: ctx.company.id,
   755→   755→      organizationId: ctx.company.organizationId,
   756→   756→      userId: ctx.user.id,
   757→   757→      employeeId: ctx.employee.id,
   758→   758→      oldValues: { status: 'pending' },
   759→   759→      newValues: { status: 'confirmed' },
   760→   760→    })
   761→   761→
   762→   762→    // Step 3: Run stock reservation for all items.
   763→   763→    // This may flip the order to 'partially_backordered' if some items
   764→   764→    // can't be fulfilled immediately.
   765→   765→    const reserveResult = await reserveOrderStock(orderId, ctx)
   766→   766→    if (!reserveResult.success) {
   767→   767→      // Non-fatal — the order is confirmed, but reservation had issues.
   768→   768→      // The results array has per-item details.
   769→   769→    }
   770→   770→
   771→   771→    await insertMetricEvent({
   772→   772→      companyId: ctx.company.id,
   773→   773→      entityType: 'order',
   774→   774→      entityId: orderId,
   775→   775→      metricKey: 'order.confirmed',
   776→   776→      numericValue: Number(order.totalOrderValue),
   777→   777→      dimensions: { confirmation_method: 'manual' },
   778→   778→    }).catch(() => {})
   779→   779→
   780→   780→    return { success: true }
   781→   781→  } catch (err) {
   782→   782→    return {
   783→   783→      success: false,
   784→   784→      error: err instanceof Error ? err.message : 'Failed to confirm order',
   785→   785→    }
   786→   786→  }
   787→   787→}

  1238→  1238→// Step 3: Dispatch + Processing/Packing actions
  1239→  1239→// ──────────────────────────────────────────────────────────────
  1240→  1240→
  1241→  1241→/**
  1242→  1242→ * Dispatch an order — deducts stock from inventory, sets tracking info.
  1243→  1243→ *
  1244→  1244→ * Business rules:
  1245→  1245→ *   - If order.status = 'pending' (confirmation was skipped): run the
  1246→  1246→ *     full confirmation/reservation logic inline first, then proceed.
  1247→  1247→ *   - If any item is still 'backordered' after reservation: BLOCK dispatch
  1248→  1248→ *     with a clear error (hard rule — no split shipments).
  1249→  1249→ *   - For each reserved item: call dispatchOrder() (inventory) which
  1250→  1250→ *     deducts on_hand, releases the reservation, and locks COGS.
  1251→  1251→ *   - The backfill_order_timestamps() trigger auto-sets confirmed_at/
  1252→  1252→ *     packed_at if they were still NULL.
  1253→  1253→ *   - If company_order_settings.require_packing_step = TRUE, verify
  1254→  1254→ *     order.packed_at IS NOT NULL before proceeding.
  1255→  1255→ */
  1256→  1256→export async function dispatchOrderAction(
  1257→  1257→  orderId: string,
  1258→  1258→  trackingNumber: string,
  1259→  1259→  courierName?: string,
  1260→  1260→): Promise<ActionResult> {
  1261→  1261→  try {
  1262→  1262→    const ctx = await getWorkspace()
  1263→  1263→    await requirePermission(ctx, PERMISSIONS.ORDERS_FULFILL)
  1264→  1264→
  1265→  1265→    const order = await db.order.findFirst({
  1266→  1266→      where: { id: orderId, companyId: ctx.company.id },
  1267→  1267→    })
  1268→  1268→    if (!order) return { success: false, error: 'Order not found' }
  1269→  1269→
  1270→  1270→    // Cannot dispatch already-dispatched/delivered/cancelled orders
  1271→  1271→    if (['dispatched', 'delivered', 'rto', 'cancelled', 'refunded'].includes(order.status)) {
  1272→  1272→      return { success: false, error: `Cannot dispatch an order that is already ${order.status}` }
  1273→  1273→    }
  1274→  1274→
  1275→  1275→    // Check packing requirement
  1276→  1276→    const settings = await db.companyOrderSetting.findUnique({
  1277→  1277→      where: { companyId: ctx.company.id },
  1278→  1278→    })
  1279→  1279→    if (settings?.requirePackingStep && !order.packedAt) {
  1280→  1280→      return {
  1281→  1281→        success: false,
  1282→  1282→        error: 'This order must be marked as packed before dispatching. Use markOrderPacked() first.',
  1283→  1283→      }
  1284→  1284→    }
  1285→  1285→
  1286→  1286→    // If order is still 'pending' (confirmation was skipped), run the
  1287→  1287→    // full confirmation + reservation logic inline first.
  1288→  1288→    if (order.status === 'pending') {
  1289→  1289→      await db.order.update({
  1290→  1290→        where: { id: orderId },
  1291→  1291→        data: { status: 'confirmed', confirmedAt: new Date() },
  1292→  1292→      })
  1293→  1293→      await reserveOrderStock(orderId, ctx)
  1294→  1294→    }
  1295→  1295→
  1296→  1296→    // Re-fetch the order to get the updated status after reservation
  1297→  1297→    const updatedOrder = await db.order.findFirst({
  1298→  1298→      where: { id: orderId },
  1299→  1299→      select: { status: true, dispatchLocationId: true, organizationId: true },
  1300→  1300→    })
  1301→  1301→    if (!updatedOrder) return { success: false, error: 'Order not found after reservation' }
  1302→  1302→
  1303→  1303→    // Check for backordered items — BLOCK dispatch if any exist
  1304→  1304→    const backorderedItems = await db.orderItem.findMany({
  1305→  1305→      where: { orderId, fulfillmentStatus: 'backordered' },
  1306→  1306→      include: {
  1307→  1307→        orgVariant: { select: { sku: true } },
  1308→  1308→      },
  1309→  1309→    })
  1310→  1310→
  1311→  1311→    if (backorderedItems.length > 0) {
  1312→  1312→      const itemList = backorderedItems
  1313→  1313→        .map((i) => `${i.orgVariant.sku} (qty: ${i.quantity})`)
  1314→  1314→        .join(', ')
  1315→  1315→      return {
  1316→  1316→        success: false,
  1317→  1317→        error: `Cannot dispatch: ${backorderedItems.length} item(s) are still backordered: ${itemList}. Receive stock or cancel those items first.`,
  1318→  1318→      }
  1319→  1319→    }
  1320→  1320→
  1321→  1321→    // Dispatch each reserved item
  1322→  1322→    const itemsToDispatch = await db.orderItem.findMany({
  1323→  1323→      where: { orderId, fulfillmentStatus: 'reserved' },
  1324→  1324→    })
  1325→  1325→
  1326→  1326→    for (const item of itemsToDispatch) {
  1327→  1327→      const locationId = item.reservedLocationId ?? updatedOrder.dispatchLocationId
  1328→  1328→      if (!locationId) {
  1329→  1329→        return { success: false, error: `Order item ${item.id} has no dispatch location` }
  1330→  1330→      }
  1331→  1331→
  1332→  1332→      const dispatchResult = await dispatchInventory({
  1333→  1333→        orgVariantId: item.orgVariantId,
  1334→  1334→        locationId,
  1335→  1335→        organizationId: updatedOrder.organizationId,
  1336→  1336→        companyId: ctx.company.id,
  1337→  1337→        employeeId: ctx.employee.id,
  1338→  1338→        quantity: item.quantity,
  1339→  1339→        orderId,
  1340→  1340→      })
  1341→  1341→
  1342→  1342→      if (!dispatchResult.success) {
  1343→  1343→        return {
  1344→  1344→          success: false,
  1345→  1345→          error: `Failed to dispatch item: ${dispatchResult.error}`,
  1346→  1346→        }
  1347→  1347→      }
  1348→  1348→
  1349→  1349→      // Mark item as dispatched
  1350→  1350→      await db.orderItem.update({
  1351→  1351→        where: { id: item.id },
  1352→  1352→        data: { fulfillmentStatus: 'dispatched', fulfilledAt: new Date() },
  1353→  1353→      })
  1354→  1354→    }
  1355→  1355→
  1356→  1356→    // Update order status — the backfill_order_timestamps() trigger
  1357→  1357→    // will auto-set confirmedAt/packedAt if they were still NULL.
  1358→  1358→    await db.order.update({
  1359→  1359→      where: { id: orderId },
  1360→  1360→      data: {
  1361→  1361→        status: 'dispatched',
  1362→  1362→        dispatchedAt: new Date(),
  1363→  1363→        trackingNumber: trackingNumber || null,
  1364→  1364→        courierName: courierName || order.courierName,
  1365→  1365→      },
  1366→  1366→    })
  1367→  1367→
  1368→  1368→    await insertAuditLog({
  1369→  1369→      action: 'order.dispatched',
  1370→  1370→      entityType: 'order',
  1371→  1371→      entityId: orderId,
  1372→  1372→      companyId: ctx.company.id,
  1373→  1373→      organizationId: ctx.company.organizationId,
  1374→  1374→      userId: ctx.user.id,
  1375→  1375→      employeeId: ctx.employee.id,
  1376→  1376→      oldValues: { status: updatedOrder.status },
  1377→  1377→      newValues: {
  1378→  1378→        status: 'dispatched',
  1379→  1379→        trackingNumber,
  1380→  1380→        courierName,
  1381→  1381→        itemsDispatched: itemsToDispatch.length,
  1382→  1382→      },
  1383→  1383→    })
  1384→  1384→
  1385→  1385→    const timeToDispatchHours = order.createdAt
  1386→  1386→      ? Math.round((new Date().getTime() - new Date(order.createdAt).getTime()) / (1000 * 60 * 60))
  1387→  1387→      : 0
  1388→  1388→    await insertMetricEvent({
  1389→  1389→      companyId: ctx.company.id,
  1390→  1390→      entityType: 'order',
  1391→  1391→      entityId: orderId,
  1392→  1392→      metricKey: 'order.dispatched',
  1393→  1393→      numericValue: Number(order.totalOrderValue),
  1394→  1394→      dimensions: {
  1395→  1395→        courier_name: courierName || order.courierName,
  1396→  1396→        employee_id: ctx.employee.id,
  1397→  1397→        skipped_confirmation: updatedOrder.status === 'pending',
  1398→  1398→        skipped_packing: !order.packedAt,
  1399→  1399→        time_to_dispatch_hours: timeToDispatchHours,
  1400→  1400→      },
  1401→  1401→    }).catch(() => {})
  1402→  1402→
  1403→  1403→    // Update customer stats
  1404→  1404→    const orderWithCustomer = await db.order.findUnique({
  1405→  1405→      where: { id: orderId },
  1406→  1406→      select: { customerId: true },
  1407→  1407→    })
  1408→  1408→    if (orderWithCustomer) {
  1409→  1409→      await updateCustomerStats(orderWithCustomer.customerId)
  1410→  1410→    }
  1411→  1411→
  1412→  1412→    return { success: true }
  1413→  1413→  } catch (err) {
  1414→  1414→    return {
  1415→  1415→      success: false,
  1416→  1416→      error: err instanceof Error ? err.message : 'Failed to dispatch order',
  1417→  1417→    }
  1418→  1418→  }
  1419→  1419→}
  1420→  1420→
  1421→  1421→/**
  1422→  1422→ * Mark an order as "processing" (being packed). Only meaningful when
  1423→  1423→ * company_order_settings.require_packing_step = TRUE.
  1424→  1424→ */
  1425→  1425→export async function markOrderProcessing(orderId: string): Promise<ActionResult> {
  1426→  1426→  try {
  1427→  1427→    const ctx = await getWorkspace()
  1428→  1428→    await requirePermission(ctx, PERMISSIONS.ORDERS_FULFILL)
  1429→  1429→
  1430→  1430→    const order = await db.order.findFirst({
