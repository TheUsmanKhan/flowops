   222→   222→function toISODate(d: Date): string {
   223→   223→  const y = d.getFullYear()
   224→   224→  const m = String(d.getMonth() + 1).padStart(2, '0')
   225→   225→  const day = String(d.getDate()).padStart(2, '0')
   226→   226→  return `${y}-${m}-${day}`
   227→   227→}
   228→   228→
   229→   229→function applyPreset(preset: DatePresetValue): { from: string; to: string } {
   230→   230→  const now = new Date()
   231→   231→  const to = new Date(now)
   232→   232→  const from = new Date(now)
   233→   233→  if (preset === 'today') {
   234→   234→    // from = today
   235→   235→  } else if (preset === '7d') {
   236→   236→    from.setDate(now.getDate() - 6)
   237→   237→  } else if (preset === '30d') {
   238→   238→    from.setDate(now.getDate() - 29)
   239→   239→  } else if (preset === 'this_month') {
   240→   240→    from.setDate(1)
   241→   241→  }
   242→   242→  return { from: toISODate(from), to: toISODate(to) }
   243→   243→}

   281→   281→function toggleArray(arr: string[], value: string): string[] {
   282→   282→  return arr.includes(value) ? arr.filter((v) => v !== value) : [...arr, value]
   283→   283→}

   285→   285→function countActiveFilters(f: FilterState): number {
   286→   286→  let n = 0
   287→   287→  if (f.statuses.length) n++
   288→   288→  if (f.paymentTypes.length) n++
   289→   289→  if (f.paymentStatuses.length) n++
   290→   290→  if (f.orderSources.length) n++
   291→   291→  if (f.dateFrom || f.dateTo) n++
   292→   292→  if (f.amountMin || f.amountMax) n++
   293→   293→  if (f.orgVariantId) n++
   294→   294→  if (f.customerId) n++
   295→   295→  if (f.courier.trim()) n++
   296→   296→  return n
   297→   297→}

   326→   326→    if (filters.orgVariantId) params.set('orgVariantId', filters.orgVariantId)
   327→   327→    if (filters.customerId) params.set('customer_id', filters.customerId)

   329→   329→    if (search.trim()) params.set('search', search.trim())

   841→   841→function FiltersSheet({
   842→   842→  filters,

   848→   848→  onClose: () => void
   849→   849→}) {
   850→   850→  // Local draft so the user can apply all changes at once on "Apply".
   851→   851→  const [draft, setDraft] = useState<FilterState>(filters)
   852→   852→  useEffect(() => {
   853→   853→    setDraft(filters)
   854→   854→  }, [filters])
   855→   855→
   856→   856→  // Products for variant picker (lazy-loaded once when sheet opens).
   857→   857→  const productsQuery = useQuery<ProductsResponse>({
   858→   858→    queryKey: ['products', 'filter-picker'],
   859→   859→    queryFn: () => api.get<ProductsResponse>('/api/products?pageSize=200'),
   860→   860→    staleTime: 60_000,
   861→   861→  })
   862→   862→
   863→   863→  const variantOptions: ProductOption[] = useMemo(() => {
   864→   864→    const list: ProductOption[] = []
   865→   865→    for (const p of productsQuery.data?.products ?? []) {
   866→   866→      for (const v of p.variants) {
   867→   867→        list.push({ productId: p.id, variantId: v.id, sku: v.sku, title: p.title })
   868→   868→      }
   869→   869→    }
   870→   870→    return list
   871→   871→  }, [productsQuery.data])
   872→   872→
   873→   873→  const [variantSearch, setVariantSearch] = useState('')
   874→   874→  const variantResults = useMemo(() => {
   875→   875→    if (!variantSearch.trim()) return variantOptions.slice(0, 50)
   876→   876→    const q = variantSearch.toLowerCase()
   877→   877→    return variantOptions
   878→   878→      .filter((v) => v.sku.toLowerCase().includes(q) || v.title.toLowerCase().includes(q))
   879→   879→      .slice(0, 50)
   880→   880→  }, [variantOptions, variantSearch])
   881→   881→
   882→   882→  // Customer search (debounced via query key + min length gate).
   883→   883→  const [customerSearch, setCustomerSearch] = useState(draft.customerLabel)
   884→   884→  const trimmedCustomer = customerSearch.trim()
   885→   885→  const customersQuery = useQuery<CustomersSearchResponse>({
   886→   886→    queryKey: ['customers', 'filter', trimmedCustomer],
   887→   887→    queryFn: () =>
   888→   888→      api.get<CustomersSearchResponse>(
   889→   889→        `/api/customers?search=${encodeURIComponent(trimmedCustomer)}&limit=10`,
   890→   890→      ),
   891→   891→    enabled: trimmedCustomer.length >= 3,
   892→   892→    staleTime: 10_000,
   893→   893→  })
   894→   894→
   895→   895→  const patch = (p: Partial<FilterState>) => setDraft((prev) => ({ ...prev, ...p }))
   896→   896→
   897→   897→  return (
   898→   898→    <SheetContent side="right" className="w-full sm:max-w-md overflow-y-auto">
   899→   899→      <SheetHeader>
   900→   900→        <SheetTitle className="flex items-center gap-2">
   901→   901→          <SlidersHorizontal className="h-4 w-4" />
   902→   902→          Filters
   903→   903→        </SheetTitle>
   904→   904→        <SheetDescription>
   905→   905→          Refine the order list by status, payment, date, amount and more.
   906→   906→        </SheetDescription>
   907→   907→      </SheetHeader>
   908→   908→
   909→   909→      <div className="flex-1 space-y-6 px-4 pb-2">
   910→   910→        {/* Status */}
   911→   911→        <FilterSection title="Status">
   912→   912→          <div className="grid grid-cols-1 gap-2">
   913→   913→            {STATUS_OPTIONS.map((opt) => (
   914→   914→              <label
   915→   915→                key={opt.value}
   916→   916→                className="flex items-center gap-2 text-sm cursor-pointer hover:bg-muted/40 rounded px-1 py-0.5"
   917→   917→              >
   918→   918→                <Checkbox
   919→   919→                  checked={draft.statuses.includes(opt.value)}
   920→   920→                  onCheckedChange={() =>
   921→   921→                    patch({ statuses: toggleArray(draft.statuses, opt.value) })
   922→   922→                  }
   923→   923→                />
   924→   924→                <span>{opt.label}</span>
   925→   925→              </label>
   926→   926→            ))}
   927→   927→          </div>
   928→   928→        </FilterSection>
   929→   929→
   930→   930→        {/* Payment Type */}
   931→   931→        <FilterSection title="Payment Type">
   932→   932→          <div className="grid grid-cols-1 gap-2">
   933→   933→            {PAYMENT_TYPE_OPTIONS.map((opt) => (
   934→   934→              <label
   935→   935→                key={opt.value}
   936→   936→                className="flex items-center gap-2 text-sm cursor-pointer hover:bg-muted/40 rounded px-1 py-0.5"
