   133→function subscriberStatusMeta(status: string): {
   134→  label: string
   135→  className: string
   136→} {
   137→  switch (status) {
   138→    case 'active':
   139→      return {
   140→        label: 'Active',

   142→      }

   144→      return {
   145→        label: 'Paused',
   146→        className: 'bg-amber-50 text-amber-700 border-amber-200',
   147→      }

   154→      return {
   155→        label: status,
   156→        className: 'bg-muted text-muted-foreground border-border',
   157→      }
   158→  }

   274→function DemoteDialog({
   275→  product,
   276→  open,
   277→  onOpenChange,
   278→}: {
   279→  product: SharedProduct | null
   280→  open: boolean
   281→  onOpenChange: (v: boolean) => void
   282→}) {
   283→  const qc = useQueryClient()
   284→  const [newScope, setNewScope] = useState<'private' | 'selective'>('private')
   285→  const [reason, setReason] = useState('')
   286→  const [warnings, setWarnings] = useState<string[] | null>(null)
   287→
   288→  const mutation = useMutation({
   289→    mutationFn: async () => {
   290→      if (!product) throw new Error('No product')
   291→      return api.post<DemoteResponse>(`/api/products/${product.id}/demote`, {
   292→        new_scope: newScope,
   293→        reason,
   294→      })
   295→    },
   296→    onSuccess: (data) => {
   297→      qc.invalidateQueries({ queryKey: ['org-catalog'] })
   298→      if (data.warnings && data.warnings.length > 0) {
   299→        // Keep dialog open with warnings surfaced as an Alert
   300→        setWarnings(data.warnings)
   301→        toast.warning('Product demoted', {
   302→          description: data.warnings.join(' '),
   303→        })
   304→      } else {
   305→        toast.success('Product demoted', {
   306→          description: `Scope changed to ${newScope}. ${
   307→            data.affected_companies.length > 0
   308→              ? `${data.affected_companies.length} subscription(s) revoked.`
   309→              : ''
   310→          }`,
   311→        })
   312→        handleClose()
   313→      }
   314→    },
   315→    onError: (err: unknown) => {
   316→      const message =
   317→        err instanceof FetchError ? err.message : 'Failed to demote product.'
   318→      toast.error('Demote failed', { description: message })
   319→    },
   320→  })
   321→
   322→  function handleClose() {
   323→    onOpenChange(false)
   324→    setNewScope('private')
   325→    setReason('')
   326→    setWarnings(null)
   327→    mutation.reset()
   328→  }
   329→
   330→  // Reset internal state whenever a new product is targeted
   331→  function handleOpenChange(v: boolean) {
   332→    if (!v) {
   333→      handleClose()
   334→    } else {
   335→      onOpenChange(true)
   336→    }
   337→  }
   338→
   339→  if (!product) return null
   340→
   341→  const canSubmit = reason.trim().length >= 3 && !mutation.isPending && !warnings
   342→
   343→  return (
   344→    <Dialog open={open} onOpenChange={handleOpenChange}>
   345→      <DialogContent className="sm:max-w-[480px]">
   346→        <DialogHeader>
   347→          <DialogTitle className="flex items-center gap-2">
   348→            <ArrowUpRight className="h-4 w-4 rotate-90 text-amber-600" />
   349→            Demote product
   350→          </DialogTitle>
   351→          <DialogDescription>
   352→            Reduce the visibility of <span className="font-medium text-foreground">{product.title}</span>.
   353→            Non-source company subscriptions will be revoked.
   354→          </DialogDescription>
   355→        </DialogHeader>
   356→
   357→        <div className="space-y-4 py-2">
   358→          {/* Current scope */}
   359→          <div className="space-y-1.5">
   360→            <Label className="text-xs text-muted-foreground">Current scope</Label>
   361→            <div className="flex items-center gap-2">
   362→              <ProductScopeBadge scope={product.productScope} />
   363→              <span className="text-xs text-muted-foreground">
   364→                {product.subscriberCount} subscriber{product.subscriberCount === 1 ? '' : 's'}
   365→              </span>
   366→            </div>
   367→          </div>
   368→
   369→          {/* Target scope */}
   370→          <div className="space-y-1.5">
   371→            <Label htmlFor="demote-scope">Target scope</Label>
   372→            <Select
   373→              value={newScope}
   374→              onValueChange={(v) => setNewScope(v as 'private' | 'selective')}
   375→              disabled={!!warnings}
   376→            >
   377→              <SelectTrigger id="demote-scope" className="w-full">
   378→                <SelectValue />
   379→              </SelectTrigger>
   380→              <SelectContent>
   381→                <SelectItem value="private">
   382→                  <span className="flex items-center gap-2">
   383→                    <Lock className="h-3.5 w-3.5" />
   384→                    Private — only source company
   385→                  </span>
   386→                </SelectItem>
   387→                <SelectItem value="selective">
   388→                  <span className="flex items-center gap-2">
   389→                    <Users className="h-3.5 w-3.5" />
   390→                    Selective — chosen companies only
   391→                  </span>
   392→                </SelectItem>
   393→              </SelectContent>
   394→            </Select>
   395→          </div>
   396→
   397→          {/* Reason */}
   398→          <div className="space-y-1.5">
   399→            <Label htmlFor="demote-reason">
   400→              Reason <span className="text-rose-500">*</span>
   401→            </Label>
   402→            <Textarea
   403→              id="demote-reason"
   404→              placeholder="Explain why this product is being demoted (min 3 chars)…"
   405→              value={reason}
   406→              onChange={(e) => setReason(e.target.value)}
   407→              rows={3}
   408→              maxLength={500}
   409→              disabled={!!warnings}
   410→            />
   411→            <p className="text-xs text-muted-foreground">
   412→              {reason.length}/500 — recorded in the audit log.
   413→            </p>
   414→          </div>
   415→
   416→          {/* Warnings Alert (shown after demote if any) */}
   417→          {warnings && warnings.length > 0 && (
   418→            <Alert className="border-amber-200 bg-amber-50 dark:bg-amber-950/40">
   419→              <AlertTriangle className="h-4 w-4 text-amber-600" />
   420→              <AlertDescription>
   421→                <div className="text-sm font-medium text-amber-900 dark:text-amber-200 mb-1">
   422→                  Demotion completed with warnings
   423→                </div>
   424→                <ul className="list-disc pl-4 space-y-0.5 text-amber-800 dark:text-amber-300">
   425→                  {warnings.map((w, i) => (
   426→                    <li key={i}>{w}</li>
   427→                  ))}
   428→                </ul>
   429→              </AlertDescription>
   430→            </Alert>
   431→          )}
   432→        </div>
   433→
   434→        <DialogFooter className="gap-2">
   435→          {warnings ? (
   436→            <Button onClick={handleClose} className="w-full sm:w-auto">
   437→              <Check className="h-4 w-4 mr-1" />
   438→              Acknowledge & close
   439→            </Button>
   440→          ) : (
   441→            <>
   442→              <Button variant="outline" onClick={handleClose} disabled={mutation.isPending}>
   443→                Cancel
   444→              </Button>
   445→              <Button
   446→                onClick={() => mutation.mutate()}
   447→                disabled={!canSubmit}
   448→                className="bg-amber-600 hover:bg-amber-700 text-white"
   449→              >
   450→                {mutation.isPending ? (
   451→                  <>
   452→                    <Loader2 className="h-4 w-4 mr-1 animate-spin" />
   453→                    Demoting…
   454→                  </>
   455→                ) : (
   456→                  <>
   457→                    <ArrowUpRight className="h-4 w-4 mr-1 rotate-90" />
   458→                    Demote product
   459→                  </>
   460→                )}
   461→              </Button>
   462→            </>
   463→          )}
   464→        </DialogFooter>
   465→      </DialogContent>
   466→    </Dialog>
   467→  )
   468→}

   474→function PromoteDialog({
   475→  product,
   476→  companies,
   477→  open,
   478→  onOpenChange,
   479→}: {
   480→  product: PromotableProduct | null
   481→  companies: CompanyRef[]
   482→  open: boolean
   483→  onOpenChange: (v: boolean) => void
   484→}) {
   485→  const qc = useQueryClient()
   486→  const [targetScope, setTargetScope] = useState<'organization' | 'selective'>('organization')
   487→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set())
   488→
   489→  const mutation = useMutation({
   490→    mutationFn: async () => {
   491→      if (!product) throw new Error('No product')
   492→      return api.post<{ success: boolean }>(`/api/products/${product.id}/promote`, {
   493→        target_scope: targetScope,
   494→        selected_company_ids: targetScope === 'selective' ? Array.from(selectedIds) : [],
   495→      })
   496→    },
   497→    onSuccess: () => {
   498→      qc.invalidateQueries({ queryKey: ['org-catalog'] })
   499→      toast.success('Product promoted', {
   500→        description: `Now visible as ${targetScope === 'organization' ? 'Organization-wide' : 'Selective'}.`,
   501→      })
   502→      handleClose()
   503→    },
   504→    onError: (err: unknown) => {
   505→      const message =
   506→        err instanceof FetchError ? err.message : 'Failed to promote product.'
   507→      toast.error('Promote failed', { description: message })
   508→    },
   509→  })
   510→
   511→  function handleClose() {
   512→    onOpenChange(false)
   513→    setTargetScope('organization')
   514→    setSelectedIds(new Set())
   515→    mutation.reset()
   516→  }
   517→
   518→  function handleOpenChange(v: boolean) {
   519→    if (!v) handleClose()
   520→    else onOpenChange(true)
   521→  }
   522→
   523→  function toggleCompany(id: string) {
   524→    setSelectedIds((prev) => {
   525→      const next = new Set(prev)
   526→      if (next.has(id)) next.delete(id)
   527→      else next.add(id)
   528→      return next
   529→    })
   530→  }
   531→
   532→  if (!product) return null
   533→
   534→  // Exclude the source company from the selective list
   535→  const selectableCompanies = companies.filter(
   536→    (c) => c.id !== product.sourceCompany.id,
   537→  )
   538→
   539→  const canSubmit =
   540→    !mutation.isPending &&
   541→    (targetScope === 'organization' || selectedIds.size > 0)
   542→
   543→  return (
   544→    <Dialog open={open} onOpenChange={handleOpenChange}>
   545→      <DialogContent className="sm:max-w-[520px]">
   546→        <DialogHeader>
   547→          <DialogTitle className="flex items-center gap-2">
   548→            <Globe className="h-4 w-4 text-emerald-600" />
   549→            Promote product
   550→          </DialogTitle>
   551→          <DialogDescription>
   552→            Share <span className="font-medium text-foreground">{product.title}</span> with other
   553→            companies in your organization.
   554→          </DialogDescription>
   555→        </DialogHeader>
   556→
   557→        <div className="space-y-4 py-2">
   558→          {/* Source company info */}
   559→          <div className="flex items-center gap-2 text-xs text-muted-foreground">
   560→            <Building2 className="h-3.5 w-3.5" />
   561→            Source: <span className="text-foreground font-medium">{product.sourceCompany.name}</span>
   562→          </div>
   563→
   564→          {/* Target scope — radio cards */}
   565→          <div className="space-y-2">
   566→            <Label>Target scope</Label>
   567→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-2">
   568→              <button
   569→                type="button"
   570→                onClick={() => setTargetScope('organization')}
   571→                className={cn(
   572→                  'flex items-start gap-3 rounded-lg border p-3 text-left transition-colors',
   573→                  targetScope === 'organization'
   574→                    ? 'border-emerald-500 bg-emerald-50 dark:bg-emerald-950/30 ring-1 ring-emerald-500/30'
   575→                    : 'border-border hover:border-emerald-400/60 hover:bg-muted/40',
   576→                )}
   577→              >
   578→                <div
   579→                  className={cn(
   580→                    'mt-0.5 flex h-4 w-4 shrink-0 items-center justify-center rounded-full border',
   581→                    targetScope === 'organization'
   582→                      ? 'border-emerald-500 bg-emerald-500'
   583→                      : 'border-muted-foreground/40',
   584→                  )}
   585→                >
   586→                  {targetScope === 'organization' && (
   587→                    <Check className="h-3 w-3 text-white" />
   588→                  )}
   589→                </div>
   590→                <div className="space-y-0.5 min-w-0">
   591→                  <div className="flex items-center gap-1.5 text-sm font-medium">
   592→                    <Globe className="h-3.5 w-3.5" />
   593→                    Organization
   594→                  </div>
   595→                  <p className="text-xs text-muted-foreground">
   596→                    Visible to every active company in the org.
   597→                  </p>
   598→                </div>
   599→              </button>
   600→
   601→              <button
   602→                type="button"
   603→                onClick={() => setTargetScope('selective')}
   604→                className={cn(
   605→                  'flex items-start gap-3 rounded-lg border p-3 text-left transition-colors',
   606→                  targetScope === 'selective'
   607→                    ? 'border-amber-500 bg-amber-50 dark:bg-amber-950/30 ring-1 ring-amber-500/30'
   608→                    : 'border-border hover:border-amber-400/60 hover:bg-muted/40',
   609→                )}
   610→              >
   611→                <div
   612→                  className={cn(
   613→                    'mt-0.5 flex h-4 w-4 shrink-0 items-center justify-center rounded-full border',
   614→                    targetScope === 'selective'
   615→                      ? 'border-amber-500 bg-amber-500'
   616→                      : 'border-muted-foreground/40',
   617→                  )}
   618→                >
   619→                  {targetScope === 'selective' && (
   620→                    <Check className="h-3 w-3 text-white" />
   621→                  )}
   622→                </div>

  1013→  const [demoteTarget, setDemoteTarget] = useState<SharedProduct | null>(null)
  1014→  const [demoteOpen, setDemoteOpen] = useState(false)
  1015→  const [promoteTarget, setPromoteTarget] = useState<PromotableProduct | null>(null)
  1016→  const [promoteOpen, setPromoteOpen] = useState(false)
  1017→
  1018→  const {
  1019→    data,
  1020→    isLoading,
  1021→    isError,
  1022→    error,
  1023→    refetch,
  1024→  } = useQuery<OrgCatalogResponse>({
  1025→    queryKey: ['org-catalog'],
  1026→    queryFn: () => api.get<OrgCatalogResponse>('/api/org/catalog'),
  1027→    staleTime: 30_000,
  1028→    retry: (failureCount, err) => {
  1029→      // Don't retry 403s — they won't self-heal
  1030→      if (err instanceof FetchError && err.status === 403) return false

  1032→    },

  1037→    mutationFn: async ({ productId, companyId }: { productId: string; companyId: string }) => {
  1038→      return api.delete<{ success: boolean }>(
  1039→        `/api/products/${productId}/selective-access?company_id=${encodeURIComponent(companyId)}`,
  1040→      )
  1041→    },
  1042→    onSuccess: (_data, variables) => {

  1044→      toast.success('Access revoked', {
  1045→        description: 'The company will no longer see this product.',

  1052→        err instanceof FetchError ? err.message : 'Failed to revoke access.'
