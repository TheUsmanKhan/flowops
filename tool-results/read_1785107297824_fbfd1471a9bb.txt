   225→const PKR = new Intl.NumberFormat('en-PK', { maximumFractionDigits: 0 })
   226→
   227→function formatPKR(n: number): string {
   228→  return `Rs. ${PKR.format(n)}`
   229→}
   230→
   231→function formatDate(iso: string | null): string {
   232→  if (!iso) return '—'
   233→  try {
   234→    return new Date(iso).toLocaleDateString('en-PK', {
   235→      day: '2-digit',
   236→      month: 'short',
   237→      year: 'numeric',
   238→    })
   239→  } catch {
   240→    return iso
   241→  }
   242→}
   243→
   244→function formatDateTime(iso: string | null): string {
   245→  if (!iso) return '—'
   246→  try {
   247→    return new Date(iso).toLocaleString('en-PK', {
   248→      day: '2-digit',
   249→      month: 'short',
   250→      hour: '2-digit',
   251→      minute: '2-digit',
   252→    })
   253→  } catch {
   254→    return iso
   255→  }
   256→}
   257→
   258→function getErrorMessage(err: unknown): string {
   259→  if (err instanceof FetchError) return err.message
   260→  if (err instanceof Error) return err.message
   261→  return 'Something went wrong.'
   262→}
   263→
   264→function formatAttributeValues(values: Record<string, string>): string {
   265→  if (!values || Object.keys(values).length === 0) return ''
   266→  return Object.entries(values)
   267→    .map(([k, v]) => `${k}: ${v}`)
   268→    .join(' · ')
   269→}

   292→    queryKey: ['order', orderId, 'activity'],

   296→      ),

   312→  }
   313→
   314→  // ── Mutations ─────────────────────────────────────────────────────────────

   316→    mutationFn: () => api.post(`/api/orders/${orderId}/confirm`),
   317→    onSuccess: () => {
   318→      toast.success('Confirm order succeeded.')
   319→      invalidateAll()
   320→    },
   321→    onError: (err) => toast.error(getErrorMessage(err)),
   322→  })
   323→  const processingMutation = useMutation({
   324→    mutationFn: () => api.post(`/api/orders/${orderId}/processing`),
   325→    onSuccess: () => {
   326→      toast.success('Mark as processing succeeded.')
   327→      invalidateAll()
   328→    },
   329→    onError: (err) => toast.error(getErrorMessage(err)),
   330→  })
   331→  const packedMutation = useMutation({
   332→    mutationFn: () => api.post(`/api/orders/${orderId}/packed`),
   333→    onSuccess: () => {
   334→      toast.success('Mark as packed succeeded.')
   335→      invalidateAll()
   336→    },
   337→    onError: (err) => toast.error(getErrorMessage(err)),
   338→  })
   339→  const deliveredMutation = useMutation({
   340→    mutationFn: () => api.post(`/api/orders/${orderId}/delivered`),
   341→    onSuccess: () => {
   342→      toast.success('Mark as delivered succeeded.')
   343→      invalidateAll()
   344→    },
   345→    onError: (err) => toast.error(getErrorMessage(err)),
   346→  })
   347→
   348→  const dispatchMutation = useMutation({
   349→    mutationFn: (vars: { tracking_number: string; courier_name?: string }) =>
   350→      api.post(`/api/orders/${orderId}/dispatch`, vars),
   351→    onSuccess: () => {
   352→      toast.success('Order dispatched.')
   353→      setDispatchDialogOpen(false)
   354→      invalidateAll()
   355→    },
   356→    onError: (err) => toast.error(getErrorMessage(err)),
   357→  })
   358→
   359→  const cancelMutation = useMutation({
   360→    mutationFn: (vars: { cancellation_reason: string }) =>
   361→      api.post(`/api/orders/${orderId}/cancel`, vars),
   362→    onSuccess: () => {
   363→      toast.success('Order cancelled.')
   364→      setCancelDialogOpen(false)
   365→      invalidateAll()
   366→    },
   367→    onError: (err) => toast.error(getErrorMessage(err)),
   368→  })
   369→
   370→  const rtoMutation = useMutation({
   371→    mutationFn: (vars: { return_reason: string }) =>
   372→      api.post(`/api/orders/${orderId}/rto`, vars),
   373→    onSuccess: () => {
   374→      toast.success('Order marked as RTO.')
   375→      setRtoDialogOpen(false)
   376→      invalidateAll()
   377→    },
   378→    onError: (err) => toast.error(getErrorMessage(err)),
   379→  })
   380→
   381→  const convertMutation = useMutation({
   382→    mutationFn: (vars: {
   383→      new_payment_type: 'partial_advance' | 'fully_prepaid'
   384→      advance_amount?: number
   385→      advance_payment_method?: string
   386→      advance_payment_reference?: string
   387→      advance_payment_screenshot_url?: string
   388→    }) => api.post(`/api/orders/${orderId}/convert-payment`, vars),

   390→      toast.success('Payment converted.')
   391→      setConvertDialogOpen(false)
   392→      invalidateAll()
   393→    },
   394→    onError: (err) => toast.error(getErrorMessage(err)),
   395→  })
   396→
   397→  const codMutation = useMutation({
   398→    mutationFn: (vars: { collected_amount: number }) =>
   399→      api.post(`/api/orders/${orderId}/cod-collected`, vars),
   400→    onSuccess: () => {
   401→      toast.success('COD marked as collected.')
   402→      setCodDialogOpen(false)
   403→      invalidateAll()
   404→    },
   405→    onError: (err) => toast.error(getErrorMessage(err)),
   406→  })
   407→
   408→  // ── Loading state ─────────────────────────────────────────────────────────
   409→  if (orderQuery.isLoading) {
   410→    return (
   411→      <div className="space-y-6">
   412→        <PageHeader
   413→          title="Order detail"
   414→          actions={
   415→            <Button variant="outline" size="sm" onClick={() => navigate({ name: 'orders' })}>
   416→              <ArrowLeft className="h-4 w-4" /> Back to Orders
   417→            </Button>
   418→          }
   419→        />

   421→          <CardContent className="p-6 space-y-3">

   424→            <Skeleton className="h-5 w-2/3" />
   425→            <Skeleton className="h-64 w-full" />
   426→          </CardContent>
   427→        </Card>
   428→      </div>
   429→    )
   430→  }
   431→
   432→  if (orderQuery.isError || !orderQuery.data) {
   433→    return (
   434→      <div className="space-y-6">
   435→        <PageHeader
   436→          title="Order detail"
   437→          actions={
   438→            <Button variant="outline" size="sm" onClick={() => navigate({ name: 'orders' })}>
   439→              <ArrowLeft className="h-4 w-4" /> Back to Orders
   440→            </Button>
   441→          }
   442→        />
   443→        <Card>
   444→          <CardContent className="p-10 text-center">
   445→            <AlertCircle className="h-10 w-10 mx-auto text-rose-500 mb-3" />
   446→            <p className="text-sm text-muted-foreground mb-4">
   447→              {getErrorMessage(orderQuery.error)}
   448→            </p>
   449→            <Button variant="outline" onClick={() => orderQuery.refetch()}>
   450→              Try again
   451→            </Button>
   452→          </CardContent>
   453→        </Card>
   454→      </div>
   455→    )
   456→  }
   457→
   458→  const { order, customer, items } = orderQuery.data
   459→
   460→  // ── Status + action availability ──────────────────────────────────────────

   462→  const canConfirm = status === 'pending' && canManage
   463→  const canProcess =
   464→    (status === 'confirmed' || status === 'partially_backordered') && canFulfill

   466→    (status === 'confirmed' ||
   467→      status === 'partially_backordered' ||
   468→      status === 'processing') &&
   469→    canFulfill &&

   472→    (status === 'confirmed' ||
   473→      status === 'partially_backordered' ||
   474→      status === 'processing') &&

   481→
   482→  const canConvertPayment = order.paymentStatus === 'cod_pending' && canManage

   484→    (status === 'dispatched' || status === 'delivered') &&
   485→    !order.codCollected &&
   486→    order.paymentType !== 'fully_prepaid' &&
   487→    canManage
   488→
   489→  const statusBadge = ORDER_STATUS_BADGE[status] ?? {
   490→    label: status,
   491→    className: 'bg-gray-100 text-gray-700 border-gray-200',
   492→  }

   495→    className: 'bg-gray-100 text-gray-700 border-gray-200',
   496→  }
   497→  const sourceBadge = SOURCE_BADGE[order.orderSource] ?? {
   498→    label: order.orderSource,
   499→    className: 'bg-gray-100 text-gray-700 border-gray-200',
   500→  }
   501→
   502→  const remainingCod = order.remainingCodAmount ?? Math.max(

   504→    order.totalOrderValue - (order.advanceAmount ?? 0),
   505→  )
   506→
   507→  return (
   508→    <div className="space-y-6">
   509→      <PageHeader
   510→        title={order.flowopsOrderNumber}
   511→        description={
   512→          order.externalOrderReference
   513→            ? `External ref: ${order.externalOrderReference}`

   515→        }
   516→        actions={
   517→          <div className="flex items-center gap-2">
   518→            <Button
   519→              variant="outline"
   520→              size="sm"
   521→              onClick={() => orderQuery.refetch()}
   522→              disabled={orderQuery.isFetching}
   523→            >
   524→              <RefreshCw className={orderQuery.isFetching ? 'h-4 w-4 animate-spin' : 'h-4 w-4'} />
   525→              Refresh
   526→            </Button>
   527→            <Button variant="outline" size="sm" onClick={() => navigate({ name: 'orders' })}>
   528→              <ArrowLeft className="h-4 w-4" /> Back to Orders
   529→            </Button>
   530→          </div>
   531→        }
   532→      />
   533→
   534→      {/* ── Status badge row ─────────────────────────────────────────────── */}
   535→      <div className="flex flex-wrap items-center gap-2">
   536→        <Badge variant="outline" className={statusBadge.className}>
   537→          {statusBadge.label}
   538→        </Badge>
   539→        <Badge variant="outline" className={paymentBadge.className}>
   540→          {paymentBadge.label}
   541→        </Badge>
   542→        <Badge variant="outline" className={sourceBadge.className}>
   543→          {sourceBadge.label}
   544→        </Badge>
   545→        {order.paymentType === 'partial_advance' && (
   546→          <Badge variant="outline" className="bg-sky-50 text-sky-700 border-sky-200">
   547→            Partial Advance
   548→          </Badge>
   549→        )}
   550→        {order.paymentType === 'fully_prepaid' && (
   551→          <Badge variant="outline" className="bg-emerald-50 text-emerald-700 border-emerald-200">
   552→            Prepaid
   553→          </Badge>
   554→        )}
   555→        {order.skippedConfirmation && (
   556→          <Badge variant="outline" className="bg-amber-50 text-amber-700 border-amber-200">
   557→            Skipped Confirmation
   558→          </Badge>
   559→        )}
   560→      </div>
   561→
   562→      {/* ── Action buttons (context-sensitive) ─────────────────────────────── */}
   563→      <Card>
   564→        <CardContent className="p-4 flex flex-wrap gap-2">
   565→          {canConfirm && (
   566→            <Button
   567→              size="sm"
   568→              onClick={() => confirmMutation.mutate()}
   569→              disabled={confirmMutation.isPending}
   570→            >
   571→              {confirmMutation.isPending ? (
   572→                <Loader2 className="h-4 w-4 animate-spin" />
   573→              ) : (
   574→                <CheckCircle2 className="h-4 w-4" />
   575→              )}
   576→              Confirm Order

  1169→    ]
  1170→  }, [order])

  1180→  }
  1181→
  1182→  // Filter to relevant steps — show cancelled/rto only if they occurred
  1183→  const visibleSteps = steps.filter((s) => {
  1184→    if (s.key === 'rto') return !!s.time

  1187→      return order.status === 'processing' || !!order.packedAt || !!order.dispatchedAt
  1188→    if (s.key === 'packed') return !!s.time
  1189→    return true
  1190→  })
  1191→
  1192→  return (
  1193→    <ol className="space-y-3">

  1197→        return (

  1199→            <div className="flex flex-col items-center">
  1200→              <div
  1201→                className={cn(
  1202→                  'flex h-7 w-7 items-center justify-center rounded-full',
  1203→                  reached ? toneClasses[s.tone] : 'bg-muted text-muted-foreground',
  1204→                )}
  1205→              >
  1206→                <Icon className="h-3.5 w-3.5" />
  1207→              </div>
  1208→              {idx < visibleSteps.length - 1 && (
  1209→                <div className={cn('w-px flex-1 my-1', reached ? 'bg-border' : 'bg-muted')} />
  1210→              )}
  1211→            </div>
  1212→            <div className="min-w-0 flex-1 pb-1">
  1213→              <p className={cn('text-sm', reached ? 'font-medium' : 'text-muted-foreground')}>
  1214→                {s.label}
  1215→              </p>
  1216→              <p className="text-xs text-muted-foreground">
  1217→                {reached ? formatDateTime(s.time) : 'Pending'}
  1218→              </p>
  1219→            </div>
  1220→          </li>
  1221→        )
  1222→      })}
  1223→    </ol>
  1224→  )
