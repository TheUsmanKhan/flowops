     1→import { db } from './db'
     2→import { Decimal } from '@prisma/client/runtime/library'
     3→import type { Prisma } from '@prisma/client'
     4→
     5→/**
     6→ * THE CORE INVENTORY FUNCTION.
     7→ *
     8→ * This is the single point of truth for all stock movements. It:
     9→ *   1. Finds or creates the inventory_pools row for variant+location
    10→ *   2. Validates sufficient stock for OUT-direction transactions
    11→ *   3. Recalculates WAC (weighted average cost) for IN-direction transactions
    12→ *   4. Updates on_hand / reserved / avg_cost on the pool
    13→ *   5. Inserts the immutable inventory_transactions ledger row
    14→ *   6. Records avg_cost_history when avg_cost changes
    15→ *   7. Flips track_inventory TRUE on made_to_order variants on first return
    16→ *
    17→ * IMPORTANT: inventory_pools is NEVER written to directly from any other
    18→ * code path — only through this function. This guarantees the ledger
    19→ * and pool always agree.
    20→ */
    21→
    22→export type TransactionType =
    23→  | 'opening_stock'
    24→  | 'purchase_received'
    25→  | 'sale_dispatched'
    26→  | 'order_reserved'
    27→  | 'order_unreserved'
    28→  | 'return_resellable'
    29→  | 'return_damaged'
    30→  | 'return_stitched_received'
    31→  | 'transfer_out'
    32→  | 'transfer_in'
    33→  | 'cycle_count_adjust'
    34→  | 'damage_writeoff'
    35→  | 'theft_writeoff'
    36→  | 'missing_writeoff'
    37→  | 'transit_loss'
    38→  | 'supplier_return'
    39→  | 'fabric_consumed_for_stitching'
    40→
    41→const OUT_TYPES: TransactionType[] = [
    42→  'sale_dispatched',
    43→  'transfer_out',
    44→  'damage_writeoff',
    45→  'theft_writeoff',
    46→  'missing_writeoff',
    47→  'transit_loss',
    48→  'supplier_return',
    49→  'fabric_consumed_for_stitching',
    50→]
    51→
    52→const WAC_RECALC_TYPES: TransactionType[] = [
    53→  'opening_stock',
    54→  'purchase_received',
    55→  'return_stitched_received',
    56→  'transfer_in',
    57→  'return_resellable',
    58→]
    59→
    60→interface ProcessTxnInput {
    61→  orgVariantId: string
    62→  locationId: string
    63→  organizationId: string
    64→  companyId?: string | null
    65→  employeeId?: string | null
    66→  transactionType: TransactionType
    67→  quantity: number // positive = in, negative = out
    68→  costPerUnit?: number | null // if null, uses current avg_cost for OUT, or new_cost for IN
    69→  referenceType?: string | null
    70→  referenceId?: string | null
    71→  notes?: string | null
    72→  metadata?: Record<string, unknown> | null
    73→}
    74→
    75→interface ProcessTxnResult {
    76→  success: boolean
    77→  transactionId?: string
    78→  poolState?: {
    79→    onHand: number
    80→    reserved: number
    81→    available: number
    82→    avgCost: number
    83→  }
    84→  error?: string
    85→}
    86→
    87→/**
    88→ * Calculate new weighted average cost.
    89→ * new_avg = (existing_qty × old_avg + new_qty × new_cost) / total_qty
    90→ */
    91→function calculateNewAvgCost(
    92→  existingQty: number,
    93→  oldAvg: number,
    94→  newQty: number,
    95→  newCost: number,
    96→): number {
    97→  const totalQty = existingQty + newQty
    98→  if (totalQty <= 0) return 0
    99→  return (existingQty * oldAvg + newQty * newCost) / totalQty
   100→}
   101→
   102→/**
   103→ * Process a single inventory transaction.
   104→ * This function is the ONLY way to modify inventory_pools.
   105→ */
   106→export async function processInventoryTransaction(
   107→  input: ProcessTxnInput,
   108→): Promise<ProcessTxnResult> {
   109→  const {
   110→    orgVariantId,
   111→    locationId,
   112→    organizationId,
   113→    companyId = null,
   114→    employeeId = null,
   115→    transactionType,
   116→    quantity,
   117→    referenceType = null,
   118→    referenceId = null,
   119→    notes = null,
   120→    metadata = null,
   121→  } = input
   122→
   123→  // For OUT transactions, quantity should be positive (we negate internally)
   124→  // For IN transactions, quantity is positive
   125→  const absQty = Math.abs(quantity)
   126→
   127→  try {
   128→    // 1. Find or create the inventory_pools row
   129→    let pool = await db.inventoryPool.findUnique({
   130→      where: {
   131→        orgVariantId_locationId: { orgVariantId, locationId },
   132→      },
   133→    })
   134→
   135→    if (!pool) {
   136→      // First transaction ever for this variant+location — create pool with zeros
   137→      pool = await db.inventoryPool.create({
   138→        data: {
   139→          orgVariantId,
   140→          locationId,
   141→          organizationId,
   142→          onHand: 0,
   143→          reserved: 0,
   144→          incoming: 0,
   145→          avgCost: 0,
   146→        },
   147→      })
   148→    }
   149→
   150→    // 2. Validate sufficient stock for OUT-direction transactions
   151→    if (OUT_TYPES.includes(transactionType)) {
   152→      const available = pool.onHand - pool.reserved
   153→      if (available < absQty) {
   154→        return {
   155→          success: false,
   156→          error: `INSUFFICIENT_STOCK: Available ${available}, requested ${absQty}`,
   157→        }
   158→      }
   159→    }
   160→
   161→    // 3. Determine cost_per_unit and compute new avg_cost
   162→    const oldAvgCost = Number(pool.avgCost)
   163→    let costPerUnit = input.costPerUnit ?? null
   164→
   165→    // For IN-direction WAC recalculation types
   166→    if (WAC_RECALC_TYPES.includes(transactionType)) {
   167→      if (costPerUnit === null) {
   168→        costPerUnit = oldAvgCost // fallback if not provided
   169→      }
   170→    }
   171→
   172→    // For OUT types: use current avg_cost if not explicitly provided
   173→    if (OUT_TYPES.includes(transactionType) && costPerUnit === null) {
   174→      costPerUnit = oldAvgCost
   175→    }
   176→
   177→    // For transfer_in: costPerUnit must be passed explicitly (sending location's cost)
   178→    if (transactionType === 'transfer_in' && costPerUnit === null) {
   179→      costPerUnit = oldAvgCost // fallback
   180→    }
   181→
   182→    const finalCostPerUnit = costPerUnit ?? 0
   183→
   184→    // 4. Compute new pool state
   185→    let newOnHand = pool.onHand
   186→    let newReserved = pool.reserved
   187→    let newAvgCost = oldAvgCost
   188→    let newIncoming = pool.incoming
   189→
   190→    switch (transactionType) {
   191→      case 'opening_stock':
   192→        newOnHand += absQty
   193→        newAvgCost = calculateNewAvgCost(pool.onHand, oldAvgCost, absQty, finalCostPerUnit)
   194→        break
   195→      case 'purchase_received':
   196→        newOnHand += absQty
   197→        newIncoming = Math.max(0, newIncoming - absQty)
   198→        newAvgCost = calculateNewAvgCost(pool.onHand, oldAvgCost, absQty, finalCostPerUnit)
   199→        break
   200→      case 'sale_dispatched':
   201→        newOnHand -= absQty
   202→        newReserved = Math.max(0, newReserved - absQty)
   203→        break
   204→      case 'order_reserved':
   205→        newReserved += absQty
   206→        break
   207→      case 'order_unreserved':
   208→        newReserved = Math.max(0, newReserved - absQty)
   209→        break
   210→      case 'return_resellable':
   211→        newOnHand += absQty
   212→        newAvgCost = calculateNewAvgCost(pool.onHand, oldAvgCost, absQty, finalCostPerUnit)
   213→        break
   214→      case 'return_stitched_received':
   215→        newOnHand += absQty
   216→        newAvgCost = calculateNewAvgCost(pool.onHand, oldAvgCost, absQty, finalCostPerUnit)
   217→        break
   218→      case 'return_damaged':
   219→        // No pool change — goes straight to stock_loss_records
   220→        break
   221→      case 'transfer_out':
   222→        newOnHand -= absQty
   223→        break
   224→      case 'transfer_in':
   225→        newOnHand += absQty
   226→        // costPerUnit is the sending location's cost — do NOT recalculate WAC
   227→        // The transferred stock keeps its original cost_per_unit exactly
   228→        newAvgCost = calculateNewAvgCost(pool.onHand, oldAvgCost, absQty, finalCostPerUnit)
   229→        break
   230→      case 'cycle_count_adjust':
   231→        // Set on_hand directly to counted value
   232→        // quantity here represents the NEW on_hand value (positive)
   233→        newOnHand = absQty
   234→        break
   235→      case 'damage_writeoff':
   236→      case 'theft_writeoff':
   237→      case 'missing_writeoff':
   238→      case 'transit_loss':
   239→        newOnHand -= absQty
   240→        break
   241→      case 'supplier_return':
   242→        newOnHand -= absQty
   243→        break
   244→      case 'fabric_consumed_for_stitching':
   245→        newOnHand -= absQty
   246→        break
   247→    }
   248→
   249→    // 5. Update timestamps
   250→    const now = new Date()
   251→    const updateData: Record<string, unknown> = {
   252→      onHand: newOnHand,
   253→      reserved: newReserved,
   254→      incoming: newIncoming,
   255→      avgCost: newAvgCost,
   256→      updatedAt: now,
   257→    }
   258→    if (
   259→      transactionType === 'purchase_received' ||
   260→      transactionType === 'opening_stock' ||
   261→      transactionType === 'return_resellable' ||
   262→      transactionType === 'return_stitched_received' ||
   263→      transactionType === 'transfer_in'
   264→    ) {
   265→      updateData.lastReceivedAt = now
   266→    }
   267→    if (transactionType === 'sale_dispatched') {
   268→      updateData.lastSoldAt = now
   269→    }
   270→    if (transactionType === 'cycle_count_adjust') {
   271→      updateData.lastCountedAt = now
   272→    }
   273→
   274→    await db.inventoryPool.update({
   275→      where: { id: pool.id },
   276→      data: updateData,
   277→    })
   278→
   279→    // 6. Handle track_inventory flip for made_to_order variants on first
   280→    //    return OR on opening_stock entry (e.g. user confirms "pre-made bulk
   281→    //    stock" for an MTO variant during product creation). One-way FALSE → TRUE.
   282→    if (
   283→      transactionType === 'return_stitched_received' ||
   284→      transactionType === 'opening_stock'
   285→    ) {
   286→      const variant = await db.orgProductVariant.findUnique({
   287→        where: { id: orgVariantId },
   288→        select: { trackInventory: true, fulfillmentType: true },
   289→      })
   290→      if (variant && !variant.trackInventory && variant.fulfillmentType === 'made_to_order') {
   291→        // ONE-WAY flip: FALSE → TRUE (never back to FALSE)
   292→        await db.orgProductVariant.update({
   293→          where: { id: orgVariantId },
   294→          data: { trackInventory: true },
   295→        })
   296→      }
   297→    }
   298→
   299→    // 7. Insert the inventory_transactions ledger row
   300→    const txnQuantity = OUT_TYPES.includes(transactionType) ? -absQty : absQty
   301→    const avgCostChanged = newAvgCost !== oldAvgCost
   302→
   303→    const txn = await db.inventoryTransaction.create({
   304→      data: {
   305→        orgVariantId,
   306→        locationId,
   307→        organizationId,
   308→        companyId,
   309→        employeeId,
   310→        transactionType,
   311→        quantity: txnQuantity,
   312→        costPerUnit: finalCostPerUnit,
   313→        avgCostBefore: oldAvgCost,
   314→        avgCostAfter: newAvgCost,
   315→        referenceType,
   316→        referenceId,
   317→        notes,
   318→        metadata: metadata ? JSON.stringify(metadata) : '{}',
   319→        recordedAt: now,
   320→      },
   321→    })
   322→
   323→    // 8. Insert avg_cost_history if avg_cost changed
   324→    if (avgCostChanged) {
   325→      await db.avgCostHistory.create({
   326→        data: {
   327→          orgVariantId,
   328→          locationId,
   329→          organizationId,
   330→          avgCostBefore: oldAvgCost,
   331→          avgCostAfter: newAvgCost,
   332→          triggeredByTxnId: txn.id,
   333→          triggerReason: transactionType,
   334→        },
   335→      })
   336→    }
   337→
   338→    return {
   339→      success: true,
   340→      transactionId: txn.id,
   341→      poolState: {
   342→        onHand: newOnHand,
   343→        reserved: newReserved,
   344→        available: newOnHand - newReserved,
   345→        avgCost: newAvgCost,
   346→      },
   347→    }
   348→  } catch (err) {
   349→    console.error('[inventory] processInventoryTransaction error:', err)
   350→    return {
   351→      success: false,
   352→      error: err instanceof Error ? err.message : 'Unknown inventory transaction error',
   353→    }
   354→  }
   355→}

   357→/**
   358→ * Check returned stock availability for a made_to_order variant.
   359→ * Returns available inventory_pools rows across all locations.
   360→ */
   361→export async function checkReturnedStockAvailability(variantId: string) {
   362→  const pools = await db.inventoryPool.findMany({
   363→    where: {
   364→      orgVariantId: variantId,
   365→      onHand: { gt: 0 },
   366→    },
   367→    include: {
   368→      location: { select: { id: true, name: true } },
   369→    },
   370→  })
   371→  return pools.map((p) => ({
   372→    locationId: p.locationId,
   373→    locationName: p.location.name,
   374→    available: p.onHand - p.reserved,
   375→    avgCost: Number(p.avgCost),
   376→  }))
   377→}

   379→/**
   380→ * Get inventory summary for a product — powers the product detail Inventory tab.
   381→ */
   382→export async function getProductInventorySummary(productId: string) {
   383→  const variants = await db.orgProductVariant.findMany({
   384→    where: { productId },
   385→    select: { id: true, sku: true, fulfillmentType: true, trackInventory: true },
   386→  })
   387→
   388→  const result: Array<{
   389→    variantId: string
   390→    sku: string
   391→    fulfillmentType: string
   392→    trackInventory: boolean
   393→    totalOnHand: number
   394→    totalReserved: number
   395→    totalAvailable: number
   396→    totalValue: number
   397→    locations: Array<{
   398→      locationId: string
   399→      locationName: string
   400→      onHand: number
   401→      reserved: number
   402→      available: number
   403→      avgCost: number
   404→      incoming: number
   405→    }>
   406→  }> = []
   407→  for (const variant of variants) {
   408→    const pools = await db.inventoryPool.findMany({
   409→      where: { orgVariantId: variant.id },
   410→      include: {
   411→        location: { select: { id: true, name: true } },
   412→      },
   413→    })
   414→
   415→    const totalOnHand = pools.reduce((sum, p) => sum + p.onHand, 0)
   416→    const totalReserved = pools.reduce((sum, p) => sum + p.reserved, 0)
   417→    const totalAvailable = totalOnHand - totalReserved
   418→    const totalValue = pools.reduce((sum, p) => sum + Number(p.onHand) * Number(p.avgCost), 0)
   419→
   420→    result.push({
   421→      variantId: variant.id,
   422→      sku: variant.sku,
   423→      fulfillmentType: variant.fulfillmentType,
   424→      trackInventory: variant.trackInventory,
   425→      totalOnHand,
   426→      totalReserved,
   427→      totalAvailable,
   428→      totalValue,
   429→      locations: pools.map((p) => ({
   430→        locationId: p.locationId,
   431→        locationName: p.location.name,
   432→        onHand: p.onHand,
   433→        reserved: p.reserved,
   434→        available: p.onHand - p.reserved,
   435→        avgCost: Number(p.avgCost),
   436→        incoming: p.incoming,
   437→      })),
   438→    })
   439→  }
   440→
   441→  return result
   442→}

   444→/**
   445→ * Generate the next PO number: PO-{year}-{sequence}
   446→ * Sequence resets per calendar year per organization.
   447→ */
   448→export async function generatePoNumber(organizationId: string): Promise<string> {
   449→  const year = new Date().getFullYear()
   450→  const prefix = `PO-${year}-`
   451→
   452→  // Count existing POs this year for this org
   453→  const count = await db.purchaseOrder.count({
   454→    where: {
   455→      organizationId,
   456→      poNumber: { startsWith: prefix },
   457→    },
   458→  })
   459→
   460→  return `${prefix}${String(count + 1).padStart(3, '0')}`
   461→}

   463→/**
   464→ * Increment incoming stock on a pool (for PO ordering).
   465→ * This is the ONLY function that writes to inventory_pools.incoming directly
   466→ * — it's a live projection field, not a ledgered movement.
   467→ * Creates the pool row if it doesn't exist.
   468→ */
   469→export async function incrementIncomingStock(
   470→  orgVariantId: string,
   471→  locationId: string,
   472→  organizationId: string,
   473→  qty: number,
   474→): Promise<void> {
   475→  await db.inventoryPool.upsert({
   476→    where: { orgVariantId_locationId: { orgVariantId, locationId } },
   477→    update: { incoming: { increment: qty } },
   478→    create: {
   479→      orgVariantId,
   480→      locationId,
   481→      organizationId,
   482→      incoming: qty,
   483→    },
   484→  })
   485→}

   487→/**
   488→ * Decrement incoming stock (never below 0).
   489→ * Used when cancelling POs or receiving against POs.
   490→ */
   491→export async function decrementIncomingStock(
   492→  orgVariantId: string,
   493→  locationId: string,
   494→  qty: number,
   495→): Promise<void> {
   496→  const pool = await db.inventoryPool.findUnique({
   497→    where: { orgVariantId_locationId: { orgVariantId, locationId } },
   498→    select: { incoming: true },
   499→  })
   500→  if (!pool) return
   501→  const newIncoming = Math.max(0, pool.incoming - qty)
   502→  await db.inventoryPool.update({
   503→    where: { orgVariantId_locationId: { orgVariantId, locationId } },
   504→    data: { incoming: newIncoming },
   505→  })
   506→}

   508→/**
   509→ * Check and fulfill a made-to-order variant.
   510→ *
   511→ * The central decision function:
   512→ * 1. Check if returned stock is available for this variant
   513→ * 2. If yes: return { source: 'existing_stock', location_id, available }
   514→ * 3. If no: create a production order, consume fabric, return { source: 'fresh_production' }
   515→ */
   516→export async function checkAndFulfillMadeToOrderVariant(
   517→  orgVariantId: string,
   518→  quantity: number,
   519→  companyId: string,
   520→  preferredLocationId?: string,
   521→): Promise<{
   522→  source: 'existing_stock' | 'fresh_production'
   523→  locationId?: string
   524→  available?: number
   525→  productionOrderId?: string
   526→  estimatedCompletionDate?: Date
   527→  error?: string
   528→}> {
   529→  // 1. Check returned stock availability
   530→  const availability = await checkReturnedStockAvailability(orgVariantId)
   531→  const totalAvailable = availability.reduce((sum, a) => sum + a.available, 0)
   532→
   533→  if (totalAvailable >= quantity) {
   534→    // Use existing stock — find the best location (most available)
   535→    const best = availability
   536→      .filter((a) => a.available > 0)
   537→      .sort((a, b) => b.available - a.available)[0]
   538→    return {
   539→      source: 'existing_stock',
   540→      locationId: best.locationId,
   541→      available: best.available,
   542→    }
   543→  }
   544→
   545→  // 2. Not enough returned stock — create a production order
   546→  const variant = await db.orgProductVariant.findUnique({
   547→    where: { id: orgVariantId },
   548→    select: {
   549→      id: true,
   550→      fabricSourceVariantId: true,
   551→      stitchingCharges: true,
   552→      productionDays: true,
   553→      organizationId: true,
   554→    },
   555→  })
   556→
   557→  if (!variant) return { source: 'fresh_production', error: 'Variant not found' }
   558→  if (!variant.fabricSourceVariantId) {
   559→    return { source: 'fresh_production', error: 'No fabric source variant linked to this made_to_order variant' }
   560→  }
   561→
   562→  // Find fabric stock at the preferred location or any location with stock
   563→  const fabricPools = await db.inventoryPool.findMany({
   564→    where: {
   565→      orgVariantId: variant.fabricSourceVariantId,
   566→      onHand: { gt: 0 },
   567→    },
   568→    include: { location: { select: { id: true, name: true } } },
   569→  })
   570→
   571→  const fabricLocation = preferredLocationId
   572→    ? fabricPools.find((p) => p.locationId === preferredLocationId)
   573→    : fabricPools[0]
   574→
   575→  if (!fabricLocation || fabricLocation.onHand - fabricLocation.reserved < quantity) {
   576→    return {
   577→      source: 'fresh_production',
   578→      error: `Insufficient fabric stock. Available: ${fabricLocation?.onHand ?? 0}, required: ${quantity}`,
   579→    }
   580→  }
   581→
   582→  const fabricCost = Number(fabricLocation.avgCost) * quantity
   583→  const estimatedCompletionDate = new Date()
   584→  estimatedCompletionDate.setDate(estimatedCompletionDate.getDate() + (variant.productionDays || 5))
   585→
   586→  // Consume fabric
   587→  const txnResult = await processInventoryTransaction({
   588→    orgVariantId: variant.fabricSourceVariantId,
   589→    locationId: fabricLocation.locationId,
   590→    organizationId: variant.organizationId,
   591→    companyId,
   592→    transactionType: 'fabric_consumed_for_stitching',
   593→    quantity,
   594→    costPerUnit: Number(fabricLocation.avgCost),
   595→    referenceType: 'production_order',
   596→  })
   597→
   598→  if (!txnResult.success) {
   599→    return { source: 'fresh_production', error: `Fabric consumption failed: ${txnResult.error}` }
   600→  }
   601→
   602→  // Create production order
   603→  const productionOrder = await db.productionOrder.create({
   604→    data: {
   605→      organizationId: variant.organizationId,
   606→      companyId,
   607→      stitchedVariantId: orgVariantId,
   608→      fabricVariantId: variant.fabricSourceVariantId,
   609→      fabricLocationId: fabricLocation.locationId,
   610→      quantity,
   611→      status: 'fabric_reserved',
   612→      stitchingCost: new Decimal(Number(variant.stitchingCharges) || 0),
   613→      fabricCost: new Decimal(fabricCost),
   614→      estimatedCompletionDate,
   615→      fabricTxnId: txnResult.transactionId ?? null,
   616→    } as Prisma.ProductionOrderUncheckedCreateInput,
   617→  })
   618→
   619→  return {
   620→    source: 'fresh_production',
   621→    productionOrderId: productionOrder.id,
   622→    estimatedCompletionDate,
   623→  }
   624→}

   626→/**
   627→ * Quarantine stock for theft/missing loss investigations.
   628→ * Directly increments inventory_pools.reserved — does NOT call
   629→ * process_inventory_transaction() since no actual movement has occurred.
   630→ * This is a soft-hold that reduces available stock without touching on_hand.
   631→ */
   632→export async function quarantineStock(
   633→  orgVariantId: string,
   634→  locationId: string,
   635→  quantity: number,
   636→): Promise<{ success: boolean; error?: string }> {
   637→  const pool = await db.inventoryPool.findUnique({
   638→    where: { orgVariantId_locationId: { orgVariantId, locationId } },
   639→  })
   640→  if (!pool) {
   641→    return { success: false, error: 'No inventory pool exists for this variant+location.' }
   642→  }
   643→  const available = pool.onHand - pool.reserved
   644→  if (available < quantity) {
   645→    return { success: false, error: `Insufficient available stock. Available: ${available}, required: ${quantity}.` }
   646→  }
   647→  await db.inventoryPool.update({
   648→    where: { id: pool.id },
   649→    data: { reserved: { increment: quantity } },
   650→  })
   651→  return { success: true }
   652→}

   656→ * Directly decrements inventory_pools.reserved.
   657→ * Used when resolving theft/missing investigations (regardless of outcome —
   658→ * the write_off path creates a separate transaction that decrements on_hand).

   660→export async function releaseQuarantine(
   661→  orgVariantId: string,
   662→  locationId: string,
   663→  quantity: number,
   664→): Promise<void> {
   665→  const pool = await db.inventoryPool.findUnique({
   666→    where: { orgVariantId_locationId: { orgVariantId, locationId } },
   667→  })

   669→  const newReserved = Math.max(0, pool.reserved - quantity)

   671→    where: { id: pool.id },
   672→    data: { reserved: newReserved },
   673→  })
   674→}

   684→/**
   685→ * Reserve stock for an order item. Increments inventory_pools.reserved
   686→ * (does NOT touch on_hand — that happens at dispatch time). Records
   687→ * an order_reserved transaction in the ledger.
   688→ *
   689→ * @returns { success, error? } — fails if insufficient available stock.
   690→ */
   691→export async function reserveStockForOrder(input: {
   692→  orgVariantId: string
   693→  locationId: string
   694→  organizationId: string
   695→  companyId: string
   696→  employeeId?: string | null
   697→  quantity: number
   698→  orderId?: string
   699→}): Promise<{ success: boolean; error?: string }> {
   700→  // Check available stock first (available = onHand - reserved)
   701→  const pool = await db.inventoryPool.findUnique({
   702→    where: {
   703→      orgVariantId_locationId: {
   704→        orgVariantId: input.orgVariantId,
   705→        locationId: input.locationId,
   706→      },
   707→    },
   708→  })
   709→  if (!pool) {
   710→    return { success: false, error: 'No inventory pool exists for this variant+location.' }
   711→  }
   712→  const available = pool.onHand - pool.reserved
   713→  if (available < input.quantity) {
   714→    return {
   715→      success: false,
   716→      error: `Insufficient available stock. Available: ${available}, required: ${input.quantity}.`,
   717→    }
   718→  }
   719→
   720→  const result = await processInventoryTransaction({
   721→    orgVariantId: input.orgVariantId,
   722→    locationId: input.locationId,
   723→    organizationId: input.organizationId,
   724→    companyId: input.companyId,
   725→    employeeId: input.employeeId,
   726→    transactionType: 'order_reserved',
   727→    quantity: input.quantity,
   728→    referenceType: 'order',
   729→    referenceId: input.orderId,
   730→  })
   731→
   732→  if (!result.success) {
   733→    return { success: false, error: result.error }
   734→  }
   735→  return { success: true }
   736→}

   803→/**
   804→ * Restock inventory for an RTO (Return To Origin) order — session-free
   805→ * version for use by courier polling jobs and webhooks (which have no user
   806→ * session and therefore can't call processOrderReturn() which uses
   807→ * getWorkspace()).
   808→ *
   809→ * For each DISPATCHED order item:
   810→ *   - Looks up the original sale_dispatched transaction to recover the
   811→ *     cost-per-unit that was locked at dispatch time.
   812→ *   - Calls processInventoryTransaction with type 'return_resellable'
   813→ *     (for stock_based items) or 'return_stitched_received' (for
   814→ *     made_to_order items) — which increments onHand AND recalculates WAC.
   815→ *   - Marks the order item with fulfillmentStatus='returned' +
   816→ *     autoProcessedAsPerfect=true + needsReview=true so it surfaces in the
   817→ *     exception-review queue for physical spot-checking (same as the manual
   818→ *     processOrderReturn path).
   819→ *
   820→ * For CONFIRMED/PROCESSING (not-yet-dispatched) reserved items: calls
   821→ * unreserveStockForOrder to release the reservation (no onHand change since
   822→ * onHand was never decremented).
   823→ *
   824→ * This function is IDEMPOTENT — it skips items whose fulfillmentStatus is
   825→ * already 'returned' (set by a prior restock call) so re-running a poll
   826→ * cycle doesn't double-restock.
   827→ *
   828→ * @returns { success, itemsRestocked } — never throws (errors logged per-item).
   829→ */
   830→export async function restockOrderForRto(
   831→  orderId: string,
   832→  context: {
   833→    organizationId: string
   834→    companyId: string
   835→    employeeId?: string | null
   836→    returnReason?: string
   837→  },
   838→): Promise<{ success: boolean; itemsRestocked: number }> {
   839→  const order = await db.order.findUnique({
   840→    where: { id: orderId },
   841→    select: {
   842→      id: true,
   843→      status: true,
   844→      dispatchLocationId: true,
   845→      organizationId: true,
   846→      companyId: true,
   847→      items: {
   848→        include: {
   849→          orgVariant: {
   850→            select: { id: true, sku: true, costPrice: true, fulfillmentType: true },
   851→          },
   852→        },
   853→      },
   854→    },
   855→  })
   856→  if (!order) return { success: false, itemsRestocked: 0 }
   857→
   858→  const locationId = order.dispatchLocationId
   859→  if (!locationId) {
   860→    console.error(`[restockOrderForRto] Order ${orderId} has no dispatchLocationId — cannot restock`)
   861→    return { success: false, itemsRestocked: 0 }
   862→  }
   863→
   864→  let itemsRestocked = 0
   865→
   866→  for (const item of order.items) {
   867→    // Idempotency: skip items already processed (fulfillmentStatus='returned')
   868→    if (item.fulfillmentStatus === 'returned') continue
   869→
   870→    if (item.fulfillmentStatus === 'dispatched') {
   871→      // Dispatched item — onHand was decremented at dispatch. Restock it.
   872→      // Recover the cost-per-unit from the original sale_dispatched txn.
   873→      const dispatchTxn = await db.inventoryTransaction.findFirst({
   874→        where: {
   875→          orgVariantId: item.orgVariantId,
   876→          locationId,
   877→          transactionType: 'sale_dispatched',
   878→          referenceType: 'order',
   879→          referenceId: orderId,
   880→        },
   881→        select: { costPerUnit: true },
   882→        orderBy: { recordedAt: 'desc' },
   883→      })
   884→      const costPerUnit = dispatchTxn ? Number(dispatchTxn.costPerUnit) : Number(item.orgVariant.costPrice)
   885→
   886→      const txnType = item.fulfillmentTypeSnapshot === 'made_to_order'
   887→        ? 'return_stitched_received'
   888→        : 'return_resellable'
   889→
   890→      const txnResult = await processInventoryTransaction({
   891→        orgVariantId: item.orgVariantId,
   892→        locationId,
   893→        organizationId: order.organizationId,
   894→        companyId: order.companyId,
   895→        employeeId: context.employeeId ?? null,
   896→        transactionType: txnType,
   897→        quantity: item.quantity,
   898→        costPerUnit,
   899→        referenceType: 'order',
   900→        referenceId: orderId,
   901→        notes: `Auto-processed RTO return (assumed ${txnType === 'return_resellable' ? 'resellable' : 'perfect'}). Reason: ${context.returnReason ?? 'courier returned'}`,
   902→      })
   903→
   904→      if (txnResult.success) {
   905→        await db.orderItem.update({
   906→          where: { id: item.id },
   907→          data: {
   908→            fulfillmentStatus: 'returned',
   909→            autoProcessedAsPerfect: true,
   910→            needsReview: true,
   911→          },
   912→        })
   913→        itemsRestocked++
   914→      } else {
   915→        console.error(`[restockOrderForRto] Failed to restock item ${item.id}: ${txnResult.error}`)
   916→      }
   917→    } else if (item.fulfillmentStatus === 'reserved') {
   918→      // Reserved but not dispatched — just release the reservation.
   919→      const unreserveResult = await unreserveStockForOrder({
   920→        orgVariantId: item.orgVariantId,
   921→        locationId,
   922→        organizationId: order.organizationId,
   923→        companyId: order.companyId,
   924→        employeeId: context.employeeId ?? null,
   925→        quantity: item.quantity,
   926→        orderId,
   927→      })
   928→      if (unreserveResult.success) {
   929→        await db.orderItem.update({
   930→          where: { id: item.id },
   931→          data: { fulfillmentStatus: 'returned' },
   932→        })
   933→        itemsRestocked++
   934→      } else {
   935→        console.error(`[restockOrderForRto] Failed to unreserve item ${item.id}: ${unreserveResult.error}`)
   936→      }
   937→    }
   938→    // backordered / pending items: no inventory action needed
   939→  }
   940→
   941→  return { success: true, itemsRestocked }
   942→}
