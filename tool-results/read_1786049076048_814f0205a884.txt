     3→/**
     4→ * BookingWorkbenchView (v2) — bulk courier-booking workbench with 3 tabs.
     5→ *
     6→ *  1. Orders             — GET /api/booking-workbench/bookable → data.orders
     7→ *  2. Exchange Shipments — GET /api/booking-workbench/bookable → data.shipments
     8→ *  3. Booking Activity   — GET /api/booking-workbench/activity?date_from=&date_to=
     9→ *
    10→ * Key behaviours:
    11→ *  • Per-row courier <Select> — defaults to row.recommendedCourierCompanyIntegrationId;
    12→ *    drives that row's <CityAutocomplete providerKey>.
    13→ *  • Bulk Apply — toolbar dropdown + "Apply to Selected" sets the courier on all
    14→ *    CHECKED rows in the active tab only.
    15→ *  • Weight auto-compute — each row's default orderType is computed via
    16→ *    calculateOrderWeightKg + determinePostExOrderType. If hasMissingWeight,
    17→ *    a ⚠️ tooltip is shown next to the Order Type dropdown. Still editable.
    18→ *  • Upload Booking — sequentially POSTs /api/booking-workbench/book for each
    19→ *    checked row using THAT ROW's courier integration id. Failures don't block
    20→ *    other rows. On success the row shows ✅ tracking number + auto-unchecks.
    21→ *  • After any successful booking, ['booking-workbench-bookable'] and
    22→ *    ['booking-workbench-activity'] queries are invalidated.
    23→ */

   193→// ── Helpers ───────────────────────────────────────────────────────────────────
   194→
   195→function rowKey(row: BookableRow): string {
   196→  return `${row.type}:${row.id}`
   197→}
   198→
   199→function computeRowOrderType(row: BookableRow): {
   200→  orderType: PostExOrderType
   201→  hasMissingWeight: boolean
   202→  totalWeightKg: number
   203→} {
   204→  const weightResult = calculateOrderWeightKg(
   205→    row.items.map((i) => ({
   206→      quantity: i.quantity,
   207→      variant: { weightKg: i.weightKg },
   208→    })),
   209→  )
   210→  const isExchangeReplacement =
   211→    row.type === 'exchange_shipment' && row.exchangeMethod === 'courier_replacement'
   212→  return {
   213→    orderType: determinePostExOrderType(
   214→      weightResult.totalWeightKg,
   215→      weightResult.hasMissingWeight,
   216→      isExchangeReplacement,
   217→    ),
   218→    hasMissingWeight: weightResult.hasMissingWeight,
   219→    totalWeightKg: weightResult.totalWeightKg,
   220→  }
   221→}
   222→
   223→function defaultRowState(row: BookableRow): RowState {
   224→  const computed = computeRowOrderType(row)
   225→  return {
   226→    customerName: row.customerName ?? '',
   227→    customerPhone: row.customerPhone ?? '',
   228→    deliveryAddress: row.deliveryAddress ?? '',
   229→    deliveryCity: row.deliveryCity ?? '',
   230→    codAmount: String(row.codAmount ?? 0),
   231→    orderType: computed.orderType,
   232→    hasMissingWeight: computed.hasMissingWeight,
   233→    totalWeightKg: computed.totalWeightKg,
   234→    companyIntegrationId: row.recommendedCourierCompanyIntegrationId ?? '',
   235→    // Seed from the stored universal courier reference fields (migration 015)
   236→    orderRefNumber: row.orderRefNumber ?? row.referenceNumber ?? '',
   237→    orderDetail: row.orderDetail ?? '',
   238→    transactionNotes: row.notesForCourier ?? '',
   239→    showAdvanced: false,
   240→    checked: false,
   241→    result: null,
   242→    isBooking: false,
   243→  }
   244→}

   246→// ── Main component ────────────────────────────────────────────────────────────
   247→
   248→export function BookingWorkbenchView() {
   249→  const queryClient = useQueryClient()
   250→  const [activeTab, setActiveTab] = useState<'orders' | 'shipments' | 'activity'>('orders')
   251→  const [search, setSearch] = useState('')
   252→  const [bulkApplyIntegrationId, setBulkApplyIntegrationId] = useState('')
   253→  const [rowStates, setRowStates] = useState<Record<string, RowState>>({})
   254→
   255→  // ── Data: bookable orders + shipments ──
   256→  const bookableQuery = useQuery<BookableResponse>({
   257→    queryKey: ['booking-workbench-bookable'],
   258→    queryFn: () => api.get<BookableResponse>('/api/booking-workbench/bookable'),
   259→    staleTime: 15_000,
   260→  })
   261→
   262→  // ── Data: courier integrations ──
   263→  const integrationsQuery = useQuery<IntegrationsResponse>({
   264→    queryKey: ['integrations', 'courier'],
   265→    queryFn: () => api.get<IntegrationsResponse>('/api/integrations?category=courier'),
   266→    staleTime: 30_000,
   267→  })
   268→
   269→  const courierIntegrations = useMemo(
   270→    () =>
   271→      (integrationsQuery.data?.integrations ?? []).filter(
   272→        (i) => i.isActive && i.provider?.category === 'courier',
   273→      ),
   274→    [integrationsQuery.data],
   275→  )
   276→
   277→  const integrationById = useMemo(() => {
   278→    const m = new Map<string, CompanyIntegration>()
   279→    for (const ci of courierIntegrations) m.set(ci.id, ci)
   280→    return m
   281→  }, [courierIntegrations])
   282→
   283→  function onTabChange(v: string) {
   284→    setActiveTab(v as 'orders' | 'shipments' | 'activity')
   285→    setSearch('')
   286→  }
   287→
   288→  // ── Rows currently visible in the active bookable tab ──
   289→  const rowsForActiveTab = useMemo<BookableRow[]>(() => {
   290→    if (activeTab === 'orders') return bookableQuery.data?.orders ?? []
   291→    if (activeTab === 'shipments') return bookableQuery.data?.shipments ?? []
   292→    return []
   293→  }, [bookableQuery.data, activeTab])
   294→
   295→  const filteredRows = useMemo(() => {
   296→    if (!search.trim()) return rowsForActiveTab
   297→    const q = search.trim().toLowerCase()
   298→    return rowsForActiveTab.filter(
   299→      (r) =>
   300→        r.referenceNumber.toLowerCase().includes(q) ||
   301→        r.customerName.toLowerCase().includes(q) ||
   302→        r.customerPhone?.toLowerCase().includes(q) ||
   303→        r.originalOrderNumber?.toLowerCase().includes(q),
   304→    )
   305→  }, [rowsForActiveTab, search])
   306→
   307→  // ── Row state helpers ──
   308→  function getRowState(row: BookableRow): RowState {
   309→    const key = rowKey(row)
   310→    return rowStates[key] ?? defaultRowState(row)
   311→  }
   312→
   313→  function patchRow(row: BookableRow, patch: Partial<RowState>) {
   314→    const key = rowKey(row)
   315→    setRowStates((prev) => ({
   316→      ...prev,
   317→      [key]: {
   318→        ...(prev[key] ?? defaultRowState(row)),
   319→        ...patch,
   320→        // Any edit clears the previous booking result so the user knows to re-submit.
   321→        result: patch.result !== undefined ? patch.result : null,
   322→      },
   323→    }))
   324→  }
   325→
   326→  // ── Select-all ──
   327→  const checkedCount = filteredRows.filter((r) => getRowState(r).checked).length
   328→  const allChecked = filteredRows.length > 0 && checkedCount === filteredRows.length
   329→
   330→  function toggleSelectAll() {
   331→    const next = { ...rowStates }
   332→    const target = !allChecked
   333→    for (const r of filteredRows) {
   334→      const key = rowKey(r)
   335→      const cur = next[key] ?? defaultRowState(r)
   336→      next[key] = { ...cur, checked: target }
   337→    }
   338→    setRowStates(next)
   339→  }
   340→
   341→  // ── Bulk Apply ──
   342→  function handleBulkApply() {
   343→    if (!bulkApplyIntegrationId) {
   344→      toast.error('Pick a courier in the Bulk Apply dropdown first.')
   345→      return
   346→    }
   347→    const checkedRows = filteredRows.filter((r) => getRowState(r).checked)
   348→    if (checkedRows.length === 0) {
   349→      toast.error('No rows selected. Tick at least one row first.')
   350→      return
   351→    }
   352→    setRowStates((prev) => {
   353→      const next = { ...prev }
   354→      for (const r of checkedRows) {
   355→        const key = rowKey(r)
   356→        const cur = next[key] ?? defaultRowState(r)
   357→        next[key] = { ...cur, companyIntegrationId: bulkApplyIntegrationId, result: null }
   358→      }
   359→      return next
   360→    })
   361→    const integ = integrationById.get(bulkApplyIntegrationId)
   362→    toast.success(
   363→      `Applied ${integ?.provider?.providerName ?? 'courier'} to ${checkedRows.length} row(s).`,
   364→    )
   365→  }
   366→
   367→  // ── Booking mutation (per-row) ──
   368→  const bookMutation = useMutation({
   369→    mutationFn: async (body: BookRequest): Promise<BookSuccess> =>
   370→      api.post<BookSuccess>('/api/booking-workbench/book', body),
   371→  })
   372→
   373→  async function handleUploadBooking() {
   374→    const checkedRows = filteredRows.filter((r) => {
   375→      const s = getRowState(r)
   376→      return s.checked && !s.isBooking
   377→    })
   378→    if (checkedRows.length === 0) {
   379→      toast.error('No rows selected. Tick the checkbox on at least one row.')
   380→      return
   381→    }
   382→    const missingCourier = checkedRows.find((r) => !getRowState(r).companyIntegrationId)
   383→    if (missingCourier) {
   384→      toast.error(`Row ${missingCourier.referenceNumber} has no courier selected.`)
   385→      return
   386→    }
   387→
   388→    toast.info(`Booking ${checkedRows.length} row(s)…`)
   389→
   390→    let successCount = 0
   391→    let failureCount = 0
   392→
   393→    // Sequential — external courier APIs typically don't tolerate parallel hits,
   394→    // and per-row errors must not block other rows.
   395→    for (const row of checkedRows) {
   396→      const state = getRowState(row)
   397→      const body: BookRequest = {
   398→        companyIntegrationId: state.companyIntegrationId,
   399→        customerName: state.customerName.trim() || undefined,
   400→        customerPhone: state.customerPhone.trim() || undefined,
   401→        deliveryAddress: state.deliveryAddress.trim() || undefined,
   402→        deliveryCity: state.deliveryCity.trim() || undefined,
   403→        codAmount: Number(state.codAmount) || 0,
   404→        orderType: state.orderType,
   405→        // Universal courier reference fields (migration 015) — only send if
   406→        // the user overrode the seeded value (the backend falls back to the
   407→        // stored value when these are undefined).
   408→        orderRefNumber: state.orderRefNumber.trim() || undefined,
   409→        itemDescription: state.orderDetail.trim() || undefined,
   410→        transactionNotes: state.transactionNotes.trim() || undefined,
   411→      }
   412→      if (row.type === 'order') body.orderId = row.id
   413→      else body.shipmentId = row.id
   414→
   415→      patchRow(row, { isBooking: true, result: null })
   416→
   417→      try {
   418→        const result = await bookMutation.mutateAsync(body)
   419→        successCount++
   420→        patchRow(row, {
   421→          isBooking: false,
   422→          checked: false,
   423→          result: { ok: true, trackingNumber: result.trackingNumber, orderType: result.orderType },
   424→        })
   425→      } catch (err) {
   426→        failureCount++
   427→        const message =
   428→          err instanceof FetchError ? err.message
   429→          : err instanceof Error ? err.message
   430→          : 'Booking failed'
   431→        patchRow(row, { isBooking: false, result: { ok: false, error: message } })
   432→      }
   433→    }
   434→
   435→    if (successCount > 0 && failureCount === 0) {
   436→      toast.success(`All ${successCount} row(s) booked successfully.`)
   437→    } else if (successCount > 0 && failureCount > 0) {
   438→      toast.warning(`${successCount} booked, ${failureCount} failed. See per-row status.`)
   439→    } else {
   440→      toast.error(`All ${failureCount} booking(s) failed. See per-row errors.`)
   441→    }
   442→
   443→    if (successCount > 0) {
   444→      queryClient.invalidateQueries({ queryKey: ['booking-workbench-bookable'] })
   445→      queryClient.invalidateQueries({ queryKey: ['booking-workbench-activity'] })
   446→      queryClient.invalidateQueries({ queryKey: ['orders'] })
   447→    }
   448→  }
   449→
   450→  // ── Outer chrome ──
   451→  return (
   452→    <div className="space-y-4">
   453→      <PageHeader
   454→        title="Booking Workbench"
   455→        description="Bulk-book confirmed orders and exchange shipments with a courier."
   456→        actions={
   457→          <Button
   458→            variant="outline"
   459→            size="sm"
   460→            onClick={() => {
   461→              bookableQuery.refetch()
   462→              integrationsQuery.refetch()
   463→            }}
   464→            disabled={bookableQuery.isFetching && !bookableQuery.isLoading}
   465→          >
   466→            <RefreshCw
   467→              className={cn('h-3.5 w-3.5', bookableQuery.isFetching && 'animate-spin')}
   468→            />
   469→            Refresh
   470→          </Button>
   471→        }
   472→      />
   473→
   474→      {courierIntegrations.length === 0 && activeTab !== 'activity' && (
   475→        <Card className="border-amber-200 bg-amber-50/40">
   476→          <CardContent className="p-4 flex items-start gap-3">
   477→            <AlertCircle className="h-5 w-5 text-amber-600 mt-0.5 shrink-0" />
   478→            <div className="text-sm text-amber-800">
   479→              No courier integrations are connected yet. Connect one in
   480→              <span className="font-medium"> Settings → Integrations</span> before booking shipments.
   481→            </div>
   482→          </CardContent>
   483→        </Card>
   484→      )}
   485→
   486→      <Tabs value={activeTab} onValueChange={onTabChange}>
   487→        <TabsList className="w-full sm:w-auto overflow-x-auto">
   488→          <TabsTrigger value="orders" className="gap-1.5">
   489→            <Truck className="h-3.5 w-3.5" />
   490→            Orders
   491→            <Badge variant="secondary" className="ml-1 text-[10px] h-4 px-1">
   492→              {bookableQuery.data?.orders.length ?? 0}
   493→            </Badge>
   494→          </TabsTrigger>
   495→          <TabsTrigger value="shipments" className="gap-1.5">
   496→            <ArrowRight className="h-3.5 w-3.5" />
   497→            Exchange Shipments
   498→            <Badge variant="secondary" className="ml-1 text-[10px] h-4 px-1">
   499→              {bookableQuery.data?.shipments.length ?? 0}
   500→            </Badge>
   501→          </TabsTrigger>
   502→          <TabsTrigger value="activity" className="gap-1.5">
   503→            <CheckCircle2 className="h-3.5 w-3.5" />
   504→            Booking Activity
   505→          </TabsTrigger>
   506→        </TabsList>
   507→
   508→        <TabsContent value="orders" className="mt-4">
   509→          <BookableTabContent
   510→            rows={filteredRows}
   511→            isLoading={bookableQuery.isLoading}
   512→            isError={bookableQuery.isError}
   513→            error={bookableQuery.error}
   514→            onRetry={() => bookableQuery.refetch()}
   515→            search={search}
   516→            setSearch={setSearch}
   517→            searchPlaceholder="Search order #, customer, phone…"
   518→            emptyHint="orders"
   519→            allChecked={allChecked}
   520→            onToggleSelectAll={toggleSelectAll}
   521→            checkedCount={checkedCount}
   522→            courierIntegrations={courierIntegrations}
   523→            integrationById={integrationById}
   524→            getRowState={getRowState}
   525→            patchRow={patchRow}
   526→            bulkApplyIntegrationId={bulkApplyIntegrationId}
   527→            onBulkApplyIntegrationIdChange={setBulkApplyIntegrationId}
   528→            onBulkApply={handleBulkApply}
   529→            onUploadBooking={handleUploadBooking}
   530→          />
   531→        </TabsContent>
   532→
   533→        <TabsContent value="shipments" className="mt-4">
   534→          <BookableTabContent
   535→            rows={filteredRows}
   536→            isLoading={bookableQuery.isLoading}
   537→            isError={bookableQuery.isError}
   538→            error={bookableQuery.error}
   539→            onRetry={() => bookableQuery.refetch()}
   540→            search={search}
   541→            setSearch={setSearch}
   542→            searchPlaceholder="Search shipment #, customer, phone…"
   543→            emptyHint="exchange shipments"
   544→            allChecked={allChecked}
   545→            onToggleSelectAll={toggleSelectAll}
   546→            checkedCount={checkedCount}
   547→            courierIntegrations={courierIntegrations}
   548→            integrationById={integrationById}
   549→            getRowState={getRowState}
   550→            patchRow={patchRow}
   551→            bulkApplyIntegrationId={bulkApplyIntegrationId}
   552→            onBulkApplyIntegrationIdChange={setBulkApplyIntegrationId}
   553→            onBulkApply={handleBulkApply}
   554→            onUploadBooking={handleUploadBooking}
   555→          />
   556→        </TabsContent>
   557→
   558→        <TabsContent value="activity" className="mt-4">
   559→          <BookingActivityTab />
   560→        </TabsContent>
   561→      </Tabs>
   562→    </div>
   563→  )
   564→}
