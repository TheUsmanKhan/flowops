   168→// ─────────────────────────────────────────────────────────────────────────────
   169→// Zod schema for the "Record a Return" form
   170→// ─────────────────────────────────────────────────────────────────────────────
   171→
   172→const recordReturnSchema = z
   173→  .object({
   174→    org_variant_id: z.string().min(1, 'Select a variant'),
   175→    quantity: z
   176→      .number({ error: 'Enter a number' })
   177→      .int('Quantity must be a whole number')
   178→      .min(1, 'Quantity must be at least 1'),
   179→    condition: z.enum(['perfect', 'good', 'open_box', 'damaged']),
   180→    fabricStitchingCost: z
   181→      .number({ error: 'Enter a number' })
   182→      .min(0, 'Cannot be negative'),
   183→    outgoingCourier: z
   184→      .number({ error: 'Enter a number' })
   185→      .min(0, 'Cannot be negative'),
   186→    returnCourier: z
   187→      .number({ error: 'Enter a number' })
   188→      .min(0, 'Cannot be negative'),
   189→    return_reason: z.enum([
   190→      'RTO',
   191→      'Refused at door',
   192→      'Size issue',
   193→      'Wrong item',
   194→      'Other',
   195→    ]),
   196→    custom_reason: z.string().optional().or(z.literal('')),
   197→    original_order_reference: z.string().optional().or(z.literal('')),
   198→    notes: z
   199→      .string()
   200→      .max(1000, 'Notes must be 1000 characters or fewer')
   201→      .optional()
   202→      .or(z.literal('')),
   203→  })
   204→  .superRefine((data, ctx) => {
   205→    const total = data.fabricStitchingCost + data.outgoingCourier + data.returnCourier
   206→    if (total <= 0) {
   207→      ctx.addIssue({
   208→        code: z.ZodIssueCode.custom,
   209→        message: 'Total cost must be greater than 0',
   210→        path: ['fabricStitchingCost'],
   211→      })
   212→    }
   213→    if (data.return_reason === 'Other' && !data.custom_reason?.trim()) {
   214→      ctx.addIssue({
   215→        code: z.ZodIssueCode.custom,
   216→        message: 'Please specify the reason',
   217→        path: ['custom_reason'],
   218→      })
   219→    }
   220→  })
   221→
   222→type RecordReturnForm = z.infer<typeof recordReturnSchema>

   228→function formatPrice(n: number): string {
   229→  return new Intl.NumberFormat('en-US', {
   230→    maximumFractionDigits: 2,
   231→    minimumFractionDigits: 0,
   232→  }).format(n)
   233→}
   234→
   235→function formatDate(iso: string | null): string {
   236→  if (!iso) return '—'
   237→  try {
   238→    const d = new Date(iso)
   239→    return d.toLocaleDateString('en-US', {
   240→      year: 'numeric',
   241→      month: 'short',
   242→      day: 'numeric',
   243→    })
   244→  } catch {
   245→    return '—'
   246→  }
   247→}
   248→
   249→function formatAttributeValues(values: Record<string, string>): string {
   250→  const entries = Object.entries(values || {})
   251→  if (entries.length === 0) return ''
   252→  return entries.map(([k, v]) => `${k}: ${v}`).join(' · ')
   253→}
   254→
   255→function getErrorMessage(err: unknown): string {
   256→  if (err instanceof FetchError) return err.message
   257→  if (err instanceof Error) return err.message
   258→  return 'Something went wrong. Please try again.'
   259→}
   260→
   261→/** Parse a numeric input value, returning 0 for empty/invalid input. */
   262→function parseNumberInput(v: string): number {
   263→  if (v === '' || v === '-') return 0
   264→  const n = Number(v)
   265→  return Number.isFinite(n) ? n : 0
   266→}

   272→export function ReturnedStitchedView() {
   273→  const queryClient = useQueryClient()
   274→  const [statusFilter, setStatusFilter] = useState<StatusFilter>('all')
   275→  const [recordOpen, setRecordOpen] = useState(false)
   276→  const [soldTarget, setSoldTarget] = useState<string | null>(null)
   277→  const [writeOffTarget, setWriteOffTarget] = useState<string | null>(null)
   278→
   279→  // ---- Stats query ----
   280→  const statsQuery = useQuery<ReturnedStats>({
   281→    queryKey: ['returned-stitched-stats'],
   282→    queryFn: () => api.get<ReturnedStats>('/api/returned-stitched/stats'),
   283→    staleTime: 30_000,
   284→  })
   285→
   286→  // ---- Items query (depends on status filter) ----
   287→  const itemsQuery = useQuery<{ items: ReturnedItem[] }>({
   288→    queryKey: ['returned-stitched', { status: statusFilter }],
   289→    queryFn: () => {
   290→      const qs = statusFilter === 'all' ? '' : `?status=${statusFilter}`
   291→      return api.get<{ items: ReturnedItem[] }>(`/api/returned-stitched${qs}`)
   292→    },
   293→    staleTime: 15_000,
   294→  })
   295→
   296→  // ---- Invalidation helper ----
   297→  const invalidateAll = () => {
   298→    void queryClient.invalidateQueries({ queryKey: ['returned-stitched'] })
   299→    void queryClient.invalidateQueries({ queryKey: ['returned-stitched-stats'] })
   300→  }
   301→
   302→  // ---- Mutations ----
   303→  const receiveMutation = useIdempotentMutation<unknown, {
   304→    org_variant_id: string
   305→    quantity: number
   306→    condition: ReturnedCondition
   307→    total_cost: number
   308→    return_reason: string
   309→    original_order_reference?: string
   310→    photos: string[]
   311→    notes?: string
   312→  }>({
   313→    url: '/api/returned-stitched',
   314→    mutationOptions: {
   315→      onSuccess: (_data, vars) => {
   316→        const msg =
   317→          vars.condition === 'damaged'
   318→            ? 'Damaged item recorded and written off.'
   319→            : 'Return recorded — item is now available stock.'
   320→        toast.success(msg)
   321→        invalidateAll()
   322→        setRecordOpen(false)
   323→      },
   324→      onError: (err) => toast.error(getErrorMessage(err)),
   325→    },
   326→  })
   327→
   328→  const markSoldMutation = useMutation({
   329→    mutationFn: async ({ id, reference }: { id: string; reference: string }) =>
   330→      api.post(`/api/returned-stitched/${id}`, {
   331→        action: 'sold',
   332→        sold_order_reference: reference,
   333→      }),
   334→    onSuccess: () => {
   335→      toast.success('Item marked as sold.')
   336→      invalidateAll()
   337→      setSoldTarget(null)
   338→    },
   339→    onError: (err) => toast.error(getErrorMessage(err)),
   340→  })
   341→
   342→  const writeOffMutation = useMutation({
   343→    mutationFn: async ({ id, reason }: { id: string; reason: string }) =>
   344→      api.post(`/api/returned-stitched/${id}`, {
   345→        action: 'write_off',
   346→        reason,
   347→      }),
   348→    onSuccess: () => {
   349→      toast.success('Item written off.')
   350→      invalidateAll()
   351→      setWriteOffTarget(null)
   352→    },
   353→    onError: (err) => toast.error(getErrorMessage(err)),
   354→  })
   355→
   356→  const items = itemsQuery.data?.items ?? []
   357→  const stats = statsQuery.data
   358→
   359→  return (
   360→    <div className="space-y-6">
   361→      <PageHeader
   362→        title="Returned Stitched Inventory"
   363→        description="Track stitched items that come back. Reuse, resell, or write them off cleanly."
   364→        actions={
   365→          <Button onClick={() => setRecordOpen(true)}>
   366→            <Plus className="h-4 w-4" /> Record a Return
   367→          </Button>
   368→        }
   369→      />
   370→
   371→      {/* ── Stats row ───────────────────────────────────────── */}
   372→      <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-3">
   373→        <StatCard
   374→          label="Available pieces"
   375→          value={stats ? `${stats.availableCount}` : null}
   376→          icon={<Package className="h-5 w-5" />}
   377→          iconClassName="bg-emerald-100 text-emerald-700"
   378→          loading={statsQuery.isLoading}
   379→        />
   380→        <StatCard
   381→          label="Total value (available)"
   382→          value={stats ? `Rs. ${formatPrice(stats.totalValue)}` : null}
   383→          icon={<DollarSign className="h-5 w-5" />}
   384→          iconClassName="bg-sky-100 text-sky-700"
   385→          loading={statsQuery.isLoading}
   386→        />
   387→        <StatCard
   388→          label="Written off this month"
   389→          value={stats ? `${stats.writtenOffThisMonth}` : null}
   390→          icon={<TrendingDown className="h-5 w-5" />}
   391→          iconClassName="bg-rose-100 text-rose-700"
   392→          loading={statsQuery.isLoading}
   393→        />
   394→      </div>
   395→
   396→      {/* ── Filter ──────────────────────────────────────────── */}
   397→      <div className="flex flex-col sm:flex-row sm:items-center gap-3 sm:justify-between">
   398→        <div className="flex items-center gap-2 text-sm text-muted-foreground">
   399→          <RotateCcw className="h-4 w-4" />
   400→          {itemsQuery.isFetching
   401→            ? 'Refreshing…'
   402→            : `${items.length} item${items.length === 1 ? '' : 's'}${
   403→                statusFilter === 'all'
   404→                  ? ''
   405→                  : ` · ${STATUS_FILTER_OPTIONS.find((o) => o.value === statusFilter)?.label ?? ''}`
   406→              }`}
   407→        </div>
   408→        <Select
   409→          value={statusFilter}
   410→          onValueChange={(v) => setStatusFilter(v as StatusFilter)}
   411→        >
   412→          <SelectTrigger className="w-full sm:w-48">
   413→            <SelectValue placeholder="Filter by status" />
   414→          </SelectTrigger>
   415→          <SelectContent>
   416→            {STATUS_FILTER_OPTIONS.map((opt) => (
   417→              <SelectItem key={opt.value} value={opt.value}>
   418→                {opt.label}
   419→              </SelectItem>
   420→            ))}
   421→          </SelectContent>
   422→        </Select>
   423→      </div>
   424→
   425→      {/* ── Table / states ──────────────────────────────────── */}
   426→      {itemsQuery.isLoading ? (
   427→        <TableSkeleton />
   428→      ) : itemsQuery.isError ? (
   429→        <Card>
   430→          <CardContent className="p-10 text-center space-y-4">
   431→            <div className="mx-auto flex h-10 w-10 items-center justify-center rounded-full bg-rose-50 text-rose-600">
   432→              <AlertCircle className="h-5 w-5" />
   433→            </div>
   434→            <p className="text-sm text-muted-foreground">
   435→              Couldn&apos;t load returned items. The server may have restarted.
   436→            </p>
   437→            <Button variant="outline" onClick={() => itemsQuery.refetch()}>
   438→              Try again
   439→            </Button>
   440→          </CardContent>
   441→        </Card>
   442→      ) : items.length === 0 ? (
   443→        <Card>
   444→          <CardContent className="p-10 text-center space-y-4">
   445→            <div className="mx-auto flex h-12 w-12 items-center justify-center rounded-full bg-muted text-muted-foreground">
   446→              <RotateCcw className="h-6 w-6" />
   447→            </div>
   448→            <div className="space-y-1">
   449→              <p className="text-sm font-medium">No returned items yet.</p>
   450→              <p className="text-sm text-muted-foreground">
   451→                Record a return when a stitched item comes back.
   452→              </p>
   453→            </div>
   454→            <Button onClick={() => setRecordOpen(true)}>
   455→              <Plus className="h-4 w-4" /> Record a Return
   456→            </Button>
   457→          </CardContent>
   458→        </Card>
   459→      ) : (
   460→        <Card>
   461→          <CardContent className="p-0">
   462→            <Table>
   463→              <TableHeader>
   464→                <TableRow>
   465→                  <TableHead>Variant</TableHead>
   466→                  <TableHead className="w-[60px] text-center">Qty</TableHead>
   467→                  <TableHead className="w-[110px]">Condition</TableHead>
   468→                  <TableHead className="w-[110px] text-right">Cost</TableHead>
   469→                  <TableHead className="w-[120px]">Status</TableHead>
   470→                  <TableHead className="w-[120px]">Received</TableHead>
   471→                  <TableHead className="w-[160px] text-right">Actions</TableHead>
   472→                </TableRow>
   473→              </TableHeader>
   474→              <TableBody>
   475→                {items.map((item) => (
   476→                  <ReturnedRow
   477→                    key={item.id}
   478→                    item={item}
   479→                    onMarkSold={(id) => setSoldTarget(id)}
   480→                    onWriteOff={(id) => setWriteOffTarget(id)}
   481→                  />
   482→                ))}
   483→              </TableBody>
   484→            </Table>
   485→          </CardContent>
   486→        </Card>
   487→      )}
   488→
   489→      {/* ── Record a Return dialog ──────────────────────────── */}
   490→      <RecordReturnDialog
   491→        open={recordOpen}
   492→        onOpenChange={(o) => {
   493→          if (!receiveMutation.isPending) setRecordOpen(o)
   494→        }}
   495→        submitting={receiveMutation.isPending}
   496→        onSubmit={(payload) => receiveMutation.mutate(payload)}
   497→      />
   498→
   499→      {/* ── Mark Sold / Write-off dialogs ───────────────────── */}
   500→      <MarkSoldDialog
   501→        open={soldTarget !== null}
   502→        onOpenChange={(o) => {
   503→          if (!markSoldMutation.isPending) setSoldTarget(o ? soldTarget : null)
   504→        }}
   505→        submitting={markSoldMutation.isPending}
   506→        onSubmit={(reference) => {
   507→          if (soldTarget) {
   508→            markSoldMutation.mutate({ id: soldTarget, reference })
   509→          }
   510→        }}
   511→      />
   512→
   513→      <WriteOffDialog
   514→        open={writeOffTarget !== null}
   515→        onOpenChange={(o) => {
   516→          if (!writeOffMutation.isPending) setWriteOffTarget(o ? writeOffTarget : null)
   517→        }}
   518→        submitting={writeOffMutation.isPending}
   519→        onSubmit={(reason) => {
   520→          if (writeOffTarget) {
   521→            writeOffMutation.mutate({ id: writeOffTarget, reason })
   522→          }
   523→        }}
   524→      />
   525→    </div>
   526→  )
   527→}
