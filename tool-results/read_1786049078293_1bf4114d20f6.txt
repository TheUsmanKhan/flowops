     3→     3→/**
     4→     4→ * BookingWorkbenchView (v2) — bulk courier-booking workbench with 3 tabs.
     5→     5→ *
     6→     6→ *  1. Orders             — GET /api/booking-workbench/bookable → data.orders
     7→     7→ *  2. Exchange Shipments — GET /api/booking-workbench/bookable → data.shipments
     8→     8→ *  3. Booking Activity   — GET /api/booking-workbench/activity?date_from=&date_to=
     9→     9→ *
    10→    10→ * Key behaviours:
    11→    11→ *  • Per-row courier <Select> — defaults to row.recommendedCourierCompanyIntegrationId;
    12→    12→ *    drives that row's <CityAutocomplete providerKey>.
    13→    13→ *  • Bulk Apply — toolbar dropdown + "Apply to Selected" sets the courier on all
    14→    14→ *    CHECKED rows in the active tab only.
    15→    15→ *  • Weight auto-compute — each row's default orderType is computed via
    16→    16→ *    calculateOrderWeightKg + determinePostExOrderType. If hasMissingWeight,
    17→    17→ *    a ⚠️ tooltip is shown next to the Order Type dropdown. Still editable.
    18→    18→ *  • Upload Booking — sequentially POSTs /api/booking-workbench/book for each
    19→    19→ *    checked row using THAT ROW's courier integration id. Failures don't block
    20→    20→ *    other rows. On success the row shows ✅ tracking number + auto-unchecks.
    21→    21→ *  • After any successful booking, ['booking-workbench-bookable'] and
    22→    22→ *    ['booking-workbench-activity'] queries are invalidated.
    23→    23→ */

   195→   195→function rowKey(row: BookableRow): string {
   196→   196→  return `${row.type}:${row.id}`
   197→   197→}
   198→   198→
   199→   199→function computeRowOrderType(row: BookableRow): {
   200→   200→  orderType: PostExOrderType
   201→   201→  hasMissingWeight: boolean
   202→   202→  totalWeightKg: number
   203→   203→} {
   204→   204→  const weightResult = calculateOrderWeightKg(
   205→   205→    row.items.map((i) => ({
   206→   206→      quantity: i.quantity,
   207→   207→      variant: { weightKg: i.weightKg },
   208→   208→    })),
   209→   209→  )
   210→   210→  const isExchangeReplacement =
   211→   211→    row.type === 'exchange_shipment' && row.exchangeMethod === 'courier_replacement'
   212→   212→  return {
   213→   213→    orderType: determinePostExOrderType(
   214→   214→      weightResult.totalWeightKg,
   215→   215→      weightResult.hasMissingWeight,
   216→   216→      isExchangeReplacement,
   217→   217→    ),
   218→   218→    hasMissingWeight: weightResult.hasMissingWeight,
   219→   219→    totalWeightKg: weightResult.totalWeightKg,
   220→   220→  }
   221→   221→}
   222→   222→
   223→   223→function defaultRowState(row: BookableRow): RowState {
   224→   224→  const computed = computeRowOrderType(row)
   225→   225→  return {
   226→   226→    customerName: row.customerName ?? '',
   227→   227→    customerPhone: row.customerPhone ?? '',
   228→   228→    deliveryAddress: row.deliveryAddress ?? '',
   229→   229→    deliveryCity: row.deliveryCity ?? '',
   230→   230→    codAmount: String(row.codAmount ?? 0),
   231→   231→    orderType: computed.orderType,
   232→   232→    hasMissingWeight: computed.hasMissingWeight,
   233→   233→    totalWeightKg: computed.totalWeightKg,
   234→   234→    companyIntegrationId: row.recommendedCourierCompanyIntegrationId ?? '',
   235→   235→    // Seed from the stored universal courier reference fields (migration 015)
   236→   236→    orderRefNumber: row.orderRefNumber ?? row.referenceNumber ?? '',
   237→   237→    orderDetail: row.orderDetail ?? '',
   238→   238→    transactionNotes: row.notesForCourier ?? '',
   239→   239→    showAdvanced: false,
   240→   240→    checked: false,
   241→   241→    result: null,
   242→   242→    isBooking: false,
   243→   243→  }
   244→   244→}

   248→   248→export function BookingWorkbenchView() {
   249→   249→  const queryClient = useQueryClient()
   250→   250→  const [activeTab, setActiveTab] = useState<'orders' | 'shipments' | 'activity'>('orders')
   251→   251→  const [search, setSearch] = useState('')
   252→   252→  const [bulkApplyIntegrationId, setBulkApplyIntegrationId] = useState('')
   253→   253→  const [rowStates, setRowStates] = useState<Record<string, RowState>>({})
   254→   254→
   255→   255→  // ── Data: bookable orders + shipments ──
   256→   256→  const bookableQuery = useQuery<BookableResponse>({
   257→   257→    queryKey: ['booking-workbench-bookable'],
   258→   258→    queryFn: () => api.get<BookableResponse>('/api/booking-workbench/bookable'),
   259→   259→    staleTime: 15_000,
   260→   260→  })
   261→   261→
   262→   262→  // ── Data: courier integrations ──
   263→   263→  const integrationsQuery = useQuery<IntegrationsResponse>({
   264→   264→    queryKey: ['integrations', 'courier'],
   265→   265→    queryFn: () => api.get<IntegrationsResponse>('/api/integrations?category=courier'),
   266→   266→    staleTime: 30_000,
   267→   267→  })
   268→   268→
   269→   269→  const courierIntegrations = useMemo(
   270→   270→    () =>
   271→   271→      (integrationsQuery.data?.integrations ?? []).filter(
   272→   272→        (i) => i.isActive && i.provider?.category === 'courier',
   273→   273→      ),
   274→   274→    [integrationsQuery.data],
   275→   275→  )
   276→   276→
   277→   277→  const integrationById = useMemo(() => {
   278→   278→    const m = new Map<string, CompanyIntegration>()
   279→   279→    for (const ci of courierIntegrations) m.set(ci.id, ci)
   280→   280→    return m
   281→   281→  }, [courierIntegrations])
   282→   282→
   283→   283→  function onTabChange(v: string) {
   284→   284→    setActiveTab(v as 'orders' | 'shipments' | 'activity')
   285→   285→    setSearch('')
   286→   286→  }
   287→   287→
   288→   288→  // ── Rows currently visible in the active bookable tab ──
   289→   289→  const rowsForActiveTab = useMemo<BookableRow[]>(() => {
   290→   290→    if (activeTab === 'orders') return bookableQuery.data?.orders ?? []
   291→   291→    if (activeTab === 'shipments') return bookableQuery.data?.shipments ?? []
   292→   292→    return []
   293→   293→  }, [bookableQuery.data, activeTab])
   294→   294→
   295→   295→  const filteredRows = useMemo(() => {
   296→   296→    if (!search.trim()) return rowsForActiveTab
   297→   297→    const q = search.trim().toLowerCase()
   298→   298→    return rowsForActiveTab.filter(
   299→   299→      (r) =>
   300→   300→        r.referenceNumber.toLowerCase().includes(q) ||
   301→   301→        r.customerName.toLowerCase().includes(q) ||
   302→   302→        r.customerPhone?.toLowerCase().includes(q) ||
   303→   303→        r.originalOrderNumber?.toLowerCase().includes(q),
   304→   304→    )
   305→   305→  }, [rowsForActiveTab, search])
   306→   306→
   307→   307→  // ── Row state helpers ──
   308→   308→  function getRowState(row: BookableRow): RowState {
   309→   309→    const key = rowKey(row)
   310→   310→    return rowStates[key] ?? defaultRowState(row)
   311→   311→  }
   312→   312→
   313→   313→  function patchRow(row: BookableRow, patch: Partial<RowState>) {
   314→   314→    const key = rowKey(row)
   315→   315→    setRowStates((prev) => ({
   316→   316→      ...prev,
   317→   317→      [key]: {
   318→   318→        ...(prev[key] ?? defaultRowState(row)),
   319→   319→        ...patch,
   320→   320→        // Any edit clears the previous booking result so the user knows to re-submit.
   321→   321→        result: patch.result !== undefined ? patch.result : null,
   322→   322→      },
   323→   323→    }))
   324→   324→  }
   325→   325→
   326→   326→  // ── Select-all ──
   327→   327→  const checkedCount = filteredRows.filter((r) => getRowState(r).checked).length
   328→   328→  const allChecked = filteredRows.length > 0 && checkedCount === filteredRows.length
   329→   329→
   330→   330→  function toggleSelectAll() {
   331→   331→    const next = { ...rowStates }
   332→   332→    const target = !allChecked
   333→   333→    for (const r of filteredRows) {
   334→   334→      const key = rowKey(r)
   335→   335→      const cur = next[key] ?? defaultRowState(r)
   336→   336→      next[key] = { ...cur, checked: target }
   337→   337→    }
   338→   338→    setRowStates(next)
   339→   339→  }
   340→   340→
   341→   341→  // ── Bulk Apply ──
   342→   342→  function handleBulkApply() {
   343→   343→    if (!bulkApplyIntegrationId) {
   344→   344→      toast.error('Pick a courier in the Bulk Apply dropdown first.')
   345→   345→      return
   346→   346→    }
   347→   347→    const checkedRows = filteredRows.filter((r) => getRowState(r).checked)
   348→   348→    if (checkedRows.length === 0) {
   349→   349→      toast.error('No rows selected. Tick at least one row first.')
   350→   350→      return
   351→   351→    }
   352→   352→    setRowStates((prev) => {
   353→   353→      const next = { ...prev }
   354→   354→      for (const r of checkedRows) {
   355→   355→        const key = rowKey(r)
   356→   356→        const cur = next[key] ?? defaultRowState(r)
   357→   357→        next[key] = { ...cur, companyIntegrationId: bulkApplyIntegrationId, result: null }
   358→   358→      }
   359→   359→      return next
   360→   360→    })
   361→   361→    const integ = integrationById.get(bulkApplyIntegrationId)
   362→   362→    toast.success(
   363→   363→      `Applied ${integ?.provider?.providerName ?? 'courier'} to ${checkedRows.length} row(s).`,
   364→   364→    )
   365→   365→  }
   366→   366→
   367→   367→  // ── Booking mutation (per-row) ──
   368→   368→  const bookMutation = useMutation({
   369→   369→    mutationFn: async (body: BookRequest): Promise<BookSuccess> =>
   370→   370→      api.post<BookSuccess>('/api/booking-workbench/book', body),
   371→   371→  })
   372→   372→
   373→   373→  async function handleUploadBooking() {
   374→   374→    const checkedRows = filteredRows.filter((r) => {
   375→   375→      const s = getRowState(r)
   376→   376→      return s.checked && !s.isBooking
   377→   377→    })
   378→   378→    if (checkedRows.length === 0) {
   379→   379→      toast.error('No rows selected. Tick the checkbox on at least one row.')
   380→   380→      return
   381→   381→    }
   382→   382→    const missingCourier = checkedRows.find((r) => !getRowState(r).companyIntegrationId)
   383→   383→    if (missingCourier) {
   384→   384→      toast.error(`Row ${missingCourier.referenceNumber} has no courier selected.`)
   385→   385→      return
   386→   386→    }
   387→   387→
   388→   388→    toast.info(`Booking ${checkedRows.length} row(s)…`)
   389→   389→
   390→   390→    let successCount = 0
   391→   391→    let failureCount = 0
   392→   392→
   393→   393→    // Sequential — external courier APIs typically don't tolerate parallel hits,
   394→   394→    // and per-row errors must not block other rows.
   395→   395→    for (const row of checkedRows) {
   396→   396→      const state = getRowState(row)
   397→   397→      const body: BookRequest = {
   398→   398→        companyIntegrationId: state.companyIntegrationId,
   399→   399→        customerName: state.customerName.trim() || undefined,
   400→   400→        customerPhone: state.customerPhone.trim() || undefined,
   401→   401→        deliveryAddress: state.deliveryAddress.trim() || undefined,
   402→   402→        deliveryCity: state.deliveryCity.trim() || undefined,
   403→   403→        codAmount: Number(state.codAmount) || 0,
   404→   404→        orderType: state.orderType,
   405→   405→        // Universal courier reference fields (migration 015) — only send if
   406→   406→        // the user overrode the seeded value (the backend falls back to the
   407→   407→        // stored value when these are undefined).
   408→   408→        orderRefNumber: state.orderRefNumber.trim() || undefined,
   409→   409→        itemDescription: state.orderDetail.trim() || undefined,
   410→   410→        transactionNotes: state.transactionNotes.trim() || undefined,
   411→   411→      }
   412→   412→      if (row.type === 'order') body.orderId = row.id
   413→   413→      else body.shipmentId = row.id
   414→   414→
   415→   415→      patchRow(row, { isBooking: true, result: null })
   416→   416→
   417→   417→      try {
   418→   418→        const result = await bookMutation.mutateAsync(body)
   419→   419→        successCount++
   420→   420→        patchRow(row, {
   421→   421→          isBooking: false,
   422→   422→          checked: false,
   423→   423→          result: { ok: true, trackingNumber: result.trackingNumber, orderType: result.orderType },
   424→   424→        })
   425→   425→      } catch (err) {
   426→   426→        failureCount++
   427→   427→        const message =
   428→   428→          err instanceof FetchError ? err.message
   429→   429→          : err instanceof Error ? err.message
   430→   430→          : 'Booking failed'
   431→   431→        patchRow(row, { isBooking: false, result: { ok: false, error: message } })
   432→   432→      }
   433→   433→    }
   434→   434→
   435→   435→    if (successCount > 0 && failureCount === 0) {
   436→   436→      toast.success(`All ${successCount} row(s) booked successfully.`)
   437→   437→    } else if (successCount > 0 && failureCount > 0) {
   438→   438→      toast.warning(`${successCount} booked, ${failureCount} failed. See per-row status.`)
   439→   439→    } else {
   440→   440→      toast.error(`All ${failureCount} booking(s) failed. See per-row errors.`)
   441→   441→    }
   442→   442→
   443→   443→    if (successCount > 0) {
   444→   444→      queryClient.invalidateQueries({ queryKey: ['booking-workbench-bookable'] })
   445→   445→      queryClient.invalidateQueries({ queryKey: ['booking-workbench-activity'] })
   446→   446→      queryClient.invalidateQueries({ queryKey: ['orders'] })
   447→   447→    }
   448→   448→  }
   449→   449→
   450→   450→  // ── Outer chrome ──
   451→   451→  return (
   452→   452→    <div className="space-y-4">
   453→   453→      <PageHeader
   454→   454→        title="Booking Workbench"
   455→   455→        description="Bulk-book confirmed orders and exchange shipments with a courier."
   456→   456→        actions={
   457→   457→          <Button
   458→   458→            variant="outline"
   459→   459→            size="sm"
   460→   460→            onClick={() => {
   461→   461→              bookableQuery.refetch()
   462→   462→              integrationsQuery.refetch()
   463→   463→            }}
   464→   464→            disabled={bookableQuery.isFetching && !bookableQuery.isLoading}
   465→   465→          >
   466→   466→            <RefreshCw
   467→   467→              className={cn('h-3.5 w-3.5', bookableQuery.isFetching && 'animate-spin')}
   468→   468→            />
   469→   469→            Refresh
   470→   470→          </Button>
   471→   471→        }
