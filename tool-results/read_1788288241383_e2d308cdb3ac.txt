   192→export default function Page() {
   193→  const { hydrated, route, user, activeCompany, employee, setSession, setHydrated, navigate } =
   194→    useAppStore()
   195→
   196→  // Track whether this is the first session fetch (cold start) vs a background
   197→  // refetch (tab refocus, reconnect, invalidation). On the first fetch we show
   198→  // a loading spinner; on background refetches we update the store silently.
   199→  const isFirstFetchRef = useRef(true)
   200→
   201→  // ─── Session hydration via TanStack Query ──────────────────────────
   202→  // This query intentionally overrides the app's global refetchOnWindowFocus:
   203→  // false default. Session validity (active employee status, permissions,
   204→  // platform-level access) is the one place where catching a change quickly
   205→  // after the user returns to a background tab actually matters — e.g., an
   206→  // employee terminated while their tab sat in the background should have
   207→  // their UI reflect that promptly on refocus, not only on their next manual
   208→  // page refresh. Every other view in this app is correctly left on the
   209→  // global default because those views display data, not gate
   210→  // security-sensitive UI. Do not "clean up" this override to match the
   211→  // global default in future refactors.
   212→  const sessionQuery = useQuery<SessionResponse>({
   213→    queryKey: ['session', 'me'],
   214→    queryFn: () => api.get<SessionResponse>('/api/auth/me'),
   215→    staleTime: 60_000,
   216→    refetchOnWindowFocus: true,
   217→    refetchOnReconnect: true,
   218→    retry: 1,
   219→  })
   220→
   221→  // Wire session data into the Zustand store.
   222→  useEffect(() => {
   223→    const session = sessionQuery.data
   224→    if (!session) return
   225→
   226→    // On the very first successful fetch, restore the URL route + set hydrated.
   227→    if (isFirstFetchRef.current) {
   228→      isFirstFetchRef.current = false
   229→      const urlRoute = queryToRoute()
   230→      const currentRoute = useAppStore.getState().route
   231→
   232→      setSession({
   233→        user: session.user,
   234→        activeCompany: session.activeCompany,
   235→        companies: session.companies,
   236→        employee: session.employee ?? undefined,
   237→      })
   238→
   239→      // If the URL has a route, restore it (overrides the default 'login')
   240→      if (urlRoute && urlRoute.name !== currentRoute.name) {
   241→        if (session.user) {
   242→          useAppStore.getState().navigate(urlRoute)
   243→        }
   244→      }
   245→    } else {
   246→      // Background refetch: update the store ONLY if the session data
   247→      // actually changed, so the UI updates without a visible loading flicker.
   248→      const currentUser = useAppStore.getState().user
   249→      const currentActiveCompanyId = useAppStore.getState().activeCompany?.id
   250→      const newActiveCompanyId = session.activeCompany?.id
   251→      const newPermissions = session.employee?.permissions ?? []
   252→      const currentPermissions = useAppStore.getState().employee?.permissions ?? []
   253→
   254→      const userChanged = currentUser?.id !== session.user?.id
   255→      const companyChanged = currentActiveCompanyId !== newActiveCompanyId
   256→      const permissionsChanged =
   257→        newPermissions.length !== currentPermissions.length ||
   258→        newPermissions.some((p, i) => p !== currentPermissions[i])
   259→
   260→      if (userChanged || companyChanged || permissionsChanged || !currentUser) {
   261→        setSession({
   262→          user: session.user,
   263→          activeCompany: session.activeCompany,
   264→          companies: session.companies,
   265→          employee: session.employee ?? undefined,
   266→        })
   267→      }
   268→    }
   269→  }, [sessionQuery.data, setSession])
   270→
   271→  // On query error (401, network failure), mark hydrated so the login screen shows.
   272→  useEffect(() => {
   273→    if (sessionQuery.error && !hydrated) {
   274→      setHydrated(true)
   275→    }
   276→  }, [sessionQuery.error, hydrated, setHydrated])
   277→
   278→  // Mark hydrated on first successful fetch (if not already set by setSession).
   279→  useEffect(() => {
   280→    if (sessionQuery.isSuccess && !hydrated) {
   281→      setHydrated(true)
   282→    }
   283→  }, [sessionQuery.isSuccess, hydrated, setHydrated])
   284→
   285→  // Browser back/forward: sync URL → Zustand state
   286→  // NOTE: The useFormGuard's use-browser-back-guard also listens to popstate.
   287→  // When the guard is active (form is dirty), it re-pushes the URL to cancel
   288→  // the back navigation. We need to make sure our handler doesn't fight it.
   289→  // We use a global flag that the guard sets when it's intercepting.
   290→  useEffect(() => {
   291→    function handlePopState() {
   292→      // If the form guard is intercepting, skip our sync — the guard will
   293→      // handle the modal, and if the user confirms, the guard calls
   294→      // window.history.back() which will trigger another popstate that we'll handle.
   295→      if (typeof window !== 'undefined' && window.__formGuardIntercepting) {
   296→        return
   297→      }
   298→      const urlRoute = queryToRoute()
   299→      if (urlRoute) {
   300→        // Use set() directly, NOT navigate(), to avoid pushing a new history entry
   301→        useAppStore.setState({ route: urlRoute })
   302→      }
   303→    }
   304→    window.addEventListener('popstate', handlePopState)
   305→    return () => window.removeEventListener('popstate', handlePopState)
   306→  }, [])
   307→
   308→  // URL sync: ensure the address bar always reflects the current route
   309→  // (covers the case where user lands on "/" with no query string)
   310→  // Must be before any early returns to satisfy React Hooks rules.
   311→  useEffect(() => {
   312→    if (hydrated) {
   313→      replaceRouteInURL(route)
   314→    }
   315→     
   316→  }, [hydrated, route.name, 'id' in route ? route.id : '', 'token' in route ? route.token : ''])
   317→
   318→  // Loading screen while hydrating (first session fetch only).
   319→  // On background refetches, sessionQuery.isFetching is true but we DON'T show
   320→  // a spinner — TanStack Query returns cached data instantly while revalidating.
   321→  if (!hydrated) {
   322→    return (
   323→      <div className="min-h-screen flex items-center justify-center bg-background">
   324→        <Loader2 className="h-6 w-6 animate-spin text-primary" />
   325→      </div>
   326→    )
   327→  }
   328→
   329→  // ---- Unauthenticated views ----
   330→  if (!user) {
   331→    if (route.name === 'register') {
   332→      return (
   333→        <AuthShell
   334→          title="Create your account"
   335→          subtitle="Start managing your e-commerce operations in minutes."
   336→        >
   337→          <RegisterForm />
   338→        </AuthShell>
   339→      )
   340→    }
   341→    if (route.name === 'forgot') {
   342→      return (
   343→        <AuthShell
   344→          title="Reset your password"
   345→          subtitle="We'll email you a secure recovery link."
   346→        >
   347→          <ForgotPasswordForm />
   348→        </AuthShell>
   349→      )
   350→    }
   351→    if (route.name === 'reset') {
   352→      return (
   353→        <AuthShell
   354→          title="Set a new password"
   355→          subtitle="Choose a strong password for your account."
   356→        >
   357→          <ResetPasswordForm token={route.token} />
   358→        </AuthShell>
   359→      )
   360→    }
   361→    // default: login
   362→    return (
   363→      <AuthShell
   364→        title="Welcome back"
   365→        subtitle="Sign in to your FlowOps workspace."
   366→      >
   367→        <LoginForm />
   368→      </AuthShell>
   369→    )
   370→  }
   371→
   372→  // ---- Authenticated but not onboarded ----
   373→  // Allow create-organization even before onboarding completes.
   374→  if (user && route.name === 'create-organization') {
   375→    return (
   376→      <DashboardShell>
   377→        <CreateOrganizationView onBack={() => navigate({ name: user.isOnboarded && activeCompany ? 'dashboard' : 'onboarding' })} />
   378→      </DashboardShell>
   379→    )
   380→  }
   381→  if (!user.isOnboarded || !activeCompany) {
   382→    if (route.name === 'create-company') {
   383→      return (
   384→        <DashboardShell>
   385→          <CreateCompanyView onBack={() => navigate({ name: 'onboarding' })} />
   386→        </DashboardShell>
   387→      )
   388→    }
   389→    return <OnboardingView />
   390→  }
   391→
   392→  // ---- Authenticated + onboarded: dashboard shell ----
   393→  return <DashboardShell>{renderRoute(route, employee)}</DashboardShell>
   394→}
