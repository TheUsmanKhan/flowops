   158→/// Company-scoped roles. System roles (Owner/Founder/Co-Founder/Investor) are
   159→/// elevated and bypass all permission checks. Custom roles use role_permissions.
   160→model Role {
   161→  id            String   @id @default(cuid())
   162→  companyId     String
   163→  company       Company  @relation(fields: [companyId], references: [id], onDelete: Cascade)
   164→  name          String
   165→  description   String?
   166→  roleTier      String   @default("standard") // elevated | standard
   167→  isSystemRole  Boolean  @default(false)
   168→  systemRoleKey String?  // owner | founder | co_founder | investor | null
   169→  // NOTE: systemRoleKey is unique PER COMPANY (not globally), so every
   170→  // company can seed its own owner/founder/co_founder/investor roles.
   171→  isActive      Boolean  @default(true)
   172→  createdById   String?
   173→  createdBy     Profile? @relation("RoleCreator", fields: [createdById], references: [id])
   174→  createdAt     DateTime @default(now())
   175→  updatedAt     DateTime @updatedAt
   176→
   177→  employees       Employee[]
   178→  rolePermissions RolePermission[]
   179→  invitations     Invitation[]
   180→
   181→  @@unique([companyId, name])
   182→  @@unique([companyId, systemRoleKey])
   183→}
