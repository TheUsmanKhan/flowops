    36→export async function listAvailableProviders(category?: 'courier' | 'ecommerce'): Promise<ActionResult<{
    37→  providers: Array<{
    38→    id: string
    39→    providerKey: string
    40→    providerName: string
    41→    category: string
    42→    logoUrl: string | null
    43→    authType: string
    44→    supportsWebhook: boolean
    45→    configSchema: string
    46→    capabilities: string
    47→    adapterStatus: string // 'live' | 'framework_ready' | 'stub' — from registry
    48→  }>
    49→}>> {
    50→  try {
    51→    const ctx = await getWorkspace()
    52→    const where: { category?: string; isActive: boolean } = { isActive: true }
    53→    if (category) where.category = category
    54→
    55→    const providers = await db.integrationProvider.findMany({
    56→      where,
    57→      orderBy: { category: 'asc' },
    58→    })
    59→
    60→    return {
    61→      success: true,
    62→      data: {
    63→        providers: providers.map((p) => ({
    64→          id: p.id,
    65→          providerKey: p.providerKey,
    66→          providerName: p.providerName,
    67→          category: p.category,
    68→          logoUrl: p.logoUrl,
    69→          authType: p.authType,
    70→          supportsWebhook: p.supportsWebhook,
    71→          configSchema: p.configSchema,
    72→          capabilities: p.capabilities,
    73→          adapterStatus: getAdapterStatus(p.providerKey),
    74→        })),
    75→      },
    76→    }
    77→  } catch (err) {
    78→    return {
    79→      success: false,
    80→      error: err instanceof Error ? err.message : 'Failed to list providers',
    81→    }
    82→  }
    83→}

    89→export async function listCompanyIntegrations(category?: 'courier' | 'ecommerce'): Promise<ActionResult<{
    90→  integrations: Array<{
    91→    id: string
    92→    connectionName: string
    93→    isActive: boolean
    94→    isDefault: boolean
    95→    connectionStatus: string
    96→    lastSyncAt: Date | null
    97→    lastError: string | null
    98→    webhookEndpointId: string | null
    99→    webhookUrl: string | null
   100→    createdAt: Date
   101→    provider: {
   102→      id: string
   103→      providerKey: string
   104→      providerName: string
   105→      category: string
   106→      logoUrl: string | null
   107→      authType: string
   108→      supportsWebhook: boolean
   109→    }
   110→  }>
   111→}>> {
   112→  try {
   113→    const ctx = await getWorkspace()
   114→
   115→    const integrations = await db.companyIntegration.findMany({
   116→      where: {
   117→        companyId: ctx.company.id,
   118→        ...(category ? { provider: { category } } : {}),
   119→      },
   120→      include: {
   121→        provider: {
   122→          select: {
   123→            id: true,
   124→            providerKey: true,
   125→            providerName: true,
   126→            category: true,
   127→            logoUrl: true,
   128→            authType: true,
   129→            supportsWebhook: true,
   130→            configSchema: true, // needed by ReconnectDialog to render credential fields
   131→          },
   132→        },
   133→      },
   134→      orderBy: { createdAt: 'desc' },
   135→    })
   136→
   137→    const appUrl = process.env.APP_URL || 'http://localhost:3000'
   138→
   139→    return {
   140→      success: true,
   141→      data: {
   142→        integrations: integrations.map((i) => ({
   143→          id: i.id,
   144→          connectionName: i.connectionName,
   145→          isActive: i.isActive,
   146→          isDefault: i.isDefault,
   147→          connectionStatus: i.connectionStatus,
   148→          lastSyncAt: i.lastSyncAt,
   149→          lastError: i.lastError,
   150→          webhookEndpointId: i.webhookEndpointId,
   151→          webhookUrl: i.webhookEndpointId
   152→            ? `${appUrl}/api/webhooks/${i.provider.providerKey}/${i.webhookEndpointId}`
   153→            : null,
   154→          createdAt: i.createdAt,
   155→          provider: i.provider,
   156→          // NOTE: credentialsEncrypted is intentionally EXCLUDED — never
   157→          // send encrypted credentials to the client, even encrypted.
   158→        })),
   159→      },
   160→    }
   161→  } catch (err) {
   162→    return {
   163→      success: false,
   164→      error: err instanceof Error ? err.message : 'Failed to list integrations',
   165→    }
   166→  }
   167→}

   169→// ──────────────────────────────────────────────────────────────
   170→// connectIntegration
   171→// ──────────────────────────────────────────────────────────────
   172→
   173→export async function connectIntegration(input: {
   174→  providerId: string
   175→  connectionName: string
   176→  credentials: Record<string, string>
   177→}): Promise<ActionResult<{
   178→  companyIntegrationId: string
   179→  webhookUrl?: string
   180→}>> {
   181→  try {
   182→    const ctx = await getWorkspace()
   183→    if (!isElevated(ctx)) {
   184→      return { success: false, error: 'Only elevated roles (Owner/Founder/Co-Founder/Investor) can configure integrations.' }
   185→    }
   186→
   187→    // Fetch the provider to get config_schema + supports_webhook
   188→    const provider = await db.integrationProvider.findUnique({
   189→      where: { id: input.providerId },
   190→    })
   191→    if (!provider) return { success: false, error: 'Provider not found' }
   192→
   193→    // Validate required credential fields from config_schema
   194→    let configFields: Array<{ key: string; label: string; type: string; required: boolean }> = []
   195→    try {
   196→      configFields = JSON.parse(provider.configSchema)
   197→    } catch {
   198→      return { success: false, error: 'Provider config_schema is invalid' }
   199→    }
   200→
   201→    for (const field of configFields) {
   202→      if (field.required) {
   203→        const value = input.credentials[field.key]
   204→        if (!value || !value.trim()) {
   205→          return { success: false, error: `Missing required field: ${field.label}` }
   206→        }
   207→      }
   208→    }
   209→
   210→    // Encrypt credentials
   211→    const credentialsEncrypted = encryptCredentials(input.credentials)
   212→
   213→    // ── Find-or-reactivate ──
   214→    // If there's an EXISTING integration for this provider + company
   215→    // (including disconnected ones), reactivate it with the new credentials.
   216→    // This gives users a clean "connect from scratch" experience after
   217→    // disconnecting — the provider appears in "Available to Connect",
   218→    // they click Connect, enter new credentials, and the old row is
   219→    // reactivated (preserving audit history) instead of creating a duplicate.
   220→    const existing = await db.companyIntegration.findFirst({
   221→      where: { companyId: ctx.company.id, providerId: input.providerId },
   222→      select: { id: true, isActive: true },
   223→    })
   224→
   225→    let integration: { id: string }
   226→    let webhookEndpointId: string | null = null
   227→    let isNewConnection = false
   228→
   229→    if (existing) {
   230→      // Reactivate the existing integration with new credentials.
   231→      // Generate new webhook endpoint ID + secret if the provider supports
   232→      // webhooks and the old ones were wiped on disconnect.
   233→      let webhookSecret: string | null = null
   234→      if (provider.supportsWebhook) {
   235→        webhookEndpointId = generateWebhookEndpointId()
   236→        webhookSecret = generateWebhookSecret()
   237→      }
   238→
   239→      integration = await db.companyIntegration.update({
   240→        where: { id: existing.id },
   241→        data: {
   242→          connectionName: input.connectionName,
   243→          credentialsEncrypted,
   244→          isActive: true,
   245→          connectionStatus: 'pending',
   246→          lastError: null,
   247→          webhookEndpointId,
   248→          webhookSecret,
   249→          // Shopify Adapter Foundation: track the employee re-connecting this
   250→          // integration (used by the webhook route to build an injected
   251→          // WorkspaceContext for webhook-triggered actions).
   252→          connectedByEmployeeId: ctx.employee.id,
   253→        },
   254→        select: { id: true },
   255→      })
   256→
   257→      insertAuditLog({
   258→        action: existing.isActive ? 'integration.credentials_updated' : 'integration.reconnected',
   259→        entityType: 'company_integration',
   260→        entityId: integration.id,
   261→        companyId: ctx.company.id,
   262→        organizationId: ctx.company.organizationId,
   263→        userId: ctx.user.id,
   264→        employeeId: ctx.employee.id,
   265→        newValues: {
   266→          provider: provider.providerKey,
   267→          connectionName: input.connectionName,
   268→          supportsWebhook: provider.supportsWebhook,
   269→        },
   270→      })
   271→    } else {
   272→      // No existing integration — create a fresh one.
   273→      let webhookSecret: string | null = null
   274→      if (provider.supportsWebhook) {
   275→        webhookEndpointId = generateWebhookEndpointId()
   276→        webhookSecret = generateWebhookSecret()
   277→      }
   278→
   279→      try {
   280→        integration = await db.companyIntegration.create({
   281→          data: {
   282→            companyId: ctx.company.id,
   283→            organizationId: ctx.company.organizationId,
   284→            providerId: input.providerId,
   285→            connectionName: input.connectionName,
   286→            credentialsEncrypted,
   287→            webhookEndpointId,
   288→            webhookSecret,
   289→            connectionStatus: 'pending',
   290→            createdBy: ctx.employee.id,
   291→            // Shopify Adapter Foundation: track the employee connecting this
   292→            // integration (used by the webhook route to build an injected
   293→            // WorkspaceContext for webhook-triggered actions).
   294→            connectedByEmployeeId: ctx.employee.id,
   295→          },
   296→          select: { id: true },
   297→        })
   298→        isNewConnection = true
   299→
   300→        insertAuditLog({
   301→          action: 'integration.connected',
   302→          entityType: 'company_integration',
   303→          entityId: integration.id,
   304→          companyId: ctx.company.id,
   305→          organizationId: ctx.company.organizationId,
   306→          userId: ctx.user.id,
   307→          employeeId: ctx.employee.id,
   308→          newValues: {
   309→            provider: provider.providerKey,
   310→            connectionName: input.connectionName,
   311→            supportsWebhook: provider.supportsWebhook,
   312→          },
   313→        })
   314→      } catch (createErr: unknown) {
   315→        // Catch the @@unique([companyId, providerId]) constraint violation.
   316→        // This means another concurrent request just created this integration
   317→        // (genuine race between two different sessions — the idempotency key
   318→        // system only protects same-session double-clicks). Re-fetch the
   319→        // now-existing row and run the same reactivation logic as the
   320→        // findFirst-then-update path above.
   321→        if (
   322→          createErr &&
   323→          typeof createErr === 'object' &&
   324→          'code' in createErr &&
   325→          (createErr as { code: string }).code === 'P2002'
   326→        ) {
   327→          // Re-fetch the row the winner created
   328→          const raceExisting = await db.companyIntegration.findFirst({
   329→            where: { companyId: ctx.company.id, providerId: input.providerId },
   330→            select: { id: true, isActive: true },
   331→          })
   332→          if (!raceExisting) {
   333→            // Extremely unlikely: the row was deleted between the constraint
   334→            // violation and this re-fetch. Let the error propagate.
   335→            throw createErr
   336→          }
   337→          // Reactivate the row (same logic as the existing-update path above)
   338→          let raceWebhookSecret: string | null = null
   339→          if (provider.supportsWebhook) {
   340→            webhookEndpointId = generateWebhookEndpointId()
   341→            raceWebhookSecret = generateWebhookSecret()
   342→          }
   343→          integration = await db.companyIntegration.update({
   344→            where: { id: raceExisting.id },
   345→            data: {
   346→              connectionName: input.connectionName,
   347→              credentialsEncrypted,
   348→              isActive: true,
   349→              connectionStatus: 'pending',
   350→              lastError: null,
   351→              webhookEndpointId,
   352→              webhookSecret: raceWebhookSecret,
   353→              // Shopify Adapter Foundation: track the employee connecting this
   354→              // integration (race-recovery path — same as the main create path).
   355→              connectedByEmployeeId: ctx.employee.id,
   356→            },
   357→            select: { id: true },
   358→          })
   359→          insertAuditLog({
   360→            action: raceExisting.isActive ? 'integration.credentials_updated' : 'integration.reconnected',
   361→            entityType: 'company_integration',
   362→            entityId: integration.id,
   363→            companyId: ctx.company.id,
   364→            organizationId: ctx.company.organizationId,
   365→            userId: ctx.user.id,
   366→            employeeId: ctx.employee.id,
   367→            newValues: {
   368→              provider: provider.providerKey,
   369→              connectionName: input.connectionName,
   370→              supportsWebhook: provider.supportsWebhook,
   371→            },
   372→          })
   373→        } else {
   374→          throw createErr
   375→        }
   376→      }
   377→    }
   378→
   379→    const appUrl = process.env.APP_URL || 'http://localhost:3000'
   380→    const webhookUrl = webhookEndpointId
   381→      ? `${appUrl}/api/webhooks/${provider.providerKey}/${webhookEndpointId}`
   382→      : undefined
   383→
   384→    return {
   385→      success: true,
   386→      data: { companyIntegrationId: integration.id, webhookUrl },
   387→    }
   388→  } catch (err) {
   389→    return {
   390→      success: false,
   391→      error: err instanceof Error ? err.message : 'Failed to connect integration',
   392→    }
   393→  }
   394→}

   397→// updateIntegrationCredentials
   398→// ──────────────────────────────────────────────────────────────
   399→
   400→export async function updateIntegrationCredentials(
   401→  companyIntegrationId: string,
   402→  credentials: Record<string, string>,
   403→): Promise<ActionResult> {
   404→  try {
   405→    const ctx = await getWorkspace()
   406→    if (!isElevated(ctx)) {
   407→      return { success: false, error: 'Only elevated roles can update integration credentials.' }
   408→    }
   409→
   410→    const integration = await db.companyIntegration.findFirst({
   411→      where: { id: companyIntegrationId, companyId: ctx.company.id },
   412→      include: { provider: true },
   413→    })
   414→    if (!integration) return { success: false, error: 'Integration not found' }
   415→
   416→    // Validate required fields
   417→    let configFields: Array<{ key: string; required: boolean; label: string }> = []
   418→    try {
   419→      configFields = JSON.parse(integration.provider.configSchema)
   420→    } catch {
   421→      return { success: false, error: 'Provider config_schema is invalid' }
   422→    }
   423→    for (const field of configFields) {
   424→      if (field.required && (!credentials[field.key] || !credentials[field.key].trim())) {
   425→        return { success: false, error: `Missing required field: ${field.label}` }
   426→      }
   427→    }
   428→
   429→    const credentialsEncrypted = encryptCredentials(credentials)
   430→
   431→    // Reactivate the integration if it was disconnected. This makes the
   432→    // credentials PATCH route double as the "reconnect" endpoint — the UI's
   433→    // Reconnect button calls this route with new credentials, and the
   434→    // integration flips back to isActive=true + connectionStatus='pending'.
   435→    // If the integration was already active, this is a no-op on isActive
   436→    // (just a credential refresh).
   437→    const wasDisconnected = !integration.isActive
   438→
   439→    await db.companyIntegration.update({
   440→      where: { id: companyIntegrationId },
   441→      data: {
   442→        credentialsEncrypted,
   443→        isActive: true, // reactivate if disconnected
   444→        connectionStatus: 'pending', // reset to pending — needs re-test
   445→        lastError: null,
   446→        // Shopify Adapter Foundation: this route doubles as the Reconnect
   447→        // button's handler — the employee clicking Reconnect is connecting
   448→        // Shopify, so update connectedByEmployeeId here too.
   449→        connectedByEmployeeId: ctx.employee.id,
   450→      },
   451→    })
   452→
   453→    insertAuditLog({
   454→      action: wasDisconnected ? 'integration.reconnected' : 'integration.credentials_updated',
   455→      entityType: 'company_integration',
   456→      entityId: companyIntegrationId,
   457→      companyId: ctx.company.id,
   458→      organizationId: ctx.company.organizationId,
   459→      userId: ctx.user.id,
   460→      employeeId: ctx.employee.id,
   461→    })
   462→
   463→    return { success: true }
   464→  } catch (err) {
   465→    return {
   466→      success: false,
   467→      error: err instanceof Error ? err.message : 'Failed to update credentials',
   468→    }
   469→  }
   470→}

   473→// disconnectIntegration

   476→export async function disconnectIntegration(companyIntegrationId: string): Promise<ActionResult> {
   477→  try {
   478→    const ctx = await getWorkspace()
   479→    if (!isElevated(ctx)) {
   480→      return { success: false, error: 'Only elevated roles can disconnect integrations.' }
   481→    }
   482→
   483→    const integration = await db.companyIntegration.findFirst({
   484→      where: { id: companyIntegrationId, companyId: ctx.company.id },
   485→      select: { id: true, isActive: true, provider: { select: { category: true } } },
   486→    })
   487→    if (!integration) return { success: false, error: 'Integration not found' }
   488→
   489→    // Idempotency guard — if already disconnected, return success without
   490→    // inserting a duplicate audit log.
   491→    if (!integration.isActive) {
   492→      return { success: true }
   493→    }
   494→
   495→    // ── Full disconnect in a single transaction ──
   496→    // 1. Deactivate the integration + update connectionStatus to 'expired'
   497→    //    (the CHECK constraint allows 'pending' | 'connected' | 'error' | 'expired'
   498→    //    — 'expired' is the closest semantic to "disconnected" without a migration).
   499→    // 2. Wipe credentials + webhook info (security — no lingering API keys).
   500→    // 3. Clear lastError + lastSyncAt (stale state from before disconnect).
   501→    // 4. If this integration was the company's default courier, clear that FK
   502→    //    so auto-booking doesn't silently fail on future orders.
   503→    // 5. Audit log (in the same transaction so we never have a deactivation
   504→    //    without an audit trail).
   505→    await db.$transaction(async (tx) => {
   506→      await tx.companyIntegration.update({
   507→        where: { id: companyIntegrationId },
   508→        data: {
   509→          isActive: false,
   510→          isDefault: false,
   511→          connectionStatus: 'expired',
   512→          credentialsEncrypted: null,
   513→          webhookEndpointId: null,
   514→          webhookSecret: null,
   515→          lastError: null,
   516→          lastSyncAt: null,
   517→        },
   518→      })
   519→
   520→      // Clear the default courier FK if it points at this integration.
   521→      // This prevents auto-booking from silently failing with "integration
   522→      // not found or inactive" on every future manual order.
   523→      if (integration.provider.category === 'courier') {
   524→        await tx.companyOrderSetting.updateMany({
   525→          where: {
   526→            companyId: ctx.company.id,
   527→            defaultCourierCompanyIntegrationId: companyIntegrationId,
   528→          },
   529→          data: {
   530→            defaultCourierCompanyIntegrationId: null,
   531→          },
   532→        })
   533→      }
   534→
   535→      await tx.auditLog.create({
   536→        data: {
   537→          action: 'integration.disconnected',
   538→          entityType: 'company_integration',
   539→          entityId: companyIntegrationId,
   540→          companyId: ctx.company.id,
   541→          organizationId: ctx.company.organizationId,
   542→          userId: ctx.user.id,
   543→          employeeId: ctx.employee.id,
   544→        },
   545→      })
   546→    })
   547→
   548→    return { success: true }
   549→  } catch (err) {
   550→    return {
   551→      success: false,
   552→      error: err instanceof Error ? err.message : 'Failed to disconnect integration',
   553→    }
   554→  }
   555→}

   558→// setDefaultIntegration
   559→// ──────────────────────────────────────────────────────────────
   560→
   561→export async function setDefaultIntegration(companyIntegrationId: string): Promise<ActionResult> {
   562→  try {
   563→    const ctx = await getWorkspace()
   564→    if (!isElevated(ctx)) {
   565→      return { success: false, error: 'Only elevated roles can set default integrations.' }
   566→    }
   567→
   568→    // Fetch the integration + its provider's category
   569→    const integration = await db.companyIntegration.findFirst({
   570→      where: { id: companyIntegrationId, companyId: ctx.company.id },
   571→      include: { provider: { select: { category: true } } },
   572→    })
   573→    if (!integration) return { success: false, error: 'Integration not found' }
   574→
   575→    // Unset is_default for all other integrations in the SAME category
   576→    // for this company (join through provider to determine category)
   577→    const sameCategoryIntegrations = await db.companyIntegration.findMany({
   578→      where: {
   579→        companyId: ctx.company.id,
   580→        id: { not: companyIntegrationId },
   581→        provider: { category: integration.provider.category },
   582→      },
   583→      select: { id: true },
   584→    })
   585→
   586→    await db.$transaction([
   587→      // Unset others
   588→      db.companyIntegration.updateMany({
   589→        where: { id: { in: sameCategoryIntegrations.map((i) => i.id) } },
   590→        data: { isDefault: false },
   591→      }),
   592→      // Set this one
   593→      db.companyIntegration.update({
   594→        where: { id: companyIntegrationId },
   595→        data: { isDefault: true },
   596→      }),
   597→    ])
   598→
   599→    insertAuditLog({
   600→      action: 'integration.set_default',
   601→      entityType: 'company_integration',
   602→      entityId: companyIntegrationId,
   603→      companyId: ctx.company.id,
   604→      organizationId: ctx.company.organizationId,
   605→      userId: ctx.user.id,
   606→      employeeId: ctx.employee.id,
   607→      newValues: { category: integration.provider.category },
   608→    })
   609→
   610→    return { success: true }
   611→  } catch (err) {
   612→    return {
   613→      success: false,
   614→      error: err instanceof Error ? err.message : 'Failed to set default integration',
   615→    }
   616→  }
   617→}

   619→// ──────────────────────────────────────────────────────────────
   620→// testIntegrationConnection
   621→// ──────────────────────────────────────────────────────────────
   622→
   623→export async function testIntegrationConnection(companyIntegrationId: string): Promise<ActionResult<{
   624→  status: string
   625→  error?: string
   626→}>> {
   627→  try {
   628→    const ctx = await getWorkspace()
   629→    if (!isElevated(ctx)) {
   630→      return { success: false, error: 'Only elevated roles can test integrations.' }
   631→    }
   632→
   633→    const integration = await db.companyIntegration.findFirst({
   634→      where: { id: companyIntegrationId, companyId: ctx.company.id },
   635→      include: { provider: true },
   636→    })
   637→    if (!integration) return { success: false, error: 'Integration not found' }
   638→    if (!integration.credentialsEncrypted) {
   639→      return { success: false, error: 'No credentials stored for this integration' }
   640→    }
   641→
   642→    // Decrypt credentials
   643→    const credentials = decryptCredentials(integration.credentialsEncrypted)
   644→
   645→    // Get the appropriate adapter via the registry
   646→    const providerKey = integration.provider.providerKey
   647→    const category = getAdapterCategory(providerKey)
   648→    if (!category) {
   649→      return { success: false, error: `No adapter registered for provider '${providerKey}'` }
   650→    }
   651→
   652→    // Call a lightweight, read-only capability check via the logged wrapper.
   653→    //
   654→    // PROVIDER-DISPATCH PATTERN (not PostEx-hardcoded):
   655→    //   1. If the adapter implements pingConnection() → use it (preferred).
   656→    //      PostEx implements this (calls fetchOperationalCities — read-only).
   657→    //      Leopard/TCS will automatically pick this up once their adapters
   658→    //      implement pingConnection() — no changes needed here.
   659→    //   2. Else if the adapter supports fetchOperationalCities() → use it
   660→    //      (covers couriers that have a cities endpoint but no explicit ping).
   661→    //   3. Else fall back to calculateRate() (legacy path — some couriers
   662→    //      expose a rate endpoint that validates credentials).
   663→    //   4. Else surface a clear "not supported" error.
   664→    //
   665→    // For ecommerce adapters: no standard "ping" — parseWebhookOrder with an
   666→    // empty payload is the minimal "does the adapter work" test.
   667→    try {
   668→      const testResult = await executeLoggedIntegrationAction<{ success: boolean; error?: string }>({
   669→        companyIntegrationId,
   670→        organizationId: ctx.company.organizationId,
   671→        actionType: 'test_connection',
   672→        direction: 'outbound',
   673→        fn: async () => {
   674→          if (category === 'courier') {
   675→            const adapter = getCourierAdapter(providerKey, credentials)
   676→
   677→            // Preferred: adapter implements pingConnection()
   678→            if (typeof adapter.pingConnection === 'function') {
   679→              return adapter.pingConnection()
   680→            }
   681→
   682→            // Fallback 1: adapter supports fetchOperationalCities() (read-only)
   683→            if (typeof adapter.fetchOperationalCities === 'function') {
   684→              const cities = await adapter.fetchOperationalCities()
   685→              if (cities.length === 0) {
   686→                return {
   687→                  success: false,
   688→                  error: `${providerKey} accepted the credentials but returned 0 cities.`,
   689→                }
   690→              }
   691→              return { success: true }
   692→            }
   693→
   694→            // Fallback 2: calculateRate() (legacy — unsupported by PostEx)
   695→            if (typeof adapter.calculateRate === 'function') {
   696→              try {
   697→                const rateResult = await adapter.calculateRate({
   698→                  fromCity: 'Karachi',
   699→                  toCity: 'Lahore',
   700→                  weightGrams: 500,
   701→                })
   702→                return {
   703→                  success: rateResult.success,
   704→                  error: rateResult.success ? undefined : rateResult.error,
   705→                }
   706→              } catch (rateErr) {
   707→                return {
   708→                  success: false,
   709→                  error: rateErr instanceof Error ? rateErr.message : 'calculateRate failed',
   710→                }
   711→              }
   712→            }
   713→
   714→            // No read-only test method available
   715→            return {
   716→              success: false,
   717→              error: `The ${providerKey} adapter does not implement a read-only connectivity check (pingConnection, fetchOperationalCities, or calculateRate). Test not supported.`,
   718→            }
   719→          } else {
   720→            const adapter = getEcommerceAdapter(providerKey, credentials)
   721→            // Ecommerce: no standard "ping" — use parseWebhookOrder with
   722→            // an empty payload as a minimal "does the adapter work" test
   723→            try {
   724→              const parsed = await adapter.parseWebhookOrder({})
   725→              return {
   726→                success: parsed.success,
   727→                error: parsed.success ? undefined : parsed.error,
   728→              }
   729→            } catch (parseErr) {
   730→              return {
   731→                success: false,
   732→                error: parseErr instanceof Error ? parseErr.message : 'parseWebhookOrder failed',
   733→              }
   734→            }
   735→          }
   736→        },
   737→      })
   738→
   739→      // executeLoggedIntegrationAction throws on failure; if we reach here
   740→      // with testResult.success=true, the connection is good.
   741→      if (testResult.success) {
   742→        await db.companyIntegration.update({
   743→          where: { id: companyIntegrationId },
   744→          data: { connectionStatus: 'connected', lastSyncAt: new Date(), lastError: null },
   745→        })
   746→        return { success: true, data: { status: 'connected' } }
   747→      }
   748→
   749→      // The ping call returned a structured failure (not an exception).
   750→      // Surface the error but DON'T mark the integration as broken — a single
   751→      // failed test ping may be transient (network blip, rate limit, etc.).
   752→      // We update lastError + connectionStatus='error' for visibility, but
   753→      // isActive stays true so the integration remains usable.
   754→      const errorMsg = testResult.error || 'Connection test failed'
   755→      await db.companyIntegration.update({
   756→        where: { id: companyIntegrationId },
   757→        data: { connectionStatus: 'error', lastError: errorMsg },
   758→      })
   759→      return { success: false, error: errorMsg, data: { status: 'error', error: errorMsg } }
   760→    } catch (testErr) {
   761→      const errorMsg = testErr instanceof Error ? testErr.message : String(testErr)
   762→      // Same non-destructive policy: log the error, don't disable the integration.
   763→      await db.companyIntegration.update({
   764→        where: { id: companyIntegrationId },
   765→        data: { connectionStatus: 'error', lastError: errorMsg },
   766→      }).catch(() => {}) // don't let a DB failure mask the original test error
   767→      // Return success=false but with the error message (don't throw —
   768→      // the caller wants to display it in the UI)
   769→      return { success: false, error: errorMsg, data: { status: 'error', error: errorMsg } }
   770→    }
   771→  } catch (err) {
   772→    return {
   773→      success: false,
   774→      error: err instanceof Error ? err.message : 'Failed to test connection',
   775→    }
   776→  }
   777→}

   779→// ──────────────────────────────────────────────────────────────
   780→// Internal helper: get a decrypted-credentials + adapter combo
   781→// for use by other server actions (courier booking, etc.)
   782→// ──────────────────────────────────────────────────────────────
   783→
   784→export async function getIntegrationAdapter(companyIntegrationId: string): Promise<{
   785→  credentials: Record<string, string>
   786→  providerKey: string
   787→  category: 'courier' | 'ecommerce'
   788→  organizationId: string
   789→} | null> {
   790→  const integration = await db.companyIntegration.findUnique({
   791→    where: { id: companyIntegrationId },
   792→    include: { provider: true },
   793→  })
   794→  if (!integration || !integration.isActive || !integration.credentialsEncrypted) return null
   795→
   796→  const credentials = decryptCredentials(integration.credentialsEncrypted)
   797→  const category = getAdapterCategory(integration.provider.providerKey)
   798→  if (!category) return null
   799→
   800→  return {
   801→    credentials,
   802→    providerKey: integration.provider.providerKey,
   803→    category,
   804→    organizationId: integration.organizationId,
   805→  }
   806→}
