     3→/**
     4→ * Client-side parent-child variant table for the product CREATION wizard.
     5→ *
     6→ * This is the client-side counterpart of the edit-page ParentChildVariantTable.
     7→ * It uses the SAME shared grouping utility (/lib/utils/variant-grouping.ts) and
     8→ * the SAME shared presentational sub-components (variant-table-parts.tsx) as
     9→ * the edit-page table — so grouping behavior and visual markup are identical
    10→ * between the two contexts.
    11→ *
    12→ * The key difference: ALL state changes here update LOCAL component state (the
    13→ * parent wizard's variant array, via the onVariantsChange callback) instead of
    14→ * calling any server action. No network calls happen for grouping, cascading,
    15→ * or overriding during the wizard flow — all of that is pure local state until
    16→ * the wizard's final submit.
    17→ *
    18→ * When hasMeaningfulGrouping is false (single-attribute or zero-attribute
    19→ * products), it renders a flat table matching the wizard's existing flat
    20→ * preview table.
    21→ */

    58→/**
    59→ * The minimum shape a wizard variant needs to participate in grouped
    60→ * pricing. The component is generic over T so any extra fields
    61→ * (opening_stock_*, stitching_*, etc.) pass through untouched.
    62→ */
    63→export interface WizardGroupableVariant {
    64→  /** Unique key (use SKU since no DB id exists yet) */
    65→  sku: string
    66→  attribute_values: Record<string, string>
    67→  cost_price: number
    68→  sale_price: number
    69→  compare_price: number | null
    70→  cost_price_synced_with_parent: boolean
    71→  sale_price_synced_with_parent: boolean
    72→  compare_price_synced_with_parent: boolean
    73→  fulfillment_type: string
    74→  is_active: boolean
    75→  is_default: boolean
    76→}
    77→
    78→interface Props<T extends WizardGroupableVariant> {
    79→  variants: T[]
    80→  selectedAttributes: GroupableAttribute[]
    81→  onVariantsChange: (variants: T[]) => void
    82→  /** Whether the user can edit (always true in the wizard if they have products.create) */
    83→  canEdit?: boolean
    84→}

    90→export function ClientSideParentChildVariantTable<T extends WizardGroupableVariant>({
    91→  variants,
    92→  selectedAttributes,
    93→  onVariantsChange,
    94→  canEdit = true,
    95→}: Props<T>) {
    96→  // Compute grouping using the SHARED utility (same as the server endpoint)
    97→  const grouping = useMemo(() => {
    98→    const parent = determineParentAttribute(selectedAttributes)
    99→    // Map wizard variants to GroupableVariant shape (id = sku for client-side keying)
   100→    const groupable = variants.map((v) => ({
   101→      id: v.sku,
   102→      attribute_values: v.attribute_values,
   103→      data: v, // preserve the full variant as opaque payload
   104→    }))
   105→    return groupVariantsByParentAttribute(groupable, parent?.name ?? null)
   106→  }, [variants, selectedAttributes])
   107→
   108→  const parentAttrName = grouping.parentAttributeName
   109→
   110→  if (!grouping.hasMeaningfulGrouping) {
   111→    return <FlatVariantTable variants={variants} canEdit={canEdit} onVariantsChange={onVariantsChange} />
   112→  }
   113→
   114→  // Extract the original variant data from the grouped result
   115→  const groupedVariants = grouping.groups.map((g) => ({
   116→    parentValue: g.parentValue,
   117→    children: g.children.map((c) => c.data),
   118→  }))
   119→
   120→  return (
   121→    <GroupedVariantTable
   122→      variants={variants}
   123→      groups={groupedVariants}
   124→      parentAttrName={parentAttrName!}
   125→      canEdit={canEdit}
   126→      onVariantsChange={onVariantsChange}
   127→    />
   128→  )
   129→}

   135→function GroupedVariantTable<T extends WizardGroupableVariant>({
   136→  variants,
   137→  groups,
   138→  parentAttrName,
   139→  canEdit,
   140→  onVariantsChange,
   141→}: {
   142→  variants: T[]
   143→  groups: Array<{ parentValue: string; children: T[] }>
   144→  parentAttrName: string
   145→  canEdit: boolean
   146→  onVariantsChange: (v: T[]) => void
   147→}) {
   148→  const [expandedGroups, setExpandedGroups] = useState<Set<string>>(
   149→    () => new Set(groups.map((g) => g.parentValue)),
   150→  )
   151→
   152→  // Auto-expand new groups
   153→  useEffect(() => {
   154→    setExpandedGroups((prev) => {
   155→      const next = new Set(prev)
   156→      for (const g of groups) next.add(g.parentValue)
   157→      return next
   158→    })
   159→  }, [groups])
   160→
   161→  function toggleGroup(parentValue: string) {
   162→    setExpandedGroups((prev) => {
   163→      const next = new Set(prev)
   164→      if (next.has(parentValue)) next.delete(parentValue)
   165→      else next.add(parentValue)
   166→      return next
   167→    })
   168→  }
   169→
   170→  return (
   171→    <div className="space-y-3">
   172→      <p className="text-xs text-muted-foreground">
   173→        Grouped by <span className="font-medium text-foreground">{parentAttrName}</span> — set
   174→        prices once per group, or override individual variants.
   175→      </p>
   176→      {groups.map((group) => (
   177→        <GroupCard
   178→          key={group.parentValue}
   179→          group={group}
   180→          parentAttrName={parentAttrName}
   181→          expanded={expandedGroups.has(group.parentValue)}
   182→          onToggle={() => toggleGroup(group.parentValue)}
   183→          canEdit={canEdit}
   184→          onVariantsChange={onVariantsChange}
   185→          allVariants={variants}
   186→        />
   187→      ))}
   188→    </div>
   189→  )
   190→}

   196→function GroupCard<T extends WizardGroupableVariant>({
   197→  group,
   198→  parentAttrName,
   199→  expanded,
   200→  onToggle,
   201→  canEdit,
   202→  onVariantsChange,
   203→  allVariants,
   204→}: {
   205→  group: { parentValue: string; children: T[] }
   206→  parentAttrName: string
   207→  expanded: boolean
   208→  onToggle: () => void
   209→  canEdit: boolean
   210→  onVariantsChange: (v: T[]) => void
   211→  allVariants: T[]
   212→}) {
   213→  const showCost = true
   214→  const showPricing = true
   215→
   216→  // Parent input local state — initializes from the first synced child
   217→  const firstSyncedCost = group.children.find((c) => c.cost_price_synced_with_parent)
   218→  const firstSyncedSale = group.children.find((c) => c.sale_price_synced_with_parent)
   219→  const firstSyncedCompare = group.children.find((c) => c.compare_price_synced_with_parent)
   220→  const [parentCost, setParentCost] = useState(
   221→    String(firstSyncedCost?.cost_price ?? group.children[0]?.cost_price ?? 0),
   222→  )
   223→  const [parentSale, setParentSale] = useState(
   224→    String(firstSyncedSale?.sale_price ?? group.children[0]?.sale_price ?? 0),
   225→  )
   226→  const [parentCompare, setParentCompare] = useState(
   227→    firstSyncedCompare?.compare_price != null
   228→      ? String(firstSyncedCompare.compare_price)
   229→      : group.children[0]?.compare_price != null
   230→        ? String(group.children[0].compare_price)
   231→        : '',
   232→  )
   233→
   234→  // Re-init when the group identity changes (different children)
   235→  const groupKey = group.children.map((c) => c.sku).join(',')
   236→  const lastGroupKeyRef = useRef(groupKey)
   237→  useEffect(() => {
   238→    if (lastGroupKeyRef.current !== groupKey) {
   239→      lastGroupKeyRef.current = groupKey
   240→      const sc = group.children.find((c) => c.cost_price_synced_with_parent)
   241→      const ss = group.children.find((c) => c.sale_price_synced_with_parent)
   242→      const sp = group.children.find((c) => c.compare_price_synced_with_parent)
   243→      setParentCost(String(sc?.cost_price ?? group.children[0]?.cost_price ?? 0))
   244→      setParentSale(String(ss?.sale_price ?? group.children[0]?.sale_price ?? 0))
   245→      setParentCompare(
   246→        sp?.compare_price != null
   247→          ? String(sp.compare_price)
   248→          : group.children[0]?.compare_price != null
   249→            ? String(group.children[0].compare_price)
   250→            : '',
   251→      )
   252→    }
   253→  }, [groupKey, group.children])
   254→
   255→  // Child attribute keys (exclude parent attribute)
   256→  const childAttrKeys = group.children.length > 0
   257→    ? Object.keys(group.children[0].attribute_values).filter((k) => k !== parentAttrName)
   258→    : []
   259→
   260→  // ── Cascade handler (pure local state, no network) ──
   261→  // Bug 1 fix: ONE handler cascades ALL THREE fields (cost, sale, compare)
   262→  // independently to their respective synced children. Each field only
   263→  // updates children whose relevant synced flag is true, so the three flags
   264→  // remain INDEPENDENT.
   265→
   266→  function applyAllToGroup() {
   267→    const cost = Number(parentCost)
   268→    const sale = Number(parentSale)
   269→    const compare = parentCompare ? Number(parentCompare) : null
   270→
   271→    if (isNaN(cost) || cost < 0) {
   272→      toast.error('Enter a valid cost price')
   273→      return
   274→    }
   275→    if (isNaN(sale) || sale < 0) {
   276→      toast.error('Enter a valid sale price')
   277→      return
   278→    }
   279→
   280→    const updated = allVariants.map((v) => {
   281→      const isChild = group.children.some((c) => c.sku === v.sku)
   282→      if (!isChild) return v
   283→      const patch: Partial<T> = {}
   284→      // Cost cascades to cost-synced children only
   285→      if (v.cost_price_synced_with_parent) {
   286→        ;(patch as Record<string, unknown>).cost_price = cost
   287→      }
   288→      // Sale cascades to sale-synced children only
   289→      if (v.sale_price_synced_with_parent) {
   290→        ;(patch as Record<string, unknown>).sale_price = sale
   291→      }
   292→      // Compare cascades to compare-synced children only
   293→      if (v.compare_price_synced_with_parent) {
   294→        ;(patch as Record<string, unknown>).compare_price = compare
   295→      }
   296→      return Object.keys(patch).length > 0 ? { ...v, ...patch } : v
   297→    })
   298→    onVariantsChange(updated)
   299→
   300→    const costCount = group.children.filter((c) => c.cost_price_synced_with_parent).length
   301→    const saleCount = group.children.filter((c) => c.sale_price_synced_with_parent).length
   302→    const compareCount = group.children.filter((c) => c.compare_price_synced_with_parent).length
   303→    toast.success(
   304→      `Applied to group — Cost: ${costCount}, Sale: ${saleCount}, Compare: ${compareCount} variant(s)`,
   305→    )
   306→  }
   307→
   308→  function updateChild(sku: string, patch: Partial<T>) {
   309→    onVariantsChange(allVariants.map((v) => (v.sku === sku ? { ...v, ...patch } : v)))
   310→  }
   311→
   312→  // Find the current synced value from siblings (for re-sync)
   313→  function getSyncedSiblingValue(field: 'cost_price' | 'sale_price' | 'compare_price'): number | null | undefined {
   314→    const synced = group.children.find((c) => {
   315→      if (field === 'cost_price') return c.cost_price_synced_with_parent
   316→      if (field === 'sale_price') return c.sale_price_synced_with_parent
   317→      return c.compare_price_synced_with_parent
   318→    })
   319→    if (synced) return synced[field] as number | null
   320→    // Fall back to the parent input value
   321→    if (field === 'cost_price') return Number(parentCost)
   322→    if (field === 'sale_price') return Number(parentSale)
   323→    return parentCompare ? Number(parentCompare) : null
   324→  }
   325→
   326→  return (
   327→    <Card>
   328→      <CardContent className="p-4 space-y-4">
   329→        {/* Collapsible header */}
   330→        <div
   331→          className="flex items-center justify-between cursor-pointer"
   332→          onClick={onToggle}
   333→        >
   334→          <div className="flex items-center gap-2">
   335→            <span className="text-base font-semibold">{group.parentValue}</span>
   336→            <Badge variant="secondary" className="text-[10px]">
   337→              {group.children.length} variant{group.children.length === 1 ? '' : 's'}
   338→            </Badge>
   339→          </div>
   340→          <Button variant="ghost" size="sm" className="h-7 text-xs">
   341→            {expanded ? 'Collapse' : 'Expand'}
   342→          </Button>
   343→        </div>
   344→
   345→        {expanded && (
   346→          <>
   347→            {/* Parent group inputs */}
   348→            <ParentGroupInputs
   349→              parentCost={parentCost}
   350→              parentSale={parentSale}
   351→              parentCompare={parentCompare}
   352→              onCostChange={setParentCost}
   353→              onSaleChange={setParentSale}
   354→              onCompareChange={setParentCompare}
   355→              onApplyAll={applyAllToGroup}
   356→              showCost={showCost}
   357→              showPricing={showPricing}
   358→              canEditCost={canEdit}
   359→              canEditPrice={canEdit}
   360→              applying={false}
   361→            />
   362→
   363→            {/* Children table */}
   364→            <div className="rounded-md border overflow-x-auto scrollbar-thin">
   365→              <table className="w-full text-sm">
   366→                <thead className="bg-muted/30">
   367→                  <tr className="text-left text-xs text-muted-foreground">
   368→                    {childAttrKeys.map((k) => (
   369→                      <th key={k} className="px-3 py-2 font-medium">{k}</th>
   370→                    ))}
   371→                    <th className="px-3 py-2 font-medium">SKU</th>
   372→                    <th className="px-3 py-2 font-medium">Fulfillment</th>
   373→                    <th className="px-3 py-2 font-medium text-right">Cost</th>
   374→                    <th className="px-3 py-2 font-medium text-right">Sale</th>
   375→                    <th className="px-3 py-2 font-medium text-right">Compare</th>
   376→                    <th className="px-3 py-2 font-medium text-center">Active</th>
   377→                    <th className="px-3 py-2 font-medium">Actions</th>
   378→                  </tr>
   379→                </thead>
   380→                <tbody>
   381→                  {group.children.map((child) => (
   382→                    <GroupedChildRow
   383→                      key={child.sku}
   384→                      child={child}
   385→                      childAttrKeys={childAttrKeys}
   386→                      canEdit={canEdit}
   387→                      onUpdate={(patch) => updateChild(child.sku, patch)}
   388→                      onResyncCost={() => {
   389→                        const val = getSyncedSiblingValue('cost_price')
   390→                        if (val == null) return
   391→                        updateChild(child.sku, {
   392→                          cost_price: val,
   393→                          cost_price_synced_with_parent: true,
   394→                        } as Partial<T>)
   395→                        toast.success(`Re-synced cost with parent (Rs. ${val})`)
   396→                      }}
   397→                      onResyncSale={() => {
   398→                        const val = getSyncedSiblingValue('sale_price')
   399→                        if (val == null) return
   400→                        updateChild(child.sku, {
   401→                          sale_price: val,
   402→                          sale_price_synced_with_parent: true,
   403→                        } as Partial<T>)
   404→                        toast.success(`Re-synced sale price with parent (Rs. ${val})`)
   405→                      }}
   406→                      onResyncCompare={() => {
   407→                        const val = getSyncedSiblingValue('compare_price')
   408→                        updateChild(child.sku, {
   409→                          compare_price: val,
   410→                          compare_price_synced_with_parent: true,
   411→                        } as Partial<T>)
   412→                        toast.success('Re-synced compare price with parent')
   413→                      }}
   414→                      canResync={
   415→                        group.children.some((c) => c.cost_price_synced_with_parent && c.sku !== child.sku) ||
   416→                        !!firstSyncedCost
   417→                      }
   418→                    />
   419→                  ))}
   420→                </tbody>
   421→              </table>
   422→            </div>
   423→          </>
   424→        )}
   425→      </CardContent>
   426→    </Card>
   427→  )
   428→}

   430→// ──────────────────────────────────────────────────────────────
   431→// Grouped child row — with sync indicators + re-sync actions
   432→// ──────────────────────────────────────────────────────────────
   433→
   434→function GroupedChildRow<T extends WizardGroupableVariant>({
   435→  child,
   436→  childAttrKeys,
   437→  canEdit,
   438→  onUpdate,
   439→  onResyncCost,
   440→  onResyncSale,
   441→  onResyncCompare,
   442→  canResync,
   443→}: {
   444→  child: T
   445→  childAttrKeys: string[]
   446→  canEdit: boolean
   447→  onUpdate: (patch: Partial<T>) => void
   448→  onResyncCost: () => void
   449→  onResyncSale: () => void
   450→  onResyncCompare: () => void
   451→  canResync: boolean
   452→}) {
   453→  const [costValue, setCostValue] = useState(String(child.cost_price))
   454→  const [saleValue, setSaleValue] = useState(child.sale_price != null ? String(child.sale_price) : '')
   455→  const [compareValue, setCompareValue] = useState(child.compare_price != null ? String(child.compare_price) : '')
   456→
   457→  // Sync local input state when the child's values change externally (e.g. after Apply to Group or re-sync)
   458→  useEffect(() => {
   459→    setCostValue(String(child.cost_price))
   460→  }, [child.cost_price])
   461→  useEffect(() => {
   462→    setSaleValue(child.sale_price != null ? String(child.sale_price) : '')
   463→  }, [child.sale_price])
   464→  useEffect(() => {
   465→    setCompareValue(child.compare_price != null ? String(child.compare_price) : '')
   466→  }, [child.compare_price])
   467→
   468→  function saveCost() {
   469→    const newCost = Number(costValue)
   470→    if (isNaN(newCost) || newCost < 0) return
   471→    if (newCost === child.cost_price) return
   472→    // Override: flip synced flag to false
   473→    onUpdate({ cost_price: newCost, cost_price_synced_with_parent: false } as Partial<T>)
   474→  }
   475→
   476→  function saveSale() {
   477→    const newSale = Number(saleValue)
   478→    if (isNaN(newSale) || newSale < 0) return
   479→    const newCompare = compareValue ? Number(compareValue) : null
   480→    const patch: Partial<T> = {}
   481→    if (newSale !== child.sale_price) {
   482→      ;(patch as Record<string, unknown>).sale_price = newSale
   483→      ;(patch as Record<string, unknown>).sale_price_synced_with_parent = false
   484→    }
   485→    if (newCompare !== child.compare_price) {
   486→      ;(patch as Record<string, unknown>).compare_price = newCompare
   487→      ;(patch as Record<string, unknown>).compare_price_synced_with_parent = false
   488→    }
   489→    if (Object.keys(patch).length > 0) onUpdate(patch)
   490→  }
   491→
   492→  function toggleActive() {
   493→    onUpdate({ is_active: !child.is_active } as Partial<T>)
   494→  }
   495→
   496→  return (
   497→    <tr className="border-t hover:bg-muted/20">
   498→      {childAttrKeys.map((k) => (
   499→        <td key={k} className="px-3 py-2 text-xs">{child.attribute_values[k] ?? '—'}</td>
   500→      ))}
   501→      <td className="px-3 py-2">
   502→        <div className="flex items-center gap-1.5">
   503→          {canEdit ? (
   504→            <Input
   505→              value={child.sku}
   506→              onChange={(e) => onUpdate({ sku: e.target.value } as Partial<T>)}
   507→              className="h-7 w-40 text-xs font-mono"
   508→            />
   509→          ) : (
   510→            <span className="font-mono text-xs">{child.sku}</span>
   511→          )}
   512→          {child.is_default && <Badge variant="secondary" className="text-[9px] py-0 px-1">DEFAULT</Badge>}
   513→        </div>
   514→      </td>
   515→      <td className="px-3 py-2">
   516→        {canEdit ? (
   517→          <Select
   518→            value={child.fulfillment_type}
   519→            onValueChange={(val) => onUpdate({ fulfillment_type: val } as Partial<T>)}
   520→          >
   521→            <SelectTrigger className="h-7 w-32 text-xs">
   522→              <SelectValue />
   523→            </SelectTrigger>
   524→            <SelectContent>
   525→              <SelectItem value="stock_based">Stock Tracked</SelectItem>
   526→              <SelectItem value="made_to_order">Made to Order</SelectItem>
   527→            </SelectContent>
   528→          </Select>
   529→        ) : (
   530→          <FulfillmentTypeBadge type={child.fulfillment_type} />
   531→        )}
   532→      </td>
   533→      {/* Cost with sync indicator */}
   534→      <td className="px-3 py-2">
   535→        <div className="flex items-center gap-1 justify-end">
   536→          {canEdit ? (
   537→            <Input
   538→              type="number"
   539→              min="0"
   540→              step="0.01"
   541→              value={costValue}
   542→              onChange={(e) => setCostValue(e.target.value)}
   543→              onBlur={saveCost}
   544→              className="h-7 w-20 text-xs text-right"
   545→            />
   546→          ) : (
   547→            <span className="text-xs">{child.cost_price}</span>
   548→          )}
   549→          <SyncIndicator synced={child.cost_price_synced_with_parent} />
   550→        </div>
   551→      </td>
   552→      {/* Sale with sync indicator */}
   553→      <td className="px-3 py-2">
   554→        <div className="flex items-center gap-1 justify-end">
   555→          {canEdit ? (
   556→            <Input
   557→              type="number"
   558→              min="0"
   559→              step="0.01"
   560→              value={saleValue}
   561→              onChange={(e) => setSaleValue(e.target.value)}
   562→              onBlur={saveSale}
   563→              className="h-7 w-20 text-xs text-right"
   564→            />
   565→          ) : (
   566→            <span className="text-xs">{child.sale_price ?? '—'}</span>
   567→          )}
   568→          <SyncIndicator synced={child.sale_price_synced_with_parent} />
   569→        </div>
   570→      </td>
   571→      {/* Compare with sync indicator */}
   572→      <td className="px-3 py-2">
   573→        <div className="flex items-center gap-1 justify-end">
   574→          {canEdit ? (
   575→            <Input
   576→              type="number"
   577→              min="0"
   578→              step="0.01"
   579→              value={compareValue}
   580→              onChange={(e) => setCompareValue(e.target.value)}
   581→              onBlur={saveSale}
   582→              className="h-7 w-20 text-xs text-right"
   583→            />
   584→          ) : (
   585→            <span className="text-xs text-muted-foreground">{child.compare_price ?? '—'}</span>
   586→          )}
   587→          <SyncIndicator synced={child.compare_price_synced_with_parent} />
   588→        </div>
   589→      </td>
   590→      <td className="px-3 py-2 text-center">
   591→        <Switch checked={child.is_active} onCheckedChange={toggleActive} disabled={!canEdit} />
   592→      </td>
   593→      <td className="px-3 py-2">
   594→        <div className="flex items-center gap-1">
   595→          {canEdit && !child.cost_price_synced_with_parent && (
   596→            <ResyncButton
   597→              label="Cost"
   598→              onClick={onResyncCost}
   599→              disabled={!canResync}
   600→              title={canResync ? 'Re-sync cost with parent' : 'No synced siblings to sync from yet'}
   601→            />
   602→          )}
   603→          {canEdit && !child.sale_price_synced_with_parent && (
   604→            <ResyncButton
   605→              label="Price"
   606→              onClick={onResyncSale}
   607→              disabled={!canResync}
   608→              title={canResync ? 'Re-sync sale price with parent' : 'No synced siblings to sync from yet'}
   609→            />
   610→          )}
   611→          {canEdit && !child.compare_price_synced_with_parent && (
   612→            <ResyncButton
   613→              label="Compare"
   614→              onClick={onResyncCompare}
   615→              disabled={!canResync}
   616→              title={canResync ? 'Re-sync compare price with parent' : 'No synced siblings to sync from yet'}
   617→            />
   618→          )}
   619→        </div>
   620→      </td>
   621→    </tr>
   622→  )
   623→}

   625→// ──────────────────────────────────────────────────────────────
   626→// Flat table — for single-attribute or zero-attribute products
   627→// (same visual layout as the wizard's existing flat preview table)
   628→// ──────────────────────────────────────────────────────────────
   629→
   630→function FlatVariantTable<T extends WizardGroupableVariant>({
   631→  variants,
   632→  canEdit,
   633→  onVariantsChange,
   634→}: {
   635→  variants: T[]
   636→  canEdit: boolean
   637→  onVariantsChange: (v: T[]) => void
   638→}) {
   639→  // Collect the union of attribute keys across all variants
   640→  const attributeKeys = useMemo(() => {
   641→    const keys = new Set<string>()
   642→    for (const v of variants) {
   643→      for (const k of Object.keys(v.attribute_values)) keys.add(k)
   644→    }
   645→    return Array.from(keys)
   646→  }, [variants])
   647→
   648→  function updateChild(sku: string, patch: Partial<T>) {
   649→    onVariantsChange(variants.map((v) => (v.sku === sku ? { ...v, ...patch } : v)))
   650→  }
   651→
   652→  if (variants.length === 0) {
   653→    return (
   654→      <p className="text-xs text-muted-foreground text-center py-4 border border-dashed rounded-lg">
   655→        No variants generated yet.
   656→      </p>
   657→    )
   658→  }
   659→
   660→  return (
   661→    <div className="rounded-lg border overflow-hidden">
   662→      <div className="flex items-center justify-between p-3 border-b bg-muted/30">
   663→        <p className="text-sm font-medium">
   664→          Generated variants ({variants.length})
   665→        </p>
   666→        <p className="text-xs text-muted-foreground">
   667→          Adjust prices, fulfillment, and active state per row.
   668→        </p>
   669→      </div>
   670→      <div className="max-h-96 overflow-y-auto scrollbar-thin">
   671→        <table className="w-full text-sm">
   672→          <thead className="bg-muted/30 sticky top-0 z-10">
   673→            <tr className="text-left text-xs text-muted-foreground">
   674→              {attributeKeys.map((k) => (
   675→                <th key={k} className="px-3 py-2 font-medium">{k}</th>
   676→              ))}
   677→              <th className="px-3 py-2 font-medium">SKU</th>
   678→              <th className="px-3 py-2 font-medium">Cost</th>
   679→              <th className="px-3 py-2 font-medium">Fulfillment</th>
   680→              <th className="px-3 py-2 font-medium">Sale price</th>
   681→              <th className="px-3 py-2 font-medium text-center">Active</th>
   682→            </tr>
   683→          </thead>
   684→          <tbody>
   685→            {variants.map((v, i) => (
   686→              <FlatChildRow
   687→                key={v.sku}
   688→                variant={v}
   689→                attributeKeys={attributeKeys}
   690→                canEdit={canEdit}
   691→                onUpdate={(patch) => updateChild(v.sku, patch)}
   692→                index={i}
   693→              />
   694→            ))}
   695→          </tbody>
   696→        </table>
   697→      </div>
   698→    </div>
   699→  )
   700→}
   701→
   702→function FlatChildRow<T extends WizardGroupableVariant>({
   703→  variant,
   704→  attributeKeys,
   705→  canEdit,
   706→  onUpdate,
   707→}: {
   708→  variant: T
   709→  attributeKeys: string[]
   710→  canEdit: boolean
   711→  onUpdate: (patch: Partial<T>) => void
   712→  index: number
   713→}) {
   714→  return (
   715→    <tr className="border-t">
   716→      {attributeKeys.map((k) => (
   717→        <td key={k} className="px-3 py-2 text-xs">
   718→          {variant.attribute_values[k] ?? '—'}

   720→      ))}
   721→      <td className="px-3 py-2">

   736→          className="h-8 w-20 text-xs"

   738→        />
   739→      </td>
   740→      <td className="px-3 py-2">
   741→        <Select
   742→          value={variant.fulfillment_type}
   743→          onValueChange={(val) => onUpdate({ fulfillment_type: val } as Partial<T>)}

   746→            <SelectValue />
   747→          </SelectTrigger>

   771→      </td>
   772→    </tr>
   773→  )
   774→}
