   193→   193→/**
   194→   194→ * Discrepancy reason options for cycle count items.
   195→   195→ * These values are sent to the backend and drive the approve-time behavior:
   196→   196→ *   - theft_suspected / unknown → shortage is QUARANTINED + a missing
   197→   197→ *     stock_loss_record is auto-created (investigation opens)
   198→   198→ *   - recording_error / transfer_not_recorded / damage_not_recorded /
   199→   199→ *     no_discrepancy → standard cycle_count_adjust (on_hand set to counted)
   200→   200→ *   - surplus (positive discrepancy) → standard adjust (on_hand set to counted)
   201→   201→ *
   202→   202→ * The backend reads these from item.discrepancyReason in the approve action.
   203→   203→ */
   204→   204→type DiscrepancyReason =
   205→   205→  | 'recording_error'
   206→   206→  | 'theft_suspected'
   207→   207→  | 'damage_not_recorded'
   208→   208→  | 'transfer_not_recorded'
   209→   209→  | 'unknown'
   210→   210→  | 'no_discrepancy'
   211→   211→
   212→   212→const DISCREPANCY_REASONS: Array<{
   213→   213→  value: DiscrepancyReason
   214→   214→  label: string
   215→   215→  hint: string
   216→   216→}> = [
   217→   217→  { value: 'no_discrepancy', label: 'No discrepancy', hint: 'Count matches — no action needed.' },
   218→   218→  { value: 'recording_error', label: 'Recording error', hint: 'Data-entry mistake in a prior receive/adjust/transfer.' },
   219→   219→  { value: 'transfer_not_recorded', label: 'Transfer not recorded', hint: 'Stock moved to/from another location but the transfer was not logged.' },
   220→   220→  { value: 'damage_not_recorded', label: 'Damage not recorded', hint: 'Units were damaged but no stock-loss report was filed.' },
   221→   221→  { value: 'theft_suspected', label: 'Theft suspected', hint: 'Shortage appears stolen — will quarantine + open a missing-stock investigation on approval.' },
   222→   222→  { value: 'unknown', label: 'Unknown', hint: 'Cause cannot be determined — will quarantine + open a missing-stock investigation on approval.' },
   223→   223→]

   227→   227→  theft_suspected: 'Theft suspected',
   228→   228→  damage_not_recorded: 'Damage not recorded',

   233→   233→
   234→   234→const PKR = new Intl.NumberFormat('en-PK', { maximumFractionDigits: 0 })
   235→   235→function formatPKR(n: number): string {
   236→   236→  return `Rs. ${PKR.format(n)}`
   237→   237→}
   238→   238→
   239→   239→function formatDate(iso: string | null): string {
   240→   240→  if (!iso) return '—'
   241→   241→  try {
   242→   242→    return new Date(iso).toLocaleDateString('en-PK', {
   243→   243→      day: '2-digit',
   244→   244→      month: 'short',
   245→   245→      year: 'numeric',
   246→   246→    })
   247→   247→  } catch {
   248→   248→    return iso
   249→   249→  }
   250→   250→}
   251→   251→
   252→   252→function getErrorMessage(err: unknown): string {
   253→   253→  if (err instanceof FetchError) return err.message
   254→   254→  if (err instanceof Error) return err.message
   255→   255→  return 'Something went wrong. Please try again.'
   256→   256→}

   261→   261→
   262→   262→export function CycleCountsView() {
   263→   263→  const can = useCan()
   264→   264→  const queryClient = useQueryClient()
   265→   265→
   266→   266→  const [search, setSearch] = useState('')
   267→   267→  const [statusFilter, setStatusFilter] = useState<string>('all')
   268→   268→  const [createOpen, setCreateOpen] = useState(false)
   269→   269→  const [selectedId, setSelectedId] = useState<string | null>(null)
   270→   270→  const [cancelTarget, setCancelTarget] = useState<CycleCountRow | null>(null)
   271→   271→
   272→   272→  const canManage = can(PERMISSIONS.INVENTORY_CYCLE_COUNT)
   273→   273→
   274→   274→  const countsQuery = useQuery<CycleCountsResponse>({
   275→   275→    queryKey: ['cycle-counts'],
   276→   276→    queryFn: () => api.get<CycleCountsResponse>('/api/cycle-counts'),
   277→   277→    staleTime: 15_000,
   278→   278→  })
   279→   279→
   280→   280→  const counts = countsQuery.data?.counts ?? []
   281→   281→
   282→   282→  const filtered = useMemo(() => {
   283→   283→    return counts.filter((c) => {
   284→   284→      if (statusFilter !== 'all' && c.status !== statusFilter) return false
   285→   285→      if (search) {
   286→   286→        const q = search.toLowerCase()
   287→   287→        return (
   288→   288→          c.countName.toLowerCase().includes(q) ||
   289→   289→          c.location.toLowerCase().includes(q)
   290→   290→        )
   291→   291→      }
   292→   292→      return true
   293→   293→    })
   294→   294→  }, [counts, statusFilter, search])
   295→   295→
   296→   296→  const invalidateAll = (id?: string) => {
   297→   297→    void queryClient.invalidateQueries({ queryKey: ['cycle-counts'] })
   298→   298→    if (id) {
   299→   299→      void queryClient.invalidateQueries({ queryKey: ['cycle-count', id] })
   300→   300→    }
   301→   301→    void queryClient.invalidateQueries({ queryKey: ['inventory-dashboard'] })
   302→   302→  }
   303→   303→
   304→   304→  // ── Action mutations ─────────────────────────────────────────────────────
   305→   305→  const startMutation = useMutation({
   306→   306→    mutationFn: async (id: string) =>
   307→   307→      api.patch(`/api/cycle-counts/${id}`, { action: 'start' }),
   308→   308→    onSuccess: () => {
   309→   309→      toast.success('Count started. Inventory snapshot captured.')
   310→   310→      if (selectedId) invalidateAll(selectedId)
   311→   311→      else invalidateAll()
   312→   312→    },
   313→   313→    onError: (err) => toast.error(getErrorMessage(err)),
   314→   314→  })
   315→   315→
   316→   316→  const approveMutation = useMutation({
   317→   317→    mutationFn: async (id: string) =>
   318→   318→      api.patch(`/api/cycle-counts/${id}`, { action: 'approve' }),
   319→   319→    onSuccess: () => {
   320→   320→      toast.success('Count approved. Adjustments applied to inventory.')
   321→   321→      if (selectedId) invalidateAll(selectedId)
   322→   322→    },
   323→   323→    onError: (err) => toast.error(getErrorMessage(err)),
   324→   324→  })
   325→   325→
   326→   326→  const cancelMutation = useMutation({
   327→   327→    mutationFn: async (id: string) =>
   328→   328→      api.patch(`/api/cycle-counts/${id}`, { action: 'cancel' }),
   329→   329→    onSuccess: () => {
   330→   330→      toast.success('Count cancelled.')
   331→   331→      setCancelTarget(null)
   332→   332→      if (selectedId) invalidateAll(selectedId)
   333→   333→    },
   334→   334→    onError: (err) => toast.error(getErrorMessage(err)),
   335→   335→  })
   336→   336→
   337→   337→  return (
   338→   338→    <div className="space-y-6">
   339→   339→      <PageHeader
   340→   340→        title="Cycle Counts"
   341→   341→        description="Reconcile system stock against physical counts. Discrepancies become inventory adjustments on approval."
   342→   342→        actions={
   343→   343→          <div className="flex items-center gap-2">
   344→   344→            <Button
   345→   345→              variant="outline"
   346→   346→              size="sm"
   347→   347→              onClick={() => countsQuery.refetch()}
   348→   348→              disabled={countsQuery.isFetching}
   349→   349→            >
   350→   350→              <RefreshCw
   351→   351→                className={countsQuery.isFetching ? 'h-4 w-4 animate-spin' : 'h-4 w-4'}
   352→   352→              />
   353→   353→              Refresh
   354→   354→            </Button>
   355→   355→            {canManage && (
   356→   356→              <Button onClick={() => setCreateOpen(true)}>
   357→   357→                <Plus className="h-4 w-4" /> New Cycle Count
   358→   358→              </Button>
   359→   359→            )}
   360→   360→          </div>
   361→   361→        }
   362→   362→      />
   363→   363→
   364→   364→      {/* ── Filter bar ─────────────────────────────────────────────────────── */}
   365→   365→      <div className="flex flex-col sm:flex-row gap-2 sm:items-center">
   366→   366→        <div className="relative flex-1">
   367→   367→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   368→   368→          <Input
   369→   369→            placeholder="Search count name or location…"
   370→   370→            className="pl-9"
   371→   371→            value={search}
   372→   372→            onChange={(e) => setSearch(e.target.value)}
   373→   373→          />
   374→   374→        </div>
   375→   375→        <Select value={statusFilter} onValueChange={setStatusFilter}>
   376→   376→          <SelectTrigger className="w-full sm:w-48">
   377→   377→            <SelectValue placeholder="All statuses" />
   378→   378→          </SelectTrigger>
   379→   379→          <SelectContent>
   380→   380→            <SelectItem value="all">All statuses</SelectItem>
   381→   381→            <SelectItem value="scheduled">Scheduled</SelectItem>
   382→   382→            <SelectItem value="in_progress">In Progress</SelectItem>
   383→   383→            <SelectItem value="pending_review">Pending Review</SelectItem>
   384→   384→            <SelectItem value="approved">Approved</SelectItem>
   385→   385→            <SelectItem value="cancelled">Cancelled</SelectItem>
   386→   386→          </SelectContent>
   387→   387→        </Select>
   388→   388→      </div>
   389→   389→
   390→   390→      <div className="grid gap-6 lg:grid-cols-5">
   391→   391→        {/* ── Left: counts list ─────────────────────────────────────────── */}
   392→   392→        <div className="lg:col-span-2 space-y-3">
   393→   393→          {countsQuery.isLoading ? (
   394→   394→            <Card>
   395→   395→              <CardContent className="p-3 space-y-2">
   396→   396→                {Array.from({ length: 5 }).map((_, i) => (
   397→   397→                  <Skeleton key={i} className="h-16" />
   398→   398→                ))}
   399→   399→              </CardContent>
   400→   400→            </Card>
   401→   401→          ) : countsQuery.isError ? (
   402→   402→            <Card>
   403→   403→              <CardContent className="p-10 text-center">
   404→   404→                <p className="text-sm text-muted-foreground mb-4">
   405→   405→                  Couldn&apos;t load cycle counts. {getErrorMessage(countsQuery.error)}
   406→   406→                </p>
   407→   407→                <Button variant="outline" onClick={() => countsQuery.refetch()}>
   408→   408→                  Try again
   409→   409→                </Button>
   410→   410→              </CardContent>
   411→   411→            </Card>
   412→   412→          ) : filtered.length === 0 ? (
   413→   413→            <EmptyState
   414→   414→              hasCounts={counts.length > 0}
   415→   415→              canManage={canManage}
   416→   416→              onCreate={() => setCreateOpen(true)}
   417→   417→            />
   418→   418→          ) : (
   419→   419→            <div className="space-y-2 max-h-[70vh] overflow-y-auto scrollbar-thin pr-1">
   420→   420→              {filtered.map((c) => {
   421→   421→                const badge = STATUS_BADGE[c.status]
   422→   422→                const isSelected = c.id === selectedId
   423→   423→                return (
   424→   424→                  <button
   425→   425→                    type="button"
   426→   426→                    key={c.id}
   427→   427→                    onClick={() => setSelectedId(c.id)}
   428→   428→                    className={`w-full text-left rounded-md border p-3 transition-colors ${
   429→   429→                      isSelected
   430→   430→                        ? 'border-primary bg-primary/5'
   431→   431→                        : 'hover:bg-muted/50 border-border'
   432→   432→                    }`}
   433→   433→                  >
   434→   434→                    <div className="flex items-start justify-between gap-2 mb-1">
   435→   435→                      <div className="min-w-0">
   436→   436→                        <p className="font-medium text-sm truncate">{c.countName}</p>
   437→   437→                        <p className="text-xs text-muted-foreground">{c.location}</p>
   438→   438→                      </div>
   439→   439→                      <Badge variant="outline" className={badge.className}>
   440→   440→                        {badge.label}
   441→   441→                      </Badge>
   442→   442→                    </div>
   443→   443→                    <div className="flex items-center justify-between text-xs text-muted-foreground mt-2">
   444→   444→                      <span>
   445→   445→                        {c.itemCount} item{c.itemCount === 1 ? '' : 's'} ·{' '}
   446→   446→                        {COUNT_TYPE_LABEL[c.countType]}
   447→   447→                      </span>
   448→   448→                      {c.status === 'approved' || c.status === 'pending_review' ? (
   449→   449→                        <span
   450→   450→                          className={`tabular-nums font-medium ${
   451→   451→                            c.totalDiscrepancies > 0
   452→   452→                              ? 'text-amber-700'
   453→   453→                              : 'text-emerald-700'
   454→   454→                          }`}
   455→   455→                        >
   456→   456→                          {c.totalDiscrepancies} discrepancies · {formatPKR(c.totalVarianceValue)}
   457→   457→                        </span>
   458→   458→                      ) : (
   459→   459→                        <span className="whitespace-nowrap">{formatDate(c.scheduledAt)}</span>
   460→   460→                      )}
   461→   461→                    </div>
   462→   462→                  </button>
   463→   463→                )
   464→   464→              })}
   465→   465→            </div>
   466→   466→          )}
   467→   467→        </div>
   468→   468→
   469→   469→        {/* ── Right: detail or empty ────────────────────────────────────── */}
   470→   470→        <div className="lg:col-span-3">
   471→   471→          {selectedId ? (
   472→   472→            <CycleCountDetailPanel
   473→   473→              countId={selectedId}
   474→   474→              onBack={() => setSelectedId(null)}
   475→   475→              onStart={() => startMutation.mutate(selectedId)}
   476→   476→              onApprove={() => approveMutation.mutate(selectedId)}
   477→   477→              onCancel={() => {
   478→   478→                const target = counts.find((c) => c.id === selectedId)
   479→   479→                if (target) setCancelTarget(target)
   480→   480→              }}
   481→   481→              starting={startMutation.isPending}
   482→   482→              approving={approveMutation.isPending}
   483→   483→              canManage={canManage}
   484→   484→            />
   485→   485→          ) : (
   486→   486→            <Card>
   487→   487→              <CardContent className="p-10 text-center">

   879→   879→function SubmitCountsButton({
   880→   880→  countId,
   881→   881→  items,
   882→   882→}: {
   883→   883→  countId: string
   884→   884→  items: CycleCountItemRow[]
   885→   885→}) {
   886→   886→  const queryClient = useQueryClient()
   887→   887→  const [open, setOpen] = useState(false)
   888→   888→  const [counts, setCounts] = useState<Record<string, string>>({})
   889→   889→  // Per-item reason + notes — only meaningful for discrepant rows
   890→   890→  const [reasons, setReasons] = useState<Record<string, DiscrepancyReason>>({})
   891→   891→  const [notes, setNotes] = useState<Record<string, string>>({})
   892→   892→

   894→   894→    if (open) {
   895→   895→      const initCounts: Record<string, string> = {}
   896→   896→      const initReasons: Record<string, DiscrepancyReason> = {}
   897→   897→      const initNotes: Record<string, string> = {}
   898→   898→      for (const i of items) {
   899→   899→        initCounts[i.id] = i.countedQuantity === null ? String(i.systemQuantity) : String(i.countedQuantity)
   900→   900→        // Preserve existing reason if re-opening after a previous submit attempt
   901→   901→        initReasons[i.id] = (i.discrepancyReason as DiscrepancyReason) ?? 'no_discrepancy'
   902→   902→        initNotes[i.id] = i.notes ?? ''
   903→   903→      }
   904→   904→      setCounts(initCounts)
   905→   905→      setReasons(initReasons)
   906→   906→      setNotes(initNotes)
   907→   907→    }
   908→   908→  }, [open, items])
   909→   909→
   910→   910→  const submitMutation = useMutation({
   911→   911→    mutationFn: async (payload: SubmitCountsPayload) =>
   912→   912→      api.patch(`/api/cycle-counts/${countId}`, payload),
   913→   913→    onSuccess: (_data, vars) => {
   914→   914→      const totalDisc = vars.counted_items.reduce((s, i) => s + i.counted_quantity, 0)
   915→   915→      toast.success(`Submitted ${totalDisc} units counted. Awaiting review.`)
   916→   916→      void queryClient.invalidateQueries({ queryKey: ['cycle-count', countId] })
   917→   917→      void queryClient.invalidateQueries({ queryKey: ['cycle-counts'] })
   918→   918→      setOpen(false)
   919→   919→    },
   920→   920→    onError: (err) => toast.error(getErrorMessage(err)),
   921→   921→  })
   922→   922→
   923→   923→  // Build a list of discrepant item IDs (counted != system)
   924→   924→  const discrepantItems = items.filter((i) => {
   925→   925→    const c = parseInt(counts[i.id] ?? '0', 10) || 0
   926→   926→    return c !== i.systemQuantity
   927→   927→  })
   928→   928→  const totalDiscrepancies = discrepantItems.length
   929→   929→

   931→   931→  const itemsNeedingReason = discrepantItems.filter((i) => {
   932→   932→    const c = parseInt(counts[i.id] ?? '0', 10) || 0
   933→   933→    return c < i.systemQuantity // shortage
   934→   934→  })
   935→   935→
   936→   936→  return (
   937→   937→    <>
   938→   938→      <Button onClick={() => setOpen(true)}>
   939→   939→        <Send className="h-4 w-4" /> Submit for Review

   943→   943→          <DialogHeader>
   944→   944→            <DialogTitle>Submit counted quantities</DialogTitle>
   945→   945→            <DialogDescription>
   946→   946→              Enter the actual counted quantity for each item. Discrepancies require a reason — theft/unknown will open a missing-stock investigation on approval.
   947→   947→            </DialogDescription>
   948→   948→          </DialogHeader>
   949→   949→          <div className="rounded-md border max-h-[50vh] overflow-y-auto scrollbar-thin">
