   235→const PKR = new Intl.NumberFormat('en-PK', { maximumFractionDigits: 0 })
   236→
   237→function formatPKR(n: number): string {
   238→  return `Rs. ${PKR.format(n)}`
   239→}
   240→
   241→function formatDate(iso: string | null): string {
   242→  if (!iso) return '—'
   243→  try {
   244→    return new Date(iso).toLocaleDateString('en-PK', {
   245→      day: '2-digit',
   246→      month: 'short',
   247→      year: 'numeric',
   248→    })
   249→  } catch {
   250→    return iso
   251→  }
   252→}
   253→
   254→function formatDateTime(iso: string | null): string {
   255→  if (!iso) return '—'
   256→  try {
   257→    return new Date(iso).toLocaleString('en-PK', {
   258→      day: '2-digit',
   259→      month: 'short',
   260→      hour: '2-digit',
   261→      minute: '2-digit',
   262→    })
   263→  } catch {
   264→    return iso
   265→  }
   266→}
   267→
   268→function getErrorMessage(err: unknown): string {
   269→  if (err instanceof FetchError) return err.message
   270→  if (err instanceof Error) return err.message
   271→  return 'Something went wrong.'
   272→}
   273→
   274→function formatAttributeValues(values: Record<string, string>): string {
   275→  if (!values || Object.keys(values).length === 0) return ''
   276→  return Object.entries(values)
   277→    .map(([k, v]) => `${k}: ${v}`)
   278→    .join(' · ')
   279→}
