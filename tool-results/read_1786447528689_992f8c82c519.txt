    90→    90→// Helper: reserveOrderStock — shared internal function
    91→    91→// ──────────────────────────────────────────────────────────────
    92→    92→// Processes ALL order_items for an order and attempts to reserve
    93→    93→// stock for each based on its fulfillment_type_snapshot.
    94→    94→//
    95→    95→// For stock_based items:
    96→    96→//   - If sufficient available stock → reserveStockForOrder(), fulfillment_status='reserved'
    97→    97→//   - If insufficient + inventory_policy='continue' → fulfillment_status='backordered'
    98→    98→//   - If insufficient + inventory_policy='deny' → outcome='failed'
    99→    99→//
   100→   100→// For made_to_order items:
   101→   101→//   - checkAndFulfillMadeToOrderVariant() checks returned stock first
   102→   102→//   - If existing_stock → fulfillment_status='reserved', returned_stitched_used=true
   103→   103→//   - If fresh_production → fulfillment_status='reserved', production_order_id set
   104→   104→//   - If error → outcome='failed'
   105→   105→//
   106→   106→// After processing: if ANY item is 'backordered', order status → 'partially_backordered'.
   107→   107→// If ALL items are 'reserved', order stays 'confirmed'.
   108→   108→//
   109→   109→// Called from:
   110→   110→//   - createManualOrder() IF the order auto-confirmed at creation
   111→   111→//   - confirmOrder() when a 'pending' order is manually confirmed
   112→   112→//   - dispatchOrderAction() if status is still 'pending' at dispatch time
   113→   113→
   114→   114→async function reserveOrderStock(
   115→   115→  orderId: string,
   116→   116→  ctx: { company: { id: string; organizationId: string }; user: { id: string }; employee: { id: string } },
   117→   117→): Promise<{
   118→   118→  success: boolean
   119→   119→  results: Array<{ orderItemId: string; outcome: 'reserved' | 'backordered' | 'failed'; reason?: string }>
   120→   120→}> {
   121→   121→  const order = await db.order.findUnique({
   122→   122→    where: { id: orderId },
   123→   123→    select: { dispatchLocationId: true, companyId: true, organizationId: true },
   124→   124→  })
   125→   125→  if (!order) return { success: false, results: [] }
   126→   126→
   127→   127→  const items = await db.orderItem.findMany({
   128→   128→    where: { orderId },
   129→   129→    include: {
   130→   130→      orgVariant: {
   131→   131→        select: { id: true, fulfillmentType: true, inventoryPolicy: true, stitchingCharges: true },
   132→   132→      },
   133→   133→    },
   134→   134→  })
   135→   135→
   136→   136→  const results: Array<{ orderItemId: string; outcome: 'reserved' | 'backordered' | 'failed'; reason?: string }> = []
   137→   137→  let hasBackordered = false
   138→   138→
   139→   139→  for (const item of items) {
   140→   140→    if (item.fulfillmentStatus === 'reserved' || item.fulfillmentStatus === 'dispatched') {
   141→   141→      // Already processed (e.g. re-confirmation attempt) — skip
   142→   142→      results.push({ orderItemId: item.id, outcome: 'reserved' })
   143→   143→      continue
   144→   144→    }
   145→   145→
   146→   146→    const locationId = item.reservedLocationId ?? order.dispatchLocationId
   147→   147→    if (!locationId) {
   148→   148→      results.push({ orderItemId: item.id, outcome: 'failed', reason: 'No dispatch location set' })
   149→   149→      continue
   150→   150→    }
   151→   151→
   152→   152→    if (item.fulfillmentTypeSnapshot === 'stock_based') {
   153→   153→      // Check available stock at the dispatch location
   154→   154→      const pool = await db.inventoryPool.findUnique({
   155→   155→        where: {
   156→   156→          orgVariantId_locationId: {
   157→   157→            orgVariantId: item.orgVariantId,
   158→   158→            locationId,
   159→   159→          },
   160→   160→        },
   161→   161→      })
   162→   162→
   163→   163→      const available = pool ? pool.onHand - pool.reserved : 0
   164→   164→
   165→   165→      if (available >= item.quantity) {
   166→   166→        // Sufficient stock — reserve it
   167→   167→        const reserveResult = await reserveStockForOrder({
   168→   168→          orgVariantId: item.orgVariantId,
   169→   169→          locationId,
   170→   170→          organizationId: order.organizationId,
   171→   171→          companyId: order.companyId,
   172→   172→          employeeId: ctx.employee.id,
   173→   173→          quantity: item.quantity,
   174→   174→          orderId,
   175→   175→        })
   176→   176→
   177→   177→        if (reserveResult.success) {
   178→   178→          await db.orderItem.update({
   179→   179→            where: { id: item.id },
   180→   180→            data: { fulfillmentStatus: 'reserved', reservedLocationId: locationId },
   181→   181→          })
   182→   182→          results.push({ orderItemId: item.id, outcome: 'reserved' })
   183→   183→        } else {
   184→   184→          results.push({ orderItemId: item.id, outcome: 'failed', reason: reserveResult.error })
   185→   185→        }
   186→   186→      } else if (item.orgVariant.inventoryPolicy === 'continue') {
   187→   187→        // Insufficient stock but backordering allowed
   188→   188→        await db.orderItem.update({
   189→   189→          where: { id: item.id },
   190→   190→          data: { fulfillmentStatus: 'backordered', backorderedAt: new Date() },
   191→   191→        })
   192→   192→        insertMetricEvent({
   193→   193→          companyId: order.companyId,
   194→   194→          entityType: 'product',
   195→   195→          entityId: item.orgVariantId,
   196→   196→          metricKey: 'order.backordered',
   197→   197→          numericValue: item.quantity,
   198→   198→          dimensions: { order_id: orderId },
   199→   199→        })
   200→   200→        hasBackordered = true
   201→   201→        results.push({
   202→   202→          orderItemId: item.id,
   203→   203→          outcome: 'backordered',
   204→   204→          reason: `Available: ${available}, required: ${item.quantity}`,
   205→   205→        })
   206→   206→      } else {
   207→   207→        // Insufficient stock + policy='deny' — fail
   208→   208→        results.push({
   209→   209→          orderItemId: item.id,
   210→   210→          outcome: 'failed',
   211→   211→          reason: `Insufficient stock (available: ${available}, required: ${item.quantity}) and inventory_policy='deny'`,
   212→   212→        })
   213→   213→      }
   214→   214→    } else if (item.fulfillmentTypeSnapshot === 'made_to_order') {
   215→   215→      // Made-to-order: check returned stock first, then trigger production
   216→   216→      const mtoResult = await checkAndFulfillMadeToOrderVariant(
   217→   217→        item.orgVariantId,
   218→   218→        item.quantity,
   219→   219→        order.companyId,
   220→   220→        locationId,
   221→   221→      )
   222→   222→
   223→   223→      if (mtoResult.source === 'existing_stock' && mtoResult.locationId) {
   224→   224→        // Returned stock available — reserve it
   225→   225→        const reserveResult = await reserveStockForOrder({
   226→   226→          orgVariantId: item.orgVariantId,
   227→   227→          locationId: mtoResult.locationId,
   228→   228→          organizationId: order.organizationId,
   229→   229→          companyId: order.companyId,
   230→   230→          employeeId: ctx.employee.id,
   231→   231→          quantity: item.quantity,
   232→   232→          orderId,
   233→   233→        })
   234→   234→
   235→   235→        if (reserveResult.success) {
   236→   236→          await db.orderItem.update({
   237→   237→            where: { id: item.id },
   238→   238→            data: {
   239→   239→              fulfillmentStatus: 'reserved',
   240→   240→              returnedStitchedUsed: true,
   241→   241→              reservedLocationId: mtoResult.locationId,
   242→   242→            },
   243→   243→          })
   244→   244→          insertMetricEvent({
   245→   245→            companyId: order.companyId,
   246→   246→            entityType: 'product',
   247→   247→            entityId: item.orgVariantId,
   248→   248→            metricKey: 'order.made_to_order_from_returned_stock',
   249→   249→            numericValue: 1,
   250→   250→            dimensions: { order_id: orderId, stitching_cost_saved: Number(item.orgVariant.stitchingCharges) || 0 },
   251→   251→          })
   252→   252→          results.push({ orderItemId: item.id, outcome: 'reserved' })
   253→   253→        } else {
   254→   254→          results.push({ orderItemId: item.id, outcome: 'failed', reason: reserveResult.error })
   255→   255→        }
   256→   256→      } else if (mtoResult.source === 'fresh_production' && mtoResult.productionOrderId) {
   257→   257→        // Fresh production triggered — link the production order
   258→   258→        await db.orderItem.update({
   259→   259→          where: { id: item.id },
   260→   260→          data: {
   261→   261→            fulfillmentStatus: 'reserved',
   262→   262→            productionOrderId: mtoResult.productionOrderId,
   263→   263→          },
   264→   264→        })
   265→   265→        // Also link the order_item back to the production order
   266→   266→        await db.productionOrder.update({
   267→   267→          where: { id: mtoResult.productionOrderId },
   268→   268→          data: { orderItemId: item.id },
   269→   269→        })
   270→   270→        insertMetricEvent({
   271→   271→          companyId: order.companyId,
   272→   272→          entityType: 'product',
   273→   273→          entityId: item.orgVariantId,
   274→   274→          metricKey: 'order.made_to_order_production_triggered',
   275→   275→          numericValue: 1,
   276→   276→          dimensions: { order_id: orderId, production_order_id: mtoResult.productionOrderId },
   277→   277→        })
   278→   278→        results.push({ orderItemId: item.id, outcome: 'reserved' })
   279→   279→      } else {
   280→   280→        // Error (no fabric source, insufficient fabric, etc.)
   281→   281→        results.push({
   282→   282→          orderItemId: item.id,
   283→   283→          outcome: 'failed',
   284→   284→          reason: mtoResult.error ?? 'Made-to-order fulfillment failed',
   285→   285→        })
   286→   286→      }
   287→   287→    }
   288→   288→  }
   289→   289→
   290→   290→  // Update order status based on results
   291→   291→  if (hasBackordered) {
   292→   292→    await db.order.update({
   293→   293→      where: { id: orderId },
   294→   294→      data: { status: 'partially_backordered' },
   295→   295→    })
   296→   296→  }
   297→   297→  // If no backordered items, order stays 'confirmed' (or whatever it was)
   298→   298→
   299→   299→  return { success: true, results }
   300→   300→}
   301→   301→

   305→   305→
   306→   306→export async function createManualOrder(
   307→   307→  input: CreateManualOrderInput,
   308→   308→): Promise<ActionResult<{
   309→   309→  orderId: string
   310→   310→  flowopsOrderNumber: string
   311→   311→  orderItems: Array<{ id: string; orgVariantId: string; quantity: number }>
   312→   312→  /** Auto-booking is now ASYNC — bookingAttempted=true means it's running in the background. */
   313→   313→  bookingAttempted: boolean
   314→   314→  bookingSucceeded?: boolean
   315→   315→  bookingError?: string
   316→   316→  bookingTrackingNumber?: string
   317→   317→}>> {
   318→   318→  try {
   319→   319→    const ctx = await getWorkspace()
   320→   320→    await requirePermission(ctx, PERMISSIONS.ORDERS_CREATE)
   321→   321→
   322→   322→    // 1. Validate
   323→   323→    const parsed = createManualOrderSchema.safeParse(input)
   324→   324→    if (!parsed.success) {
   325→   325→      return {
   326→   326→        success: false,
   327→   327→        error: parsed.error.issues[0]?.message ?? 'Invalid order data',
   328→   328→      }
   329→   329→    }
   330→   330→    const d = parsed.data
   331→   331→
   332→   332→    // ──────────────────────────────────────────────────────────────
   333→   333→    // 2. PARALLEL GROUP 1 — all independent pre-create reads.
   334→   334→    //    Previously these ran SEQUENTIALLY (customer → variants → settings →
   335→   335→    //    order-number → …), costing ~4× the per-query network latency.
   336→   336→    //    Now they run in a single Promise.all() batch. None of these depend
   337→   337→    //    on each other's results:
   338→   338→    //      - customer resolution: depends on ctx + d only
   339→   339→    //      - variant fetch: depends on ctx.company.organizationId + d.items
   340→   340→    //      - companyOrderSetting: depends on ctx.company.id (fetched ONCE here
   341→   341→    //        — previously fetched TWICE, once for requireOrderConfirmation
   342→   342→    //        and again for courierBookingMode)
   343→   343→    //      - generateOrderNumber: SQL function, depends on ctx.company.id only
   344→   344→    // ──────────────────────────────────────────────────────────────
   345→   345→    const variantIds = d.items.map((i) => i.org_variant_id)
   346→   346→    const [custResult, variants, orderSettings, flowopsOrderNumber] = await Promise.all([
   347→   347→      // ── Customer resolution (existing lookup OR inline new-customer creation) ──
   348→   348→      (async () => {
   349→   349→        let customerId: string
   350→   350→        let customerName: string
   351→   351→        let usedCustomerAddressId: string | null = null
   352→   352→        let usedCustomerPhoneId: string | null = null
   353→   353→        let saveAddressForNextTime = d.save_address_for_next_time ?? false
   354→   354→        let selectedSavedAddressId: string | null = null
   355→   355→
   356→   356→        if (d.customer_id) {
   357→   357→          // Existing customer path — verify they belong to this org.
   358→   358→          const existing = await db.customer.findFirst({
   359→   359→            where: { id: d.customer_id, organizationId: ctx.company.organizationId },
   360→   360→            select: { id: true, name: true },
   361→   361→          })
   362→   362→          if (!existing) return { ok: false as const, error: 'Customer not found' }
   363→   363→          customerId = existing.id
   364→   364→          customerName = existing.name
   365→   365→
   366→   366→          // If BOTH a saved address AND a saved phone were selected, verify
   367→   367→          // them in PARALLEL (independent lookups). Otherwise do whichever
   368→   368→          // is present.
   369→   369→          if (d.used_customer_address_id && d.used_customer_phone_id) {
   370→   370→            const [addr, phone] = await Promise.all([
   371→   371→              db.customerAddress.findFirst({
   372→   372→                where: { id: d.used_customer_address_id, customerId: existing.id, organizationId: ctx.company.organizationId },
   373→   373→                select: { id: true },
   374→   374→              }),
   375→   375→              db.customerPhone.findFirst({
   376→   376→                where: { id: d.used_customer_phone_id, customerId: existing.id, organizationId: ctx.company.organizationId },
   377→   377→                select: { id: true },
   378→   378→              }),
   379→   379→            ])
   380→   380→            if (addr) {
   381→   381→              usedCustomerAddressId = addr.id
   382→   382→              selectedSavedAddressId = addr.id
   383→   383→            }
   384→   384→            if (phone) {
   385→   385→              usedCustomerPhoneId = phone.id
   386→   386→            }
   387→   387→          } else if (d.used_customer_address_id) {
   388→   388→            const addr = await db.customerAddress.findFirst({
   389→   389→              where: { id: d.used_customer_address_id, customerId: existing.id, organizationId: ctx.company.organizationId },
   390→   390→              select: { id: true },
   391→   391→            })
   392→   392→            if (addr) {
   393→   393→              usedCustomerAddressId = addr.id
   394→   394→              selectedSavedAddressId = addr.id
   395→   395→            }
   396→   396→          } else if (d.used_customer_phone_id) {
   397→   397→            const phone = await db.customerPhone.findFirst({
   398→   398→              where: { id: d.used_customer_phone_id, customerId: existing.id, organizationId: ctx.company.organizationId },
   399→   399→              select: { id: true },
   400→   400→            })
   401→   401→            if (phone) {
   402→   402→              usedCustomerPhoneId = phone.id
   403→   403→            }
   404→   404→          }
   405→   405→        } else if (d.new_customer) {
   406→   406→          // Inline new-customer path — create the customer + phones +
   407→   407→          // addresses via the INTERNAL variant (skips the redundant
   408→   408→          // getWorkspace + requirePermission that createCustomer() would
   409→   409→          // re-run). createCustomerInternal also returns the default
   410→   410→          // address + primary phone IDs, saving two findFirst queries.
   411→   411→          const createResult = await createCustomerInternal(ctx, d.new_customer)
   412→   412→          if (!createResult.success || !createResult.data) {
   413→   413→            return { ok: false as const, error: createResult.error ?? 'Failed to create customer' }
   414→   414→          }
   415→   415→          customerId = createResult.data.customerId
   416→   416→          customerName = d.new_customer.name.trim()
   417→   417→          // For a brand-new customer, the first address they provided IS the
   418→   418→          // saved default address. The internal variant returns its ID
   419→   419→          // directly — no extra query needed.
   420→   420→          usedCustomerAddressId = createResult.data.defaultAddressId
   421→   421→          selectedSavedAddressId = createResult.data.defaultAddressId
   422→   422→          usedCustomerPhoneId = createResult.data.primaryPhoneId
   423→   423→          // For a brand-new customer, we already saved the address as part of
   424→   424→          // createCustomer — don't double-save it.
   425→   425→          saveAddressForNextTime = false
   426→   426→        } else {
   427→   427→          return { ok: false as const, error: 'Either customer_id or new_customer is required' }
   428→   428→        }
   429→   429→
   430→   430→        return {
   431→   431→          ok: true as const,
   432→   432→          customerId,
   433→   433→          customerName,
   434→   434→          usedCustomerAddressId,
   435→   435→          usedCustomerPhoneId,
   436→   436→          saveAddressForNextTime,
   437→   437→          selectedSavedAddressId,
   438→   438→        }
   439→   439→      })(),
   440→   440→      // ── Variant fetch (already batched via `in` clause) ──
   441→   441→      db.orgProductVariant.findMany({
   442→   442→        where: { id: { in: variantIds }, organizationId: ctx.company.organizationId },
   443→   443→        include: {
   444→   444→          product: { select: { title: true } },
   445→   445→          companyPricing: { where: { companyId: ctx.company.id } },
   446→   446→        },
   447→   447→      }),
   448→   448→      // ── Company order settings — fetched ONCE (was twice) ──
   449→   449→      db.companyOrderSetting.findUnique({
   450→   450→        where: { companyId: ctx.company.id },
   451→   451→        select: {
   452→   452→          requireOrderConfirmation: true,
   453→   453→          courierBookingMode: true,
   454→   454→          defaultCourierCompanyIntegrationId: true,
   455→   455→        },
   456→   456→      }),
   457→   457→      // ── Order number (SQL function, independent) ──
   458→   458→      generateOrderNumber(ctx.company.id),
   459→   459→    ])
   460→   460→
   461→   461→    if (!custResult.ok) {
   462→   462→      return { success: false, error: custResult.error }
   463→   463→    }
   464→   464→    const {
   465→   465→      customerId,
   466→   466→      customerName,
   467→   467→      usedCustomerAddressId,
   468→   468→      usedCustomerPhoneId,
   469→   469→      saveAddressForNextTime,
   470→   470→      selectedSavedAddressId,
   471→   471→    } = custResult
   472→   472→
   473→   473→    if (variants.length !== variantIds.length) {
   474→   474→      return { success: false, error: 'One or more variants not found in this organization' }
   475→   475→    }
   476→   476→
   477→   477→    // 3. recipient_name defaults to the customer's name if not explicitly overridden.
   478→   478→    const recipientName = (d.recipient_name && d.recipient_name.trim()) || customerName
   479→   479→
   480→   480→    // 4. Compute subtotal + build order items data (CPU-only, no DB)
   481→   481→    let subtotal = 0
   482→   482→    const orderItemsData: Array<{
   483→   483→      orgVariantId: string
   484→   484→      quantity: number
   485→   485→      unitPrice: number
   486→   486→      lineTotal: number
   487→   487→      fulfillmentTypeSnapshot: string
   488→   488→    }> = []
   489→   489→
   490→   490→    const orderDetailParts: string[] = []
   491→   491→
   492→   492→    for (const item of d.items) {
   493→   493→      const variant = variants.find((v) => v.id === item.org_variant_id)
   494→   494→      if (!variant) continue
   495→   495→
   496→   496→      const unitPrice =
   497→   497→        item.unit_price ??
   498→   498→        (variant.companyPricing[0]?.salePrice
   499→   499→          ? Number(variant.companyPricing[0].salePrice)
   500→   500→          : Number(variant.costPrice))
   501→   501→
   502→   502→      const lineTotal = item.quantity * unitPrice
   503→   503→      subtotal += lineTotal
   504→   504→
   505→   505→      orderItemsData.push({
   506→   506→        orgVariantId: item.org_variant_id,
   507→   507→        quantity: item.quantity,
   508→   508→        unitPrice,
   509→   509→        lineTotal,
   510→   510→        fulfillmentTypeSnapshot: variant.fulfillmentType,
   511→   511→      })
   512→   512→
   513→   513→      const attrParts: string[] = []
   514→   514→      try {
   515→   515→        const attrs = JSON.parse(variant.attributeValues || '{}') as Record<string, string>
   516→   516→        for (const [k, v] of Object.entries(attrs)) {
   517→   517→          if (v) attrParts.push(`${k}: ${v}`)
   518→   518→        }
   519→   519→      } catch {
   520→   520→        // ignore parse errors
   521→   521→      }
   522→   522→      const inner = [variant.sku, ...attrParts].filter(Boolean).join(', ')

   525→   525→      )
   526→   526→    }
   527→   527→
   528→   528→    // 5. Compute totals (CPU-only)
   529→   529→    const discountAmount = d.discount_amount ?? 0
   530→   530→    const courierCharges = d.courier_charges ?? 0
   531→   531→    const estimatedDeliveryCharge = d.estimated_delivery_charge ?? 0
   532→   532→    const taxAmount = d.tax_amount ?? 0

   534→   534→

   547→   547→      advancePaymentMethod = d.advance_payment_method || null
   548→   548→      advancePaymentReference = d.advance_payment_reference || null
   549→   549→      advancePaidAt = new Date()
   550→   550→    } else if (d.payment_type === 'fully_prepaid') {
   551→   551→      paymentStatus = 'fully_prepaid'
   552→   552→      paymentSource = 'manual_conversion'
   553→   553→      advanceAmount = totalOrderValue
   554→   554→      advancePaymentMethod = d.advance_payment_method || null
   555→   555→      advancePaymentReference = d.advance_payment_reference || null
   556→   556→      advancePaidAt = new Date()
   557→   557→    }
   558→   558→
   559→   559→    const remainingCodAmount = totalOrderValue - (advanceAmount ?? 0)
   560→   560→
   561→   561→    // 7. Determine initial order status — uses the ALREADY-FETCHED
   562→   562→    //    companyOrderSetting (no separate query). Previously this did a
   563→   563→    //    dedicated findUnique for requireOrderConfirmation.

   565→   565→    let confirmedAt: Date | null = null
   566→   566→    let skippedConfirmation = false
   567→   567→
   568→   568→    if (paymentStatus === 'advance_paid' || paymentStatus === 'fully_prepaid') {
   569→   569→      orderStatus = 'confirmed'
   570→   570→      confirmedAt = new Date()
   571→   571→    } else {
   572→   572→      if (!orderSettings?.requireOrderConfirmation) {
   573→   573→        orderStatus = 'confirmed'
   574→   574→        confirmedAt = new Date()
   575→   575→        skippedConfirmation = true
   576→   576→      }
   577→   577→    }
   578→   578→
   579→   579→    // 8. Universal courier reference fields (migration 015).
   580→   580→    const orderRefNumber =
   581→   581→      (d.order_ref_number && d.order_ref_number.trim()) || flowopsOrderNumber
   582→   582→    const orderDetail =
   583→   583→      (d.order_detail && d.order_detail.trim()) || orderDetailParts.join(', ')
   584→   584→
   585→   585→    // 9. Create the order (sequential — needs all above)
   586→   586→    const order = await db.order.create({
   587→   587→      data: {
   588→   588→        organizationId: ctx.company.organizationId,
   589→   589→        companyId: ctx.company.id,
   590→   590→        flowopsOrderNumber,
   591→   591→        orderSource: 'manual',
   592→   592→        customerId,
   593→   593→        recipientName,
   594→   594→        usedCustomerAddressId,
   595→   595→        usedCustomerPhoneId,
   596→   596→        status: orderStatus,
   597→   597→        paymentType: d.payment_type,
   598→   598→        paymentStatus,
   599→   599→        paymentSource,
   600→   600→        subtotal,
   601→   601→        discountAmount: discountAmount || null,
   602→   602→        discountReason: d.discount_reason || null,
   603→   603→        courierCharges: courierCharges || null,
   604→   604→        estimatedDeliveryCharge: estimatedDeliveryCharge || null,
   605→   605→        taxAmount: taxAmount || null,
   606→   606→        taxLabel: d.tax_label?.trim() || null,
   607→   607→        totalOrderValue,
   608→   608→        advanceAmount,
   609→   609→        advancePaymentMethod,
   610→   610→        advancePaymentReference,
   611→   611→        advancePaidAt,
   612→   612→        remainingCodAmount,
   613→   613→        deliveryAddress: d.delivery_address,
   614→   614→        deliveryCity: d.delivery_city,
   615→   615→        courierName: d.courier_name || null,
   616→   616→        courierCompanyIntegrationId: d.courier_company_integration_id || null,
   617→   617→        recommendedCourierCompanyIntegrationId: d.courier_company_integration_id || null,

   619→   619→        notesForCourier: d.notes_for_courier || null,

   627→   627→    })
   628→   628→
   629→   629→    // 10. Batch-create order items — SINGLE query via createManyAndReturn
   630→   630→    //     (was N sequential inserts in a loop). Returns the created rows so
   631→   631→    //     we can populate `createdItems` for the response.
   632→   632→    const createdItemRows = await db.orderItem.createManyAndReturn({
   633→   633→      data: orderItemsData.map((itemData) => ({
   634→   634→        orderId: order.id,
   635→   635→        orgVariantId: itemData.orgVariantId,
   636→   636→        organizationId: ctx.company.organizationId,
   637→   637→        quantity: itemData.quantity,
   638→   638→        unitPrice: itemData.unitPrice,
   639→   639→        lineTotal: itemData.lineTotal,
   640→   640→        fulfillmentStatus: 'reserved', // PLACEHOLDER — reserveOrderStock validates/adjusts
   641→   641→        fulfillmentTypeSnapshot: itemData.fulfillmentTypeSnapshot,
   642→   642→        reservedLocationId: d.dispatch_location_id,
   643→   643→      })),
   644→   644→      select: { id: true, orgVariantId: true, quantity: true },
   645→   645→    })
   646→   646→    const createdItems = createdItemRows.map((r) => ({
   647→   647→      id: r.id,
   648→   648→      orgVariantId: r.orgVariantId,
   649→   649→      quantity: r.quantity,

   651→   651→
   652→   652→    // 11. Audit log (fire-and-forget — see src/lib/audit.ts)

   654→   654→      action: 'order.created',
   655→   655→      entityType: 'order',

   657→   657→      companyId: ctx.company.id,
   658→   658→      organizationId: ctx.company.organizationId,
   659→   659→      userId: ctx.user.id,
   660→   660→      employeeId: ctx.employee.id,
   661→   661→      newValues: {
   662→   662→        flowopsOrderNumber,
   663→   663→        status: orderStatus,
   664→   664→        paymentType: d.payment_type,
   665→   665→        paymentStatus,
   666→   666→        itemCount: orderItemsData.length,
   667→   667→        totalOrderValue,
   668→   668→        customerId,
   669→   669→      },
   670→   670→    })
   671→   671→
   672→   672→    // 12. Customer Management System integration — address lastUsedAt +
   673→   673→    //     save-one-off-address + recompute cached stats. The
   674→   674→    //     saveAddressForNextTime branch MUST be sequential (the order.update

   676→   676→    //     branch) and updateCustomerStats are independent → run in parallel.
   677→   677→    if (selectedSavedAddressId) {
   678→   678→      // Bump lastUsedAt on the saved address + recompute customer stats
   679→   679→      // in parallel (independent writes to different tables).
   680→   680→      await Promise.all([
   681→   681→        markAddressAsUsed(selectedSavedAddressId),
   682→   682→        updateCustomerStats(customerId),
   683→   683→      ])
   684→   684→    } else if (saveAddressForNextTime && d.delivery_address && d.delivery_city) {
   685→   685→      // One-off address typed + user opted to save it for next time.
   686→   686→      // Persist as a non-default customer_addresses row, then link the
   687→   687→      // order to it + recompute customer stats in parallel.
   688→   688→      const savedAddr = await db.customerAddress.create({
   689→   689→        data: {
   690→   690→          customerId,
   691→   691→          organizationId: ctx.company.organizationId,
   692→   692→          label: null,
   693→   693→          address: d.delivery_address,
   694→   694→          city: d.delivery_city,
   695→   695→          isDefault: false,

   697→   697→        },
   698→   698→      })

   701→   701→          where: { id: order.id },

   705→   705→      ])
   706→   706→    } else {
   707→   707→      // No address to save — just recompute stats.
   708→   708→      await updateCustomerStats(customerId)
   709→   709→    }
   710→   710→
   711→   711→    // 13. If the order auto-confirmed, run the stock reservation logic.
   712→   712→    //     Sequential — reserveOrderStock reads the order items created above.
   713→   713→    if (orderStatus === 'confirmed') {
   714→   714→      await reserveOrderStock(order.id, ctx)
   715→   715→    }
   716→   716→
   717→   717→    // 14. AUTO-BOOKING — uses the ALREADY-FETCHED orderSettings (no 2nd
   718→   718→    //     findUnique). Fires in the background (PostEx can take 50-100s).
   719→   719→    let bookingAttempted = false
   720→   720→    let bookingSucceeded = false
   721→   721→    let bookingError: string | undefined
   722→   722→    let bookingTrackingNumber: string | undefined
   723→   723→
   724→   724→    if (orderStatus === 'confirmed') {
   725→   725→      const shouldAutoBook =
   726→   726→        orderSettings?.courierBookingMode === 'automatic' &&
   727→   727→        (d.courier_company_integration_id || orderSettings?.defaultCourierCompanyIntegrationId)
   728→   728→

   730→   730→        bookingAttempted = true
   731→   731→        ;(async () => {
   732→   732→          try {
   733→   733→            const { maybeAutoBookOrder } = await import('./booking.actions')
   734→   734→            const bookResult = await maybeAutoBookOrder(order.id, 'manual', orderStatus)
   735→   735→            if (bookResult.success) {
   736→   736→              console.log(`[createManualOrder] Background auto-booking succeeded for ${flowopsOrderNumber}: tracking=${bookResult.data?.trackingNumber}`)
   737→   737→            } else {
   738→   738→              console.warn(`[createManualOrder] Background auto-booking failed for ${flowopsOrderNumber}: ${bookResult.error}`)
   739→   739→            }
   740→   740→          } catch (err) {
   741→   741→            console.error(`[createManualOrder] Background auto-booking threw for ${flowopsOrderNumber}:`, err)
   742→   742→          }
   743→   743→        })()
   744→   744→      }
   745→   745→    }
   746→   746→
   747→   747→    insertMetricEvent({
   748→   748→      companyId: ctx.company.id,
   749→   749→      entityType: 'order',
   750→   750→      entityId: order.id,
   751→   751→      metricKey: 'order.created',
   752→   752→      numericValue: Number(totalOrderValue),
   753→   753→      dimensions: { order_source: 'manual', payment_type: d.payment_type, company_id: ctx.company.id },
   754→   754→    })
   755→   755→
   756→   756→    return {
   757→   757→      success: true,
   758→   758→      data: {
   759→   759→        orderId: order.id,
   760→   760→        flowopsOrderNumber,
   761→   761→        orderItems: createdItems,
   762→   762→        bookingAttempted,
   763→   763→        bookingSucceeded,
   764→   764→        bookingError,
   765→   765→        bookingTrackingNumber,
   766→   766→      },
   767→   767→    }
   768→   768→  } catch (err) {
   769→   769→    return {
   770→   770→      success: false,
   771→   771→      error: err instanceof Error ? err.message : 'Failed to create order',
   772→   772→    }
   773→   773→  }
   774→   774→}
