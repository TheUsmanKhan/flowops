     1→     1→import { db } from './db'
     2→     2→import { Decimal } from '@prisma/client/runtime/library'
     3→     3→import type { Prisma } from '@prisma/client'

     5→     5→/**
     6→     6→ * THE CORE INVENTORY FUNCTION.
     7→     7→ *
     8→     8→ * This is the single point of truth for all stock movements. It:
     9→     9→ *   1. Finds or creates the inventory_pools row for variant+location
    10→    10→ *   2. Validates sufficient stock for OUT-direction transactions
    11→    11→ *   3. Recalculates WAC (weighted average cost) for IN-direction transactions
    12→    12→ *   4. Updates on_hand / reserved / avg_cost on the pool
    13→    13→ *   5. Inserts the immutable inventory_transactions ledger row
    14→    14→ *   6. Records avg_cost_history when avg_cost changes
    15→    15→ *   7. Flips track_inventory TRUE on made_to_order variants on first return
    16→    16→ *
    17→    17→ * IMPORTANT: inventory_pools is NEVER written to directly from any other
    18→    18→ * code path — only through this function. This guarantees the ledger
    19→    19→ * and pool always agree.
    20→    20→ */
    21→    21→
    22→    22→export type TransactionType =
    23→    23→  | 'opening_stock'
    24→    24→  | 'purchase_received'
    25→    25→  | 'sale_dispatched'
    26→    26→  | 'order_reserved'
    27→    27→  | 'order_unreserved'
    28→    28→  | 'return_resellable'
    29→    29→  | 'return_damaged'
    30→    30→  | 'return_stitched_received'
    31→    31→  | 'transfer_out'
    32→    32→  | 'transfer_in'
    33→    33→  | 'cycle_count_adjust'
    34→    34→  | 'damage_writeoff'
    35→    35→  | 'theft_writeoff'
    36→    36→  | 'missing_writeoff'
    37→    37→  | 'transit_loss'
    38→    38→  | 'supplier_return'
    39→    39→  | 'fabric_consumed_for_stitching'
    40→    40→
    41→    41→const OUT_TYPES: TransactionType[] = [
    42→    42→  'sale_dispatched',
    43→    43→  'transfer_out',
    44→    44→  'damage_writeoff',
    45→    45→  'theft_writeoff',
    46→    46→  'missing_writeoff',
    47→    47→  'transit_loss',
    48→    48→  'supplier_return',
    49→    49→  'fabric_consumed_for_stitching',
    50→    50→]
    51→    51→
    52→    52→const WAC_RECALC_TYPES: TransactionType[] = [
    53→    53→  'opening_stock',
    54→    54→  'purchase_received',
    55→    55→  'return_stitched_received',
    56→    56→  'transfer_in',
    57→    57→  'return_resellable',
    58→    58→]
    59→    59→
    60→    60→interface ProcessTxnInput {
    61→    61→  orgVariantId: string
    62→    62→  locationId: string
    63→    63→  organizationId: string
    64→    64→  companyId?: string | null
    65→    65→  employeeId?: string | null
    66→    66→  transactionType: TransactionType
    67→    67→  quantity: number // positive = in, negative = out
    68→    68→  costPerUnit?: number | null // if null, uses current avg_cost for OUT, or new_cost for IN
    69→    69→  referenceType?: string | null
    70→    70→  referenceId?: string | null
    71→    71→  notes?: string | null
    72→    72→  metadata?: Record<string, unknown> | null
    73→    73→}
    74→    74→
    75→    75→interface ProcessTxnResult {
    76→    76→  success: boolean
    77→    77→  transactionId?: string
    78→    78→  poolState?: {
    79→    79→    onHand: number
    80→    80→    reserved: number
    81→    81→    available: number
    82→    82→    avgCost: number
    83→    83→  }
    84→    84→  error?: string
    85→    85→}
    86→    86→
    87→    87→/**
    88→    88→ * Calculate new weighted average cost.
    89→    89→ * new_avg = (existing_qty × old_avg + new_qty × new_cost) / total_qty
    90→    90→ */
    91→    91→function calculateNewAvgCost(
    92→    92→  existingQty: number,
    93→    93→  oldAvg: number,
    94→    94→  newQty: number,
    95→    95→  newCost: number,
    96→    96→): number {
    97→    97→  const totalQty = existingQty + newQty
    98→    98→  if (totalQty <= 0) return 0
    99→    99→  return (existingQty * oldAvg + newQty * newCost) / totalQty
   100→   100→}

   102→   102→/**
   103→   103→ * Process a single inventory transaction.
   104→   104→ * This function is the ONLY way to modify inventory_pools.
   105→   105→ */
   106→   106→export async function processInventoryTransaction(
   107→   107→  input: ProcessTxnInput,
   108→   108→): Promise<ProcessTxnResult> {
   109→   109→  const {
   110→   110→    orgVariantId,
   111→   111→    locationId,
   112→   112→    organizationId,
   113→   113→    companyId = null,
   114→   114→    employeeId = null,
   115→   115→    transactionType,
   116→   116→    quantity,
   117→   117→    referenceType = null,
   118→   118→    referenceId = null,
   119→   119→    notes = null,
   120→   120→    metadata = null,
   121→   121→  } = input
   122→   122→
   123→   123→  // For OUT transactions, quantity should be positive (we negate internally)
   124→   124→  // For IN transactions, quantity is positive
   125→   125→  const absQty = Math.abs(quantity)
   126→   126→
   127→   127→  try {
   128→   128→    // 1. Find or create the inventory_pools row
   129→   129→    let pool = await db.inventoryPool.findUnique({
   130→   130→      where: {
   131→   131→        orgVariantId_locationId: { orgVariantId, locationId },
   132→   132→      },
   133→   133→    })
   134→   134→
   135→   135→    if (!pool) {
   136→   136→      // First transaction ever for this variant+location — create pool with zeros
   137→   137→      pool = await db.inventoryPool.create({
   138→   138→        data: {
   139→   139→          orgVariantId,
   140→   140→          locationId,
   141→   141→          organizationId,
   142→   142→          onHand: 0,
   143→   143→          reserved: 0,
   144→   144→          incoming: 0,
   145→   145→          avgCost: 0,
   146→   146→        },
   147→   147→      })
   148→   148→    }
   149→   149→
   150→   150→    // 2. Validate sufficient stock for OUT-direction transactions
   151→   151→    if (OUT_TYPES.includes(transactionType)) {
   152→   152→      const available = pool.onHand - pool.reserved
   153→   153→      if (available < absQty) {
   154→   154→        return {
   155→   155→          success: false,
   156→   156→          error: `INSUFFICIENT_STOCK: Available ${available}, requested ${absQty}`,
   157→   157→        }
   158→   158→      }
   159→   159→    }
   160→   160→
   161→   161→    // 3. Determine cost_per_unit and compute new avg_cost
   162→   162→    const oldAvgCost = Number(pool.avgCost)
   163→   163→    let costPerUnit = input.costPerUnit ?? null
   164→   164→
   165→   165→    // For IN-direction WAC recalculation types
   166→   166→    if (WAC_RECALC_TYPES.includes(transactionType)) {
   167→   167→      if (costPerUnit === null) {
   168→   168→        costPerUnit = oldAvgCost // fallback if not provided
   169→   169→      }
   170→   170→    }
   171→   171→
   172→   172→    // For OUT types: use current avg_cost if not explicitly provided
   173→   173→    if (OUT_TYPES.includes(transactionType) && costPerUnit === null) {
   174→   174→      costPerUnit = oldAvgCost
   175→   175→    }
   176→   176→
   177→   177→    // For transfer_in: costPerUnit must be passed explicitly (sending location's cost)
   178→   178→    if (transactionType === 'transfer_in' && costPerUnit === null) {
   179→   179→      costPerUnit = oldAvgCost // fallback
   180→   180→    }
   181→   181→
   182→   182→    const finalCostPerUnit = costPerUnit ?? 0
   183→   183→
   184→   184→    // 4. Compute new pool state
   185→   185→    let newOnHand = pool.onHand
   186→   186→    let newReserved = pool.reserved
   187→   187→    let newAvgCost = oldAvgCost
   188→   188→    let newIncoming = pool.incoming
   189→   189→
   190→   190→    switch (transactionType) {
   191→   191→      case 'opening_stock':
   192→   192→        newOnHand += absQty
   193→   193→        newAvgCost = calculateNewAvgCost(pool.onHand, oldAvgCost, absQty, finalCostPerUnit)
   194→   194→        break
   195→   195→      case 'purchase_received':
   196→   196→        newOnHand += absQty
   197→   197→        newIncoming = Math.max(0, newIncoming - absQty)
   198→   198→        newAvgCost = calculateNewAvgCost(pool.onHand, oldAvgCost, absQty, finalCostPerUnit)
   199→   199→        break
   200→   200→      case 'sale_dispatched':
   201→   201→        newOnHand -= absQty
   202→   202→        newReserved = Math.max(0, newReserved - absQty)
   203→   203→        break
   204→   204→      case 'order_reserved':
   205→   205→        newReserved += absQty
   206→   206→        break
   207→   207→      case 'order_unreserved':
   208→   208→        newReserved = Math.max(0, newReserved - absQty)
   209→   209→        break
   210→   210→      case 'return_resellable':
   211→   211→        newOnHand += absQty
   212→   212→        newAvgCost = calculateNewAvgCost(pool.onHand, oldAvgCost, absQty, finalCostPerUnit)
   213→   213→        break
   214→   214→      case 'return_stitched_received':
   215→   215→        newOnHand += absQty
   216→   216→        newAvgCost = calculateNewAvgCost(pool.onHand, oldAvgCost, absQty, finalCostPerUnit)
   217→   217→        break
   218→   218→      case 'return_damaged':
   219→   219→        // No pool change — goes straight to stock_loss_records
   220→   220→        break
   221→   221→      case 'transfer_out':
   222→   222→        newOnHand -= absQty
   223→   223→        break
   224→   224→      case 'transfer_in':
   225→   225→        newOnHand += absQty
   226→   226→        // costPerUnit is the sending location's cost — do NOT recalculate WAC
   227→   227→        // The transferred stock keeps its original cost_per_unit exactly
   228→   228→        newAvgCost = calculateNewAvgCost(pool.onHand, oldAvgCost, absQty, finalCostPerUnit)
   229→   229→        break
   230→   230→      case 'cycle_count_adjust':
   231→   231→        // Set on_hand directly to counted value
   232→   232→        // quantity here represents the NEW on_hand value (positive)
   233→   233→        newOnHand = absQty
   234→   234→        break
   235→   235→      case 'damage_writeoff':
   236→   236→      case 'theft_writeoff':
   237→   237→      case 'missing_writeoff':
   238→   238→      case 'transit_loss':
   239→   239→        newOnHand -= absQty
   240→   240→        break
   241→   241→      case 'supplier_return':
   242→   242→        newOnHand -= absQty
   243→   243→        break
   244→   244→      case 'fabric_consumed_for_stitching':
   245→   245→        newOnHand -= absQty
   246→   246→        break
   247→   247→    }
   248→   248→
   249→   249→    // 5. Update timestamps
   250→   250→    const now = new Date()
   251→   251→    const updateData: Record<string, unknown> = {
   252→   252→      onHand: newOnHand,
   253→   253→      reserved: newReserved,
   254→   254→      incoming: newIncoming,
   255→   255→      avgCost: newAvgCost,
   256→   256→      updatedAt: now,
   257→   257→    }
   258→   258→    if (
   259→   259→      transactionType === 'purchase_received' ||
   260→   260→      transactionType === 'opening_stock' ||
   261→   261→      transactionType === 'return_resellable' ||
   262→   262→      transactionType === 'return_stitched_received' ||
   263→   263→      transactionType === 'transfer_in'
   264→   264→    ) {
   265→   265→      updateData.lastReceivedAt = now
   266→   266→    }
   267→   267→    if (transactionType === 'sale_dispatched') {
   268→   268→      updateData.lastSoldAt = now
   269→   269→    }
   270→   270→    if (transactionType === 'cycle_count_adjust') {
   271→   271→      updateData.lastCountedAt = now
   272→   272→    }
   273→   273→
   274→   274→    await db.inventoryPool.update({
   275→   275→      where: { id: pool.id },
   276→   276→      data: updateData,
   277→   277→    })
   278→   278→
   279→   279→    // 6. Handle track_inventory flip for made_to_order variants on first
   280→   280→    //    return OR on opening_stock entry (e.g. user confirms "pre-made bulk
   281→   281→    //    stock" for an MTO variant during product creation). One-way FALSE → TRUE.
   282→   282→    if (
   283→   283→      transactionType === 'return_stitched_received' ||
   284→   284→      transactionType === 'opening_stock'
   285→   285→    ) {
   286→   286→      const variant = await db.orgProductVariant.findUnique({
   287→   287→        where: { id: orgVariantId },
   288→   288→        select: { trackInventory: true, fulfillmentType: true },
   289→   289→      })
   290→   290→      if (variant && !variant.trackInventory && variant.fulfillmentType === 'made_to_order') {
   291→   291→        // ONE-WAY flip: FALSE → TRUE (never back to FALSE)
   292→   292→        await db.orgProductVariant.update({
   293→   293→          where: { id: orgVariantId },
   294→   294→          data: { trackInventory: true },
   295→   295→        })
   296→   296→      }
   297→   297→    }
   298→   298→
   299→   299→    // 7. Insert the inventory_transactions ledger row
   300→   300→    const txnQuantity = OUT_TYPES.includes(transactionType) ? -absQty : absQty
   301→   301→    const avgCostChanged = newAvgCost !== oldAvgCost
   302→   302→
   303→   303→    const txn = await db.inventoryTransaction.create({
   304→   304→      data: {
   305→   305→        orgVariantId,
   306→   306→        locationId,
   307→   307→        organizationId,
   308→   308→        companyId,
   309→   309→        employeeId,
   310→   310→        transactionType,
   311→   311→        quantity: txnQuantity,
   312→   312→        costPerUnit: finalCostPerUnit,
   313→   313→        avgCostBefore: oldAvgCost,
   314→   314→        avgCostAfter: newAvgCost,
   315→   315→        referenceType,
   316→   316→        referenceId,
   317→   317→        notes,
   318→   318→        metadata: metadata ? JSON.stringify(metadata) : '{}',
   319→   319→        recordedAt: now,
   320→   320→      },
   321→   321→    })
   322→   322→
   323→   323→    // 8. Insert avg_cost_history if avg_cost changed
   324→   324→    if (avgCostChanged) {
   325→   325→      await db.avgCostHistory.create({
   326→   326→        data: {
   327→   327→          orgVariantId,
   328→   328→          locationId,
   329→   329→          organizationId,
   330→   330→          avgCostBefore: oldAvgCost,
   331→   331→          avgCostAfter: newAvgCost,
   332→   332→          triggeredByTxnId: txn.id,
   333→   333→          triggerReason: transactionType,
   334→   334→        },
   335→   335→      })
   336→   336→    }
   337→   337→
   338→   338→    return {
   339→   339→      success: true,
   340→   340→      transactionId: txn.id,
   341→   341→      poolState: {
   342→   342→        onHand: newOnHand,
   343→   343→        reserved: newReserved,
   344→   344→        available: newOnHand - newReserved,
   345→   345→        avgCost: newAvgCost,
   346→   346→      },
   347→   347→    }
   348→   348→  } catch (err) {
   349→   349→    console.error('[inventory] processInventoryTransaction error:', err)
   350→   350→    return {
   351→   351→      success: false,
   352→   352→      error: err instanceof Error ? err.message : 'Unknown inventory transaction error',
   353→   353→    }
   354→   354→  }
   355→   355→}

   361→   361→export async function checkReturnedStockAvailability(variantId: string) {

   363→   363→    where: {

   371→   371→  return pools.map((p) => ({

   379→   379→/**
   380→   380→ * Get inventory summary for a product — powers the product detail Inventory tab.
   381→   381→ */
   382→   382→export async function getProductInventorySummary(productId: string) {
   383→   383→  const variants = await db.orgProductVariant.findMany({
   384→   384→    where: { productId },
   385→   385→    select: { id: true, sku: true, fulfillmentType: true, trackInventory: true },
   386→   386→  })
   387→   387→
   388→   388→  const result: Array<{
   389→   389→    variantId: string
   390→   390→    sku: string
   391→   391→    fulfillmentType: string
   392→   392→    trackInventory: boolean
   393→   393→    totalOnHand: number
   394→   394→    totalReserved: number
   395→   395→    totalAvailable: number
   396→   396→    totalValue: number
   397→   397→    locations: Array<{
   398→   398→      locationId: string
   399→   399→      locationName: string
   400→   400→      onHand: number
   401→   401→      reserved: number
   402→   402→      available: number
   403→   403→      avgCost: number
   404→   404→      incoming: number
   405→   405→    }>
   406→   406→  }> = []
   407→   407→  for (const variant of variants) {
   408→   408→    const pools = await db.inventoryPool.findMany({
   409→   409→      where: { orgVariantId: variant.id },
   410→   410→      include: {
   411→   411→        location: { select: { id: true, name: true } },
   412→   412→      },
   413→   413→    })
   414→   414→
   415→   415→    const totalOnHand = pools.reduce((sum, p) => sum + p.onHand, 0)
   416→   416→    const totalReserved = pools.reduce((sum, p) => sum + p.reserved, 0)
   417→   417→    const totalAvailable = totalOnHand - totalReserved
   418→   418→    const totalValue = pools.reduce((sum, p) => sum + Number(p.onHand) * Number(p.avgCost), 0)
   419→   419→
   420→   420→    result.push({
   421→   421→      variantId: variant.id,
   422→   422→      sku: variant.sku,
   423→   423→      fulfillmentType: variant.fulfillmentType,
   424→   424→      trackInventory: variant.trackInventory,
   425→   425→      totalOnHand,
   426→   426→      totalReserved,
   427→   427→      totalAvailable,
   428→   428→      totalValue,
   429→   429→      locations: pools.map((p) => ({
   430→   430→        locationId: p.locationId,
   431→   431→        locationName: p.location.name,
   432→   432→        onHand: p.onHand,
   433→   433→        reserved: p.reserved,
   434→   434→        available: p.onHand - p.reserved,
   435→   435→        avgCost: Number(p.avgCost),
   436→   436→        incoming: p.incoming,
   437→   437→      })),
   438→   438→    })
   439→   439→  }
   440→   440→
   441→   441→  return result
   442→   442→}

   449→   449→  const year = new Date().getFullYear()
   450→   450→  const prefix = `PO-${year}-`

   463→   463→/**
   464→   464→ * Increment incoming stock on a pool (for PO ordering).
   465→   465→ * This is the ONLY function that writes to inventory_pools.incoming directly
   466→   466→ * — it's a live projection field, not a ledgered movement.
   467→   467→ * Creates the pool row if it doesn't exist.

   469→   469→export async function incrementIncomingStock(

   479→   479→      orgVariantId,
   480→   480→      locationId,

   487→   487→/**
   488→   488→ * Decrement incoming stock (never below 0).
   489→   489→ * Used when cancelling POs or receiving against POs.
   490→   490→ */
   491→   491→export async function decrementIncomingStock(
   492→   492→  orgVariantId: string,
   493→   493→  locationId: string,
   494→   494→  qty: number,
   495→   495→): Promise<void> {
   496→   496→  const pool = await db.inventoryPool.findUnique({
   497→   497→    where: { orgVariantId_locationId: { orgVariantId, locationId } },
   498→   498→    select: { incoming: true },
   499→   499→  })
   500→   500→  if (!pool) return
   501→   501→  const newIncoming = Math.max(0, pool.incoming - qty)
   502→   502→  await db.inventoryPool.update({

   504→   504→    data: { incoming: newIncoming },
   505→   505→  })
   506→   506→}

   508→   508→/**
   509→   509→ * Check and fulfill a made-to-order variant.
   510→   510→ *
   511→   511→ * The central decision function:
   512→   512→ * 1. Check if returned stock is available for this variant
   513→   513→ * 2. If yes: return { source: 'existing_stock', location_id, available }
   514→   514→ * 3. If no: create a production order, consume fabric, return { source: 'fresh_production' }
   515→   515→ */
   516→   516→export async function checkAndFulfillMadeToOrderVariant(
   517→   517→  orgVariantId: string,
   518→   518→  quantity: number,
   519→   519→  companyId: string,
   520→   520→  preferredLocationId?: string,
   521→   521→): Promise<{
   522→   522→  source: 'existing_stock' | 'fresh_production'
   523→   523→  locationId?: string
   524→   524→  available?: number
   525→   525→  productionOrderId?: string
   526→   526→  estimatedCompletionDate?: Date
   527→   527→  error?: string
   528→   528→}> {
   529→   529→  // 1. Check returned stock availability
   530→   530→  const availability = await checkReturnedStockAvailability(orgVariantId)
   531→   531→  const totalAvailable = availability.reduce((sum, a) => sum + a.available, 0)
   532→   532→
   533→   533→  if (totalAvailable >= quantity) {
   534→   534→    // Use existing stock — find the best location (most available)
   535→   535→    const best = availability
   536→   536→      .filter((a) => a.available > 0)
   537→   537→      .sort((a, b) => b.available - a.available)[0]
   538→   538→    return {
   539→   539→      source: 'existing_stock',
   540→   540→      locationId: best.locationId,
   541→   541→      available: best.available,
   542→   542→    }
   543→   543→  }
   544→   544→
   545→   545→  // 2. Not enough returned stock — create a production order
   546→   546→  const variant = await db.orgProductVariant.findUnique({
   547→   547→    where: { id: orgVariantId },
   548→   548→    select: {
   549→   549→      id: true,
   550→   550→      fabricSourceVariantId: true,
   551→   551→      stitchingCharges: true,
   552→   552→      productionDays: true,
   553→   553→      organizationId: true,
   554→   554→    },
   555→   555→  })
   556→   556→
   557→   557→  if (!variant) return { source: 'fresh_production', error: 'Variant not found' }
   558→   558→  if (!variant.fabricSourceVariantId) {
   559→   559→    return { source: 'fresh_production', error: 'No fabric source variant linked to this made_to_order variant' }
   560→   560→  }
   561→   561→
   562→   562→  // Find fabric stock at the preferred location or any location with stock
   563→   563→  const fabricPools = await db.inventoryPool.findMany({
   564→   564→    where: {
   565→   565→      orgVariantId: variant.fabricSourceVariantId,
   566→   566→      onHand: { gt: 0 },
   567→   567→    },
   568→   568→    include: { location: { select: { id: true, name: true } } },
   569→   569→  })
   570→   570→
   571→   571→  const fabricLocation = preferredLocationId
   572→   572→    ? fabricPools.find((p) => p.locationId === preferredLocationId)
   573→   573→    : fabricPools[0]
   574→   574→
   575→   575→  if (!fabricLocation || fabricLocation.onHand - fabricLocation.reserved < quantity) {
   576→   576→    return {
   577→   577→      source: 'fresh_production',
   578→   578→      error: `Insufficient fabric stock. Available: ${fabricLocation?.onHand ?? 0}, required: ${quantity}`,
   579→   579→    }
   580→   580→  }
   581→   581→
   582→   582→  const fabricCost = Number(fabricLocation.avgCost) * quantity
   583→   583→  const estimatedCompletionDate = new Date()
   584→   584→  estimatedCompletionDate.setDate(estimatedCompletionDate.getDate() + (variant.productionDays || 5))
   585→   585→
   586→   586→  // Consume fabric
   587→   587→  const txnResult = await processInventoryTransaction({
   588→   588→    orgVariantId: variant.fabricSourceVariantId,
   589→   589→    locationId: fabricLocation.locationId,
   590→   590→    organizationId: variant.organizationId,
   591→   591→    companyId,
   592→   592→    transactionType: 'fabric_consumed_for_stitching',
   593→   593→    quantity,
   594→   594→    costPerUnit: Number(fabricLocation.avgCost),
   595→   595→    referenceType: 'production_order',
   596→   596→  })
   597→   597→
   598→   598→  if (!txnResult.success) {
   599→   599→    return { source: 'fresh_production', error: `Fabric consumption failed: ${txnResult.error}` }
   600→   600→  }
   601→   601→
   602→   602→  // Create production order
   603→   603→  const productionOrder = await db.productionOrder.create({
   604→   604→    data: {
   605→   605→      organizationId: variant.organizationId,
   606→   606→      companyId,
   607→   607→      stitchedVariantId: orgVariantId,
   608→   608→      fabricVariantId: variant.fabricSourceVariantId,
   609→   609→      fabricLocationId: fabricLocation.locationId,
   610→   610→      quantity,
   611→   611→      status: 'fabric_reserved',
   612→   612→      stitchingCost: new Decimal(Number(variant.stitchingCharges) || 0),
   613→   613→      fabricCost: new Decimal(fabricCost),
   614→   614→      estimatedCompletionDate,
   615→   615→      fabricTxnId: txnResult.transactionId ?? null,
   616→   616→    } as Prisma.ProductionOrderUncheckedCreateInput,
   617→   617→  })
   618→   618→
   619→   619→  return {
   620→   620→    source: 'fresh_production',
   621→   621→    productionOrderId: productionOrder.id,
   622→   622→    estimatedCompletionDate,
   623→   623→  }
   624→   624→}

   626→   626→/**
   627→   627→ * Quarantine stock for theft/missing loss investigations.
   628→   628→ * Directly increments inventory_pools.reserved — does NOT call
   629→   629→ * process_inventory_transaction() since no actual movement has occurred.
   630→   630→ * This is a soft-hold that reduces available stock without touching on_hand.
   631→   631→ */
   632→   632→export async function quarantineStock(
   633→   633→  orgVariantId: string,
   634→   634→  locationId: string,
   635→   635→  quantity: number,
   636→   636→): Promise<{ success: boolean; error?: string }> {
   637→   637→  const pool = await db.inventoryPool.findUnique({
   638→   638→    where: { orgVariantId_locationId: { orgVariantId, locationId } },
   639→   639→  })
   640→   640→  if (!pool) {
   641→   641→    return { success: false, error: 'No inventory pool exists for this variant+location.' }
   642→   642→  }
   643→   643→  const available = pool.onHand - pool.reserved
   644→   644→  if (available < quantity) {
   645→   645→    return { success: false, error: `Insufficient available stock. Available: ${available}, required: ${quantity}.` }
   646→   646→  }
   647→   647→  await db.inventoryPool.update({
   648→   648→    where: { id: pool.id },
   649→   649→    data: { reserved: { increment: quantity } },
   650→   650→  })
   651→   651→  return { success: true }
   652→   652→}
   653→   653→
   654→   654→/**
   655→   655→ * Release quarantined stock (reverse of quarantineStock).
   656→   656→ * Directly decrements inventory_pools.reserved.
   657→   657→ * Used when resolving theft/missing investigations (regardless of outcome —
   658→   658→ * the write_off path creates a separate transaction that decrements on_hand).
   659→   659→ */
   660→   660→export async function releaseQuarantine(
   661→   661→  orgVariantId: string,
   662→   662→  locationId: string,
   663→   663→  quantity: number,
   664→   664→): Promise<void> {
   665→   665→  const pool = await db.inventoryPool.findUnique({
   666→   666→    where: { orgVariantId_locationId: { orgVariantId, locationId } },
   667→   667→  })
   668→   668→  if (!pool) return
   669→   669→  const newReserved = Math.max(0, pool.reserved - quantity)
   670→   670→  await db.inventoryPool.update({
   671→   671→    where: { id: pool.id },
   672→   672→    data: { reserved: newReserved },
   673→   673→  })
   674→   674→}

   684→   684→/**
   685→   685→ * Reserve stock for an order item. Increments inventory_pools.reserved
   686→   686→ * (does NOT touch on_hand — that happens at dispatch time). Records
   687→   687→ * an order_reserved transaction in the ledger.
   688→   688→ *
   689→   689→ * @returns { success, error? } — fails if insufficient available stock.
   690→   690→ */
   691→   691→export async function reserveStockForOrder(input: {
   692→   692→  orgVariantId: string
   693→   693→  locationId: string
   694→   694→  organizationId: string
   695→   695→  companyId: string
   696→   696→  employeeId?: string | null
   697→   697→  quantity: number
   698→   698→  orderId?: string
   699→   699→}): Promise<{ success: boolean; error?: string }> {
   700→   700→  // Check available stock first (available = onHand - reserved)
   701→   701→  const pool = await db.inventoryPool.findUnique({
   702→   702→    where: {
   703→   703→      orgVariantId_locationId: {
   704→   704→        orgVariantId: input.orgVariantId,
   705→   705→        locationId: input.locationId,
   706→   706→      },
   707→   707→    },
   708→   708→  })
   709→   709→  if (!pool) {
   710→   710→    return { success: false, error: 'No inventory pool exists for this variant+location.' }
   711→   711→  }
   712→   712→  const available = pool.onHand - pool.reserved
   713→   713→  if (available < input.quantity) {
   714→   714→    return {
   715→   715→      success: false,
   716→   716→      error: `Insufficient available stock. Available: ${available}, required: ${input.quantity}.`,
   717→   717→    }
   718→   718→  }
   719→   719→
   720→   720→  const result = await processInventoryTransaction({
   721→   721→    orgVariantId: input.orgVariantId,
   722→   722→    locationId: input.locationId,
   723→   723→    organizationId: input.organizationId,
   724→   724→    companyId: input.companyId,
   725→   725→    employeeId: input.employeeId,
   726→   726→    transactionType: 'order_reserved',
   727→   727→    quantity: input.quantity,
   728→   728→    referenceType: 'order',
   729→   729→    referenceId: input.orderId,
   730→   730→  })
   731→   731→
   732→   732→  if (!result.success) {
   733→   733→    return { success: false, error: result.error }
   734→   734→  }
   735→   735→  return { success: true }
   736→   736→}
   737→   737→

   739→   739→ * Unreserve stock for an order item (e.g. on order cancellation).

   747→   747→  companyId: string
   748→   748→  employeeId?: string | null
   749→   749→  quantity: number

   751→   751→}): Promise<{ success: boolean; error?: string }> {

   756→   756→    companyId: input.companyId,
   757→   757→    employeeId: input.employeeId,
   758→   758→    transactionType: 'order_unreserved',
   759→   759→    quantity: input.quantity,

   762→   762→  })
   763→   763→
   764→   764→  if (!result.success) {

   770→   770→/**
   771→   771→ * Dispatch stock for an order item — deducts on_hand AND releases

   780→   780→  employeeId?: string | null

   787→   787→    organizationId: input.organizationId,

   803→   803→/**
   804→   804→ * Restock inventory for an RTO (Return To Origin) order — session-free
   805→   805→ * version for use by courier polling jobs and webhooks (which have no user
   806→   806→ * session and therefore can't call processOrderReturn() which uses
   807→   807→ * getWorkspace()).
   808→   808→ *
   809→   809→ * For each DISPATCHED order item:
   810→   810→ *   - Looks up the original sale_dispatched transaction to recover the
   811→   811→ *     cost-per-unit that was locked at dispatch time.
   812→   812→ *   - Calls processInventoryTransaction with type 'return_resellable'
   813→   813→ *     (for stock_based items) or 'return_stitched_received' (for
   814→   814→ *     made_to_order items) — which increments onHand AND recalculates WAC.
   815→   815→ *   - Marks the order item with fulfillmentStatus='returned' +
   816→   816→ *     autoProcessedAsPerfect=true + needsReview=true so it surfaces in the
   817→   817→ *     exception-review queue for physical spot-checking (same as the manual
   818→   818→ *     processOrderReturn path).
   819→   819→ *
   820→   820→ * For CONFIRMED/PROCESSING (not-yet-dispatched) reserved items: calls
   821→   821→ * unreserveStockForOrder to release the reservation (no onHand change since
   822→   822→ * onHand was never decremented).
   823→   823→ *
   824→   824→ * This function is IDEMPOTENT — it skips items whose fulfillmentStatus is
   825→   825→ * already 'returned' (set by a prior restock call) so re-running a poll
   826→   826→ * cycle doesn't double-restock.
   827→   827→ *
   828→   828→ * @returns { success, itemsRestocked } — never throws (errors logged per-item).
   829→   829→ */
   830→   830→export async function restockOrderForRto(
   831→   831→  orderId: string,
   832→   832→  context: {
   833→   833→    organizationId: string
   834→   834→    companyId: string
   835→   835→    employeeId?: string | null
   836→   836→    returnReason?: string
   837→   837→  },
   838→   838→): Promise<{ success: boolean; itemsRestocked: number }> {
   839→   839→  const order = await db.order.findUnique({
   840→   840→    where: { id: orderId },
   841→   841→    select: {
   842→   842→      id: true,
   843→   843→      status: true,
   844→   844→      dispatchLocationId: true,
   845→   845→      organizationId: true,
   846→   846→      companyId: true,
   847→   847→      items: {
   848→   848→        include: {
   849→   849→          orgVariant: {
   850→   850→            select: { id: true, sku: true, costPrice: true, fulfillmentType: true },
   851→   851→          },
   852→   852→        },
   853→   853→      },
   854→   854→    },
   855→   855→  })
   856→   856→  if (!order) return { success: false, itemsRestocked: 0 }
   857→   857→
   858→   858→  const locationId = order.dispatchLocationId
   859→   859→  if (!locationId) {
   860→   860→    console.error(`[restockOrderForRto] Order ${orderId} has no dispatchLocationId — cannot restock`)
   861→   861→    return { success: false, itemsRestocked: 0 }
   862→   862→  }
   863→   863→
   864→   864→  let itemsRestocked = 0
   865→   865→
   866→   866→  for (const item of order.items) {
   867→   867→    // Idempotency: skip items already processed (fulfillmentStatus='returned')
   868→   868→    if (item.fulfillmentStatus === 'returned') continue
   869→   869→
   870→   870→    if (item.fulfillmentStatus === 'dispatched') {
   871→   871→      // Dispatched item — onHand was decremented at dispatch. Restock it.
   872→   872→      // Recover the cost-per-unit from the original sale_dispatched txn.
   873→   873→      const dispatchTxn = await db.inventoryTransaction.findFirst({
   874→   874→        where: {
   875→   875→          orgVariantId: item.orgVariantId,
   876→   876→          locationId,
   877→   877→          transactionType: 'sale_dispatched',
   878→   878→          referenceType: 'order',
   879→   879→          referenceId: orderId,
   880→   880→        },
   881→   881→        select: { costPerUnit: true },
   882→   882→        orderBy: { recordedAt: 'desc' },
   883→   883→      })
   884→   884→      const costPerUnit = dispatchTxn ? Number(dispatchTxn.costPerUnit) : Number(item.orgVariant.costPrice)
   885→   885→
   886→   886→      const txnType = item.fulfillmentTypeSnapshot === 'made_to_order'
   887→   887→        ? 'return_stitched_received'
   888→   888→        : 'return_resellable'
   889→   889→
   890→   890→      const txnResult = await processInventoryTransaction({
   891→   891→        orgVariantId: item.orgVariantId,
   892→   892→        locationId,
   893→   893→        organizationId: order.organizationId,
   894→   894→        companyId: order.companyId,
   895→   895→        employeeId: context.employeeId ?? null,
   896→   896→        transactionType: txnType,
   897→   897→        quantity: item.quantity,
   898→   898→        costPerUnit,
   899→   899→        referenceType: 'order',
   900→   900→        referenceId: orderId,
   901→   901→        notes: `Auto-processed RTO return (assumed ${txnType === 'return_resellable' ? 'resellable' : 'perfect'}). Reason: ${context.returnReason ?? 'courier returned'}`,
   902→   902→      })
   903→   903→
   904→   904→      if (txnResult.success) {
   905→   905→        await db.orderItem.update({
   906→   906→          where: { id: item.id },
   907→   907→          data: {
   908→   908→            fulfillmentStatus: 'returned',
   909→   909→            autoProcessedAsPerfect: true,
   910→   910→            needsReview: true,
   911→   911→          },
   912→   912→        })
   913→   913→        itemsRestocked++
   914→   914→      } else {
   915→   915→        console.error(`[restockOrderForRto] Failed to restock item ${item.id}: ${txnResult.error}`)
   916→   916→      }
   917→   917→    } else if (item.fulfillmentStatus === 'reserved') {
   918→   918→      // Reserved but not dispatched — just release the reservation.
   919→   919→      const unreserveResult = await unreserveStockForOrder({
   920→   920→        orgVariantId: item.orgVariantId,
   921→   921→        locationId,
   922→   922→        organizationId: order.organizationId,
   923→   923→        companyId: order.companyId,
   924→   924→        employeeId: context.employeeId ?? null,
   925→   925→        quantity: item.quantity,
   926→   926→        orderId,
   927→   927→      })
   928→   928→      if (unreserveResult.success) {
   929→   929→        await db.orderItem.update({
   930→   930→          where: { id: item.id },
   931→   931→          data: { fulfillmentStatus: 'returned' },
   932→   932→        })
   933→   933→        itemsRestocked++
   934→   934→      } else {
   935→   935→        console.error(`[restockOrderForRto] Failed to unreserve item ${item.id}: ${unreserveResult.error}`)
   936→   936→      }
   937→   937→    }
   938→   938→    // backordered / pending items: no inventory action needed
   939→   939→  }
   940→   940→
   941→   941→  return { success: true, itemsRestocked }
   942→   942→}
