     1→     1→import { db } from './db'
     2→     2→import { Decimal } from '@prisma/client/runtime/library'
     3→     3→import type { Prisma } from '@prisma/client'
     4→     4→import { insertAuditLog } from './audit'
     5→     5→
     6→     6→/**
     7→     7→ * THE CORE INVENTORY FUNCTION.
     8→     8→ *
     9→     9→ * This is the single point of truth for all stock movements. It:
    10→    10→ *   1. Finds or creates the inventory_pools row for variant+location
    11→    11→ *   2. Validates sufficient stock for OUT-direction transactions
    12→    12→ *   3. Recalculates WAC (weighted average cost) for IN-direction transactions
    13→    13→ *   4. Updates on_hand / reserved / avg_cost on the pool
    14→    14→ *   5. Inserts the immutable inventory_transactions ledger row
    15→    15→ *   6. Records avg_cost_history when avg_cost changes
    16→    16→ *   7. Flips track_inventory TRUE on made_to_order variants on first return
    17→    17→ *
    18→    18→ * ATOMICITY (INV-006 fix): Steps 1–7 are wrapped in a single
    19→    19→ * `db.$transaction` — either all writes commit, or none do. The
    20→    20→ * "ledger and pool always agree" guarantee now holds under partial failure.
    21→    21→ *
    22→    22→ * RESERVATION INVARIANT (INV-001 fix): For onHand-reducing transaction
    23→    23→ * types (cycle_count_adjust, damage_writeoff, theft_writeoff,
    24→    24→ * missing_writeoff, transit_loss, supplier_return,
    25→    25→ * fabric_consumed_for_stitching, transfer_out), if the new onHand drops
    26→    26→ * below the current reserved count, the newest-reserved OrderItems for
    27→    27→ * this variant+location are bumped to 'backordered' (oldest first
    28→    28→ * protected) until `reserved <= onHand` is restored. If bumping all
    29→    29→ * matching OrderItems still doesn't close the gap (ghost reservations
    30→    30→ * with no OrderItem row), newReserved is clamped to newOnHand and a
    31→    31→ * WARNING audit log is emitted.
    32→    32→ *
    33→    33→ * IMPORTANT: inventory_pools is NEVER written to directly from any other
    34→    34→ * code path — only through this function. This guarantees the ledger
    35→    35→ * and pool always agree.
    36→    36→ */
    37→    37→
    38→    38→export type TransactionType =
    39→    39→  | 'opening_stock'
    40→    40→  | 'purchase_received'
    41→    41→  | 'sale_dispatched'
    42→    42→  | 'order_reserved'
    43→    43→  | 'order_unreserved'
    44→    44→  | 'return_resellable'
    45→    45→  | 'return_damaged'
    46→    46→  | 'return_stitched_received'
    47→    47→  | 'transfer_out'
    48→    48→  | 'transfer_in'
    49→    49→  | 'cycle_count_adjust'
    50→    50→  | 'manual_adjustment_in'
    51→    51→  | 'damage_writeoff'
    52→    52→  | 'theft_writeoff'
    53→    53→  | 'missing_writeoff'
    54→    54→  | 'transit_loss'
    55→    55→  | 'supplier_return'
    56→    56→  | 'fabric_consumed_for_stitching'
    57→    57→
    58→    58→const OUT_TYPES: TransactionType[] = [
    59→    59→  'sale_dispatched',
    60→    60→  'transfer_out',
    61→    61→  'damage_writeoff',
    62→    62→  'theft_writeoff',
    63→    63→  'missing_writeoff',
    64→    64→  'transit_loss',
    65→    65→  'supplier_return',
    66→    66→  'fabric_consumed_for_stitching',
    67→    67→]
    68→    68→
    69→    69→const WAC_RECALC_TYPES: TransactionType[] = [
    70→    70→  'opening_stock',
    71→    71→  'purchase_received',
    72→    72→  'return_stitched_received',
    73→    73→  'transfer_in',
    74→    74→  'return_resellable',
    75→    75→]
    76→    76→
    77→    77→/**
    78→    78→ * Transaction types that REDUCE onHand (rather than just moving it between
    79→    79→ * pools, like transfer_out which is also a reducer at the source pool).
    80→    80→ *
    81→    81→ * Used by the reservation-invariant protection (INV-001 fix): if any of
    82→    82→ * these transaction types drops `newOnHand < newReserved`, the
    83→    83→ * newest-reserved OrderItems at this variant+location are bumped to
    84→    84→ * 'backordered' until the invariant is restored.
    85→    85→ */
    86→    86→const ONHAND_REDUCING_TYPES: TransactionType[] = [
    87→    87→  'cycle_count_adjust',
    88→    88→  'damage_writeoff',
    89→    89→  'theft_writeoff',
    90→    90→  'missing_writeoff',
    91→    91→  'transit_loss',
    92→    92→  'supplier_return',
    93→    93→  'fabric_consumed_for_stitching',
    94→    94→  'transfer_out',
    95→    95→]
    96→    96→
    97→    97→interface ProcessTxnInput {
    98→    98→  orgVariantId: string
    99→    99→  locationId: string
   100→   100→  organizationId: string
   101→   101→  companyId?: string | null
   102→   102→  employeeId?: string | null
   103→   103→  transactionType: TransactionType
   104→   104→  quantity: number // positive = in, negative = out
   105→   105→  costPerUnit?: number | null // if null, uses current avg_cost for OUT, or new_cost for IN
   106→   106→  referenceType?: string | null
   107→   107→  referenceId?: string | null
   108→   108→  notes?: string | null
   109→   109→  metadata?: Record<string, unknown> | null
   110→   110→}
   111→   111→
   112→   112→interface ProcessTxnResult {
   113→   113→  success: boolean
   114→   114→  transactionId?: string
   115→   115→  poolState?: {
   116→   116→    onHand: number
   117→   117→    reserved: number
   118→   118→    available: number
   119→   119→    avgCost: number
   120→   120→  }
   121→   121→  error?: string
   122→   122→}
   123→   123→
   124→   124→/**
   125→   125→ * Calculate new weighted average cost.
   126→   126→ * new_avg = (existing_qty × old_avg + new_qty × new_cost) / total_qty
   127→   127→ */
   128→   128→function calculateNewAvgCost(
   129→   129→  existingQty: number,
   130→   130→  oldAvg: number,
   131→   131→  newQty: number,
   132→   132→  newCost: number,
   133→   133→): number {
   134→   134→  const totalQty = existingQty + newQty
   135→   135→  if (totalQty <= 0) return 0
   136→   136→  return (existingQty * oldAvg + newQty * newCost) / totalQty
   137→   137→}
   138→   138→
   139→   139→/**
   140→   140→ * Process a single inventory transaction.
   141→   141→ * This function is the ONLY way to modify inventory_pools.
   142→   142→ */
   143→   143→export async function processInventoryTransaction(
   144→   144→  input: ProcessTxnInput,
   145→   145→): Promise<ProcessTxnResult> {
   146→   146→  const {
   147→   147→    orgVariantId,
   148→   148→    locationId,
   149→   149→    organizationId,
   150→   150→    companyId = null,
   151→   151→    employeeId = null,
   152→   152→    transactionType,
   153→   153→    quantity,
   154→   154→    referenceType = null,
   155→   155→    referenceId = null,
   156→   156→    notes = null,
   157→   157→    metadata = null,
   158→   158→  } = input
   159→   159→
   160→   160→  // For OUT transactions, quantity should be positive (we negate internally)
   161→   161→  // For IN transactions, quantity is positive
   162→   162→  const absQty = Math.abs(quantity)
   163→   163→
   164→   164→  try {
   165→   165→    // INV-006 fix: wrap the entire sequence (pool find/create → validate →
   166→   166→    // compute → optional reservation-bump → pool.update → ledger.create →
   167→   167→    // avgCostHistory.create) in a single database transaction. Either all
   168→   168→    // writes commit, or none do — restoring the "ledger and pool always
   169→   169→    // agree" guarantee claimed in the header comment.
   170→   170→    const result = await db.$transaction(async (tx) => {
   171→   171→      // 1. Find or create the inventory_pools row
   172→   172→      let pool = await tx.inventoryPool.findUnique({
   173→   173→        where: {
   174→   174→          orgVariantId_locationId: { orgVariantId, locationId },
   175→   175→        },
   176→   176→      })
   177→   177→
   178→   178→      if (!pool) {
   179→   179→        // First transaction ever for this variant+location — create pool with zeros
   180→   180→        pool = await tx.inventoryPool.create({
   181→   181→          data: {
   182→   182→            orgVariantId,
   183→   183→            locationId,
   184→   184→            organizationId,
   185→   185→            onHand: 0,
   186→   186→            reserved: 0,
   187→   187→            incoming: 0,
   188→   188→            avgCost: 0,
   189→   189→          },
   190→   190→        })
   191→   191→      }
   192→   192→
   193→   193→      // 2. Validate sufficient stock for OUT-direction transactions
   194→   194→      if (OUT_TYPES.includes(transactionType)) {
   195→   195→        const available = pool.onHand - pool.reserved
   196→   196→        if (available < absQty) {
   197→   197→          // Throw to abort the transaction — caught below and converted
   198→   198→          // back to a structured INSUFFICIENT_STOCK error response.
   199→   199→          throw new Error(
   200→   200→            `INSUFFICIENT_STOCK: Available ${available}, requested ${absQty}`,
   201→   201→          )
   202→   202→        }
   203→   203→      }
   204→   204→
   205→   205→      // 3. Determine cost_per_unit and compute new avg_cost
   206→   206→      const oldAvgCost = Number(pool.avgCost)
   207→   207→      let costPerUnit = input.costPerUnit ?? null
   208→   208→
   209→   209→      // For IN-direction WAC recalculation types
   210→   210→      if (WAC_RECALC_TYPES.includes(transactionType)) {
   211→   211→        if (costPerUnit === null) {
   212→   212→          costPerUnit = oldAvgCost // fallback if not provided
   213→   213→        }
   214→   214→      }
   215→   215→
   216→   216→      // For OUT types: use current avg_cost if not explicitly provided
   217→   217→      if (OUT_TYPES.includes(transactionType) && costPerUnit === null) {
   218→   218→        costPerUnit = oldAvgCost
   219→   219→      }
   220→   220→
   221→   221→      // For transfer_in: costPerUnit must be passed explicitly (sending location's cost)
   222→   222→      if (transactionType === 'transfer_in' && costPerUnit === null) {
   223→   223→        costPerUnit = oldAvgCost // fallback
   224→   224→      }
   225→   225→
   226→   226→      const finalCostPerUnit = costPerUnit ?? 0
   227→   227→
   228→   228→      // 4. Compute new pool state
   229→   229→      let newOnHand = pool.onHand
   230→   230→      let newReserved = pool.reserved
   231→   231→      let newAvgCost = oldAvgCost
   232→   232→      let newIncoming = pool.incoming
   233→   233→
   234→   234→      switch (transactionType) {
   235→   235→        case 'opening_stock':
   236→   236→          newOnHand += absQty
   237→   237→          newAvgCost = calculateNewAvgCost(pool.onHand, oldAvgCost, absQty, finalCostPerUnit)
   238→   238→          break
   239→   239→        case 'purchase_received':
   240→   240→          newOnHand += absQty
   241→   241→          newIncoming = Math.max(0, newIncoming - absQty)
   242→   242→          newAvgCost = calculateNewAvgCost(pool.onHand, oldAvgCost, absQty, finalCostPerUnit)
   243→   243→          break
   244→   244→        case 'sale_dispatched':
   245→   245→          newOnHand -= absQty
   246→   246→          newReserved = Math.max(0, newReserved - absQty)
   247→   247→          break
   248→   248→        case 'order_reserved':
   249→   249→          newReserved += absQty
   250→   250→          break
   251→   251→        case 'order_unreserved':
   252→   252→          newReserved = Math.max(0, newReserved - absQty)
   253→   253→          break
   254→   254→        case 'return_resellable':
   255→   255→          newOnHand += absQty
   256→   256→          newAvgCost = calculateNewAvgCost(pool.onHand, oldAvgCost, absQty, finalCostPerUnit)
   257→   257→          break
   258→   258→        case 'return_stitched_received':
   259→   259→          newOnHand += absQty
   260→   260→          newAvgCost = calculateNewAvgCost(pool.onHand, oldAvgCost, absQty, finalCostPerUnit)
   261→   261→          break
   262→   262→        case 'return_damaged':
   263→   263→          // No pool change — goes straight to stock_loss_records
   264→   264→          break
   265→   265→        case 'transfer_out':
   266→   266→          newOnHand -= absQty
   267→   267→          break
   268→   268→        case 'transfer_in':
   269→   269→          newOnHand += absQty
   270→   270→          // costPerUnit is the sending location's cost — do NOT recalculate WAC
   271→   271→          // The transferred stock keeps its original cost_per_unit exactly
   272→   272→          newAvgCost = calculateNewAvgCost(pool.onHand, oldAvgCost, absQty, finalCostPerUnit)
   273→   273→          break
   274→   274→        case 'cycle_count_adjust':
   275→   275→          // Set on_hand directly to counted value
   276→   276→          // quantity here represents the NEW on_hand value (positive)
   277→   277→          newOnHand = absQty
   278→   278→          break
   279→   279→        case 'manual_adjustment_in':
   280→   280→          // Manual positive adjustment — INCREMENT on_hand by the quantity
   281→   281→          // (unlike cycle_count_adjust which SETS on_hand to the quantity)
   282→   282→          newOnHand += absQty
   283→   283→          break
   284→   284→        case 'damage_writeoff':
   285→   285→        case 'theft_writeoff':
   286→   286→        case 'missing_writeoff':
   287→   287→        case 'transit_loss':
   288→   288→          newOnHand -= absQty
   289→   289→          break
   290→   290→        case 'supplier_return':
   291→   291→          newOnHand -= absQty
   292→   292→          break
   293→   293→        case 'fabric_consumed_for_stitching':
   294→   294→          newOnHand -= absQty
   295→   295→          break
   296→   296→      }
   297→   297→
   298→   298→      // --- RESERVATION INVARIANT PROTECTION (INV-001 fix) ---
   299→   299→      // After computing newOnHand, check if reserved > onHand.
   300→   300→      // For onHand-reducing transaction types, this means the available pool
   301→   301→      // has dropped below the reservations held against it — bump the
   302→   302→      // newest-reserved OrderItems to 'backordered' (oldest protected).
   303→   303→      if (
   304→   304→        ONHAND_REDUCING_TYPES.includes(transactionType) &&
   305→   305→        newReserved > newOnHand
   306→   306→      ) {
   307→   307→        const shortfall = newReserved - newOnHand
   308→   308→
   309→   309→        // Find reserved OrderItems for this variant+location, oldest first.
   310→   310→        // We iterate the list newest-first (reverse) to bump the most
   311→   311→        // recent reservations while protecting the earliest / oldest ones.
   312→   312→        const reservedItems = await tx.orderItem.findMany({
   313→   313→          where: {
   314→   314→            orgVariantId,
   315→   315→            reservedLocationId: locationId,
   316→   316→            fulfillmentStatus: 'reserved',
   317→   317→          },
   318→   318→          orderBy: { createdAt: 'asc' },
   319→   319→          select: {
   320→   320→            id: true,
   321→   321→            quantity: true,
   322→   322→            orderId: true,
   323→   323→            createdAt: true,
   324→   324→          },
   325→   325→        })
   326→   326→
   327→   327→        let bumpedQty = 0
   328→   328→        const reviewReason = `Inventory shortage — converted to backorder (${transactionType}${referenceId ? ', ref: ' + referenceId : ''})`
   329→   329→
   330→   330→        // Process newest first (end of the sorted-asc list)
   331→   331→        for (
   332→   332→          let i = reservedItems.length - 1;
   333→   333→          i >= 0 && bumpedQty < shortfall;
   334→   334→          i--
   335→   335→        ) {
   336→   336→          const item = reservedItems[i]
   337→   337→
   338→   338→          // Unreserve: decrement newReserved by the full item quantity
   339→   339→          newReserved = Math.max(0, newReserved - item.quantity)
   340→   340→
   341→   341→          // Set OrderItem to backordered + flag for human review
   342→   342→          await tx.orderItem.update({
   343→   343→            where: { id: item.id },
   344→   344→            data: {
   345→   345→              fulfillmentStatus: 'backordered',
   346→   346→              needsReview: true,
   347→   347→              needsReviewReason: reviewReason,
   348→   348→            },
   349→   349→          })
   350→   350→
   351→   351→          // Recompute the parent order's aggregated status from line items
   352→   352→          await tx.$queryRaw`SELECT recompute_order_status(${item.orderId}::TEXT)`
   353→   353→
   354→   354→          bumpedQty += item.quantity
   355→   355→        }
   356→   356→
   357→   357→        // Edge case: still not enough — "ghost" reservations exist on the
   358→   358→        // pool with no matching OrderItem row (e.g. orphaned by a past bug
   359→   359→        // or manually injected). Clamp newReserved to newOnHand to restore
   360→   360→        // the invariant and emit a WARNING audit log so an operator can
   361→   361→        // investigate the discrepancy.
   362→   362→        if (newReserved > newOnHand) {
   363→   363→          const clampedReserved = newReserved
   364→   364→          const shortfallRemaining = clampedReserved - newOnHand
   365→   365→          newReserved = newOnHand
   366→   366→
   367→   367→          insertAuditLog({
   368→   368→            action: 'inventory.reservation_clamp',
   369→   369→            entityType: 'inventory_pool',
   370→   370→            entityId: pool.id,
   371→   371→            organizationId,
   372→   372→            companyId,
   373→   373→            employeeId,
   374→   374→            oldValues: { reserved: clampedReserved, onHand: newOnHand },
   375→   375→            newValues: {
   376→   376→              reserved: newReserved,
   377→   377→              shortfallRemaining,
   378→   378→            },
   379→   379→            metadata: {
   380→   380→              transactionType,
   381→   381→              referenceType,
   382→   382→              referenceId,
   383→   383→              orgVariantId,
   384→   384→              locationId,
   385→   385→              shortfall,
   386→   386→              bumpedQty,
   387→   387→              reason: 'reservation_bump_exhausted_ghost_reservation',
   388→   388→            },
   389→   389→          })
   390→   390→        }
   391→   391→      }
   392→   392→      // --- END INV-001 fix ---
   393→   393→
   394→   394→      // 5. Update timestamps
   395→   395→      const now = new Date()
   396→   396→      const updateData: Record<string, unknown> = {
   397→   397→        onHand: newOnHand,
   398→   398→        reserved: newReserved,
   399→   399→        incoming: newIncoming,
   400→   400→        avgCost: newAvgCost,
   401→   401→        updatedAt: now,
   402→   402→      }
   403→   403→      if (
   404→   404→        transactionType === 'purchase_received' ||
   405→   405→        transactionType === 'opening_stock' ||
   406→   406→        transactionType === 'return_resellable' ||
   407→   407→        transactionType === 'return_stitched_received' ||
   408→   408→        transactionType === 'transfer_in'
   409→   409→      ) {
   410→   410→        updateData.lastReceivedAt = now
   411→   411→      }
   412→   412→      if (transactionType === 'sale_dispatched') {
   413→   413→        updateData.lastSoldAt = now
   414→   414→      }
   415→   415→      if (transactionType === 'cycle_count_adjust') {
   416→   416→        updateData.lastCountedAt = now
   417→   417→      }
   418→   418→
   419→   419→      await tx.inventoryPool.update({
   420→   420→        where: { id: pool.id },
   421→   421→        data: updateData,
   422→   422→      })
   423→   423→
   424→   424→      // 6. Handle track_inventory flip for made_to_order variants on first
   425→   425→      //    return OR on opening_stock entry (e.g. user confirms "pre-made bulk
   426→   426→      //    stock" for an MTO variant during product creation). One-way FALSE → TRUE.
   427→   427→      if (
   428→   428→        transactionType === 'return_stitched_received' ||
   429→   429→        transactionType === 'opening_stock'
   430→   430→      ) {
   431→   431→        const variant = await tx.orgProductVariant.findUnique({
   432→   432→          where: { id: orgVariantId },
   433→   433→          select: { trackInventory: true, fulfillmentType: true },
   434→   434→        })
   435→   435→        if (variant && !variant.trackInventory && variant.fulfillmentType === 'made_to_order') {
   436→   436→          // ONE-WAY flip: FALSE → TRUE (never back to FALSE)
   437→   437→          await tx.orgProductVariant.update({
   438→   438→            where: { id: orgVariantId },
   439→   439→            data: { trackInventory: true },
   440→   440→          })
   441→   441→        }
   442→   442→      }
   443→   443→
   444→   444→      // 7. Insert the inventory_transactions ledger row
   445→   445→      const txnQuantity = OUT_TYPES.includes(transactionType) ? -absQty : absQty
   446→   446→      const avgCostChanged = newAvgCost !== oldAvgCost
   447→   447→
   448→   448→      const txn = await tx.inventoryTransaction.create({
   449→   449→        data: {
   450→   450→          orgVariantId,
   451→   451→          locationId,
   452→   452→          organizationId,
   453→   453→          companyId,
   454→   454→          employeeId,
   455→   455→          transactionType,
   456→   456→          quantity: txnQuantity,
   457→   457→          costPerUnit: finalCostPerUnit,
   458→   458→          avgCostBefore: oldAvgCost,
   459→   459→          avgCostAfter: newAvgCost,
   460→   460→          referenceType,
   461→   461→          referenceId,
   462→   462→          notes,
   463→   463→          metadata: metadata ? JSON.stringify(metadata) : '{}',
   464→   464→          recordedAt: now,
   465→   465→        },
   466→   466→      })
   467→   467→
   468→   468→      // 8. Insert avg_cost_history if avg_cost changed
   469→   469→      if (avgCostChanged) {
   470→   470→        await tx.avgCostHistory.create({
   471→   471→          data: {
   472→   472→            orgVariantId,
   473→   473→            locationId,
   474→   474→            organizationId,
   475→   475→            avgCostBefore: oldAvgCost,
   476→   476→            avgCostAfter: newAvgCost,
   477→   477→            triggeredByTxnId: txn.id,
   478→   478→            triggerReason: transactionType,
   479→   479→          },
   480→   480→        })
   481→   481→      }
   482→   482→
   483→   483→      return {
   484→   484→        success: true,
   485→   485→        transactionId: txn.id,
   486→   486→        poolState: {
   487→   487→          onHand: newOnHand,
   488→   488→          reserved: newReserved,
   489→   489→          available: newOnHand - newReserved,
   490→   490→          avgCost: newAvgCost,
   491→   491→        },
   492→   492→      }
   493→   493→    })
   494→   494→
   495→   495→    return result
   496→   496→  } catch (err) {
   497→   497→    // INSUFFICIENT_STOCK is thrown from inside the transaction to force a
   498→   498→    // rollback. Convert it back to a structured error response (preserving
   499→   499→    // the original behavior callers depend on) without logging it as an
   500→   500→    // error — it is a validation failure, not a runtime fault.
   501→   501→    const msg = err instanceof Error ? err.message : 'Unknown inventory transaction error'
   502→   502→    if (msg.startsWith('INSUFFICIENT_STOCK:')) {
   503→   503→      return { success: false, error: msg }
   504→   504→    }
   505→   505→    console.error('[inventory] processInventoryTransaction error:', err)
   506→   506→    return { success: false, error: msg }
   507→   507→  }
   508→   508→}

   552→   552→/**
   553→   553→ * Canonical returned-stitched receipt processor (INV-002 fix).
   554→   554→ *
   555→   555→ * Single entry point for receiving returned-stitched items — unifies the
   556→   556→ * previously split flow across two routes:
   557→   557→ *   - POST /api/returned-stitched                       (created register row only)
   558→   558→ *   - POST /api/inventory/receive-returned-stitched     (created txn only)
   559→   559→ *
   560→   560→ * Both routes now delegate here so that EVERY receipt creates BOTH the
   561→   561→ * ReturnedStitchedInventory register row AND the corresponding ledger /
   562→   562→ * loss entry, with the bidirectional link set:
   563→   563→ *
   564→   564→ *   Non-damaged path (condition ∈ perfect|good|open_box):
   565→   565→ *     1. Calls processInventoryTransaction({ type: 'return_stitched_received' })
   566→   566→ *        → increments onHand, recalculates WAC, flips track_inventory=TRUE
   567→   567→ *       on made_to_order variants (one-way).
   568→   568→ *     2. Creates ReturnedStitchedInventory with status='available' and
   569→   569→ *        inventoryTxnId=txn.id (the link).
   570→   570→ *
   571→   571→ *   Damaged path (condition = 'damaged'):
   572→   572→ *     1. Calls recordStockLoss({ createInventoryTransaction: false })
   573→   573→ *        → creates a StockLossRecord (dedup-safe, sourceModule='returned_stitched')
   574→   574→ *        but does NOT decrement onHand (the returned item was never added to
   575→   575→ *        stock, so there's nothing to remove — preserves the behavior of the
   576→   576→ *        existing /api/inventory/receive-returned-stitched damaged path).
   577→   577→ *     2. Creates ReturnedStitchedInventory with status='written_off',
   578→   578→ *        writtenOffAt=now, writtenOffById=employeeId, inventoryTxnId=NULL
   579→   579→ *        (no stock movement occurred).
   580→   580→ *
   581→   581→ * ATOMICITY: The register-row creation runs inside db.$transaction. Note
   582→   582→ * that processInventoryTransaction / recordStockLoss internally use the
   583→   583→ * global db client (not a passed-in tx), so they execute as SEPARATE
   584→   584→ * transactions — full cross-call atomicity would require refactoring those
   585→   585→ * helpers to accept a tx client (out of INV-002 scope, same caveat as the
   586→   586→ * INV-006 fix). The current arrangement matches the behavior of the
   587→   587→ * existing receive-returned-stitched route and is strictly better than the
   588→   588→ * prior split flow: the register row is now consistently created with the
   589→   589→ * link set, instead of being skipped entirely.
   590→   590→ */
   591→   591→export async function processReturnedStitchedReceipt(
   592→   592→  input: ProcessReturnedStitchedInput,
   593→   593→): Promise<ProcessReturnedStitchedResult> {
   594→   594→  const {
   595→   595→    organizationId,
   596→   596→    companyId,
   597→   597→    orgVariantId,
   598→   598→    locationId,
   599→   599→    quantity,
   600→   600→    condition,
   601→   601→    totalCost,
   602→   602→    suggestedResalePrice = null,
   603→   603→    originalOrderReference = null,
   604→   604→    returnReason,
   605→   605→    photos = [],
   606→   606→    notes = null,
   607→   607→    employeeId,
   608→   608→  } = input
   609→   609→
   610→   610→  if (quantity <= 0) {
   611→   611→    return { success: false, error: 'Quantity must be positive.' }
   612→   612→  }
   613→   613→
   614→   614→  const isDamaged = condition === 'damaged'
   615→   615→  const costPerUnit = totalCost / quantity
   616→   616→
   617→   617→  try {
   618→   618→    if (isDamaged) {
   619→   619→      // ── Damaged path: record loss, no stock movement ──
   620→   620→      const { recordStockLoss } = await import('@/lib/stock-loss')
   621→   621→      const lossResult = await recordStockLoss({
   622→   622→        organizationId,
   623→   623→        companyId,
   624→   624→        orgVariantId,
   625→   625→        locationId,
   626→   626→        lossType: 'damaged',
   627→   627→        sourceModule: 'returned_stitched',
   628→   628→        quantity,
   629→   629→        costPerUnit,
   630→   630→        employeeId,
   631→   631→        subType: 'confirmed',
   632→   632→        damageType: 'other',
   633→   633→        responsibleParty: 'courier',
   634→   634→        notes: `Damaged returned stitched item. ${notes || ''}`,
   635→   635→        // createInventoryTransaction=false — this endpoint does NOT add
   636→   636→        // stock for damaged items (the loss is just recorded, stock stays
   637→   637→        // unchanged since the returned item was never added in the first place)
   638→   638→        createInventoryTransaction: false,
   639→   639→      })
   640→   640→
   641→   641→      if (!lossResult.success) {
   642→   642→        return { success: false, error: `Failed to record damaged loss: ${lossResult.error}` }
   643→   643→      }
   644→   644→
   645→   645→      // Create the ReturnedStitchedInventory register row with status='written_off'.
   646→   646→      // inventoryTxnId stays NULL — no stock movement occurred. Wrapped in
   647→   647→      // db.$transaction per the INV-002 spec (single write here, but the
   648→   648→      // wrapper documents the atomicity intent for future multi-write extensions).
   649→   649→      const record = await db.$transaction(async (tx) => {
   650→   650→        return tx.returnedStitchedInventory.create({
   651→   651→          data: {
   652→   652→            organizationId,
   653→   653→            companyId,
   654→   654→            orgVariantId,
   655→   655→            quantity,
   656→   656→            condition,
   657→   657→            totalCost,
   658→   658→            suggestedResalePrice,
   659→   659→            originalOrderReference,
   660→   660→            returnReason,
   661→   661→            status: 'written_off',
   662→   662→            photos: JSON.stringify(photos),
   663→   663→            notes,
   664→   664→            receivedById: employeeId,
   665→   665→            writtenOffAt: new Date(),
   666→   666→            writtenOffById: employeeId,
   667→   667→            writeOffReason: 'Damaged on return',
   668→   668→            inventoryTxnId: null,
   669→   669→          },
   670→   670→        })
   671→   671→      })
   672→   672→
   673→   673→      return {
   674→   674→        success: true,
   675→   675→        recordId: record.id,
   676→   676→        inventoryTxnId: null,
   677→   677→        lossRecordId: lossResult.lossRecordId ?? null,
   678→   678→        status: 'written_off',
   679→   679→        wasDuplicate: lossResult.wasDuplicate,
   680→   680→      }
   681→   681→    }
   682→   682→
   683→   683→    // ── Non-damaged path: increment stock via processInventoryTransaction ──
   684→   684→    const txnResult = await processInventoryTransaction({
   685→   685→      orgVariantId,
   686→   686→      locationId,
   687→   687→      organizationId,
   688→   688→      companyId,
   689→   689→      employeeId,
   690→   690→      transactionType: 'return_stitched_received',
   691→   691→      quantity,
   692→   692→      costPerUnit,
   693→   693→      referenceType: originalOrderReference ? 'order' : 'manual',
   694→   694→      referenceId: originalOrderReference || null,
   695→   695→      notes: `Returned stitched item (${condition}). ${notes || ''}`,
   696→   696→    })
   697→   697→
   698→   698→    if (!txnResult.success) {
   699→   699→      return { success: false, error: `Failed to receive returned item: ${txnResult.error}` }
   700→   700→    }
   701→   701→
   702→   702→    // Create the ReturnedStitchedInventory register row with status='available'
   703→   703→    // and link it to the inventory transaction id. Wrapped in db.$transaction
   704→   704→    // per the INV-002 spec.
   705→   705→    const record = await db.$transaction(async (tx) => {
   706→   706→      return tx.returnedStitchedInventory.create({
   707→   707→        data: {
   708→   708→          organizationId,
   709→   709→          companyId,
   710→   710→          orgVariantId,
   711→   711→          quantity,
   712→   712→          condition,
   713→   713→          totalCost,
   714→   714→          suggestedResalePrice,
   715→   715→          originalOrderReference,
   716→   716→          returnReason,
   717→   717→          status: 'available',
   718→   718→          photos: JSON.stringify(photos),
   719→   719→          notes,
   720→   720→          receivedById: employeeId,
   721→   721→          inventoryTxnId: txnResult.transactionId ?? null,
   722→   722→        },
   723→   723→      })
   724→   724→    })
   725→   725→
   726→   726→    return {
   727→   727→      success: true,
   728→   728→      recordId: record.id,
   729→   729→      inventoryTxnId: txnResult.transactionId ?? null,
   730→   730→      lossRecordId: null,
   731→   731→      status: 'available',
   732→   732→    }
   733→   733→  } catch (err) {
   734→   734→    const msg = err instanceof Error ? err.message : 'Unknown returned-stitched receipt error'
   735→   735→    console.error('[inventory] processReturnedStitchedReceipt error:', err)
   736→   736→    return { success: false, error: msg }
   737→   737→  }
   738→   738→}

   827→   827→/**
   828→   828→ * Generate a unique, sequential PO number per organization per year.
   829→   829→ *
   830→   830→ * Format: PO-YYYY-NNN (e.g. PO-2026-001, PO-2026-002, ...)
   831→   831→ *
   832→   832→ * ATOMIC & RACE-FREE: uses the get_next_sequence_number() Postgres function
   833→   833→ * which does INSERT ... ON CONFLICT DO UPDATE ... RETURNING in a single
   834→   834→ * atomic statement. Two concurrent calls will each get a DIFFERENT number
   835→   835→ * (guaranteed by Postgres's row-level locking).
   836→   836→ *
   837→   837→ * This REPLACES the old count+1 pattern which raced under concurrency —
   838→   838→ * two simultaneous PO creations would generate the same number, causing
   839→   839→ * a unique-constraint 500 error. See INVENTORY_AUDIT.md CRITICAL #2.
   840→   840→ *
   841→   841→ * The sequence counter persists in the number_sequences table, so it
   842→   842→ * survives restarts and is consistent across all server instances.
   843→   843→ */
   844→   844→export async function generatePoNumber(organizationId: string): Promise<string> {
   845→   845→  const year = new Date().getFullYear()
   846→   846→  const prefix = `PO-${year}-`
   847→   847→
   848→   848→  const result = await db.$queryRaw<{ n: number }[]>`
   849→   849→    SELECT get_next_sequence_number(${organizationId}::TEXT, 'po_number', ${year}::INT) AS n
   850→   850→  `
   851→   851→  const seq = result[0].n
   852→   852→
   853→   853→  return `${prefix}${String(seq).padStart(3, '0')}`
   854→   854→}

   901→   901→/**
   902→   902→ * Check and fulfill a made-to-order variant.
   903→   903→ *
   904→   904→ * The central decision function:
   905→   905→ * 1. Check if returned stock is available for this variant
   906→   906→ * 2. If yes: return { source: 'existing_stock', location_id, available }
   907→   907→ * 3. If no: create a production order, consume fabric, return { source: 'fresh_production' }
   908→   908→ */
   909→   909→export async function checkAndFulfillMadeToOrderVariant(
   910→   910→  orgVariantId: string,
   911→   911→  quantity: number,
   912→   912→  companyId: string,
   913→   913→  preferredLocationId?: string,
   914→   914→): Promise<{
   915→   915→  source: 'existing_stock' | 'fresh_production'
   916→   916→  locationId?: string
   917→   917→  available?: number
   918→   918→  productionOrderId?: string
   919→   919→  estimatedCompletionDate?: Date
   920→   920→  error?: string
   921→   921→}> {
   922→   922→  // 1. Check returned stock availability
   923→   923→  const availability = await checkReturnedStockAvailability(orgVariantId)
   924→   924→  const totalAvailable = availability.reduce((sum, a) => sum + a.available, 0)
   925→   925→
   926→   926→  if (totalAvailable >= quantity) {
   927→   927→    // Use existing stock — find the best location (most available)
   928→   928→    const best = availability
   929→   929→      .filter((a) => a.available > 0)
   930→   930→      .sort((a, b) => b.available - a.available)[0]
   931→   931→    return {
   932→   932→      source: 'existing_stock',
   933→   933→      locationId: best.locationId,
   934→   934→      available: best.available,
   935→   935→    }
   936→   936→  }
   937→   937→
   938→   938→  // 2. Not enough returned stock — create a production order
   939→   939→  const variant = await db.orgProductVariant.findUnique({
   940→   940→    where: { id: orgVariantId },
   941→   941→    select: {
   942→   942→      id: true,
   943→   943→      fabricSourceVariantId: true,
   944→   944→      stitchingCharges: true,
   945→   945→      productionDays: true,
   946→   946→      organizationId: true,
   947→   947→    },
   948→   948→  })
   949→   949→
   950→   950→  if (!variant) return { source: 'fresh_production', error: 'Variant not found' }
   951→   951→  if (!variant.fabricSourceVariantId) {
   952→   952→    return { source: 'fresh_production', error: 'No fabric source variant linked to this made_to_order variant' }
   953→   953→  }
   954→   954→
   955→   955→  // Find fabric stock at the preferred location or any location with stock
   956→   956→  const fabricPools = await db.inventoryPool.findMany({
   957→   957→    where: {
   958→   958→      orgVariantId: variant.fabricSourceVariantId,
   959→   959→      onHand: { gt: 0 },
   960→   960→    },
   961→   961→    include: { location: { select: { id: true, name: true } } },
   962→   962→  })
   963→   963→
   964→   964→  const fabricLocation = preferredLocationId
   965→   965→    ? fabricPools.find((p) => p.locationId === preferredLocationId)
   966→   966→    : fabricPools[0]
   967→   967→
   968→   968→  if (!fabricLocation || fabricLocation.onHand - fabricLocation.reserved < quantity) {
   969→   969→    return {
   970→   970→      source: 'fresh_production',
   971→   971→      error: `Insufficient fabric stock. Available: ${fabricLocation?.onHand ?? 0}, required: ${quantity}`,
   972→   972→    }
   973→   973→  }
   974→   974→
   975→   975→  const fabricCost = Number(fabricLocation.avgCost) * quantity
   976→   976→  const estimatedCompletionDate = new Date()
   977→   977→  estimatedCompletionDate.setDate(estimatedCompletionDate.getDate() + (variant.productionDays || 5))
   978→   978→
   979→   979→  // INV-004 fix: previously the fabric_consumed_for_stitching
   980→   980→  // InventoryTransaction was created BEFORE the ProductionOrder record —
   981→   981→  // leaving the forward link (InventoryTransaction.referenceId →
   982→   982→  // ProductionOrder.id) NULL because the PO id didn't exist yet.
   983→   983→  // Now we create the ProductionOrder FIRST (with fabricTxnId=null),
   984→   984→  // then consume fabric with referenceId=productionOrder.id, then
   985→   985→  // backfill fabricTxnId on the ProductionOrder. All three writes are
   986→   986→  // wrapped in a db.$transaction so a failure in fabric consumption
   987→   987→  // rolls back the ProductionOrder creation (no orphan POs).
   988→   988→  // processInventoryTransaction internally uses db.$transaction (since
   989→   989→  // the INV-006 fix), which Prisma nests as a savepoint inside this
   990→   990→  // outer transaction.
   991→   991→  try {
   992→   992→    const result = await db.$transaction(async (tx) => {
   993→   993→      // 1. Create the ProductionOrder record (fabricTxnId is NULL
   994→   994→      //    at this point — backfilled in step 3).
   995→   995→      const po = await tx.productionOrder.create({
   996→   996→        data: {
   997→   997→          organizationId: variant.organizationId,
   998→   998→          companyId,
   999→   999→          stitchedVariantId: orgVariantId,
  1000→  1000→          fabricVariantId: variant.fabricSourceVariantId,
  1001→  1001→          fabricLocationId: fabricLocation.locationId,
  1002→  1002→          quantity,
  1003→  1003→          status: 'fabric_reserved',
  1004→  1004→          stitchingCost: new Decimal(Number(variant.stitchingCharges) || 0),
  1005→  1005→          fabricCost: new Decimal(fabricCost),
  1006→  1006→          estimatedCompletionDate,
  1007→  1007→          fabricTxnId: null,
  1008→  1008→        } as Prisma.ProductionOrderUncheckedCreateInput,
  1009→  1009→      })
  1010→  1010→
  1011→  1011→      // 2. Consume fabric with referenceId=po.id so the
  1012→  1012→      //    InventoryTransaction → ProductionOrder forward link is set
  1013→  1013→      //    at creation time (no subsequent mutation needed).
  1014→  1014→      const txnResult = await processInventoryTransaction({
  1015→  1015→        orgVariantId: variant.fabricSourceVariantId,
  1016→  1016→        locationId: fabricLocation.locationId,
  1017→  1017→        organizationId: variant.organizationId,
  1018→  1018→        companyId,
  1019→  1019→        transactionType: 'fabric_consumed_for_stitching',
  1020→  1020→        quantity,
  1021→  1021→        costPerUnit: Number(fabricLocation.avgCost),
  1022→  1022→        referenceType: 'production_order',
  1023→  1023→        referenceId: po.id,
  1024→  1024→      })
  1025→  1025→
  1026→  1026→      if (!txnResult.success) {
  1027→  1027→        // Throwing aborts the outer db.$transaction, rolling back the
  1028→  1028→        // ProductionOrder.create above.
  1029→  1029→        throw new Error(`Fabric consumption failed: ${txnResult.error}`)
  1030→  1030→      }
  1031→  1031→
  1032→  1032→      // 3. Backfill fabricTxnId on the ProductionOrder so the reverse
  1033→  1033→      //    link (ProductionOrder → InventoryTransaction) is set.
  1034→  1034→      await tx.productionOrder.update({
  1035→  1035→        where: { id: po.id },
  1036→  1036→        data: { fabricTxnId: txnResult.transactionId ?? null },
  1037→  1037→      })
  1038→  1038→
  1039→  1039→      return po
  1040→  1040→    })
  1041→  1041→
  1042→  1042→    return {
  1043→  1043→      source: 'fresh_production',
  1044→  1044→      productionOrderId: result.id,
  1045→  1045→      estimatedCompletionDate,
  1046→  1046→    }
  1047→  1047→  } catch (err) {
  1048→  1048→    const message = err instanceof Error ? err.message : 'Unknown error'
  1049→  1049→    return { source: 'fresh_production', error: message }
  1050→  1050→  }
  1051→  1051→}

  1240→  1240→/**

  1242→  1242→ * version for use by courier polling jobs and webhooks (which have no user
  1243→  1243→ * session and therefore can't call processOrderReturn() which uses
  1244→  1244→ * getWorkspace()).
  1245→  1245→ *
  1246→  1246→ * For each DISPATCHED order item:
  1247→  1247→ *   - Looks up the original sale_dispatched transaction to recover the
  1248→  1248→ *     cost-per-unit that was locked at dispatch time.
  1249→  1249→ *   - Calls processInventoryTransaction with type 'return_resellable'
  1250→  1250→ *     (for stock_based items) or 'return_stitched_received' (for
  1251→  1251→ *     made_to_order items) — which increments onHand AND recalculates WAC.
  1252→  1252→ *   - Marks the order item with fulfillmentStatus='returned' +
  1253→  1253→ *     autoProcessedAsPerfect=true + needsReview=true so it surfaces in the
  1254→  1254→ *     exception-review queue for physical spot-checking (same as the manual
  1255→  1255→ *     processOrderReturn path).
  1256→  1256→ *
  1257→  1257→ * For CONFIRMED/PROCESSING (not-yet-dispatched) reserved items: calls
  1258→  1258→ * unreserveStockForOrder to release the reservation (no onHand change since
  1259→  1259→ * onHand was never decremented).
  1260→  1260→ *
  1261→  1261→ * This function is IDEMPOTENT — it skips items whose fulfillmentStatus is
  1262→  1262→ * already 'returned' (set by a prior restock call) so re-running a poll
  1263→  1263→ * cycle doesn't double-restock.
  1264→  1264→ *
  1265→  1265→ * @returns { success, itemsRestocked } — never throws (errors logged per-item).
  1266→  1266→ */
  1267→  1267→export async function restockOrderForRto(
  1268→  1268→  orderId: string,
  1269→  1269→  context: {
  1270→  1270→    organizationId: string
  1271→  1271→    companyId: string
  1272→  1272→    employeeId?: string | null
  1273→  1273→    returnReason?: string
  1274→  1274→  },
  1275→  1275→): Promise<{ success: boolean; itemsRestocked: number }> {
  1276→  1276→  const order = await db.order.findUnique({
  1277→  1277→    where: { id: orderId },
  1278→  1278→    select: {
  1279→  1279→      id: true,
  1280→  1280→      status: true,
  1281→  1281→      dispatchLocationId: true,
  1282→  1282→      organizationId: true,
  1283→  1283→      companyId: true,
  1284→  1284→      items: {
  1285→  1285→        include: {
  1286→  1286→          orgVariant: {
  1287→  1287→            select: { id: true, sku: true, costPrice: true, fulfillmentType: true },
  1288→  1288→          },
  1289→  1289→        },
  1290→  1290→      },
  1291→  1291→    },
  1292→  1292→  })
  1293→  1293→  if (!order) return { success: false, itemsRestocked: 0 }
  1294→  1294→
  1295→  1295→  const locationId = order.dispatchLocationId
  1296→  1296→  if (!locationId) {
  1297→  1297→    console.error(`[restockOrderForRto] Order ${orderId} has no dispatchLocationId — cannot restock`)
  1298→  1298→    return { success: false, itemsRestocked: 0 }
  1299→  1299→  }
  1300→  1300→
  1301→  1301→  let itemsRestocked = 0
  1302→  1302→
  1303→  1303→  for (const item of order.items) {
  1304→  1304→    // Idempotency: skip items already processed (fulfillmentStatus='returned')
  1305→  1305→    if (item.fulfillmentStatus === 'returned') continue
  1306→  1306→

  1308→  1308→      // Dispatched item — onHand was decremented at dispatch. Restock it.

  1311→  1311→        where: {
  1312→  1312→          orgVariantId: item.orgVariantId,
  1313→  1313→          locationId,
  1314→  1314→          transactionType: 'sale_dispatched',
  1315→  1315→          referenceType: 'order',
  1316→  1316→          referenceId: orderId,
  1317→  1317→        },
  1318→  1318→        select: { costPerUnit: true },
  1319→  1319→        orderBy: { recordedAt: 'desc' },
  1320→  1320→      })
  1321→  1321→      const costPerUnit = dispatchTxn ? Number(dispatchTxn.costPerUnit) : Number(item.orgVariant.costPrice)
  1322→  1322→
  1323→  1323→      const txnType = item.fulfillmentTypeSnapshot === 'made_to_order'
  1324→  1324→        ? 'return_stitched_received'
  1325→  1325→        : 'return_resellable'
  1326→  1326→
  1327→  1327→      const txnResult = await processInventoryTransaction({
  1328→  1328→        orgVariantId: item.orgVariantId,
  1329→  1329→        locationId,
  1330→  1330→        organizationId: order.organizationId,
  1331→  1331→        companyId: order.companyId,
  1332→  1332→        employeeId: context.employeeId ?? null,
  1333→  1333→        transactionType: txnType,
  1334→  1334→        quantity: item.quantity,
  1335→  1335→        costPerUnit,
  1336→  1336→        referenceType: 'order',
  1337→  1337→        referenceId: orderId,
  1338→  1338→        notes: `Auto-processed RTO return (assumed ${txnType === 'return_resellable' ? 'resellable' : 'perfect'}). Reason: ${context.returnReason ?? 'courier returned'}`,
  1339→  1339→      })
  1340→  1340→
  1341→  1341→      if (txnResult.success) {

  1343→  1343→          where: { id: item.id },
  1344→  1344→          data: {
  1345→  1345→            fulfillmentStatus: 'returned',
  1346→  1346→            autoProcessedAsPerfect: true,
  1347→  1347→            needsReview: true,
  1348→  1348→          },
  1349→  1349→        })
  1350→  1350→        itemsRestocked++
  1351→  1351→      } else {
  1352→  1352→        console.error(`[restockOrderForRto] Failed to restock item ${item.id}: ${txnResult.error}`)
  1353→  1353→      }
  1354→  1354→    } else if (item.fulfillmentStatus === 'reserved') {
  1355→  1355→      // Reserved but not dispatched — just release the reservation.
  1356→  1356→      const unreserveResult = await unreserveStockForOrder({
  1357→  1357→        orgVariantId: item.orgVariantId,
  1358→  1358→        locationId,
  1359→  1359→        organizationId: order.organizationId,
  1360→  1360→        companyId: order.companyId,
  1361→  1361→        employeeId: context.employeeId ?? null,
  1362→  1362→        quantity: item.quantity,
  1363→  1363→        orderId,
  1364→  1364→      })
  1365→  1365→      if (unreserveResult.success) {
  1366→  1366→        await db.orderItem.update({
  1367→  1367→          where: { id: item.id },
  1368→  1368→          data: { fulfillmentStatus: 'returned' },
  1369→  1369→        })
  1370→  1370→        itemsRestocked++
  1371→  1371→      } else {
  1372→  1372→        console.error(`[restockOrderForRto] Failed to unreserve item ${item.id}: ${unreserveResult.error}`)
  1373→  1373→      }
  1374→  1374→    }
  1375→  1375→    // backordered / pending items: no inventory action needed
  1376→  1376→  }
  1377→  1377→
  1378→  1378→  return { success: true, itemsRestocked }
  1379→  1379→}
  1380→  1380→
