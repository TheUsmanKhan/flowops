     1→/**
     2→ * Booking Server Actions — reusable courier booking logic.
     3→ *
     4→ * Extracted from /api/booking-workbench/book/route.ts so that the same
     5→ * booking logic can be called from:
     6→ *   1. The manual Booking Workbench (via the API route)
     7→ *   2. Auto-booking after order creation (when courierBookingMode='automatic')
     8→ *
     9→ * The function handles ONLY orders (not exchange shipments) — exchange
    10→ * shipment booking is handled by the exchange-shipment flow separately.
    11→ */
    12→
    13→import { db } from '@/lib/db'
    14→import { getWorkspace, requirePermission, ApiError } from '@/lib/workspace'
    15→import { PERMISSIONS } from '@/lib/permissions'
    16→import { insertAuditLog } from '@/lib/audit'
    17→import { insertMetricEvent } from '@/lib/metrics'
    18→import { decryptCredentials } from '@/lib/utils/encryption'
    19→import { getCourierAdapter } from '@/lib/integrations/registry'
    20→import { executeLoggedIntegrationAction } from '@/lib/integrations/logged-call'
    21→import { revalidateCityAtBookingTime } from '@/lib/integrations/city-matcher'
    22→import { determinePostExOrderType } from '@/lib/integrations/couriers/postex.order-type'
    23→import { calculateOrderWeightKg } from '@/lib/utils/order-weight'
    24→import {
    25→  parseLeopardPreferences,
    26→  buildLeopardSpecialInstructions,
    27→} from '@/lib/integrations/couriers/leopard-preferences'
    28→import type { BookShipmentInput, BookShipmentResult } from '@/lib/integrations/types'
    29→
    30→interface ActionResult<T = unknown> {
    31→  success: boolean
    32→  data?: T
    33→  error?: string
    34→}
    35→
    36→export interface BookOrderResult {
    37→  trackingNumber: string
    38→  orderType: string
    39→  providerStatus: string | null
    40→  courierBookingStatus: 'booked' | 'failed'
    41→}
    42→
    43→export interface BookOrderOptions {
    44→  /** The order to book. */
    45→  orderId: string
    46→  /** The courier integration to book with. */
    47→  companyIntegrationId: string
    48→  /** Optional per-booking overrides (from the Workbench UI or auto-booking). */
    49→  customerName?: string
    50→  customerPhone?: string
    51→  deliveryAddress?: string
    52→  deliveryCity?: string
    53→  codAmount?: number
    54→  orderType?: string
    55→  transactionNotes?: string
    56→  itemDescription?: string
    57→  orderRefNumber?: string
    58→  pickupAddressCode?: string
    59→}
    60→
    61→/**
    62→ * Book a single order with the selected courier.
    63→ *
    64→ * This is the SINGLE source of truth for order booking logic. Both the
    65→ * manual Booking Workbench (via /api/booking-workbench/book) and the
    66→ * auto-booking hook in createManualOrder() call this function.
    67→ *
    68→ * Flow:
    69→ *   1. Fetch the order + items + variant weights + customer
    70→ *   2. Validate the city (with live PostEx fallback on cache miss)
    71→ *   3. Compute weight + orderType
    72→ *   4. Get the pickup address code (from override or default)
    73→ *   5. Build the BookShipmentInput (using stored orderRefNumber/orderDetail/
    74→ *      notesForCourier as defaults, with per-call overrides taking precedence)
    75→ *   6. Call the courier adapter's bookShipment() via executeLoggedIntegrationAction
    76→ *   7. Update the order with trackingNumber + courierBookingStatus='booked'
    77→ *
    78→ * On failure: sets courierBookingStatus='failed' on the order and returns
    79→ * success=false with the error message. Does NOT throw.
    80→ */
    81→export async function bookOrderWithCourier(
    82→  options: BookOrderOptions,
    83→): Promise<ActionResult<BookOrderResult>> {
    84→  // Extract orderId + companyIntegrationId BEFORE the try block so they're
    85→  // accessible in the catch block (const is block-scoped in JS).
    86→  const { orderId, companyIntegrationId } = options
    87→  if (!orderId || !companyIntegrationId) {
    88→    return { success: false, error: 'orderId and companyIntegrationId are required' }
    89→  }
    90→
    91→  // ════════════════════════════════════════════════════════════════
    92→  // TEMPORARY DIAGNOSTIC INSTRUMENTATION — booking flow timing.
    93→  // Measures each step's wall-clock duration and logs a structured
    94→  // breakdown at the end. Will be removed after diagnosis.
    95→  // ════════════════════════════════════════════════════════════════
    96→  const T: Record<string, number> = {}
    97→  const marks: Record<string, number> = {}
    98→  function mark(label: string) { marks[label] = Date.now() }
    99→  function measure(from: string, to: string, key: string) { T[key] = marks[to] - marks[from] }
   100→  const flowStart = Date.now()
   101→  mark('start')
   102→
   103→  try {
   104→    mark('authStart')
   105→    const ctx = await getWorkspace()
   106→    await requirePermission(ctx, PERMISSIONS.ORDERS_FULFILL)
   107→    mark('authEnd')
   108→    measure('authStart', 'authEnd', '1_auth_resolution')
   109→
   110→    // ── Fetch the integration ──
   111→    mark('integrationFetchStart')
   112→    const integration = await db.companyIntegration.findFirst({
   113→      where: { id: companyIntegrationId, companyId: ctx.company.id, isActive: true },
   114→      include: { provider: true },
   115→    })
   116→    mark('integrationFetchEnd')
   117→    measure('integrationFetchStart', 'integrationFetchEnd', '2a_integration_fetch')
   118→    if (!integration) {
   119→      const reason = `Courier integration not found or inactive (integrationId: ${companyIntegrationId}).`
   120→      await db.order.update({
   121→        where: { id: orderId },
   122→        data: { courierBookingStatus: 'failed', courierBookingFailureReason: reason },
   123→      })
   124→      return { success: false, error: reason }
   125→    }
   126→
   127→    const providerKey = integration.provider.providerKey
   128→    if (providerKey !== 'postex') {
   129→      return {
   130→        success: false,
   131→        error: `Booking not yet implemented for provider '${providerKey}'.`,
   132→      }
   133→    }
   134→
   135→    // ── Fetch the order with items + variant weights + customer ──
   136→    mark('orderLoadStart')
   137→    const order = await db.order.findFirst({
   138→      where: { id: orderId, companyId: ctx.company.id },
   139→      include: {
   140→        customer: {
   141→          select: {
   142→            id: true, name: true,
   143→            phones: { select: { id: true, phoneRaw: true, isPrimary: true }, orderBy: { isPrimary: 'desc' } },
   144→          },
   145→        },
   146→        items: {
   147→          include: {
   148→            orgVariant: {
   149→              select: { id: true, sku: true, weightKg: true, attributeValues: true, product: { select: { title: true } } },
   150→            },
   151→          },
   152→        },
   153→      },
   154→    })
   155→    mark('orderLoadEnd')
   156→    measure('orderLoadStart', 'orderLoadEnd', '2b_order_load')
   157→    if (!order) {
   158→      return { success: false, error: 'Order not found.' }
   159→    }
   160→
   161→    // ── Apply overrides + stored-value defaults ──
   162→    const customerName = options.customerName?.trim() || order.customer?.name || 'Customer'
   163→    const customerPhone =
   164→      options.customerPhone?.trim() ||
   165→      order.customer?.phones.find((p) => p.isPrimary)?.phoneRaw ||
   166→      order.customer?.phones[0]?.phoneRaw || ''
   167→    const deliveryAddress = options.deliveryAddress?.trim() || order.deliveryAddress || ''
   168→    const deliveryCity = options.deliveryCity?.trim() || order.deliveryCity || ''
   169→    const codAmount = options.codAmount ?? Number(order.remainingCodAmount ?? order.totalOrderValue ?? 0)
   170→    // Universal courier reference fields (migration 015): per-call override >
   171→    // stored value > flowops order number
   172→    const orderRefNumber =
   173→      options.orderRefNumber?.trim() ||
   174→      (order.orderRefNumber && order.orderRefNumber.trim()) ||
   175→      order.flowopsOrderNumber
   176→    const itemDescription =
   177→      options.itemDescription?.trim() ||
   178→      (order.orderDetail && order.orderDetail.trim()) ||
   179→      order.items.map((i) => `${i.orgVariant.product.title} (${i.orgVariant.sku}) ×${i.quantity}`).join(', ')
   180→    const transactionNotes =
   181→      options.transactionNotes?.trim() || (order.notesForCourier && order.notesForCourier.trim()) || ''
   182→
   183→    // ── Leopard preferences: optionally append product details to transaction note ──
   184→    // The CompanyIntegration.preferencesJson column stores per-integration
   185→    // preferences (currently only Leopard transactionNote prefs). If this is a
   186→    // Leopard integration AND the user has enabled the auto-append feature,
   187→    // we build a combined special-instructions string from the order's line
   188→    // items and the existing user note.
   189→    let effectiveTransactionNotes = transactionNotes
   190→    if (providerKey === 'leopard' && integration.preferencesJson) {
   191→      const prefs = parseLeopardPreferences(integration.preferencesJson)
   192→      if (prefs.transactionNote.enabled) {
   193→        effectiveTransactionNotes = buildLeopardSpecialInstructions(
   194→          transactionNotes,
   195→          order.items.map((i) => ({
   196→            productTitle: i.orgVariant.product.title,
   197→            sku: i.orgVariant.sku,
   198→            attributeValues: i.orgVariant.attributeValues ?? null,
   199→            quantity: i.quantity,
   200→          })),
   201→          prefs.transactionNote,
   202→        )
   203→      }
   204→    }
   205→
   206→    if (!deliveryCity) {
   207→      const reason = 'Delivery city is required.'
   208→      await db.order.update({
   209→        where: { id: orderId },
   210→        data: { courierBookingStatus: 'failed', courierBookingFailureReason: reason },
   211→      })
   212→      return { success: false, error: reason }
   213→    }
   214→    if (!customerPhone) {
   215→      const reason = 'Customer phone is required.'
   216→      await db.order.update({
   217→        where: { id: orderId },
   218→        data: { courierBookingStatus: 'failed', courierBookingFailureReason: reason },
   219→      })
   220→      return { success: false, error: reason }
   221→    }
   222→
   223→    // ── Validate city (with live courier fallback + staleness check) ──
   224→    // revalidateCityAtBookingTime() is provider-agnostic — it uses the
   225→    // adapter's fetchOperationalCities() for the live fallback.
   226→    mark('cityValidateStart')
   227→    const cityValid = await revalidateCityAtBookingTime(
   228→      providerKey,
   229→      deliveryCity,
   230→      integration.id,
   231→      // Country guard (Phase 3): skip Pakistan-courier-city validation for
   232→      // non-Pakistan orders. International cities aren't in the Pakistan-
   233→      // sourced courier_operational_cities cache; blocking on that would be
   234→      // wrong. For non-Pakistan, the function short-circuits to `true`.
   235→      order.deliveryCountry ?? undefined,
   236→    )
   237→    mark('cityValidateEnd')
   238→    measure('cityValidateStart', 'cityValidateEnd', '4_city_validation')
   239→    if (!cityValid) {
   240→      const reason = `City not recognized: "${deliveryCity}" is not available for delivery with ${integration.provider.providerName}. The city may need to be resolved or the courier may not serve this area.`
   241→      await db.order.update({
   242→        where: { id: orderId },
   243→        data: {
   244→          courierCityStatus: 'unresolved',
   245→          courierBookingStatus: 'failed',
   246→          courierBookingFailureReason: reason,
   247→          courierCompanyIntegrationId: integration.id,
   248→          courierName: integration.provider.providerName,
   249→        },
   250→      })
   251→      return { success: false, error: reason }
   252→    }
   253→
   254→    // ── Compute weight ──
   255→    mark('weightStart')
   256→    const weightResult = calculateOrderWeightKg(
   257→      order.items.map((i) => ({
   258→        quantity: i.quantity,
   259→        variant: { weightKg: i.orgVariant.weightKg as { toNumber: () => number } | number | null },
   260→      })),
   261→    )
   262→    mark('weightEnd')
   263→    measure('weightStart', 'weightEnd', '3_weight_calc')
   264→
   265→    // ── Compute orderType (PostEx-specific; Leopard doesn't use this concept) ──
   266→    // Leopard's shipment_type field is a DIFFERENT thing (optional, defaults to "overnight")
   267→    // — do NOT apply PostEx's Normal/Overland/Replacement logic to Leopard.
   268→    mark('orderTypeStart')
   269→    let orderType: string | undefined
   270→    if (providerKey === 'postex') {
   271→      orderType = options.orderType || determinePostExOrderType(
   272→        weightResult.totalWeightKg,
   273→        weightResult.hasMissingWeight,
   274→        false, // isExchangeReplacement=false for regular orders
   275→      )
   276→    } else {
   277→      orderType = options.orderType // pass through for other couriers (may be undefined)
   278→    }
   279→    mark('orderTypeEnd')
   280→    measure('orderTypeStart', 'orderTypeEnd', '5_orderType_calc')
   281→
   282→    // ── Get pickup address code ──
   283→    // Priority: per-call override > order's persisted pickupAddressId >
   284→    // integration's default.
   285→    mark('pickupAddrStart')
   286→    let pickupAddressCode = options.pickupAddressCode
   287→    if (!pickupAddressCode && order.pickupAddressId) {
   288→      const orderAddr = await db.courierPickupAddress.findFirst({
   289→        where: { id: order.pickupAddressId, companyIntegrationId: integration.id },
   290→        select: { providerAddressCode: true },
   291→      })
   292→      pickupAddressCode = orderAddr?.providerAddressCode
   293→    }
   294→    if (!pickupAddressCode) {
   295→      const defaultAddr = await db.courierPickupAddress.findFirst({
   296→        where: { companyIntegrationId: integration.id, isDefault: true },
   297→        select: { providerAddressCode: true },
   298→      })
   299→      pickupAddressCode = defaultAddr?.providerAddressCode
   300→    }
   301→    mark('pickupAddrEnd')
   302→    measure('pickupAddrStart', 'pickupAddrEnd', '6_pickup_addr_resolution')
   303→
   304→    // ── For Leopard: resolve delivery city NAME to numeric cityId ──
   305→    // Leopard requires numeric city IDs (integers), NOT city name strings.
   306→    // PostEx accepts city name strings directly.
   307→    let resolvedDeliveryCity = deliveryCity
   308→    if (providerKey === 'leopard') {
   309→      const cityRecord = await db.courierOperationalCity.findFirst({
   310→        where: {
   311→          providerKey: 'leopard',
   312→          cityName: { equals: deliveryCity, mode: 'insensitive' },
   313→          isDeliveryCity: true,
   314→        },
   315→        select: { cityId: true },
   316→      })
   317→      if (!cityRecord?.cityId) {
   318→        const reason = `Could not resolve city "${deliveryCity}" to a Leopard numeric city ID. Sync Leopard cities first.`
   319→        await db.order.update({
   320→          where: { id: orderId },
   321→          data: {
   322→            courierCityStatus: 'unresolved',
   323→            courierBookingStatus: 'failed',
   324→            courierBookingFailureReason: reason,
   325→            courierCompanyIntegrationId: integration.id,
   326→            courierName: integration.provider.providerName,
   327→          },
   328→        })
   329→        return { success: false, error: reason }
   330→      }
   331→      resolvedDeliveryCity = cityRecord.cityId // numeric ID as string
   332→    }
   333→
   334→    // ── Build the BookShipmentInput ──
   335→    const bookInput: BookShipmentInput = {
   336→      orderNumber: orderRefNumber,
   337→      recipientName: customerName,
   338→      recipientPhone: customerPhone,
   339→      deliveryAddress,
   340→      deliveryCity: resolvedDeliveryCity, // numeric ID for Leopard, name for PostEx
   341→      pickupLocationAddress: '',
   342→      pickupLocationCity: '',
   343→      weightGrams: Math.round(weightResult.totalWeightKg * 1000), // KG → grams
   344→      codAmount,
   345→      itemDescription,
   346→      pickupAddressCode,
   347→      orderType,
   348→      quantity: order.items.reduce((sum, i) => sum + i.quantity, 0),
   349→      transactionNotes: effectiveTransactionNotes,
   350→    }
   351→
   352→    // ── Call the courier adapter ──
   353→    // executeLoggedIntegrationAction wraps the adapter call AND does an
   354→    // awaited DB write (integration_action_logs insert) in its finally
   355→    // block. To isolate the EXTERNAL API call time from the log write, we
   356→    // measure the adapter call separately from the wrapper.
   357→    const credentials = decryptCredentials(integration.credentialsEncrypted!)
   358→    const adapter = getCourierAdapter(providerKey, credentials)
   359→
   360→    // Measure the adapter call (external API) in isolation
   361→    mark('adapterCallStart')
   362→    let adapterResult: BookShipmentResult
   363→    try {
   364→      adapterResult = await adapter.bookShipment(bookInput)
   365→    } catch (err) {
   366→      mark('adapterCallEnd')
   367→      measure('adapterCallStart', 'adapterCallEnd', '7_external_api_call')
   368→      T['7_external_api_call'] = marks['adapterCallEnd'] - marks['adapterCallStart']
   369→      throw err
   370→    }
   371→    mark('adapterCallEnd')
   372→    measure('adapterCallStart', 'adapterCallEnd', '7_external_api_call')
   373→
   374→    // Now wrap in the logged action — but the fn is a no-op since we
   375→    // already called the adapter. We pass the captured result so the
   376→    // logged-call wrapper only does its DB write (timing it separately).
   377→    mark('loggedActionWriteStart')
   378→    const bookResult = await executeLoggedIntegrationAction<BookShipmentResult>({
   379→      companyIntegrationId: integration.id,
   380→      organizationId: ctx.company.organizationId,
   381→      actionType: 'book_shipment',
   382→      direction: 'outbound',
   383→      relatedEntityType: 'order',
   384→      relatedEntityId: orderId,
   385→      fn: async () => adapterResult,
   386→      // Log the actual request payload sent to the courier (for audit/debugging
   387→      // and future backfill scenarios). bookInput contains the delivery city,
   388→      // address, COD amount, weight, etc. — no credentials (those are in the
   389→      // adapter instance, not in bookInput).
   390→      requestPayload: bookInput,
   391→    })
   392→    mark('loggedActionWriteEnd')
   393→    measure('loggedActionWriteStart', 'loggedActionWriteEnd', '8_action_log_write')
   394→
   395→    if (!bookResult.success || !bookResult.trackingNumber) {
   396→      const reason = bookResult.error || 'Booking failed — no tracking number returned.'
   397→      await db.order.update({
   398→        where: { id: orderId },
   399→        data: {
   400→          courierBookingStatus: 'failed',
   401→          courierBookingFailureReason: reason,
   402→          courierCompanyIntegrationId: integration.id,
   403→          courierName: integration.provider.providerName,
   404→        },
   405→      })
   406→      return { success: false, error: reason }
   407→    }
   408→
   409→    // ── Map the providerStatus to canonical courierSubStatus ──
   410→    // Provider-agnostic: PostEx uses mapPostExStatus, Leopard uses a simpler
   411→    // mapping (just 'Booked' → null for now, since Leopard doesn't return a
   412→    // meaningful initial sub-status; Prompt 7 will add the full status map).
   413→    let mappedBookingStatus: { courierSubStatus: string | null } | null = null
   414→    if (bookResult.providerStatus && providerKey === 'postex') {
   415→      const { mapPostExStatus } = await import('@/lib/integrations/couriers/postex.status-map')
   416→      mappedBookingStatus = mapPostExStatus(bookResult.providerStatus)
   417→    } else if (providerKey === 'leopard') {
   418→      // Leopard returns 'Booked' as the initial status — map to 'slip_generated'
   419→      // (the canonical sub-status for "booked but not yet pickup_requested")
   420→      mappedBookingStatus = { courierSubStatus: 'slip_generated' }
   421→    }
   422→
   423→    // ── Download the slip_link if provided (Leopard-specific) ──
   424→    // Store our own copy — don't trust external courier URLs long-term.
   425→    let courierSlipStoragePath: string | null = null
   426→    if (bookResult.slipLink) {
   427→      try {
   428→        const fs = await import('fs/promises')
   429→        const path = await import('path')
   430→        const dir = path.join(process.cwd(), 'public', 'uploads', 'courier-slips', ctx.company.id)
   431→        await fs.mkdir(dir, { recursive: true })
   432→        const filename = `slip-${order.flowopsOrderNumber}-${Date.now()}.pdf`
   433→        const filepath = path.join(dir, filename)
   434→        const slipResponse = await fetch(bookResult.slipLink)
   435→        if (slipResponse.ok) {
   436→          const arrayBuffer = await slipResponse.arrayBuffer()
   437→          await fs.writeFile(filepath, Buffer.from(arrayBuffer))
   438→          courierSlipStoragePath = `/uploads/courier-slips/${ctx.company.id}/${filename}`
   439→        }
   440→      } catch (slipErr) {
   441→        console.error('[booking] Failed to download courier slip:', slipErr)
   442→        // Non-fatal — booking still succeeded
   443→      }
   444→    }
   445→
   446→    mark('orderUpdateStart')
   447→    await db.order.update({
   448→      where: { id: orderId },
   449→      data: {
   450→        courierCompanyIntegrationId: integration.id,
   451→        trackingNumber: bookResult.trackingNumber,
   452→        courierCityStatus: 'matched',
   453→        courierSubStatus: mappedBookingStatus?.courierSubStatus ?? null,
   454→        courierName: integration.provider.providerName,
   455→        courierBookingStatus: 'booked',
   456→        courierBookingFailureReason: null,
   457→        // Store our own copy of the courier slip PDF (if downloaded)
   458→        ...(courierSlipStoragePath ? { courierSlipStoragePath } : {}),
   459→        // Persist the (possibly corrected) delivery city + address on the order
   460→        deliveryCity: resolvedDeliveryCity || deliveryCity,
   461→        ...(deliveryAddress ? { deliveryAddress } : {}),
   462→      },
   463→    })
   464→    mark('orderUpdateEnd')
   465→    measure('orderUpdateStart', 'orderUpdateEnd', '9_order_update')
   466→
   467→    // ── Propagate city correction back to the customer's saved address ──
   468→    // If the city was corrected during booking (e.g., via the mismatch
   469→    // resolver or manual override), and the order used a SAVED customer
   470→    // address (usedCustomerAddressId is non-null), update that
   471→    // CustomerAddress row so the corrected city shows up on future orders
   472→    // for this customer — the same correction won't need to be repeated.
   473→    // If the order used a one-off address (usedCustomerAddressId is null),
   474→    // skip propagation — we only fix the order's own address (above).
   475→    if (order.usedCustomerAddressId && resolvedDeliveryCity) {
   476→      db.customerAddress.update({
   477→        where: { id: order.usedCustomerAddressId },
   478→        data: { city: resolvedDeliveryCity },
   479→      }).catch(() => {
   480→        // Non-fatal: if this fails, the order is still booked successfully.
   481→      })
   482→    }
   483→
   484→    // Audit log (fire-and-forget — see src/lib/audit.ts)
   485→    mark('auditMetricStart')
   486→    insertAuditLog({
   487→      action: 'order.auto_booked',
   488→      entityType: 'order',
   489→      entityId: orderId,
   490→      companyId: ctx.company.id,
   491→      organizationId: ctx.company.organizationId,
   492→      userId: ctx.user.id,
   493→      employeeId: ctx.employee.id,
   494→      newValues: {
   495→        trackingNumber: bookResult.trackingNumber,
   496→        courierIntegrationId: integration.id,
   497→        orderType,
   498→        providerStatus: bookResult.providerStatus,
   499→        slipLink: bookResult.slipLink ?? null,
   500→        courierSlipStoragePath,
   501→      },
   502→    })
   503→    mark('auditMetricEnd')
   504→    measure('auditMetricStart', 'auditMetricEnd', '10_audit_metric')
   505→
   506→    // ════════════════════════════════════════════════════════════════
   507→    // LOG THE TIMING BREAKDOWN
   508→    // ════════════════════════════════════════════════════════════════
   509→    const totalMs = Date.now() - flowStart
   510→    const codebaseMs = totalMs - (T['7_external_api_call'] ?? 0)
   511→    console.log(JSON.stringify({
   512→      __BOOKING_TIMING__: true,
   513→      orderId,
   514→      orderNumber: order.flowopsOrderNumber,
   515→      provider: providerKey,
   516→      steps_ms: T,
   517→      external_api_ms: T['7_external_api_call'] ?? 0,
   518→      codebase_ms: codebaseMs,
   519→      total_ms: totalMs,
   520→    }))
   521→
   522→    return {
   523→      success: true,
   524→      data: {
   525→        trackingNumber: bookResult.trackingNumber,
   526→        orderType,
   527→        providerStatus: bookResult.providerStatus ?? null,
   528→        courierBookingStatus: 'booked',
   529→      },
   530→    }
   531→  } catch (err) {
   532→    // Any uncaught error (credential decryption, network, etc.) — persist
   533→    // the failure status so the order shows up in the Workbench for retry.
   534→    const reason = err instanceof Error ? err.message : 'Failed to book order'
   535→    console.error(JSON.stringify({
   536→      __BOOKING_TIMING_ERROR__: true,
   537→      orderId,
   538→      steps_ms: T,
   539→      total_ms: Date.now() - flowStart,
   540→      error: reason,
   541→    }))
   542→    try {
   543→      await db.order.update({
   544→        where: { id: orderId },
   545→        data: { courierBookingStatus: 'failed', courierBookingFailureReason: reason },
   546→      })
   547→    } catch {
   548→      // Best-effort — don't mask the original error
   549→    }
   550→    return { success: false, error: reason }
   551→  }
   552→}

   554→/**
   555→ * Auto-book an order if the company's courierBookingMode is 'automatic'.
   556→ *
   557→ * Called from createManualOrder() AFTER the order is created + stock is
   558→ * reserved. Reads the company's order settings to decide whether to book.
   559→ *
   560→ * This function is NON-BLOCKING for order creation — if booking fails, the
   561→ * order is still created successfully (with courierBookingStatus='failed'),
   562→ * and the user can retry from the Booking Workbench. The error is logged
   563→ * and returned in the result so the caller can surface a toast.
   564→ *
   565→ * Skips auto-booking when:
   566→ *   - courierBookingMode !== 'automatic'
   567→ *   - No defaultCourierCompanyIntegrationId is set
   568→ *   - The order source is not 'manual' (Shopify/Daraz always manual per UI)
   569→ *   - The order is not yet confirmed (pending orders shouldn't auto-book)
   570→ */
   571→export async function maybeAutoBookOrder(
   572→  orderId: string,
   573→  orderSource: string,
   574→  orderStatus: string,
   575→): Promise<ActionResult<BookOrderResult>> {
   576→  try {
   577→    // Only auto-book manual orders that are confirmed (or processing)
   578→    if (orderSource !== 'manual') {
   579→      return { success: false, error: 'Auto-booking skipped: order source is not manual.' }
   580→    }
   581→    if (orderStatus !== 'confirmed' && orderStatus !== 'processing') {
   582→      return { success: false, error: `Auto-booking skipped: order status is '${orderStatus}' (not confirmed).` }
   583→    }
   584→
   585→    const ctx = await getWorkspace()
   586→
   587→    // Read the company's order settings
   588→    const settings = await db.companyOrderSetting.findUnique({
   589→      where: { companyId: ctx.company.id },
   590→    })
   591→    if (!settings) {
   592→      return { success: false, error: 'No order settings found for this company.' }
   593→    }
   594→
   595→    if (settings.courierBookingMode !== 'automatic') {
   596→      return { success: false, error: `Auto-booking skipped: courierBookingMode is '${settings.courierBookingMode}'.` }
   597→    }
   598→
   599→    // ── Determine which integration to book with ──
   600→    // Priority: the courier the user selected on the order form
   601→    // (Order.courierCompanyIntegrationId) → falls back to the company's
   602→    // default courier (CompanyOrderSetting.defaultCourierCompanyIntegrationId).
   603→    // This fixes the bug where auto-booking didn't fire when the user
   604→    // selected a courier on the form but the company default was null
   605→    // (e.g. after a disconnect/reconnect cycle).
   606→    const order = await db.order.findUnique({
   607→      where: { id: orderId },
   608→      select: { courierCompanyIntegrationId: true, flowopsOrderNumber: true },
   609→    })
   610→    if (!order) {
   611→      return { success: false, error: 'Order not found.' }
   612→    }
   613→
   614→    const integrationId = order.courierCompanyIntegrationId || settings.defaultCourierCompanyIntegrationId
   615→    if (!integrationId) {
   616→      return { success: false, error: 'Auto-booking skipped: no courier selected on the order and no default courier set in Order Settings.' }
   617→    }
   618→
   619→    // ── Early bail: check if the integration is still active ──
   620→    const defaultIntegration = await db.companyIntegration.findFirst({
   621→      where: { id: integrationId, companyId: ctx.company.id },
   622→      select: { id: true, isActive: true, connectionName: true, provider: { select: { providerName: true } } },
   623→    })
   624→    if (!defaultIntegration) {
   625→      return {
   626→        success: false,
   627→        error: 'Courier integration was deleted. Update Order Settings or select a different courier.',
   628→      }
   629→    }
   630→    if (!defaultIntegration.isActive) {
   631→      return {
   632→        success: false,
   633→        error: `Courier integration "${defaultIntegration.provider.providerName} — ${defaultIntegration.connectionName}" is disconnected. Reconnect it in Integrations settings or choose a different courier.`,
   634→      }
   635→    }
   636→
   637→    // Fire the booking
   638→    const result = await bookOrderWithCourier({
   639→      orderId,
   640→      companyIntegrationId: integrationId,
   641→    })
   642→
   643→    if (result.success) {
   644→      // Metric event for successful auto-booking
   645→      insertMetricEvent({
   646→        companyId: ctx.company.id,
   647→        entityType: 'order',
   648→        entityId: orderId,
   649→        metricKey: 'order.auto_booked',
   650→        numericValue: 1,
   651→        dimensions: { provider: 'postex', mode: 'automatic' },
   652→      })
   653→    } else {
   654→      // Metric event for failed auto-booking
   655→      insertMetricEvent({
   656→        companyId: ctx.company.id,
   657→        entityType: 'order',
   658→        entityId: orderId,
   659→        metricKey: 'order.auto_booking_failed',
   660→        numericValue: 1,
   661→        dimensions: { reason: result.error?.slice(0, 100) ?? 'unknown' },
   662→      })
   663→    }
   664→
   665→    return result
   666→  } catch (err) {
   667→    return {
   668→      success: false,
   669→      error: err instanceof Error ? err.message : 'Auto-booking failed',
   670→    }
   671→  }
   672→}

   674→/**
   675→ * Book an exchange shipment with courier
   676→ */
   677→export async function bookExchangeShipmentWithCourier(
   678→  options: BookOrderOptions & { shipmentId: string }
   679→): Promise<ActionResult<BookOrderResult>> {
   680→  try {
   681→    const ctx = await getWorkspace()
   682→    const shipment = await db.exchangeShipment.findFirst({
   683→      where: {
   684→        id: options.shipmentId,
   685→        orderExchange: { companyId: ctx.company.id },
   686→      },
   687→      include: {
   688→        orderExchange: {
   689→          include: {
   690→            order: {
   691→              include: {
   692→                customer: true,
   693→                shippingAddress: true,
   694→              },
   695→            },
   696→          },
   697→        },
   698→        newOrgVariant: {
   699→          include: {
   700→            product: true,
   701→          },
   702→        },
   703→        shippingAddress: true,
   704→      },
   705→    })
   706→
   707→    if (!shipment) {
   708→      return { success: false, error: 'Exchange shipment not found.' }
   709→    }
   710→
   711→    const integration = await db.companyIntegration.findFirst({
   712→      where: {
   713→        id: options.companyIntegrationId,
   714→        companyId: ctx.company.id,
   715→        isActive: true,
   716→      },
   717→      include: { provider: true },
   718→    })
   719→
   720→    if (!integration) {
   721→      return { success: false, error: 'Courier integration not found or inactive.' }
   722→    }
   723→
   724→    const providerKey = integration.provider.providerKey
   725→
   726→    const customerName =
   727→      options.customerName?.trim() ||
   728→      shipment.shippingAddress?.name ||
   729→      shipment.orderExchange.order.customer?.name ||
   730→      'Customer'
   731→    const customerPhone =
   732→      options.customerPhone?.trim() ||
   733→      shipment.shippingAddress?.phone ||
   734→      shipment.orderExchange.order.customer?.phone ||
   735→      ''
   736→    const deliveryAddress =
   737→      options.deliveryAddress?.trim() ||
   738→      shipment.shippingAddress?.address1 ||
   739→      shipment.orderExchange.order.shippingAddress?.address1 ||
   740→      ''
   741→    const deliveryCity =
   742→      options.deliveryCity?.trim() ||
   743→      shipment.shippingAddress?.city ||
   744→      shipment.orderExchange.order.shippingAddress?.city ||
   745→      ''
   746→    const codAmount = options.codAmount !== undefined ? options.codAmount : Number(shipment.codAmount || 0)
   747→    const orderRefNumber =
   748→      options.orderRefNumber?.trim() ||
   749→      shipment.exchangeShipmentNumber
   750→    const itemDescription =
   751→      options.itemDescription?.trim() ||
   752→      (shipment.orderDetail && shipment.orderDetail.trim()) ||
   753→      `${shipment.newOrgVariant.product.title} (${shipment.newOrgVariant.sku}) ×${shipment.quantity}`
   754→    const transactionNotes = options.transactionNotes?.trim() || ''
   755→    const isExchangeReplacement = shipment.orderExchange.exchangeMethod === 'courier_replacement'
   756→
   757→    if (!deliveryCity) {
   758→      return { success: false, error: 'Delivery city is required.' }
   759→    }
   760→    if (!customerPhone) {
   761→      return { success: false, error: 'Customer phone is required.' }
   762→    }
   763→
   764→    const cityValid = await revalidateCityAtBookingTime(providerKey, deliveryCity, integration.id, shipment.shippingAddress?.country ?? undefined)
   765→    if (!cityValid) {
   766→      await db.exchangeShipment.update({
   767→        where: { id: shipment.id },
   768→        data: { courierCityStatus: 'unresolved' },
   769→      })
   770→      return {
   771→        success: false,
   772→        error: `City "${deliveryCity}" is not available for delivery with ${integration.provider.providerName}.`,
   773→      }
   774→    }
   775→
   776→    const weightResult = calculateOrderWeightKg([
   777→      {
   778→        quantity: shipment.quantity,
   779→        variant: { weightKg: shipment.newOrgVariant.weightKg ? Number(shipment.newOrgVariant.weightKg) : null },
   780→      },
   781→    ])
   782→
   783→    const orderType = options.orderType || determinePostExOrderType(
   784→      weightResult.totalWeightKg,
   785→      weightResult.hasMissingWeight,
   786→      isExchangeReplacement,
   787→    )
   788→
   789→    let pickupAddressCode = options.pickupAddressCode
   790→    if (!pickupAddressCode) {
   791→      const defaultAddr = await db.courierPickupAddress.findFirst({
   792→        where: { companyIntegrationId: integration.id, isDefault: true },
   793→        select: { providerAddressCode: true },
   794→      })
   795→      pickupAddressCode = defaultAddr?.providerAddressCode
   796→    }
   797→
   798→    const bookInput: BookShipmentInput = {
   799→      orderNumber: orderRefNumber,
   800→      recipientName: customerName,
   801→      recipientPhone: customerPhone,
   802→      deliveryAddress,
   803→      deliveryCity,
   804→      pickupLocationAddress: '',
   805→      pickupLocationCity: '',
   806→      weightGrams: Math.round(weightResult.totalWeightKg * 1000),
   807→      codAmount,
   808→      itemDescription,
   809→      pickupAddressCode,
   810→      orderType,
   811→      quantity: shipment.quantity,
   812→      transactionNotes,
   813→    }
   814→
   815→    const credentials = decryptCredentials(integration.credentialsEncrypted!)
   816→    const adapter = getCourierAdapter(providerKey, credentials)
   817→
   818→    const bookResult = await executeLoggedIntegrationAction<BookShipmentResult>({
   819→      companyIntegrationId: integration.id,
   820→      organizationId: ctx.company.organizationId,
   821→      actionType: 'book_shipment',
   822→      direction: 'outbound',
   823→      relatedEntityType: 'exchange_shipment',
   824→      relatedEntityId: shipment.id,
   825→      fn: async () => adapter.bookShipment(bookInput),
   826→    })
   827→
   828→    if (!bookResult.success || !bookResult.trackingNumber) {
   829→      return {
   830→        success: false,
   831→        error: bookResult.error || 'Booking failed — no tracking number returned.',
   832→      }
   833→    }
   834→
   835→    const { mapPostExStatus } = await import('@/lib/integrations/couriers/postex.status-map')
   836→    const mappedBookingStatus = bookResult.providerStatus
   837→      ? mapPostExStatus(bookResult.providerStatus)
   838→      : null
   839→
   840→    await db.exchangeShipment.update({
   841→      where: { id: shipment.id },
   842→      data: {
   843→        courierCompanyIntegrationId: integration.id,
   844→        trackingNumber: bookResult.trackingNumber,
   845→        courierCityStatus: 'matched',
   846→        courierSubStatus: mappedBookingStatus?.courierSubStatus ?? null,
   847→        courierBookingStatus: 'booked',
   848→      },
   849→    })
   850→
   851→    return {
   852→      success: true,
   853→      data: {
   854→        trackingNumber: bookResult.trackingNumber,
   855→        orderType,
   856→        providerStatus: bookResult.providerStatus ?? null,
   857→        courierBookingStatus: 'booked',
   858→      },
   859→    }
   860→  } catch (err) {
   861→    return {
   862→      success: false,
   863→      error: err instanceof Error ? err.message : 'Exchange shipment booking failed',
   864→    }
   865→  }
   866→}

   891→/**
   892→ * Batch booking action: processes multiple orders/exchange shipments sequentially
   893→ * and returns per-item success/failure statuses.
   894→ */
   895→export async function bookOrdersBatch(
   896→  companyIntegrationId: string,
   897→  items: BatchBookingItem[],
   898→): Promise<ActionResult<{ results: BatchBookingResultItem[] }>> {
   899→  const results: BatchBookingResultItem[] = []
   900→
   901→  for (const item of items) {
   902→    if (item.entityType === 'order') {
   903→      const res = await bookOrderWithCourier({
   904→        orderId: item.entityId,
   905→        companyIntegrationId,
   906→        customerName: item.customerName,
   907→        customerPhone: item.customerPhone,
   908→        deliveryAddress: item.deliveryAddress,
   909→        deliveryCity: item.deliveryCity,
   910→        codAmount: item.codAmount,
   911→        orderType: item.orderType,
   912→        transactionNotes: item.transactionNotes,
   913→        itemDescription: item.itemDescription,
   914→        orderRefNumber: item.orderRefNumber,
   915→        pickupAddressCode: item.pickupAddressCode,
   916→      })
   917→      if (res.success && res.data) {
   918→        results.push({
   919→          entityType: 'order',
   920→          entityId: item.entityId,
   921→          success: true,
   922→          trackingNumber: res.data.trackingNumber,
   923→        })
   924→      } else {
   925→        results.push({
   926→          entityType: 'order',
   927→          entityId: item.entityId,
   928→          success: false,
   929→          error: res.error ?? 'Booking failed',
   930→        })
   931→      }
   932→    } else {
   933→      const res = await bookExchangeShipmentWithCourier({
   934→        shipmentId: item.entityId,
   935→        companyIntegrationId,
   936→        customerName: item.customerName,
   937→        customerPhone: item.customerPhone,
   938→        deliveryAddress: item.deliveryAddress,
   939→        deliveryCity: item.deliveryCity,
   940→        codAmount: item.codAmount,
   941→        orderType: item.orderType,
   942→        transactionNotes: item.transactionNotes,
   943→        itemDescription: item.itemDescription,
   944→        orderRefNumber: item.orderRefNumber,
   945→        pickupAddressCode: item.pickupAddressCode,
   946→      })
   947→      if (res.success && res.data) {
   948→        results.push({
   949→          entityType: 'exchange_shipment',
   950→          entityId: item.entityId,
   951→          success: true,
   952→          trackingNumber: res.data.trackingNumber,
   953→        })
   954→      } else {
   955→        results.push({
   956→          entityType: 'exchange_shipment',
   957→          entityId: item.entityId,
   958→          success: false,
   959→          error: res.error ?? 'Booking failed',
   960→        })
   961→      }
   962→    }
   963→  }
   964→
   965→  return {
   966→    success: true,
   967→    data: { results },
   968→  }
   969→}
