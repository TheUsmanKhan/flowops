    91→// ──────────────────────────────────────────────────────────────
    92→// Processes ALL order_items for an order and attempts to reserve
    93→// stock for each based on its fulfillment_type_snapshot.
    94→//
    95→// For stock_based items:
    96→//   - If sufficient available stock → reserveStockForOrder(), fulfillment_status='reserved'
    97→//   - If insufficient + inventory_policy='continue' → fulfillment_status='backordered'
    98→//   - If insufficient + inventory_policy='deny' → outcome='failed'
    99→//
   100→// For made_to_order items:
   101→//   - checkAndFulfillMadeToOrderVariant() checks returned stock first
   102→//   - If existing_stock → fulfillment_status='reserved', returned_stitched_used=true
   103→//   - If fresh_production → fulfillment_status='reserved', production_order_id set
   104→//   - If error → outcome='failed'
   105→//
   106→// After processing: if ANY item is 'backordered', order status → 'partially_backordered'.
   107→// If ALL items are 'reserved', order stays 'confirmed'.
   108→//
   109→// Called from:
   110→//   - createManualOrder() IF the order auto-confirmed at creation
   111→//   - confirmOrder() when a 'pending' order is manually confirmed
   112→//   - dispatchOrderAction() if status is still 'pending' at dispatch time
   113→
   114→async function reserveOrderStock(
   115→  orderId: string,
   116→  ctx: { company: { id: string; organizationId: string }; user: { id: string }; employee: { id: string } },
   117→): Promise<{
   118→  success: boolean
   119→  results: Array<{ orderItemId: string; outcome: 'reserved' | 'backordered' | 'failed'; reason?: string }>
   120→}> {
   121→  const order = await db.order.findUnique({
   122→    where: { id: orderId },
   123→    select: { dispatchLocationId: true, companyId: true, organizationId: true },
   124→  })
   125→  if (!order) return { success: false, results: [] }
   126→
   127→  const items = await db.orderItem.findMany({
   128→    where: { orderId },
   129→    include: {
   130→      orgVariant: {
   131→        select: { id: true, fulfillmentType: true, inventoryPolicy: true, stitchingCharges: true },
   132→      },
   133→    },
   134→  })
   135→
   136→  const results: Array<{ orderItemId: string; outcome: 'reserved' | 'backordered' | 'failed'; reason?: string }> = []
   137→  let hasBackordered = false
   138→
   139→  for (const item of items) {
   140→    if (item.fulfillmentStatus === 'reserved' || item.fulfillmentStatus === 'dispatched') {
   141→      // Already processed (e.g. re-confirmation attempt) — skip
   142→      results.push({ orderItemId: item.id, outcome: 'reserved' })
   143→      continue
   144→    }
   145→
   146→    const locationId = item.reservedLocationId ?? order.dispatchLocationId
   147→    if (!locationId) {
   148→      results.push({ orderItemId: item.id, outcome: 'failed', reason: 'No dispatch location set' })
   149→      continue
   150→    }
   151→
   152→    if (item.fulfillmentTypeSnapshot === 'stock_based') {
   153→      // Check available stock at the dispatch location
   154→      const pool = await db.inventoryPool.findUnique({
   155→        where: {
   156→          orgVariantId_locationId: {
   157→            orgVariantId: item.orgVariantId,
   158→            locationId,
   159→          },
   160→        },
   161→      })
   162→
   163→      const available = pool ? pool.onHand - pool.reserved : 0
   164→
   165→      if (available >= item.quantity) {
   166→        // Sufficient stock — reserve it
   167→        const reserveResult = await reserveStockForOrder({
   168→          orgVariantId: item.orgVariantId,
   169→          locationId,
   170→          organizationId: order.organizationId,
   171→          companyId: order.companyId,
   172→          employeeId: ctx.employee.id,
   173→          quantity: item.quantity,
   174→          orderId,
   175→        })
   176→
   177→        if (reserveResult.success) {
   178→          await db.orderItem.update({
   179→            where: { id: item.id },
   180→            data: { fulfillmentStatus: 'reserved', reservedLocationId: locationId },
   181→          })
   182→          results.push({ orderItemId: item.id, outcome: 'reserved' })
   183→        } else {
   184→          results.push({ orderItemId: item.id, outcome: 'failed', reason: reserveResult.error })
   185→        }
   186→      } else if (item.orgVariant.inventoryPolicy === 'continue') {
   187→        // Insufficient stock but backordering allowed
   188→        await db.orderItem.update({
   189→          where: { id: item.id },
   190→          data: { fulfillmentStatus: 'backordered', backorderedAt: new Date() },
   191→        })
   192→        await insertMetricEvent({
   193→          companyId: order.companyId,
   194→          entityType: 'product',
   195→          entityId: item.orgVariantId,
   196→          metricKey: 'order.backordered',
   197→          numericValue: item.quantity,
   198→          dimensions: { order_id: orderId },
   199→        }).catch(() => {})
   200→        hasBackordered = true
   201→        results.push({
   202→          orderItemId: item.id,
   203→          outcome: 'backordered',
   204→          reason: `Available: ${available}, required: ${item.quantity}`,
   205→        })
   206→      } else {
   207→        // Insufficient stock + policy='deny' — fail
   208→        results.push({
   209→          orderItemId: item.id,
   210→          outcome: 'failed',
   211→          reason: `Insufficient stock (available: ${available}, required: ${item.quantity}) and inventory_policy='deny'`,
   212→        })
   213→      }
   214→    } else if (item.fulfillmentTypeSnapshot === 'made_to_order') {
   215→      // Made-to-order: check returned stock first, then trigger production
   216→      const mtoResult = await checkAndFulfillMadeToOrderVariant(
   217→        item.orgVariantId,
   218→        item.quantity,
   219→        order.companyId,
   220→        locationId,
   221→      )
   222→
   223→      if (mtoResult.source === 'existing_stock' && mtoResult.locationId) {
   224→        // Returned stock available — reserve it
   225→        const reserveResult = await reserveStockForOrder({
   226→          orgVariantId: item.orgVariantId,
   227→          locationId: mtoResult.locationId,
   228→          organizationId: order.organizationId,
   229→          companyId: order.companyId,
   230→          employeeId: ctx.employee.id,
   231→          quantity: item.quantity,
   232→          orderId,
   233→        })
   234→
   235→        if (reserveResult.success) {
   236→          await db.orderItem.update({
   237→            where: { id: item.id },
   238→            data: {
   239→              fulfillmentStatus: 'reserved',
   240→              returnedStitchedUsed: true,
   241→              reservedLocationId: mtoResult.locationId,
   242→            },
   243→          })
   244→          await insertMetricEvent({
   245→            companyId: order.companyId,
   246→            entityType: 'product',
   247→            entityId: item.orgVariantId,
   248→            metricKey: 'order.made_to_order_from_returned_stock',
   249→            numericValue: 1,
   250→            dimensions: { order_id: orderId, stitching_cost_saved: Number(item.orgVariant.stitchingCharges) || 0 },
   251→          }).catch(() => {})
   252→          results.push({ orderItemId: item.id, outcome: 'reserved' })
   253→        } else {
   254→          results.push({ orderItemId: item.id, outcome: 'failed', reason: reserveResult.error })
   255→        }
   256→      } else if (mtoResult.source === 'fresh_production' && mtoResult.productionOrderId) {
   257→        // Fresh production triggered — link the production order
   258→        await db.orderItem.update({
   259→          where: { id: item.id },
   260→          data: {
   261→            fulfillmentStatus: 'reserved',
   262→            productionOrderId: mtoResult.productionOrderId,
   263→          },
   264→        })
   265→        // Also link the order_item back to the production order
   266→        await db.productionOrder.update({
   267→          where: { id: mtoResult.productionOrderId },
   268→          data: { orderItemId: item.id },
   269→        })
   270→        await insertMetricEvent({
   271→          companyId: order.companyId,
   272→          entityType: 'product',
   273→          entityId: item.orgVariantId,
   274→          metricKey: 'order.made_to_order_production_triggered',
   275→          numericValue: 1,
   276→          dimensions: { order_id: orderId, production_order_id: mtoResult.productionOrderId },
   277→        }).catch(() => {})
   278→        results.push({ orderItemId: item.id, outcome: 'reserved' })
   279→      } else {
   280→        // Error (no fabric source, insufficient fabric, etc.)
   281→        results.push({
   282→          orderItemId: item.id,
   283→          outcome: 'failed',
   284→          reason: mtoResult.error ?? 'Made-to-order fulfillment failed',
   285→        })
   286→      }
   287→    }
   288→  }
   289→
   290→  // Update order status based on results
   291→  if (hasBackordered) {
   292→    await db.order.update({
   293→      where: { id: orderId },
   294→      data: { status: 'partially_backordered' },
   295→    })
   296→  }
   297→  // If no backordered items, order stays 'confirmed' (or whatever it was)
   298→
   299→  return { success: true, results }
   300→}

   302→// ──────────────────────────────────────────────────────────────

   304→// ──────────────────────────────────────────────────────────────
   305→
   306→export async function createManualOrder(
   307→  input: CreateManualOrderInput,
   308→): Promise<ActionResult<{
   309→  orderId: string
   310→  flowopsOrderNumber: string
   311→  orderItems: Array<{ id: string; orgVariantId: string; quantity: number }>
   312→  /** Auto-booking is now ASYNC — bookingAttempted=true means it's running in the background. */
   313→  bookingAttempted: boolean
   314→  bookingSucceeded?: boolean
   315→  bookingError?: string
   316→  bookingTrackingNumber?: string
   317→}>> {
   318→  try {
   319→    const ctx = await getWorkspace()
   320→    await requirePermission(ctx, PERMISSIONS.ORDERS_CREATE)
   321→
   322→    // 1. Validate
   323→    const parsed = createManualOrderSchema.safeParse(input)
   324→    if (!parsed.success) {
   325→      return {
   326→        success: false,
   327→        error: parsed.error.issues[0]?.message ?? 'Invalid order data',
   328→      }
   329→    }
   330→    const d = parsed.data
   331→
   332→    // 2. Resolve customer (existing customer_id OR inline new_customer).
   333→    //    Also resolve optional saved address/phone selection and recipient_name.
   334→    let customerId: string
   335→    let customerName: string
   336→    let usedCustomerAddressId: string | null = null
   337→    let usedCustomerPhoneId: string | null = null
   338→    let saveAddressForNextTime = d.save_address_for_next_time ?? false
   339→    // Whether the delivery_address came from a saved customer_addresses row
   340→    // (vs. typed as a one-off). Used to decide whether to bump lastUsedAt.
   341→    let selectedSavedAddressId: string | null = null
   342→
   343→    if (d.customer_id) {
   344→      // Existing customer path — verify they belong to this org.
   345→      const existing = await db.customer.findFirst({
   346→        where: { id: d.customer_id, organizationId: ctx.company.organizationId },
   347→        select: { id: true, name: true },
   348→      })
   349→      if (!existing) return { success: false, error: 'Customer not found' }
   350→      customerId = existing.id
   351→      customerName = existing.name
   352→
   353→      // If a saved address was selected, verify it belongs to this customer.
   354→      if (d.used_customer_address_id) {
   355→        const addr = await db.customerAddress.findFirst({
   356→          where: {
   357→            id: d.used_customer_address_id,
   358→            customerId: existing.id,
   359→            organizationId: ctx.company.organizationId,
   360→          },
   361→          select: { id: true, address: true, city: true },
   362→        })
   363→        if (addr) {
   364→          usedCustomerAddressId = addr.id
   365→          selectedSavedAddressId = addr.id
   366→          // The delivery_address snapshot is the order's own editable copy.
   367→          // If the caller passed the same address text, we trust it; if they
   368→          // passed different text (edited it), we still record the link to
   369→          // the saved address for lastUsedAt tracking but keep the edited text
   370→          // as the order's snapshot. Either way, delivery_address comes from
   371→          // the input — we do NOT override it here.
   372→        }
   373→      }
   374→
   375→      // If a saved phone was selected, verify it belongs to this customer.
   376→      if (d.used_customer_phone_id) {
   377→        const phone = await db.customerPhone.findFirst({
   378→          where: {
   379→            id: d.used_customer_phone_id,
   380→            customerId: existing.id,
   381→            organizationId: ctx.company.organizationId,
   382→          },
   383→          select: { id: true },
   384→        })
   385→        if (phone) {
   386→          usedCustomerPhoneId = phone.id
   387→        }
   388→      }
   389→    } else if (d.new_customer) {
   390→      // Inline new-customer path — create the customer + their phones +
   391→      // addresses now via the Customer Management System. The first phone is
   392→      // primary and the first address is default (validated by createCustomerSchema).
   393→      const createResult = await createCustomer(d.new_customer)
   394→      if (!createResult.success || !createResult.data) {
   395→        return { success: false, error: createResult.error ?? 'Failed to create customer' }
   396→      }
   397→      customerId = createResult.data.customerId
   398→      customerName = d.new_customer.name.trim()
   399→      // For a brand-new customer, the first address they provided IS the
   400→      // saved default address. Mark it as the selected address so we bump
   401→      // its lastUsedAt after order creation.
   402→      const newAddr = await db.customerAddress.findFirst({
   403→        where: { customerId, isDefault: true },
   404→        select: { id: true },
   405→      })
   406→      if (newAddr) {
   407→        usedCustomerAddressId = newAddr.id
   408→        selectedSavedAddressId = newAddr.id
   409→      }
   410→      // Same for the primary phone.
   411→      const newPhone = await db.customerPhone.findFirst({
   412→        where: { customerId, isPrimary: true },
   413→        select: { id: true },
   414→      })
   415→      if (newPhone) {
   416→        usedCustomerPhoneId = newPhone.id
   417→      }
   418→      // For a brand-new customer, we already saved the address as part of
   419→      // createCustomer — don't double-save it.
   420→      saveAddressForNextTime = false
   421→    } else {
   422→      return { success: false, error: 'Either customer_id or new_customer is required' }
   423→    }
   424→
   425→    // 3. recipient_name defaults to the customer's name if not explicitly overridden.
   426→    const recipientName = (d.recipient_name && d.recipient_name.trim()) || customerName
   427→
   428→    // 3. Fetch variants + their pricing for this company + fulfillment_type snapshot
   429→    const variantIds = d.items.map((i) => i.org_variant_id)
   430→    const variants = await db.orgProductVariant.findMany({
   431→      where: { id: { in: variantIds }, organizationId: ctx.company.organizationId },
   432→      include: {
   433→        product: { select: { title: true } },
   434→        companyPricing: { where: { companyId: ctx.company.id } },
   435→      },
   436→    })
   437→
   438→    if (variants.length !== variantIds.length) {
   439→      return { success: false, error: 'One or more variants not found in this organization' }
   440→    }
   441→
   442→    // 4. Compute subtotal + build order items data
   443→    let subtotal = 0
   444→    const orderItemsData: Array<{
   445→      orgVariantId: string
   446→      quantity: number
   447→      unitPrice: number
   448→      lineTotal: number
   449→      fulfillmentTypeSnapshot: string
   450→    }> = []
   451→
   452→    // Build the order_detail string as we iterate — universal courier
   453→    // reference field (migration 015). Format:
   454→    //   "Product Title (SKU-001, Size: M, Color: Blue) ×2, ..."
   455→    // Auto-generated but editable from the order-create form. If the caller
   456→    // supplies an explicit order_detail, we honour it verbatim.
   457→    const orderDetailParts: string[] = []
   458→
   459→    for (const item of d.items) {
   460→      const variant = variants.find((v) => v.id === item.org_variant_id)
   461→      if (!variant) continue
   462→
   463→      // Use provided unit_price OR fall back to company pricing OR variant cost
   464→      const unitPrice =
   465→        item.unit_price ??
   466→        (variant.companyPricing[0]?.salePrice
   467→          ? Number(variant.companyPricing[0].salePrice)
   468→          : Number(variant.costPrice))
   469→
   470→      const lineTotal = item.quantity * unitPrice
   471→      subtotal += lineTotal
   472→
   473→      orderItemsData.push({
   474→        orgVariantId: item.org_variant_id,
   475→        quantity: item.quantity,
   476→        unitPrice,
   477→        lineTotal,
   478→        fulfillmentTypeSnapshot: variant.fulfillmentType,
   479→      })
   480→
   481→      // Append to orderDetail: "Product Title (SKU-001, Size: M) ×2"
   482→      const attrParts: string[] = []
   483→      try {
   484→        const attrs = JSON.parse(variant.attributeValues || '{}') as Record<string, string>
   485→        for (const [k, v] of Object.entries(attrs)) {
   486→          if (v) attrParts.push(`${k}: ${v}`)
   487→        }
   488→      } catch {
   489→        // ignore parse errors
   490→      }
   491→      const inner = [variant.sku, ...attrParts].filter(Boolean).join(', ')
   492→      orderDetailParts.push(
   493→        `${variant.product.title}${inner ? ` (${inner})` : ''} ×${item.quantity}`,
   494→      )
   495→    }
   496→
   497→    // 5. Compute totals
   498→    const discountAmount = d.discount_amount ?? 0
   499→    const courierCharges = d.courier_charges ?? 0
   500→    const estimatedDeliveryCharge = d.estimated_delivery_charge ?? 0
   501→    const taxAmount = d.tax_amount ?? 0
   502→    // Total = subtotal + courier charges + delivery charge + tax - discount
   503→    // (delivery charge and tax are ADDITIVE — not absorbed into subtotal)
   504→    const totalOrderValue = subtotal + courierCharges + estimatedDeliveryCharge + taxAmount - discountAmount
   505→
   506→    // 6. Determine payment status + source
   507→    let paymentStatus: string = 'cod_pending'
   508→    let paymentSource: string = 'cod_native'
   509→    let advanceAmount: number | null = null
   510→    let advancePaymentMethod: string | null = null
   511→    let advancePaymentReference: string | null = null
   512→    let advancePaidAt: Date | null = null
   513→
   514→    if (d.payment_type === 'partial_advance') {
   515→      paymentStatus = 'advance_paid'
   516→      paymentSource = 'manual_conversion'
   517→      advanceAmount = d.advance_amount ?? 0
   518→      advancePaymentMethod = d.advance_payment_method || null
   519→      advancePaymentReference = d.advance_payment_reference || null
   520→      advancePaidAt = new Date()
   521→    } else if (d.payment_type === 'fully_prepaid') {
   522→      paymentStatus = 'fully_prepaid'
   523→      paymentSource = 'manual_conversion'
   524→      advanceAmount = totalOrderValue
   525→      advancePaymentMethod = d.advance_payment_method || null
   526→      advancePaymentReference = d.advance_payment_reference || null
   527→      advancePaidAt = new Date()
   528→    }
   529→
   530→    // remainingCodAmount is NOT a GENERATED column in the DB — the application
   531→    // MUST compute it as totalOrderValue - (advanceAmount ?? 0).
   532→    const remainingCodAmount = totalOrderValue - (advanceAmount ?? 0)
   533→
   534→    // 7. Determine initial order status
   535→    let orderStatus: string = 'pending'
   536→    let confirmedAt: Date | null = null
   537→    let skippedConfirmation = false
   538→
   539→    if (paymentStatus === 'advance_paid' || paymentStatus === 'fully_prepaid') {
   540→      // Payment itself is the confirmation signal — bypasses require_order_confirmation
   541→      orderStatus = 'confirmed'
   542→      confirmedAt = new Date()
   543→    } else {
   544→      // COD order — check company_order_settings
   545→      const settings = await db.companyOrderSetting.findUnique({
   546→        where: { companyId: ctx.company.id },
   547→      })
   548→      if (!settings?.requireOrderConfirmation) {
   549→        orderStatus = 'confirmed'
   550→        confirmedAt = new Date()
   551→        skippedConfirmation = true
   552→      }
   553→    }
   554→
   555→    // 8. Generate order number
   556→    const flowopsOrderNumber = await generateOrderNumber(ctx.company.id)
   557→
   558→    // 8a. Universal courier reference fields (migration 015).
   559→    // orderRefNumber: defaults to flowopsOrderNumber if the caller left it
   560→    // blank — every courier has a "reference" field so this is a core OMS
   561→    // field, mapped to the courier's own ref at booking time.
   562→    const orderRefNumber =
   563→      (d.order_ref_number && d.order_ref_number.trim()) || flowopsOrderNumber
   564→    // orderDetail: caller-provided takes precedence, otherwise use the
   565→    // auto-generated string from the cart items (product title + SKU +
   566→    // variant attributes + quantity).
   567→    const orderDetail =
   568→      (d.order_detail && d.order_detail.trim()) || orderDetailParts.join(', ')
   569→
   570→    // 9. Create the order
   571→    const order = await db.order.create({
   572→      data: {
   573→        organizationId: ctx.company.organizationId,
   574→        companyId: ctx.company.id,
   575→        flowopsOrderNumber,
   576→        orderSource: 'manual',
   577→        customerId,
   578→        // Customer Management System integration (migration 002):
   579→        //   - recipientName: the name to deliver to (defaults to customer.name)
   580→        //   - usedCustomerAddressId / usedCustomerPhoneId: which saved
   581→        //     customer_addresses / customer_phones rows were used for this
   582→        //     order, for lastUsedAt tracking and reporting. Nullable when a
   583→        //     one-off address/phone was typed without saving.
   584→        recipientName,
   585→        usedCustomerAddressId,
   586→        usedCustomerPhoneId,
   587→        status: orderStatus,
   588→        paymentType: d.payment_type,
   589→        paymentStatus,
   590→        paymentSource,
   591→        subtotal,
   592→        discountAmount: discountAmount || null,
   593→        discountReason: d.discount_reason || null,
   594→        courierCharges: courierCharges || null,
   595→        estimatedDeliveryCharge: estimatedDeliveryCharge || null,
   596→        taxAmount: taxAmount || null,
   597→        taxLabel: d.tax_label?.trim() || null,
   598→        totalOrderValue,
   599→        advanceAmount,
   600→        advancePaymentMethod,
   601→        advancePaymentReference,
   602→        advancePaidAt,
   603→        remainingCodAmount,
   604→        deliveryAddress: d.delivery_address,
   605→        deliveryCity: d.delivery_city,
   606→        courierName: d.courier_name || null,
   607→        courierCompanyIntegrationId: d.courier_company_integration_id || null,
   608→        recommendedCourierCompanyIntegrationId: d.courier_company_integration_id || null,
   609→        dispatchLocationId: d.dispatch_location_id,
   610→        notesForCourier: d.notes_for_courier || null,
   611→        // Per-order pickup address override (null = use integration default at booking time)
   612→        pickupAddressId: d.pickup_address_id || null,
   613→        // Universal courier reference fields (migration 015)
   614→        orderRefNumber,
   615→        orderDetail,
   616→        skippedConfirmation,
   617→        confirmedAt,
   618→        createdBy: ctx.employee.id,
   619→      },
   620→    })
   621→
   622→    // 10. Create order items (fulfillment_status = 'reserved' as PLACEHOLDER)
   623→    const createdItems: Array<{ id: string; orgVariantId: string; quantity: number }> = []
   624→    for (const itemData of orderItemsData) {
   625→      const item = await db.orderItem.create({
   626→        data: {
   627→          orderId: order.id,
   628→          orgVariantId: itemData.orgVariantId,
   629→          organizationId: ctx.company.organizationId,
   630→          quantity: itemData.quantity,
   631→          unitPrice: itemData.unitPrice,
   632→          lineTotal: itemData.lineTotal,
   633→          fulfillmentStatus: 'reserved', // PLACEHOLDER — Step 3 will validate/adjust
   634→          fulfillmentTypeSnapshot: itemData.fulfillmentTypeSnapshot,
   635→          reservedLocationId: d.dispatch_location_id,
   636→        },
   637→      })
   638→      createdItems.push({
   639→        id: item.id,
   640→        orgVariantId: item.orgVariantId,
   641→        quantity: item.quantity,
   642→      })
   643→    }
   644→
   645→    // 11. Audit log
   646→    await insertAuditLog({
   647→      action: 'order.created',
   648→      entityType: 'order',
   649→      entityId: order.id,
   650→      companyId: ctx.company.id,
   651→      organizationId: ctx.company.organizationId,
   652→      userId: ctx.user.id,
   653→      employeeId: ctx.employee.id,
   654→      newValues: {
   655→        flowopsOrderNumber,
   656→        status: orderStatus,
   657→        paymentType: d.payment_type,
   658→        paymentStatus,
   659→        itemCount: orderItemsData.length,
   660→        totalOrderValue,
   661→        customerId,
   662→      },
   663→    })
   664→
   665→    // 12. Customer Management System integration:
   666→    //   a. If a saved address was selected, bump its lastUsedAt to NOW() so
   667→    //      the UI can sort saved addresses by recency.
   668→    //   b. If a brand-new one-off address was typed AND the caller set
   669→    //      save_address_for_next_time=true, persist it as a new
   670→    //      customer_addresses row (and link it via usedCustomerAddressId
   671→    //      for future tracking). This is the "Save this address for next
   672→    //      time" UX from Step 3's frontend.
   673→    //   c. Recompute cached customer stats (total_orders_count, etc.) and
   674→    //      auto-flag if RTO threshold crossed.
   675→    if (selectedSavedAddressId) {
   676→      await markAddressAsUsed(selectedSavedAddressId)
   677→    } else if (saveAddressForNextTime && d.delivery_address && d.delivery_city) {
   678→      // One-off address typed + user opted to save it for next time.
   679→      // Persist as a non-default customer_addresses row (default stays as-is
   680→      // unless the user explicitly changes it via the customer profile).
   681→      const savedAddr = await db.customerAddress.create({
   682→        data: {
   683→          customerId,
   684→          organizationId: ctx.company.organizationId,
   685→          label: null,
   686→          address: d.delivery_address,
   687→          city: d.delivery_city,
   688→          isDefault: false,
   689→          lastUsedAt: new Date(),
   690→        },
   691→      })
   692→      // Link the order to the newly-saved address for future tracking.
   693→      await db.order.update({
   694→        where: { id: order.id },
   695→        data: { usedCustomerAddressId: savedAddr.id },
   696→      })
   697→    }
   698→
   699→    await updateCustomerStats(customerId)
   700→
   701→    // 13. If the order auto-confirmed (payment-driven or company setting),
   702→    // run the stock reservation logic immediately.
   703→    if (orderStatus === 'confirmed') {
   704→      await reserveOrderStock(order.id, ctx)
   705→    }
   706→
   707→    // 14. AUTO-BOOKING (Phase 3): if the company's courierBookingMode is
   708→    // 'automatic' AND the order is confirmed, automatically book the courier.
   709→    //
   710→    // ASYNCHRONOUS: PostEx's API can take 50-100 seconds to respond. Running
   711→    // the booking synchronously would block the order creation response for
   712→    // that long, causing a 502 Bad Gateway from the ALB. Instead, we fire the
   713→    // booking in the background and return immediately. The order is created
   714→    // with courierBookingStatus='not_booked', and the background task updates
   715→    // it to 'booked' or 'failed' when PostEx responds.
   716→    //
   717→    // The frontend shows "Order created. Booking in progress…" and can poll
   718→    // the order detail to see when the tracking number appears.
   719→    let bookingAttempted = false
   720→    let bookingSucceeded = false
   721→    let bookingError: string | undefined
   722→    let bookingTrackingNumber: string | undefined
   723→
   724→    if (orderStatus === 'confirmed') {
   725→      // Check if auto-booking should fire (quick synchronous check — no API call)
   726→      try {
   727→        const settings = await db.companyOrderSetting.findUnique({
   728→          where: { companyId: ctx.company.id },
   729→          select: { courierBookingMode: true, defaultCourierCompanyIntegrationId: true },
   730→        })
   731→        const shouldAutoBook =
   732→          settings?.courierBookingMode === 'automatic' &&
   733→          (d.courier_company_integration_id || settings?.defaultCourierCompanyIntegrationId)
   734→
   735→        if (shouldAutoBook) {
   736→          bookingAttempted = true // we're going to attempt it in the background
   737→          // Fire-and-forget: the booking runs in the background. The order
   738→          // creation returns immediately with bookingSucceeded=undefined
   739→          // (meaning "in progress"). When the background task completes, it
   740→          // updates the order's courierBookingStatus + trackingNumber.
   741→          ;(async () => {
   742→            try {
   743→              const { maybeAutoBookOrder } = await import('./booking.actions')
   744→              const bookResult = await maybeAutoBookOrder(order.id, 'manual', orderStatus)
   745→              if (bookResult.success) {
   746→                console.log(`[createManualOrder] Background auto-booking succeeded for ${flowopsOrderNumber}: tracking=${bookResult.data?.trackingNumber}`)
   747→              } else {
   748→                console.warn(`[createManualOrder] Background auto-booking failed for ${flowopsOrderNumber}: ${bookResult.error}`)
   749→              }
   750→            } catch (err) {
   751→              console.error(`[createManualOrder] Background auto-booking threw for ${flowopsOrderNumber}:`, err)
   752→            }
   753→          })()
   754→        }
   755→      } catch {
   756→        // Settings check failed — non-fatal, just skip auto-booking
   757→      }
   758→    }
   759→
   760→    await insertMetricEvent({
   761→      companyId: ctx.company.id,
   762→      entityType: 'order',
   763→      entityId: order.id,
   764→      metricKey: 'order.created',
   765→      numericValue: Number(totalOrderValue),
   766→      dimensions: { order_source: 'manual', payment_type: d.payment_type, company_id: ctx.company.id },
   767→    }).catch(() => {})
   768→
   769→    return {
   770→      success: true,
   771→      data: {
   772→        orderId: order.id,
   773→        flowopsOrderNumber,
   774→        orderItems: createdItems,
   775→        // Phase 5: booking result — distinct from order creation success.
   776→        // The frontend uses bookingAttempted + bookingSucceeded to show TWO
   777→        // separate messages: "Order created" + (if applicable) "Booking failed".
   778→        bookingAttempted,
   779→        bookingSucceeded,
   780→        bookingError,
   781→        bookingTrackingNumber,
   782→      },
   783→    }
   784→  } catch (err) {
   785→    return {
   786→      success: false,
   787→      error: err instanceof Error ? err.message : 'Failed to create order',
   788→    }
   789→  }
   790→}

   793→// createOrderFromShopifyWebhook — STUB (structured but not wired)
   794→// ──────────────────────────────────────────────────────────────
   795→
   796→export async function createOrderFromShopifyWebhook(
   797→  payload: ShopifyOrderWebhook,
   798→  companyId: string,
   799→  organizationId: string,
   800→): Promise<ActionResult<{ orderId: string; flowopsOrderNumber: string }>> {
   801→  try {
   802→    const ctx = await getWorkspace()
   803→    // This function would be called by a future webhook handler.
   804→    // For now it's structured and unit-testable with a mock payload.
   805→
   806→    const parsed = shopifyOrderWebhookSchema.safeParse(payload)
   807→    if (!parsed.success) {
   808→      return { success: false, error: 'Invalid Shopify webhook payload' }
   809→    }
   810→    const d = parsed.data
   811→
   812→    // Map Shopify financial_status → FlowOps payment_status
   813→    let paymentStatus: string
   814→    let paymentSource: string = 'shopify_gateway'
   815→    let paymentType: string
   816→    let orderStatus: string = 'pending'
   817→    let confirmedAt: Date | null = null
   818→    let advanceAmount: number | null = null
   819→    let advancePaidAt: Date | null = null
   820→
   821→    switch (d.financial_status) {
   822→      case 'paid':
   823→        paymentStatus = 'fully_prepaid'
   824→        paymentType = 'fully_prepaid'
   825→        orderStatus = 'confirmed'
   826→        confirmedAt = new Date()
   827→        advancePaidAt = new Date()
   828→        // advanceAmount will be set to totalOrderValue below (after total_price is parsed)
   829→        break
   830→      case 'partially_paid':
   831→        paymentStatus = 'advance_paid'
   832→        paymentType = 'partial_advance'
   833→        orderStatus = 'confirmed'
   834→        confirmedAt = new Date()
   835→        advancePaidAt = new Date()
   836→        // Shopify webhook does not carry the partial amount paid — leave advanceAmount null.
   837→        // remainingCodAmount will be computed as totalOrderValue - 0 = totalOrderValue.
   838→        break
   839→      default:
   840→        paymentStatus = 'cod_pending'
   841→        paymentType = 'full_cod'
   842→        paymentSource = 'cod_native'
   843→        // Check company settings for COD orders
   844→        const settings = await db.companyOrderSetting.findUnique({ where: { companyId } })
   845→        if (!settings?.requireOrderConfirmation) {
   846→          orderStatus = 'confirmed'
   847→          confirmedAt = new Date()
   848→        }
   849→    }
   850→
   851→    // Resolve customer via the cross-platform matching strategy (Step 1's
   852→    // match_or_create_customer() SQL function, wrapped by
   853→    // matchOrCreateExternalCustomer). This handles the layered:
   854→    //   1. exact_identity (existing Shopify customer mapping)
   855→    //   2. phone_match (normalized phone)
   856→    //   3. email_match
   857→    //   4. create new customer + primary phone + external identity mapping
   858→    //
   859→    // The customer's name comes from Shopify's first_name + last_name.
   860→    // Phone/email come from the Shopify payload (may be null for guest orders).
   861→    const shopifyCustomerName =
   862→      `${d.customer.first_name ?? ''} ${d.customer.last_name ?? ''}`.trim() ||
   863→      'Unknown Shopify Customer'
   864→
   865→    const matchResult = await matchOrCreateExternalCustomer({
   866→      platform: 'shopify',
   867→      external_customer_id: String(d.customer.id ?? d.id),
   868→      phone: d.customer.phone || undefined,
   869→      email: d.customer.email || undefined,
   870→      name: shopifyCustomerName,
   871→      organizationId,
   872→    })
   873→    if (!matchResult.success || !matchResult.data) {
   874→      return {
   875→        success: false,
   876→        error: matchResult.error ?? 'Could not resolve customer from Shopify payload',
   877→      }
   878→    }
   879→    const customerId = matchResult.data.customerId
   880→
   881→    // Fetch the resolved customer (for recipient_name default + address lookup).
   882→    const customer = await db.customer.findUnique({
   883→      where: { id: customerId },
   884→      include: {
   885→        phones: { where: { isPrimary: true }, take: 1 },
   886→        addresses: { where: { isDefault: true }, take: 1 },
   887→      },
   888→    })
   889→    if (!customer) {
   890→      return { success: false, error: 'Resolved customer not found' }
   891→    }
   892→
   893→    // If Shopify sent a default_address AND the customer has no saved address
   894→    // yet (newly created), persist it as their default customer_addresses row.
   895→    // For existing customers, we leave their saved addresses alone — the
   896→    // Shopify address becomes a one-off delivery snapshot on this order only.
   897→    let usedCustomerAddressId: string | null = null
   898→    let usedCustomerPhoneId: string | null = customer.phones[0]?.id ?? null
   899→    const shopifyAddress1 = d.customer.default_address?.address1 || null
   900→    const shopifyCity = d.customer.default_address?.city || null
   901→
   902→    if (shopifyAddress1 && shopifyCity && customer.addresses.length === 0) {
   903→      const newAddr = await db.customerAddress.create({
   904→        data: {
   905→          customerId: customer.id,
   906→          organizationId,
   907→          label: 'Shopify Default',
   908→          address: shopifyAddress1,
   909→          city: shopifyCity,
   910→          isDefault: true,
   911→          lastUsedAt: new Date(),
   912→        },
   913→      })
   914→      usedCustomerAddressId = newAddr.id
   915→    } else if (customer.addresses.length > 0) {
   916→      // Existing customer with a saved default address — use it for this order.
   917→      usedCustomerAddressId = customer.addresses[0].id
   918→      await markAddressAsUsed(usedCustomerAddressId)
   919→    }
   920→
   921→    const recipientName = shopifyCustomerName
   922→
   923→    // Resolve variants by SKU
   924→    const skus = d.line_items.map((li) => li.sku).filter(Boolean) as string[]
   925→    const variants = await db.orgProductVariant.findMany({
   926→      where: { sku: { in: skus }, organizationId },
   927→    })
   928→
   929→    // Compute subtotal
   930→    let subtotal = 0
   931→    const orderItemsData: Array<{
   932→      orgVariantId: string
   933→      quantity: number
   934→      unitPrice: number
   935→      fulfillmentTypeSnapshot: string
   936→    }> = []
   937→
   938→    for (const li of d.line_items) {
   939→      const variant = variants.find((v) => v.sku === li.sku)
   940→      if (!variant) continue // skip unmatched items for now
   941→      const unitPrice = parseFloat(li.price)
   942→      subtotal += unitPrice * li.quantity
   943→      orderItemsData.push({
   944→        orgVariantId: variant.id,
   945→        quantity: li.quantity,
   946→        unitPrice,
   947→        fulfillmentTypeSnapshot: variant.fulfillmentType,
   948→      })
   949→    }
   950→
   951→    if (orderItemsData.length === 0) {
   952→      return { success: false, error: 'No matching variants found for Shopify line items' }
   953→    }
   954→
   955→    const totalOrderValue = parseFloat(d.total_price)
   956→
   957→    // For fully_prepaid Shopify orders, advanceAmount = totalOrderValue.
   958→    if (paymentStatus === 'fully_prepaid') {
   959→      advanceAmount = totalOrderValue
   960→    }
   961→
   962→    // remainingCodAmount is NOT a GENERATED column in the DB — the application
   963→    // MUST compute it as totalOrderValue - (advanceAmount ?? 0).
   964→    const remainingCodAmount = totalOrderValue - (advanceAmount ?? 0)
   965→
   966→    const flowopsOrderNumber = await generateOrderNumber(companyId)
   967→
   968→    // Build orderDetail from Shopify line items for the universal courier
   969→    // reference field (migration 015). Shopify doesn't expose variant
   970→    // attributes cleanly in the webhook payload, so we use a simpler
   971→    // format: "Product Title (SKU) ×qty, ..." — matching the existing
   972→    // booking-workbench book route's itemDescription format.
   973→    const orderDetailParts: string[] = []
   974→    for (const li of d.line_items) {
   975→      const variant = variants.find((v) => v.sku === li.sku)
   976→      const title = variant?.sku ?? li.sku ?? 'Item'
   977→      orderDetailParts.push(`${title} ×${li.quantity}`)
   978→    }
   979→    const orderDetail = orderDetailParts.join(', ')
   980→
   981→    const order = await db.order.create({
   982→      data: {
   983→        organizationId,
   984→        companyId,
   985→        flowopsOrderNumber,
   986→        orderSource: 'shopify',
   987→        externalOrderReference: d.name,
   988→        externalOrderId: String(d.id),
   989→        customerId,
   990→        // Customer Management System integration (migration 002):
   991→        recipientName,
   992→        usedCustomerAddressId,
   993→        usedCustomerPhoneId,
   994→        status: orderStatus,
   995→        paymentType,
   996→        paymentStatus,
   997→        paymentSource,
   998→        subtotal,
   999→        totalOrderValue,
  1000→        advanceAmount,
  1001→        advancePaidAt,
  1002→        remainingCodAmount,
  1003→        confirmedAt,
  1004→        // delivery_address is the order's own editable snapshot. Use the
  1005→        // Shopify address if provided; otherwise fall back to the saved
  1006→        // default address's text (if any).
  1007→        deliveryAddress: shopifyAddress1 || customer.addresses[0]?.address || null,
  1008→        deliveryCity: shopifyCity || customer.addresses[0]?.city || null,
  1009→        // Universal courier reference fields (migration 015):
  1010→        // orderRefNumber defaults to the Shopify order name (e.g. "#1001")
  1011→        // — staff can override post-creation if needed. orderDetail is the
  1012→        // auto-generated item summary above.
  1013→        orderRefNumber: d.name || flowopsOrderNumber,
  1014→        orderDetail,
  1015→      },
  1016→    })
  1017→
  1018→    for (const itemData of orderItemsData) {
  1019→      await db.orderItem.create({
  1020→        data: {
  1021→          orderId: order.id,
  1022→          orgVariantId: itemData.orgVariantId,
  1023→          organizationId,
  1024→          quantity: itemData.quantity,
  1025→          unitPrice: itemData.unitPrice,
  1026→          lineTotal: itemData.quantity * itemData.unitPrice,
  1027→          fulfillmentStatus: 'reserved',
  1028→          fulfillmentTypeSnapshot: itemData.fulfillmentTypeSnapshot,
  1029→        },
  1030→      })
  1031→    }
  1032→
  1033→    await insertAuditLog({
  1034→      action: 'order.created_from_shopify',
  1035→      entityType: 'order',
  1036→      entityId: order.id,
  1037→      companyId,
  1038→      organizationId,
  1039→      userId: ctx.user.id,
  1040→      employeeId: ctx.employee.id,
  1041→      newValues: {
  1042→        flowopsOrderNumber,
  1043→        externalOrderReference: d.name,
  1044→        financialStatus: d.financial_status,
  1045→        totalOrderValue,
  1046→      },
  1047→    })
  1048→
  1049→    await updateCustomerStats(customerId)
  1050→
  1051→    await insertMetricEvent({
  1052→      companyId,
  1053→      entityType: 'order',
  1054→      entityId: order.id,
  1055→      metricKey: 'order.created',
  1056→      numericValue: Number(totalOrderValue),
  1057→      dimensions: { order_source: 'shopify', payment_type: paymentType, company_id: companyId },
  1058→    }).catch(() => {})
  1059→
  1060→    return { success: true, data: { orderId: order.id, flowopsOrderNumber } }
  1061→  } catch (err) {
  1062→    return {
  1063→      success: false,
  1064→      error: err instanceof Error ? err.message : 'Failed to create order from Shopify webhook',
  1065→    }
  1066→  }
  1067→}

  1073→export async function confirmOrder(orderId: string): Promise<ActionResult> {
  1074→  try {
  1075→    const ctx = await getWorkspace()
  1076→    await requirePermission(ctx, PERMISSIONS.ORDERS_MANAGE)
  1077→
  1078→    const order = await db.order.findFirst({
  1079→      where: { id: orderId, companyId: ctx.company.id },
  1080→    })
  1081→    if (!order) return { success: false, error: 'Order not found' }
  1082→    if (order.status !== 'pending') {
  1083→      return { success: false, error: `Order is already ${order.status} (cannot confirm)` }
  1084→    }
  1085→
  1086→    await db.order.update({
  1087→      where: { id: orderId },
  1088→      data: { status: 'confirmed', confirmedAt: new Date() },
  1089→    })
  1090→
  1091→    await insertAuditLog({
  1092→      action: 'order.confirmed',
  1093→      entityType: 'order',
  1094→      entityId: orderId,
  1095→      companyId: ctx.company.id,
  1096→      organizationId: ctx.company.organizationId,
  1097→      userId: ctx.user.id,
  1098→      employeeId: ctx.employee.id,
  1099→      oldValues: { status: 'pending' },
  1100→      newValues: { status: 'confirmed' },
  1101→    })
  1102→
  1103→    // Step 3: Run stock reservation for all items.
  1104→    // This may flip the order to 'partially_backordered' if some items
  1105→    // can't be fulfilled immediately.
  1106→    const reserveResult = await reserveOrderStock(orderId, ctx)
  1107→    if (!reserveResult.success) {
  1108→      // Non-fatal — the order is confirmed, but reservation had issues.
  1109→      // The results array has per-item details.
  1110→    }
  1111→
  1112→    await insertMetricEvent({
  1113→      companyId: ctx.company.id,
  1114→      entityType: 'order',
  1115→      entityId: orderId,
  1116→      metricKey: 'order.confirmed',

  1376→export async function cancelOrder(input: CancelOrderInput): Promise<ActionResult> {
  1377→  try {
  1378→    const ctx = await getWorkspace()
  1379→    await requirePermission(ctx, PERMISSIONS.ORDERS_CANCEL)
  1380→
  1381→    const parsed = cancelOrderSchema.safeParse(input)
  1382→    if (!parsed.success) {
  1383→      return { success: false, error: parsed.error.issues[0]?.message ?? 'Invalid input' }
  1384→    }
  1385→    const d = parsed.data
  1386→
  1387→    const order = await db.order.findFirst({
  1388→      where: { id: d.order_id, companyId: ctx.company.id },
  1389→    })
  1390→    if (!order) return { success: false, error: 'Order not found' }
  1391→
  1392→    const nonCancellableStatuses = ['dispatched', 'delivered', 'rto', 'cancelled', 'refunded']
  1393→    if (nonCancellableStatuses.includes(order.status)) {
  1394→      return {
  1395→        success: false,
  1396→        error: `Cannot cancel an order that is already ${order.status} (use the returns flow instead)`,
  1397→      }
  1398→    }
  1399→
  1400→    // ── Phase 3: Set physicalUnpackRequired if the order was already
  1401→    // processing/packed (a physical parcel likely already exists) ──
  1402→    const physicalUnpackRequired =
  1403→      order.status === 'processing' ||
  1404→      order.packedAt !== null
  1405→
  1406→    await db.order.update({
  1407→      where: { id: d.order_id },
  1408→      data: {
  1409→        status: 'cancelled',
  1410→        cancelledAt: new Date(),
  1411→        cancellationReason: d.cancellation_reason,
  1412→        physicalUnpackRequired,
  1413→      },
  1414→    })
  1415→
  1416→    // Step 3: Unreserve stock for any items with fulfillment_status='reserved'.
  1417→    // Items with fulfillment_status='backordered' need no inventory action
  1418→    // (no reservation ever existed for them) — they'll be orphaned since
  1419→    // the order is now cancelled and won't be picked up by future
  1420→    // checkAndFulfillBackorders() runs (which skip cancelled orders).
  1421→    const reservedItems = await db.orderItem.findMany({
  1422→      where: { orderId: d.order_id, fulfillmentStatus: 'reserved' },
  1423→    })
  1424→
  1425→    for (const item of reservedItems) {
  1426→      const locationId = item.reservedLocationId ?? order.dispatchLocationId
  1427→      if (!locationId) continue
  1428→
  1429→      await unreserveStockForOrder({
  1430→        orgVariantId: item.orgVariantId,
  1431→        locationId,
  1432→        organizationId: order.organizationId,
  1433→        companyId: ctx.company.id,
  1434→        employeeId: ctx.employee.id,
  1435→        quantity: item.quantity,
  1436→        orderId: d.order_id,
  1437→      })
  1438→    }
  1439→
  1440→    await insertAuditLog({
  1441→      action: 'order.cancelled',
  1442→      entityType: 'order',
  1443→      entityId: d.order_id,
  1444→      companyId: ctx.company.id,
  1445→      organizationId: ctx.company.organizationId,
  1446→      userId: ctx.user.id,
  1447→      employeeId: ctx.employee.id,
  1448→      oldValues: { status: order.status },
  1449→      newValues: { status: 'cancelled', reason: d.cancellation_reason, unreservedItems: reservedItems.length },
  1450→    })
  1451→
  1452→    await insertMetricEvent({
  1453→      companyId: ctx.company.id,
  1454→      entityType: 'order',
  1455→      entityId: d.order_id,
  1456→      metricKey: 'order.cancelled',
  1457→      numericValue: Number(order.totalOrderValue),
  1458→      dimensions: { cancellation_reason: d.cancellation_reason, had_reserved_items: reservedItems.length > 0 },
  1459→    }).catch(() => {})
  1460→
  1461→    // Recompute cached customer stats (cancelled orders are excluded from
  1462→    // total_orders_count, so this count drops). Non-fatal.
  1463→    await updateCustomerStats(order.customerId).catch(() => {})
  1464→
  1465→    return { success: true }
  1466→  } catch (err) {
  1467→    return {
  1468→      success: false,
  1469→      error: err instanceof Error ? err.message : 'Failed to cancel order',
  1470→    }
  1471→  }
  1472→}

  1649→                0,
