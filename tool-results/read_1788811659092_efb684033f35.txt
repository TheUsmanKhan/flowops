    41→// ──────────────────────────────────────────────────────────────
    42→// Phone format conversion helper
    43→// ──────────────────────────────────────────────────────────────
    44→
    45→/**
    46→ * Convert a Pakistani phone number to PostEx's required format: 03XXXXXXXXX.
    47→ *
    48→ * PostEx's API requires customerPhone in format "03xxxxxxxxx" (11 digits,
    49→ * starting with 03). Our CRM stores phones in E.164 format (+92XXXXXXXXXX).
    50→ *
    51→ * Conversion rules:
    52→ *   +92 3XX XXXXXXX → 03XX XXXXXXX  (strip +92, prepend 0)
    53→ *   92 3XX XXXXXXX  → 03XX XXXXXXX  (strip 92, prepend 0)
    54→ *   03XXXXXXXXX     → 03XXXXXXXXX  (already correct)
    55→ *   3XXXXXXXXX      → 03XXXXXXXXX  (prepend 0)
    56→ *   Anything else   → return as-is (let PostEx validate/reject)
    57→ */
    58→function convertToPostExPhone(phone: string): string {
    59→  const trimmed = phone.trim()
    60→
    61→  // Already in 03XXXXXXXXX format
    62→  if (/^03\d{9}$/.test(trimmed)) {
    63→    return trimmed
    64→  }
    65→
    66→  // +92 3XX... → 03XX...
    67→  if (trimmed.startsWith('+92')) {
    68→    const rest = trimmed.substring(3) // remove +92
    69→    if (rest.startsWith('3')) {
    70→      return '0' + rest
    71→    }
    72→    return '0' + rest // fallback
    73→  }
    74→
    75→  // 92 3XX... → 03XX...
    76→  if (trimmed.startsWith('92') && trimmed.length >= 11) {
    77→    const rest = trimmed.substring(2) // remove 92
    78→    if (rest.startsWith('3')) {
    79→      return '0' + rest
    80→    }
    81→    return '0' + rest
    82→  }
    83→
    84→  // 3XX... (missing leading 0) → 03XX...
    85→  if (/^3\d{9}$/.test(trimmed)) {
    86→    return '0' + trimmed
    87→  }
    88→
    89→  // Unknown format — return as-is, let PostEx validate
    90→  return trimmed
    91→}

   160→  async bookShipment(input: BookShipmentInput): Promise<BookShipmentResult> {
   161→    // NOTE: City validation is handled by the CALLER (booking-workbench/book
   162→    // route or the auto-booking server action) via revalidateCityAtBookingTime()
   163→    // with the companyIntegrationId — which enables the live PostEx fallback.
   164→    // The adapter doesn't have the integration ID, so it can't do the live
   165→    // fallback. We trust the caller's pre-validation.
   166→
   167→    // Build the request body — ONLY fields that exist in the confirmed API.
   168→    // NO weight/handling/itemsQty/paymentMethod/orderTags fields.
   169→    const body: Record<string, unknown> = {
   170→      cityName: input.deliveryCity,
   171→      customerName: input.recipientName,
   172→      customerPhone: convertToPostExPhone(input.recipientPhone),
   173→      deliveryAddress: input.deliveryAddress,
   174→      invoiceDivision: 1, // default 1 (split airway bills for larger orders)
   175→      invoicePayment: input.codAmount,
   176→      items: input.quantity ?? 1,
   177→      orderDetail: input.itemDescription || 'Order',
   178→      orderRefNumber: input.orderNumber,
   179→      orderType: input.orderType ?? 'Normal',
   180→      transactionNotes: input.transactionNotes ?? '',
   181→      pickupAddressCode: input.pickupAddressCode ?? '',
   182→    }
   183→    // NOTE: storeAddressCode is intentionally OMITTED. Sending it (even as
   184→    // empty string) causes PostEx to validate it and reject with
   185→    // "INVALID MERCHANT STORE ADDRESS CODE". PostEx uses its default store
   186→    // address when the field is absent.
   187→
   188→    const response = await fetch(`${POSTEX_BASE_URL}/v3/create-order`, {
   189→      method: 'POST',
   190→      headers: {
   191→        'Content-Type': 'application/json',
   192→        token: this.token,
   193→      },
   194→      body: JSON.stringify(body),
   195→    })
   196→
   197→    const json: PostExApiResponse<PostExCreateOrderDist> = await response.json()
   198→
   199→    if (json.statusCode === 200 || json.statusCode === '200') {
   200→      const trackingNumber = json.dist?.trackingNumber
   201→      if (!trackingNumber) {
   202→        return {
   203→          success: false,
   204→          error: 'PostEx returned success but no tracking number in response.',
   205→          rawResponse: json,
   206→        }
   207→      }
   208→
   209→      return {
   210→        success: true,
   211→        trackingNumber,
   212→        providerStatus: json.dist?.orderStatus ?? 'Unbooked',
   213→        rawResponse: json,
   214→      }
   215→    }
   216→
   217→    return {
   218→      success: false,
   219→      error: json.statusMessage || `PostEx API returned statusCode ${json.statusCode}`,
   220→      rawResponse: json,
   221→    }
   222→  }

   224→  // ──────────────────────────────────────────────────────────────
   225→  // 2. trackShipment — GET v1/track-order/{trackingNumber}
   226→  // ──────────────────────────────────────────────────────────────
   227→
   228→  async trackShipment(trackingNumber: string): Promise<TrackShipmentResult> {
   229→    const response = await fetch(
   230→      `${POSTEX_BASE_URL}/v1/track-order/${encodeURIComponent(trackingNumber)}`,
   231→      {
   232→        method: 'GET',
   233→        headers: {
   234→          token: this.token,
   235→        },
   236→      },
   237→    )
   238→
   239→    const json: PostExApiResponse<PostExTrackOrderDist> = await response.json()
   240→
   241→    if (json.statusCode === 200 || json.statusCode === '200') {
   242→      const dist = json.dist
   243→      if (!dist) {
   244→        return {
   245→          success: false,
   246→          error: 'PostEx returned success but no dist in response.',
   247→          rawResponse: json,
   248→        }
   249→      }
   250→
   251→      const postexStatus = dist.transactionStatus || 'Unbooked'
   252→      const mapped = mapPostExStatus(postexStatus)
   253→
   254→      // Map to the generic TrackShipmentResult status enum
   255→      let genericStatus: TrackShipmentResult['status'] = 'booked'
   256→      if (mapped.triggerDelivered) genericStatus = 'delivered'
   257→      else if (mapped.triggerRto) genericStatus = 'returned'
   258→      else if (mapped.triggerDispatch) genericStatus = 'in_transit'
   259→      else if (mapped.orderStatus === 'cancelled') genericStatus = 'failed' // reuse 'failed' for cancelled-by-merchant/expired
   260→
   261→      return {
   262→        success: true,
   263→        status: genericStatus,
   264→        lastUpdateAt: (dist.transactionDate as string) ?? undefined,
   265→        rawResponse: {
   266→          ...dist, // includes transactionStatusHistory for audit/display
   267→          mappedSubStatus: mapped.courierSubStatus,
   268→          needsShipperAdvice: mapped.needsShipperAdvice,
   269→          unrecognized: mapped.unrecognized,
   270→        },
   271→      }
   272→    }
   273→
   274→    return {
   275→      success: false,
   276→      error: json.statusMessage || `PostEx API returned statusCode ${json.statusCode}`,
   277→      rawResponse: json,
   278→    }
   279→  }

   281→  // ──────────────────────────────────────────────────────────────
   282→  // 3. trackBulkShipments — GET v1/track-bulk-order
   283→  // ──────────────────────────────────────────────────────────────
   284→
   285→  async trackBulkShipments(trackingNumbers: string[]): Promise<TrackShipmentResult[]> {
   286→    if (trackingNumbers.length === 0) return []
   287→
   288→    // Chunk into groups of 50 (default — PostEx docs don't specify a limit,
   289→    // but 50 is a reasonable batch size to avoid URL length issues)
   290→    const CHUNK_SIZE = 50
   291→    const results: TrackShipmentResult[] = []
   292→
   293→    for (let i = 0; i < trackingNumbers.length; i += CHUNK_SIZE) {
   294→      const chunk = trackingNumbers.slice(i, i + CHUNK_SIZE)
   295→
   296→      // PostEx's bulk tracking API uses GET with query params.
   297→      // The parameter name is "TrackingNumbers" (capital T, capital N)
   298→      // — confirmed from the live API error message:
   299→      // "Required List parameter 'TrackingNumbers' is not present"
   300→      const queryString = chunk.map((t) => `TrackingNumbers=${encodeURIComponent(t)}`).join('&')
   301→
   302→      let response: Response
   303→      try {
   304→        response = await fetch(
   305→          `${POSTEX_BASE_URL}/v1/track-bulk-order?${queryString}`,
   306→          {
   307→            method: 'GET',
   308→            headers: {
   309→              token: this.token,
   310→            },
   311→          },
   312→        )
   313→      } catch (fetchErr) {
   314→        // Network error on the bulk call — fall back to single-track for this chunk
   315→        console.warn('[postex] bulk track fetch failed, falling back to single-track:', fetchErr instanceof Error ? fetchErr.message : fetchErr)
   316→        for (const tn of chunk) {
   317→          results.push(await this.trackShipment(tn))
   318→        }
   319→        continue
   320→      }
   321→
   322→      // If the HTTP status itself is an error (400/500), fall back to single-track.
   323→      // PostEx's bulk endpoint intermittently returns 400 "Required List parameter
   324→      // 'TrackingNumbers' is not present" even when the param IS present — this is
   325→      // a known PostEx API instability. Falling back to single-track (which uses
   326→      // a different endpoint /v1/track-order/{tn}) is reliable.
   327→      if (!response.ok) {
   328→        console.warn(`[postex] bulk track returned HTTP ${response.status}, falling back to single-track for ${chunk.length} orders`)
   329→        for (const tn of chunk) {
   330→          results.push(await this.trackShipment(tn))
   331→        }
   332→        continue
   333→      }
   334→
   335→      const json: PostExApiResponse<PostExBulkTrackItem[]> = await response.json()
   336→
   337→      if (json.statusCode === 200 || json.statusCode === '200') {
   338→        const dist = json.dist ?? []
   339→        for (const item of dist) {
   340→          if (!item.trackingResponse) {
   341→            results.push({
   342→              success: false,
   343→              error: item.message || 'No tracking response for this number.',
   344→              rawResponse: item,
   345→            })
   346→            continue
   347→          }
   348→
   349→          const postexStatus = item.trackingResponse.transactionStatus || 'Unbooked'
   350→          const mapped = mapPostExStatus(postexStatus)
   351→
   352→          let genericStatus: TrackShipmentResult['status'] = 'booked'
   353→          if (mapped.triggerDelivered) genericStatus = 'delivered'
   354→          else if (mapped.triggerRto) genericStatus = 'returned'
   355→          else if (mapped.triggerDispatch) genericStatus = 'in_transit'
   356→          else if (mapped.orderStatus === 'cancelled') genericStatus = 'failed' // cancelled-by-merchant/expired
   357→
   358→          results.push({
   359→            success: true,
   360→            status: genericStatus,
   361→            lastUpdateAt: (item.trackingResponse.transactionDate as string) ?? undefined,
   362→            rawResponse: {
   363→              ...item.trackingResponse,
   364→              mappedSubStatus: mapped.courierSubStatus,
   365→              needsShipperAdvice: mapped.needsShipperAdvice,
   366→              unrecognized: mapped.unrecognized,
   367→            },
   368→          })
   369→        }
   370→      } else {
   371→        // API returned a non-200 statusCode in the JSON body — fall back to
   372→        // single-track for this chunk (same as the HTTP-error case above).
   373→        console.warn(`[postex] bulk track returned statusCode ${json.statusCode} (${json.statusMessage}), falling back to single-track for ${chunk.length} orders`)
   374→        for (const tn of chunk) {
   375→          results.push(await this.trackShipment(tn))
   376→        }
   377→      }
   378→    }
   379→
   380→    return results
   381→  }

   387→  async cancelShipment(trackingNumber: string): Promise<CancelShipmentResult> {
   388→    // NOTE: Only meaningful while PostEx status is Unbooked or Booked (per

   395→        token: this.token,
   396→      },
   397→      body: JSON.stringify({ trackingNumber }),

   399→

   401→      return { success: true }
   402→    }

   406→    }
   407→
   408→    let errorMessage = `PostEx cancel API returned HTTP ${response.status}`
   409→    try {
   410→      const json = await response.json()
   411→      errorMessage = json.statusMessage || errorMessage
   412→    } catch {
   413→      // Response body not JSON — use the HTTP status message
   414→    }
   415→
   416→    return { success: false, error: errorMessage }
   417→  }

   445→  // ──────────────────────────────────────────────────────────────

   447→  /**
   448→   * Fetch ALL operational cities from PostEx.
   449→   *
   450→   * PostEx's v2/get-operational-city endpoint accepts an optional
   451→   * `operationalCityType` query param (Pickup | Delivery | null), but in
   452→   * practice PostEx's API returns errors ("statusCode undefined") when this
   453→   * param is set. Only the unfiltered call (no param) works reliably.
   454→   *
   455→   * We make a SINGLE unfiltered call with a 30-second timeout. This returns
   456→   * all cities with their isPickupCity/isDeliveryCity flags already set.
   457→   */
   458→  async fetchOperationalCities(): Promise<OperationalCity[]> {
   459→    const url = `${POSTEX_BASE_URL}/v2/get-operational-city`
   460→
   461→    const controller = new AbortController()
   462→    const timeout = setTimeout(() => controller.abort(), 30_000)
   463→    try {
   464→      const response = await fetch(url, {
   465→        method: 'GET',
   466→        headers: { token: this.token },
   467→        signal: controller.signal,
   468→      })
   469→      const json: PostExApiResponse<PostExOperationalCity[]> = await response.json()
   470→
   471→      if (json.statusCode === 200 || json.statusCode === '200') {
   472→        const dist = json.dist ?? []
   473→        if (dist.length === 0) {
   474→          throw new Error('PostEx cities API returned 0 cities. Check the integration token.')
   475→        }
   476→        return dist.map((c) => ({
   477→          cityName: c.operationalCityName,
   478→          cityId: undefined,
   479→          isPickupCity: c.isPickupCity === true || c.isPickupCity === 'true',
   480→          isDeliveryCity: c.isDeliveryCity === true || c.isDeliveryCity === 'true',
   481→        }))
   482→      }
   483→
   484→      throw new Error(
   485→        json.statusMessage || `PostEx cities API returned statusCode ${json.statusCode}`,
   486→      )
   487→    } finally {
   488→      clearTimeout(timeout)
   489→    }
   490→  }

   519→        }
   520→      }

   522→    } catch (e) {
   523→      return {
   524→        success: false,

   588→    contactPersonName: string
   589→    phone1: string
   590→    phone2?: string
   591→  }>> {
   592→    const response = await fetch(
   593→      `${POSTEX_BASE_URL}/v1/get-merchant-address`,
   594→      {
   595→        method: 'GET',
   596→        headers: {
   597→          token: this.token,
   598→        },
   599→      },
   600→    )
   601→
   602→    const json: PostExApiResponse<PostExMerchantAddress[]> = await response.json()
   603→
   604→    if (json.statusCode === 200 || json.statusCode === '200') {
   605→      const dist = json.dist ?? []
   606→      return dist.map((a) => ({
   607→        providerAddressCode: a.addressCode,
   608→        label: a.address, // use the address as the label (PostEx doesn't return a label)
   609→        address: a.address,
   610→        cityName: a.cityName,
   611→        contactPersonName: a.contactPersonName,

   622→  // ──────────────────────────────────────────────────────────────
   623→  // 10. generateLoadSheet — POST v2/generate-load-sheet
   624→  // ──────────────────────────────────────────────────────────────
   625→
   626→  async generateLoadSheet(
   627→    trackingNumbers: string[],
   628→    pickupAddress?: string,
   629→  ): Promise<{ success: boolean; rawResponse?: unknown; error?: string; pdfBase64?: string }> {
   630→    if (trackingNumbers.length === 0) {
   631→      return { success: false, error: 'No tracking numbers provided.' }
   632→    }
   633→
   634→    const body = {
   635→      pickupAddress: pickupAddress ?? '',
   636→      trackingNumbers,
   637→    }
   638→
   639→    const response = await fetch(
   640→      `${POSTEX_BASE_URL}/v2/generate-load-sheet`,
   641→      {
   642→        method: 'POST',
   643→        headers: {
   644→          'Content-Type': 'application/json',
   645→          token: this.token,
   646→        },
   647→        body: JSON.stringify(body),
   648→      },
   649→    )
   650→
   651→    // PostEx returns a PDF file for this endpoint, not JSON.
   652→    if (response.status === 200) {
   653→      // The response is a PDF binary — capture it as base64 so the caller
   654→      // can store it in our own file storage (not an external courier URL
   655→      // that might expire — same principle as Proof of Delivery).
   656→      const arrayBuffer = await response.arrayBuffer()
   657→      const buffer = Buffer.from(arrayBuffer)
   658→      const pdfBase64 = buffer.toString('base64')
   659→
   660→      return {
   661→        success: true,
   662→        pdfBase64,
   663→        rawResponse: {
   664→          contentType: response.headers.get('content-type'),
   665→          contentLength: response.headers.get('content-length'),
   666→          status: response.status,
   667→          pdfSizeBytes: buffer.length,
   668→        },
   669→      }
   670→    }
   671→
   672→    // Non-200: try to parse as JSON for the error message
   673→    let errorMsg = `PostEx load-sheet API returned HTTP ${response.status}`
   674→    try {
   675→      const json = await response.json()
   676→      errorMsg = json.statusMessage || json.message || errorMsg
   677→    } catch {
   678→      // Not JSON — use the HTTP status error
   679→    }
   680→
   681→    return {
   682→      success: false,
   683→      error: errorMsg,
   684→    }
   685→  }

   687→  // ──────────────────────────────────────────────────────────────
   688→  // 11. fetchPaymentStatus — GET v1/payment-status/{trackingNumber}
   689→  // ──────────────────────────────────────────────────────────────
   690→
   691→  /**
   692→   * Fetch the payment/settlement status for a tracking number.
   693→   *
   694→   * PostEx's Payment Status API response:
   695→   *   { statusCode, statusMessage, dist: { orderRefNumber, trackingNumber,
   696→   *     settle (boolean), settlementDate, upfrontPaymentDate,
   697→   *     cprNumber_1, reservePaymentDate, cprNumber_2 } }
   698→   *
   699→   * IMPORTANT: PostEx's Payment Status API does NOT break out delivery charge
   700→   * as a separate field. It only provides settlement status (boolean), dates,
   701→   * and CPR numbers. The `actualDeliveryCharge` field on Order/exchange_shipments
   702→   * CANNOT be auto-populated from this API — it would need to come from a
   703→   * different source (e.g. manual entry or a reconciliation report).
   704→   */
   705→  async fetchPaymentStatus(trackingNumber: string): Promise<{
   706→    success: boolean
   707→    settled: boolean
   708→    settlementDate: string | null
   709→    upfrontPaymentDate: string | null
   710→    cprNumber1: string | null
   711→    cprNumber2: string | null
   712→    error?: string
   713→  }> {
   714→    const response = await fetch(
   715→      `${POSTEX_BASE_URL}/v1/payment-status/${encodeURIComponent(trackingNumber)}`,
   716→      {
   717→        method: 'GET',
   718→        headers: {
   719→          token: this.token,
   720→        },
   721→      },
   722→    )
   723→
   724→    if (response.status === 404) {
   725→      return { success: false, settled: false, settlementDate: null, upfrontPaymentDate: null, cprNumber1: null, cprNumber2: null, error: 'Order not found on PostEx.' }
   726→    }
   727→
   728→    const json: PostExApiResponse<{
   729→      orderRefNumber: string
   730→      trackingNumber: string
   731→      settle: boolean
   732→      settlementDate: string
   733→      upfrontPaymentDate: string
   734→      cprNumber_1: string
   735→      reservePaymentDate: string
   736→      cprNumber_2: string
   737→    }> = await response.json()
   738→
   739→    if (json.statusCode === 200 || json.statusCode === '200') {
   740→      const dist = json.dist
   741→      return {
   742→        success: true,
   743→        settled: dist?.settle ?? false,
   744→        settlementDate: dist?.settlementDate ?? null,
   745→        upfrontPaymentDate: dist?.upfrontPaymentDate ?? null,
   746→        cprNumber1: dist?.cprNumber_1 ?? null,
   747→        cprNumber2: dist?.cprNumber_2 ?? null,
   748→      }
   749→    }
   750→
   751→    return {
   752→      success: false,
   753→      settled: false,
   754→      settlementDate: null,
   755→      upfrontPaymentDate: null,
   756→      cprNumber1: null,
   757→      cprNumber2: null,
   758→      error: json.statusMessage || `PostEx payment-status API returned statusCode ${json.statusCode}`,
   759→    }
   760→  }
