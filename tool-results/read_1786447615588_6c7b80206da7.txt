    13→ *      bulk tracking API, and triggers OMS state transitions for any
    14→ *      status changes.
    15→ *
    16→ * SCHEDULING NOTE: pollPostExOrderStatuses() needs to run every 30 minutes

    41→// ──────────────────────────────────────────────────────────────
    42→// 1. generatePostExLoadSheet — standalone batch action
    43→// ──────────────────────────────────────────────────────────────
    44→
    45→/**
    46→ * Generate a PostEx load sheet (pickup manifest) for a batch of tracking numbers.
    47→ *
    48→ * Can be called:
    49→ *   (a) Manually from a "Generate Load Sheet" button for multiple Unbooked orders.
    50→ *   (b) Automatically right after a successful bookShipment() call (chained).
    51→ *
    52→ * @param companyIntegrationId - The PostEx company integration to use
    53→ * @param trackingNumbers - List of PostEx tracking numbers to include
    54→ * @param pickupAddress - Optional pickup address text (from the address book)
    55→ */
    56→export async function generatePostExLoadSheet(
    57→  companyIntegrationId: string,
    58→  trackingNumbers: string[],
    59→  pickupAddress?: string,
    60→): Promise<ActionResult> {
    61→  try {
    62→    const ctx = await getWorkspace()
    63→    if (!isElevated(ctx)) {
    64→      return { success: false, error: 'Only elevated roles can generate load sheets.' }
    65→    }
    66→
    67→    const integration = await db.companyIntegration.findFirst({
    68→      where: {
    69→        id: companyIntegrationId,
    70→        companyId: ctx.company.id,
    71→        provider: { providerKey: 'postex' },
    72→      },
    73→      include: { provider: true },
    74→    })
    75→    if (!integration) {
    76→      return { success: false, error: 'PostEx integration not found.' }
    77→    }
    78→
    79→    const credentials = decryptCredentials(integration.credentialsEncrypted!)
    80→    const adapter = getCourierAdapter('postex', credentials)
    81→
    82→    if (!adapter.generateLoadSheet) {
    83→      return { success: false, error: 'PostEx adapter does not support load sheet generation.' }
    84→    }
    85→
    86→    const result = await executeLoggedIntegrationAction<{ success: boolean; rawResponse?: unknown; error?: string }>({
    87→      companyIntegrationId,
    88→      organizationId: integration.organizationId,
    89→      actionType: 'generate_load_sheet',
    90→      direction: 'outbound',
    91→      fn: async () => adapter.generateLoadSheet!(trackingNumbers, pickupAddress),
    92→    })
    93→
    94→    insertAuditLog({
    95→      action: 'postex.load_sheet_generated',
    96→      entityType: 'company_integration',
    97→      entityId: companyIntegrationId,
    98→      companyId: ctx.company.id,
    99→      organizationId: ctx.company.organizationId,
   100→      userId: ctx.user.id,
   101→      employeeId: ctx.employee.id,
   102→      newValues: {
   103→        trackingNumbersCount: trackingNumbers.length,
   104→        pickupAddress: pickupAddress ?? null,
   105→      },
   106→    })
   107→
   108→    return { success: result.success, data: result }
   109→  } catch (err) {
   110→    return {
   111→      success: false,
   112→      error: err instanceof Error ? err.message : 'Failed to generate load sheet',
   113→    }
   114→  }
   115→}

   117→// ──────────────────────────────────────────────────────────────
   118→// 2. pollPostExOrderStatuses — the polling job
   119→// ──────────────────────────────────────────────────────────────
   120→
   121→/**
   122→ * Poll PostEx for status updates on all active orders AND exchange_shipments.
   123→ *
   124→ * Queries all records with a PostEx courier integration where status is NOT
   125→ * 'delivered' or 'rto' (permanent exclusion — these are never re-polled).
   126→ * Batches tracking numbers, calls trackBulkShipments(), and for each result:
   127→ *   - If status changed → triggers the appropriate OMS function
   128→ *   - If status unchanged → only updates lastPolledAt
   129→ *
   130→ * IDEMPOTENT: calling twice with no underlying change produces no duplicate
   131→ * audit/metric entries beyond the tracking API call log itself.
   132→ *
   133→ * SCHEDULING: needs to run every 30 minutes. Infrastructure-level scheduling
   134→ * (cron, Vercel Cron, etc.) still needs to be connected — same pattern as
   135→ * the city-sync job.
   136→ */

   138→// ──────────────────────────────────────────────────────────────
   139→// 2b. trackSingleOrderStatus — single-order immediate tracking (uses trackShipment)
   140→// ──────────────────────────────────────────────────────────────
   141→
   142→/**
   143→ * Track a SINGLE order's courier status immediately via the adapter's
   144→ * trackShipment() method (NOT the bulk API). This is the function wired to the
   145→ * manual "Refresh Courier Status" button on the Order Detail page.
   146→ *
   147→ * Why a separate function (not just calling pollPostExOrderStatuses)?
   148→ *   - The bulk poll iterates ALL active orders across ALL companies — wasteful
   149→ *     for a single user-initiated check.
   150→ *   - trackShipment() is the adapter's single-tracking method (GET /v1/track-order/{tn}),
   151→ *     which is faster and cheaper than the bulk API for one order.
   152→ *   - This gives the user an immediate, per-order refresh without waiting for
   153→ *     the next cron cycle.
   154→ *
   155→ * Reuses the SAME status-transition logic as the bulk poll (performOrderDispatch
   156→ * for in_transit, mark delivered for delivered, RTO handling, etc.) by calling
   157→ * the shared functions directly.
   158→ *
   159→ * @param orderId - The order to track
   160→ * @returns { success, data: { status, subStatus, updated } }
   161→ */
   162→export async function trackSingleOrderStatus(orderId: string): Promise<ActionResult<{
   163→  status: string
   164→  subStatus: string | null
   165→  updated: boolean
   166→}>> {
   167→  try {
   168→    const ctx = await getWorkspace()
   169→    if (!isElevated(ctx)) {
   170→      return { success: false, error: 'Only elevated roles can refresh courier status.' }
   171→    }
   172→
   173→    // Fetch the order (workspace-scoped)
   174→    const order = await db.order.findFirst({
   175→      where: { id: orderId, companyId: ctx.company.id },
   176→      select: {
   177→        id: true,
   178→        trackingNumber: true,
   179→        courierCompanyIntegrationId: true,
   180→        status: true,
   181→        courierSubStatus: true,
   182→        flowopsOrderNumber: true,
   183→      },
   184→    })
   185→    if (!order) return { success: false, error: 'Order not found' }
   186→    if (!order.trackingNumber) return { success: false, error: 'Order has no tracking number' }
   187→    if (!order.courierCompanyIntegrationId) {
   188→      return { success: false, error: 'Order has no courier integration' }
   189→    }
   190→
   191→    // Fetch the integration + decrypt credentials
   192→    const integration = await db.companyIntegration.findUnique({
   193→      where: { id: order.courierCompanyIntegrationId },
   194→      include: { provider: true },
   195→    })
   196→    if (!integration || !integration.credentialsEncrypted) {
   197→      return { success: false, error: 'Courier integration not found or no credentials' }
   198→    }
   199→
   200→    const credentials = decryptCredentials(integration.credentialsEncrypted)
   201→    const providerKey = integration.provider.providerKey
   202→    const adapter = getCourierAdapter(providerKey, credentials)
   203→
   204→    // Call trackShipment() via the logged wrapper (single-order, NOT bulk)
   205→    const trackResult = await executeLoggedIntegrationAction<TrackShipmentResult>({
   206→      companyIntegrationId: integration.id,
   207→      organizationId: integration.organizationId,
   208→      actionType: 'track_shipment',
   209→      direction: 'outbound',
   210→      relatedEntityType: 'order',
   211→      relatedEntityId: order.id,
   212→      fn: async () => adapter.trackShipment(order.trackingNumber!),
   213→    })
   214→
   215→    // Extract mapped subStatus from raw response
   216→    const raw = trackResult.rawResponse as Record<string, unknown> | undefined
   217→    const mappedSubStatus = (raw?.mappedSubStatus as string) ?? null
   218→    const needsShipperAdvice = (raw?.needsShipperAdvice as boolean) ?? false
   219→    const unrecognized = (raw?.unrecognized as boolean) ?? false
   220→
   221→    const subStatusChanged = mappedSubStatus !== order.courierSubStatus
   222→    const now = new Date()
   223→
   224→    // Update lastPolledAt + courierSubStatus + flags (same as bulk poll)
   225→    await db.order.update({
   226→      where: { id: order.id },
   227→      data: {
   228→        lastPolledAt: now,
   229→        courierSubStatus: mappedSubStatus,
   230→        needsShipperAdvice,
   231→        unrecognizedCourierStatus: unrecognized,
   232→      },
   233→    })
   234→
   235→    // Apply the SAME status transitions as the bulk poll
   236→    if (subStatusChanged) {
   237→      // in_transit → auto-dispatch (with inventory deduction)
   238→      if (trackResult.status === 'in_transit') {
   239→        if (order.status === 'confirmed' || order.status === 'processing') {
   240→          const dispatchResult = await performOrderDispatch(order.id, { source: 'auto_poll' })
   241→          if (!dispatchResult.success && !dispatchResult.skipped) {
   242→            console.error(`[trackSingle] Failed to auto-dispatch order ${order.id}: ${dispatchResult.error}`)
   243→          }
   244→        }
   245→      }
   246→
   247→      // delivered → mark as delivered
   248→      if (trackResult.status === 'delivered') {
   249→        const freshOrder = await db.order.findUnique({ where: { id: order.id }, select: { status: true } })
   250→        if (freshOrder && (freshOrder.status === 'confirmed' || freshOrder.status === 'processing')) {
   251→          await performOrderDispatch(order.id, { source: 'auto_poll' }).catch(() => {})
   252→        }
   253→        if (freshOrder && freshOrder.status !== 'delivered' && freshOrder.status !== 'cancelled' && freshOrder.status !== 'refunded') {
   254→          await db.order.update({
   255→            where: { id: order.id },
   256→            data: { status: 'delivered', deliveredAt: new Date() },
   257→          })
   258→        }
   259→      }
   260→
   261→      // returned → RTO (release reservation if still confirmed/processing)
   262→      if (trackResult.status === 'returned') {
   263→        const freshOrder = await db.order.findUnique({
   264→          where: { id: order.id },
   265→          select: { status: true, dispatchLocationId: true, organizationId: true },
   266→        })
   267→        if (freshOrder && freshOrder.status !== 'rto' && freshOrder.status !== 'cancelled' && freshOrder.status !== 'refunded') {
   268→          if (freshOrder.status === 'confirmed' || freshOrder.status === 'processing') {
   269→            const reservedItems = await db.orderItem.findMany({
   270→              where: { orderId: order.id, fulfillmentStatus: 'reserved' },
   271→            })
   272→            for (const item of reservedItems) {
   273→              const locationId = item.reservedLocationId ?? freshOrder.dispatchLocationId
   274→              if (!locationId) continue
   275→              try {
   276→                await unreserveStockForOrder({
   277→                  orgVariantId: item.orgVariantId,
   278→                  locationId,
   279→                  organizationId: freshOrder.organizationId,
   280→                  companyId: integration.companyId,
   281→                  quantity: item.quantity,
   282→                  orderId: order.id,
   283→                })
   284→              } catch (e) {
   285→                console.error(`[trackSingle] Failed to unreserve stock for item ${item.id}:`, e)
   286→              }
   287→            }
   288→          }
   289→          await db.order.update({
   290→            where: { id: order.id },
   291→            data: { status: 'rto', returnedAt: new Date() },
   292→          })
   293→        }
   294→      }
   295→    }
   296→
   297→    return {
   298→      success: true,
   299→      data: {
   300→        status: trackResult.status ?? 'unknown',
   301→        subStatus: mappedSubStatus,
   302→        updated: subStatusChanged,
   303→      },
   304→    }
   305→  } catch (err) {
   306→    return {
   307→      success: false,
   308→      error: err instanceof Error ? err.message : 'Failed to track single order',
   309→    }
   310→  }
   311→}

   313→export async function pollPostExOrderStatuses(): Promise<ActionResult<{
   314→  polledOrders: number
   315→  polledShipments: number
   316→  statusChanges: number
   317→  errors: string[]
   318→}>> {
   319→  const errors: string[] = []
   320→  let polledOrders = 0
   321→  let polledShipments = 0
   322→  let statusChanges = 0
   323→
   324→  try {
   325→    // Find all active PostEx company integrations
   326→    const postexIntegrations = await db.companyIntegration.findMany({
   327→      where: {
   328→        isActive: true,
   329→        provider: { providerKey: 'postex' },
   330→        credentialsEncrypted: { not: null },
   331→      },
   332→      include: { provider: true },
   333→    })
   334→
   335→    if (postexIntegrations.length === 0) {
   336→      return { success: true, data: { polledOrders: 0, polledShipments: 0, statusChanges: 0, errors: [] } }
   337→    }
   338→
   339→    for (const integration of postexIntegrations) {
   340→      try {
   341→        const credentials = decryptCredentials(integration.credentialsEncrypted!)
   342→        const adapter = getCourierAdapter('postex', credentials)
   343→
   344→        if (!adapter.trackBulkShipments) {
   345→          errors.push(`PostEx adapter for integration ${integration.id} does not support bulk tracking.`)
   346→          continue
   347→        }
   348→
   349→        // ── Fetch active Orders (not delivered/rto) with PostEx tracking ──
   350→        const activeOrders = await db.order.findMany({
   351→          where: {
   352→            courierCompanyIntegrationId: integration.id,
   353→            status: { notIn: ['delivered', 'rto', 'cancelled', 'refunded'] },
   354→            trackingNumber: { not: null },
   355→          },
   356→          select: {
   357→            id: true,
   358→            trackingNumber: true,
   359→            courierSubStatus: true, // we'll use this to detect changes
   360→            status: true,
   361→          },
   362→        })
   363→
   364→        // ── Fetch active Exchange Shipments (not delivered/rto) with PostEx tracking ──
   365→        const activeShipments = await db.exchangeShipment.findMany({
   366→          where: {
   367→            courierCompanyIntegrationId: integration.id,
   368→            status: { notIn: ['delivered', 'rto', 'cancelled'] },
   369→            trackingNumber: { not: null },
   370→          },
   371→          select: {
   372→            id: true,
   373→            trackingNumber: true,
   374→            courierSubStatus: true,
   375→            status: true,
   376→          },
   377→        })
   378→
   379→        // Combine all tracking numbers for bulk API call
   380→        const allTrackingNumbers: Array<{ trackingNumber: string; type: 'order' | 'shipment'; id: string; currentSubStatus: string | null; currentStatus: string }> = [
   381→          ...activeOrders.map((o) => ({
   382→            trackingNumber: o.trackingNumber!,
   383→            type: 'order' as const,
   384→            id: o.id,
   385→            currentSubStatus: o.courierSubStatus ?? null,
   386→            currentStatus: o.status,
   387→          })),
   388→          ...activeShipments.map((s) => ({
   389→            trackingNumber: s.trackingNumber!,
   390→            type: 'shipment' as const,
   391→            id: s.id,
   392→            currentSubStatus: s.courierSubStatus ?? null,
   393→            currentStatus: s.status,
   394→          })),
   395→        ]
   396→
   397→        if (allTrackingNumbers.length === 0) continue
   398→
   399→        // Call the bulk tracking API via the logged wrapper
   400→        const trackingResults = await executeLoggedIntegrationAction<TrackShipmentResult[]>({
   401→          companyIntegrationId: integration.id,
   402→          organizationId: integration.organizationId,
   403→          actionType: 'track_shipment_bulk',
   404→          direction: 'outbound',
   405→          fn: async () => adapter.trackBulkShipments!(allTrackingNumbers.map((t) => t.trackingNumber)),
   406→        })
   407→
   408→        // Map results back to records by tracking number
   409→        const resultsByTrackingNumber = new Map<string, TrackShipmentResult>()
   410→        for (let i = 0; i < allTrackingNumbers.length; i++) {
   411→          if (trackingResults[i]) {
   412→            resultsByTrackingNumber.set(allTrackingNumbers[i].trackingNumber, trackingResults[i])
   413→          }
   414→        }
   415→
   416→        const now = new Date()
   417→
   418→        for (const entry of allTrackingNumbers) {
   419→          const result = resultsByTrackingNumber.get(entry.trackingNumber)
   420→          if (!result || !result.success) {
   421→            // Tracking failed for this number — update lastPolledAt only
   422→            if (entry.type === 'order') {
   423→              await db.order.update({ where: { id: entry.id }, data: { lastPolledAt: now } }).catch((e) => {
   424→                console.error(`[poll] Failed to update lastPolledAt for order ${entry.id}:`, e)
   425→              })
   426→              polledOrders++
   427→            } else {
   428→              await db.exchangeShipment.update({ where: { id: entry.id }, data: { lastPolledAt: now } }).catch((e) => {
   429→                console.error(`[poll] Failed to update lastPolledAt for shipment ${entry.id}:`, e)
   430→              })
   431→              polledShipments++
   432→            }
   433→            continue
   434→          }
   435→
   436→          // Extract the mapped subStatus from the raw response
   437→          const raw = result.rawResponse as Record<string, unknown> | undefined
   438→          const mappedSubStatus = (raw?.mappedSubStatus as string) ?? null
   439→          const needsShipperAdvice = (raw?.needsShipperAdvice as boolean) ?? false
   440→          const unrecognized = (raw?.unrecognized as boolean) ?? false
   441→
   442→          // Check if status changed (compare subStatus)
   443→          const subStatusChanged = mappedSubStatus !== entry.currentSubStatus
   444→
   445→          if (entry.type === 'order') {
   446→            // Update lastPolledAt + courierSubStatus + flags
   447→            // NOTE: we also persist unrecognizedCourierStatus on Order (was
   448→            // previously only persisted on ExchangeShipment — bug fix).
   449→            await db.order.update({
   450→              where: { id: entry.id },
   451→              data: {
   452→                lastPolledAt: now,
   453→                courierSubStatus: mappedSubStatus,
   454→                needsShipperAdvice: needsShipperAdvice,
   455→                unrecognizedCourierStatus: unrecognized,
   456→              },
   457→            }).catch((e) => {
   458→              console.error(`[poll] Failed to update courier status for order ${entry.id}:`, e)
   459→            })
   460→            polledOrders++
   461→
   462→            if (subStatusChanged) {
   463→              statusChanges++
   464→
   465→              // ── "Picked By PostEx" → auto-dispatch ──
   466→              // When PostEx picks up the package, the item has physically left
   467→              // the warehouse. Call performOrderDispatch() which runs the FULL
   468→              // inventory deduction (processInventoryTransaction('sale_dispatched')
   469→              // per item — decrements onHand, releases reserved, locks WAC) +
   470→              // sets status='dispatched' + audit/metric. This is the fix for the
   471→              // critical bug where polling previously set status='dispatched' via
   472→              // direct db.order.update() WITHOUT inventory deduction.
   473→              if (result.status === 'in_transit') {
   474→                try {
   475→                  const order = await db.order.findUnique({
   476→                    where: { id: entry.id },
   477→                    select: { status: true, flowopsOrderNumber: true },
   478→                  })
   479→                  if (order && (order.status === 'confirmed' || order.status === 'processing')) {
   480→                    const dispatchResult = await performOrderDispatch(entry.id, { source: 'auto_poll' })
   481→                    if (dispatchResult.success) {
   482→                      console.log(`[poll] Auto-dispatched ${order.flowopsOrderNumber} (PostEx picked up; inventory deducted${dispatchResult.skipped ? ' — skipped, already dispatched' : ''})`)
   483→                    } else if (!dispatchResult.skipped) {
   484→                      console.error(`[poll] Failed to auto-dispatch order ${entry.id}: ${dispatchResult.error}`)
   485→                      errors.push(`Failed to auto-dispatch order ${entry.id}: ${dispatchResult.error}`)
   486→                    }
   487→                  }
   488→                } catch (e) {
   489→                  console.error(`[poll] Failed to auto-dispatch order ${entry.id}:`, e)
   490→                  errors.push(`Failed to auto-dispatch order ${entry.id}: ${e}`)
   491→                }
   492→              }
   493→
   494→              // ── "Delivered" → mark as delivered ──
   495→              if (result.status === 'delivered') {
   496→                try {
   497→                  // First ensure the order is dispatched WITH inventory deduction
   498→                  // (auto-dispatch if still confirmed/processing). performOrderDispatch
   499→                  // creates the sale_dispatched txn so onHand is correctly decremented.
   500→                  const order = await db.order.findUnique({
   501→                    where: { id: entry.id },
   502→                    select: { status: true, flowopsOrderNumber: true },
   503→                  })
   504→                  if (order && (order.status === 'confirmed' || order.status === 'processing')) {
   505→                    const dispatchResult = await performOrderDispatch(entry.id, { source: 'auto_poll' })
   506→                    if (!dispatchResult.success && !dispatchResult.skipped) {
   507→                      console.error(`[poll] Failed to auto-dispatch order ${entry.id} before marking delivered: ${dispatchResult.error}`)
   508→                      errors.push(`Failed to auto-dispatch order ${entry.id} before delivered: ${dispatchResult.error}`)
   509→                    }
   510→                  }
   511→                  if (order && order.status !== 'delivered' && order.status !== 'cancelled' && order.status !== 'refunded') {
   512→                    // Mark as delivered directly (bypass markOrderDelivered's
   513→                    // getWorkspace() which breaks multi-tenant polling).
   514→                    // No inventory change at delivery time — the dispatch txn
   515→                    // already decremented onHand.
   516→                    await db.order.update({
   517→                      where: { id: entry.id },
   518→                      data: {
   519→                        status: 'delivered',
   520→                        deliveredAt: new Date(),
   521→                      },
   522→                    })
   523→                    console.log(`[poll] Marked order ${order.flowopsOrderNumber} as delivered (PostEx confirmed delivery)`)
   524→                  }
   525→                } catch (e) {
   526→                  console.error(`[poll] Failed to mark order ${entry.id} as delivered:`, e)
   527→                  errors.push(`Failed to mark order ${entry.id} as delivered: ${e}`)
   528→                }
   529→              }
   530→
   531→              // ── "Returned" → mark as RTO ──
   532→              // RTO means the item came back. The correct inventory treatment is
   533→              // to RELEASE the reservation (order_unreserved) — NOT create a
   534→              // sale_dispatched txn, because the item never stayed sold. This
   535→              // matches the retroactive cleanup fix for RTO orders.
   536→              //
   537→              // NOTE: if the order was already 'dispatched' (has a sale_dispatched
   538→              // txn from a previous in_transit polling cycle), the onHand was
   539→              // already decremented. The correct return would be a return_resellable
   540→              // txn to add stock back, but that requires per-item cost lookup and
   541→              // the full processOrderReturn() logic (which uses getWorkspace()).
   542→              // This is a pre-existing gap in the polling RTO path — NOT introduced
   543→              // by this fix. For orders that go directly confirmed→rto (never
   544→              // dispatched), this path correctly releases the reservation without
   545→              // touching onHand.
   546→              if (result.status === 'returned') {
   547→                try {
   548→                  const order = await db.order.findUnique({
   549→                    where: { id: entry.id },
   550→                    select: { status: true, flowopsOrderNumber: true, dispatchLocationId: true, organizationId: true },
   551→                  })
   552→                  if (order && order.status !== 'rto' && order.status !== 'cancelled' && order.status !== 'refunded') {
   553→                    // If still confirmed/processing, release the reservation for
   554→                    // all reserved items (the item came back — don't dispatch it).
   555→                    if (order.status === 'confirmed' || order.status === 'processing') {
   556→                      const reservedItems = await db.orderItem.findMany({
   557→                        where: { orderId: entry.id, fulfillmentStatus: 'reserved' },
   558→                      })
   559→                      for (const item of reservedItems) {
   560→                        const locationId = item.reservedLocationId ?? order.dispatchLocationId
   561→                        if (!locationId) continue
   562→                        try {
   563→                          await unreserveStockForOrder({
   564→                            orgVariantId: item.orgVariantId,
   565→                            locationId,
   566→                            organizationId: order.organizationId,
   567→                            companyId: integration.companyId,
   568→                            quantity: item.quantity,
   569→                            orderId: entry.id,
   570→                          })
   571→                        } catch (e) {
   572→                          console.error(`[poll] Failed to unreserve stock for RTO item ${item.id}:`, e)
   573→                        }
   574→                      }
   575→                    }
   576→                    // Mark as RTO directly (bypass processOrderReturn's
   577→                    // getWorkspace() which breaks multi-tenant polling)
   578→                    await db.order.update({
   579→                      where: { id: entry.id },
   580→                      data: {
   581→                        status: 'rto',
   582→                        returnedAt: new Date(),
   583→                      },
   584→                    })
   585→                    console.log(`[poll] Marked order ${order.flowopsOrderNumber} as RTO (PostEx returned; reservation released)`)
   586→                  }
   587→                } catch (e) {
   588→                  console.error(`[poll] Failed to mark order ${entry.id} as RTO:`, e)
   589→                  errors.push(`Failed to mark order ${entry.id} as RTO: ${e}`)
   590→                }
   591→              }
   592→              // ── "Cancelled by merchant" / "Expired" → cancel the order ──
   593→              // PostEx returns "Un-Assigned By Me" when the merchant cancels
   594→              // on the PostEx portal, or "Expired" when the booking expires.
   595→              // Both map to genericStatus='failed' + courierSubStatus='cancelled_by_merchant' or 'expired'.
   596→              // We need to cancel the order in FlowOps and unreserve stock.
   597→              if (result.status === 'failed' && mappedSubStatus &&
   598→                  (mappedSubStatus === 'cancelled_by_merchant' || mappedSubStatus === 'expired')) {
   599→                try {
   600→                  const order = await db.order.findUnique({
   601→                    where: { id: entry.id },
   602→                    select: { status: true, flowopsOrderNumber: true, dispatchLocationId: true, organizationId: true },
   603→                  })
   604→                  if (order && order.status !== 'cancelled' && order.status !== 'delivered' && order.status !== 'rto') {
   605→                    // Unreserve stock for reserved items
   606→                    const reservedItems = await db.orderItem.findMany({
   607→                      where: { orderId: entry.id, fulfillmentStatus: 'reserved' },
   608→                    })
   609→                    for (const item of reservedItems) {
   610→                      const locationId = item.reservedLocationId ?? order.dispatchLocationId
   611→                      if (!locationId) continue
   612→                      try {
   613→                        await unreserveStockForOrder({
   614→                          orgVariantId: item.orgVariantId,
   615→                          locationId,
   616→                          organizationId: order.organizationId,
   617→                          companyId: integration.companyId,
   618→                          employeeId: integration.createdBy ?? '',
   619→                          quantity: item.quantity,
   620→                          orderId: entry.id,
   621→                        })
   622→                      } catch (e) {
   623→                        console.error(`[poll] Failed to unreserve stock for item ${item.id}:`, e)
   624→                      }
   625→                    }
   626→
   627→                    // Cancel the order
   628→                    await db.order.update({
   629→                      where: { id: entry.id },
   630→                      data: {
   631→                        status: 'cancelled',
   632→                        cancelledAt: new Date(),
   633→                        cancellationReason: `Courier booking ${mappedSubStatus === 'cancelled_by_merchant' ? 'cancelled by merchant on PostEx portal' : 'expired on PostEx'} (detected via status polling)`,
   634→                        courierBookingStatus: 'cancelled',
   635→                      },
   636→                    })
   637→                    console.log(`[poll] Auto-cancelled ${order.flowopsOrderNumber} (PostEx status: ${mappedSubStatus})`)
   638→                  }
   639→                } catch (e) {
   640→                  console.error(`[poll] Failed to auto-cancel order ${entry.id}:`, e)
   641→                  errors.push(`Failed to auto-cancel order ${entry.id}: ${e}`)
   642→                }
   643→              }
   644→            }
   645→
   646→            // ── Payment Status lookup (Phase 3 — migration 012) ──
   647→            if (result.status === 'delivered' || result.status === 'returned') {
   648→              try {
   649→                const paymentStatus = await executeLoggedIntegrationAction<{
   650→                  success: boolean
   651→                  settled: boolean
   652→                  settlementDate: string | null
   653→                  error?: string
   654→                }>({
   655→                  companyIntegrationId: integration.id,
   656→                  organizationId: integration.organizationId,
   657→                  actionType: 'fetch_payment_status',
   658→                  direction: 'outbound',
   659→                  relatedEntityType: 'order',
   660→                  relatedEntityId: entry.id,
   661→                  fn: async () => {
   662→                    const postexAdapter = adapter as unknown as {
   663→                      fetchPaymentStatus: (tn: string) => Promise<{
   664→                        success: boolean
   665→                        settled: boolean
   666→                        settlementDate: string | null
   667→                        error?: string
   668→                      }>
   669→                    }
   670→                    if (typeof postexAdapter.fetchPaymentStatus === 'function') {
   671→                      return postexAdapter.fetchPaymentStatus(entry.trackingNumber)
   672→                    }
   673→                    return { success: false, settled: false, settlementDate: null }
   674→                  },
   675→                })
   676→
   677→                if (paymentStatus?.settled) {
   678→                  insertAuditLog({
   679→                    action: 'postex.payment_settled',
   680→                    entityType: 'order',
   681→                    entityId: entry.id,
   682→                    companyId: integration.companyId,
   683→                    organizationId: integration.organizationId,
   684→                    newValues: {
   685→                      trackingNumber: entry.trackingNumber,
   686→                      settlementDate: paymentStatus.settlementDate,
   687→                    },
   688→                  })
   689→                }
   690→              } catch (e) {
   691→                console.error(`[poll] Payment status lookup failed for ${entry.id}:`, e)
   692→              }
   693→            }
   694→          } else {
   695→            // Exchange shipment
   696→            await db.exchangeShipment.update({
   697→              where: { id: entry.id },
   698→              data: {
   699→                lastPolledAt: now,
   700→                courierSubStatus: mappedSubStatus,
   701→                needsShipperAdvice: needsShipperAdvice,
   702→                unrecognizedCourierStatus: unrecognized,
   703→              },
   704→            }).catch((e) => {
   705→              console.error(`[poll] Failed to update courier status for shipment ${entry.id}:`, e)
   706→            })
   707→            polledShipments++
   708→
   709→            if (subStatusChanged) {
   710→              statusChanges++
   711→
   712→              // ── "Picked By PostEx" → auto-dispatch exchange shipment ──
   713→              // Call performExchangeShipmentDispatch() which runs the FULL
   714→              // inventory deduction (dispatchOrder → sale_dispatched txn —
   715→              // decrements onHand, releases reserved) + sets status='dispatched'
   716→              // + updates parent exchange status + audit/metric. This is the fix
   717→              // for the bug where polling previously set status='dispatched' via
   718→              // direct db.exchangeShipment.update() WITHOUT inventory deduction.
   719→              if (result.status === 'in_transit') {
   720→                try {
   721→                  const shipment = await db.exchangeShipment.findUnique({
   722→                    where: { id: entry.id },
   723→                    select: { status: true, exchangeShipmentNumber: true },
   724→                  })
   725→                  if (shipment && shipment.status === 'confirmed') {
   726→                    const dispatchResult = await performExchangeShipmentDispatch(entry.id, { source: 'auto_poll' })
   727→                    if (dispatchResult.success) {
   728→                      console.log(`[poll] Auto-dispatched exchange shipment ${shipment.exchangeShipmentNumber} (inventory deducted${dispatchResult.skipped ? ' — skipped, already dispatched' : ''})`)
   729→                    } else if (!dispatchResult.skipped) {
   730→                      console.error(`[poll] Failed to auto-dispatch shipment ${entry.id}: ${dispatchResult.error}`)
   731→                      errors.push(`Failed to auto-dispatch shipment ${entry.id}: ${dispatchResult.error}`)
   732→                    }
   733→                  }
   734→                } catch (e) {
   735→                  console.error(`[poll] Failed to auto-dispatch shipment ${entry.id}:`, e)
   736→                  errors.push(`Failed to auto-dispatch shipment ${entry.id}: ${e}`)
   737→                }
   738→              }
   739→
   740→              if (result.status === 'delivered') {
   741→                try {
   742→                  const shipment = await db.exchangeShipment.findUnique({
   743→                    where: { id: entry.id },
   744→                    select: { status: true, exchangeShipmentNumber: true },
   745→                  })
   746→                  if (shipment && shipment.status !== 'delivered' && shipment.status !== 'cancelled') {
   747→                    // Auto-dispatch first if needed (with inventory deduction)
   748→                    if (shipment.status === 'confirmed') {
   749→                      const dispatchResult = await performExchangeShipmentDispatch(entry.id, { source: 'auto_poll' })
   750→                      if (!dispatchResult.success && !dispatchResult.skipped) {
   751→                        console.error(`[poll] Failed to auto-dispatch shipment ${entry.id} before delivered: ${dispatchResult.error}`)
   752→                        errors.push(`Failed to auto-dispatch shipment ${entry.id} before delivered: ${dispatchResult.error}`)
   753→                      }
   754→                    }
   755→                    // Mark as delivered (no inventory change at delivery — dispatch txn already decremented onHand)
   756→                    await db.exchangeShipment.update({
   757→                      where: { id: entry.id },
   758→                      data: { status: 'delivered', deliveredAt: new Date() },
   759→                    })
   760→                    console.log(`[poll] Marked exchange shipment ${shipment.exchangeShipmentNumber} as delivered`)
   761→                  }
   762→                } catch (e) {
   763→                  console.error(`[poll] Failed to mark shipment ${entry.id} as delivered:`, e)
   764→                  errors.push(`Failed to mark exchange shipment ${entry.id} as delivered: ${e}`)
   765→                }
   766→              }
   767→
   768→              // ── "Returned" → mark exchange shipment as RTO ──
   769→              // The replacement item itself was returned by the courier.
   770→              // Calls performExchangeShipmentRto() which restores inventory
   771→              // (return_resellable or return_stitched_received) + sets status='rto'
   772→              // + sets parent order_exchanges to 'exchange_item_returned' (terminal).
   773→              // This is INTENTIONALLY terminal — no automatic re-exchange/refund.
   774→              if (result.status === 'returned') {
   775→                try {
   776→                  const shipment = await db.exchangeShipment.findUnique({
   777→                    where: { id: entry.id },
   778→                    select: { status: true, exchangeShipmentNumber: true },
   779→                  })
   780→                  if (shipment && shipment.status !== 'rto' && shipment.status !== 'cancelled' &&
   781→                      (shipment.status === 'dispatched' || shipment.status === 'delivered')) {
   782→                    const rtoResult = await performExchangeShipmentRto(entry.id, {
   783→                      source: 'auto_poll',
   784→                      returnReason: 'Courier returned the replacement item (detected via PostEx status polling)',
   785→                    })
   786→                    if (rtoResult.success) {
   787→                      console.log(`[poll] Marked exchange shipment ${shipment.exchangeShipmentNumber} as RTO (inventory restored${rtoResult.skipped ? ' — skipped, already RTO' : ''})`)
   788→                    } else if (!rtoResult.skipped) {
   789→                      console.error(`[poll] Failed to mark shipment ${entry.id} as RTO: ${rtoResult.error}`)
   790→                      errors.push(`Failed to mark exchange shipment ${entry.id} as RTO: ${rtoResult.error}`)
   791→                    }
   792→                  }
   793→                } catch (e) {
   794→                  console.error(`[poll] Failed to mark shipment ${entry.id} as RTO:`, e)
   795→                  errors.push(`Failed to mark exchange shipment ${entry.id} as RTO: ${e}`)
   796→                }
   797→              }
   798→            }
   799→          }
   800→        }
   801→      } catch (integrationErr) {
   802→        errors.push(
   803→          `Failed to poll integration ${integration.id}: ${integrationErr instanceof Error ? integrationErr.message : String(integrationErr)}`,
   804→        )
   805→      }
   806→    }
   807→
   808→    // Audit log for the polling run (non-fatal) — use the first integration's
   809→    // company/org context so the audit row is queryable (was previously empty strings).
   810→    const firstIntegration = postexIntegrations[0]
   811→    insertAuditLog({
   812→      action: 'postex.status_poll_completed',
   813→      entityType: 'company_integration',
   814→      entityId: firstIntegration?.id ?? '',
   815→      companyId: firstIntegration?.companyId ?? '',
   816→      organizationId: firstIntegration?.organizationId ?? '',
   817→      newValues: { polledOrders, polledShipments, statusChanges, errorCount: errors.length },
   818→    })
   819→
   820→    insertMetricEvent({
   821→      companyId: firstIntegration?.companyId ?? '',
   822→      entityType: 'company_integration',
   823→      entityId: firstIntegration?.id ?? '',
   824→      metricKey: 'postex.status_poll_completed',
   825→      numericValue: statusChanges,
   826→      dimensions: { polled_orders: polledOrders, polled_shipments: polledShipments, errors: errors.length },
   827→    })
   828→
   829→    return {
   830→      success: true,
   831→      data: { polledOrders, polledShipments, statusChanges, errors },
   832→    }
   833→  } catch (err) {
   834→    return {
   835→      success: false,
   836→      error: err instanceof Error ? err.message : 'Failed to poll PostEx statuses',
   837→      data: { polledOrders, polledShipments, statusChanges, errors },
   838→    }
   839→  }
   840→}
